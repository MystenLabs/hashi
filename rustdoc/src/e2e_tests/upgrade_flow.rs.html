32/// Poll until every node's watcher reports `package_id` as the active
33/// package — the PackageUpgraded handler in watcher.rs must update
34/// OnchainState's package_versions map on all nodes. Prints per-node
35/// diagnostics before failing on timeout.
36pub async fn wait_for_package_convergence(
37    networks: &TestNetworks,
38    package_id: Address,
39    max_wait: std::time::Duration,
40) -> Result<()> {
41    tracing::info!("waiting for all nodes to detect the new package version...");
42    let wait_start = std::time::Instant::now();
43    // Only nodes that are actually running: a pending member (the key-rotation
44    // tests hold one back to start mid-test) has no Hashi instance yet — its
45    // accessor panics — and when it does boot, its fresh scrape sees the
46    // post-upgrade chain, so there is nothing to converge.
47    loop {
48        let all_updated = networks
49            .hashi_network
50            .nodes()
51            .iter()
52            .filter(|node| node.is_running())
53            .all(|node| node.hashi().onchain_state().package_id() == Some(package_id));
54        if all_updated {
55            return Ok(());
56        }
57        if wait_start.elapsed() > max_wait {
58            for (i, node) in networks
59                .hashi_network
60                .nodes()
61                .iter()
62                .filter(|node| node.is_running())
63                .enumerate()
64            {
65                let latest = node.hashi().onchain_state().package_id();
66                let versions = node
67                    .hashi()
68                    .onchain_state()
69                    .state()
70                    .package_versions()
71                    .clone();
72                tracing::info!("node {i}: package_id={latest:?}, versions={versions:?}");
73            }
74            anyhow::bail!("timeout: not all nodes detected the new package version");
75        }
76        tokio::time::sleep(std::time::Duration::from_millis(500)).await;
77    }
78}

80/// Prepare an upgrade package by copying the current source and patching it.
81///
82/// 1. Copies `<test_dir>/packages/hashi` to `<test_dir>/packages/hashi-upgrade`
83/// 2. Sets `PACKAGE_VERSION` in `versioning.move` to `target_version`
84///    (whatever the source currently declares — a no-op rewrite when the tree
85///    already carries the target, a real bump when the chain has moved past it)
86/// 3. Sets `published-at` in `Move.toml` to `published_at` — the chain's
87///    LATEST published package id, which the upgrade ticket checks against
88///
89/// Returns the path to the patched package directory.
90pub fn prepare_upgrade_package(
91    test_dir: &Path,
92    published_at: Address,
93    target_version: u64,
94) -> Result<PathBuf> {
95    let src = test_dir.join("packages/hashi");
96    let dst = test_dir.join("packages/hashi-upgrade");
97
98    anyhow::ensure!(
99        src.exists(),
100        "source package not found at {}",
101        src.display()
102    );
103
104    // Copy the package; the builder's auto-upgrade may have left a previous
105    // copy behind, and `cp -r` into an existing directory would nest.
106    let _ = std::fs::remove_dir_all(&dst);
107    let output = std::process::Command::new("cp")
108        .args(["-r", &src.to_string_lossy(), &dst.to_string_lossy()])
109        .output()?;
110    anyhow::ensure!(
111        output.status.success(),
112        "failed to copy package: {}",
113        String::from_utf8_lossy(&output.stderr)
114    );
115
116    // Patch versioning.move: set PACKAGE_VERSION to the target version.
117    let versioning_path = dst.join("sources/core/versioning.move");
118    let versioning_src = std::fs::read_to_string(&versioning_path)?;
119    const CONST_PREFIX: &str = "const PACKAGE_VERSION: u64 = ";
120    let mut replaced = false;
121    let patched: String = versioning_src
122        .lines()
123        .map(|line| {
124            if line.starts_with(CONST_PREFIX) {
125                replaced = true;
126                format!("{CONST_PREFIX}{target_version};\n")
127            } else {
128                format!("{line}\n")
129            }
130        })
131        .collect();
132    anyhow::ensure!(
133        replaced,
134        "PACKAGE_VERSION constant not found in versioning.move"
135    );
136    std::fs::write(&versioning_path, patched)?;
137
138    // Patch Move.toml: add published-at
139    let move_toml_path = dst.join("Move.toml");
140    let move_toml = std::fs::read_to_string(&move_toml_path)?;
141    let patched_toml = move_toml.replace(
142        "[package]",
143        &format!("[package]\npublished-at = \"{}\"", published_at),
144    );
145    std::fs::write(&move_toml_path, patched_toml)?;
146
147    // Add a trivial new-in-this-version module to prove new code is callable
148    // post-upgrade. Re-added on every prepare so successive upgrades stay
149    // module-compatible.
150    let test_module_path = dst.join("sources/upgrade_canary.move");
151    std::fs::write(
152        &test_module_path,
153        format!(
154            "module hashi::upgrade_canary;\n\npublic fun version(): u64 {{ {target_version} }}\n"
155        ),
156    )?;
157
158    // Clean build artifacts from the copy
159    let _ = std::fs::remove_dir_all(dst.join("build"));
160
161    tracing::info!(
162        "upgrade package prepared at {} (published-at = {})",
163        dst.display(),
164        published_at
165    );
166
167    Ok(dst)
168}

170/// [`build_upgrade_package`] behind a cross-process disk cache.
171///
172/// Every builder boot compiles the byte-identical artifact (verified: the
173/// per-test `published-at` patch does not reach the compiled modules or
174/// digest), and nextest runs each test in its own process — so on CI's
175/// 4-core runners the compile was repeated per test, starving the network
176/// it was booting next to. Keyed on the patched source (with the
177/// `published-at` line masked), the target version, and the sui toolchain;
178/// a mkdir lock dedups concurrent builders, and every fallback path builds
179/// rather than fails — the cache can only add speed, never wrongness.
180fn build_upgrade_package_cached(
181    upgrade_path: &Path,
182    client_config: Option<&Path>,
183    target_version: u64,
184) -> Result<(Publish, Vec<u8>)> {
185    use fastcrypto::hash::HashFunction;
186
187    let key = {
188        let mut hasher = fastcrypto::hash::Sha256::default();
189        hasher.update(target_version.to_le_bytes());
190        let sui_version = std::process::Command::new(sui_binary())
191            .arg("--version")
192            .output()
193            .map(|o| o.stdout)
194            .unwrap_or_default();
195        hasher.update(&sui_version);
196        let mut files: Vec<PathBuf> = walk_files(upgrade_path)?;
197        files.sort();
198        for file in &files {
199            let rel = file.strip_prefix(upgrade_path).unwrap_or(file);
200            // `build/` is output, not input; `published-at` is per-test but
201            // verified not to reach the artifact.
202            if rel.starts_with("build") {
203                continue;
204            }
205            hasher.update(rel.to_string_lossy().as_bytes());
206            let contents = std::fs::read(file)?;
207            if rel == Path::new("Move.toml") {
208                let masked: String = String::from_utf8_lossy(&contents)
209                    .lines()
210                    .filter(|line| !line.trim_start().starts_with("published-at"))
211                    .collect::<Vec<_>>()
212                    .join("\n");
213                hasher.update(masked.as_bytes());
214            } else {
215                hasher.update(&contents);
216            }
217        }
218        hex_encode(&hasher.finalize().digest)
219    };
220
221    let cache_root = std::env::temp_dir().join("hashi-e2e-upgrade-build-cache");
222    let final_path = cache_root.join(format!("{key}.json"));
223
224    if let Some(cached) = read_cached_build(&final_path) {
225        tracing::info!("upgrade package cache hit ({key})");
226        return Ok(cached);
227    }
228
229    // mkdir is atomic: exactly one process wins the right to build; the rest
230    // wait for the artifact to appear. Timeouts and stale locks fall through
231    // to building — duplicated work over a wedged run.
232    std::fs::create_dir_all(&cache_root)?;
233    let lock = cache_root.join(format!("{key}.lock"));
234    if std::fs::create_dir(&lock).is_err() {
235        let stale = std::time::Duration::from_secs(900);
236        let started = std::time::Instant::now();
237        while started.elapsed() < std::time::Duration::from_secs(600) {
238            if let Some(cached) = read_cached_build(&final_path) {
239                tracing::info!("upgrade package cache hit after waiting ({key})");
240                return Ok(cached);
241            }
242            if std::fs::metadata(&lock)
243                .and_then(|m| m.modified())
244                .map(|t| t.elapsed().unwrap_or_default() > stale)
245                .unwrap_or(true)
246            {
247                break;
248            }
249            std::thread::sleep(std::time::Duration::from_secs(2));
250        }
251        tracing::warn!("upgrade build lock wait expired; building anyway ({key})");
252    }
253
254    let built = build_upgrade_package(sui_binary(), upgrade_path, client_config, target_version);
255    let _ = std::fs::remove_dir(&lock);
256    let (compiled, digest) = built?;
257
258    // Write-then-rename so readers never see a partial file.
259    let tmp = cache_root.join(format!("{key}.tmp-{}", std::process::id()));
260    if let Ok(bytes) = serde_json::to_vec(&(&compiled, &digest))
261        && std::fs::write(&tmp, bytes).is_ok()
262    {
263        let _ = std::fs::rename(&tmp, &final_path);
264    }
265    Ok((compiled, digest))
266}

268fn read_cached_build(path: &Path) -> Option<(Publish, Vec<u8>)> {
269    let bytes = std::fs::read(path).ok()?;
270    serde_json::from_slice(&bytes).ok()
271}

273fn walk_files(dir: &Path) -> Result<Vec<PathBuf>> {
274    let mut out = Vec::new();
275    let mut stack = vec![dir.to_path_buf()];
276    while let Some(d) = stack.pop() {
277        for entry in std::fs::read_dir(&d)? {
278            let path = entry?.path();
279            if path.is_dir() {
280                stack.push(path);
281            } else {
282                out.push(path);
283            }
284        }
285    }
286    Ok(out)
287}

289fn hex_encode(bytes: &[u8]) -> String {
290    bytes.iter().map(|b| format!("{b:02x}")).collect()
291}

293/// Run the full upgrade lifecycle: prepare → build → propose → vote → execute+publish+finalize.
294///
295/// Returns the new package ID on success.
296pub async fn execute_full_upgrade(networks: &mut TestNetworks) -> Result<Address> {
297    // Running nodes only: a pending member (started mid-test by the
298    // key-rotation tests) has no Hashi instance to read state from or vote
299    // with — and is not a registered committee member yet anyway.
300    let nodes: Vec<_> = networks
301        .hashi_network
302        .nodes()
303        .iter()
304        .filter(|node| node.is_running())
305        .collect();
306    anyhow::ensure!(!nodes.is_empty(), "no running nodes to drive the upgrade");
307    let hashi_ids = networks.hashi_network.ids();
308
309    let mut executors: Vec<SuiTxExecutor> = nodes
310        .iter()
311        .map(|node| {
312            let hashi = node.hashi();
313            SuiTxExecutor::from_config(&hashi.config, hashi.onchain_state())
314        })
315        .collect::<Result<_>>()?;
316
317    // 1. Prepare the upgrade package (copy + patch), targeting one past the
318    // chain's latest published version. `published-at` must name the LATEST
319    // published package (not the original): upgrading v2 -> v3 with a
320    // v1 `published-at` fails the upgrade ticket's package-id check.
321    let (current_version, current_package_id) = {
322        let state = nodes[0].hashi().onchain_state().state();
323        let versions = state.package_versions();
324        match (versions.latest_version(), versions.latest_id()) {
325            (Some(version), Some(id)) => (version, id),
326            _ => anyhow::bail!("onchain state has no package versions yet"),
327        }
328    };
329    let target_version = current_version + 1;
330    let test_dir = networks.dir();
331    let upgrade_path = prepare_upgrade_package(test_dir, current_package_id, target_version)?;
332
333    let client_config_path = test_dir.join("sui/client.yaml");
334    let client_config = client_config_path
335        .exists()
336        .then_some(client_config_path.as_path());
337
338    // 2. Build the upgrade
339    tracing::info!("building upgrade package from {}", upgrade_path.display());
340    let (compiled, digest) =
341        build_upgrade_package_cached(&upgrade_path, client_config, target_version)?;
342    tracing::info!("upgrade package built, digest: {digest:?}");
343
344    // 3. Propose the upgrade
345    tracing::info!("proposing upgrade...");
346    let creator = executors[0].sender();
347    let create_tx = build_create_proposal_transaction(
348        hashi_ids,
349        creator,
350        CreateProposalParams::Upgrade {
351            digest: digest.clone(),
352            metadata: vec![("reason".to_string(), "upgrade test".to_string())],
353        },
354    );
355    let response = executors[0].execute(create_tx).await?;
356    anyhow::ensure!(
357        response.transaction().effects().status().success(),
358        "create Upgrade proposal failed"
359    );
360
361    let proposal_id = extract_proposal_id_from_response(&response)?;
362    tracing::info!("upgrade proposal {proposal_id} created");
363
364    // 4. All other nodes vote (upgrade requires 100% quorum)
365    let upgrade_type_tag = TypeTag::Struct(Box::new(StructTag::new(
366        hashi_ids.package_id,
367        Identifier::from_static("upgrade"),
368        Identifier::from_static("Upgrade"),
369        vec![],
370    )));
371
372    for executor in &mut executors[1..] {
373        let voter = executor.sender();
374        let vote_tx =
375            build_vote_transaction(hashi_ids, voter, proposal_id, upgrade_type_tag.clone());
376        let vote_resp = executor.execute(vote_tx).await?;
377        anyhow::ensure!(
378            vote_resp.transaction().effects().status().success(),
379            "vote on Upgrade proposal failed"
380        );
381    }
382    tracing::info!("all nodes voted on upgrade proposal");
383
384    // 5. Execute upgrade + publish + finalize in one PTB
385    tracing::info!("executing upgrade (execute + publish + finalize in one PTB)...");
386    let upgrade_tx =
387        build_upgrade_execution_transaction(hashi_ids, current_package_id, proposal_id, compiled);
388    let upgrade_resp = executors[0].execute(upgrade_tx).await?;
389    anyhow::ensure!(
390        upgrade_resp.transaction().effects().status().success(),
391        "upgrade execute+publish+finalize failed: {:?}",
392        upgrade_resp.transaction().effects().status()
393    );
394
395    let new_package_id = extract_new_package_id_from_response(&upgrade_resp)?;
396    tracing::info!("upgrade complete! new package: {new_package_id}");
397    Ok(new_package_id)
398}

400/// Propose + vote + execute a DisableVersion governance action.
401///
402/// `execute_package_id` is the package whose `disable_version::execute` is called.
403/// When disabling an old version after upgrade, this must be the NEW package ID
404/// (whose `PACKAGE_VERSION` differs from the version being disabled).
405pub async fn disable_version(
406    executors: &mut [SuiTxExecutor],
407    hashi_ids: HashiIds,
408    version: u64,
409    execute_package_id: Address,
410) -> Result<()> {
411    let creator = executors[0].sender();
412    let create_tx = build_create_proposal_transaction(
413        hashi_ids,
414        creator,
415        CreateProposalParams::DisableVersion {
416            version,
417            metadata: vec![],
418        },
419    );
420    let response = executors[0].execute(create_tx).await?;
421    anyhow::ensure!(
422        response.transaction().effects().status().success(),
423        "create DisableVersion proposal failed"
424    );
425
426    let proposal_id = extract_proposal_id_from_response(&response)?;
427
428    let disable_version_type = TypeTag::Struct(Box::new(StructTag::new(
429        hashi_ids.package_id,
430        Identifier::from_static("disable_version"),
431        Identifier::from_static("DisableVersion"),
432        vec![],
433    )));
434
435    for executor in &mut executors[1..] {
436        let voter = executor.sender();
437        let vote_tx =
438            build_vote_transaction(hashi_ids, voter, proposal_id, disable_version_type.clone());
439        let vote_resp = executor.execute(vote_tx).await?;
440        anyhow::ensure!(
441            vote_resp.transaction().effects().status().success(),
442            "vote on DisableVersion proposal failed"
443        );
444    }
445
446    let execute_tx = build_execute_proposal_transaction(
447        hashi_ids,
448        proposal_id,
449        execute_package_id,
450        "disable_version",
451    )?;
452    let exec_resp = executors[0].execute(execute_tx).await?;
453    anyhow::ensure!(
454        exec_resp.transaction().effects().status().success(),
455        "execute DisableVersion proposal failed"
456    );
457
458    tracing::info!("version {version} disabled");
459    Ok(())
