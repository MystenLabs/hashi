4//! Publish a checked-in Move *bytecode snapshot* into a local Sui network.
5//!
6//! Hashi keeps a checked-in bytecode snapshot of the deployed package
7//! (`crates/hashi/tests/move_upgrade_snapshots/<network>/v<version>/`) so the
8//! upgrade-compatibility CI gate can run hermetically. This module reuses that
9//! same snapshot for a *stronger* test: publish the deployed v1 bytecode into a
10//! fresh local net and then upgrade it to the current source, proving the real
11//! "deployed bytecode → current source" upgrade end-to-end (not just the static
12//! compatibility check).
13//!
14//! ## Mechanism
15//!
16//! 1. Read every `*.mv` in the snapshot directory.
17//! 2. Rebase each module's self-address from the package's *runtime* id to
18//!    `0x0`. This is the **inverse** of the compat test's
19//!    `substitute_self_address`: the on-chain `module_map` (and thus the
20//!    snapshot) stores modules self-addressed at the runtime id, while a fresh
21//!    `publish` on Sui assigns a brand-new object id and expects the incoming
22//!    modules to be self-addressed at `0x0` (what a source build of an
23//!    upgradeable package emits).
24//! 3. Re-serialize at each module's *original* binary version (via
25//!    `serialize_with_version`, NOT `serialize()` — the latter forces
26//!    `VERSION_MAX`, which could exceed what the local net accepts).
27//! 4. Build a `Publish { modules, dependencies: [0x1, 0x2, 0x3] }` — the
28//!    framework-only deps present on any local net, per the on-chain linkage
29//!    table.
30//! 5. Publish via the production `hashi::publish::publish_package` path, which
31//!    also extracts the `Hashi` shared object + `UpgradeCap` — a strong signal
32//!    the snapshot published as a real, init-run package.
33

46/// Default snapshot directory: the deployed `testnet` package's snapshot,
47/// `../hashi/tests/move_upgrade_snapshots/testnet/v<version>`, with the
48/// version taken from `packages/hashi/Published.toml` — the same source of
49/// truth the compat CI gate derives its snapshot locations from. A deployment
50/// bumps `Published.toml`, so this test follows it to the new snapshot
51/// instead of silently exercising stale bytecode.
52pub fn default_snapshot_dir() -> Result<PathBuf> {
53    let crates_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("..");
54    let published = crates_dir
55        .join("..")
56        .join("packages")
57        .join("hashi")
58        .join("Published.toml");
59    let entries = hashi::published::published_entries(&published)?;
60    let entry = entries.get("testnet").ok_or_else(|| {
61        anyhow::anyhow!(
62            "no `testnet` entry in {} — the snapshot tests exercise the testnet deployment",
63            published.display()
64        )
65    })?;
66    Ok(crates_dir
67        .join("hashi")
68        .join("tests")
69        .join("move_upgrade_snapshots")
70        .join("testnet")
71        .join(format!("v{}", entry.version)))
72}

74/// The self-address a module declares in its bytecode.
75fn module_self_address(module: &CompiledModule) -> Result<AccountAddress> {
76    let idx = module.self_handle().address;
77    module
78        .address_identifiers
79        .get(idx.0 as usize)
80        .copied()
81        .ok_or_else(|| anyhow::anyhow!("module has an invalid self-address index"))
82}
83
84/// Rebase a module's self-address from `runtime` to `0x0`.
85///
86/// INVERSE of the compat test's `substitute_self_address` (which rebases
87/// `0x0` → runtime for comparison). Here we go the other way because we are
88/// *publishing* the snapshot as a brand-new package: the adapter assigns a
89/// fresh object id and requires the incoming modules to be `0x0`-addressed.
90fn rebase_runtime_to_zero(module: &mut CompiledModule, runtime: AccountAddress) -> Result<()> {
91    let self_addr_idx = module.self_handle().address;
92    let name = module.identifier_at(module.self_handle().name).to_string();
93
94    let addr = module
95        .address_identifiers
96        .get_mut(self_addr_idx.0 as usize)
97        .ok_or_else(|| anyhow::anyhow!("module `{name}` has an invalid self-address index"))?;
98
99    anyhow::ensure!(
100        *addr == runtime,
101        "module `{name}` self-address is {addr}, expected snapshot runtime id {runtime}"
102    );
103
104    *addr = AccountAddress::ZERO;
105    Ok(())
106}

108/// Read the snapshot's runtime package id.
109///
110/// Every snapshot module is self-addressed at the deployed package's
111/// runtime/original id. We derive it from the first module's bytecode
112/// (matching `MovePackage::original_package_id`) rather than trusting the
113/// manifest, and assert every module agrees.
114fn read_runtime_id(modules: &[CompiledModule]) -> Result<AccountAddress> {
115    let runtime = module_self_address(
116        modules
117            .first()
118            .ok_or_else(|| anyhow::anyhow!("snapshot has no modules"))?,
119    )?;
120    anyhow::ensure!(
121        runtime != AccountAddress::ZERO,
122        "snapshot modules unexpectedly carry a 0x0 self-address"
123    );
124    for module in modules {
125        let addr = module_self_address(module)?;
126        anyhow::ensure!(
127            addr == runtime,
128            "module `{}` self-address is {addr}, expected snapshot runtime id {runtime}",
129            module.identifier_at(module.self_handle().name),
130        );
131    }
132    Ok(runtime)
133}

135/// Load every `*.mv` in `snapshot_dir`, rebase each module's self-address from
136/// the runtime id to `0x0`, re-serialize at its original binary version, and
137/// return a `Publish` payload with framework-only dependencies (`[0x1, 0x2,
138/// 0x3]`) ready to hand to `hashi::publish::publish_package`.
139pub fn load_snapshot_publish(snapshot_dir: &Path) -> Result<sui_sdk_types::Publish> {
140    anyhow::ensure!(
141        snapshot_dir.is_dir(),
142        "snapshot dir missing: {}",
143        snapshot_dir.display()
144    );
145
146    let mut mv_paths: Vec<PathBuf> = std::fs::read_dir(snapshot_dir)
147        .with_context(|| format!("reading snapshot directory {}", snapshot_dir.display()))?
148        .map(|e| e.map(|e| e.path()).map_err(anyhow::Error::from))
149        .collect::<Result<Vec<_>>>()?
150        .into_iter()
151        .filter(|p| p.extension().and_then(|e| e.to_str()) == Some("mv"))
152        .collect();
153    mv_paths.sort();
154    anyhow::ensure!(
155        !mv_paths.is_empty(),
156        "no .mv files in {}",
157        snapshot_dir.display()
158    );
159
160    // Deserialize first so we can derive the runtime id from the bytecode.
161    let mut modules: Vec<CompiledModule> = mv_paths
162        .iter()
163        .map(|path| {
164            let raw = std::fs::read(path).with_context(|| format!("reading {}", path.display()))?;
165            CompiledModule::deserialize_with_defaults(&raw)
166                .map_err(|e| anyhow::anyhow!("deserialize {}: {e:?}", path.display()))
167        })
168        .collect::<Result<_>>()?;
169
170    let runtime = read_runtime_id(&modules)?;
171
172    let mut serialized = Vec::with_capacity(modules.len());
173    for module in &mut modules {
174        rebase_runtime_to_zero(module, runtime)?;
175
176        // Preserve the binary version the snapshot was serialized at (do NOT
177        // bump to VERSION_MAX — that could exceed what the local net accepts).
178        let version = module.version;
179        let mut bytes = Vec::new();
180        module
181            .serialize_with_version(version, &mut bytes)
182            .map_err(|e| anyhow::anyhow!("re-serialize module: {e:?}"))?;
183        serialized.push(bytes);
184    }
185
186    // Dependencies = the sui framework, per the on-chain linkage table. These
187    // three are present on any local net.
188    let dependencies: Vec<Address> = vec![
189        Address::from_static("0x1"),
190        Address::from_static("0x2"),
191        Address::from_static("0x3"),
192    ];
193
194    Ok(sui_sdk_types::Publish {
195        modules: serialized,
196        dependencies,
197    })
198}

200/// Publish the bytecode snapshot at `snapshot_dir` as a fresh package, driven
201/// by `private_key`. Mirrors the `publish` module but sources the modules from a
202/// checked-in bytecode snapshot instead of a source build.
203///
204/// Like the source-build path, the `UpgradeCap` is transferred to the sender
205/// and left there until `hashi::finish_publish` (the launch switch) hands it
206/// into on-chain custody — so the returned [`PublishOutput`] slots directly
207/// into the normal genesis/launch sequencing.
208pub async fn publish_snapshot(
209    snapshot_dir: &Path,
210    client: &mut Client,
211    private_key: &Ed25519PrivateKey,
212) -> Result<PublishOutput> {
213    let publish = load_snapshot_publish(snapshot_dir)?;
214    tracing::info!(
215        modules = publish.modules.len(),
216        dir = %snapshot_dir.display(),
217        "publishing bytecode snapshot (runtime -> 0x0 rebased)"
218    );
219    hashi::publish::publish_package(client, &private_key.clone().into(), publish).await
220}
