39impl BitcoinNodeHandle {
40    pub fn new(rpc_port: u16, data_dir: PathBuf, bitcoin_core_path: PathBuf) -> Result<Self> {
41        Self::new_with_args(rpc_port, data_dir, bitcoin_core_path, &[])
42    }
43
44    fn new_with_args(
45        rpc_port: u16,
46        data_dir: PathBuf,
47        bitcoin_core_path: PathBuf,
48        extra_args: &[String],
49    ) -> Result<Self> {
50        let rpc_url = format!("http://127.0.0.1:{}", rpc_port);
51        let p2p_port = get_available_port();
52        info!(
53            "Starting Bitcoin node with RPC at {} and P2P port {}",
54            rpc_url, p2p_port
55        );
56
57        let stdout_name = data_dir.join("bitcoin.stdout");
58        let stdout = std::fs::File::create(&stdout_name)?;
59        let stderr_name = data_dir.join("bitcoin.stderr");
60        let stderr = std::fs::File::create(&stderr_name)?;
61
62        let mut process = Command::new(&bitcoin_core_path)
63            .arg("-regtest")
64            .arg("-server")
65            .arg(format!("-datadir={}", data_dir.display()))
66            .arg(format!("-rpcport={}", rpc_port))
67            // Use -bind instead of -port to prevent Bitcoin Core from
68            // also opening an onion listener on port+1.
69            .arg(format!("-bind=127.0.0.1:{}", p2p_port))
70            .arg("-listen=1")
71            .arg(format!("-rpcuser={}", RPC_USER))
72            .arg(format!("-rpcpassword={}", RPC_PASSWORD))
73            .arg("-rpcbind=127.0.0.1")
74            .arg("-rpcallowip=127.0.0.1")
75            .arg("-fallbackfee=0.0001")
76            .arg("-acceptnonstdtxn=1")
77            .arg("-blockfilterindex=1") // Enable compact block filters (BIP-158)
78            .arg("-peerblockfilters=1") // Serve filters to peers (BIP-157)
79            .arg("-txindex=1") // Enable transaction index for RPC queries
80            .arg("-listenonion=0")
81            .args(extra_args)
82            .stdout(stdout)
83            .stderr(stderr)
84            .spawn()
85            .map_err(|e| {
86                anyhow!(
87                    "Failed to start bitcoind: {}. Make sure bitcoind is installed and in PATH",
88                    e
89                )
90            })?;
91
92        // Give bitcoind a moment to start up and bind to the RPC port
93        std::thread::sleep(Duration::from_millis(500));
94
95        // Check if process exited already
96        match process.try_wait() {
97            Ok(Some(status)) => {
98                return Err(anyhow!(
99                    "bitcoind exited immediately with status: {:?}",
100                    status
101                ));
102            }
103            Ok(None) => {
104                info!("Bitcoin process spawned with PID: {:?}", process.id());
105            }
106            Err(e) => {
107                warn!("Error checking process status: {}", e);
108            }
109        }
110
111        let rpc_client = Client::new_with_auth(
112            &rpc_url,
113            Auth::UserPass(RPC_USER.to_string(), RPC_PASSWORD.to_string()),
114        )?;
115        Ok(Self {
116            rpc_client,
117            data_dir,
118            process,
119            stdout_path: stdout_name,
120            stderr_path: stderr_name,
121            rpc_url,
122            rpc_port,
123            p2p_port,
124        })
125    }
126
127    pub async fn wait_until_ready(&self) -> Result<()> {
128        let start = std::time::Instant::now();
129        let timeout = Duration::from_secs(BITCOIN_CORE_STARTUP_TIMEOUT_SECS);
130        loop {
131            if start.elapsed() > timeout {
132                return Err(anyhow!(
133                    "Bitcoin Core failed to start within timeout. {}",
134                    self.startup_diagnostics()
135                ));
136            }
137            match self.rpc_client.get_blockchain_info() {
138                Ok(_) => {
139                    info!("Bitcoin node is ready");
140                    match self.rpc_client.create_wallet("test") {
141                        Ok(_) => info!("Created test wallet"),
142                        Err(e) => info!("Wallet creation: {}", e),
143                    }
144                    return Ok(());
145                }
146                Err(e) => {
147                    let elapsed = start.elapsed().as_secs();
148                    if elapsed.is_multiple_of(STARTUP_LOG_INTERVAL_SECS) {
149                        info!("Waiting for Bitcoin node to be ready ({}s): {}", elapsed, e);
150                    }
151                    tokio::time::sleep(Duration::from_millis(100)).await;
152                }
153            }
154        }
155    }
156
157    fn startup_diagnostics(&self) -> String {
158        crate::tail_logs(&[
159            ("stderr", &self.stderr_path),
160            ("stdout", &self.stdout_path),
161            ("debug.log", &self.data_dir.join("regtest/debug.log")),
162        ])
163    }

165    pub fn generate_blocks(&self, count: u64) -> Result<Vec<BlockHash>> {
166        let blocks = self
167            .rpc_client
168            .generate_to_address(count as usize, &self.get_new_address()?)?
169            .into_model()
170            .map_err(|e| anyhow!("invalid block hash: {e}"))?
171            .0;
172        info!("Generated {} blocks", count);
173        Ok(blocks)
174    }
175
176    pub fn send_to_address(&self, address: &BitcoinAddress, amount: Amount) -> Result<Txid> {
177        let txid = self
178            .rpc_client
179            .send_to_address(address, amount)?
180            .into_model()
181            .map_err(|e| anyhow!("invalid txid: {e}"))?
182            .txid;
183        info!("Sent {} to {}: {}", amount, address, txid);
184        Ok(txid)
185    }

201    pub fn get_block_count(&self) -> Result<u64> {

203    }

205    pub async fn wait_for_transaction(&self, txid: &Txid, timeout: Duration) -> Result<()> {
206        let start = std::time::Instant::now();
207        loop {
208            if start.elapsed() > timeout {
209                return Err(anyhow!("Transaction {} not found within timeout", txid));
210            }
211            match self.rpc_client.get_transaction(*txid) {
212                Ok(_) => {
213                    info!("Transaction {} confirmed", txid);
214                    return Ok(());
215                }
216                Err(_) => {
217                    tokio::time::sleep(Duration::from_millis(100)).await;
218                }
219            }
220        }
221    }

226        Ok(())
227    }

250impl Drop for BitcoinNodeHandle {
251    fn drop(&mut self) {
252        if let Err(e) = self.stop() {
253            warn!("Failed to stop Bitcoin node gracefully: {}", e);
254        }
255        if let Err(e) = self.process.kill() {
256            warn!("Failed to kill Bitcoin node process: {}", e);
257        }
258    }
259}

261pub struct BitcoinNodeBuilder {
262    dir: Option<PathBuf>,
263    initial_blocks: u64,
264    bitcoin_core_path: Option<PathBuf>,
265    rpc_port: Option<u16>,
266    extra_args: Vec<String>,
267}

269impl BitcoinNodeBuilder {
270    pub fn new() -> Self {
271        Self {
272            initial_blocks: DEFAULT_INITIAL_BLOCKS,
273            bitcoin_core_path: None,
274            dir: None,
275            rpc_port: None,
276            extra_args: Vec::new(),
277        }
278    }
279
280    pub fn with_initial_blocks(mut self, blocks: u64) -> Self {
281        self.initial_blocks = blocks;
282        self
283    }
284
285    pub fn with_bitcoin_core_path(mut self, path: PathBuf) -> Self {
286        self.bitcoin_core_path = Some(path);
287        self
288    }
289
290    pub fn dir(mut self, dir: &Path) -> Self {
291        self.dir = Some(dir.to_owned());
292        self
293    }
294
295    pub fn with_rpc_port(mut self, port: u16) -> Self {
296        self.rpc_port = Some(port);
297        self
298    }
299
300    pub fn with_extra_args(mut self, args: impl IntoIterator<Item = impl Into<String>>) -> Self {
301        self.extra_args.extend(args.into_iter().map(Into::into));
302        self
303    }
304
305    pub async fn build(self) -> Result<BitcoinNodeHandle> {
306        let bitcoin_core_path = self
307            .bitcoin_core_path
308            .unwrap_or_else(|| PathBuf::from("bitcoind"));
309        let rpc_port = self.rpc_port.unwrap_or_else(get_available_port);
310
311        let data_dir = self.dir.ok_or_else(|| anyhow!("no data_dir configured"))?;
312
313        let node_handle = BitcoinNodeHandle::new_with_args(
314            rpc_port,
315            data_dir,
316            bitcoin_core_path,
317            &self.extra_args,
318        )?;
319        node_handle.wait_until_ready().await?;
320        if self.initial_blocks > 0 {
321            node_handle.generate_blocks(self.initial_blocks)?;
322        }
323        info!(
324            "Created Bitcoin node at RPC port {} with {} initial blocks",
325            rpc_port, self.initial_blocks
326        );
327        Ok(node_handle)
328    }
329}

357    fn wallet_script(node: &BitcoinNodeHandle) -> Result<ScriptBuf> {
358        let address: String = node
359            .rpc_client()
360            .call("getnewaddress", &[json!(""), json!("bech32")])?;
361        Ok(address
362            .parse::<bitcoin::Address<_>>()?
363            .require_network(bitcoin::Network::Regtest)?
364            .script_pubkey())
365    }

367    fn signed_spend(
368        node: &BitcoinNodeHandle,
369        input: OutPoint,
370        output: Vec<TxOut>,
371    ) -> Result<Transaction> {
372        let unsigned = Transaction {
373            version: bitcoin::transaction::Version::TWO,
374            lock_time: bitcoin::absolute::LockTime::ZERO,
375            input: vec![TxIn {
376                previous_output: input,
377                script_sig: ScriptBuf::new(),
378                sequence: bitcoin::Sequence::MAX,
379                witness: bitcoin::Witness::new(),
380            }],
381            output,
382        };
383        let signed: Value = node.rpc_client().call(
384            "signrawtransactionwithwallet",
385            &[json!(serialize_hex(&unsigned))],
386        )?;
387        anyhow::ensure!(
388            signed["complete"] == true,
389            "Incomplete wallet signature: {signed}"
390        );
391        Ok(deserialize_hex(signed["hex"].as_str().ok_or_else(
392            || anyhow!("Missing signed transaction hex: {signed}"),
393        )?)?)
394    }

396    fn confirmed_coins(node: &BitcoinNodeHandle) -> Result<Vec<(OutPoint, Amount)>> {
397        let coins: Vec<Value> = node.rpc_client().call("listunspent", &[json!(1)])?;
398        coins
399            .into_iter()
400            .map(|coin| {
401                Ok((
402                    OutPoint {
403                        txid: serde_json::from_value(coin["txid"].clone())?,
404                        vout: serde_json::from_value(coin["vout"].clone())?,
405                    },
406                    Amount::from_btc(
407                        coin["amount"]
408                            .as_f64()
409                            .ok_or_else(|| anyhow!("Missing coin amount: {coin}"))?,
410                    )?,
411                ))
412            })
413            .collect()
414    }

416    fn mempool(node: &BitcoinNodeHandle) -> Result<Vec<Txid>> {
417        Ok(node.rpc_client().call("getrawmempool", &[])?)
418    }

420    fn broadcast(node: &BitcoinNodeHandle, tx: &Transaction) -> Result<()> {
421        let _: Txid = node
422            .rpc_client()
423            .call("sendrawtransaction", &[json!(serialize_hex(tx))])?;
424        Ok(())
425    }

427    pub(crate) fn fill_mempool_until_evicted(
428        node: &BitcoinNodeHandle,
429        txids: &[Txid],
430    ) -> Result<()> {
431        // Independent confirmed inputs avoid ancestor limits and CPFP effects.
432        // Each large transaction pays about 10 sat/vB.
433        let script = wallet_script(node)?;
434        let data = bitcoin::script::PushBytesBuf::try_from(vec![0; 80_000])?;
435        let data_script = bitcoin::script::Builder::new()
436            .push_opcode(bitcoin::opcodes::all::OP_RETURN)
437            .push_slice(data)
438            .into_script();
439        for (input, amount) in confirmed_coins(node)? {
440            let filler = signed_spend(
441                node,
442                input,
443                vec![
444                    TxOut {
445                        value: amount - Amount::from_sat(810_000),
446                        script_pubkey: script.clone(),
447                    },
448                    TxOut {
449                        value: Amount::ZERO,
450                        script_pubkey: data_script.clone(),
451                    },
452                ],
453            )?;
454            // Equal-feerate fillers tie for eviction, so Core may trim the one
455            // just submitted. That still raises the floor; check the pool below.
456            if let Err(error) = broadcast(node, &filler)
457                && !error.to_string().contains("mempool full")
458            {
459                return Err(error);
460            }
461            let pool = mempool(node)?;
462            let info: Value = node.rpc_client().call("getmempoolinfo", &[])?;
463            let floor = info["mempoolminfee"]
464                .as_f64()
465                .ok_or_else(|| anyhow!("Missing mempool floor: {info}"))?;
466            let relay_floor = info["minrelaytxfee"]
467                .as_f64()
468                .ok_or_else(|| anyhow!("Missing relay floor: {info}"))?;
469            if txids.iter().all(|txid| !pool.contains(txid)) && floor > relay_floor {
470                return Ok(());
471            }
472        }
473        anyhow::bail!(
474            "Exhausted confirmed coins before eviction and dynamic mempool floor increase"
475        )
476    }

493    #[tokio::test(flavor = "multi_thread")]
494    async fn test_monitor_broadcast_package_recovers_evicted_parent() -> Result<()> {
495        crate::test_helpers::init_test_logging();
496        let dir = tempfile::tempdir()?;
497        let node = BitcoinNodeBuilder::new()
498            .dir(dir.path())
499            .with_initial_blocks(210)
500            .with_extra_args(["-maxmempool=5", "-datacarriersize=100000"])
501            .build()
502            .await?;
503        let mut coins = confirmed_coins(&node)?.into_iter();
504        let (input, amount) = coins.next().ok_or_else(|| anyhow!("No mature coinbase"))?;
505        let script = wallet_script(&node)?;
506        let parent = signed_spend(
507            &node,
508            input,
509            vec![TxOut {
510                value: amount - Amount::from_sat(200),
511                script_pubkey: script.clone(),
512            }],
513        )?;
514        broadcast(&node, &parent)?;
515        let child = signed_spend(
516            &node,
517            OutPoint {
518                txid: parent.compute_txid(),
519                vout: 0,
520            },
521            vec![TxOut {
522                value: parent.output[0].value - Amount::from_sat(10_000),
523                script_pubkey: script.clone(),
524            }],
525        )?;
526        assert!(mempool(&node)?.contains(&parent.compute_txid()));
527
528        fill_mempool_until_evicted(&node, &[parent.compute_txid()])?;
529        assert!(
530            !mempool(&node)?.contains(&parent.compute_txid()),
531            "The low-fee parent must actually be evicted"
532        );
533        let info: Value = node.rpc_client().call("getmempoolinfo", &[])?;
534        let floor_sat_vb = info["mempoolminfee"]
535            .as_f64()
536            .ok_or_else(|| anyhow!("Missing mempool floor: {info}"))?
537            * 100_000.0;
538        assert!(floor_sat_vb > 200.0 / parent.vsize() as f64);
539        assert!(10_200.0 / (parent.vsize() + child.vsize()) as f64 > floor_sat_vb);
540
541        let (monitor, _service) = Monitor::run(
542            monitor_config(&node),
543            Arc::new(hashi::metrics::Metrics::new_default()),
544        )?;
545        // submitpackage can return an RPC success envelope containing rejection.
546        // The monitor must propagate that rejection, not report success.
547        assert!(
548            monitor
549                .broadcast_package(vec![child.clone()])
550                .await
551                .is_err()
552        );
553        let pool = mempool(&node)?;
554        assert!(!pool.contains(&child.compute_txid()));
555        assert!(!pool.contains(&parent.compute_txid()));
556
557        monitor
558            .broadcast_package(vec![parent.clone(), child.clone()])
559            .await?;
560        let pool = mempool(&node)?;
561        assert!(pool.contains(&parent.compute_txid()));
562        assert!(pool.contains(&child.compute_txid()));
563
564        // Already-in-mempool results are idempotent success.
565        monitor
566            .broadcast_package(vec![parent.clone(), child.clone()])
567            .await?;
568        let pool = mempool(&node)?;
569        assert!(pool.contains(&parent.compute_txid()));
570        assert!(pool.contains(&child.compute_txid()));
571        Ok(())
572    }

576        crate::test_helpers::init_test_logging();

601        )?;
602        assert!(parent_info["confirmations"].as_u64().unwrap_or(0) > 0);
603        assert!(!mempool(&node)?.contains(&parent.compute_txid()));
604        let child = signed_spend(
605            &node,
606            OutPoint {
607                txid: parent.compute_txid(),
608                vout: 0,
609            },
610            vec![TxOut {
611                value: parent.output[0].value - Amount::from_sat(10_000),
612                script_pubkey: script,
613            }],
614        )?;
615        assert!(!mempool(&node)?.contains(&child.compute_txid()));
616        let (monitor, _service) = Monitor::run(
617            monitor_config(&node),
618            Arc::new(hashi::metrics::Metrics::new_default()),
619        )?;
620        monitor.broadcast_package(vec![child.clone()]).await?;
621        let pool = mempool(&node)?;
622        assert!(pool.contains(&child.compute_txid()));
