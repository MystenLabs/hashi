27pub struct HashiNodeHandle {
28    config: HashiConfig,
29    /// The running service and Hashi instance. Both are dropped together on shutdown
30    /// to ensure the database lock is released before a new instance can be created.
31    service: Option<(Service, Arc<Hashi>)>,
32}
33
34impl Drop for HashiNodeHandle {
35    /// Tear the node down and wait (bounded) until nothing holds its database
36    /// any more.
37    ///
38    /// Dropping the `Service` aborts the node's tasks, but a `spawn_blocking`
39    /// closure that a task already queued (e.g. the post-reconfig major
40    /// compaction) keeps running on the blocking pool with its own `db`
41    /// clone. The test's environment directory is deleted right after the
42    /// handles drop; if that happens under such a closure, its storage
43    /// workers crash on the missing files and the closure can wedge, and the
44    /// test runtime then waits on it forever (nextest kills the test at
45    /// terminate-after). The database lock is released only once every
46    /// handle is gone, so a successful re-open is the exact "no DB work in
47    /// flight" signal; the same idiom `create_hashi_retry` uses for restarts.
48    fn drop(&mut self) {
49        const MAX_WAIT: std::time::Duration = std::time::Duration::from_secs(5);
50        const STEP: std::time::Duration = std::time::Duration::from_millis(100);
51
52        if self.service.take().is_none() {
53            return;
54        }
55        let Some(db_path) = self.config.db.as_deref() else {
56            return;
57        };
58        let started = std::time::Instant::now();
59        loop {
60            match hashi::db::Database::open(db_path) {
61                Ok(_probe) => return,
62                Err(_) if started.elapsed() < MAX_WAIT => std::thread::sleep(STEP),
63                Err(e) => {
64                    tracing::warn!(
65                        "Hashi node database still locked {:?} after shutdown; \
66                         tearing down anyway: {e}",
67                        started.elapsed()
68                    );
69                    return;
70                }
71            }
72        }
73    }
74}

76impl HashiNodeHandle {
77    pub fn new(config: HashiConfig) -> Result<Self> {
78        Ok(Self {

84    pub async fn start(&mut self) -> Result<()> {
85        if self.service.is_some() {
86            anyhow::bail!("Hashi node already started");
87        }
88        let hashi = Self::create_hashi_retry(&self.config).await?;
89        let service = hashi.clone().start().await?;
90        self.service = Some((service, hashi));
91        Ok(())
92    }
93
94    pub fn is_running(&self) -> bool {
95        self.service.is_some()
96    }
97
98    fn create_hashi(config: &HashiConfig) -> Result<Arc<Hashi>> {
99        let server_version = ServerVersion::new("test-hashi", "0.1.0");
100        let registry = prometheus::Registry::new();
101        Hashi::new_with_registry(server_version, None, config.clone(), &registry)
102    }
103
104    /// Create a Hashi instance with retry logic for database lock contention.
105    ///
106    /// After shutdown, there may be a brief delay before the database lock is released.
107    async fn create_hashi_retry(config: &HashiConfig) -> Result<Arc<Hashi>> {
108        const MAX_ATTEMPTS: u32 = 10;
109
110        for attempt in 1..=MAX_ATTEMPTS {
111            match Self::create_hashi(config) {
112                Ok(hashi) => return Ok(hashi),
113                Err(e) if attempt == MAX_ATTEMPTS => return Err(e),
114                Err(e) => {
115                    tracing::debug!(
116                        "Failed to create Hashi (attempt {attempt}/{MAX_ATTEMPTS}): {e}"
117                    );
118                    tokio::time::sleep(POLL_INTERVAL).await;
119                }
120            }
121        }
122        unreachable!()
123    }
124
125    pub async fn shutdown(&mut self) {
126        let Some((service, _hashi)) = self.service.take() else {
127            tracing::warn!("Hashi node not running, cannot shutdown");
128            return;
129        };
130        let result = service.shutdown().await;
131        if let Err(e) = result {
132            tracing::warn!("Hashi shutdown error: {e}");
133        }
134    }
135
136    pub async fn restart(&mut self) -> Result<()> {
137        self.shutdown().await;
138        self.start().await
139    }

141    /// Open the node's DB independently (node must be stopped).
142    pub fn open_db(&self) -> Result<hashi::db::Database> {
143        assert!(
144            self.service.is_none(),
145            "Cannot open DB while node is running"
146        );
147        let db_path = self.config.db.as_deref().expect("db path not set");
148        for _ in 0..10 {
149            if let Ok(db) = hashi::db::Database::open(db_path) {
150                return Ok(db);
151            }
152            std::thread::sleep(POLL_INTERVAL);
153        }
154        hashi::db::Database::open(db_path)
155    }

190    pub async fn wait_for_mpc_key(&self, timeout: std::time::Duration) -> Result<()> {
191        tokio::time::timeout(timeout, self.wait_for_mpc_key_inner())

193            .map_err(|_| anyhow::anyhow!("MPC key timed out after {:?}", timeout))?

196    async fn wait_for_mpc_key_inner(&self) -> Result<()> {
197        loop {
198            if let Some(mpc_handle) = self.hashi().mpc_handle()
199                && mpc_handle.public_key().is_some()
200                && self.hashi().signing_verifying_key().is_some()
201            {
202                return Ok(());
203            }
204            tokio::time::sleep(POLL_INTERVAL).await;
205        }

243    pub fn current_epoch(&self) -> Option<u64> {
244        self.hashi()
245            .onchain_state_opt()
246            .map(|s| s.state().hashi().committees.epoch())
247    }
248
249    pub async fn wait_for_epoch(
250        &self,
251        target_epoch: u64,
252        timeout: std::time::Duration,
253    ) -> Result<()> {
254        tokio::time::timeout(timeout, self.wait_for_epoch_inner(target_epoch))
255            .await
256            .map_err(|_| anyhow::anyhow!("Timed out waiting for Hashi epoch {target_epoch}"))
257    }
258
259    async fn wait_for_epoch_inner(&self, target_epoch: u64) {
260        loop {
261            let onchain_state = match self.hashi().onchain_state_opt() {
262                Some(state) => state,
263                None => {
264                    tokio::time::sleep(POLL_INTERVAL).await;
265                    continue;
266                }
267            };
268            let epoch = onchain_state.state().hashi().committees.epoch();
269            if epoch >= target_epoch {
270                return;
271            }
272            tokio::time::sleep(POLL_INTERVAL).await;
273        }
274    }

314    /// Send the launch switch (`hashi::finish_publish`): wait until every
315    /// `expected` validator is fully registered, assert genesis has not
316    /// started, then configure the deploy and hand the `UpgradeCap` into
317    /// on-chain custody — unlocking the genesis `start_reconfig`. The
318    /// initial committee is exactly the set of fully-registered validators
319    /// at this moment.
320    pub async fn launch_genesis(
321        &self,
322        client: &mut sui_rpc::Client,
323        publisher: &sui_crypto::ed25519::Ed25519PrivateKey,
324        expected: &[Address],
325        timeout: std::time::Duration,
326    ) -> Result<()> {
327        wait_for_registered_validators(&self.nodes[0], expected, timeout).await?;
328
329        // The gate must have held while validators registered: even though
330        // every node has been retrying start_reconfig, genesis cannot begin
331        // before the launch tx below.
332        if let Some(onchain) = self.nodes[0].hashi().onchain_state_opt() {
333            anyhow::ensure!(
334                onchain.current_committee().is_none()
335                    && onchain
336                        .state()
337                        .hashi()
338                        .committees
339                        .pending_epoch_change()
340                        .is_none(),
341                "genesis started before the launch tx (finish_publish)"
342            );
343        }
344
345        hashi::publish::finish_publish(
346            client,
347            &publisher.clone().into(),
348            &self.ids,
349            self.upgrade_cap_id,
350            hashi::constants::BITCOIN_REGTEST_CHAIN_ID,
351            &self.guardian,
352            &hashi::publish::BitcoinConfigOverrides::default(),
353        )
354        .await?;
355        debug!("finish_publish sent — genesis unlocked");
356        Ok(())
357    }
358

360        futures::future::try_join_all(self.nodes.iter_mut().map(|node| node.restart())).await?;
361        Ok(())
362    }

368    pub async fn register_and_start_pending_node(&mut self, client: sui_rpc::Client) -> Result<()> {
369        // The registration call must route through the package that is enabled
370        // NOW (the default boot upgrades and disables v1 — the original id
371        // aborts at `versioning::assert_version_enabled`). The pending node has
372        // no Hashi instance to read the chain through, so borrow a running
373        // node's onchain state for the executor's version routing.
374        let onchain_state = self
375            .nodes
376            .iter()
377            .find(|n| n.is_running())
378            .map(|n| n.hashi().onchain_state().clone());
379        let node = self
380            .nodes
381            .iter_mut()
382            .find(|n| n.service.is_none())
383            .ok_or_else(|| anyhow::anyhow!("no pending nodes to start"))?;
384        // The node's ports were picked (and released) when the network was
385        // built, and have sat unbound through everything the test did since —
386        // minutes, under the upgraded-chain boot — so on a busy CI runner
387        // another process may hold them by now, and the gRPC server panics on
388        // bind. Re-pick immediately before the registration that publishes the
389        // endpoint on-chain, shrinking the reservation window to milliseconds.
390        let listen_addr =
391            std::net::SocketAddr::from(([127, 0, 0, 1], hashi::config::get_available_port()));
392        node.config.listen_address = Some(listen_addr);
393        node.config.endpoint_url = Some(format!("https://{listen_addr}"));
394        node.config.metrics_http_address = Some(std::net::SocketAddr::from((
395            [127, 0, 0, 1],
396            hashi::config::get_available_port(),
397        )));
398        register_onchain(client, &node.config, onchain_state.as_ref()).await?;
399        node.start().await?;
400        Ok(())
401    }
402
403    /// Start every validator that isn't already running, staggered to avoid the
404    /// Kyoto peer-ban cascade on regtest. Does NOT register them on-chain —
405    /// registration is the operator's responsibility (e.g. via the CLI in
406    /// localnet `--manual` mode); a node that boots already-registered simply
407    /// publishes its next-epoch keys and proceeds to genesis.
408    pub async fn start_pending_validators(&mut self) -> Result<()> {
409        let mut first = true;
410        for node in self.nodes.iter_mut().filter(|n| !n.is_running()) {
411            if !first {
412                tokio::time::sleep(std::time::Duration::from_secs(2)).await;
413            }
414            first = false;
415            node.start().await?;
416        }
417        Ok(())
418    }
419}

506    pub async fn build(
507        self,
508        dir: &Path,
509        sui: &SuiNetworkHandle,
510        bitcoin: &BitcoinNodeHandle,
511        hashi_ids: HashiIds,
512        upgrade_cap_id: Address,
513        guardian: hashi::publish::GuardianConfig,
514    ) -> Result<HashiNetwork> {
515        // Start a mock screener server for integration tests
516        let (screener_addr, screener_service) =
517            hashi_screener::test_utils::start_mock_screener_server().await;
518        let screener_endpoint = format!("http://{}", screener_addr);
519
520        let bitcoin_rpc = bitcoin.rpc_url().to_owned();
521        let sui_rpc = sui.rpc_url.clone();
522        // Interpose the severable proxy for the selected node before the
523        // configs are built, so that node dials the proxy from boot.
524        let sui_rpc_proxy = match self.sui_rpc_proxy_node {
525            Some(index) => {
526                assert!(index < self.num_nodes, "proxied node index out of range");
527                let target: std::net::SocketAddr = sui_rpc
528                    .strip_prefix("http://")
529                    .ok_or_else(|| anyhow::anyhow!("unexpected sui rpc url format: {sui_rpc}"))?
530                    .parse()?;
531                Some(crate::tcp_proxy::TcpProxy::start(target).await?)
532            }
533            None => None,
534        };
535        let service_info = sui
536            .client
537            .clone()
538            .ledger_client()
539            .get_service_info(GetServiceInfoRequest::default())
540            .await?
541            .into_inner();
542
543        // Resolve the corrupt shares target index to a validator address.
544        let corrupt_target_address = self.test_corrupt_shares_target.map(|idx| {
545            *sui.validator_keys
546                .keys()
547                .nth(idx)
548                .expect("corrupt target index out of range")
549        });
550
551        let mut configs = Vec::with_capacity(self.num_nodes);
552        for (i, (validator_address, private_key)) in
553            sui.validator_keys.iter().take(self.num_nodes).enumerate()
554        {
555            let mut config = HashiConfig::new_for_testing();
556            config.test_weight_divisor = self.test_weight_divisor;
557            config.test_batch_size_per_weight = self.test_batch_size_per_weight;
558            config.withdrawal_batching_delay_ms = self.withdrawal_batching_delay_ms;
559            config.withdrawal_max_batch_size = self.withdrawal_max_batch_size;
560            config.max_mempool_chain_depth = self.max_mempool_chain_depth;
561            // All nodes EXCEPT the target corrupt shares for the target.
562            if let Some(target_addr) = corrupt_target_address
563                && Some(i) != self.test_corrupt_shares_target
564            {
565                config.test_corrupt_shares_for = Some(target_addr);
566            }
567            // Deliberately NO local `guardian_endpoint`: nodes must resolve
568            // the guardian client lazily from the on-chain guardian_url set
569            // by the launch tx, so every e2e run exercises the
570            // guardian-set-up-last path.
571            config.hashi_ids = Some(hashi_ids);
572            config.validator_address = Some(*validator_address);
573            config.operator_private_key = Some(private_key.to_pem()?);
574            config.sui_rpc = match (&sui_rpc_proxy, self.sui_rpc_proxy_node) {
575                (Some(proxy), Some(index)) if index == i => Some(proxy.url()),
576                _ => Some(sui_rpc.clone()),
577            };
578            config.bitcoin_rpc = Some(bitcoin_rpc.clone());
579            config.bitcoin_rpc_auth = Some(hashi::btc_monitor::config::BtcRpcAuth::UserPass(
580                crate::bitcoin_node::RPC_USER.into(),
581                crate::bitcoin_node::RPC_PASSWORD.into(),
582            ));
583            config.bitcoin_trusted_peers = Some(vec![bitcoin.p2p_address()]);
584            config.bitcoin_chain_id = Some(hashi::constants::BITCOIN_REGTEST_CHAIN_ID.to_string());
585            config.sui_chain_id = service_info.chain_id.clone();
586            config.screener_endpoint = Some(screener_endpoint.clone());
587            config.db = Some(dir.join(validator_address.to_string()));
588            configs.push(config);
589        }
590
591        let initially_active = self.num_initially_active_nodes.unwrap_or(configs.len());
592        assert!(
593            initially_active <= configs.len(),
594            "initially_active ({initially_active}) must be <= num_nodes ({})",
595            configs.len()
596        );
597        // Nodes register themselves on startup, and will trigger
598        // start_reconfig + DKG + end_reconfig automatically once the
599        // publisher sends finish_publish (the launch switch), which the
600        // harness does after all initially-active validators are registered.
601        let mut nodes = Vec::with_capacity(configs.len());
602        for config in configs {
603            let node_handle = HashiNodeHandle::new(config)?;
604            nodes.push(node_handle);
605        }
606        // Start only the active nodes.
607        // Stagger startup so each validator's Kyoto light client finishes its
608        // initial compact-filter-header sync before the next one connects to
609        // the same bitcoind P2P peer. Without this delay all Kyoto instances
610        // race on CFilter headers, triggering peer bans and crashes on regtest
611        // (which has no DNS seeds for recovery).
612        for (i, node) in nodes[..initially_active].iter_mut().enumerate() {
613            if i > 0 {
614                tokio::time::sleep(std::time::Duration::from_secs(2)).await;
615            }
616            node.start().await?;
617            debug!(
618                "Created Hashi node {} at listen: {}, endpoint: {}, metrics: {}",
619                node.config.validator_address()?,
620                node.listen_address(),
621                node.endpoint_url(),
622                node.metrics_address()
623            );
624        }
625
626        let network = HashiNetwork {
627            ids: hashi_ids,
628            upgrade_cap_id,
629            guardian,
630            nodes,
631            sui_rpc_proxy,
632            _screener_service: screener_service,
633        };
634
635        // Unlock genesis, then wait for the initial committee to appear
636        // on-chain, which indicates that the genesis bootstrap
637        // (start_reconfig → DKG → end_reconfig) has completed. Skipped when
638        // no nodes are started up front (e.g. localnet `--manual` mode,
639        // where registration and launch are driven externally after the
640        // validators are launched).
641        if initially_active > 0 {
642            // The genesis committee is formed from whoever is fully
643            // registered when start_reconfig lands, so every
644            // initially-active validator must be registered (with its
645            // next-epoch keys) before the launch tx unlocks it.
646            let expected = network.nodes[..initially_active]
647                .iter()
648                .map(|node| node.config.validator_address())
649                .collect::<Result<Vec<_>>>()?;
650            let publisher = sui
651                .user_keys
652                .first()
653                .ok_or_else(|| anyhow::anyhow!("no publisher key in Sui network handle"))?;
654            let mut client = sui.client.clone();
655            network
656                .launch_genesis(
657                    &mut client,
658                    publisher,
659                    &expected,
660                    std::time::Duration::from_secs(120),
661                )
662                .await?;
663
664            let genesis_timeout = std::time::Duration::from_secs(120);
665            tokio::time::timeout(genesis_timeout, async {
666                loop {
667                    if let Some(onchain) = network.nodes[0].hashi().onchain_state_opt()
668                        && onchain.current_committee().is_some()
669                        && onchain
670                            .state()
671                            .hashi()
672                            .committees
673                            .pending_epoch_change()
674                            .is_none()
675                    {
676                        break;
677                    }
678                    tokio::time::sleep(POLL_INTERVAL).await;
679                }
680            })
681            .await
682            .map_err(|_| anyhow::anyhow!("Timed out waiting for initial committee to form"))?;
683            debug!("Initial committee formed on-chain");
684        }
685
686        Ok(network)
687    }
688}

690/// Wait until every `expected` validator is fully registered on-chain:
691/// present in the members bag with its next-epoch encryption key set (the
692/// last piece a node registers before it is eligible for the genesis
693/// committee).
694pub async fn wait_for_registered_validators(
695    node: &HashiNodeHandle,
696    expected: &[Address],
697    timeout: std::time::Duration,
698) -> Result<()> {
699    tokio::time::timeout(timeout, async {
700        loop {
701            if let Some(onchain) = node.hashi().onchain_state_opt()
702                && expected.iter().all(|address| {
703                    onchain
704                        .state()
705                        .hashi()
706                        .committees
707                        .members()
708                        .get(address)
709                        .is_some_and(|m| m.next_epoch_encryption_public_key().is_some())
710                })
711            {
712                return;
713            }
714            tokio::time::sleep(POLL_INTERVAL).await;
715        }
716    })
717    .await
718    .map_err(|_| anyhow::anyhow!("Timed out waiting for validators to register on-chain"))
719}

727/// `onchain_state` supplies the executor's version routing so the register
728/// call executes the package that is enabled at submission time. Without it
729/// the executor falls back to the ORIGINAL package id, which aborts once the
730/// post-upgrade boot has disabled v1.
731async fn register_onchain(
732    client: sui_rpc::Client,
733    config: &HashiConfig,
734    onchain_state: Option<&hashi::onchain::OnchainState>,
735) -> Result<()> {
736    let signer = config.operator_private_key()?;
737    let hashi_ids = config.hashi_ids();
738    let mut executor = hashi::sui_tx_executor::SuiTxExecutor::new(client, signer, hashi_ids);
739    if let Some(onchain_state) = onchain_state {
740        executor = executor.with_onchain_state(onchain_state);
741    }
742    executor
743        .execute_register_or_update_validator(config, None, None, None, true)
744        .await
745        .map(|_| ())
746}

748/// `call_package_id` is the package the Move call routes through — the
749/// chain's LATEST package (resolve it from a node's onchain state after the
750/// boot-time upgrade); the original id aborts once v1 is disabled.
751pub async fn update_tls_public_key(
752    client: sui_rpc::Client,
753    config: &HashiConfig,
754    call_package_id: Address,
755) -> Result<()> {
756    let hashi_ids = config.hashi_ids();
757    let private_key = config.operator_private_key()?;
758    let validator_address = config.validator_address()?;
759    let tls_key = config.tls_public_key()?;
760
761    let mut executor = hashi::sui_tx_executor::SuiTxExecutor::new(client, private_key, hashi_ids);
762
763    let mut builder = TransactionBuilder::new();
764
765    let hashi_arg = builder.object(
766        ObjectInput::new(hashi_ids.hashi_object_id)
767            .as_shared()
768            .with_mutable(true),
769    );
770    let validator_address_arg = builder.pure(&validator_address);
771    let tls_key_arg = builder.pure(&tls_key.as_bytes().to_vec());
772
773    builder.move_call(
774        Function::new(
775            call_package_id,
776            Identifier::from_static("validator"),
777            Identifier::from_static("update_tls_public_key"),
778        ),
779        vec![hashi_arg, validator_address_arg, tls_key_arg],
780    );
781
782    let response = executor.execute(builder).await?;
783    assert!(
784        response.transaction().effects().status().success(),
785        "update_tls_public_key failed"
786    );
787
788    Ok(())
789}
