27pub struct HashiNodeHandle {
28    config: HashiConfig,
29    /// The running service and Hashi instance. Both are dropped together on shutdown
30    /// to ensure the database lock is released before a new instance can be created.
31    service: Option<(Service, Arc<Hashi>)>,
32}
33
34impl Drop for HashiNodeHandle {
35    /// Tear the node down and wait (bounded) until nothing holds its database
36    /// any more.
37    ///
38    /// Dropping the `Service` aborts the node's tasks, but a `spawn_blocking`
39    /// closure that a task already queued (e.g. the post-reconfig major
40    /// compaction) keeps running on the blocking pool with its own `db`
41    /// clone. The test's environment directory is deleted right after the
42    /// handles drop; if that happens under such a closure, its storage
43    /// workers crash on the missing files and the closure can wedge, and the
44    /// test runtime then waits on it forever (nextest kills the test at
45    /// terminate-after). The database lock is released only once every
46    /// handle is gone, so a successful re-open is the exact "no DB work in
47    /// flight" signal; the same idiom `create_hashi_retry` uses for restarts.
48    fn drop(&mut self) {
49        const MAX_WAIT: std::time::Duration = std::time::Duration::from_secs(5);
50        const STEP: std::time::Duration = std::time::Duration::from_millis(100);
51
52        if self.service.take().is_none() {
53            return;
54        }
55        let Some(db_path) = self.config.db.as_deref() else {
56            return;
57        };
58        let started = std::time::Instant::now();
59        loop {
60            match hashi::db::Database::open(db_path) {
61                Ok(_probe) => return,
62                Err(_) if started.elapsed() < MAX_WAIT => std::thread::sleep(STEP),
63                Err(e) => {
64                    tracing::warn!(
65                        "Hashi node database still locked {:?} after shutdown; \
66                         tearing down anyway: {e}",
67                        started.elapsed()
68                    );
69                    return;
70                }
71            }
72        }
73    }
74}

76impl HashiNodeHandle {
77    pub fn new(config: HashiConfig) -> Result<Self> {
78        Ok(Self {

84    pub async fn start(&mut self) -> Result<()> {
85        if self.service.is_some() {
86            anyhow::bail!("Hashi node already started");
87        }
88        let hashi = Self::create_hashi_retry(&self.config).await?;
89        let service = hashi.clone().start().await?;
90        self.service = Some((service, hashi));
91        Ok(())
92    }
93
94    pub fn is_running(&self) -> bool {
95        self.service.is_some()
96    }
97
98    fn create_hashi(config: &HashiConfig) -> Result<Arc<Hashi>> {
99        let server_version = ServerVersion::new("test-hashi", "0.1.0");
100        let registry = prometheus::Registry::new();
101        Hashi::new_with_registry(server_version, None, config.clone(), &registry)
102    }
103
104    /// Create a Hashi instance with retry logic for database lock contention.
105    ///
106    /// After shutdown, there may be a brief delay before the database lock is released.
107    async fn create_hashi_retry(config: &HashiConfig) -> Result<Arc<Hashi>> {
108        const MAX_ATTEMPTS: u32 = 10;
109
110        for attempt in 1..=MAX_ATTEMPTS {
111            match Self::create_hashi(config) {
112                Ok(hashi) => return Ok(hashi),
113                Err(e) if attempt == MAX_ATTEMPTS => return Err(e),
114                Err(e) => {
115                    tracing::debug!(
116                        "Failed to create Hashi (attempt {attempt}/{MAX_ATTEMPTS}): {e}"
117                    );
118                    tokio::time::sleep(POLL_INTERVAL).await;
119                }
120            }
121        }
122        unreachable!()
123    }
124
125    pub async fn shutdown(&mut self) {
126        let Some((service, _hashi)) = self.service.take() else {
127            tracing::warn!("Hashi node not running, cannot shutdown");
128            return;
129        };
130        let result = service.shutdown().await;
131        if let Err(e) = result {
132            tracing::warn!("Hashi shutdown error: {e}");
133        }
134    }
135
136    pub async fn restart(&mut self) -> Result<()> {
137        self.shutdown().await;
138        self.start().await
139    }

141    /// Open the node's DB independently (node must be stopped).
142    pub fn open_db(&self) -> Result<hashi::db::Database> {
143        assert!(
144            self.service.is_none(),
145            "Cannot open DB while node is running"
146        );
147        let db_path = self.config.db.as_deref().expect("db path not set");
148        for _ in 0..10 {
149            if let Ok(db) = hashi::db::Database::open(db_path) {
150                return Ok(db);
151            }
152            std::thread::sleep(POLL_INTERVAL);
153        }
154        hashi::db::Database::open(db_path)
155    }

190    pub async fn wait_for_mpc_key(&self, timeout: std::time::Duration) -> Result<()> {
191        tokio::time::timeout(timeout, self.wait_for_mpc_key_inner())
192            .await
193            .map_err(|_| anyhow::anyhow!("MPC key timed out after {:?}", timeout))?

196    async fn wait_for_mpc_key_inner(&self) -> Result<()> {
197        loop {
198            if let Some(mpc_handle) = self.hashi().mpc_handle()
199                && mpc_handle.public_key().is_some()
200                && self.hashi().signing_verifying_key().is_some()
201            {
202                return Ok(());
203            }
204            tokio::time::sleep(POLL_INTERVAL).await;
205        }

243    pub fn current_epoch(&self) -> Option<u64> {
244        self.hashi()
245            .onchain_state_opt()
246            .map(|s| s.state().hashi().committees.epoch())
247    }
248
249    pub async fn wait_for_epoch(
250        &self,
251        target_epoch: u64,
252        timeout: std::time::Duration,
253    ) -> Result<()> {
254        tokio::time::timeout(timeout, self.wait_for_epoch_inner(target_epoch))
255            .await
256            .map_err(|_| anyhow::anyhow!("Timed out waiting for Hashi epoch {target_epoch}"))
257    }
258
259    async fn wait_for_epoch_inner(&self, target_epoch: u64) {
260        loop {
261            let onchain_state = match self.hashi().onchain_state_opt() {
262                Some(state) => state,
263                None => {
264                    tokio::time::sleep(POLL_INTERVAL).await;
265                    continue;
266                }
267            };
268            let epoch = onchain_state.state().hashi().committees.epoch();
269            if epoch >= target_epoch {
270                return;
271            }
272            tokio::time::sleep(POLL_INTERVAL).await;
273        }
274    }

308    pub fn nodes_mut(&mut self) -> &mut [HashiNodeHandle] {

311
312    /// Send the launch switch (`hashi::finish_publish`): wait until every
313    /// `expected` validator is fully registered, assert genesis has not
314    /// started, then configure the deploy and hand the `UpgradeCap` into
315    /// on-chain custody — unlocking the genesis `start_reconfig`. The
316    /// initial committee is exactly the set of fully-registered validators
317    /// at this moment.
318    pub async fn launch_genesis(
319        &self,
320        client: &mut sui_rpc::Client,
321        publisher: &sui_crypto::ed25519::Ed25519PrivateKey,
322        expected: &[Address],
323        timeout: std::time::Duration,
324    ) -> Result<()> {
325        wait_for_registered_validators(&self.nodes[0], expected, timeout).await?;
326
327        // The gate must have held while validators registered: even though
328        // every node has been retrying start_reconfig, genesis cannot begin
329        // before the launch tx below.
330        if let Some(onchain) = self.nodes[0].hashi().onchain_state_opt() {
331            anyhow::ensure!(
332                onchain.current_committee().is_none()
333                    && onchain
334                        .state()
335                        .hashi()
336                        .committees
337                        .pending_epoch_change()
338                        .is_none(),
339                "genesis started before the launch tx (finish_publish)"
340            );
341        }
342
343        hashi::publish::finish_publish(
344            client,
345            &publisher.clone().into(),
346            &self.ids,
347            self.upgrade_cap_id,
348            hashi::constants::BITCOIN_REGTEST_CHAIN_ID,
349            &self.guardian,
350            &hashi::publish::BitcoinConfigOverrides::default(),
351        )
352        .await?;
353        debug!("finish_publish sent — genesis unlocked");
354        Ok(())
355    }
356
357    pub async fn restart(&mut self) -> Result<()> {
358        futures::future::try_join_all(self.nodes.iter_mut().map(|node| node.restart())).await?;
359        Ok(())
360    }

366    pub async fn register_and_start_pending_node(&mut self, client: sui_rpc::Client) -> Result<()> {
367        // The registration call must route through the package that is enabled
368        // NOW (the default boot upgrades and disables v1 — the original id
369        // aborts at `versioning::assert_version_enabled`). The pending node has
370        // no Hashi instance to read the chain through, so borrow a running
371        // node's onchain state for the executor's version routing.
372        let onchain_state = self
373            .nodes
374            .iter()
375            .find(|n| n.is_running())
376            .map(|n| n.hashi().onchain_state().clone());
377        let node = self
378            .nodes
379            .iter_mut()
380            .find(|n| n.service.is_none())
381            .ok_or_else(|| anyhow::anyhow!("no pending nodes to start"))?;
382        // The node's ports were picked (and released) when the network was
383        // built, and have sat unbound through everything the test did since —
384        // minutes, under the upgraded-chain boot — so on a busy CI runner
385        // another process may hold them by now, and the gRPC server panics on
386        // bind. Re-pick immediately before the registration that publishes the
387        // endpoint on-chain, shrinking the reservation window to milliseconds.
388        let listen_addr =
389            std::net::SocketAddr::from(([127, 0, 0, 1], hashi::config::get_available_port()));
390        node.config.listen_address = Some(listen_addr);
391        node.config.endpoint_url = Some(format!("https://{listen_addr}"));
392        node.config.metrics_http_address = Some(std::net::SocketAddr::from((
393            [127, 0, 0, 1],
394            hashi::config::get_available_port(),
395        )));
396        register_onchain(client, &node.config, onchain_state.as_ref()).await?;
397        node.start().await?;
398        Ok(())
399    }
400
401    /// Start every validator that isn't already running, staggered to avoid the
402    /// Kyoto peer-ban cascade on regtest. Does NOT register them on-chain —
403    /// registration is the operator's responsibility (e.g. via the CLI in
404    /// localnet `--manual` mode); a node that boots already-registered simply
405    /// publishes its next-epoch keys and proceeds to genesis.
406    pub async fn start_pending_validators(&mut self) -> Result<()> {
407        let mut first = true;
408        for node in self.nodes.iter_mut().filter(|n| !n.is_running()) {
409            if !first {
410                tokio::time::sleep(std::time::Duration::from_secs(2)).await;
411            }
412            first = false;
413            node.start().await?;
414        }
415        Ok(())
416    }
417}

504    pub async fn build(
505        self,
506        dir: &Path,
507        sui: &SuiNetworkHandle,
508        bitcoin: &BitcoinNodeHandle,
509        hashi_ids: HashiIds,
510        upgrade_cap_id: Address,
511        guardian: hashi::publish::GuardianConfig,
512    ) -> Result<HashiNetwork> {
513        let bitcoin_rpc = bitcoin.rpc_url().to_owned();
514        let sui_rpc = sui.rpc_url.clone();
515        // Interpose the severable proxy for the selected node before the
516        // configs are built, so that node dials the proxy from boot.
517        let sui_rpc_proxy = match self.sui_rpc_proxy_node {
518            Some(index) => {
519                assert!(index < self.num_nodes, "proxied node index out of range");
520                let target: std::net::SocketAddr = sui_rpc
521                    .strip_prefix("http://")
522                    .ok_or_else(|| anyhow::anyhow!("unexpected sui rpc url format: {sui_rpc}"))?
523                    .parse()?;
524                Some(crate::tcp_proxy::TcpProxy::start(target).await?)
525            }
526            None => None,
527        };
528        let service_info = sui
529            .client
530            .clone()
531            .ledger_client()
532            .get_service_info(GetServiceInfoRequest::default())
533            .await?
534            .into_inner();
535
536        // Resolve the corrupt shares target index to a validator address.
537        let corrupt_target_address = self.test_corrupt_shares_target.map(|idx| {
538            *sui.validator_keys
539                .keys()
540                .nth(idx)
541                .expect("corrupt target index out of range")
542        });
543
544        let mut configs = Vec::with_capacity(self.num_nodes);
545        for (i, (validator_address, private_key)) in
546            sui.validator_keys.iter().take(self.num_nodes).enumerate()
547        {
548            let mut config = HashiConfig::new_for_testing();
549            config.test_weight_divisor = self.test_weight_divisor;
550            config.test_batch_size_per_weight = self.test_batch_size_per_weight;
551            config.withdrawal_batching_delay_ms = self.withdrawal_batching_delay_ms;
552            config.withdrawal_max_batch_size = self.withdrawal_max_batch_size;
553            config.max_mempool_chain_depth = self.max_mempool_chain_depth;
554            // All nodes EXCEPT the target corrupt shares for the target.
555            if let Some(target_addr) = corrupt_target_address
556                && Some(i) != self.test_corrupt_shares_target
557            {
558                config.test_corrupt_shares_for = Some(target_addr);
559            }
560            // Deliberately NO local `guardian_endpoint`: nodes must resolve
561            // the guardian client lazily from the on-chain guardian_url set
562            // by the launch tx, so every e2e run exercises the
563            // guardian-set-up-last path.
564            config.hashi_ids = Some(hashi_ids);
565            config.validator_address = Some(*validator_address);
566            config.operator_private_key = Some(private_key.to_pem()?);
567            config.sui_rpc = match (&sui_rpc_proxy, self.sui_rpc_proxy_node) {
568                (Some(proxy), Some(index)) if index == i => Some(proxy.url()),
569                _ => Some(sui_rpc.clone()),
570            };
571            config.bitcoin_rpc = Some(bitcoin_rpc.clone());
572            config.bitcoin_rpc_auth = Some(hashi::btc_monitor::config::BtcRpcAuth::UserPass(
573                crate::bitcoin_node::RPC_USER.into(),
574                crate::bitcoin_node::RPC_PASSWORD.into(),
575            ));
576            config.bitcoin_trusted_peers = Some(vec![bitcoin.p2p_address()]);
577            config.bitcoin_chain_id = Some(hashi::constants::BITCOIN_REGTEST_CHAIN_ID.to_string());
578            config.sui_chain_id = service_info.chain_id.clone();
579            let node_name = validator_address.to_string();
580            config.backup_dir = dir.join("backups").join(&node_name);
581            config.db = Some(dir.join(node_name));
582            configs.push(config);
583        }
584
585        let initially_active = self.num_initially_active_nodes.unwrap_or(configs.len());
586        assert!(
587            initially_active <= configs.len(),
588            "initially_active ({initially_active}) must be <= num_nodes ({})",
589            configs.len()
590        );
591        // Nodes register themselves on startup, and will trigger
592        // start_reconfig + DKG + end_reconfig automatically once the
593        // publisher sends finish_publish (the launch switch), which the
594        // harness does after all initially-active validators are registered.
595        let mut nodes = Vec::with_capacity(configs.len());
596        for config in configs {
597            let node_handle = HashiNodeHandle::new(config)?;
598            nodes.push(node_handle);
599        }
600        // Start only the active nodes.
601        // Stagger startup so each validator's Kyoto light client finishes its
602        // initial compact-filter-header sync before the next one connects to
603        // the same bitcoind P2P peer. Without this delay all Kyoto instances
604        // race on CFilter headers, triggering peer bans and crashes on regtest
605        // (which has no DNS seeds for recovery).
606        for (i, node) in nodes[..initially_active].iter_mut().enumerate() {
607            if i > 0 {
608                tokio::time::sleep(std::time::Duration::from_secs(2)).await;
609            }
610            node.start().await?;
611            debug!(
612                "Created Hashi node {} at listen: {}, endpoint: {}, metrics: {}",
613                node.config.validator_address()?,
614                node.listen_address(),
615                node.endpoint_url(),
616                node.metrics_address()
617            );
618        }
619
620        let network = HashiNetwork {
621            ids: hashi_ids,
622            upgrade_cap_id,
623            guardian,
624            nodes,
625            sui_rpc_proxy,
626        };
627
628        // Unlock genesis, then wait for the initial committee to appear
629        // on-chain, which indicates that the genesis bootstrap
630        // (start_reconfig → DKG → end_reconfig) has completed. Skipped when
631        // no nodes are started up front (e.g. localnet `--manual` mode,
632        // where registration and launch are driven externally after the
633        // validators are launched).
634        if initially_active > 0 {
635            // The genesis committee is formed from whoever is fully
636            // registered when start_reconfig lands, so every
637            // initially-active validator must be registered (with its
638            // next-epoch keys) before the launch tx unlocks it.
639            let expected = network.nodes[..initially_active]
640                .iter()
641                .map(|node| node.config.validator_address())
642                .collect::<Result<Vec<_>>>()?;
643            let publisher = sui
644                .user_keys
645                .first()
646                .ok_or_else(|| anyhow::anyhow!("no publisher key in Sui network handle"))?;
647            let mut client = sui.client.clone();
648            network
649                .launch_genesis(
650                    &mut client,
651                    publisher,
652                    &expected,
653                    std::time::Duration::from_secs(120),
654                )
655                .await?;
656
657            let genesis_timeout = std::time::Duration::from_secs(120);
658            tokio::time::timeout(genesis_timeout, async {
659                loop {
660                    if let Some(onchain) = network.nodes[0].hashi().onchain_state_opt()
661                        && onchain.current_committee().is_some()
662                        && onchain
663                            .state()
664                            .hashi()
665                            .committees
666                            .pending_epoch_change()
667                            .is_none()
668                    {
669                        break;
670                    }
671                    tokio::time::sleep(POLL_INTERVAL).await;
672                }
673            })
674            .await
675            .map_err(|_| anyhow::anyhow!("Timed out waiting for initial committee to form"))?;
676            debug!("Initial committee formed on-chain");
677        }
678
679        Ok(network)
680    }
681}

683/// Wait until every `expected` validator is fully registered on-chain:
684/// present in the members bag with its next-epoch encryption key set (the
685/// last piece a node registers before it is eligible for the genesis
686/// committee).
687pub async fn wait_for_registered_validators(
688    node: &HashiNodeHandle,
689    expected: &[Address],
690    timeout: std::time::Duration,
691) -> Result<()> {
692    tokio::time::timeout(timeout, async {
693        loop {
694            if let Some(onchain) = node.hashi().onchain_state_opt()
695                && expected.iter().all(|address| {
696                    onchain
697                        .state()
698                        .hashi()
699                        .committees
700                        .members()
701                        .get(address)
702                        .is_some_and(|m| m.next_epoch_encryption_public_key().is_some())
703                })
704            {
705                return;
706            }
707            tokio::time::sleep(POLL_INTERVAL).await;
708        }
709    })
710    .await
711    .map_err(|_| anyhow::anyhow!("Timed out waiting for validators to register on-chain"))
712}

720/// `onchain_state` supplies the executor's version routing so the register
721/// call executes the package that is enabled at submission time. Without it
722/// the executor falls back to the ORIGINAL package id, which aborts once the
723/// post-upgrade boot has disabled v1.
724async fn register_onchain(
725    client: sui_rpc::Client,
726    config: &HashiConfig,
727    onchain_state: Option<&hashi::onchain::OnchainState>,
728) -> Result<()> {
729    let signer = config.operator_private_key()?;
730    let hashi_ids = config.hashi_ids();
731    let mut executor = hashi::sui_tx_executor::SuiTxExecutor::new(client, signer, hashi_ids);
732    if let Some(onchain_state) = onchain_state {
733        executor = executor.with_onchain_state(onchain_state);
734    }
735    executor
736        .execute_register_or_update_validator(config, None, None, None, true)
737        .await
738        .map(|_| ())
739}

741/// `call_package_id` is the package the Move call routes through — the
742/// chain's LATEST package (resolve it from a node's onchain state after the
743/// boot-time upgrade); the original id aborts once v1 is disabled.
744pub async fn update_tls_public_key(
745    client: sui_rpc::Client,
746    config: &HashiConfig,
747    call_package_id: Address,
748) -> Result<()> {
749    let hashi_ids = config.hashi_ids();
750    let private_key = config.operator_private_key()?;
751    let validator_address = config.validator_address()?;
752    let tls_key = config.tls_public_key()?;
753
754    let mut executor = hashi::sui_tx_executor::SuiTxExecutor::new(client, private_key, hashi_ids);
755
756    let mut builder = TransactionBuilder::new();
757
758    let hashi_arg = builder.object(
759        ObjectInput::new(hashi_ids.hashi_object_id)
760            .as_shared()
761            .with_mutable(true),
762    );
763    let validator_address_arg = builder.pure(&validator_address);
764    let tls_key_arg = builder.pure(&tls_key.as_bytes().to_vec());
765
766    let preimage = hashi_types::committee::tls_proof_of_possession_preimage(
767        hashi_ids.hashi_object_id,
768        validator_address,
769        *tls_key.as_bytes(),
770    );
771    let pop = ed25519_dalek::Signer::sign(&config.tls_private_key()?, &preimage);
772    let tls_pop_arg = builder.pure(&pop.to_bytes().to_vec());
773
774    builder.move_call(
775        Function::new(
776            call_package_id,
777            Identifier::from_static("validator"),
778            Identifier::from_static("update_tls_public_key"),
779        ),
780        vec![hashi_arg, validator_address_arg, tls_key_arg, tls_pop_arg],
781    );
782
783    let response = executor.execute(builder).await?;
784    assert!(
785        response.transaction().effects().status().success(),
786        "update_tls_public_key failed"
787    );
788
789    Ok(())
790}
