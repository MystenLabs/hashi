54pub async fn get_hbtc_balance(
55    sui_client: &mut sui_rpc::Client,
56    package_id: Address,
57    owner: Address,
58) -> Result<u64> {
59    let btc_type = format!("{package_id}::btc::BTC");
60    let btc_struct_tag: StructTag = btc_type.parse()?;
61    let request = GetBalanceRequest::default()
62        .with_owner(owner.to_string())
63        .with_coin_type(btc_struct_tag.to_string());
64
65    let response = sui_client
66        .state_client()
67        .get_balance(request)
68        .await?
69        .into_inner();
70
71    let balance = response.balance().balance_opt().unwrap_or(0);
72    debug!("hBTC balance for {owner}: {balance} sats");
73    Ok(balance)
74}

76pub async fn wait_for_deposit_confirmation(
77    sui_client: &mut sui_rpc::Client,
78    request_id: Address,
79    timeout: Duration,
80) -> Result<()> {
81    info!("Waiting for deposit confirmation for request_id: {request_id}");
82
83    let start = std::time::Instant::now();
84    let read_mask = FieldMask::from_paths([Checkpoint::path_builder()
85        .transactions()
86        .events()
87        .events()
88        .contents()
89        .finish()]);
90    let mut subscription = sui_client
91        .subscription_client()
92        .subscribe_checkpoints(SubscribeCheckpointsRequest::default().with_read_mask(read_mask))
93        .await?
94        .into_inner();
95
96    while let Some(item) = subscription.next().await {
97        if start.elapsed() > timeout {
98            return Err(anyhow!(
99                "Timeout waiting for deposit confirmation after {timeout:?}"
100            ));
101        }
102
103        let checkpoint = match item {
104            Ok(cp) => cp,
105            Err(e) => {
106                debug!("Error in checkpoint stream: {e}");
107                continue;
108            }
109        };
110
111        for txn in checkpoint.checkpoint().transactions() {
112            for event in txn.events().events() {
113                if event.contents().name().contains("DepositConfirmed")
114                    && let Ok(evt) = DepositConfirmed::from_bcs(event.contents().value())
115                    && evt.request_id == request_id
116                {
117                    info!("Deposit confirmed for request_id: {request_id}");
118                    return Ok(());
119                }
120            }
121        }
122    }
123
124    Err(anyhow!("Checkpoint subscription ended unexpectedly"))
125}

127pub fn lookup_vout(
128    networks: &TestNetworks,
129    txid: Txid,
130    address: BitcoinAddress,
131    amount: u64,
132) -> Result<usize> {
133    let tx = networks
134        .bitcoin_node
135        .rpc_client()
136        .get_raw_transaction(txid)
137        .and_then(|r| r.transaction().map_err(Into::into))?;
138    let vout = tx
139        .output
140        .iter()
141        .position(|output| {
142            output.value == Amount::from_sat(amount)
143                && output.script_pubkey == address.script_pubkey()
144        })
145        .ok_or_else(|| anyhow!("Could not find output with amount {amount} and deposit address"))?;
146    debug!("Found deposit in tx output {vout}");
147    Ok(vout)
148}

150/// Deposit BTC and wait for the validators to auto-confirm it via the full
151/// observe → sign → confirm path. Returns the hBTC recipient address.
152pub async fn create_deposit_and_wait(
153    networks: &mut TestNetworks,
154    amount_sats: u64,
155) -> Result<Address> {
156    let user_key = networks.sui_network.user_keys.first().unwrap();
157    let hbtc_recipient = user_key.public_key().derive_address();
158    let hashi = networks.hashi_network.nodes()[0].hashi().clone();
159    // `get_deposit_address` internally reads the on-chain MPC key, which is
160    // populated atomically during `end_reconfig` and is guaranteed
161    // available once `HashiNetworkBuilder::build()` returns.
162    let deposit_address = hashi.get_deposit_address(Some(&hbtc_recipient))?;
163
164    info!("Sending Bitcoin to deposit address...");
165    let txid = networks
166        .bitcoin_node
167        .send_to_address(&deposit_address, Amount::from_sat(amount_sats))?;
168    info!("Transaction sent: {txid}");
169
170    info!("Mining blocks for confirmation...");
171    let blocks_to_mine = 10;
172    networks.bitcoin_node.generate_blocks(blocks_to_mine)?;
173    info!("{blocks_to_mine} blocks mined");
174
175    info!("Creating deposit request on Sui...");
176    let vout = lookup_vout(networks, txid, deposit_address, amount_sats)?;
177    let mut executor = SuiTxExecutor::from_config(&hashi.config, hashi.onchain_state())?
178        .with_signer(user_key.clone().into());
179    let request_id = executor
180        .execute_create_deposit_request(
181            txid_to_address(&txid),
182            vout as u32,
183            amount_sats,
184            Some(hbtc_recipient),
185        )
186        .await?;
187    info!("Deposit request created: {request_id}");
188
189    // Mine blocks in the background so the leader's BTC-block-driven
190    // deposit processing loop fires.
191    let _miner = BackgroundMiner::start(&networks.bitcoin_node);
192    wait_for_deposit_confirmation(
193        &mut networks.sui_network.client,
194        request_id,
195        Duration::from_secs(300),
196    )
197    .await?;
198    info!("Deposit confirmed on Sui");
199
200    Ok(hbtc_recipient)
201}

203/// Extract the witness program (the bytes after the segwit version +
204/// push-length prefix) from a P2WPKH (`0x00 0x14 …20`) or P2TR
205/// (`0x51 0x20 …32`) address. This is the destination form the withdrawal
206/// entry function expects.
207pub fn extract_witness_program(address: &BitcoinAddress) -> Result<Vec<u8>> {
208    let script = address.script_pubkey();
209    let bytes = script.as_bytes();
210    match bytes {
211        [0x00, 0x14, rest @ ..] if rest.len() == 20 => Ok(rest.to_vec()),
212        [0x51, 0x20, rest @ ..] if rest.len() == 32 => Ok(rest.to_vec()),
213        _ => Err(anyhow!(
214            "Unsupported script pubkey for withdrawal: {script}"
215        )),
216    }
217}

219/// A live checkpoint subscription to be consumed by
220/// [`WithdrawalConfirmations::wait_for`].
221///
222/// Two-phase on purpose: open the subscription with
223/// [`subscribe_withdrawal_confirmations`] BEFORE kicking off whatever
224/// produces the confirmation (the background miner), so an early event
225/// cannot land in the gap before the stream exists and be missed.
226pub struct WithdrawalConfirmations {
227    subscription: tonic::Streaming<SubscribeCheckpointsResponse>,
228}
229
230/// Open a checkpoint subscription for awaiting `WithdrawalConfirmed` events.
231pub async fn subscribe_withdrawal_confirmations(
232    sui_client: &mut sui_rpc::Client,
233) -> Result<WithdrawalConfirmations> {
234    let subscription_read_mask = FieldMask::from_paths([Checkpoint::path_builder()
235        .transactions()
236        .events()
237        .events()
238        .contents()
239        .finish()]);
240    let subscription = sui_client
241        .subscription_client()
242        .subscribe_checkpoints(
243            SubscribeCheckpointsRequest::default().with_read_mask(subscription_read_mask),
244        )
245        .await?
246        .into_inner();
247    Ok(WithdrawalConfirmations { subscription })
248}
249
250impl WithdrawalConfirmations {
251    /// Return the `WithdrawalConfirmed` event covering `request_id`, or error
252    /// on timeout. Confirmation is the terminal on-chain milestone of a
253    /// withdrawal: it fires only after the committee has generated
254    /// presignatures, threshold-signed the BTC transaction, the guardian
255    /// co-signed, and the leader observed the mined tx — so waiting on it
256    /// exercises the whole signing path. Confirmations for other requests are
257    /// skipped, so concurrent withdrawals cannot satisfy the wait.
258    pub async fn wait_for(
259        mut self,
260        request_id: Address,
261        timeout: Duration,
262    ) -> Result<WithdrawalConfirmed> {
263        info!("Waiting for confirmation of withdrawal request {request_id}...");
264
265        let deadline = tokio::time::Instant::now() + timeout;
266        loop {
267            // Deadline the stream read itself: a stalled subscription must
268            // fail the wait, not hang it.
269            let item = tokio::time::timeout_at(deadline, self.subscription.next())
270                .await
271                .map_err(|_| {
272                    anyhow!("Timeout waiting for withdrawal confirmation after {timeout:?}")
273                })?
274                .ok_or_else(|| anyhow!("Checkpoint subscription ended unexpectedly"))?;
275
276            let checkpoint = match item {
277                Ok(checkpoint) => checkpoint,
278                Err(e) => {
279                    debug!("Error in checkpoint stream: {}", e);
280                    continue;
281                }
282            };
283
284            debug!(
285                "Received checkpoint {}, checking for WithdrawalConfirmed...",
286                checkpoint.cursor()
287            );
288
289            for txn in checkpoint.checkpoint().transactions() {
290                for event in txn.events().events() {
291                    if !event.contents().name().contains("WithdrawalConfirmed") {
292                        continue;
293                    }
294                    match WithdrawalConfirmed::from_bcs(event.contents().value()) {
295                        Ok(event_data) if event_data.request_ids.contains(&request_id) => {
296                            info!(
297                                withdrawal_txn_id = %event_data.withdrawal_txn_id,
298                                txid = %event_data.txid,
299                                "Withdrawal confirmed!"
300                            );
301                            return Ok(event_data);
302                        }
303                        Ok(event_data) => {
304                            debug!(
305                                withdrawal_txn_id = %event_data.withdrawal_txn_id,
306                                "WithdrawalConfirmed for other requests, still waiting"
307                            );
308                        }
309                        Err(e) => {
310                            debug!("Failed to parse WithdrawalConfirmed: {}", e);
311                        }
312                    }
313                }
314            }
315        }
316    }
317}

319/// Request a withdrawal of `amount_sats` to a fresh regtest address and wait
320/// for it to be confirmed on Sui, returning the `WithdrawalConfirmed` event.
321///
322/// Mirrors [`create_deposit_and_wait`] for the withdrawal side: it drives the
323/// full presignature-generation + threshold-signing path, mining Bitcoin blocks
324/// in the background so the signed transaction confirms and the leader reports
325/// `WithdrawalConfirmed`.
326pub async fn create_withdrawal_and_wait(
327    networks: &mut TestNetworks,
328    amount_sats: u64,
329) -> Result<WithdrawalConfirmed> {
330    let hashi = networks.hashi_network.nodes()[0].hashi().clone();
331    let user_key = networks.sui_network.user_keys.first().unwrap();
332    let btc_destination = networks.bitcoin_node.get_new_address()?;
333    let destination_bytes = extract_witness_program(&btc_destination)?;
334    info!("Requesting withdrawal of {amount_sats} sats to {btc_destination}");
335
336    // Subscribe before creating the request or mining, so the confirmation
337    // cannot land before the stream exists.
338    let confirmations =
339        subscribe_withdrawal_confirmations(&mut networks.sui_network.client).await?;
340
341    let mut executor = SuiTxExecutor::from_config(&hashi.config, hashi.onchain_state())?
342        .with_signer(user_key.clone().into());
343    let withdrawal_request_id = executor
344        .execute_create_withdrawal_request(amount_sats, destination_bytes)
345        .await?;
346    info!("Withdrawal request created: {withdrawal_request_id}");
347
348    // Mine in the background so the signed BTC transaction confirms and the
349    // leader's block-driven loop reports the withdrawal on Sui.
350    let _miner = BackgroundMiner::start(&networks.bitcoin_node);
351    let confirmed = confirmations
352        .wait_for(withdrawal_request_id, Duration::from_secs(300))
353        .await?;
354    info!("Withdrawal confirmed on Sui");
355
356    Ok(confirmed)
357}

359/// Wait until every node's object mirror shows the spent withdrawal
360/// inputs cleaned up: at least one `spent_utxos` tombstone exists and no
361/// `utxo_records` entry is still marked spent.
362///
363/// The cleanup transaction (`cleanup_spent_utxos`) emits no event, so
364/// this passing proves both halves of the eventless-write path: the
365/// leader's GC decided on the cleanup from its mirror, and every node's
366/// mirror applied the resulting Field deletions from the object stream
367/// alone — no rescrape exists to paper over a miss.
368pub async fn wait_for_spent_utxo_cleanup(networks: &TestNetworks, timeout: Duration) -> Result<()> {
369    info!("Waiting for the spent-UTXO cleanup to reach every node's mirror...");
370    let deadline = std::time::Instant::now() + timeout;
371    loop {
372        let laggard =
373            networks
374                .hashi_network
375                .nodes()
376                .iter()
377                .enumerate()
378                .find_map(|(index, node)| {
379                    let state = node.hashi().onchain_state();
380                    let pending = state
381                        .utxo_records()
382                        .values()
383                        .filter(|record| record.spent_epoch.is_some())
384                        .count();
385                    let tombstones = state.spent_utxos_entries().len();
386                    (pending > 0 || tombstones == 0).then_some((index, pending, tombstones))
387                });
388        let Some((index, pending, tombstones)) = laggard else {
389            info!("Every node's mirror shows the spent UTXOs cleaned up");
390            return Ok(());
391        };
392        if std::time::Instant::now() >= deadline {
393            return Err(anyhow!(
394                "Timeout after {timeout:?} waiting for the spent-UTXO cleanup: node {index}'s \
395                 mirror still shows {pending} spent record(s) and {tombstones} tombstone(s)"
396            ));
397        }
398        tokio::time::sleep(Duration::from_millis(500)).await;
399    }
400}

402/// Mines one block per second on Bitcoin regtest until stopped.
403/// Stops automatically when dropped.
404pub struct BackgroundMiner {
405    stop_flag: Arc<AtomicBool>,
406    handle: Option<std::thread::JoinHandle<()>>,
407}
408
409impl BackgroundMiner {
410    pub fn start(bitcoin_node: &BitcoinNodeHandle) -> Self {
411        let stop_flag = Arc::new(AtomicBool::new(false));
412        let stop_clone = stop_flag.clone();
413        let rpc_url = bitcoin_node.rpc_url().to_string();
414        let handle = std::thread::spawn(move || {
415            let rpc = corepc_client::client_sync::v29::Client::new_with_auth(
416                &rpc_url,
417                corepc_client::client_sync::Auth::UserPass(
418                    crate::bitcoin_node::RPC_USER.to_string(),
419                    crate::bitcoin_node::RPC_PASSWORD.to_string(),
420                ),
421            )
422            .expect("failed to create mining RPC client");
423            let addr = rpc.new_address().expect("failed to get mining address");
424            while !stop_clone.load(Ordering::Relaxed) {
425                let _ = rpc.generate_to_address(1, &addr);
426                std::thread::sleep(Duration::from_secs(1));
427            }
428        });
429        Self {
430            stop_flag,
431            handle: Some(handle),
432        }
433    }
434}
435
436impl Drop for BackgroundMiner {
437    fn drop(&mut self) {
438        self.stop_flag.store(true, Ordering::Relaxed);
439        if let Some(handle) = self.handle.take() {
440            let _ = handle.join();
441        }
442    }
