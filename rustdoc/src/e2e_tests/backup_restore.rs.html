4//! End-to-end backup/restore round-trip test.
5//!
6//! Verifies that a node can be taken offline, have its config + DB backed
7//! up to an encrypted archive, have the rest of the network rotate several
8//! epochs without it, and then be restored from the archive and successfully
9//! rejoin the network.
10//!
11//! This complements the unit tests in `crates/hashi/src/backup.rs` and
12//! `crates/hashi/src/cli/commands/backup.rs`, which prove the archive format
13//! and on-disk layout correctness. The unique value this test adds is that
14//! the rest of the validator network actually accepts the restored node and
15//! drives it through a key rotation.
16

42    const DKG_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(120);
43    const ROTATION_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(480);

45    fn assert_nodes_agree_on_mpc_key(nodes: &[HashiNodeHandle]) {
46        let pk = nodes[0].hashi().mpc_handle().unwrap().public_key().unwrap();
47        for (i, node) in nodes.iter().enumerate().skip(1) {
48            let node_pk = node.hashi().mpc_handle().unwrap().public_key().unwrap();
49            assert_eq!(pk, node_pk, "Node {i} public key differs from node 0");
50        }
51    }

53    async fn wait_for_rotation(nodes: &[HashiNodeHandle], target_epoch: u64) -> u64 {
54        let futures: Vec<_> = nodes
55            .iter()
56            .map(|node| node.wait_for_epoch(target_epoch, ROTATION_TIMEOUT))
57            .collect();
58        let results: Vec<Result<()>> = futures::future::join_all(futures).await;
59        for (i, result) in results.into_iter().enumerate() {
60            result.unwrap_or_else(|e| panic!("Node {i} failed to reach epoch {target_epoch}: {e}"));
61        }
62        nodes[0].current_epoch().unwrap()
63    }

65    /// Generate a fresh OpenPGP keypair, write its secret key to a
66    /// file, and return `(public_cert, secret_key_file_path)` ready to be
67    /// passed to `backup::save` and `backup::restore` respectively.
68    fn generate_pgp_keypair(dir: &Path) -> (String, PathBuf) {
69        let (public, secret) = mock_pgp_keypair();
70        let secret_key_path = dir.join("backup-secret-key.asc");
71        std::fs::write(&secret_key_path, secret).unwrap();
72        (public, secret_key_path)
73    }

75    /// Materialise an in-memory `HashiConfig` (the node's runtime config) to
76    /// a TOML file on disk. Returns the written path.
77    fn write_node_config_to_disk(config: &HashiConfig, dir: &Path) -> PathBuf {
78        let path = dir.join("node-config.toml");
79        config.save(&path).unwrap();
80        path
81    }

83    fn scheduled_successes(node: &HashiNodeHandle, name: &str) -> f64 {
84        node.metrics_registry()
85            .gather()
86            .iter()
87            .find(|family| family.name() == name)
88            .unwrap_or_else(|| panic!("Missing scheduled backup counter {name}"))
89            .get_metric()[0]
90            .get_counter()
91            .as_ref()
92            .expect("counter")
93            .value()
94    }

96    async fn wait_for_scheduled_backup(node: &HashiNodeHandle) -> Result<()> {
97        tokio::time::timeout(ROTATION_TIMEOUT, async {
98            let mut poll = tokio::time::interval(std::time::Duration::from_millis(100));
99            loop {
100                poll.tick().await;
101                let local =
102                    scheduled_successes(node, "hashi_backup_scheduled_local_successes_total");
103                let remote =
104                    scheduled_successes(node, "hashi_backup_scheduled_remote_successes_total");
105                if local >= 1.0 && remote >= 1.0 {
106                    assert_eq!(local, 1.0, "Expected one scheduled local backup");
107                    assert_eq!(remote, 1.0, "Expected one scheduled S3 upload");
108                    return;
109                }
110            }
111        })
112        .await
113        .context("Scheduled local backup and S3 upload did not both complete")
114    }

126    #[ignore = "requires live S3 bucket, HASHI_BACKUP_S3_BUCKET, HASHI_BACKUP_S3_REGION, AWS credentials"]
127    async fn test_manual_backup_to_s3_and_restore() -> Result<()> {
128        const EPOCH: u64 = 42;
129        let s3 = live_s3_config()?;
130        let node_dir = tempfile::tempdir()?;
131        let recovery_dir = tempfile::tempdir()?;
132        let (recipient, secret_key_path) = generate_pgp_keypair(recovery_dir.path());
133        // Armor headers do not change the configured encryption recipients.
134        let recipient_override = recipient.replacen(
135            "-----BEGIN PGP PUBLIC KEY BLOCK-----",
136            "-----BEGIN PGP PUBLIC KEY BLOCK-----\nComment: Manual S3 backup override",
137            1,
138        );
139        assert_ne!(recipient_override, recipient);
140        let db_path = node_dir.path().join("db");
141        let archive_dir = node_dir.path().join("archives");
142        let mut config = HashiConfig::new_for_testing();
143        config.db = Some(db_path.clone());
144        config.backup_dir = archive_dir.clone();
145        config.backup_pgp_cert = PgpPublicCert::new(recipient)?;
146        config.backup_s3 = Some(s3.clone());
147        config.sui_chain_id = Some("hashi-backup-e2e".into());
148        config.hashi_ids = Some(HashiIds {
149            package_id: Address::new(rand::random()),
150            hashi_object_id: Address::new(rand::random()),
151        });
152        config.validator_address = Some(Address::new(rand::random()));
153        let config_path = write_node_config_to_disk(&config, node_dir.path());
154        let original_config = std::fs::read(&config_path)?;
155        let original_key = EncryptionPrivateKey::new(&mut rand::thread_rng());
156        {
157            let db = Database::open(&db_path)?;
158            db.store_encryption_key(EPOCH, &original_key)?;
159        }
160
161        let archive =
162            commands::backup::save(&config_path, Some(recipient_override), &archive_dir, false)
163                .await?;
164        let uri = format!(
165            "s3://{}/{}{}",
166            s3.bucket,
167            config.backup_s3_namespace().expect("S3 configured"),
168            archive.file_name().unwrap().to_str().unwrap(),
169        );
170        assert!(archive.is_file(), "Manual save must retain a local archive");
171        node_dir.close()?;
172        assert!(!archive.exists());
173        assert!(!config_path.exists());
174        assert!(!db_path.exists());
175
176        let restore_dir = tempfile::tempdir()?;
177        commands::backup::restore_from_s3(
178            &uri,
179            &s3.region,
180            None,
181            RestoreDecryptor::LocalSecretKey { secret_key_path },
182            restore_dir.path(),
183        )
184        .await?;
185        let extracted_dir = restore_dir.path().join(extract_dir_name(&archive)?);
186        assert_eq!(
187            std::fs::read(extracted_dir.join("node-config.toml"))?,
188            original_config,
189        );
190        let restored_db = Database::open(&extracted_dir.join(DB_SNAPSHOT_TAR_PREFIX))?;
191        assert_eq!(
192            restored_db
193                .get_encryption_key(EPOCH)?
194                .expect("Archived epoch key must survive remote recovery"),
195            original_key,
196        );
197        assert!(!config_path.exists());
198        assert!(!db_path.exists());
199        Ok(())
200    }

208    /// Full round-trip test:
209    ///
210    /// 1. DKG on 4 nodes + one key rotation so node 0's DB contains entries
211    ///    across multiple keyspaces.
212    /// 2. Use either a scheduled S3 archive or stop node 0 and manually save
213    ///    an encrypted archive using an externally held OpenPGP keypair.
214    /// 3. Delete node 0's on-disk state entirely (simulating "machine lost,
215    ///    only the backup remains").
216    /// 4. Force two rotations without node 0 — the rest of the network
217    ///    continues operating several epochs ahead.
218    /// 5. Restore into private staging, then explicitly install the config
219    ///    and DB at destinations chosen by the test, not by the manifest.
220    /// 6. Restart node 0 and force one more rotation so it rejoins as a
221    ///    catching-up member.
222    /// 7. Assert all 4 nodes still agree on the original MPC public key.
223    #[tokio::test(flavor = "multi_thread")]
224    async fn test_backup_restore_round_trip_and_rejoin() -> Result<()> {
225        backup_restore_round_trip_and_rejoin(None).await
226    }

228    async fn backup_restore_round_trip_and_rejoin(s3: Option<BackupS3Config>) -> Result<()> {
229        const TEST_NUM_NODES: usize = 4;
230
231        tracing_subscriber::fmt()
232            .with_test_writer()
233            .with_env_filter(
234                tracing_subscriber::EnvFilter::from_default_env()
235                    .add_directive(tracing::Level::INFO.into()),
236            )
237            .try_init()
238            .ok();
239
240        let mut test_networks = TestNetworksBuilder::new()
241            .with_nodes(TEST_NUM_NODES)
242            .build()
243            .await?;
244
245        // 1. DKG on all 4 nodes.
246        {
247            let nodes = test_networks.hashi_network().nodes();
248            let futs: Vec<_> = nodes
249                .iter()
250                .map(|n| n.wait_for_mpc_key(DKG_TIMEOUT))
251                .collect();
252            let results: Vec<Result<()>> = futures::future::join_all(futs).await;
253            for (i, r) in results.into_iter().enumerate() {
254                r.unwrap_or_else(|e| panic!("Node {i} DKG failed: {e}"));
255            }
256            assert_nodes_agree_on_mpc_key(nodes);
257        }
258        let original_mpc_key = test_networks.hashi_network().nodes()[0]
259            .hashi()
260            .mpc_handle()
261            .unwrap()
262            .public_key()
263            .unwrap();
264        // The recovery key remains outside the failed node's state.
265        let config_dir = tempfile::Builder::new()
266            .prefix("hashi-backup-e2e-")
267            .tempdir()?;
268        let (recipient, secret_key_path) = generate_pgp_keypair(config_dir.path());
269        let node_config_path = config_dir.path().join("node-config.toml");
270        if let Some(s3) = &s3 {
271            let node = &mut test_networks.hashi_network_mut().nodes_mut()[0];
272            node.shutdown().await;
273            node.config_mut().backup_s3 = Some(s3.clone());
274            node.config_mut().backup_pgp_cert = PgpPublicCert::new(recipient.clone())?;
275            node.config().save(&node_config_path)?;
276            node.config_path = Some(node_config_path.clone());
277            node.start().await?;
278            node.wait_for_mpc_key(DKG_TIMEOUT).await?;
279            for name in [
280                "hashi_backup_scheduled_local_successes_total",
281                "hashi_backup_scheduled_remote_successes_total",
282            ] {
283                assert_eq!(scheduled_successes(node, name), 0.0);
284            }
285        }
286        let initial_epoch = test_networks.hashi_network().nodes()[0]
287            .current_epoch()
288            .unwrap();
289
290        // One pre-backup rotation so the DB has rotation_messages rows, not
291        // just the initial DKG state.
292        test_networks.sui_network.force_close_epoch().await?;
293        wait_for_rotation(test_networks.hashi_network().nodes(), initial_epoch + 1).await;
294        assert_nodes_agree_on_mpc_key(test_networks.hashi_network().nodes());
295        if s3.is_some() {
296            // Observe the real epoch-change worker, not a direct backup call.
297            wait_for_scheduled_backup(&test_networks.hashi_network().nodes()[0]).await?;
298        }
299
300        // The on-chain epoch can advance before node 0 prepares its next
301        // keys. Wait for both records before stopping it for a manual save.
302        // The scheduled archive already has this ordering: next-epoch key
303        // preparation precedes the backup request.
304        tokio::time::timeout(ROTATION_TIMEOUT, async {
305            let node = &test_networks.hashi_network().nodes()[0];
306            let mut poll = tokio::time::interval(std::time::Duration::from_millis(100));
307            loop {
308                poll.tick().await;
309                if node
310                    .hashi()
311                    .db
312                    .get_encryption_key(initial_epoch + 2)?
313                    .is_some()
314                    && node
315                        .hashi()
316                        .db
317                        .get_signing_key(initial_epoch + 2)?
318                        .is_some()
319                {
320                    return Ok::<(), anyhow::Error>(());
321                }
322            }
323        })
324        .await
325        .context("Node 0 did not prepare its next-epoch private keys before backup")??;
326
327        // 2. Stop node 0.
328        test_networks.hashi_network_mut().nodes_mut()[0]
329            .shutdown()
330            .await;
331
332        // Reopening also waits for shutdown to release the DB lock. Capture
333        // only the current and next epoch's private keys: both must precede
334        // the selected archive, unlike arbitrary later DB writes.
335        let archived_key_epochs = [initial_epoch + 1, initial_epoch + 2];
336        let expected_keys = {
337            let db = test_networks.hashi_network().nodes()[0].open_db()?;
338            let mut keys = Vec::with_capacity(archived_key_epochs.len());
339            for epoch in archived_key_epochs {
340                let encryption_key = db
341                    .get_encryption_key(epoch)?
342                    .with_context(|| format!("Source DB lacks encryption key for epoch {epoch}"))?;
343                let signing_key = db
344                    .get_signing_key(epoch)?
345                    .with_context(|| format!("Source DB lacks signing key for epoch {epoch}"))?;
346                keys.push((epoch, encryption_key, bcs::to_bytes(&signing_key)?));
347            }
348            keys
349        };
350
351        // Snapshot the config now, before any deletion below touches the
352        // filesystem layout.
353        let node0_config = test_networks.hashi_network().nodes()[0].config().clone();
354        let original_db_path = node0_config
355            .db
356            .as_ref()
357            .expect("node 0 must have a db path")
358            .clone();
359
360        let (tarball, remote_uri) = if let Some(s3) = &s3 {
361            let archives = std::fs::read_dir(&node0_config.backup_dir)?
362                .map(|entry| entry.map(|entry| entry.path()))
363                .collect::<std::io::Result<Vec<_>>>()?;
364            assert_eq!(archives.len(), 1, "Expected the scheduled archive only");
365            let tarball = archives.into_iter().next().unwrap();
366            assert!(tarball.to_string_lossy().ends_with(".tar.asc"));
367            let uri = format!(
368                "s3://{}/{}{}",
369                s3.bucket,
370                node0_config.backup_s3_namespace().expect("S3 configured"),
371                tarball.file_name().unwrap().to_str().unwrap(),
372            );
373            (tarball, Some(uri))
374        } else {
375            write_node_config_to_disk(&node0_config, config_dir.path());
376            // Keep only the manual recovery copy outside the failed machine.
377            let save_out_dir = config_dir.path().join("manual");
378            let tarball =
379                commands::backup::save(&node_config_path, Some(recipient), &save_out_dir, true)
380                    .await?;
381            (tarball, None)
382        };
383
384        // 4. Destroy node 0's on-disk state so recovery actually has to put
385        //    things back. The config has its own tempdir; both the DB and
386        //    configured backups live under the TestNetworks tempdir, which
387        //    stays alive because the handle still owns it.
388        std::fs::remove_dir_all(&original_db_path)?;
389        std::fs::remove_file(&node_config_path)?;
390        if node0_config.backup_dir.exists() {
391            std::fs::remove_dir_all(&node0_config.backup_dir)?;
392        }
393        if remote_uri.is_some() {
394            assert!(
395                !tarball.exists(),
396                "Remote recovery must have no local archive"
397            );
398        }
399
400        // 5. Two more rotations without node 0. The surviving nodes advance
401        //    the Hashi epoch; node 0's backed-up DB is now several epochs
402        //    stale relative to the live network state.
403        for target in 2..=3 {
404            test_networks.sui_network.force_close_epoch().await?;
405            wait_for_rotation(
406                &test_networks.hashi_network().nodes()[1..],
407                initial_epoch + target,
408            )
409            .await;
410        }
411
412        // 6. Restore only into private staging. Keep staging on the DB's
413        //    filesystem so the explicit DB installation below can use rename.
414        let restore_out_dir = tempfile::Builder::new()
415            .prefix("hashi-restore-out-")
416            .tempdir_in(original_db_path.parent().expect("DB must have a parent"))?;
417        let decryptor = RestoreDecryptor::LocalSecretKey { secret_key_path };
418        if let Some(uri) = remote_uri {
419            commands::backup::restore_from_s3(
420                &uri,
421                &s3.as_ref().unwrap().region,
422                None,
423                decryptor,
424                restore_out_dir.path(),
425            )
426            .await?;
427        } else {
428            commands::backup::restore(&tarball, decryptor, restore_out_dir.path())?;
429        }
430
431        // Extraction must not recreate the original locations. Installation
432        // destinations come from the test's pre-backup config, never from
433        // paths supplied by the archive's manifest.
434        assert!(!original_db_path.exists());
435        assert!(!node_config_path.exists());
436        let extracted_dir = restore_out_dir.path().join(extract_dir_name(&tarball)?);
437        std::fs::copy(extracted_dir.join("node-config.toml"), &node_config_path)?;
438        std::fs::rename(
439            extracted_dir.join(DB_SNAPSHOT_TAR_PREFIX),
440            &original_db_path,
441        )?;
442
443        // Sanity: explicit installation restored the artefacts needed by
444        // the node before restarting it.
445        assert!(
446            original_db_path.is_dir(),
447            "installation did not recreate db at {}",
448            original_db_path.display()
449        );
450        assert!(
451            node_config_path.is_file(),
452            "installation did not recreate node config at {}",
453            node_config_path.display()
454        );
455
456        // Check actual archived private keys before startup can generate
457        // replacements or recover fresh shares from the other validators.
458        {
459            let db = Database::open(&original_db_path)?;
460            for (epoch, expected_encryption_key, expected_signing_key) in &expected_keys {
461                let encryption_key = db.get_encryption_key(*epoch)?.with_context(|| {
462                    format!("Restored DB lacks encryption key for epoch {epoch}")
463                })?;
464                let signing_key = db
465                    .get_signing_key(*epoch)?
466                    .with_context(|| format!("Restored DB lacks signing key for epoch {epoch}"))?;
467                assert_eq!(
468                    &encryption_key, expected_encryption_key,
469                    "Recovery changed the encryption private key for epoch {epoch}",
470                );
471                assert!(
472                    bcs::to_bytes(&signing_key)? == *expected_signing_key,
473                    "Recovery changed the signing private key for epoch {epoch}",
474                );
475            }
476        } // Drop the DB handle before restarting node 0.
477
478        // 7. Restart node 0. It may not have valid shares for the current
479        //    epoch yet — that's fine, we just need the server up so the
480        //    upcoming rotation can deliver fresh shares.
481        *test_networks.hashi_network_mut().nodes_mut()[0].config_mut() =
482            HashiConfig::load(&node_config_path)?;
483        test_networks.hashi_network_mut().nodes_mut()[0]
484            .start()
485            .await?;
486        test_networks.hashi_network().nodes()[0]
487            .wait_for_mpc_key(ROTATION_TIMEOUT)
488            .await
489            .ok();
490        assert_eq!(
491            test_networks.hashi_network().nodes()[0]
492                .hashi()
493                .metrics
494                .mpc_committee_key_lost_total
495                .get(),
496            0,
497            "Restored node must not replace lost committee keys before rejoining",
498        );
499
500        // 8. Force one more rotation; node 0 rejoins as a catching-up
501        //    member and must end up with the same MPC pubkey as the rest.
502        test_networks.sui_network.force_close_epoch().await?;
503        let nodes = test_networks.hashi_network().nodes();
504        let futs: Vec<_> = nodes
505            .iter()
506            .map(|n| n.wait_for_epoch(initial_epoch + 4, ROTATION_TIMEOUT))
507            .collect();
508        let results: Vec<Result<()>> = futures::future::join_all(futs).await;
509        for (i, r) in results.into_iter().enumerate() {
510            r.unwrap_or_else(|e| {
511                panic!("Node {i} failed to reach epoch {}: {e}", initial_epoch + 4)
512            });
513        }
514
515        assert_nodes_agree_on_mpc_key(test_networks.hashi_network().nodes());
516        assert_eq!(
517            test_networks.hashi_network().nodes()[0]
518                .hashi()
519                .mpc_handle()
520                .unwrap()
521                .public_key()
522                .unwrap(),
523            original_mpc_key,
524            "Recovery must preserve the original MPC public key",
525        );
526        assert_eq!(
527            test_networks.hashi_network().nodes()[0]
528                .hashi()
529                .metrics
530                .mpc_committee_key_lost_total
531                .get(),
532            0,
533            "Restored node must rejoin without replacing lost committee keys",
534        );
535        Ok(())
536    }
