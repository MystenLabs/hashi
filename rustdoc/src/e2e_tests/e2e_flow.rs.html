187        };

191            .hashi_network
192            .nodes()
193            .iter()
194            .map(|n| n.wait_for_epoch(target_epoch, Duration::from_secs(480)))
195            .collect();
196        for (i, r) in futures::future::join_all(futs)

198            .into_iter()
199            .enumerate()
200        {
201            r.unwrap_or_else(|e| panic!("Node {i} failed to reach epoch {target_epoch}: {e}"));
202        }

349        networks: &TestNetworks,
350        request_id: Address,
351        confirmations: u32,
352        required_confirmations: u32,
353    ) -> Result<()> {
354        let expected = format!("has {confirmations}/{required_confirmations} confirmations");
355        for (index, node) in networks.hashi_network.nodes().iter().enumerate() {
356            let node_hashi = node.hashi();
357            let deposit_request = node_hashi
358                .onchain_state()
359                .deposit_requests()
360                .into_iter()
361                .find(|request| request.id == request_id)
362                .ok_or_else(|| anyhow!("node {index} has no deposit request {request_id}"))?;
363            match node_hashi.validate_deposit_request(&deposit_request).await {
364                Err(UnapprovedDepositError::BitcoinNotConfirmed(error)) => assert!(
365                    error.to_string().contains(&expected),
366                    "node {index} reported an unexpected confirmation error: {error}"
367                ),
368                Err(error) => panic!(
369                    "node {index} returned an unexpected validation error at {confirmations} confirmations: {error}"
370                ),
371                Ok(()) => panic!(
372                    "node {index} accepted the deposit at {confirmations}/{required_confirmations} confirmations"
373                ),
374            }
375        }
376        Ok(())
377    }

392    /// Wait for a withdrawal transaction to be confirmed on the Bitcoin chain.
393    /// The output to `destination` must be at most `max_amount` and at least
394    /// `min_amount` (to account for variable miner fees deducted from the user).
395    async fn wait_for_withdrawal_tx_success(
396        bitcoin_node: &crate::BitcoinNodeHandle,
397        txid: &Txid,
398        destination: &BitcoinAddress,
399        max_amount: Amount,
400        min_amount: Amount,
401        timeout: Duration,
402    ) -> Result<()> {
403        let start = std::time::Instant::now();
404
405        let check_output = |tx: &bitcoin::Transaction| -> bool {
406            tx.output.iter().any(|output| {
407                output.value <= max_amount
408                    && output.value >= min_amount
409                    && output.script_pubkey == destination.script_pubkey()
410            })
411        };
412
413        // Wait until the tx is visible (either in mempool or already confirmed).
414        loop {
415            if bitcoin_node.rpc_client().get_mempool_entry(*txid).is_ok() {
416                info!("Withdrawal tx {} is in mempool", txid);
417                break;
418            }
419            // The background miner may have already confirmed it.
420            if let Ok(info) = bitcoin_node.rpc_client().get_raw_transaction_verbose(*txid)
421                && info.confirmations.unwrap_or(0) > 0
422            {
423                info!("Withdrawal tx {} is already confirmed", txid);
424                let tx = bitcoin_node
425                    .rpc_client()
426                    .get_raw_transaction(*txid)
427                    .and_then(|r| r.transaction().map_err(Into::into))?;
428                if !check_output(&tx) {
429                    return Err(anyhow!(
430                        "Withdrawal tx {} is confirmed but does not pay [{}, {}] to {}",
431                        txid,
432                        min_amount,
433                        max_amount,
434                        destination
435                    ));
436                }
437                info!("Withdrawal tx {} confirmed with expected output", txid);
438                return Ok(());
439            }
440            if start.elapsed() >= timeout {
441                return Err(anyhow!(
442                    "Withdrawal tx {} was not seen in mempool within {:?}",
443                    txid,
444                    timeout
445                ));
446            }
447            tokio::time::sleep(Duration::from_millis(200)).await;
448        }
449
450        loop {
451            let mined_blocks = bitcoin_node.generate_blocks(1)?;
452            let block_hash = mined_blocks
453                .last()
454                .copied()
455                .ok_or_else(|| anyhow!("Expected at least one mined block"))?;
456            let block = bitcoin_node.rpc_client().get_block(block_hash)?;
457
458            if !block.txdata.iter().any(|tx| tx.compute_txid() == *txid) {
459                if start.elapsed() >= timeout {
460                    return Err(anyhow!(
461                        "Withdrawal tx {} did not confirm within {:?}",
462                        txid,
463                        timeout
464                    ));
465                }
466                tokio::time::sleep(Duration::from_millis(200)).await;
467                continue;
468            }
469
470            let tx = bitcoin_node
471                .rpc_client()
472                .get_raw_transaction(*txid)
473                .and_then(|r| r.transaction().map_err(Into::into))?;
474            if !check_output(&tx) {
475                return Err(anyhow!(
476                    "Withdrawal tx {} is confirmed but does not pay [{}, {}] to {}",
477                    txid,
478                    min_amount,
479                    max_amount,
480                    destination
481                ));
482            }
483
484            info!(
485                "Withdrawal tx {} confirmed in block {} with expected output",
486                txid, block_hash
487            );
488            return Ok(());
489        }
490    }

515    async fn test_passive_bitcoin_deposit_discovery() -> Result<()> {
516        init_test_logging();
517        info!("=== Starting Passive Bitcoin Deposit Discovery Test ===");
518
519        let networks = setup_test_networks(
520            TestNetworksBuilder::new()
521                .with_nodes(4)
522                .with_onchain_config(
523                    "bitcoin_deposit_time_delay_ms",
524                    hashi_types::move_types::ConfigValue::U64(60_000),
525                ),
526        )
527        .await?;
528        let user_key = networks.sui_network.user_keys.first().unwrap().clone();
529        let hbtc_recipient = user_key.public_key().derive_address();
530        let hashi = networks.hashi_network.nodes()[0].hashi().clone();
531        let deposit_address = hashi.get_deposit_address(Some(&hbtc_recipient))?;
532        let amount_sats = 31_337u64;
533        let amount = Amount::from_sat(amount_sats);
534        let rpc = networks.bitcoin_node.rpc_client();
535
536        let miner_fee = Amount::from_sat(1_000);
537        let unspent = rpc
538            .list_unspent()?
539            .into_model()?
540            .0
541            .into_iter()
542            .find(|output| {
543                output.spendable
544                    && output
545                        .amount
546                        .to_unsigned()
547                        .is_ok_and(|value| value > amount + miner_fee)
548            })
549            .ok_or_else(|| anyhow!("wallet has no spendable output for deposit transaction"))?;
550        let input_amount = unspent.amount.to_unsigned()?;
551        let change_address = networks.bitcoin_node.get_new_address()?;
552        let raw_tx = bitcoin::Transaction {
553            version: bitcoin::transaction::Version::TWO,
554            lock_time: bitcoin::absolute::LockTime::ZERO,
555            input: vec![bitcoin::TxIn {
556                previous_output: bitcoin::OutPoint::new(unspent.txid, unspent.vout),
557                script_sig: bitcoin::ScriptBuf::new(),
558                sequence: bitcoin::Sequence::MAX,
559                witness: bitcoin::Witness::default(),
560            }],
561            output: vec![
562                bitcoin::TxOut {
563                    value: amount,
564                    script_pubkey: deposit_address.script_pubkey(),
565                },
566                bitcoin::TxOut {
567                    value: input_amount - amount - miner_fee,
568                    script_pubkey: change_address.script_pubkey(),
569                },
570            ],
571        };
572        let signed = rpc
573            .sign_raw_transaction_with_wallet(&raw_tx)?
574            .into_model()?;
575        if !signed.complete {
576            return Err(anyhow!(
577                "wallet did not completely sign deposit transaction: {:?}",
578                signed.errors
579            ));
580        }
581        let deposit_tx = signed.tx;
582        let txid = deposit_tx.compute_txid();
583
584        wait_for_kyoto_sync(&networks, Duration::from_secs(30)).await?;
585        let baseline_not_found = outpoints_with_status(&networks, "not_found");
586        let baseline_one_confirmation = outpoints_with_status(&networks, "1");
587        let mut executor = SuiTxExecutor::from_config(&hashi.config, hashi.onchain_state())?
588            .with_signer(user_key.into());
589        let request_id = executor
590            .execute_create_deposit_request(
591                txid_to_address(&txid),
592                0,
593                amount_sats,
594                Some(hbtc_recipient),
595            )
596            .await?;
597
598        // Every node must track the outpoint before the deposit block is
599        // mined; a node that misses the block's passive scan would stay
600        // unchecked, as followers only run discovery when asked to sign.
601        wait_for_deposit_request_mirrored(&networks, request_id, Duration::from_secs(30)).await?;
602        wait_for_outpoint_status(
603            &networks,
604            "not_found",
605            &baseline_not_found,
606            1,
607            Duration::from_secs(30),
608        )
609        .await?;
610        assert_deposit_request_unapproved(&networks, request_id);
611
612        let broadcast_txid = rpc.send_raw_transaction(&deposit_tx)?.txid()?;
613        assert_eq!(
614            broadcast_txid, txid,
615            "Bitcoin Core broadcast a different txid"
616        );
617        networks.bitcoin_node.generate_blocks(1)?;
618        wait_for_outpoint_status(
619            &networks,
620            "1",
621            &baseline_one_confirmation,
622            networks.hashi_network.nodes().len(),
623            Duration::from_secs(30),
624        )
625        .await?;
626        assert_deposit_request_unapproved(&networks, request_id);
627
628        networks.bitcoin_node.generate_blocks(9)?;
629        wait_for_deposit_approval(&networks, request_id, Duration::from_secs(120)).await?;
630
631        info!("=== Passive Bitcoin Deposit Discovery Test Passed ===");
632        Ok(())
633    }

1298                    events.len(),
1299                    timeout
1300                ));
1301            }

1325                                debug!("Failed to parse WithdrawalConfirmed: {}", e);

1327                        }

1942    async fn wait_for_batched_withdrawal_picked(
1943        sui_client: &mut sui_rpc::Client,
1944        min_requests: usize,
1945        timeout: Duration,
1946    ) -> Result<WithdrawalPickedForProcessing> {
1947        let start = std::time::Instant::now();
1948        let subscription_read_mask = FieldMask::from_paths([Checkpoint::path_builder()
1949            .transactions()
1950            .events()
1951            .events()
1952            .contents()
1953            .finish()]);
1954        let mut subscription = sui_client
1955            .subscription_client()
1956            .subscribe_checkpoints(
1957                SubscribeCheckpointsRequest::default().with_read_mask(subscription_read_mask),
1958            )
1959            .await?
1960            .into_inner();
1961
1962        while let Some(item) = subscription.next().await {
1963            if start.elapsed() > timeout {
1964                return Err(anyhow!(
1965                    "Timeout waiting for batched WithdrawalPickedForProcessing \
1966                     (min_requests={min_requests}) after {:?}",
1967                    timeout
1968                ));
1969            }
1970            let checkpoint = match item {
1971                Ok(checkpoint) => checkpoint,
1972                Err(e) => {
1973                    debug!("Error in checkpoint stream: {}", e);
1974                    continue;
1975                }
1976            };
1977            for txn in checkpoint.checkpoint().transactions() {
1978                for event in txn.events().events() {
1979                    if event
1980                        .contents()
1981                        .name()
1982                        .contains("WithdrawalPickedForProcessing")
1983                    {
1984                        match WithdrawalPickedForProcessing::from_bcs(event.contents().value()) {
1985                            Ok(data) if data.request_ids.len() >= min_requests => {
1986                                info!(
1987                                    withdrawal_txn_id = %data.withdrawal_txn_id,
1988                                    request_count = %data.request_ids.len(),
1989                                    "Batched withdrawal picked"
1990                                );
1991                                return Ok(data);
1992                            }
1993                            Ok(data) => {
1994                                info!(
1995                                    "WithdrawalPickedForProcessing with {} request(s) \
1996                                     (waiting for batch of ≥{})",
1997                                    data.request_ids.len(),
1998                                    min_requests,
1999                                );
2000                            }
2001                            Err(e) => {
2002                                debug!("Failed to parse WithdrawalPickedForProcessing: {}", e);
2003                            }
2004                        }
2005                    }
2006                }
2007            }
2008            tokio::time::sleep(Duration::from_millis(100)).await;
2009        }
2010        Err(anyhow!("Checkpoint subscription ended unexpectedly"))
2011    }

2672    async fn test_leader_destroys_expired_tob_cert_buckets() -> Result<()> {

2675
2676        let mut networks = TestNetworksBuilder::new().with_nodes(4).build().await?;
2677        for node in networks.hashi_network.nodes() {
2678            node.wait_for_mpc_key(Duration::from_secs(120)).await?;
2679        }
2680
2681        let state = networks.hashi_network.nodes()[0]
2682            .hashi()
2683            .onchain_state()
2684            .clone();
2685        let genesis_epoch = state.epoch();
2686        assert!(
2687            state.tob_bucket_keys().iter().any(
2688                |(key, _)| key.epoch == genesis_epoch && key.protocol_type == ProtocolType::Dkg
2689            ),
2690            "expected a genesis DKG bucket after DKG"
2691        );
2692
2693        // The presig refill writes the nonce buckets the GC will prune;
2694        // wait for the first ones rather than racing the refill loop.
2695        info!("Waiting for genesis-epoch nonce buckets from the presig refill...");
2696        let deadline = tokio::time::Instant::now() + Duration::from_secs(120);
2697        let genesis_nonce_batches: Vec<u32> = loop {
2698            let batches: Vec<u32> = state
2699                .tob_bucket_keys()
2700                .into_iter()
2701                .filter_map(|(key, _)| {
2702                    (key.epoch == genesis_epoch
2703                        && key.protocol_type == ProtocolType::NonceGeneration)
2704                        .then_some(key.batch_index)
2705                        .flatten()
2706                })
2707                .collect();
2708            if !batches.is_empty() {
2709                break batches;
2710            }
2711            if tokio::time::Instant::now() >= deadline {
2712                anyhow::bail!("no genesis-epoch nonce bucket appeared in the mirror");
2713            }
2714            tokio::time::sleep(Duration::from_millis(500)).await;
2715        };
2716        info!("Genesis nonce batches present: {genesis_nonce_batches:?}");
2717
2718        // Two epoch advances put the genesis nonce buckets past the
2719        // Move floor (`current_epoch >= epoch + 2`); the key-generation
2720        // retention floor is longer, so the DKG bucket must survive.
2721        for step in 1..=2u64 {
2722            info!("Forcing Sui epoch close {step}/2...");
2723            networks.sui_network.force_close_epoch().await?;
2724            let target = genesis_epoch + step;
2725            for node in networks.hashi_network.nodes() {
2726                node.wait_for_epoch(target, Duration::from_secs(180))
2727                    .await?;
2728            }
2729            info!("All nodes rotated into epoch {target}");
2730        }
2731
2732        // The leader GC runs on its checkpoint tick; wait until every
2733        // node's mirror shows no expired nonce bucket left.
2734        info!("Waiting for the leader to prune the expired nonce buckets...");
2735        let deadline = tokio::time::Instant::now() + Duration::from_secs(120);
2736        'wait: loop {
2737            let current = state.epoch();
2738            let laggard =
2739                networks
2740                    .hashi_network
2741                    .nodes()
2742                    .iter()
2743                    .enumerate()
2744                    .find_map(|(index, node)| {
2745                        let expired: Vec<_> = node
2746                            .hashi()
2747                            .onchain_state()
2748                            .tob_bucket_keys()
2749                            .into_iter()
2750                            .filter(|(key, _)| {
2751                                key.protocol_type == ProtocolType::NonceGeneration
2752                                    && current >= key.epoch.saturating_add(2)
2753                            })
2754                            .collect();
2755                        (!expired.is_empty()).then_some((index, expired))
2756                    });
2757            let Some((index, expired)) = laggard else {
2758                break 'wait;
2759            };
2760            if tokio::time::Instant::now() >= deadline {
2761                anyhow::bail!(
2762                    "node {index}'s mirror still holds expired nonce buckets: {expired:?}"
2763                );
2764            }
2765            tokio::time::sleep(Duration::from_millis(500)).await;
2766        }
2767        info!("Expired nonce buckets are gone from every node's mirror");
2768
2769        // Chain truth: the genesis nonce buckets must be gone on-chain
2770        // too, not just from the mirrors — and the genesis DKG bucket
2771        // must still exist, held by the key-generation retention floor.
2772        for batch_index in genesis_nonce_batches {
2773            let onchain = fetch_tob_certs_from_chain(
2774                &networks,
2775                hashi_types::move_types::TobKey {
2776                    epoch: genesis_epoch,
2777                    batch_index: Some(batch_index),
2778                    protocol_type: ProtocolType::NonceGeneration,
2779                },
2780            )
2781            .await?;
2782            assert!(
2783                onchain.is_none(),
2784                "genesis nonce bucket (batch {batch_index}) still exists on-chain"
2785            );
2786        }
2787        let dkg_bucket = fetch_tob_certs_from_chain(
2788            &networks,
2789            hashi_types::move_types::TobKey {
2790                epoch: genesis_epoch,
2791                batch_index: None,
2792                protocol_type: ProtocolType::Dkg,
2793            },
2794        )
2795        .await?;
2796        assert!(
2797            dkg_bucket.is_some(),
2798            "the genesis DKG bucket was pruned below its retention floor"
2799        );
2800
2801        // The destroy deletions routed cleanly, and the surviving
2802        // buckets still match the chain.
2803        assert_no_unrouted_objects(&networks);
2804        assert_tob_mirror_parity(&networks).await?;
2805
2806        info!("=== TOB Cert GC E2E Test Passed ===");
2807        Ok(())
