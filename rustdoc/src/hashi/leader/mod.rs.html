55/// Throttle for the "binary unsupported for the on-chain version" leader
56/// warning. `node_is_leader` runs every checkpoint (and from re-checking
57/// tasks), so an unthrottled `warn!` would flood logs for the whole time the
58/// chain sits ahead of this binary; the `hashi_package_version_unsupported`
59/// metric is the durable signal, so we log on first detection and then only
60/// periodically. Stores the last checkpoint height it warned at (0 = never).
61static LAST_UNSUPPORTED_WARN_CHECKPOINT: std::sync::atomic::AtomicU64 =
62    std::sync::atomic::AtomicU64::new(0);
63const UNSUPPORTED_WARN_EVERY_CHECKPOINTS: u64 = 240;

238    /// [`retry_peer_call`] against the peer's bridge-service client, logging
239    /// the final error. Returns `None` once attempts are exhausted or on a
240    /// non-transport error.
241    async fn call_peer_with_retry<Resp, F, Fut>(
242        inner: &Arc<Hashi>,
243        validator: Address,
244        what: &str,
245        call: F,
246    ) -> Option<tonic::Response<Resp>>
247    where
248        F: FnMut(BridgeServiceClient<BoxedChannel>) -> Fut,
249        Fut: Future<Output = Result<tonic::Response<Resp>, tonic::Status>>,
250    {
251        retry_peer_call(
252            validator,
253            what,
254            || inner.onchain_state().bridge_service_client(&validator),
255            call,
256        )
257        .await
258        .inspect_err(|status| error!("Failed to get {what} from {validator}: {status}"))
259        .ok()
260    }

270    #[tracing::instrument(name = "leader", skip_all)]
271    async fn run(mut self) {
272        info!("Starting leader service");
273
274        // Wait for DKG to complete before processing any checkpoints.
275        let mpc_handle = self.inner.mpc_handle().expect("MpcHandle not initialized");
276        info!("Waiting for MPC key to become available...");
277        mpc_handle.wait_for_key_ready().await;
278        info!("MPC key is ready, starting leader loop");
279
280        let mut checkpoint_rx = self.inner.onchain_state().subscribe_checkpoint();
281        let mut btc_block_rx = self.inner.btc_monitor().subscribe_block_height();
282        let mut deposit_work_rx = self
283            .inner
284            .onchain_state()
285            .deposit_tracker()
286            .subscribe_work();
287
288        loop {
289            trace!("Waiting for next checkpoint or task completion...");
290            tokio::select! {
291                wait_result = checkpoint_rx.changed() => {
292                    if let Err(e) = wait_result {
293                        error!("Error waiting for checkpoint change: {e}");
294                        break;
295                    }
296                    let (checkpoint_height, checkpoint_timestamp_ms) = {
297                        let checkpoint_info = checkpoint_rx.borrow_and_update();
298                        (checkpoint_info.height, checkpoint_info.timestamp_ms)
299                    };
300
301                    // Every node reports its withdrawal semantics selection,
302                    // not just the current leader.
303                    // Heartbeat before checking leadership so followers report liveness too.
304                    self.inner.metrics.task_heartbeat("leader_loop");
305                    let was_leader = self.is_leader;
306                    if self.update_leadership(checkpoint_height) {
307                        debug!("Checkpoint {checkpoint_height}: We are the leader node");
308                        self.process_stale_unapproved_deposits_if_new_epoch();
309                        // Also replay deposit work dropped while processing
310                        // was stopped (e.g. a reconfig halt consumed a tracker
311                        // notification) and pick up confirmation-threshold
312                        // changes; neither produces another tracker event.
313                        if !was_leader || self.needs_actionable_deposit_reload() {
314                            self.process_actionable_unapproved_deposits();
315                        }
316                        self.process_approved_deposit_requests();
317                        self.check_reconcile_guardian_committee();
318                        self.process_unapproved_withdrawal_requests(checkpoint_timestamp_ms);
319                        self.process_approved_withdrawal_requests(checkpoint_timestamp_ms);
320                        self.process_unsigned_withdrawal_txns();
321                        self.process_signed_withdrawal_txns();
322                        self.check_delete_expired_deposit_requests(checkpoint_timestamp_ms);
323                        self.check_delete_proposals(checkpoint_timestamp_ms);
324                        self.check_cleanup_spent_utxos(checkpoint_timestamp_ms);
325                        self.check_archive_confirmed_withdrawals(checkpoint_timestamp_ms);
326                        self.check_prune_tob_certs(checkpoint_timestamp_ms);
327                    } else {
328                        trace!("We are not the leader node");
329                        // Deposit tasks outlive the leader's turn, but not a halt.
330                        self.check_halt_deposit_processing();
331                    }
332                }
333                wait_result = deposit_work_rx.changed() => {
334                    if let Err(e) = wait_result {
335                        error!("Error waiting for deposit tracker work change: {e}");
336                        break;
337                    }
338                    if self.is_leader {
339                        debug!("Deposit tracker changed: processing deposit requests");
340                        self.process_actionable_unapproved_deposits();
341                    }
342                }
343                wait_result = btc_block_rx.changed() => {
344                    if let Err(e) = wait_result {
345                        error!("Error waiting for Bitcoin block height change: {e}");
346                        break;
347                    }
348
349                    let block_sequence = self.inner.btc_monitor().block_sequence();
350                    self.activate_unapproved_deposits_for_btc_block(block_sequence);
351                    self.schedule_withdrawal_checks_for_btc_block();
352                }
353                Some(result) = self.unapproved_deposit_tasks.join_next() => {
354                    self.handle_completed_unapproved_deposit_task(result);
355                    while let Some(result) = self.unapproved_deposit_tasks.try_join_next() {
356                        self.handle_completed_unapproved_deposit_task(result);
357                    }
358                }
359                Some(result) = self.approved_deposit_tasks.join_next() => {
360                    self.handle_completed_approved_deposit_task(result);
361                    while let Some(result) = self.approved_deposit_tasks.try_join_next() {
362                        self.handle_completed_approved_deposit_task(result);
363                    }
364                }
365                Some(result) = self.withdrawal_signing_tasks.join_next() => {
366                    self.handle_completed_withdrawal_signing_task(result);
367                }
368                Some(result) = self.withdrawal_broadcast_tasks.join_next() => {
369                    self.handle_completed_withdrawal_broadcast_task(result);
370                }
371                Some(result) = self.withdrawal_btc_block_check_tasks.join_next() => {
372                    self.handle_completed_withdrawal_btc_block_check_task(result);
373                }
374                Some(result) = self.withdrawal_approval_tasks.join_next() => {
375                    self.handle_completed_withdrawal_approval_task(result);
376                    while let Some(result) = self.withdrawal_approval_tasks.try_join_next() {
377                        self.handle_completed_withdrawal_approval_task(result);
378                    }
379                }
380                Some(result) = OptionFuture::from(self.withdrawal_commitment_task.as_mut()) => {
381                    self.withdrawal_commitment_task = None;
382                    Self::log_task_result("withdrawal_commitment", result);
383                }
384                Some(result) = OptionFuture::from(self.deposit_gc_task.as_mut()) => {
385                    self.deposit_gc_task = None;
386                    Self::log_task_result("deposit_gc", result);
387                }
388                Some(result) = OptionFuture::from(self.proposal_gc_task.as_mut()) => {
389                    self.proposal_gc_task = None;
390                    Self::log_task_result("proposal_gc", result);
391                }
392                Some(result) = OptionFuture::from(self.utxo_cleanup_gc_task.as_mut()) => {
393                    self.utxo_cleanup_gc_task = None;
394                    // Re-arm the scan when the task did work (more may exist
395                    // past the per-GC cap, or new spends raced in) or failed
396                    // (retry after backoff); a barren scan stays disarmed
397                    // until a withdrawal confirms. Rescheduling is left to
398                    // the leader-gated checkpoint arm: respawning here ran
399                    // ungated (leader or not) with no backoff, which kept a
400                    // node resubmitting cleanups indefinitely.
401                    match &result {
402                        Ok(Ok(0)) => self.utxo_cleanup_retry.clear(),
403                        Ok(Ok(_)) => {
404                            self.utxo_cleanup_retry.clear();
405                            self.utxo_cleanup_scan_needed = true;
406                        }
407                        _ => {
408                            self.utxo_cleanup_retry.record_failure(
409                                garbage_collection::UtxoCleanupErrorKind::Failed,
410                                checkpoint_rx.borrow().timestamp_ms,
411                            );
412                            self.utxo_cleanup_scan_needed = true;
413                        }
414                    }
415                    Self::log_task_result("utxo_cleanup_gc", result);
416                }
417                Some(result) = OptionFuture::from(self.withdrawal_archive_gc_task.as_mut()) => {
418                    self.withdrawal_archive_gc_task = None;
419                    // Same re-arm policy as the UTXO cleanup arm above.
420                    match &result {
421                        Ok(Ok(0)) => self.withdrawal_archive_retry.clear(),
422                        Ok(Ok(_)) => {
423                            self.withdrawal_archive_retry.clear();
424                            self.withdrawal_archive_scan_needed = true;
425                        }
426                        _ => {
427                            self.withdrawal_archive_retry.record_failure(
428                                garbage_collection::WithdrawalArchiveErrorKind::Failed,
429                                checkpoint_rx.borrow().timestamp_ms,
430                            );
431                            self.withdrawal_archive_scan_needed = true;
432                        }
433                    }
434                    Self::log_task_result("withdrawal_archive_gc", result);
435                }
436                Some(result) = OptionFuture::from(self.tob_prune_task.as_mut()) => {
437                    self.tob_prune_task = None;
438                    match &result {
439                        Ok(Ok(outcome)) => {
440                            self.tob_prune_retry.clear();
441                            // Only a fully-drained sweep closes the epoch
442                            // gate; a cap-limited one re-runs next tick.
443                            if outcome.drained {
444                                self.last_tob_prune_epoch = Some(outcome.swept_epoch);
445                            }
446                        }
447                        _ => self.tob_prune_retry.record_failure(
448                            garbage_collection::TobPruneErrorKind::Failed,
449                            checkpoint_rx.borrow().timestamp_ms,
450                        ),
451                    }
452                    Self::log_task_result("tob_prune", result);
453                }
454                Some(result) = OptionFuture::from(self.guardian_committee_reconcile_task.as_mut()) => {
455                    self.guardian_committee_reconcile_task = None;
456                    // On failure, clear the epoch gate so the next tick retries
457                    // (e.g. transient guardian downtime); success holds the gate
458                    // until the hashi epoch advances again.
459                    if !matches!(&result, Ok(Ok(()))) {
460                        self.last_guardian_reconcile_epoch = None;
461                    }
462                    Self::log_task_result("guardian_committee_reconcile", result);
463                }
464
465            }
466        }
467    }

469    fn log_task_result<T>(label: &str, result: Result<anyhow::Result<T>, tokio::task::JoinError>) {
470        match result {
471            Ok(Ok(_)) => {}
472            Ok(Err(err)) => error!("{label} task failed: {err:#?}"),
473            Err(err) if err.is_panic() => error!("{label} task panicked: {err}"),
474            Err(err) => error!("{label} task failed to join: {err}"),
475        }

488    /// Whether this node is the leader for the given checkpoint. An
489    /// associated function (rather than a method) so long-running leader
490    /// tasks can re-check leadership mid-flight and stop driving work that
491    /// has rotated to another node.
492    pub(super) fn node_is_leader(inner: &Arc<Hashi>, checkpoint_height: u64) -> bool {
493        match inner.onchain_state().autonomous_halt_reason() {
494            None => {}
495            Some(crate::onchain::HaltReason::Paused) => {
496                debug!("Bridge is paused, not acting as leader");
497                return false;
498            }
499            Some(crate::onchain::HaltReason::BinaryUnsupported {
500                supported_max,
501                live_max,
502            }) => {
503                // Throttle: loud on first detection, then periodic (the metric
504                // carries the steady-state signal).
505                use std::sync::atomic::Ordering;
506                let last = LAST_UNSUPPORTED_WARN_CHECKPOINT.load(Ordering::Relaxed);
507                if last == 0
508                    || checkpoint_height.saturating_sub(last) >= UNSUPPORTED_WARN_EVERY_CHECKPOINTS
509                {
510                    LAST_UNSUPPORTED_WARN_CHECKPOINT
511                        .store(checkpoint_height.max(1), Ordering::Relaxed);
512                    warn!(
513                        supported_max,
514                        live_max,
515                        checkpoint_height,
516                        "binary supports no live on-chain package version; not acting as leader — \
517                         upgrade required"
518                    );
519                }
520                return false;
521            }
522        }
523
524        match inner.config.force_run_as_leader() {
525            ForceRunAsLeader::Always => return true,
526            ForceRunAsLeader::Never => return false,
527            ForceRunAsLeader::Default => (),
528        }
529
530        let Some(committee) = inner.onchain_state().current_committee() else {
531            // TODO: do we need to do anything when bootstrapping? At genesis there is no committee.
532            return false;
533        };
534        let this_validator_address = inner
535            .config
536            .validator_address()
537            .expect("No configured validator address");
538        let Some(this_validator_idx) = committee
539            .index_of(&this_validator_address)
540            .map(|i| i as u64)
541        else {
542            // We are not in the committee yet, so we cannot be the leader
543            return false;
544        };
545        let num_validators = committee.members().len() as u64;
546
547        let current_turn = checkpoint_height / NUM_CONSECUTIVE_LEADER_CHECKPOINTS;
548        let is_leader = (current_turn % num_validators) == this_validator_idx;
549
550        trace!("Node index {this_validator_idx} is leader node: {is_leader}");
551        is_leader
552    }
553
554    /// Only the checkpoint arm of the leader loop calls this: leadership is a
555    /// function of checkpoint height, so it cannot change between checkpoints,
556    /// and a single writer keeps its regain catch-up reliable. Other arms read
557    /// the cached `is_leader`.
558    fn update_leadership(&mut self, checkpoint_height: u64) -> bool {
559        let is_leader = Self::node_is_leader(&self.inner, checkpoint_height);
560        self.set_leadership(is_leader);
561        is_leader
562    }
563
564    fn set_leadership(&mut self, is_leader: bool) {
565        self.inner.metrics.is_leader.set(i64::from(is_leader));
566        if self.is_leader && !is_leader {
567            self.stop_scheduling_deposits();
568        }
569        self.is_leader = is_leader;
570    }
571}

573fn parse_member_signature(
574    member_signature: hashi_types::proto::MemberSignature,
575) -> anyhow::Result<MemberSignature> {
576    let epoch = member_signature
577        .epoch
578        .ok_or(anyhow::anyhow!("No epoch in MemberSignature"))?;
579    let address_string = member_signature
580        .address
581        .ok_or(anyhow::anyhow!("No address in MemberSignature"))?;
582    let address = address_string
583        .parse::<Address>()
584        .map_err(|e| anyhow::anyhow!("Unable to parse Address: {}", e))?;
585    let signature = BLS12381Signature::from_bytes(
586        member_signature
587            .signature
588            .ok_or(anyhow::anyhow!("No signature in MemberSignature"))?
589            .as_bytes(),
590    )?;
591    Ok(MemberSignature::new(epoch, address, signature))
592}

594/// Invoke a peer's signing RPC, retrying on transient transport failures, and
595/// hand back the peer's final status otherwise. A retry goes out on a new
596/// connection because the peer's shared tonic channel reconnects lazily once
597/// the old one is torn down.
598async fn retry_peer_call<C, Resp, F, Fut>(
599    validator: Address,
600    what: &str,
601    mut client: impl FnMut() -> Option<C>,
602    mut call: F,
603) -> Result<tonic::Response<Resp>, tonic::Status>
604where
605    F: FnMut(C) -> Fut,
606    Fut: Future<Output = Result<tonic::Response<Resp>, tonic::Status>>,
607{
608    const MAX_ATTEMPTS: u32 = 3;
609    let mut attempt = 1;
610    loop {
611        let Some(client) = client() else {
612            return Err(tonic::Status::unavailable(format!(
613                "no bridge-service client for validator {validator}"
614            )));
615        };
616        match call(client).await {
617            Err(status) if attempt < MAX_ATTEMPTS && is_retriable_transport(&status) => {
618                warn!(
619                    "Failed to get {what} from {validator} (attempt {attempt}/{MAX_ATTEMPTS}): \
620                     {status}; retrying on a fresh connection"
621                );
622                tokio::time::sleep(Duration::from_millis(50 * u64::from(attempt))).await;
623                attempt += 1;
624            }
625            result => return result,
626        }
627    }
628}

635/// Peers refuse a request their mirror shows already approved, or a withdrawal
636/// it shows finalized, with `AlreadyExists` (`deposit_refusal_status`,
637/// `withdrawal_approval_refusal_status`, `withdrawal_signing_refusal_status`).
638fn is_already_approved_refusal(status: &tonic::Status) -> bool {
639    status.code() == tonic::Code::AlreadyExists

642enum NoQuorum {
643    AlreadyApproved,
644    StaleCommittee { epoch: u64, peer_epoch: u64 },
645    Short { weight: u64, required_weight: u64 },
646}
647
648/// Add each member's signature to `aggregator` until it reaches a certificate
649/// quorum, stopping early once members reporting the request already approved,
650/// or signing at a newer epoch than the aggregator's committee, rule a quorum out.
651async fn collect_signatures<T: IntentMessage + Clone>(
652    sig_tasks: &mut JoinSet<(u64, Result<MemberSignature, NoSignature>)>,
653    aggregator: &mut BlsSignatureAggregator<'_, T>,
654    total_weight: u64,
655) -> Result<(), NoQuorum> {
656    let required_weight = certificate_threshold(total_weight);
657    // Past `total - required`, quorum is out of reach, and that is more weight than
658    // faulty members can hold, so what those members report is true.
659    let quorum_slack = total_weight.saturating_sub(required_weight);
660    let mut already_approved_weight = 0;
661    let mut newer_epoch_weight = 0;
662    while let Some(result) = sig_tasks.join_next().await {
663        let Ok((weight, reply)) = result else {
664            continue;
665        };
666        match reply {
667            Ok(sig) if sig.epoch() > aggregator.epoch() => {
668                debug!(
669                    "{} signed at epoch {}, ahead of committee epoch {}",
670                    sig.address(),
671                    sig.epoch(),
672                    aggregator.epoch()
673                );
674                newer_epoch_weight += weight;
675                if newer_epoch_weight > quorum_slack {
676                    return Err(NoQuorum::StaleCommittee {
677                        epoch: aggregator.epoch(),
678                        peer_epoch: sig.epoch(),
679                    });
680                }
681            }
682            Ok(sig) => {
683                if let Err(e) = aggregator.add_signature(sig) {
684                    error!("Failed to add member signature: {e}");
685                }
686            }
687            Err(NoSignature::AlreadyApproved) => {
688                already_approved_weight += weight;
689                if already_approved_weight > quorum_slack {
690                    return Err(NoQuorum::AlreadyApproved);
691                }
692            }
693            Err(NoSignature::Failed) => {}
694        }
695        if aggregator.weight() >= required_weight {
696            break;
697        }
698    }
699
700    if aggregator.weight() < required_weight {
701        return Err(NoQuorum::Short {
702            weight: aggregator.weight(),
703            required_weight,
704        });
705    }
706    Ok(())
707}

709/// Whether a failed peer RPC is worth retrying. Under sustained load a peer's
710/// HTTP/2 server tears the whole multiplexed connection down — `GoAway`
711/// (surfaced as `Internal`), broken pipe (`Unknown`), or the usual
712/// `Unavailable` — failing every in-flight request to that peer at once. The
713/// peer signing RPCs are idempotent, so retrying these transport-class codes is
714/// safe and lets the request land on a fresh connection.
715pub(crate) fn is_retriable_transport(status: &tonic::Status) -> bool {
716    matches!(
717        status.code(),
718        tonic::Code::Unavailable
719            | tonic::Code::Unknown
720            | tonic::Code::Internal
721            | tonic::Code::Cancelled
722            | tonic::Code::DeadlineExceeded
723    )
724}
