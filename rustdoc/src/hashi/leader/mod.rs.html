51/// Throttle for the "binary unsupported for the on-chain version" leader
52/// warning. `node_is_leader` runs every checkpoint (and from re-checking
53/// tasks), so an unthrottled `warn!` would flood logs for the whole time the
54/// chain sits ahead of this binary; the `hashi_package_version_unsupported`
55/// metric is the durable signal, so we log on first detection and then only
56/// periodically. Stores the last checkpoint height it warned at (0 = never).
57static LAST_UNSUPPORTED_WARN_CHECKPOINT: std::sync::atomic::AtomicU64 =
58    std::sync::atomic::AtomicU64::new(0);
59const UNSUPPORTED_WARN_EVERY_CHECKPOINTS: u64 = 240;

231    /// Invoke a peer's bridge-service signing RPC, retrying on transient
232    /// transport failures. `call` is handed a freshly-fetched client each
233    /// attempt so the retry reconnects (tonic reconnects lazily) rather than
234    /// reusing the connection the peer just tore down. Returns `None` once
235    /// attempts are exhausted or on a non-transport error.
236    async fn call_peer_with_retry<Resp, F, Fut>(
237        inner: &Arc<Hashi>,
238        validator: Address,
239        what: &str,
240        mut call: F,
241    ) -> Option<tonic::Response<Resp>>
242    where
243        F: FnMut(BridgeServiceClient<BoxedChannel>) -> Fut,
244        Fut: Future<Output = Result<tonic::Response<Resp>, tonic::Status>>,
245    {
246        const MAX_ATTEMPTS: u32 = 3;
247        for attempt in 1..=MAX_ATTEMPTS {
248            let Some(client) = inner.onchain_state().bridge_service_client(&validator) else {
249                error!("Cannot find bridge-service client for validator: {validator:?}");
250                return None;
251            };
252            match call(client).await {
253                Ok(response) => return Some(response),
254                Err(status) if attempt < MAX_ATTEMPTS && is_retriable_transport(&status) => {
255                    warn!(
256                        "Failed to get {what} from {validator} (attempt {attempt}/{MAX_ATTEMPTS}): \
257                         {status}; retrying on a fresh connection"
258                    );
259                    tokio::time::sleep(Duration::from_millis(50 * u64::from(attempt))).await;
260                }
261                Err(status) => {
262                    error!("Failed to get {what} from {validator}: {status}");
263                    return None;
264                }
265            }
266        }
267        None
268    }

278    #[tracing::instrument(name = "leader", skip_all)]
279    async fn run(mut self) {
280        info!("Starting leader service");
281
282        // Wait for DKG to complete before processing any checkpoints.
283        let mpc_handle = self.inner.mpc_handle().expect("MpcHandle not initialized");
284        info!("Waiting for MPC key to become available...");
285        mpc_handle.wait_for_key_ready().await;
286        info!("MPC key is ready, starting leader loop");
287
288        let mut checkpoint_rx = self.inner.onchain_state().subscribe_checkpoint();
289        let mut btc_block_rx = self.inner.btc_monitor().subscribe_block_height();
290        let mut deposit_work_rx = self
291            .inner
292            .onchain_state()
293            .deposit_tracker()
294            .subscribe_work();
295
296        loop {
297            trace!("Waiting for next checkpoint or task completion...");
298            tokio::select! {
299                wait_result = checkpoint_rx.changed() => {
300                    if let Err(e) = wait_result {
301                        error!("Error waiting for checkpoint change: {e}");
302                        break;
303                    }
304                    let (checkpoint_height, checkpoint_timestamp_ms) = {
305                        let checkpoint_info = checkpoint_rx.borrow_and_update();
306                        (checkpoint_info.height, checkpoint_info.timestamp_ms)
307                    };
308
309                    // Every node reports its withdrawal semantics selection,
310                    // not just the current leader.
311                    // Heartbeat before checking leadership so followers report liveness too.
312                    self.inner.metrics.task_heartbeat("leader_loop");
313                    let was_leader = self.is_leader;
314                    if self.update_leadership(checkpoint_height) {
315                        debug!("Checkpoint {checkpoint_height}: We are the leader node");
316                        self.process_stale_unapproved_deposits_if_new_epoch();
317                        // Also replay deposit work dropped while processing
318                        // was stopped (e.g. a reconfig halt consumed a tracker
319                        // notification) and pick up confirmation-threshold
320                        // changes; neither produces another tracker event.
321                        if !was_leader || self.needs_actionable_deposit_reload() {
322                            self.process_actionable_unapproved_deposits();
323                        }
324                        self.process_approved_deposit_requests();
325                        self.check_reconcile_guardian_committee();
326                        self.process_unapproved_withdrawal_requests(checkpoint_timestamp_ms);
327                        self.process_approved_withdrawal_requests(checkpoint_timestamp_ms);
328                        self.process_unsigned_withdrawal_txns();
329                        self.process_signed_withdrawal_txns();
330                        self.check_delete_expired_deposit_requests(checkpoint_timestamp_ms);
331                        self.check_delete_proposals(checkpoint_timestamp_ms);
332                        self.check_cleanup_spent_utxos(checkpoint_timestamp_ms);
333                        self.check_archive_confirmed_withdrawals(checkpoint_timestamp_ms);
334                        self.check_prune_tob_certs(checkpoint_timestamp_ms);
335                    } else {
336                        trace!("We are not the leader node");
337                    }
338                }
339                wait_result = deposit_work_rx.changed() => {
340                    if let Err(e) = wait_result {
341                        error!("Error waiting for deposit tracker work change: {e}");
342                        break;
343                    }
344                    if self.is_leader {
345                        debug!("Deposit tracker changed: processing deposit requests");
346                        self.process_actionable_unapproved_deposits();
347                    }
348                }
349                wait_result = btc_block_rx.changed() => {
350                    if let Err(e) = wait_result {
351                        error!("Error waiting for Bitcoin block height change: {e}");
352                        break;
353                    }
354
355                    let block_sequence = self.inner.btc_monitor().block_sequence();
356                    self.activate_unapproved_deposits_for_btc_block(block_sequence);
357                    self.schedule_withdrawal_checks_for_btc_block();
358                }
359                Some(result) = self.unapproved_deposit_tasks.join_next() => {
360                    self.handle_completed_unapproved_deposit_task(result);
361                    while let Some(result) = self.unapproved_deposit_tasks.try_join_next() {
362                        self.handle_completed_unapproved_deposit_task(result);
363                    }
364                }
365                Some(result) = self.approved_deposit_tasks.join_next() => {
366                    self.handle_completed_approved_deposit_task(result);
367                    while let Some(result) = self.approved_deposit_tasks.try_join_next() {
368                        self.handle_completed_approved_deposit_task(result);
369                    }
370                }
371                Some(result) = self.withdrawal_signing_tasks.join_next() => {
372                    self.handle_completed_withdrawal_signing_task(result);
373                }
374                Some(result) = self.withdrawal_broadcast_tasks.join_next() => {
375                    self.handle_completed_withdrawal_broadcast_task(result);
376                }
377                Some(result) = self.withdrawal_btc_block_check_tasks.join_next() => {
378                    self.handle_completed_withdrawal_btc_block_check_task(result);
379                }
380                Some(result) = self.withdrawal_approval_tasks.join_next() => {
381                    self.handle_completed_withdrawal_approval_task(result);
382                    while let Some(result) = self.withdrawal_approval_tasks.try_join_next() {
383                        self.handle_completed_withdrawal_approval_task(result);
384                    }
385                }
386                Some(result) = OptionFuture::from(self.withdrawal_commitment_task.as_mut()) => {
387                    self.withdrawal_commitment_task = None;
388                    Self::log_task_result("withdrawal_commitment", result);
389                }
390                Some(result) = OptionFuture::from(self.deposit_gc_task.as_mut()) => {
391                    self.deposit_gc_task = None;
392                    Self::log_task_result("deposit_gc", result);
393                }
394                Some(result) = OptionFuture::from(self.proposal_gc_task.as_mut()) => {
395                    self.proposal_gc_task = None;
396                    Self::log_task_result("proposal_gc", result);
397                }
398                Some(result) = OptionFuture::from(self.utxo_cleanup_gc_task.as_mut()) => {
399                    self.utxo_cleanup_gc_task = None;
400                    // Re-arm the scan when the task did work (more may exist
401                    // past the per-GC cap, or new spends raced in) or failed
402                    // (retry after backoff); a barren scan stays disarmed
403                    // until a withdrawal confirms. Rescheduling is left to
404                    // the leader-gated checkpoint arm: respawning here ran
405                    // ungated (leader or not) with no backoff, which kept a
406                    // node resubmitting cleanups indefinitely.
407                    match &result {
408                        Ok(Ok(0)) => self.utxo_cleanup_retry.clear(),
409                        Ok(Ok(_)) => {
410                            self.utxo_cleanup_retry.clear();
411                            self.utxo_cleanup_scan_needed = true;
412                        }
413                        _ => {
414                            self.utxo_cleanup_retry.record_failure(
415                                garbage_collection::UtxoCleanupErrorKind::Failed,
416                                checkpoint_rx.borrow().timestamp_ms,
417                            );
418                            self.utxo_cleanup_scan_needed = true;
419                        }
420                    }
421                    Self::log_task_result("utxo_cleanup_gc", result);
422                }
423                Some(result) = OptionFuture::from(self.withdrawal_archive_gc_task.as_mut()) => {
424                    self.withdrawal_archive_gc_task = None;
425                    // Same re-arm policy as the UTXO cleanup arm above.
426                    match &result {
427                        Ok(Ok(0)) => self.withdrawal_archive_retry.clear(),
428                        Ok(Ok(_)) => {
429                            self.withdrawal_archive_retry.clear();
430                            self.withdrawal_archive_scan_needed = true;
431                        }
432                        _ => {
433                            self.withdrawal_archive_retry.record_failure(
434                                garbage_collection::WithdrawalArchiveErrorKind::Failed,
435                                checkpoint_rx.borrow().timestamp_ms,
436                            );
437                            self.withdrawal_archive_scan_needed = true;
438                        }
439                    }
440                    Self::log_task_result("withdrawal_archive_gc", result);
441                }
442                Some(result) = OptionFuture::from(self.tob_prune_task.as_mut()) => {
443                    self.tob_prune_task = None;
444                    match &result {
445                        Ok(Ok(outcome)) => {
446                            self.tob_prune_retry.clear();
447                            // Only a fully-drained sweep closes the epoch
448                            // gate; a cap-limited one re-runs next tick.
449                            if outcome.drained {
450                                self.last_tob_prune_epoch = Some(outcome.swept_epoch);
451                            }
452                        }
453                        _ => self.tob_prune_retry.record_failure(
454                            garbage_collection::TobPruneErrorKind::Failed,
455                            checkpoint_rx.borrow().timestamp_ms,
456                        ),
457                    }
458                    Self::log_task_result("tob_prune", result);
459                }
460                Some(result) = OptionFuture::from(self.guardian_committee_reconcile_task.as_mut()) => {
461                    self.guardian_committee_reconcile_task = None;
462                    // On failure, clear the epoch gate so the next tick retries
463                    // (e.g. transient guardian downtime); success holds the gate
464                    // until the hashi epoch advances again.
465                    if !matches!(&result, Ok(Ok(()))) {
466                        self.last_guardian_reconcile_epoch = None;
467                    }
468                    Self::log_task_result("guardian_committee_reconcile", result);
469                }
470
471            }
472        }
473    }

475    fn log_task_result<T>(label: &str, result: Result<anyhow::Result<T>, tokio::task::JoinError>) {
476        match result {
477            Ok(Ok(_)) => {}
478            Ok(Err(err)) => error!("{label} task failed: {err:#?}"),
479            Err(err) if err.is_panic() => error!("{label} task panicked: {err}"),
480            Err(err) => error!("{label} task failed to join: {err}"),
481        }
482    }

494    /// Whether this node is the leader for the given checkpoint. An
495    /// associated function (rather than a method) so long-running leader
496    /// tasks can re-check leadership mid-flight and stop driving work that
497    /// has rotated to another node.
498    pub(super) fn node_is_leader(inner: &Arc<Hashi>, checkpoint_height: u64) -> bool {
499        match inner.onchain_state().autonomous_halt_reason() {
500            None => {}
501            Some(crate::onchain::HaltReason::Paused) => {
502                debug!("Bridge is paused, not acting as leader");
503                return false;
504            }
505            Some(crate::onchain::HaltReason::BinaryUnsupported {
506                supported_max,
507                live_max,
508            }) => {
509                // Throttle: loud on first detection, then periodic (the metric
510                // carries the steady-state signal).
511                use std::sync::atomic::Ordering;
512                let last = LAST_UNSUPPORTED_WARN_CHECKPOINT.load(Ordering::Relaxed);
513                if last == 0
514                    || checkpoint_height.saturating_sub(last) >= UNSUPPORTED_WARN_EVERY_CHECKPOINTS
515                {
516                    LAST_UNSUPPORTED_WARN_CHECKPOINT
517                        .store(checkpoint_height.max(1), Ordering::Relaxed);
518                    warn!(
519                        supported_max,
520                        live_max,
521                        checkpoint_height,
522                        "binary supports no live on-chain package version; not acting as leader — \
523                         upgrade required"
524                    );
525                }
526                return false;
527            }
528        }
529
530        match inner.config.force_run_as_leader() {
531            ForceRunAsLeader::Always => return true,
532            ForceRunAsLeader::Never => return false,
533            ForceRunAsLeader::Default => (),
534        }
535
536        let Some(committee) = inner.onchain_state().current_committee() else {
537            // TODO: do we need to do anything when bootstrapping? At genesis there is no committee.
538            return false;
539        };
540        let this_validator_address = inner
541            .config
542            .validator_address()
543            .expect("No configured validator address");
544        let Some(this_validator_idx) = committee
545            .index_of(&this_validator_address)
546            .map(|i| i as u64)
547        else {
548            // We are not in the committee yet, so we cannot be the leader
549            return false;
550        };
551        let num_validators = committee.members().len() as u64;
552
553        let current_turn = checkpoint_height / NUM_CONSECUTIVE_LEADER_CHECKPOINTS;
554        let is_leader = (current_turn % num_validators) == this_validator_idx;
555
556        trace!("Node index {this_validator_idx} is leader node: {is_leader}");
557        is_leader
558    }
559
560    /// Only the checkpoint arm of the leader loop calls this: leadership is a
561    /// function of checkpoint height, so it cannot change between checkpoints,
562    /// and a single writer keeps its regain catch-up reliable. Other arms read
563    /// the cached `is_leader`.
564    fn update_leadership(&mut self, checkpoint_height: u64) -> bool {
565        let is_leader = Self::node_is_leader(&self.inner, checkpoint_height);
566        self.inner.metrics.is_leader.set(i64::from(is_leader));
567        if self.is_leader && !is_leader {
568            self.stop_deposit_processing();
569        }
570        self.is_leader = is_leader;
571        is_leader
572    }
573}

575fn parse_member_signature(
576    member_signature: hashi_types::proto::MemberSignature,
577) -> anyhow::Result<MemberSignature> {
578    let epoch = member_signature
579        .epoch
580        .ok_or(anyhow::anyhow!("No epoch in MemberSignature"))?;
581    let address_string = member_signature
582        .address
583        .ok_or(anyhow::anyhow!("No address in MemberSignature"))?;
584    let address = address_string
585        .parse::<Address>()
586        .map_err(|e| anyhow::anyhow!("Unable to parse Address: {}", e))?;
587    let signature = BLS12381Signature::from_bytes(
588        member_signature
589            .signature
590            .ok_or(anyhow::anyhow!("No signature in MemberSignature"))?
591            .as_bytes(),
592    )?;
593    Ok(MemberSignature::new(epoch, address, signature))
594}

596/// Whether a failed peer RPC is worth retrying. Under sustained load a peer's
597/// HTTP/2 server tears the whole multiplexed connection down — `GoAway`
598/// (surfaced as `Internal`), broken pipe (`Unknown`), or the usual
599/// `Unavailable` — failing every in-flight request to that peer at once. The
600/// peer signing RPCs are idempotent, so retrying these transport-class codes is
601/// safe and lets the request land on a fresh connection.
602fn is_retriable_transport(status: &tonic::Status) -> bool {
603    matches!(
604        status.code(),
605        tonic::Code::Unavailable
606            | tonic::Code::Unknown
607            | tonic::Code::Internal
608            | tonic::Code::Cancelled
609            | tonic::Code::DeadlineExceeded
610    )
611}

613#[cfg(test)]
614mod tests {
615    use super::is_retriable_transport;
616    use tonic::Code;
617    use tonic::Status;
618
619    #[test]
620    fn classifies_transport_errors_as_retriable() {
621        // A peer tearing the connection down surfaces as these codes: GoAway ->
622        // Internal, broken pipe -> Unknown, plus the usual transient codes.
623        for code in [
624            Code::Unavailable,
625            Code::Unknown,
626            Code::Internal,
627            Code::Cancelled,
628            Code::DeadlineExceeded,
629        ] {
630            assert!(
631                is_retriable_transport(&Status::new(code, "boom")),
632                "{code:?} should be retried"
633            );
634        }
635        // Genuine application rejections from the peer must not be retried.
636        for code in [
637            Code::InvalidArgument,
638            Code::FailedPrecondition,
639            Code::PermissionDenied,
640            Code::NotFound,
641            Code::AlreadyExists,
642        ] {
643            assert!(
644                !is_retriable_transport(&Status::new(code, "nope")),
645                "{code:?} should not be retried"
646            );
647        }
648    }
