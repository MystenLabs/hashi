51    /// Approve every unapproved withdrawal, one task and one
52    /// transaction per request (mirroring the deposit approval pool). Each
53    /// task holds its request in `inflight_withdrawal_approvals` through the
54    /// object-mirror wait, so a later cycle cannot re-approve it while other
55    /// requests keep flowing — a single request's mirror wait must not stall
56    /// the rest of a batch past the commitment path's batching delay.
57    pub(super) fn process_unapproved_withdrawal_requests(&mut self, checkpoint_timestamp_ms: u64) {
58        debug!("Entering process_unapproved_withdrawal_requests");
59        if self.is_reconfiguring() {
60            debug!("Reconfig in progress, skipping withdrawal approval processing");
61            return;
62        }
63
64        let mut unapproved: Vec<_> = self
65            .inner
66            .onchain_state()
67            .withdrawal_requests()
68            .into_iter()
69            .filter(|r| r.awaits_approval())
70            .collect();
71        unapproved.sort_by_key(|r| r.created_timestamp_ms);
72
73        let unapproved_ids: Vec<Address> = unapproved.iter().map(|r| r.id).collect();
74        self.withdrawal_approval_retry_tracker
75            .prune(&unapproved_ids);
76
77        let to_process: Vec<_> = unapproved
78            .into_iter()
79            .filter(|r| !self.inflight_withdrawal_approvals.contains(&r.id))
80            .filter(|r| {
81                !self
82                    .withdrawal_approval_retry_tracker
83                    .should_skip(&r.id, checkpoint_timestamp_ms)
84            })
85            .collect();
86
87        self.inner
88            .metrics
89            .leader_items_in_backoff
90            .with_label_values(&["withdrawal_approval"])
91            .set(
92                self.withdrawal_approval_retry_tracker
93                    .in_backoff_count(checkpoint_timestamp_ms) as i64,
94            );
95
96        if to_process.is_empty() {
97            return;
98        }
99
100        let this_validator_address = self
101            .inner
102            .config
103            .validator_address()
104            .expect("No configured validator address");
105
106        let members = self
107            .inner
108            .onchain_state()
109            .current_committee_members()
110            .expect("No current committee members");
111
112        let committee = self
113            .inner
114            .onchain_state()
115            .current_committee()
116            .expect("No current committee");
117
118        let max_concurrent = self.inner.config.max_concurrent_leader_job_tasks();
119        for request in to_process {
120            // Whatever does not fit is picked up again on a later checkpoint
121            // tick; the worklist is recomputed from the mirror every cycle.
122            if self.withdrawal_approval_tasks.len() >= max_concurrent {
123                break;
124            }
125
126            let inner = self.inner.clone();
127            let retry_tracker = self.withdrawal_approval_retry_tracker.clone();
128            let members = members.clone();
129            let committee = committee.clone();
130            self.inflight_withdrawal_approvals.insert(request.id);
131            self.withdrawal_approval_tasks.spawn(async move {
132                let request_id = request.id;
133                let task_result = tokio::time::timeout(
134                    LEADER_TASK_TIMEOUT,
135                    Self::process_unapproved_withdrawal_request(
136                        inner.clone(),
137                        retry_tracker.clone(),
138                        request,
139                        checkpoint_timestamp_ms,
140                        this_validator_address,
141                        &members,
142                        &committee,
143                    ),
144                )
145                .await;
146
147                match task_result {
148                    Ok(Ok(())) => {}
149                    Ok(Err(err)) => {
150                        let kind = WithdrawalApprovalErrorKind::TaskFailed;
151                        inner
152                            .metrics
153                            .leader_retries_total
154                            .with_label_values(&["withdrawal_approval", &format!("{kind:?}")])
155                            .inc();
156                        retry_tracker.record_failure(kind, request_id, checkpoint_timestamp_ms);
157                        error!(request_id = %request_id, "Withdrawal approval failed: {err:#}");
158                    }
159                    Err(_) => {
160                        let kind = WithdrawalApprovalErrorKind::TimedOut;
161                        inner
162                            .metrics
163                            .leader_retries_total
164                            .with_label_values(&["withdrawal_approval", &format!("{kind:?}")])
165                            .inc();
166                        retry_tracker.record_failure(kind, request_id, checkpoint_timestamp_ms);
167                    }
168                }
169
170                request_id
171            });
172        }
173    }

175    /// Completion handler for the withdrawal approval task pool: release the
176    /// in-flight guard. Failures were already recorded into the retry
177    /// tracker inside the task.
178    pub(super) fn handle_completed_withdrawal_approval_task(
179        &mut self,
180        result: Result<Address, tokio::task::JoinError>,
181    ) {
182        match result {
183            Ok(request_id) => {
184                self.inflight_withdrawal_approvals.remove(&request_id);
185            }
186            Err(err) if err.is_panic() => std::panic::resume_unwind(err.into_panic()),
187            Err(err) => error!("Withdrawal approval task failed to join: {err}"),
188        }
189    }

191    /// Approve one withdrawal request end-to-end: validate and sign locally,
192    /// collect a certificate quorum, submit the approval on its own
193    /// transaction, and wait for the object mirror to apply it. Handled
194    /// failures are recorded into the retry tracker and return `Ok(())`;
195    /// `Err` is reserved for unexpected task failures.
196    #[tracing::instrument(level = "info", skip_all, fields(request_id = %request.id))]
197    async fn process_unapproved_withdrawal_request(
198        inner: Arc<Hashi>,
199        retry_tracker: RetryTracker<WithdrawalApprovalErrorKind>,
200        request: WithdrawalRequest,
201        checkpoint_timestamp_ms: u64,
202        this_validator_address: Address,
203        members: &[CommitteeMember],
204        committee: &hashi_types::committee::RuntimeCommittee,
205    ) -> anyhow::Result<()> {
206        let approval = WithdrawalRequestApproval {
207            request_id: request.id,
208        };
209
210        // Validate, screen, and sign locally first
211        let local_sig = match inner
212            .validate_and_sign_withdrawal_request_approval(&approval)
213            .await
214        {
215            Ok(sig) => {
216                retry_tracker.clear(&request.id);
217                parse_member_signature(sig).unwrap()
218            }
219            Err(e) => {
220                let kind = e.kind();
221                warn!(error = %e, error_kind = ?kind, "Failed to build withdrawal commitment");
222                inner
223                    .metrics
224                    .leader_retries_total
225                    .with_label_values(&["withdrawal_approval", &format!("{kind:?}")])
226                    .inc();
227                retry_tracker.record_failure(kind, request.id, checkpoint_timestamp_ms);
228                return Ok(());
229            }
230        };
231
232        let proto_request = approval.to_proto();
233
234        let mut aggregator =
235            committee.signature_aggregator(inner.config.hashi_ids().hashi_object_id, approval);
236        if let Err(e) = aggregator.add_signature(local_sig) {
237            error!("Failed to add local approval signature: {e}");
238        }
239
240        // Fan out signature requests to remote members in parallel.
241        let mut sig_tasks = JoinSet::new();
242        for member in members {
243            if member.validator_address() == this_validator_address {
244                continue;
245            }
246            let inner = inner.clone();
247            let proto_request = proto_request.clone();
248            let member = member.clone();
249            sig_tasks.spawn(async move {
250                let reply =
251                    Self::request_withdrawal_approval_signature(&inner, proto_request, &member)
252                        .await;
253                (member.weight(), reply)
254            });
255        }
256
257        if let Err(no_quorum) =
258            collect_signatures(&mut sig_tasks, &mut aggregator, committee.total_weight()).await
259        {
260            let kind = match no_quorum {
261                NoQuorum::AlreadyApproved => {
262                    debug!("Peers report the withdrawal request already approved");
263                    WithdrawalApprovalErrorKind::AlreadyApproved
264                }
265                NoQuorum::Short {
266                    weight,
267                    required_weight,
268                } => {
269                    error!("Insufficient approval signatures: weight {weight} < {required_weight}");
270                    WithdrawalApprovalErrorKind::FailedQuorum
271                }
272            };
273            inner
274                .metrics
275                .leader_retries_total
276                .with_label_values(&["withdrawal_approval", &format!("{kind:?}")])
277                .inc();
278            retry_tracker.record_failure(kind, request.id, checkpoint_timestamp_ms);
279            return Ok(());
280        }
281
282        let cert = match aggregator.finish() {
283            Ok(signed) => signed.committee_signature().clone(),
284            Err(e) => {
285                error!("Failed to build approval certificate: {e}");
286                return Ok(());
287            }
288        };
289
290        // Submit this request's approval on its own transaction — a duplicate
291        // approval (another leader's stale mirror across a handoff) aborts
292        // on-chain and must not take unrelated approvals down with it — and
293        // wait for the object mirror to apply it before completing, matching
294        // the deposit approval path. The subtask holds its slot through the
295        // wait, so the next approval cycle reads a mirror that already
296        // reflects this approval instead of re-approving it (IOP-662).
297        let mut executor = SuiTxExecutor::from_hashi(inner.clone())?;
298        let checkpoint = executor
299            .execute_approve_withdrawal_request(request.id, &cert)
300            .await
301            .inspect(|checkpoint| {
302                inner
303                    .metrics
304                    .sui_tx_submissions_total
305                    .with_label_values(&["approve_withdrawal", "success"])
306                    .inc();
307                info!(checkpoint, "Successfully submitted withdrawal approval");
308            })
309            .inspect_err(|e| {
310                inner
311                    .metrics
312                    .sui_tx_submissions_total
313                    .with_label_values(&["approve_withdrawal", "failure"])
314                    .inc();
315                error!("Failed to submit withdrawal approval: {e}");
316            });
317        let Ok(checkpoint) = checkpoint else {
318            let kind = WithdrawalApprovalErrorKind::SubmitFailed;
319            inner
320                .metrics
321                .leader_retries_total
322                .with_label_values(&["withdrawal_approval", &format!("{kind:?}")])
323                .inc();
324            retry_tracker.record_failure(kind, request.id, checkpoint_timestamp_ms);
325            return Ok(());
326        };
327        inner
328            .onchain_state()
329            .wait_until_checkpoint(checkpoint)
330            .await;
331        Ok(())
332    }

334    #[tracing::instrument(level = "debug", skip_all, fields(validator = %member.validator_address()))]
335    async fn request_withdrawal_approval_signature(
336        inner: &Arc<Hashi>,
337        proto_request: SignWithdrawalRequestApprovalRequest,
338        member: &CommitteeMember,
339    ) -> Result<MemberSignature, NoSignature> {
340        let validator_address = member.validator_address();
341        trace!("Requesting withdrawal request approval signature");
342
343        let response = match retry_peer_call(
344            validator_address,
345            "withdrawal request approval signature",
346            || {
347                inner
348                    .onchain_state()
349                    .bridge_service_client(&validator_address)
350            },
351            move |mut client| {
352                let request = proto_request.clone();
353                async move { client.sign_withdrawal_request_approval(request).await }
354            },
355        )
356        .await
357        {
358            Ok(response) => response,
359            Err(status) if is_already_approved_refusal(&status) => {
360                debug!("{validator_address} reports the withdrawal request already approved");
361                return Err(NoSignature::AlreadyApproved);
362            }
363            Err(e) => {
364                error!(
365                    "Failed to get withdrawal request approval signature from {validator_address}: {e}"
366                );
367                return Err(NoSignature::Failed);
368            }
369        };
370
371        trace!(
372            "Retrieved withdrawal request approval signature from {}",
373            validator_address
374        );
375
376        response
377            .into_inner()
378            .member_signature
379            .ok_or_else(|| anyhow::anyhow!("No member_signature in response"))
380            .and_then(parse_member_signature)
381            .inspect_err(|e| {
382                error!(
383                    "Failed to parse member signature from withdrawal request approval response from {}: {e}",
384                    validator_address
385                );
386            })
387            .map_err(|_| NoSignature::Failed)
388    }

394    pub(super) fn process_approved_withdrawal_requests(&mut self, checkpoint_timestamp_ms: u64) {
395        debug!("Entering process_approved_withdrawal_requests");
396        if self.is_reconfiguring() {
397            debug!("Reconfig in progress, skipping withdrawal commitment processing");
398            return;
399        }
400
401        if self.withdrawal_commitment_task.is_some() {
402            debug!("Withdrawal commitment task already in-flight, skipping");
403            return;
404        }
405
406        // Pairs with the spawn-side `max_concurrent = 1` cap: don't
407        // double-commit before the prior signing task has spawned.
408        if self.inner.onchain_state().has_unsigned_withdrawal_txn() {
409            debug!("Unsigned withdrawal txn already on-chain, skipping commitment");
410            return;
411        }
412
413        let mut approved: Vec<_> = self
414            .inner
415            .onchain_state()
416            .withdrawal_requests()
417            .into_iter()
418            .filter(|r| r.awaits_commitment())
419            .collect();
420        approved.sort_by_key(|r| r.created_timestamp_ms);
421
422        // Prune stuck-warn entries so a re-stuck request warns again.
423        let pending_ids: HashSet<Address> = approved.iter().map(|r| r.id).collect();
424        self.stuck_withdrawal_warned
425            .retain(|id| pending_ids.contains(id));
426
427        if self
428            .withdrawal_commitment_retry_tracker
429            .should_skip(checkpoint_timestamp_ms)
430        {
431            self.inner
432                .metrics
433                .leader_items_in_backoff
434                .with_label_values(&["withdrawal_commitment"])
435                .set(
436                    self.withdrawal_commitment_retry_tracker
437                        .in_backoff_count(checkpoint_timestamp_ms) as i64,
438                );
439            return;
440        }
441
442        self.inner
443            .metrics
444            .leader_items_in_backoff
445            .with_label_values(&["withdrawal_commitment"])
446            .set(
447                self.withdrawal_commitment_retry_tracker
448                    .in_backoff_count(checkpoint_timestamp_ms) as i64,
449            );
450
451        if approved.is_empty() {
452            return;
453        }
454
455        // Skip oversize requests (would HOL-block forever) and take the
456        // longest prefix of the rest that fits current capacity. The
457        // dropped tail flips `at_capacity` so we don't sit on demand for
458        // the full batching window.
459        let (batch, at_capacity) = if let Some(limiter) = self.inner.local_limiter() {
460            let timestamp_secs = checkpoint_timestamp_ms / 1000;
461            let view = limiter.view();
462            let max_bucket = view.config.max_bucket_capacity;
463            let capacity = view.capacity_at(timestamp_secs);
464
465            let mut batch: Vec<WithdrawalRequest> = Vec::new();
466            let mut cumulative = 0u64;
467            let mut at_capacity = false;
468            for req in approved {
469                if req.btc_amount > max_bucket {
470                    if self.stuck_withdrawal_warned.insert(req.id) {
471                        warn!(
472                            request_id = %req.id,
473                            btc_amount = req.btc_amount,
474                            max_bucket_capacity = max_bucket,
475                            "Withdrawal exceeds limiter max bucket; skipping"
476                        );
477                        self.inner
478                            .metrics
479                            .guardian_limiter_stuck_oversize_skipped_total
480                            .inc();
481                    }
482                    continue;
483                }
484                let Some(next) = cumulative.checked_add(req.btc_amount) else {
485                    at_capacity = true;
486                    break;
487                };
488                if next > capacity {
489                    at_capacity = true;
490                    break;
491                }
492                cumulative = next;
493                batch.push(req);
494            }
495
496            if batch.is_empty() {
497                // All-oversize (already warned) or refill-bound head.
498                self.inner
499                    .metrics
500                    .guardian_limiter_batch_stuck_head_total
501                    .inc();
502                return;
503            }
504            if at_capacity {
505                self.inner
506                    .metrics
507                    .guardian_limiter_batch_truncated_total
508                    .inc();
509            }
510            (batch, at_capacity)
511        } else {
512            debug!("No local guardian limiter yet; not committing withdrawals");
513            return;
514        };
515
516        let max_batch = withdrawal_fire_threshold(self.inner.config.withdrawal_max_batch_size());
517        let delay_ms = self.inner.config.withdrawal_batching_delay_ms();
518
519        let batch_is_full = batch.len() >= max_batch;
520        let oldest_has_waited = batch
521            .first()
522            .is_some_and(|r| checkpoint_timestamp_ms >= r.created_timestamp_ms + delay_ms);
523
524        if !batch_is_full && !oldest_has_waited && !at_capacity {
525            debug!(
526                "Holding {} approved request(s): oldest is {}ms old, \
527                 waiting for {}ms delay or {} requests",
528                batch.len(),
529                checkpoint_timestamp_ms.saturating_sub(batch[0].created_timestamp_ms),
530                delay_ms,
531                max_batch,
532            );
533            return;
534        }
535
536        let inner = self.inner.clone();
537        let retry_tracker = self.withdrawal_commitment_retry_tracker.clone();
538
539        self.withdrawal_commitment_task =
540            Some(AbortOnDropHandle::new(tokio::task::spawn(async move {
541                let task_result = tokio::time::timeout(
542                    LEADER_TASK_TIMEOUT,
543                    Self::process_approved_withdrawal_request_batch(
544                        inner.clone(),
545                        retry_tracker.clone(),
546                        batch,
547                        checkpoint_timestamp_ms,
548                    ),
549                )
550                .await;
551
552                match task_result {
553                    Ok(result) => result,
554                    Err(_) => {
555                        let kind = WithdrawalCommitmentErrorKind::TimedOut;
556                        inner
557                            .metrics
558                            .leader_retries_total
559                            .with_label_values(&["withdrawal_commitment", &format!("{kind:?}")])
560                            .inc();
561                        Err(anyhow::anyhow!(
562                            "withdrawal commitment timed out after {LEADER_TASK_TIMEOUT:?}"
563                        ))
564                    }
565                }
566            })));
567    }

569    #[tracing::instrument(level = "info", skip_all, fields(batch_size = requests.len()))]
570    async fn process_approved_withdrawal_request_batch(
571        inner: Arc<Hashi>,
572        retry_tracker: GlobalRetryTracker<WithdrawalCommitmentErrorKind>,
573        requests: Vec<WithdrawalRequest>,
574        checkpoint_timestamp_ms: u64,
575    ) -> anyhow::Result<()> {
576        info!(
577            withdrawal_request_ids = ?requests.iter().map(|r| r.id).collect::<Vec<_>>(),
578            "Processing batch of {} approved withdrawal request(s)",
579            requests.len(),
580        );
581
582        let approval = match build_checked_commitment(&inner, requests).await {
583            Ok(Some(approval)) => {
584                retry_tracker.clear();
585                approval
586            }
587            Ok(None) => {
588                info!("An unsigned withdrawal appeared; skipping this batch");
589                retry_tracker.clear();
590                return Ok(());
591            }
592            Err(e) => {
593                let kind = e.kind();
594                match kind {
595                    WithdrawalCommitmentErrorKind::CommitmentCheckFailed => {
596                        error!("Withdrawal batch not proposed: {e}")
597                    }
598                    WithdrawalCommitmentErrorKind::UtxoSelectionFailed => {}
599                    _ => warn!("Withdrawal commitment attempt failed: {e}"),
600                }
601                inner
602                    .metrics
603                    .leader_retries_total
604                    .with_label_values(&["withdrawal_commitment", &format!("{kind:?}")])
605                    .inc();
606                retry_tracker.record_failure(kind, checkpoint_timestamp_ms);
607                return Ok(());
608            }
609        };
610
611        // Fan out to committee for BLS signatures over the commitment message
612        let members = inner
613            .onchain_state()
614            .current_committee_members()
615            .expect("No current committee members");
616
617        let committee = inner
618            .onchain_state()
619            .current_committee()
620            .expect("No current committee");
621
622        let required_weight = certificate_threshold(committee.total_weight());
623        let proto_request = approval.to_proto();
624
625        // Fan out signature requests to all members in parallel.
626        let mut sig_tasks = JoinSet::new();
627        for member in members {
628            let inner = inner.clone();
629            let proto_request = proto_request.clone();
630            sig_tasks.spawn(async move {
631                Self::request_withdrawal_tx_commitment_signature(&inner, proto_request, &member)
632                    .await
633            });
634        }
635
636        // Collect signatures, stopping once we reach quorum.
637        let mut aggregator = committee
638            .signature_aggregator(inner.config.hashi_ids().hashi_object_id, approval.clone());
639        while let Some(result) = sig_tasks.join_next().await {
640            let Ok(Some(sig)) = result else { continue };
641            if let Err(e) = aggregator.add_signature(sig) {
642                error!("Failed to add withdrawal commitment signature: {e}");
643            }
644            if aggregator.weight() >= required_weight {
645                break;
646            }
647        }
648
649        if aggregator.weight() < required_weight {
650            inner
651                .metrics
652                .leader_retries_total
653                .with_label_values(&["withdrawal_commitment", "FailedQuorum"])
654                .inc();
655            retry_tracker.record_failure(
656                WithdrawalCommitmentErrorKind::FailedQuorum,
657                checkpoint_timestamp_ms,
658            );
659            error!(
660                "Insufficient withdrawal commitment signatures: weight {} < {required_weight}",
661                aggregator.weight()
662            );
663            return Ok(());
664        }
665
666        let signed_approval = match aggregator.finish() {
667            Ok(signed_approval) => signed_approval,
668            Err(e) => {
669                error!("Failed to build withdrawal commitment certificate: {e}");
670                return Ok(());
671            }
672        };
673
674        // Proactively trigger a presig refill if this commit will allocate
675        // indices beyond the current pool.
676        {
677            let num_inputs = approval.selected_utxos.len() as u64;
678            let num_consumed = inner.onchain_state().state().hashi().num_consumed_presigs;
679            let needed_end = num_consumed + num_inputs;
680            if let Some(signing_manager) = inner.current_signing_manager() {
681                let available_end = signing_manager.available_presig_end_index();
682                if needed_end > available_end {
683                    info!(
684                        "Presig pool may be insufficient for this withdrawal: \
685                         need index {needed_end}, pool ends at {available_end}. \
686                         Triggering proactive refill.",
687                    );
688                    signing_manager.trigger_refill();
689                }
690            }
691        }
692
693        // Submit commit_withdrawal_tx to Sui
694        Self::submit_commit_withdrawal_tx(&inner, &approval, signed_approval.committee_signature())
695            .await
696            .inspect(|()| {
697                inner
698                    .metrics
699                    .sui_tx_submissions_total
700                    .with_label_values(&["commit_withdrawal", "success"])
701                    .inc();
702            })
703            .inspect_err(|e| {
704                inner
705                    .metrics
706                    .sui_tx_submissions_total
707                    .with_label_values(&["commit_withdrawal", "failure"])
708                    .inc();
709                error!("Failed to submit commit_withdrawal_tx: {e}");
710            })?;
711
712        Ok(())
713    }
714
715    #[tracing::instrument(level = "debug", skip_all, fields(validator = %member.validator_address()))]
716    async fn request_withdrawal_tx_commitment_signature(
717        inner: &Arc<Hashi>,
718        proto_request: SignWithdrawalTxConstructionRequest,
719        member: &CommitteeMember,
720    ) -> Option<MemberSignature> {
721        let validator_address = member.validator_address();
722        trace!("Requesting withdrawal tx commitment signature");
723
724        let response = Self::call_peer_with_retry(
725            inner,
726            validator_address,
727            "withdrawal tx commitment signature",
728            move |mut client| {
729                let request = proto_request.clone();
730                async move { client.sign_withdrawal_tx_construction(request).await }
731            },
732        )
733        .await?;
734
735        trace!(
736            "Retrieved withdrawal approval signature from {}",
737            validator_address
738        );
739
740        response
741            .into_inner()
742            .member_signature
743            .ok_or_else(|| anyhow::anyhow!("No member_signature in response"))
744            .and_then(parse_member_signature)
745            .inspect_err(|e| {
746                error!(
747                    "Failed to parse member signature from withdrawal approval response from {}: {e}",
748                    validator_address
749                );
750            })
751            .ok()
752    }

754    async fn submit_commit_withdrawal_tx(
755        inner: &Arc<Hashi>,
756        approval: &WithdrawalTxCommitment,
757        cert: &CommitteeSignature,
758    ) -> anyhow::Result<()> {
759        info!(

765        executor.execute_commit_withdrawal_tx(approval, cert).await
766    }

806/// The batch size at which the scheduler stops holding for more demand.
807/// Capped by the executable request cap: commitment construction trims the
808/// batch to it, so a threshold above the cap can never be met and only pays
809/// the full batching delay before committing the capped batch anyway.
810fn withdrawal_fire_threshold(config_max: usize) -> usize {
811    config_max.min(crate::utxo_pool::CoinSelectionParams::MAX_WITHDRAWAL_REQUESTS)
812}
813
814const COMMITMENT_CHECK_ROUNDS: usize = 4;
815
816async fn build_checked_commitment(
817    inner: &Hashi,
818    mut requests: Vec<WithdrawalRequest>,
819) -> Result<Option<WithdrawalTxCommitment>, WithdrawalCommitmentError> {
820    let commit_gate_open = || !inner.onchain_state().has_unsigned_withdrawal_txn();
821    let mut excluded_inputs = BTreeSet::new();
822    for _ in 0..COMMITMENT_CHECK_ROUNDS {
823        let approval = match inner
824            .build_withdrawal_tx_commitment(&requests, &excluded_inputs)
825            .await
826        {
827            Ok(approval) => approval,
828            Err(e) if !commit_gate_open() => {
829                debug!("Batch build failed, and an unsigned withdrawal has appeared: {e}");
830                return Ok(None);
831            }
832            Err(e) => return Err(e),
833        };
834        let refusal = match inner.validate_withdrawal_tx_commitment(&approval).await {
835            Ok(()) => return Ok(commit_gate_open().then_some(approval)),
836            Err(e) if !commit_gate_open() => {
837                debug!(
838                    "Commit check refused the batch, and an unsigned withdrawal has appeared: {e:#}"
839                );
840                return Ok(None);
841            }
842            Err(refusal) => refusal,
843        };
844        if refusal.downcast_ref::<FeeEstimateUnavailable>().is_some() {
845            return Err(WithdrawalCommitmentError::FeeEstimateFailed(refusal));
846        }
847        let Some(refused) = refusal.downcast_ref::<RefusedItems>() else {
848            return Err(WithdrawalCommitmentError::CommitmentCheckFailed(refusal));
849        };
850        warn!(
851            count = refused.0.len(),
852            "Commit check refused items; leaving them out of the withdrawal batch: {refused}"
853        );
854        for item in &refused.0 {
855            inner
856                .metrics
857                .withdrawal_commitment_left_out_total
858                .with_label_values(&[item.item.label(), item.reason])
859                .inc();
860        }
861        leave_out(refused, &mut requests, &mut excluded_inputs);
862        if requests.is_empty() {
863            return Err(WithdrawalCommitmentError::CommitmentCheckFailed(
864                anyhow::anyhow!("the commit check refused every request in the batch"),
865            ));
866        }
867    }
868    Err(WithdrawalCommitmentError::CommitmentCheckFailed(
869        anyhow::anyhow!(
870            "the commit check still refused the batch after {COMMITMENT_CHECK_ROUNDS} rounds"
871        ),
872    ))
873}
874
875fn leave_out(
876    refused: &RefusedItems,
877    requests: &mut Vec<WithdrawalRequest>,
878    excluded_inputs: &mut BTreeSet<UtxoId>,
879) {
880    for item in &refused.0 {
881        match item.item {
882            CommitmentItem::Request(id) => requests.retain(|r| r.id != id),
883            CommitmentItem::Input(id) => {
884                excluded_inputs.insert(id);
885            }
886        }
887    }
888}
