24const CONFIG_TIMEOUT: Duration = Duration::from_secs(30);
25const CONNECT_TIMEOUT: Duration = Duration::from_secs(10);
26const TRANSFER_TIMEOUT: Duration = Duration::from_secs(15 * 60);

28#[derive(Debug, Clone, PartialEq, Eq)]
29pub enum ObjectVersion {
30    S3(String),
31    Gcs(i64),
32}
33
34#[derive(Debug)]
35pub struct UploadReceipt {
36    pub version: Option<ObjectVersion>,
37    pub e_tag: Option<String>,
38}
39
40#[derive(Debug, thiserror::Error)]
41pub enum UploadError {
42    #[error("remote backup already exists")]
43    AlreadyExists(#[source] anyhow::Error),
44    #[error(transparent)]
45    Other(#[from] anyhow::Error),
46}
47
48impl UploadError {
49    pub(crate) fn context(self, message: String) -> Self {
50        match self {
51            Self::AlreadyExists(error) => Self::AlreadyExists(error.context(message)),
52            Self::Other(error) => Self::Other(error.context(message)),
53        }
54    }
55}

57#[derive(Clone)]
58pub enum BackupRemoteClient {
59    S3(aws_sdk_s3::Client),
60    Gcs(Box<Storage>),
61}

63impl BackupRemoteClient {
64    fn provider(&self) -> BackupProvider {
65        match self {
66            Self::S3(_) => BackupProvider::S3,
67            Self::Gcs(_) => BackupProvider::Gcs,
68        }
69    }

71    pub async fn connect(config: &BackupRemoteConfig) -> anyhow::Result<Self> {
72        tokio::time::timeout(CONFIG_TIMEOUT, async {
73            match config {
74                BackupRemoteConfig::S3(config) => {
75                    let config = aws_config::defaults(BehaviorVersion::latest())
76                        .region(Region::new(config.region().to_owned()))
77                        .retry_config(RetryConfig::standard().with_max_attempts(1))
78                        .timeout_config(timeouts())
79                        .load()
80                        .await;
81                    let mut builder = client_builder(Builder::from(&config));
82                    // Preserve the existing S3-compatible endpoint convention.
83                    if std::env::var_os("AWS_ENDPOINT_URL_S3").is_some() {
84                        builder = builder.force_path_style(true);
85                    }
86                    Ok(Self::S3(aws_sdk_s3::Client::from_conf(builder.build())))
87                }
88                BackupRemoteConfig::Gcs(_) => {
89                    crate::init_crypto_provider();
90                    let client = Storage::builder()
91                        .with_retry_policy(NeverRetry)
92                        .with_read_resume_policy(NeverResume)
93                        .build()
94                        .await
95                        .context("failed to load GCS backup configuration")?;
96                    Ok(Self::Gcs(Box::new(client)))
97                }
98            }
99        })
100        .await
101        .context("timed out loading remote backup configuration")?
102    }

104    pub async fn upload(
105        &self,
106        bucket: &str,
107        key: &str,
108        archive: &Path,
109    ) -> Result<UploadReceipt, UploadError> {
110        // The outer deadline also covers credential resolution and opening/streaming
111        // the file, not just the SDK's HTTP request. Never remove the local archive.
112        tokio::time::timeout(TRANSFER_TIMEOUT, async {
113            match self {
114                Self::S3(client) => {
115                    let body = ByteStream::from_path(archive)
116                        .await
117                        .context("failed to open backup archive for upload")?;
118                    client
119                        .put_object()
120                        .bucket(bucket)
121                        .key(key)
122                        .if_none_match("*")
123                        .body(body)
124                        .send()
125                        .await
126                        .map(|receipt| UploadReceipt {
127                            version: receipt.version_id.map(ObjectVersion::S3),
128                            e_tag: receipt.e_tag,
129                        })
130                        .map_err(|error| {
131                            let already_exists = error
132                                .raw_response()
133                                .is_some_and(|response| response.status().as_u16() == 412);
134                            let error =
135                                anyhow::Error::new(error).context("S3 backup PutObject failed");
136                            if already_exists {
137                                UploadError::AlreadyExists(error)
138                            } else {
139                                UploadError::Other(error)
140                            }
141                        })
142                }
143                Self::Gcs(client) => {
144                    let file = tokio::fs::File::open(archive)
145                        .await
146                        .context("failed to open backup archive for upload")?;
147                    let receipt = client
148                        .write_object(format!("projects/_/buckets/{bucket}"), key, file)
149                        .set_if_generation_match(0_i64)
150                        // Supply CRC before creation so GCS rejects corrupt data before publishing.
151                        .precompute_checksums()
152                        .await
153                        .context("failed to precompute GCS backup upload checksums")?
154                        .send_unbuffered()
155                        .await
156                        .map_err(|error| {
157                            let already_exists = error.http_status_code() == Some(412);
158                            let error =
159                                anyhow::Error::new(error).context("GCS backup upload failed");
160                            if already_exists {
161                                UploadError::AlreadyExists(error)
162                            } else {
163                                UploadError::Other(error)
164                            }
165                        })?;
166                    if receipt.generation <= 0 {
167                        return Err(anyhow::anyhow!(
168                            "GCS upload returned a nonpositive generation"
169                        )
170                        .into());
171                    }
172                    Ok(UploadReceipt {
173                        version: Some(ObjectVersion::Gcs(receipt.generation)),
174                        e_tag: (!receipt.etag.is_empty()).then_some(receipt.etag),
175                    })
176                }
177            }
178        })
179        .await
180        .context("remote backup upload deadline exceeded")
181        .map_err(UploadError::from)
182        .and_then(|result| result)
183    }

185    /// Stream an object into a caller-owned file. The caller owns private file
186    /// creation and cleanup, including cleanup if this future is cancelled.
187    pub async fn download(
188        &self,
189        bucket: &str,
190        key: &str,
191        version: Option<&ObjectVersion>,
192        destination: &mut tokio::fs::File,
193    ) -> anyhow::Result<()> {
194        // Include credentials, response body consumption, and file writes in the
195        // deadline: SDK operation timeouts alone only cover the response headers.
196        tokio::time::timeout(TRANSFER_TIMEOUT, async {
197            match self {
198                Self::S3(client) => {
199                    let version_id = match version {
200                        Some(ObjectVersion::S3(value)) => Some(value.clone()),
201                        Some(ObjectVersion::Gcs(_)) => anyhow::bail!("GCS generation cannot select an S3 object"),
202                        None => None,
203                    };
204            let mut response = client
205                .get_object()
206                .bucket(bucket)
207                .key(key)
208                .set_version_id(version_id)
209                .send()
210                .await
211                .context("S3 backup GetObject failed")?;
212            let expected_length = response.content_length();
213            let mut downloaded = 0_u64;
214            while let Some(chunk) = response
215                .body
216                .try_next()
217                .await
218                .context("failed to read S3 backup body")?
219            {
220                destination
221                    .write_all(&chunk)
222                    .await
223                    .context("failed to write downloaded backup")?;
224                downloaded += chunk.len() as u64;
225            }
226            if let Some(expected) = expected_length {
227                anyhow::ensure!(
228                    u64::try_from(expected).ok() == Some(downloaded),
229                    "incomplete S3 backup body: expected {expected} bytes, received {downloaded}"
230                );
231            }
232                }
233                Self::Gcs(client) => {
234                    // The SDK leaves '%' and '\' unescaped. Protect literal keys from
235                    // server-side percent decoding and URL backslash normalization.
236                    let escapes = key.bytes().filter(|&byte| matches!(byte, b'%' | b'\\')).count();
237                    let mut object = String::with_capacity(key.len() + 2 * escapes);
238                    for character in key.chars() {
239                        match character {
240                            '%' => object.push_str("%25"),
241                            '\\' => object.push_str("%5C"),
242                            other => object.push(other),
243                        }
244                    }
245                    let mut request = client
246                        .read_object(format!("projects/_/buckets/{bucket}"), object)
247                        .with_automatic_decompression(false);
248                    match version {
249                        Some(ObjectVersion::Gcs(generation)) => {
250                            anyhow::ensure!(*generation > 0, "GCS generation must be positive");
251                            request = request.set_generation(*generation);
252                        }
253                        Some(ObjectVersion::S3(_)) => anyhow::bail!("S3 version ID cannot select a GCS object"),
254                        None => {}
255                    }
256                    let mut response = request.send().await.context("GCS backup download failed")?;
257                    let expected = response.object().size;
258                    let mut downloaded = 0_u64;
259                    // Read through EOF: checksum validation can fail after the last data chunk.
260                    while let Some(chunk) = response.next().await.transpose()
261                        .context("failed to read GCS backup body")?
262                    {
263                        destination.write_all(&chunk).await
264                            .context("failed to write downloaded backup")?;
265                        downloaded += chunk.len() as u64;
266                    }
267                    anyhow::ensure!(
268                        u64::try_from(expected).ok() == Some(downloaded),
269                        "incomplete GCS backup body: expected {expected} bytes, received {downloaded}"
270                    );
271                }
272            }
273            destination
274                .flush()
275                .await
276                .context("failed to flush downloaded backup")?;
277            Ok::<(), anyhow::Error>(())
278        })
279        .await
280        .context("remote backup download deadline exceeded")
281        .and_then(|result| result)
282        .with_context(|| {
283            let scheme = self.provider().scheme();
284            format!("failed to download backup from {scheme}://{bucket}/{key} (revision {version:?})")
285        })
286    }

325fn timeouts() -> TimeoutConfig {
326    TimeoutConfig::builder()
327        .connect_timeout(CONNECT_TIMEOUT)
328        // The SDK read timeout covers the request through response headers, not
329        // idle reads, so it must allow the full upload transfer.
330        .read_timeout(TRANSFER_TIMEOUT)
331        .operation_timeout(TRANSFER_TIMEOUT)
332        .operation_attempt_timeout(TRANSFER_TIMEOUT)
333        .build()
334}
335
336fn client_builder(builder: Builder) -> Builder {
337    builder
338        .retry_config(RetryConfig::standard().with_max_attempts(1))
339        .timeout_config(timeouts())
340}

364    async fn put_object(
365        State(store): State<Arc<Mutex<ObjectStore>>>,
366        headers: HeaderMap,
367        body: Bytes,
368    ) -> (StatusCode, &'static str) {
369        let mut store = store.lock().await;
370        store.requests += 1;
371        if store.fail {
372            return (
373                StatusCode::SERVICE_UNAVAILABLE,
374                "<Error><Code>SlowDown</Code><Message>try later</Message></Error>",
375            );
376        }
377        if headers.get("if-none-match").and_then(|h| h.to_str().ok()) != Some("*") {
378            return (StatusCode::BAD_REQUEST, "missing no-overwrite condition");
379        }
380        if store.object.is_some() {
381            return (
382                StatusCode::PRECONDITION_FAILED,
383                "<Error><Code>PreconditionFailed</Code></Error>",
384            );
385        }
386        store.object = Some(body);
387        (StatusCode::OK, "")
388    }
389
390    async fn get_object(
391        State(store): State<Arc<Mutex<ObjectStore>>>,
392        Query(query): Query<HashMap<String, String>>,
393    ) -> (StatusCode, Bytes) {
394        let mut store = store.lock().await;
395        store.requests += 1;
396        if store.fail {
397            return (
398                StatusCode::SERVICE_UNAVAILABLE,
399                Bytes::from_static(
400                    b"<Error><Code>SlowDown</Code><Message>try later</Message></Error>",
401                ),
402            );
403        }
404        let object = match query.get("versionId") {
405            Some(version) => store.versions.get(version),
406            None => store.object.as_ref(),
407        };
408        match object {
409            Some(bytes) => (StatusCode::OK, bytes.clone()),
410            None => (
411                StatusCode::NOT_FOUND,
412                Bytes::from_static(b"<Error><Code>NoSuchKey</Code></Error>"),
413            ),
414        }
415    }

447    async fn conditional_upload_does_not_overwrite_existing_object() {
448        let (client, store, server) = server(false).await;
449        let directory = tempfile::tempdir().unwrap();
450        let archive = directory.path().join("epoch-7.tar");
451        let original = b"published archive bytes";

453        client
454            .upload("backup-bucket", "testnet/validator/epoch-7.tar", &archive)

456            .unwrap();
457        let remote_bytes = store.lock().await.object.clone().unwrap();
458        assert!(!remote_bytes.is_empty());

544    // This fixture exercises the real SDK's JSON HTTP transport without changing
545    // production endpoint or credential discovery.
546    #[derive(Default)]
547    struct GcsStore {
548        requests: Vec<(axum::http::Method, String, HashMap<String, String>)>,
549        uploaded: Option<Bytes>,
550        fail: bool,
551        interrupted: bool,
552        bad_checksum: bool,
553        session_url: String,
554    }
555
556    async fn gcs_request(
557        State(store): State<Arc<Mutex<GcsStore>>>,
558        method: axum::http::Method,
559        uri: axum::http::Uri,
560        Query(query): Query<HashMap<String, String>>,
561        body: Bytes,
562    ) -> axum::response::Response {
563        use axum::body::Body;
564        use axum::response::IntoResponse;
565
566        let mut store = store.lock().await;
567        store
568            .requests
569            .push((method.clone(), uri.path().to_owned(), query.clone()));
570        if store.fail {
571            return (
572                StatusCode::SERVICE_UNAVAILABLE,
573                axum::Json(serde_json::json!({
574                    "error": {"code": 503, "message": "service unavailable"}
575                })),
576            )
577                .into_response();
578        }
579        if method == axum::http::Method::POST {
580            if query.get("ifGenerationMatch").map(String::as_str) != Some("0") {
581                return (StatusCode::BAD_REQUEST, "missing create-only condition").into_response();
582            }
583            if query.get("uploadType").map(String::as_str) == Some("resumable") {
584                // A competing writer publishes after initiation but before completion.
585                store.uploaded = Some(Bytes::from_static(b"competing archive"));
586                return (StatusCode::OK, [("location", store.session_url.clone())]).into_response();
587            }
588            if store.uploaded.is_some() {
589                return (
590                    StatusCode::PRECONDITION_FAILED,
591                    axum::Json(serde_json::json!({
592                        "error": {"code": 412, "message": "object already exists"}
593                    })),
594                )
595                    .into_response();
596            }
597            // Require CRC in the first metadata part, before accepting the object.
598            // CRC32C("123456789") = 0xe3069283, big-endian base64 "4waSgw==".
599            let metadata = std::str::from_utf8(&body)
600                .ok()
601                .and_then(|body| body.split_once("\r\n\r\n"))
602                .and_then(|(_, rest)| rest.split_once("\r\n--"))
603                .and_then(|(metadata, _)| serde_json::from_str::<serde_json::Value>(metadata).ok());
604            if metadata.as_ref().and_then(|value| value["crc32c"].as_str()) != Some("4waSgw==") {
605                return (
606                    StatusCode::BAD_REQUEST,
607                    "missing or incorrect initial CRC32C",
608                )
609                    .into_response();
610            }
611            store.uploaded = Some(body);
612            return axum::Json(serde_json::json!({
613                "bucket": "backup-bucket",
614                "name": "testnet/validator/epoch-7.tar",
615                "generation": "123",
616                "etag": "test-etag"
617            }))
618            .into_response();
619        }
620        if method == axum::http::Method::PUT && uri.path() == "/upload-session" {
621            return (
622                StatusCode::PRECONDITION_FAILED,
623                axum::Json(serde_json::json!({
624                    "error": {"code": 412, "message": "object already exists"}
625                })),
626            )
627                .into_response();
628        }
629        // Only these objects exist; decoding percent escapes or normalizing backslashes selects a missing object.
630        if !matches!(
631            uri.path(),
632            "/storage/v1/b/backup-bucket/o/testnet%2Fvalidator%2Fepoch-7.tar"
633                | "/storage/v1/b/backup-bucket/o/literal%252Fprefix%5C..%5Cname%2F..%2Farchive.tar.asc"
634        ) {
635            return (StatusCode::NOT_FOUND, "object not found").into_response();
636        }
637
638        // CRC32C("123456789") = 0xe3069283, encoded big-endian as base64.
639        let historical = query.get("generation").map(String::as_str) == Some("123");
640        let bytes = if historical {
641            "123456789"
642        } else {
643            "latest archive"
644        };
645        let mut response = axum::http::Response::builder()
646            .header("x-goog-generation", if historical { "123" } else { "456" })
647            .header("x-goog-stored-content-length", bytes.len().to_string());
648        if historical {
649            response = response.header(
650                "x-goog-hash",
651                if store.bad_checksum {
652                    "crc32c=AAAAAA=="
653                } else {
654                    "crc32c=4waSgw=="
655                },
656            );
657        }
658        let interrupted = store.interrupted;
659        let body = Body::from_stream(futures::stream::unfold(0, move |chunk| async move {
660            match chunk {
661                0 => Some((Ok(Bytes::copy_from_slice(&bytes.as_bytes()[..3])), 1)),
662                1 if interrupted => {
663                    // Flush the first chunk before the transport is interrupted.
664                    tokio::time::sleep(Duration::from_millis(20)).await;
665                    Some((Err(std::io::Error::other("interrupted body")), 2))
666                }
667                1 => Some((Ok(Bytes::copy_from_slice(&bytes.as_bytes()[3..])), 2)),
668                _ => None,
669            }
670        }));
671        response.body(body).unwrap()
672    }

674    async fn gcs_server(
675        resumable: bool,
676    ) -> (
677        BackupRemoteClient,
678        Arc<Mutex<GcsStore>>,
679        tokio::task::JoinHandle<()>,
680    ) {
681        let store = Arc::new(Mutex::new(GcsStore::default()));
682        let router = Router::new()
683            .fallback(gcs_request)
684            .with_state(store.clone());
685        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
686        let endpoint = format!("http://{}", listener.local_addr().unwrap());
687        store.lock().await.session_url = format!("{endpoint}/upload-session");
688        let task = tokio::spawn(async move {
689            axum::serve(listener, router).await.unwrap();
690        });
691        let client = if resumable {
692            BackupRemoteClient::Gcs(Box::new(
693                BackupRemoteClient::gcs_test_builder(&endpoint)
694                    .with_resumable_upload_threshold(0_usize)
695                    .build()
696                    .await
697                    .unwrap(),
698            ))
699        } else {
700            BackupRemoteClient::for_gcs_test_endpoint(&endpoint).await
701        };
702        (client, store, task)
703    }
