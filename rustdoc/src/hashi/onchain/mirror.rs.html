4//! The lossless object mirror — the production state's update path.
5//!
6//! The transport is transaction-granular and filtered:
7//! `SubscribeTransactions` with an `affected_object == hashi root`
8//! filter delivers exactly the Hashi-touching transactions, each
9//! rendered with its object set, checkpoint, and position within the
10//! checkpoint; `ListTransactions` with the same filter replays history
11//! from the watermark cursor, so bootstrap and reconnects are gap-free
12//! without rescraping. Transactions arrive in chain order, and the
13//! mirror ratchets over `(checkpoint, transaction_index)` so a
14//! transaction delivered by both replay and the live stream applies
15//! exactly once.
16//!
17//! Each applied transaction updates `OnchainState`'s `types::Hashi`
18//! directly through the route/apply layers; side effects that used to
19//! ride the event path come back as transition-derived
20//! [`apply::Effect`]s and are acted on here: notifications, the local
21//! withdrawal limiter, package-version reconciliation, and the
22//! withdrawal duration metrics.
23//!
24//! The transport requires a server that renders per-transaction object
25//! sets on the filtered APIs and populates `Watermark.checkpoint` on
26//! subscription frames (Sui v1.76 with the object-set and watermark
27//! fixes). Older servers are not supported: the stream fails and the
28//! watcher keeps reconnecting rather than degrading to a lossy mode.
29

57/// Reconnect if the stream goes silent this long. The filtered stream
58/// emits watermark or cursor progress frames periodically even when
59/// nothing matches the filter, so silence genuinely means a broken
60/// stream.
61const STREAM_STALL_TIMEOUT: Duration = Duration::from_secs(120);
62
63const RECONNECT_DELAY: Duration = Duration::from_secs(5);
64
65/// Read mask for `ExecutedTransaction` frames on the transaction
66/// stream and the replay list.
67const TX_READ_MASK: [&str; 7] = [
68    "digest",
69    "checkpoint",
70    "transaction_index",
71    "timestamp",
72    "effects.status",
73    "effects.changed_objects",
74    "objects.objects.bcs",
75];

77/// The mirror's transport bookkeeping. The mirrored values themselves
78/// live in `OnchainState`; this is everything the loop needs alongside
79/// them: the routing table, the per-object index, the delivery
80/// ratchet, and the replay cursor.
81struct Mirror {
82    routing: route::RoutingTable,
83    index: route::ObjectIndex,
84    /// Checkpoint through which the mirror is complete: every Hashi
85    /// transaction at or below it has been applied.
86    watermark_checkpoint: u64,
87    /// The last applied transaction's `(checkpoint, transaction_index)`.
88    /// Transactions are delivered in chain order on every source, so
89    /// anything at or below this position has already been applied.
90    applied_position: (u64, u64),
91    /// Opaque ledger cursor for resuming the transaction replay list;
92    /// `None` until the first watermark arrives (replay then starts
93    /// from `watermark_checkpoint`).
94    watermark_cursor: Option<bytes::Bytes>,
95}
96
97impl Mirror {
98    fn from_seed(seed: route::MirrorSeed) -> Self {
99        Self {
100            routing: seed.routing,
101            index: seed.index,
102            watermark_checkpoint: seed.floor,
103            // The scrape reflects everything through the floor; skip any
104            // replayed transaction at or before it.
105            applied_position: (seed.floor, u64::MAX),
106            watermark_cursor: None,
107        }
108    }
109}

111#[tracing::instrument(name = "state_watcher", skip_all)]
112pub(super) async fn run(
113    sui_rpc_url: String,
114    state: OnchainState,
115    seed: route::MirrorSeed,
116    metrics: Option<std::sync::Arc<Metrics>>,
117) {
118    // The boot scrape seeded both the mirrored state and this
119    // bookkeeping; `None` after a failed replay forces a re-bootstrap
120    // from a fresh scrape.
121    let mut mirror = Some(Mirror::from_seed(seed));
122    loop {
123        if let Err(e) =
124            run_transactions(&sui_rpc_url, &state, &mut mirror, metrics.as_deref()).await
125        {
126            tracing::warn!("state watcher stream ended: {e:#}; reconnecting");
127        }
128        tokio::time::sleep(RECONNECT_DELAY).await;
129    }
130}

132fn build_client(sui_rpc_url: &str, state: &OnchainState) -> Result<Client> {
133    // A fresh client per attempt: re-subscribing on the same channel can
134    // reuse a wedged h2 connection — the one whose stream just stalled —
135    // and silently hang again. The shared deadline layer bounds the
136    // request/handshake awaits (subscribe, each replay list page, the
137    // re-bootstrap scrape) that the stall timeout does not cover, while
138    // sparing established subscription streams.
139    let mut client = crate::sui_rpc_client::new_sui_rpc_client(sui_rpc_url)?;
140    if let Some(limit) = state.grpc_max_decoding_message_size() {
141        client = client.with_max_decoding_message_size(limit);
142    }
143    Ok(client)
144}

146/// Re-bootstrap the mirror from a fresh scrape if a failed replay tore
147/// it down. The scraped state is installed wholesale, so the local
148/// limiter may have missed fully-signed transitions during the gap —
149/// ask the reconcile loop to re-align it.
150async fn ensure_bootstrapped(
151    client: &Client,
152    state: &OnchainState,
153    mirror: &mut Option<Mirror>,
154) -> Result<()> {
155    if mirror.is_some() {
156        return Ok(());
157    }
158    // Always `Full` — this replaces the authoritative snapshot, and only
159    // a full scrape yields a seed to rebuild the mirror from.
160    let (_, hashi, seed) = super::scrape_hashi(
161        client.clone(),
162        state.hashi_id(),
163        state.package_id_original(),
164        super::ScrapeScope::Full,
165    )
166    .await?;
167    let seed = seed.context("a full scrape must produce a mirror seed")?;
168    tracing::info!(
169        floor = seed.floor,
170        objects = seed.index.len(),
171        "object mirror re-bootstrapped from scrape"
172    );
173    // An upgrade that landed during the gap produces no cap diff (the
174    // scrape installs state wholesale), so no `PackageUpgraded` effect
175    // fires; the scraped cap itself carries the latest version pair.
176    let latest_package = hashi
177        .config
178        .upgrade_cap
179        .as_ref()
180        .map(|cap| (cap.version, cap.package));
181    state.replace_hashi_state(hashi);
182    if let Some((version, package)) = latest_package {
183        state.add_package_version(version, package);
184    }
185    // A reset, not an advance: the scrape replaced the state wholesale,
186    // so the watermark must say exactly what the scrape covers — even
187    // if that is below where the mirror was before the replay failure.
188    state.reset_state_watermark(seed.floor);
189    *mirror = Some(Mirror::from_seed(seed));
190    state.request_limiter_reconcile();
191    if let Some(metrics) = state.metrics() {
192        metrics.watcher_rebootstrap_total.inc();
193    }
194    Ok(())
195}

197fn require_mirror(mirror: &mut Option<Mirror>) -> Result<&mut Mirror> {
198    mirror
199        .as_mut()
200        .ok_or_else(|| anyhow!("object mirror missing while frames are flowing"))
201}

203async fn run_transactions(
204    sui_rpc_url: &str,
205    state: &OnchainState,
206    mirror: &mut Option<Mirror>,
207    metrics: Option<&Metrics>,
208) -> Result<()> {
209    let mut client = build_client(sui_rpc_url, state)?;
210    let filter = TransactionFilter::matching(tx_filter::affected_object(state.hashi_id()));
211
212    let mut subscribe_request = SubscribeTransactionsRequest::default();
213    subscribe_request.read_mask = Some(FieldMask::from_paths(TX_READ_MASK));
214    subscribe_request.filter = Some(filter.clone());
215    let mut subscription = client
216        .subscription_client()
217        .subscribe_transactions(subscribe_request)
218        .await
219        .context("subscribe_transactions failed")?
220        .into_inner();
221
222    // Wait for the first frame: it proves the stream is live, and its
223    // watermark checkpoint is the subscription's starting coverage
224    // point — everything at or before it must come from replay, and
225    // the subscription delivers everything after. That makes it
226    // exactly the replay target, and — unlike a clock-stream sample —
227    // it speaks for this subscription itself, so it cannot be frozen
228    // by an outage that severed both streams. The position ratchet
229    // absorbs the overlap between replay and stream.
230    let first = tokio::time::timeout(STREAM_STALL_TIMEOUT, subscription.next())
231        .await
232        .context("timed out waiting for the first transaction frame")?
233        .context("transaction stream closed before the first frame")??;
234    let target = first
235        .watermark
236        .as_ref()
237        .and_then(|watermark| watermark.checkpoint)
238        .context("the first transaction frame carried no watermark checkpoint")?;
239    ensure_bootstrapped(&client, state, mirror).await?;
240
241    if let Err(e) = replay_transactions(&mut client, &filter, state, mirror, metrics, target).await
242    {
243        // A failed replay leaves an unknowable gap; rebuild.
244        *mirror = None;
245        return Err(e).context("transaction replay failed; object mirror reset");
246    }
247    tracing::info!(
248        target,
249        "object mirror caught up; consuming the filtered transaction stream"
250    );
251
252    handle_tx_frame(
253        state,
254        require_mirror(mirror)?,
255        first.transaction.as_ref(),
256        first.watermark.as_ref(),
257        metrics,
258    )?;
259    loop {
260        let item = tokio::time::timeout(STREAM_STALL_TIMEOUT, subscription.next())
261            .await
262            .context("transaction stream stalled")?;
263        let response = item.context("transaction stream closed")??;
264        handle_tx_frame(
265            state,
266            require_mirror(mirror)?,
267            response.transaction.as_ref(),
268            response.watermark.as_ref(),
269            metrics,
270        )?;
271        // Coverage advances from the server's own watermark claims:
272        // progress frames carry `Watermark.checkpoint` periodically
273        // (observed every 25 checkpoints on quiet filters) — plus each
274        // applied transaction's own checkpoint.
275    }
276}

278/// Replay `ListTransactions` (same filter and mask as the
279/// subscription) from the mirror's watermark until coverage reaches
280/// `target` — the live stream's starting coverage point.
281async fn replay_transactions(
282    client: &mut Client,
283    filter: &TransactionFilter,
284    state: &OnchainState,
285    mirror: &mut Option<Mirror>,
286    metrics: Option<&Metrics>,
287    target: u64,
288) -> Result<()> {
289    loop {
290        let (cursor, floor) = {
291            let mirror = require_mirror(mirror)?;
292            (mirror.watermark_cursor.clone(), mirror.watermark_checkpoint)
293        };
294        if floor >= target {
295            return Ok(());
296        }
297        let mut request = ListTransactionsRequest::default();
298        request.read_mask = Some(FieldMask::from_paths(TX_READ_MASK));
299        // The opaque cursor is authoritative once we have one; the
300        // checkpoint floor covers the first pass after bootstrap.
301        request.start_checkpoint = cursor.is_none().then_some(floor);
302        request.filter = Some(filter.clone());
303        request.options = cursor.map(|after| {
304            let mut options = QueryOptions::default();
305            options.after = Some(after);
306            options
307        });
308        let response = client
309            .ledger_client()
310            .list_transactions(request)
311            .await
312            .context("list_transactions failed")?;
313        // The response header carries the server's indexed checkpoint
314        // height at request time — the coverage proof when watermark
315        // checkpoints are unset.
316        let indexed_height = response.checkpoint_height();
317        let mut stream = response.into_inner();
318
319        let mut end_reason = None;
320        while let Some(item) = stream.next().await {
321            let response = item.context("transaction replay stream errored")?;
322            handle_tx_frame(
323                state,
324                require_mirror(mirror)?,
325                response.transaction.as_ref(),
326                response.watermark.as_ref(),
327                metrics,
328            )?;
329            if let Some(end) = response.end.as_ref() {
330                end_reason = end.reason.and_then(|r| QueryEndReason::try_from(r).ok());
331            }
332        }
333        let reached = require_mirror(mirror)?.watermark_checkpoint;
334        match end_reason {
335            Some(QueryEndReason::LedgerTip) => {
336                // LedgerTip means every matching transaction through
337                // the indexed tip was delivered.
338                if reached >= target {
339                    return Ok(());
340                }
341                if let Some(height) = indexed_height
342                    && height >= target
343                {
344                    advance_watermark(state, require_mirror(mirror)?, height);
345                    return Ok(());
346                }
347                // The list index trails the live stream; give it a beat.
348                tracing::debug!(target, ?indexed_height, "replay short of target; retrying");
349                tokio::time::sleep(Duration::from_secs(1)).await;
350            }
351            // Item or scan limits: resume from the advanced cursor.
352            Some(_) => {}
353            None => anyhow::bail!("transaction replay stream ended without a QueryEnd frame"),
354        }
355    }
356}

358fn advance_watermark(state: &OnchainState, mirror: &mut Mirror, covered: u64) {
359    mirror.watermark_checkpoint = mirror.watermark_checkpoint.max(covered);
360    state.advance_state_watermark(mirror.watermark_checkpoint);
361}

363/// Handle one transaction-stream frame: apply the transaction (if the
364/// frame carries one and the position ratchet hasn't passed it), then
365/// fold in the watermark.
366fn handle_tx_frame(
367    state: &OnchainState,
368    mirror: &mut Mirror,
369    tx: Option<&proto::ExecutedTransaction>,
370    frame_watermark: Option<&proto::Watermark>,
371    metrics: Option<&Metrics>,
372) -> Result<()> {
373    if let Some(tx) = tx {
374        apply_tx_frame(state, mirror, tx, metrics)?;
375    }
376    if let Some(watermark) = frame_watermark {
377        if let Some(cursor) = watermark.cursor.as_ref() {
378            mirror.watermark_cursor = Some(cursor.clone());
379        }
380        if let Some(checkpoint) = watermark.checkpoint {
381            if tx.is_none() {
382                tracing::debug!(checkpoint, "transaction stream coverage claim");
383            }
384            advance_watermark(state, mirror, checkpoint);
385        }
386    }
387    Ok(())
388}

390/// Apply one filtered-stream transaction to the mirror, ratcheting on
391/// its `(checkpoint, transaction_index)` position, then act on the
392/// transition-derived effects.
393fn apply_tx_frame(
394    state: &OnchainState,
395    mirror: &mut Mirror,
396    tx: &proto::ExecutedTransaction,
397    metrics: Option<&Metrics>,
398) -> Result<()> {
399    let Some(view) = apply::TxView::from_proto(tx)? else {
400        return Ok(());
401    };
402    if view.position() <= mirror.applied_position {
403        tracing::debug!(position = ?view.position(), "skipping an already-applied transaction");
404        return Ok(());
405    }
406    let out = {
407        let mut guard = state.state_mut();
408        let state = &mut *guard;
409        apply::apply_transaction(
410            &mut state.hashi,
411            &mut state.package_versions,
412            &mut mirror.routing,
413            &mut mirror.index,
414            &view,
415        )
416    };
417    for (id, type_) in &out.unrouted {
418        tracing::warn!(object = %id, r#type = %type_, digest = tx.digest(),
419            "object mirror could not route a changed object");
420        if let Some(metrics) = metrics {
421            metrics.watcher_unrouted_objects_total.inc();
422        }
423    }
424    mirror.applied_position = view.position();
425    // In-order delivery proves everything before this transaction's
426    // checkpoint has been delivered, but not that its own checkpoint is
427    // complete — a later Hashi transaction of the same checkpoint may
428    // still be in flight, and claiming the full checkpoint here would
429    // wake `wait_until_checkpoint` waiters (the GC scan gate, the
430    // rejoin loop) before it applies. The transaction itself may only
431    // claim its predecessor; the frame's own watermark, folded by the
432    // caller right after this, claims the full checkpoint exactly when
433    // the server has delivered all of it.
434    advance_watermark(state, mirror, view.checkpoint.saturating_sub(1));
435    if let Some(metrics) = metrics {
436        metrics.watcher_applied_txns_total.inc();
437    }
438    handle_effects(state, view.timestamp_ms, out.effects);
439    Ok(())
440}

442/// Act on the side effects one applied transaction derived from state
443/// transitions. Replay re-derives the same effects in order, and the
444/// apply layer's version guards make each transition fire exactly once
445/// across replay/live overlap.
446fn handle_effects(state: &OnchainState, timestamp_ms: u64, effects: Vec<apply::Effect>) {
447    for effect in effects {
448        tracing::debug!(?effect, "object mirror effect");
449        match effect {
450            apply::Effect::ValidatorInfoUpdated(validator) => {
451                state.notify(Notification::ValidatorInfoUpdated(validator));
452            }
453            apply::Effect::ReconfigStarted(epoch) => {
454                state.notify(Notification::StartReconfig(epoch));
455            }
456            apply::Effect::PackageUpgraded { package, version } => {
457                // The apply layer already extended the version map
458                // under the state guard; this is the operator signal.
459                tracing::info!(%package, version, "package upgraded; version map extended");
460            }
461            apply::Effect::WithdrawalTxnFullySigned(txn) => {
462                withdrawal_txn_fully_signed(state, timestamp_ms, &txn);
463            }
464            apply::Effect::WithdrawalTxnRemoved(txn) => {
465                withdrawal_txn_removed(state, timestamp_ms, &txn);
466            }
467        }
468    }
469}

471/// A withdrawal transaction became fully signed (2-of-2 witness
472/// complete): advance the local limiter and observe pick-to-sign.
473///
474/// The advance uses the applying transaction's checkpoint timestamp
475/// (~sign time) rather than `txn.created_timestamp_ms` (creation time)
476/// to stay in lockstep with the guardian's `last_updated_at`. The
477/// apply layer emits this transition exactly once per signing (version
478/// guards skip replayed and duplicated frames), so no additional
479/// idempotency gate is needed.
480fn withdrawal_txn_fully_signed(
481    state: &OnchainState,
482    timestamp_ms: u64,
483    txn: &types::WithdrawalTransaction,
484) {
485    tracing::info!(withdrawal_txn_id = %txn.id, "Withdrawal signatures stored on-chain");
486    state
487        .state_mut()
488        .withdrawal_signed_at_ms
489        .insert(txn.id, timestamp_ms);
490    if let Some(metrics) = state.metrics() {
491        let pick_to_sign = timestamp_ms.saturating_sub(txn.created_timestamp_ms);
492        metrics
493            .withdrawal_duration_seconds
494            .with_label_values(&["pick_to_sign"])
495            .observe(Duration::from_millis(pick_to_sign).as_secs_f64());
496    }
497
498    let amount_sats = withdrawal_limiter_consumption_amount(txn);
499    let timestamp_secs = timestamp_ms / 1000;
500    let Some(limiter) = state.local_limiter() else {
501        if let Some(metrics) = state.metrics() {
502            metrics
503                .guardian_limiter_apply_total
504                .with_label_values(&[crate::metrics::GUARDIAN_LIMITER_OUTCOME_NO_LIMITER])
505                .inc();
506        }
507        return;
508    };
509    let seq = limiter.next_seq();
510    let result = limiter.apply_consume(seq, timestamp_secs, amount_sats);
511    if let Some(metrics) = state.metrics() {
512        metrics.record_limiter_apply(&result);
513    }
514    match &result {
515        Ok(()) => {
516            if let Some(metrics) = state.metrics() {
517                metrics.guardian_limiter_anchor_events_total.inc();
518                let view = limiter.view();
519                metrics.record_limiter_state(&view.state, &view.config);
520            }
521            tracing::info!(
522                seq,
523                amount_sats,
524                timestamp_secs,
525                withdrawal_txn_id = %txn.id,
526                "Local limiter advanced from the on-chain fully-signed transition",
527            );
528        }
529        Err(e) => {
530            if let Some(metrics) = state.metrics() {
531                metrics.guardian_limiter_drifted.set(1);
532            }
533            tracing::error!(
534                ?e,
535                seq,
536                withdrawal_txn_id = %txn.id,
537                "Local limiter apply_consume failed; node is now drifted from guardian"
538            );
539        }
540    }
541}

543/// A withdrawal transaction left the in-flight bag (confirmed and
544/// moved to the historical record, or deleted): observe the
545/// sign-to-confirm and total durations.
546fn withdrawal_txn_removed(
547    state: &OnchainState,
548    timestamp_ms: u64,
549    txn: &types::WithdrawalTransaction,
550) {
551    tracing::info!(withdrawal_txn_id = %txn.id, "Withdrawal transaction left the in-flight bag");
552    let signed_at = state.state_mut().withdrawal_signed_at_ms.remove(&txn.id);
553    let Some(metrics) = state.metrics() else {
554        return;
555    };
556    if let Some(signed_at) = signed_at {
557        let sign_to_confirm = timestamp_ms.saturating_sub(signed_at);
558        metrics
559            .withdrawal_duration_seconds
560            .with_label_values(&["sign_to_confirm"])
561            .observe(Duration::from_millis(sign_to_confirm).as_secs_f64());
562    }
563    let total = timestamp_ms.saturating_sub(txn.created_timestamp_ms);
564    metrics
565        .withdrawal_duration_seconds
566        .with_label_values(&["total"])
567        .observe(Duration::from_millis(total).as_secs_f64());
568}
