4//! Ownership-based routing for the object-driven mirror.
5//!
6//! Every object the watcher receives is routed to a logical slot by
7//! walking its ownership: a dynamic field's owner is a container UID
8//! embedded in the Hashi root or `BitcoinState`, and a dynamic object
9//! field's owner is a wrapper `Field` whose own owner is a container
10//! UID. The [`RoutingTable`] holds both mappings; the [`ObjectIndex`]
11//! remembers, per applied object id, its last-applied version (for
12//! idempotent replay) and what the object means (so deletions and moves
13//! can retire the right mirror entry without needing the object's
14//! pre-deletion contents).
15

25/// A logical position in the mirrored state. Routing an object to a
26/// slot decides which `types::Hashi` map it lands in — or that it is
27/// known and deliberately not mirrored.
28#[derive(Clone, Copy, Debug, PartialEq, Eq)]
29pub(super) enum Slot {
30    // Plain-Bag containers whose Field values are mirrored.
31    Members,
32    Committees,
33    UtxoRecords,
34    SpentUtxos,
35    // ObjectBag containers whose child objects are mirrored.
36    DepositRequests,
37    WithdrawalRequests,
38    WithdrawalTxns,
39    Treasury,
40    ProposalsActive,
41    ProposalsExecuted,
42    // Known containers that are deliberately not mirrored. Routing an
43    // object here is not an error; it just doesn't update the mirror.
44    DepositProcessed,
45    WithdrawalProcessed,
46    ConfirmedTxns,
47    Tob,
48    /// The `LinkedTable` inside a `tob` bag entry (`EpochCertsV1.certs`);
49    /// its children are dealer submission nodes.
50    TobCerts,
51    /// The `BitcoinState.user_requests` table of per-user request bags.
52    UserRequests,
53    /// One per-user `Bag` stored as a `user_requests` value.
54    UserRequestBag,
55}
56
57/// Container UIDs and DOF wrappers, resolved to slots.
58#[derive(Debug)]
59pub(super) struct RoutingTable {
60    hashi_id: Address,
61    bitcoin_state_field_id: Address,
62    /// Container UID (a `Bag`/`ObjectBag`/`Table` id embedded in the
63    /// root or `BitcoinState`, or an interior container learned from a
64    /// mirrored write) -> the slot its children belong to.
65    containers: BTreeMap<Address, Slot>,
66    /// DOF wrapper field id -> the container UID that owns the wrapper.
67    /// A DOF value's owner is its wrapper, so this is the extra hop
68    /// needed to route values. Maintained from wrapper writes/deletes
69    /// in the stream and populated wholesale by the scrape.
70    wrapper_parents: BTreeMap<Address, Address>,
71}
72
73impl RoutingTable {
74    pub(super) fn new(hashi_id: Address, bitcoin_state_field_id: Address) -> Self {
75        Self {
76            hashi_id,
77            bitcoin_state_field_id,
78            containers: BTreeMap::new(),
79            wrapper_parents: BTreeMap::new(),
80        }
81    }
82
83    pub(super) fn hashi_id(&self) -> Address {
84        self.hashi_id
85    }
86
87    pub(super) fn bitcoin_state_field_id(&self) -> Address {
88        self.bitcoin_state_field_id
89    }
90
91    /// (Re)register the container UIDs embedded in the Hashi root.
92    /// Inner UIDs are stable for an object's lifetime, so after the
93    /// first call this is a no-op refresh.
94    pub(super) fn set_root_containers(&mut self, root: &move_types::Hashi) {
95        self.containers
96            .insert(root.committees.members.id, Slot::Members);
97        self.containers
98            .insert(root.committees.committees.id, Slot::Committees);
99        self.containers
100            .insert(root.treasury.objects.id, Slot::Treasury);
101        self.containers
102            .insert(root.proposals.active.id, Slot::ProposalsActive);
103        self.containers
104            .insert(root.proposals.executed.id, Slot::ProposalsExecuted);
105        self.containers.insert(root.tob.id, Slot::Tob);
106    }
107
108    /// (Re)register the container UIDs embedded in `BitcoinState`.
109    pub(super) fn set_bitcoin_state_containers(&mut self, state: &move_types::BitcoinState) {
110        self.containers
111            .insert(state.deposit_queue.requests.id, Slot::DepositRequests);
112        self.containers
113            .insert(state.deposit_queue.processed.id, Slot::DepositProcessed);
114        self.containers
115            .insert(state.withdrawal_queue.requests.id, Slot::WithdrawalRequests);
116        self.containers.insert(
117            state.withdrawal_queue.processed.id,
118            Slot::WithdrawalProcessed,
119        );
120        self.containers.insert(
121            state.withdrawal_queue.withdrawal_txns.id,
122            Slot::WithdrawalTxns,
123        );
124        self.containers.insert(
125            state.withdrawal_queue.confirmed_txns.id,
126            Slot::ConfirmedTxns,
127        );
128        self.containers
129            .insert(state.utxo_pool.utxo_records.id, Slot::UtxoRecords);
130        self.containers
131            .insert(state.utxo_pool.spent_utxos.id, Slot::SpentUtxos);
132        self.containers
133            .insert(state.user_requests.id, Slot::UserRequests);
134    }
135
136    /// Register an interior container discovered while applying a
137    /// routed write (e.g. the `LinkedTable` inside a tob entry), so its
138    /// children route to a known-ignored slot instead of tripping the
139    /// unrouted counter.
140    pub(super) fn register_interior(&mut self, id: Address, slot: Slot) {
141        self.containers.insert(id, slot);
142    }
143
144    pub(super) fn register_wrapper(&mut self, wrapper: Address, container: Address) {
145        self.wrapper_parents.insert(wrapper, container);
146    }
147
148    pub(super) fn remove_wrapper(&mut self, wrapper: &Address) {
149        self.wrapper_parents.remove(wrapper);
150    }
151
152    pub(super) fn slot_of_container(&self, id: &Address) -> Option<Slot> {
153        self.containers.get(id).copied()
154    }
155
156    /// Resolve the owner UID of an object-owned object to a slot:
157    /// either the owner is a container UID directly (plain dynamic
158    /// field), or it is a DOF wrapper one hop below a container.
159    pub(super) fn resolve_owner(&self, owner: &Address) -> Option<Slot> {
160        if let Some(slot) = self.containers.get(owner) {
161            return Some(*slot);
162        }
163        let container = self.wrapper_parents.get(owner)?;
164        self.containers.get(container).copied()
165    }
166}

168/// What an applied object id means to the mirror. Carried per id so a
169/// later deletion or cross-bag move can retire the right entry without
170/// the object's pre-state contents.
171#[derive(Clone, Debug, PartialEq, Eq)]
172pub(super) enum TrackedKind {
173    HashiRoot,
174    BitcoinStateField,
175    /// A DOF wrapper field. Retiring it prunes `wrapper_parents`.
176    DofWrapper {
177        container: Address,
178    },
179    Member(Address),
180    Committee(u64),
181    CommitteeHandoff(u64),
182    UtxoRecord(UtxoId),
183    SpentUtxo(UtxoId),
184    DepositRequest(Address),
185    WithdrawalRequest(Address),
186    WithdrawalTxn(Address),
187    TreasuryCap(TypeTag),
188    MetadataCap(TypeTag),
189    Proposal {
190        executed: bool,
191        id: Address,
192    },
193    /// Routed to a known-unmirrored slot; tracked only for versioning.
194    Ignored,
195}
196
197#[derive(Debug)]
198pub(super) struct TrackedObject {
199    pub version: u64,
200    pub kind: TrackedKind,
201}
202
203/// Last-applied version and meaning per object id.
204#[derive(Debug, Default)]
205pub(super) struct ObjectIndex {
206    entries: BTreeMap<Address, TrackedObject>,
207}
208
209impl ObjectIndex {
210    pub(super) fn new() -> Self {
211        Self::default()
212    }
213
214    pub(super) fn get(&self, id: &Address) -> Option<&TrackedObject> {
215        self.entries.get(id)
216    }
217
218    /// True when a write at `version` is stale — i.e. the mirror has
219    /// already applied this version or a newer one. Replayed and
220    /// duplicated frames hit this.
221    pub(super) fn is_stale_write(&self, id: &Address, version: u64) -> bool {
222        self.entries
223            .get(id)
224            .is_some_and(|entry| entry.version >= version)
225    }
226
227    pub(super) fn record(&mut self, id: Address, version: u64, kind: TrackedKind) {
228        self.entries.insert(id, TrackedObject { version, kind });
229    }
230
231    pub(super) fn remove(&mut self, id: &Address) -> Option<TrackedObject> {
232        self.entries.remove(id)
233    }
234
235    pub(super) fn len(&self) -> usize {
236        self.entries.len()
237    }
238}

240/// Everything a bootstrap scrape learns beyond the mirrored values:
241/// the routing table, the per-object index, and the minimum checkpoint
242/// height any scrape response was served at. Replaying the filtered
243/// transaction stream from `floor` (in order, through the tip)
244/// converges the mirror even where a page was served behind another —
245/// index versions make re-applied writes idempotent.
246#[derive(Debug)]
247pub(super) struct MirrorSeed {
248    pub routing: RoutingTable,
249    pub index: ObjectIndex,
250    pub floor: u64,
251}
252
253impl MirrorSeed {
254    pub(super) fn new(hashi_id: Address, bitcoin_state_field_id: Address) -> Self {
255        Self {
256            routing: RoutingTable::new(hashi_id, bitcoin_state_field_id),
257            index: ObjectIndex::new(),
258            floor: u64::MAX,
259        }
260    }
261
262    /// Fold one response's serving height into the replay floor.
263    pub(super) fn observe_height(&mut self, height: u64) {
264        self.floor = self.floor.min(height);
265    }
266
267    /// Merge the per-container seed entries a scrape helper produced.
268    pub(super) fn absorb(&mut self, seed: ContainerSeed) {
269        self.observe_height(seed.height);
270        for (id, version, kind) in seed.entries {
271            if let TrackedKind::DofWrapper { container } = &kind {
272                self.routing.register_wrapper(id, *container);
273            }
274            self.index.record(id, version, kind);
275        }
276        for (id, slot) in seed.interior {
277            self.routing.register_interior(id, slot);
278        }
279    }
280}
281
282/// One scrape helper's contribution to the [`MirrorSeed`].
283#[derive(Debug)]
284pub(super) struct ContainerSeed {
285    /// The minimum checkpoint height across this helper's pages.
286    pub height: u64,
287    /// (object id, version, meaning) triples for the index.
288    pub entries: Vec<(Address, u64, TrackedKind)>,
289    /// Interior containers discovered inside mirrored values.
290    pub interior: Vec<(Address, Slot)>,
291}
292
293impl Default for ContainerSeed {
294    fn default() -> Self {
295        Self {
296            // The neutral element for the min-fold: a helper that read
297            // no pages must not drag the replay floor down.
298            height: u64::MAX,
299            entries: Vec::new(),
300            interior: Vec::new(),
301        }
302    }
303}
304
305impl ContainerSeed {
306    pub(super) fn merge(&mut self, other: ContainerSeed) {
307        self.height = self.height.min(other.height);
308        self.entries.extend(other.entries);
309        self.interior.extend(other.interior);
310    }
311}

313#[cfg(test)]
314mod tests {
315    use super::*;
316
317    fn addr(byte: u8) -> Address {
318        Address::from_bytes([byte; 32]).unwrap()
319    }
320
321    fn bag(id: Address) -> move_types::Bag {
322        move_types::Bag { id, size: 0 }
323    }
324
325    fn routing_with_bitcoin_state() -> (RoutingTable, move_types::BitcoinState) {
326        let state = move_types::BitcoinState {
327            id: addr(0x10),
328            deposit_queue: move_types::DepositRequestQueue {
329                requests: bag(addr(0x11)),
330                processed: bag(addr(0x12)),
331            },
332            withdrawal_queue: move_types::WithdrawalRequestQueue {
333                requests: bag(addr(0x13)),
334                processed: bag(addr(0x14)),
335                withdrawal_txns: bag(addr(0x15)),
336                confirmed_txns: bag(addr(0x16)),
337            },
338            utxo_pool: move_types::UtxoPool {
339                utxo_records: bag(addr(0x17)),
340                spent_utxos: bag(addr(0x18)),
341            },
342            user_requests: move_types::Table {
343                id: addr(0x19),
344                size: 0,
345            },
346        };
347        let mut routing = RoutingTable::new(addr(0x01), addr(0x02));
348        routing.set_bitcoin_state_containers(&state);
349        (routing, state)
350    }
351
352    #[test]
353    fn resolve_owner_routes_container_uids_directly() {
354        let (routing, _) = routing_with_bitcoin_state();
355        assert_eq!(routing.resolve_owner(&addr(0x17)), Some(Slot::UtxoRecords));
356        assert_eq!(
357            routing.resolve_owner(&addr(0x14)),
358            Some(Slot::WithdrawalProcessed)
359        );
360        assert_eq!(routing.resolve_owner(&addr(0x77)), None);
361    }
362
363    #[test]
364    fn resolve_owner_routes_dof_values_through_their_wrapper() {
365        let (mut routing, _) = routing_with_bitcoin_state();
366        let wrapper = addr(0x20);
367        routing.register_wrapper(wrapper, addr(0x15));
368        assert_eq!(routing.resolve_owner(&wrapper), Some(Slot::WithdrawalTxns));
369
370        routing.remove_wrapper(&wrapper);
371        assert_eq!(routing.resolve_owner(&wrapper), None);
372    }
373
374    #[test]
375    fn wrapper_under_unknown_container_does_not_resolve() {
376        let (mut routing, _) = routing_with_bitcoin_state();
377        let wrapper = addr(0x21);
378        routing.register_wrapper(wrapper, addr(0x99));
379        assert_eq!(routing.resolve_owner(&wrapper), None);
380    }
381
382    #[test]
383    fn stale_write_detection_is_version_monotonic() {
384        let mut index = ObjectIndex::new();
385        let id = addr(0x30);
386        assert!(!index.is_stale_write(&id, 5));
387
388        index.record(id, 5, TrackedKind::Ignored);
389        assert!(index.is_stale_write(&id, 4));
390        assert!(index.is_stale_write(&id, 5));
391        assert!(!index.is_stale_write(&id, 6));
392    }
