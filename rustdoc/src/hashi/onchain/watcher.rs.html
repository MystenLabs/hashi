4//! The watcher's two streams.
5//!
6//! 1. The clock stream (here): an unfiltered checkpoint subscription
7//!    with a minimal scalar mask. It keeps [`CheckpointInfo`] — the
8//!    leader tick, timestamps, and Sui epoch-change detection — fresh
9//!    regardless of Hashi activity, and costs a few scalar fields per
10//!    checkpoint.
11//! 2. The state stream ([`super::mirror`]): the filtered transaction
12//!    stream that applies every Hashi-touching transaction's changed
13//!    objects to the mirror.
14
15use std::sync::Arc;

17use futures::StreamExt;

19use sui_rpc::field::FieldMaskUtil;
20use sui_rpc::proto::proto_to_timestamp_ms;

22use sui_rpc::proto::sui::rpc::v2::SubscribeCheckpointsRequest;

24use crate::metrics::Metrics;
25use crate::onchain::CheckpointInfo;

27use crate::onchain::OnchainState;

29/// Reconnect if the checkpoint stream goes silent this long. A half-open h2
30/// stream yields neither an item nor an error, so an unbounded read hangs — the
31/// SDK's keepalive only trips on a fully-dead connection, not a live one whose
32/// server silently stopped sending checkpoints.
33const CHECKPOINT_STREAM_STALL_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(120);
34
35const RECONNECT_DELAY: std::time::Duration = std::time::Duration::from_secs(5);
36
37/// Run the clock stream and the object mirror side by side.
38pub async fn watcher(
39    sui_rpc_url: String,
40    state: OnchainState,
41    seed: super::route::MirrorSeed,
42    metrics: Option<Arc<Metrics>>,
43) {
44    tokio::join!(
45        clock_watcher(sui_rpc_url.clone(), state.clone(), metrics.clone()),
46        super::mirror::run(sui_rpc_url, state, seed, metrics),
47    );
48}
49
50#[tracing::instrument(name = "clock_watcher", skip_all)]
51async fn clock_watcher(sui_rpc_url: String, state: OnchainState, metrics: Option<Arc<Metrics>>) {
52    let read_mask = FieldMask::from_paths([
53        Checkpoint::path_builder().sequence_number(),
54        Checkpoint::path_builder().summary().timestamp(),
55        Checkpoint::path_builder().summary().epoch(),
56    ]);
57
58    loop {
59        // Reconnect with a fresh client each iteration: re-subscribing on
60        // the same channel can reuse a wedged h2 connection — the one
61        // whose stream just stalled — and silently hang again.
62        let mut client = match crate::sui_rpc_client::new_sui_rpc_client(&sui_rpc_url) {
63            Ok(client) => client,
64            Err(e) => {
65                tracing::warn!("error creating Sui RPC client: {e}");
66                tokio::time::sleep(RECONNECT_DELAY).await;
67                continue;
68            }
69        };
70
71        let mut subscription = match client
72            .subscription_client()
73            .subscribe_checkpoints(
74                SubscribeCheckpointsRequest::default().with_read_mask(read_mask.clone()),
75            )
76            .await
77        {
78            Ok(subscription) => subscription,
79            Err(e) => {
80                tracing::warn!("error trying to subscribe to checkpoints: {e}");
81                tokio::time::sleep(RECONNECT_DELAY).await;
82                continue;
83            }
84        }
85        .into_inner();
86
87        while let Ok(Some(item)) =
88            tokio::time::timeout(CHECKPOINT_STREAM_STALL_TIMEOUT, subscription.next()).await
89        {
90            let checkpoint = match item {
91                Ok(checkpoint) => checkpoint,
92                Err(e) => {
93                    tracing::warn!("error in checkpoint stream: {e}");
94                    break;
95                }
96            };
97
98            let height = checkpoint.cursor();
99            tracing::trace!("received checkpoint {height}");
100            let timestamp_ms = checkpoint
101                .checkpoint()
102                .summary()
103                .timestamp
104                .and_then(|t| proto_to_timestamp_ms(t).ok())
105                .unwrap_or(0);
106            let epoch = checkpoint.checkpoint().summary().epoch();
107            let previous_epoch = state.latest_checkpoint_epoch();
108            if epoch != previous_epoch {
109                tracing::debug!("Sui epoch changed from {previous_epoch} to {epoch}");
110                state.notify(Notification::SuiEpochChanged(epoch));
111            }
112
113            state.update_latest_checkpoint_info(CheckpointInfo {
114                height,
115                timestamp_ms,
116                epoch,
117            });
118
119            if let Some(metrics) = &metrics {
120                metrics.update_onchain_state(&state);
121            }
122        }
123
124        // The stream stalled, errored, or closed. Loop to rebuild the
125        // client and re-subscribe; the clock carries no state, so
126        // nothing needs recovering.
127        tracing::warn!("clock checkpoint stream ended; reconnecting");
128        tokio::time::sleep(RECONNECT_DELAY).await;
129    }
130}
