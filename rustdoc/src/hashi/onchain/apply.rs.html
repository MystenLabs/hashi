4//! Structural apply of one transaction's changed objects into the
5//! mirror.
6//!
7//! The unit of work is a [`TxView`]: the full post-state of every
8//! object a Hashi-touching transaction wrote, plus the ids it deleted.
9//! [`apply_transaction`] routes each object by ownership (see
10//! [`super::route`]) and updates the matching `types::Hashi` map, so
11//! the mirror no longer depends on events — a Move mutation with no
12//! `event::emit` still arrives here as an object write.
13//!
14//! Side effects that used to ride the event path (limiter advance,
15//! notifications) are returned as [`Effect`]s derived from state
16//! transitions; the caller owns acting on them.
17

38/// One successful transaction's object changes, decoded from the gRPC
39/// payload into `sui_sdk_types` values.
40#[derive(Debug)]
41pub(super) struct TxView {
42    pub checkpoint: u64,
43    /// Zero-based position within the checkpoint. Together with
44    /// `checkpoint` this is the transaction's total order on the chain;
45    /// the watcher ratchets over `(checkpoint, transaction_index)` so a
46    /// transaction delivered by both replay and the live stream applies
47    /// exactly once.
48    pub transaction_index: u64,
49    /// The transaction's checkpoint timestamp; drives the limiter
50    /// advance and the withdrawal duration metrics.
51    pub timestamp_ms: u64,
52    pub changes: Vec<TxChange>,
53    /// Changed objects whose `output_state` was missing, unrecognized,
54    /// or introduced by a newer proto than this build knows. The mirror
55    /// cannot know what happened to them; apply surfaces each through
56    /// the unrouted tripwire rather than dropping it silently.
57    pub unknown_states: Vec<(Address, &'static str)>,
58}
59
60impl TxView {
61    /// The transaction's position in the chain's total order.
62    pub(super) fn position(&self) -> (u64, u64) {
63        (self.checkpoint, self.transaction_index)
64    }
65}

67#[derive(Debug)]
68pub(super) enum TxChange {
69    /// A created or mutated object, at its post-transaction state.
70    Written(Box<Object>),
71    /// The object no longer exists after this transaction (deleted or
72    /// wrapped). Ids are never reused on Sui, so removal needs no
73    /// version guard: within each ordered source a write for this id
74    /// always precedes its deletion, and re-applying an overlap replays
75    /// the write and the deletion in order.
76    Deleted { id: Address },
77}
78
79/// Objects keyed by (id, version), BCS-decoded from a proto
80/// `ObjectSet`. The set carries both input and output versions of the
81/// same id, so the version is part of the key; each changed object is
82/// paired with its output version.
83type ObjectPool = HashMap<(Address, u64), Object>;
84
85/// Decode a transaction's proto `ObjectSet` into an [`ObjectPool`].
86/// Requires the `bcs` field in the read mask.
87fn decode_object_pool(set: Option<&proto::ObjectSet>) -> anyhow::Result<ObjectPool> {
88    let mut objects = ObjectPool::new();
89    let Some(set) = set else {
90        return Ok(objects);
91    };
92    for obj in &set.objects {
93        let Some(bytes) = obj.bcs.as_ref().and_then(|b| b.value.as_ref()) else {
94            continue;
95        };
96        let object: Object =
97            bcs::from_bytes(bytes).context("failed to BCS-decode an object from an object set")?;
98        objects.insert((object.object_id(), object.version()), object);
99    }
100    Ok(objects)
101}

103impl TxView {
104    /// Decode a proto `ExecutedTransaction` (as delivered by the
105    /// filtered transaction stream) into a `TxView`, sourcing
106    /// post-state objects from its per-transaction object set. Returns
107    /// `Ok(None)` for transactions that did not execute successfully —
108    /// their object changes are gas-only and carry no Hashi state.
109    pub(super) fn from_proto(tx: &proto::ExecutedTransaction) -> anyhow::Result<Option<Self>> {
110        let effects = tx
111            .effects
112            .as_ref()
113            .context("ExecutedTransaction is missing effects")?;
114        // Check success before decoding the object set: a failed
115        // transaction's changes are gas-only and discarded, so its
116        // objects are neither worth decoding nor worth failing on.
117        if !effects
118            .status
119            .as_ref()
120            .context("TransactionEffects is missing status")?
121            .success()
122        {
123            return Ok(None);
124        }
125        let mut pool = decode_object_pool(tx.objects.as_ref())?;
126
127        let mut changes = Vec::new();
128        let mut unknown_states = Vec::new();
129        for changed in &effects.changed_objects {
130            let id: Address = changed
131                .object_id
132                .as_deref()
133                .context("ChangedObject is missing object_id")?
134                .parse()
135                .context("invalid ChangedObject object_id")?;
136            let state = changed
137                .output_state
138                .and_then(|s| OutputObjectState::try_from(s).ok())
139                .unwrap_or(OutputObjectState::Unknown);
140            match state {
141                OutputObjectState::ObjectWrite => {
142                    let version = changed
143                        .output_version
144                        .context("written ChangedObject is missing output_version")?;
145                    let object = pool.remove(&(id, version)).with_context(|| {
146                        let mut available: Vec<String> = pool
147                            .keys()
148                            .map(|(id, version)| format!("{id}@v{version}"))
149                            .collect();
150                        available.sort();
151                        available.truncate(8);
152                        format!(
153                            "object {id} v{version} not in the object set ({} present: \
154                             [{}]) — the read mask needs objects.objects.bcs",
155                            pool.len(),
156                            available.join(", ")
157                        )
158                    })?;
159                    changes.push(TxChange::Written(Box::new(object)));
160                }
161                OutputObjectState::DoesNotExist => changes.push(TxChange::Deleted { id }),
162                // Package objects are not mirrored (the upgrade effect
163                // derives from the root's `UpgradeCap` change), so don't
164                // demand one from the object set — failing here would
165                // kill the stream on every upgrade if a server ever
166                // omitted the package object.
167                OutputObjectState::PackageWrite => {}
168                // Accumulator writes carry no object contents and no
169                // Hashi state.
170                OutputObjectState::AccumulatorWrite => {}
171                // Missing, unrecognized, or newer-than-this-build
172                // states: the one place a change could slip through
173                // undecoded, so it feeds the tripwire instead.
174                state => unknown_states.push((id, state.as_str_name())),
175            }
176        }
177
178        // The bare prost accessors default missing fields to zero; a
179        // zeroed position would fall below the delivery ratchet and the
180        // transaction would be dropped silently, and a zeroed timestamp
181        // would corrupt the limiter window. Fail the frame instead —
182        // the watcher reconnects and replays.
183        Ok(Some(Self {
184            checkpoint: tx
185                .checkpoint_opt()
186                .context("ExecutedTransaction is missing checkpoint")?,
187            transaction_index: tx
188                .transaction_index_opt()
189                .context("ExecutedTransaction is missing transaction_index")?,
190            timestamp_ms: proto_to_timestamp_ms(
191                tx.timestamp
192                    .context("ExecutedTransaction is missing timestamp")?,
193            )
194            .context("invalid ExecutedTransaction timestamp")?,
195            changes,
196            unknown_states,
197        }))
198    }
199}

201/// A side effect derived from a state transition during apply. The
202/// caller (the watcher) owns notifications, the limiter, and metrics.
203#[derive(Debug)]
204pub(super) enum Effect {
205    /// A member's `MemberInfo` entry was written.
206    ValidatorInfoUpdated(Address),
207    /// The root's `pending_epoch_change` became set for this epoch.
208    ReconfigStarted(u64),
209    /// The root's `UpgradeCap` now points at a new package id. The
210    /// cap's counter starts at 1 on publish and increments in lockstep
211    /// with the package chain, so `version` is the new package's
212    /// version. Apply extends the version map itself (under the same
213    /// state guard); this effect is the caller's signal to log.
214    PackageUpgraded { package: Address, version: u64 },
215    /// A withdrawal transaction transitioned to fully signed in this
216    /// transaction (2-of-2 witness complete). Drives the local limiter
217    /// and the pick-to-sign metric.
218    WithdrawalTxnFullySigned(Box<types::WithdrawalTransaction>),
219    /// A withdrawal transaction left the in-flight bag (confirmed and
220    /// moved to the historical record, or deleted).
221    WithdrawalTxnRemoved(Box<types::WithdrawalTransaction>),
222}

224#[derive(Debug, Default)]
225pub(super) struct ApplyOutcome {
226    pub effects: Vec<Effect>,
227    /// Objects owned by a UID we don't recognize, or carrying a type no
228    /// handler decodes. Every entry is a potential mirror gap; the
229    /// caller surfaces them as the lossless-coverage tripwire.
230    pub unrouted: Vec<(Address, String)>,
231}

233/// Apply one successful transaction's changes to the mirror.
234///
235/// Order matters within a transaction: the root and `BitcoinState` are
236/// applied first (they carry the container UIDs), then DOF wrappers
237/// (they extend the routing table), then remaining writes (committee
238/// handoffs last, because their certificate references a committee that
239/// may be written in the same transaction), then deletions, which route
240/// through the object index rather than the routing table so no
241/// pre-deletion contents are needed.
242pub(super) fn apply_transaction(
243    hashi: &mut types::Hashi,
244    packages: &mut move_types::PackageVersions,
245    routing: &mut RoutingTable,
246    index: &mut ObjectIndex,
247    tx: &TxView,
248) -> ApplyOutcome {
249    let mut out = ApplyOutcome::default();
250
251    for (id, state) in &tx.unknown_states {
252        out.unrouted.push((*id, format!("output_state {state}")));
253    }
254
255    // Pass 1: the Hashi root and the BitcoinState dynamic field.
256    for change in &tx.changes {
257        let TxChange::Written(obj) = change else {
258            continue;
259        };
260        if obj.object_id() == routing.hashi_id() {
261            apply_root(hashi, packages, routing, index, obj, &mut out);
262        } else if obj.object_id() == routing.bitcoin_state_field_id() {
263            apply_bitcoin_state(routing, index, obj, &mut out);
264        }
265    }
266
267    // Pass 2: DOF wrappers, so values created in the same transaction
268    // can resolve their container.
269    for change in &tx.changes {
270        let TxChange::Written(obj) = change else {
271            continue;
272        };
273        let Some(struct_) = obj.as_struct() else {
274            continue;
275        };
276        if is_dof_wrapper(struct_.object_type()) {
277            apply_wrapper(routing, index, obj, &mut out);
278        }
279    }
280
281    // Pass 3: all other writes, in two rounds. Writes that register an
282    // interior container go first (a tob entry or per-user request bag
283    // is rewritten in the same transaction that touches its children,
284    // because the child mutation bumps the embedded container's size),
285    // so those children route as known-ignored instead of unrouted.
286    // Committee handoffs are deferred past both rounds.
287    let mut handoffs: Vec<&Object> = Vec::new();
288    let mut deferred: Vec<&Object> = Vec::new();
289    for change in &tx.changes {
290        let TxChange::Written(obj) = change else {
291            continue;
292        };
293        let id = obj.object_id();
294        if id == routing.hashi_id() || id == routing.bitcoin_state_field_id() {
295            continue;
296        }
297        let Some(struct_) = obj.as_struct() else {
298            // Package writes: the upgrade effect is derived from the
299            // root's UpgradeCap change, not the package object.
300            continue;
301        };
302        if is_dof_wrapper(struct_.object_type()) {
303            continue;
304        }
305        if is_committee_handoff(struct_.object_type()) {
306            handoffs.push(obj);
307            continue;
308        }
309        let registers_interior = matches!(obj.owner(), Owner::Object(parent)
310        if matches!(
311            routing.resolve_owner(parent),
312            Some(Slot::Tob | Slot::UserRequests)
313        ));
314        if registers_interior {
315            apply_write(hashi, packages, routing, index, obj, &mut out);
316        } else {
317            deferred.push(obj);
318        }
319    }
320    for obj in deferred {
321        apply_write(hashi, packages, routing, index, obj, &mut out);
322    }
323
324    // Pass 4: committee handoffs, after any same-transaction committee.
325    for obj in handoffs {
326        apply_write(hashi, packages, routing, index, obj, &mut out);
327    }
328
329    // Pass 5: deletions, routed by what the mirror knows the id to be.
330    for change in &tx.changes {
331        let TxChange::Deleted { id } = change else {
332            continue;
333        };
334        let Some(entry) = index.remove(id) else {
335            continue;
336        };
337        retire(hashi, routing, id, &entry.kind, &mut out.effects);
338    }
339
340    out
341}

348fn apply_root(
349    hashi: &mut types::Hashi,
350    packages: &mut move_types::PackageVersions,
351    routing: &mut RoutingTable,
352    index: &mut ObjectIndex,
353    obj: &Object,
354    out: &mut ApplyOutcome,
355) {
356    let id = obj.object_id();
357    if index.is_stale_write(&id, obj.version()) {
358        return;
359    }
360    let Some(struct_) = obj.as_struct() else {
361        return;
362    };
363    let root: move_types::Hashi = match bcs::from_bytes(struct_.contents()) {
364        Ok(root) => root,
365        Err(e) => {
366            tracing::error!("failed to decode the Hashi root object: {e}");
367            return;
368        }
369    };
370    routing.set_root_containers(&root);
371
372    let move_types::Hashi {
373        id: _,
374        committees,
375        config,
376        versioning,
377        treasury: _,
378        proposals: _,
379        tob: _,
380        num_consumed_presigs,
381    } = root;
382
383    let new_config = super::convert_move_config(config, versioning);
384    if let (Some(old), Some(new)) = (&hashi.config.upgrade_cap, &new_config.upgrade_cap)
385        && old.package != new.package
386    {
387        out.effects.push(Effect::PackageUpgraded {
388            package: new.package,
389            version: new.version,
390        });
391    }
392    // The cap always points at the latest package version; record it
393    // before pass 3 so objects of a just-upgraded package route (or
394    // trip the mirror-gap tripwire) within the same transaction.
395    if let Some(cap) = &new_config.upgrade_cap {
396        packages.insert(cap.version, cap.package);
397    }
398    hashi.config = new_config;
399    hashi.num_consumed_presigs = num_consumed_presigs;
400
401    let new_pending = committees.pending_epoch_change.as_ref().map(|p| p.epoch);
402    if let Some(epoch) = new_pending
403        && hashi.committees.pending_epoch_change() != new_pending
404    {
405        out.effects.push(Effect::ReconfigStarted(epoch));
406    }
407    hashi
408        .committees
409        .set_epoch(committees.epoch)
410        .set_pending_epoch_change(new_pending)
411        .set_mpc_public_key(committees.mpc_public_key);
412
413    index.record(id, obj.version(), TrackedKind::HashiRoot);
414}

416fn apply_bitcoin_state(
417    routing: &mut RoutingTable,
418    index: &mut ObjectIndex,
419    obj: &Object,
420    _out: &mut ApplyOutcome,
421) {
422    let id = obj.object_id();
423    if index.is_stale_write(&id, obj.version()) {
424        return;
425    }
426    let Some(struct_) = obj.as_struct() else {
427        return;
428    };
429    let field: move_types::Field<move_types::BitcoinStateKey, move_types::BitcoinState> =
430        match bcs::from_bytes(struct_.contents()) {
431            Ok(field) => field,
432            Err(e) => {
433                tracing::error!("failed to decode the BitcoinState dynamic field: {e}");
434                return;
435            }
436        };
437    routing.set_bitcoin_state_containers(&field.value);
438    index.record(id, obj.version(), TrackedKind::BitcoinStateField);
439}

441fn apply_wrapper(
442    routing: &mut RoutingTable,
443    index: &mut ObjectIndex,
444    obj: &Object,
445    out: &mut ApplyOutcome,
446) {
447    let id = obj.object_id();
448    let Owner::Object(container) = obj.owner() else {
449        out.unrouted
450            .push((id, "DOF wrapper not object-owned".into()));
451        return;
452    };
453    // Our container set is closed and known a priori (root and
454    // BitcoinState inner UIDs), so a wrapper under an unknown container
455    // belongs to some foreign object a root-touching transaction also
456    // mutated. Don't register or track it.
457    if routing.slot_of_container(container).is_none() {
458        tracing::debug!(wrapper = %id, container = %container,
459            "ignoring a DOF wrapper under a foreign container");
460        return;
461    }
462    // Registration is idempotent, so it runs even for stale replayed
463    // writes — the mapping must exist for value routing regardless.
464    routing.register_wrapper(id, *container);
465    if index.is_stale_write(&id, obj.version()) {
466        return;
467    }
468    index.record(
469        id,
470        obj.version(),
471        TrackedKind::DofWrapper {
472            container: *container,
473        },
474    );
475}

477fn apply_write(
478    hashi: &mut types::Hashi,
479    packages: &move_types::PackageVersions,
480    routing: &mut RoutingTable,
481    index: &mut ObjectIndex,
482    obj: &Object,
483    out: &mut ApplyOutcome,
484) {
485    let id = obj.object_id();
486    let Some(struct_) = obj.as_struct() else {
487        return;
488    };
489    let slot = match obj.owner() {
490        // Address-owned objects (gas coins, user funds) are not part
491        // of the mirrored state.
492        Owner::Address(_) => return,
493        // Immutable objects (packages) carry no mirrored state either.
494        Owner::Immutable => return,
495        Owner::Object(parent) => match routing.resolve_owner(parent) {
496            Some(slot) => Some(slot),
497            // A value whose wrapper isn't in this transaction and isn't
498            // registered: fall back to what the mirror already knows
499            // the id to be.
500            None => index.get(&id).and_then(|e| slot_for_kind(&e.kind)),
501        },
502        // The only hashi shared object is the root, handled in pass 1;
503        // other shared objects a root-touching transaction mutates
504        // (e.g. the system CoinRegistry) are foreign state.
505        _ => {
506            tracing::debug!(object = %id, "ignoring a foreign shared object");
507            return;
508        }
509    };
510    let Some(slot) = slot else {
511        // An unroutable object is a mirror gap only when it carries a
512        // hashi-package type (from any published version — the boot
513        // scrape seeds the full history, and upgrades extend it);
514        // foreign objects under foreign owners are expected companions
515        // of root-touching transactions.
516        if packages.contains(struct_.object_type().address()) {
517            out.unrouted.push((id, struct_.object_type().to_string()));
518        } else {
519            tracing::debug!(object = %id, r#type = %struct_.object_type(),
520                "ignoring a foreign object-owned write");
521        }
522        return;
523    };
524
525    if index.is_stale_write(&id, obj.version()) {
526        return;
527    }
528
529    let tag = struct_.object_type();
530    let contents = struct_.contents();
531    let new_kind = match slot {
532        Slot::Members => decode::<move_types::Field<Address, move_types::MemberInfo>>(
533            contents, &id,
534        )
535        .map(|field| {
536            let info = super::convert_move_member_info(field.value);
537            let validator = info.validator_address;
538            hashi.committees.update_validator(info);
539            out.effects.push(Effect::ValidatorInfoUpdated(validator));
540            TrackedKind::Member(validator)
541        }),
542        Slot::Committees => {
543            // The committees bag is heterogeneous: per-epoch committees
544            // and committee handoffs, distinguished by value type.
545            if is_committee_handoff(tag) {
546                decode::<
547                    move_types::Field<
548                        move_types::CommitteeHandoffKey,
549                        move_types::CommitteeHandoff,
550                    >,
551                >(contents, &id)
552                .and_then(|field| {
553                    let from_epoch = field.name.epoch;
554                    apply_committee_handoff(hashi, from_epoch, field.value)
555                        .then_some(TrackedKind::CommitteeHandoff(from_epoch))
556                })
557            } else if move_types::is_field_with_value(tag, "committee", "Committee") {
558                decode::<move_types::Field<u64, move_types::Committee>>(contents, &id).map(
559                    |field| {
560                        let epoch = field.name;
561                        hashi
562                            .committees
563                            .committees_mut()
564                            .insert(epoch, super::convert_move_committee(field.value.clone()));
565                        // Keep the verbatim on-chain committee too: it is
566                        // the only form a `CommitteeTransitionRequest`
567                        // may embed (see `CommitteeSet::raw_committees`).
568                        hashi
569                            .committees
570                            .raw_committees_mut()
571                            .insert(epoch, field.value);
572                        TrackedKind::Committee(epoch)
573                    },
574                )
575            } else {
576                None
577            }
578        }
579        Slot::UtxoRecords => decode::<move_types::Field<types::UtxoId, move_types::UtxoRecord>>(
580            contents, &id,
581        )
582        .map(|field| {
583            let utxo_id = field.value.utxo.id;
584            hashi
585                .bitcoin_mut()
586                .utxo_pool
587                .utxo_records
588                .insert(utxo_id, field.value);
589            TrackedKind::UtxoRecord(utxo_id)
590        }),
591        Slot::SpentUtxos => {
592            decode::<move_types::Field<types::UtxoId, u64>>(contents, &id).map(|field| {
593                hashi
594                    .bitcoin_mut()
595                    .utxo_pool
596                    .spent_utxos
597                    .insert(field.name, field.value);
598                TrackedKind::SpentUtxo(field.name)
599            })
600        }
601        Slot::DepositRequests => {
602            decode::<move_types::DepositRequest>(contents, &id).map(|request| {
603                let request_id = request.id;
604                hashi
605                    .bitcoin_mut()
606                    .deposit_queue
607                    .requests
608                    .insert(request_id, request);
609                TrackedKind::DepositRequest(request_id)
610            })
611        }
612        Slot::WithdrawalRequests => {
613            decode::<move_types::WithdrawalRequest>(contents, &id).map(|request| {
614                let request_id = request.id;
615                hashi
616                    .bitcoin_mut()
617                    .withdrawal_queue
618                    .requests
619                    .insert(request_id, request);
620                TrackedKind::WithdrawalRequest(request_id)
621            })
622        }
623        Slot::WithdrawalTxns => {
624            decode::<move_types::WithdrawalTransaction>(contents, &id).map(|txn| {
625                let txn_id = txn.id;
626                let withdrawal_queue = &mut hashi.bitcoin_mut().withdrawal_queue;
627                let was_fully_signed = withdrawal_queue
628                    .withdrawal_txns
629                    .get(&txn_id)
630                    .is_some_and(|t| t.is_fully_signed());
631                if !was_fully_signed && txn.is_fully_signed() {
632                    out.effects
633                        .push(Effect::WithdrawalTxnFullySigned(Box::new(txn.clone())));
634                }
635                withdrawal_queue.withdrawal_txns.insert(txn_id, txn);
636                TrackedKind::WithdrawalTxn(txn_id)
637            })
638        }
639        Slot::Treasury => {
640            let type_tag = TypeTag::Struct(Box::new(tag.clone()));
641            if let Some(cap) = types::TreasuryCap::try_from_contents(&type_tag, contents) {
642                let coin_type = cap.coin_type.clone();
643                hashi.treasury.treasury_caps.insert(coin_type.clone(), cap);
644                Some(TrackedKind::TreasuryCap(coin_type))
645            } else if let Some(cap) = types::MetadataCap::try_from_contents(&type_tag, contents) {
646                let coin_type = cap.coin_type.clone();
647                hashi.treasury.metadata_caps.insert(coin_type.clone(), cap);
648                Some(TrackedKind::MetadataCap(coin_type))
649            } else {
650                None
651            }
652        }
653        Slot::ProposalsActive | Slot::ProposalsExecuted => {
654            let type_tag = TypeTag::Struct(Box::new(tag.clone()));
655            super::decode_proposal(&type_tag, contents).map(|proposal| {
656                let executed = slot == Slot::ProposalsExecuted;
657                let proposal_id = proposal.id;
658                let bucket = if executed {
659                    &mut hashi.proposals.executed
660                } else {
661                    &mut hashi.proposals.active
662                };
663                bucket.insert(proposal_id, proposal);
664                TrackedKind::Proposal {
665                    executed,
666                    id: proposal_id,
667                }
668            })
669        }
670        Slot::Tob => {
671            // Not mirrored, but the entry's LinkedTable UID owns dealer
672            // submission nodes; register it so they route as ignored.
673            decode::<move_types::Field<super::TobKey, move_types::EpochCertsV1>>(contents, &id).map(
674                |field| {
675                    routing.register_interior(field.value.certs.id, Slot::TobCerts);
676                    TrackedKind::Ignored
677                },
678            )
679        }
680        Slot::UserRequests => {
681            // Same shape: each per-user value is a Bag with its own UID.
682            decode::<move_types::Field<Address, move_types::Bag>>(contents, &id).map(|field| {
683                routing.register_interior(field.value.id, Slot::UserRequestBag);
684                TrackedKind::Ignored
685            })
686        }
687        Slot::DepositProcessed
688        | Slot::WithdrawalProcessed
689        | Slot::ConfirmedTxns
690        | Slot::TobCerts
691        | Slot::UserRequestBag => Some(TrackedKind::Ignored),
692    };
693
694    let Some(new_kind) = new_kind else {
695        out.unrouted.push((id, struct_.object_type().to_string()));
696        return;
697    };
698
699    // A write whose meaning changed is a move between containers (e.g.
700    // a proposal executing, a withdrawal txn confirming): retire the
701    // old mirror entry before the new one takes over.
702    if let Some(old) = index.get(&id)
703        && old.kind != new_kind
704    {
705        let old_kind = old.kind.clone();
706        retire(hashi, routing, &id, &old_kind, &mut out.effects);
707    }
708
709    index.record(id, obj.version(), new_kind);
710}

712/// Insert a committee handoff, converting through the raw Move
713/// committee its certificate signs. Returns false when the referenced
714/// committee is unknown (the mirror can't validate the cert shape).
715///
716/// The committee embedded in the stored transition is the verbatim
717/// on-chain one: Move's `submit_committee_handoff` verified this cert
718/// over a `CommitteeTransitionRequest` rebuilt from the stored
719/// committee, so those are the only bytes the cert matches (see
720/// `CommitteeSet::raw_committees`).
721fn apply_committee_handoff(
722    hashi: &mut types::Hashi,
723    from_epoch: u64,
724    handoff: move_types::CommitteeHandoff,
725) -> bool {
726    let Some(raw) = hashi.committees.raw_committee(handoff.next_epoch).cloned() else {
727        tracing::error!(
728            from_epoch,
729            next_epoch = handoff.next_epoch,
730            "committee handoff references a committee the mirror doesn't hold"
731        );
732        return false;
733    };
734    match super::convert_move_committee_handoff(handoff, raw) {
735        Ok(signed) => {
736            hashi
737                .committees
738                .committee_handoffs_mut()
739                .insert(from_epoch, signed);
740            true
741        }
742        Err(e) => {
743            tracing::error!(from_epoch, "invalid committee handoff: {e}");
744            false
745        }
746    }
747}

749/// Remove whatever the mirror holds for a retired object id. Used both
750/// for deletions and for cross-container moves.
751fn retire(
752    hashi: &mut types::Hashi,
753    routing: &mut RoutingTable,
754    id: &Address,
755    kind: &TrackedKind,
756    effects: &mut Vec<Effect>,
757) {
758    match kind {
759        TrackedKind::HashiRoot | TrackedKind::BitcoinStateField => {
760            tracing::error!(?kind, "refusing to retire a root object");
761        }
762        TrackedKind::DofWrapper { .. } => {
763            routing.remove_wrapper(id);
764        }
765        TrackedKind::Member(validator) => {
766            hashi.committees.remove_validator(validator);
767        }
768        TrackedKind::Committee(epoch) => {
769            hashi.committees.committees_mut().remove(epoch);
770            hashi.committees.raw_committees_mut().remove(epoch);
771        }
772        TrackedKind::CommitteeHandoff(epoch) => {
773            hashi.committees.committee_handoffs_mut().remove(epoch);
774        }
775        TrackedKind::UtxoRecord(utxo_id) => {
776            hashi.bitcoin_mut().utxo_pool.utxo_records.remove(utxo_id);
777        }
778        TrackedKind::SpentUtxo(utxo_id) => {
779            hashi.bitcoin_mut().utxo_pool.spent_utxos.remove(utxo_id);
780        }
781        TrackedKind::DepositRequest(id) => {
782            hashi.bitcoin_mut().deposit_queue.requests.remove(id);
783        }
784        TrackedKind::WithdrawalRequest(id) => {
785            hashi.bitcoin_mut().withdrawal_queue.requests.remove(id);
786        }
787        TrackedKind::WithdrawalTxn(id) => {
788            if let Some(txn) = hashi
789                .bitcoin_mut()
790                .withdrawal_queue
791                .withdrawal_txns
792                .remove(id)
793            {
794                effects.push(Effect::WithdrawalTxnRemoved(Box::new(txn)));
795            }
796        }
797        TrackedKind::TreasuryCap(coin_type) => {
798            hashi.treasury.treasury_caps.remove(coin_type);
799        }
800        TrackedKind::MetadataCap(coin_type) => {
801            hashi.treasury.metadata_caps.remove(coin_type);
802        }
803        TrackedKind::Proposal { executed, id } => {
804            let bucket = if *executed {
805                &mut hashi.proposals.executed
806            } else {
807                &mut hashi.proposals.active
808            };
809            bucket.remove(id);
810        }
811        TrackedKind::Ignored => {}
812    }
813}

815/// Slot equivalent of an already-tracked kind, used to route value-only
816/// mutations whose wrapper is neither in the transaction nor registered.
817fn slot_for_kind(kind: &TrackedKind) -> Option<Slot> {
818    match kind {
819        TrackedKind::HashiRoot
820        | TrackedKind::BitcoinStateField
821        | TrackedKind::DofWrapper { .. } => None,
822        TrackedKind::Member(_) => Some(Slot::Members),
823        TrackedKind::Committee(_) | TrackedKind::CommitteeHandoff(_) => Some(Slot::Committees),
824        TrackedKind::UtxoRecord(_) => Some(Slot::UtxoRecords),
825        TrackedKind::SpentUtxo(_) => Some(Slot::SpentUtxos),
826        TrackedKind::DepositRequest(_) => Some(Slot::DepositRequests),
827        TrackedKind::WithdrawalRequest(_) => Some(Slot::WithdrawalRequests),
828        TrackedKind::WithdrawalTxn(_) => Some(Slot::WithdrawalTxns),
829        TrackedKind::TreasuryCap(_) | TrackedKind::MetadataCap(_) => Some(Slot::Treasury),
830        TrackedKind::Proposal { executed, .. } => Some(if *executed {
831            Slot::ProposalsExecuted
832        } else {
833            Slot::ProposalsActive
834        }),
835        TrackedKind::Ignored => None,
836    }
837}

839/// Decode a routed object's contents. On failure, log and return
840/// `None`; the caller records the object as unrouted (the tripwire).
841fn decode<T: serde::de::DeserializeOwned>(contents: &[u8], id: &Address) -> Option<T> {
842    match bcs::from_bytes(contents) {
843        Ok(value) => Some(value),
844        Err(e) => {
845            tracing::error!("failed to BCS-decode routed object {id}: {e}");
846            None
847        }
848    }
