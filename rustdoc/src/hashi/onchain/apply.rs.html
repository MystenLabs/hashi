4//! Structural apply of one transaction's changed objects into the
5//! mirror.
6//!
7//! The unit of work is a [`TxView`]: the full post-state of every
8//! object a Hashi-touching transaction wrote, plus the ids it deleted.
9//! [`apply_transaction`] routes each object by ownership (see
10//! [`super::route`]) and updates the matching `types::Hashi` map, so
11//! the mirror no longer depends on events — a Move mutation with no
12//! `event::emit` still arrives here as an object write.
13//!
14//! Side effects that used to ride the event path (limiter advance,
15//! notifications) are returned as [`Effect`]s derived from state
16//! transitions; the caller owns acting on them.
17

41/// One successful transaction's object changes, decoded from the gRPC
42/// payload into `sui_sdk_types` values.
43#[derive(Debug)]
44pub(super) struct TxView {
45    pub checkpoint: u64,
46    /// Zero-based position within the checkpoint. Together with
47    /// `checkpoint` this is the transaction's total order on the chain;
48    /// the watcher ratchets over `(checkpoint, transaction_index)` so a
49    /// transaction delivered by both replay and the live stream applies
50    /// exactly once.
51    pub transaction_index: u64,
52    /// The transaction's checkpoint timestamp; drives the limiter
53    /// advance and the withdrawal duration metrics.
54    pub timestamp_ms: u64,
55    pub changes: Vec<TxChange>,
56    /// Changed objects whose `output_state` was missing, unrecognized,
57    /// or introduced by a newer proto than this build knows. The mirror
58    /// cannot know what happened to them; apply surfaces each through
59    /// the unrouted tripwire rather than dropping it silently.
60    pub unknown_states: Vec<(Address, &'static str)>,
61}
62
63impl TxView {
64    /// The transaction's position in the chain's total order.
65    pub(super) fn position(&self) -> (u64, u64) {
66        (self.checkpoint, self.transaction_index)
67    }
68}

70#[derive(Debug)]
71pub(super) enum TxChange {
72    /// A created or mutated object, at its post-transaction state.
73    Written(Box<Object>),
74    /// The object no longer exists after this transaction (deleted or
75    /// wrapped). Ids are never reused on Sui, so removal needs no
76    /// version guard: within each ordered source a write for this id
77    /// always precedes its deletion, and re-applying an overlap replays
78    /// the write and the deletion in order.
79    Deleted { id: Address },
80}
81
82/// Objects keyed by (id, version), BCS-decoded from a proto
83/// `ObjectSet`. The set carries both input and output versions of the
84/// same id, so the version is part of the key; each changed object is
85/// paired with its output version.
86type ObjectPool = HashMap<(Address, u64), Object>;
87
88/// Decode a transaction's proto `ObjectSet` into an [`ObjectPool`].
89/// Requires the `bcs` field in the read mask.
90fn decode_object_pool(set: Option<&proto::ObjectSet>) -> anyhow::Result<ObjectPool> {
91    let mut objects = ObjectPool::new();
92    let Some(set) = set else {
93        return Ok(objects);
94    };
95    for obj in &set.objects {
96        let Some(bytes) = obj.bcs.as_ref().and_then(|b| b.value.as_ref()) else {
97            continue;
98        };
99        let object: Object =
100            bcs::from_bytes(bytes).context("failed to BCS-decode an object from an object set")?;
101        objects.insert((object.object_id(), object.version()), object);
102    }
103    Ok(objects)
104}

106impl TxView {
107    /// Decode a proto `ExecutedTransaction` (as delivered by the
108    /// filtered transaction stream) into a `TxView`, sourcing
109    /// post-state objects from its per-transaction object set. Returns
110    /// `Ok(None)` for transactions that did not execute successfully —
111    /// their object changes are gas-only and carry no Hashi state.
112    pub(super) fn from_proto(tx: &proto::ExecutedTransaction) -> anyhow::Result<Option<Self>> {
113        let effects = tx
114            .effects
115            .as_ref()
116            .context("ExecutedTransaction is missing effects")?;
117        // Check success before decoding the object set: a failed
118        // transaction's changes are gas-only and discarded, so its
119        // objects are neither worth decoding nor worth failing on.
120        if !effects
121            .status
122            .as_ref()
123            .context("TransactionEffects is missing status")?
124            .success()
125        {
126            return Ok(None);
127        }
128        let mut pool = decode_object_pool(tx.objects.as_ref())?;
129
130        let mut changes = Vec::new();
131        let mut unknown_states = Vec::new();
132        for changed in &effects.changed_objects {
133            let id: Address = changed
134                .object_id
135                .as_deref()
136                .context("ChangedObject is missing object_id")?
137                .parse()
138                .context("invalid ChangedObject object_id")?;
139            let state = changed
140                .output_state
141                .and_then(|s| OutputObjectState::try_from(s).ok())
142                .unwrap_or(OutputObjectState::Unknown);
143            match state {
144                OutputObjectState::ObjectWrite => {
145                    let version = changed
146                        .output_version
147                        .context("written ChangedObject is missing output_version")?;
148                    let object = pool.remove(&(id, version)).with_context(|| {
149                        let mut available: Vec<String> = pool
150                            .keys()
151                            .map(|(id, version)| format!("{id}@v{version}"))
152                            .collect();
153                        available.sort();
154                        available.truncate(8);
155                        format!(
156                            "object {id} v{version} not in the object set ({} present: \
157                             [{}]) — the read mask needs objects.objects.bcs",
158                            pool.len(),
159                            available.join(", ")
160                        )
161                    })?;
162                    changes.push(TxChange::Written(Box::new(object)));
163                }
164                OutputObjectState::DoesNotExist => changes.push(TxChange::Deleted { id }),
165                // Package objects are not mirrored (the upgrade effect
166                // derives from the root's `UpgradeCap` change), so don't
167                // demand one from the object set — failing here would
168                // kill the stream on every upgrade if a server ever
169                // omitted the package object.
170                OutputObjectState::PackageWrite => {}
171                // Accumulator writes carry no object contents and no
172                // Hashi state.
173                OutputObjectState::AccumulatorWrite => {}
174                // Missing, unrecognized, or newer-than-this-build
175                // states: the one place a change could slip through
176                // undecoded, so it feeds the tripwire instead.
177                state => unknown_states.push((id, state.as_str_name())),
178            }
179        }
180
181        // The bare prost accessors default missing fields to zero; a
182        // zeroed position would fall below the delivery ratchet and the
183        // transaction would be dropped silently, and a zeroed timestamp
184        // would corrupt the limiter window. Fail the frame instead —
185        // the watcher reconnects and replays.
186        Ok(Some(Self {
187            checkpoint: tx
188                .checkpoint_opt()
189                .context("ExecutedTransaction is missing checkpoint")?,
190            transaction_index: tx
191                .transaction_index_opt()
192                .context("ExecutedTransaction is missing transaction_index")?,
193            timestamp_ms: proto_to_timestamp_ms(
194                tx.timestamp
195                    .context("ExecutedTransaction is missing timestamp")?,
196            )
197            .context("invalid ExecutedTransaction timestamp")?,
198            changes,
199            unknown_states,
200        }))
201    }
202}

204/// A side effect derived from a state transition during apply. The
205/// caller (the watcher) owns notifications, the limiter, and metrics.
206#[derive(Debug)]
207pub(super) enum Effect {
208    /// A member's `MemberInfo` entry was written.
209    ValidatorInfoUpdated(Address),
210    /// The root's `pending_epoch_change` became set for this epoch.
211    ReconfigStarted(u64),
212    /// The root's `UpgradeCap` now points at a new package id. The
213    /// cap's counter starts at 1 on publish and increments in lockstep
214    /// with the package chain, so `version` is the new package's
215    /// version. Apply extends the version map itself (under the same
216    /// state guard); this effect is the caller's signal to log.
217    PackageUpgraded { package: Address, version: u64 },
218    /// An active deposit request was created or updated.
219    DepositRequestUpserted {
220        request_id: Address,
221        outpoint: bitcoin::OutPoint,
222    },
223    /// A deposit request left the active queue.
224    DepositRequestRemoved(Address),
225    /// A withdrawal transaction transitioned to fully signed in this
226    /// transaction (2-of-2 witness complete). Drives the local limiter
227    /// and the pick-to-sign metric.
228    WithdrawalTxnFullySigned(Box<types::WithdrawalTransaction>),
229    /// A withdrawal transaction's `confirmed_timestamp_ms` went from
230    /// `None` to `Some` in this transaction: the v2 in-place confirm,
231    /// which leaves the txn in the in-flight bag until a later archival
232    /// GC moves it out. Drives the sign-to-confirm and total duration
233    /// metrics in the deferred-archival world (under v1 bytecode the
234    /// confirm moves the txn instead, so this never fires there).
235    WithdrawalTxnConfirmed(Box<types::WithdrawalTransaction>),
236    /// A withdrawal transaction left the in-flight bag (confirmed and
237    /// moved to the historical record, or deleted).
238    WithdrawalTxnRemoved(Box<types::WithdrawalTransaction>),
239}

241#[derive(Debug, Default)]
242pub(super) struct ApplyOutcome {
243    pub effects: Vec<Effect>,
244    /// Objects owned by a UID we don't recognize, or carrying a type no
245    /// handler decodes. Every entry is a potential mirror gap; the
246    /// caller surfaces them as the lossless-coverage tripwire.
247    pub unrouted: Vec<(Address, String)>,
248}

250/// Apply one successful transaction's changes to the mirror.
251///
252/// Order matters within a transaction: the root and `BitcoinState` are
253/// applied first (they carry the container UIDs), then DOF wrappers
254/// (they extend the routing table), then remaining writes (committee
255/// handoffs last, because their certificate references a committee that
256/// may be written in the same transaction), then deletions, which route
257/// through the object index rather than the routing table so no
258/// pre-deletion contents are needed.
259pub(super) fn apply_transaction(
260    hashi: &mut types::Hashi,
261    packages: &mut move_types::PackageVersions,
262    routing: &mut RoutingTable,
263    index: &mut ObjectIndex,
264    tx: &TxView,
265) -> ApplyOutcome {
266    let mut out = ApplyOutcome::default();
267
268    for (id, state) in &tx.unknown_states {
269        out.unrouted.push((*id, format!("output_state {state}")));
270    }
271
272    // Pass 1: the Hashi root and the BitcoinState dynamic field.
273    for change in &tx.changes {
274        let TxChange::Written(obj) = change else {
275            continue;
276        };
277        if obj.object_id() == routing.hashi_id() {
278            apply_root(hashi, packages, routing, index, obj, &mut out);
279        } else if obj.object_id() == routing.bitcoin_state_field_id() {
280            apply_bitcoin_state(routing, index, obj, &mut out);
281        }
282    }
283
284    // Pass 2: DOF wrappers, so values created in the same transaction
285    // can resolve their container.
286    for change in &tx.changes {
287        let TxChange::Written(obj) = change else {
288            continue;
289        };
290        let Some(struct_) = obj.as_struct() else {
291            continue;
292        };
293        if is_dof_wrapper(struct_.object_type()) {
294            apply_wrapper(routing, index, obj, &mut out);
295        }
296    }
297
298    // Pass 3: all other writes, in two rounds. Writes that register an
299    // interior container go first (a tob entry or per-user request bag
300    // is rewritten in the same transaction that touches its children,
301    // because the child mutation bumps the embedded container's size),
302    // so those children route as known-ignored instead of unrouted.
303    // Committee handoffs are deferred past both rounds.
304    let mut handoffs: Vec<&Object> = Vec::new();
305    let mut deferred: Vec<&Object> = Vec::new();
306    for change in &tx.changes {
307        let TxChange::Written(obj) = change else {
308            continue;
309        };
310        let id = obj.object_id();
311        if id == routing.hashi_id() || id == routing.bitcoin_state_field_id() {
312            continue;
313        }
314        let Some(struct_) = obj.as_struct() else {
315            // Package writes: the upgrade effect is derived from the
316            // root's UpgradeCap change, not the package object.
317            continue;
318        };
319        if is_dof_wrapper(struct_.object_type()) {
320            continue;
321        }
322        if is_committee_handoff(struct_.object_type()) {
323            handoffs.push(obj);
324            continue;
325        }
326        let registers_interior = matches!(obj.owner(), Owner::Object(parent)
327        if matches!(
328            routing.resolve_owner(parent),
329            Some(Slot::Tob | Slot::UserRequests)
330        ));
331        if registers_interior {
332            apply_write(hashi, packages, routing, index, obj, &mut out);
333        } else {
334            deferred.push(obj);
335        }
336    }
337    for obj in deferred {
338        apply_write(hashi, packages, routing, index, obj, &mut out);
339    }
340
341    // Pass 4: committee handoffs, after any same-transaction committee.
342    for obj in handoffs {
343        apply_write(hashi, packages, routing, index, obj, &mut out);
344    }
345
346    // Pass 5: deletions, routed by what the mirror knows the id to be.
347    for change in &tx.changes {
348        let TxChange::Deleted { id } = change else {
349            continue;
350        };
351        let Some(entry) = index.remove(id) else {
352            continue;
353        };
354        retire(hashi, routing, id, &entry.kind, &mut out.effects);
355    }
356
357    out
358}

365fn apply_root(
366    hashi: &mut types::Hashi,
367    packages: &mut move_types::PackageVersions,
368    routing: &mut RoutingTable,
369    index: &mut ObjectIndex,
370    obj: &Object,
371    out: &mut ApplyOutcome,
372) {
373    let id = obj.object_id();
374    if index.is_stale_write(&id, obj.version()) {
375        return;
376    }
377    let Some(struct_) = obj.as_struct() else {
378        return;
379    };
380    let root: move_types::Hashi = match bcs::from_bytes(struct_.contents()) {
381        Ok(root) => root,
382        Err(e) => {
383            tracing::error!("failed to decode the Hashi root object: {e}");
384            return;
385        }
386    };
387    routing.set_root_containers(&root);
388
389    let move_types::Hashi {
390        id: _,
391        committees,
392        config,
393        epoch_config,
394        versioning,
395        treasury: _,
396        proposals: _,
397        tob: _,
398        num_consumed_presigs,
399    } = root;
400
401    let new_config = super::convert_move_config(config, versioning);
402    if let (Some(old), Some(new)) = (&hashi.config.upgrade_cap, &new_config.upgrade_cap)
403        && old.package != new.package
404    {
405        out.effects.push(Effect::PackageUpgraded {
406            package: new.package,
407            version: new.version,
408        });
409    }
410    // The cap always points at the latest package version; record it
411    // before pass 3 so objects of a just-upgraded package route (or
412    // trip the mirror-gap tripwire) within the same transaction.
413    if let Some(cap) = &new_config.upgrade_cap {
414        packages.insert(cap.version, cap.package);
415    }
416    hashi.config = new_config;
417    hashi.epoch_config = epoch_config;
418    hashi.num_consumed_presigs = num_consumed_presigs;
419
420    let new_pending = committees.pending_epoch_change.as_ref().map(|p| p.epoch);
421    if let Some(epoch) = new_pending
422        && hashi.committees.pending_epoch_change() != new_pending
423    {
424        out.effects.push(Effect::ReconfigStarted(epoch));
425    }
426    hashi
427        .committees
428        .set_epoch(committees.epoch)
429        .set_pending_epoch_change(new_pending)
430        .set_mpc_public_key(committees.mpc_public_key);
431
432    index.record(id, obj.version(), TrackedKind::HashiRoot);
433}

435fn apply_bitcoin_state(
436    routing: &mut RoutingTable,
437    index: &mut ObjectIndex,
438    obj: &Object,
439    _out: &mut ApplyOutcome,
440) {
441    let id = obj.object_id();
442    if index.is_stale_write(&id, obj.version()) {
443        return;
444    }
445    let Some(struct_) = obj.as_struct() else {
446        return;
447    };
448    let field: move_types::Field<move_types::BitcoinStateKey, move_types::BitcoinState> =
449        match bcs::from_bytes(struct_.contents()) {
450            Ok(field) => field,
451            Err(e) => {
452                tracing::error!("failed to decode the BitcoinState dynamic field: {e}");
453                return;
454            }
455        };
456    routing.set_bitcoin_state_containers(&field.value);
457    index.record(id, obj.version(), TrackedKind::BitcoinStateField);
458}

460fn apply_wrapper(
461    routing: &mut RoutingTable,
462    index: &mut ObjectIndex,
463    obj: &Object,
464    out: &mut ApplyOutcome,
465) {
466    let id = obj.object_id();
467    let Owner::Object(container) = obj.owner() else {
468        out.unrouted
469            .push((id, "DOF wrapper not object-owned".into()));
470        return;
471    };
472    // Our container set is closed and known a priori (root and
473    // BitcoinState inner UIDs), so a wrapper under an unknown container
474    // belongs to some foreign object a root-touching transaction also
475    // mutated. Don't register or track it.
476    if routing.slot_of_container(container).is_none() {
477        tracing::debug!(wrapper = %id, container = %container,
478            "ignoring a DOF wrapper under a foreign container");
479        return;
480    }
481    // Registration is idempotent, so it runs even for stale replayed
482    // writes — the mapping must exist for value routing regardless.
483    routing.register_wrapper(id, *container);
484    if index.is_stale_write(&id, obj.version()) {
485        return;
486    }
487    index.record(
488        id,
489        obj.version(),
490        TrackedKind::DofWrapper {
491            container: *container,
492        },
493    );
494}

496fn apply_write(
497    hashi: &mut types::Hashi,
498    packages: &move_types::PackageVersions,
499    routing: &mut RoutingTable,
500    index: &mut ObjectIndex,
501    obj: &Object,
502    out: &mut ApplyOutcome,
503) {
504    let id = obj.object_id();
505    let Some(struct_) = obj.as_struct() else {
506        return;
507    };
508    let slot = match obj.owner() {
509        // Address-owned objects (gas coins, user funds) are not part
510        // of the mirrored state.
511        Owner::Address(_) => return,
512        // Immutable objects (packages) carry no mirrored state either.
513        Owner::Immutable => return,
514        Owner::Object(parent) => match routing.resolve_owner(parent) {
515            Some(slot) => Some(slot),
516            // A value whose wrapper isn't in this transaction and isn't
517            // registered: fall back to what the mirror already knows
518            // the id to be.
519            None => index.get(&id).and_then(|e| slot_for_kind(&e.kind)),
520        },
521        // The only hashi shared object is the root, handled in pass 1;
522        // other shared objects a root-touching transaction mutates
523        // (e.g. the system CoinRegistry) are foreign state.
524        _ => {
525            tracing::debug!(object = %id, "ignoring a foreign shared object");
526            return;
527        }
528    };
529    let Some(slot) = slot else {
530        // An unroutable object is a mirror gap only when it carries a
531        // hashi-package type (from any published version — the boot
532        // scrape seeds the full history, and upgrades extend it);
533        // foreign objects under foreign owners are expected companions
534        // of root-touching transactions.
535        if packages.contains(struct_.object_type().address()) {
536            out.unrouted.push((id, struct_.object_type().to_string()));
537        } else {
538            tracing::debug!(object = %id, r#type = %struct_.object_type(),
539                "ignoring a foreign object-owned write");
540        }
541        return;
542    };
543
544    if index.is_stale_write(&id, obj.version()) {
545        return;
546    }
547
548    let tag = struct_.object_type();
549    let contents = struct_.contents();
550    let new_kind = match slot {
551        Slot::Members => decode::<move_types::Field<Address, move_types::MemberInfo>>(
552            contents, &id,
553        )
554        .map(|field| {
555            let info = super::convert_move_member_info(field.value);
556            let validator = info.validator_address;
557            hashi.committees.update_validator(info);
558            out.effects.push(Effect::ValidatorInfoUpdated(validator));
559            TrackedKind::Member(validator)
560        }),
561        Slot::Committees => {
562            // The committees bag is heterogeneous: per-epoch committees
563            // and committee handoffs, distinguished by value type.
564            if is_committee_handoff(tag) {
565                decode::<
566                    move_types::Field<
567                        move_types::CommitteeHandoffKey,
568                        move_types::CommitteeHandoff,
569                    >,
570                >(contents, &id)
571                .and_then(|field| {
572                    let from_epoch = field.name.epoch;
573                    apply_committee_handoff(hashi, from_epoch, field.value)
574                        .then_some(TrackedKind::CommitteeHandoff(from_epoch))
575                })
576            } else if move_types::is_field_with_value(tag, "committee", "Committee") {
577                decode::<move_types::Field<u64, move_types::Committee>>(contents, &id).map(
578                    |field| {
579                        let epoch = field.name;
580                        hashi
581                            .committees
582                            .committees_mut()
583                            .insert(epoch, super::convert_move_committee(field.value.clone()));
584                        // Keep the verbatim on-chain committee too: it is
585                        // the only form a `CommitteeTransitionRequest`
586                        // may embed (see `CommitteeSet::raw_committees`).
587                        hashi
588                            .committees
589                            .raw_committees_mut()
590                            .insert(epoch, field.value);
591                        TrackedKind::Committee(epoch)
592                    },
593                )
594            } else {
595                None
596            }
597        }
598        Slot::UtxoRecords => decode::<move_types::Field<types::UtxoId, move_types::UtxoRecord>>(
599            contents, &id,
600        )
601        .map(|field| {
602            let utxo_id = field.value.utxo.id;
603            hashi
604                .bitcoin_mut()
605                .utxo_pool
606                .utxo_records
607                .insert(utxo_id, field.value);
608            TrackedKind::UtxoRecord(utxo_id)
609        }),
610        Slot::SpentUtxos => {
611            decode::<move_types::Field<types::UtxoId, u64>>(contents, &id).map(|field| {
612                hashi
613                    .bitcoin_mut()
614                    .utxo_pool
615                    .spent_utxos
616                    .insert(field.name, field.value);
617                TrackedKind::SpentUtxo(field.name)
618            })
619        }
620        Slot::DepositRequests => {
621            decode::<move_types::DepositRequest>(contents, &id).map(|request| {
622                let request_id = request.id;
623                out.effects.push(Effect::DepositRequestUpserted {
624                    request_id,
625                    outpoint: request.utxo.id.into(),
626                });
627                hashi
628                    .bitcoin_mut()
629                    .deposit_queue
630                    .requests
631                    .insert(request_id, request);
632                TrackedKind::DepositRequest(request_id)
633            })
634        }
635        Slot::WithdrawalRequests => {
636            decode::<move_types::WithdrawalRequest>(contents, &id).map(|request| {
637                let request_id = request.id;
638                hashi
639                    .bitcoin_mut()
640                    .withdrawal_queue
641                    .requests
642                    .insert(request_id, request);
643                TrackedKind::WithdrawalRequest(request_id)
644            })
645        }
646        Slot::WithdrawalTxns => {
647            decode::<move_types::WithdrawalTransaction>(contents, &id).map(|txn| {
648                let txn_id = txn.id;
649                let withdrawal_queue = &mut hashi.bitcoin_mut().withdrawal_queue;
650                let old = withdrawal_queue.withdrawal_txns.get(&txn_id);
651                let was_fully_signed = old.is_some_and(|t| t.is_fully_signed());
652                let was_confirmed = old.is_some_and(|t| t.confirmed_timestamp_ms.is_some());
653                if !was_fully_signed && txn.is_fully_signed() {
654                    out.effects
655                        .push(Effect::WithdrawalTxnFullySigned(Box::new(txn.clone())));
656                }
657                if !was_confirmed && txn.is_confirmed() {
658                    out.effects
659                        .push(Effect::WithdrawalTxnConfirmed(Box::new(txn.clone())));
660                }
661                withdrawal_queue.withdrawal_txns.insert(txn_id, txn);
662                TrackedKind::WithdrawalTxn(txn_id)
663            })
664        }
665        Slot::Treasury => {
666            let type_tag = TypeTag::Struct(Box::new(tag.clone()));
667            if let Some(cap) = types::TreasuryCap::try_from_contents(&type_tag, contents) {
668                let coin_type = cap.coin_type.clone();
669                hashi.treasury.treasury_caps.insert(coin_type.clone(), cap);
670                Some(TrackedKind::TreasuryCap(coin_type))
671            } else if let Some(cap) = types::MetadataCap::try_from_contents(&type_tag, contents) {
672                let coin_type = cap.coin_type.clone();
673                hashi.treasury.metadata_caps.insert(coin_type.clone(), cap);
674                Some(TrackedKind::MetadataCap(coin_type))
675            } else {
676                None
677            }
678        }
679        Slot::ProposalsActive | Slot::ProposalsExecuted => {
680            let type_tag = TypeTag::Struct(Box::new(tag.clone()));
681            super::decode_proposal(&type_tag, contents).map(|proposal| {
682                let executed = slot == Slot::ProposalsExecuted;
683                let proposal_id = proposal.id;
684                let bucket = if executed {
685                    &mut hashi.proposals.executed
686                } else {
687                    &mut hashi.proposals.active
688                };
689                bucket.insert(proposal_id, proposal);
690                TrackedKind::Proposal {
691                    executed,
692                    id: proposal_id,
693                }
694            })
695        }
696        Slot::Tob => {
697            // A bucket Field is rewritten on every submission (the
698            // embedded LinkedTable's head/tail/size live in it); the
699            // nodes themselves arrive as TobCerts writes. The two
700            // bucket structs are BCS-identical, so one decode serves
701            // both; the Field's value-side type selects the node
702            // layout, and an unrecognized one falls through to the
703            // unrouted tripwire rather than misdecoding.
704            tob_bucket_layout(packages, tag).and_then(|layout| {
705                decode::<move_types::Field<move_types::TobKey, move_types::EpochCertsV1>>(
706                    contents, &id,
707                )
708                .map(|field| {
709                    let key = field.name;
710                    let certs_id = field.value.certs.id;
711                    routing.register_interior(certs_id, Slot::TobCerts);
712                    routing.register_tob_table(certs_id, key);
713                    let bucket = hashi
714                        .tob
715                        .buckets
716                        .entry(key)
717                        .or_insert_with(|| types::TobBucket {
718                            layout,
719                            certs_id,
720                            head: None,
721                            size: 0,
722                            nodes: std::collections::BTreeMap::new(),
723                        });
724                    bucket.layout = layout;
725                    bucket.certs_id = certs_id;
726                    bucket.head = field.value.certs.head;
727                    bucket.size = field.value.certs.size;
728                    TrackedKind::TobBucket(key)
729                })
730            })
731        }
732        Slot::TobCerts => {
733            // The owner is the bucket's LinkedTable UID; a node write is
734            // a new dealer submission, or a neighbor's prev/next link
735            // update from a same-transaction insertion. The node's own
736            // value type says whether the submission is bare or stamped;
737            // both are stored in the normalized stamped form.
738            let parent = match obj.owner() {
739                Owner::Object(parent) => *parent,
740                _ => return,
741            };
742            routing.tob_key_of_table(&parent).and_then(|key| {
743                decode_tob_node(packages, tag, contents, &id).map(|(dealer, node)| {
744                    if let Some(bucket) = hashi.tob.buckets.get_mut(&key) {
745                        bucket.nodes.insert(dealer, node);
746                    }
747                    TrackedKind::TobCert { key, dealer }
748                })
749            })
750        }
751        Slot::UserRequests => {
752            // Same shape: each per-user value is a Bag with its own UID.
753            decode::<move_types::Field<Address, move_types::Bag>>(contents, &id).map(|field| {
754                routing.register_interior(field.value.id, Slot::UserRequestBag);
755                TrackedKind::Ignored
756            })
757        }
758        Slot::DepositProcessed
759        | Slot::WithdrawalProcessed
760        | Slot::ConfirmedTxns
761        | Slot::UserRequestBag => Some(TrackedKind::Ignored),
762    };
763
764    let Some(new_kind) = new_kind else {
765        out.unrouted.push((id, struct_.object_type().to_string()));
766        return;
767    };
768
769    // A write whose meaning changed is a move between containers (e.g.
770    // a proposal executing, a withdrawal txn confirming): retire the
771    // old mirror entry before the new one takes over.
772    if let Some(old) = index.get(&id)
773        && old.kind != new_kind
774    {
775        let old_kind = old.kind.clone();
776        retire(hashi, routing, &id, &old_kind, &mut out.effects);
777    }
778
779    index.record(id, obj.version(), new_kind);
780}

782/// Insert a committee handoff, converting through the raw Move
783/// committee its certificate signs. Returns false when the referenced
784/// committee is unknown (the mirror can't validate the cert shape).
785///
786/// The committee embedded in the stored transition is the verbatim
787/// on-chain one: Move's `submit_committee_handoff` verified this cert
788/// over a `CommitteeTransitionRequest` rebuilt from the stored
789/// committee, so those are the only bytes the cert matches (see
790/// `CommitteeSet::raw_committees`).
791fn apply_committee_handoff(
792    hashi: &mut types::Hashi,
793    from_epoch: u64,
794    handoff: move_types::CommitteeHandoff,
795) -> bool {
796    let Some(raw) = hashi.committees.raw_committee(handoff.next_epoch).cloned() else {
797        tracing::error!(
798            from_epoch,
799            next_epoch = handoff.next_epoch,
800            "committee handoff references a committee the mirror doesn't hold"
801        );
802        return false;
803    };
804    match super::convert_move_committee_handoff(handoff, raw) {
805        Ok(signed) => {
806            hashi
807                .committees
808                .committee_handoffs_mut()
809                .insert(from_epoch, signed);
810            true
811        }
812        Err(e) => {
813            tracing::error!(from_epoch, "invalid committee handoff: {e}");
814            false
815        }
816    }
817}

819/// Remove whatever the mirror holds for a retired object id. Used both
820/// for deletions and for cross-container moves.
821fn retire(
822    hashi: &mut types::Hashi,
823    routing: &mut RoutingTable,
824    id: &Address,
825    kind: &TrackedKind,
826    effects: &mut Vec<Effect>,
827) {
828    match kind {
829        TrackedKind::HashiRoot | TrackedKind::BitcoinStateField => {
830            tracing::error!(?kind, "refusing to retire a root object");
831        }
832        TrackedKind::DofWrapper { .. } => {
833            routing.remove_wrapper(id);
834        }
835        TrackedKind::Member(validator) => {
836            hashi.committees.remove_validator(validator);
837        }
838        TrackedKind::Committee(epoch) => {
839            hashi.committees.committees_mut().remove(epoch);
840            hashi.committees.raw_committees_mut().remove(epoch);
841        }
842        TrackedKind::CommitteeHandoff(epoch) => {
843            hashi.committees.committee_handoffs_mut().remove(epoch);
844        }
845        TrackedKind::UtxoRecord(utxo_id) => {
846            hashi.bitcoin_mut().utxo_pool.utxo_records.remove(utxo_id);
847        }
848        TrackedKind::SpentUtxo(utxo_id) => {
849            hashi.bitcoin_mut().utxo_pool.spent_utxos.remove(utxo_id);
850        }
851        TrackedKind::DepositRequest(id) => {
852            hashi.bitcoin_mut().deposit_queue.requests.remove(id);
853            effects.push(Effect::DepositRequestRemoved(*id));
854        }
855        TrackedKind::WithdrawalRequest(id) => {
856            hashi.bitcoin_mut().withdrawal_queue.requests.remove(id);
857        }
858        TrackedKind::WithdrawalTxn(id) => {
859            if let Some(txn) = hashi
860                .bitcoin_mut()
861                .withdrawal_queue
862                .withdrawal_txns
863                .remove(id)
864            {
865                effects.push(Effect::WithdrawalTxnRemoved(Box::new(txn)));
866            }
867        }
868        TrackedKind::TreasuryCap(coin_type) => {
869            hashi.treasury.treasury_caps.remove(coin_type);
870        }
871        TrackedKind::MetadataCap(coin_type) => {
872            hashi.treasury.metadata_caps.remove(coin_type);
873        }
874        TrackedKind::Proposal { executed, id } => {
875            let bucket = if *executed {
876                &mut hashi.proposals.executed
877            } else {
878                &mut hashi.proposals.active
879            };
880            bucket.remove(id);
881        }
882        TrackedKind::TobBucket(key) => {
883            // A destroyed bucket takes its LinkedTable's routing entries
884            // with it; the node deletions in the same transaction then
885            // resolve through the index alone.
886            if let Some(bucket) = hashi.tob.buckets.remove(key) {
887                routing.remove_tob_table(&bucket.certs_id);
888                routing.remove_interior(&bucket.certs_id);
889            }
890        }
891        TrackedKind::TobCert { key, dealer } => {
892            // The bucket may already be retired if its deletion was
893            // processed first within the same transaction.
894            if let Some(bucket) = hashi.tob.buckets.get_mut(key) {
895                bucket.nodes.remove(dealer);
896            }
897        }
898        TrackedKind::Ignored => {}
899    }
900}

902/// Slot equivalent of an already-tracked kind, used to route value-only
903/// mutations whose wrapper is neither in the transaction nor registered.
904fn slot_for_kind(kind: &TrackedKind) -> Option<Slot> {
905    match kind {
906        TrackedKind::HashiRoot
907        | TrackedKind::BitcoinStateField
908        | TrackedKind::DofWrapper { .. } => None,
909        TrackedKind::Member(_) => Some(Slot::Members),
910        TrackedKind::Committee(_) | TrackedKind::CommitteeHandoff(_) => Some(Slot::Committees),
911        TrackedKind::UtxoRecord(_) => Some(Slot::UtxoRecords),
912        TrackedKind::SpentUtxo(_) => Some(Slot::SpentUtxos),
913        TrackedKind::DepositRequest(_) => Some(Slot::DepositRequests),
914        TrackedKind::WithdrawalRequest(_) => Some(Slot::WithdrawalRequests),
915        TrackedKind::WithdrawalTxn(_) => Some(Slot::WithdrawalTxns),
916        TrackedKind::TreasuryCap(_) | TrackedKind::MetadataCap(_) => Some(Slot::Treasury),
917        TrackedKind::Proposal { executed, .. } => Some(if *executed {
918            Slot::ProposalsExecuted
919        } else {
920            Slot::ProposalsActive
921        }),
922        TrackedKind::TobBucket(_) => Some(Slot::Tob),
923        TrackedKind::TobCert { .. } => Some(Slot::TobCerts),
924        TrackedKind::Ignored => None,
925    }
926}

963/// Decode a TOB dealer submission node in whichever layout its own
964/// value type reports — the value side of a linked-table entry is
965/// `0x2::linked_table::Node<address, V>` and `V` is the submission
966/// type — normalized to the stamped form (`timestamp_ms: 0` for bare
967/// submissions). `None` surfaces the write as unrouted.
968fn decode_tob_node(
969    packages: &move_types::PackageVersions,
970    tag: &sui_sdk_types::StructTag,
971    contents: &[u8],
972    id: &Address,
973) -> Option<(
974    Address,
975    move_types::LinkedTableNode<Address, move_types::StampedDealerSubmissionV1>,
976)> {
977    let node_tag = dynamic_field_value_tag(tag)?;
978    if node_tag.address() != &Address::TWO
979        || node_tag.module() != "linked_table"
980        || node_tag.name() != "Node"
981    {
982        return None;
983    }
984    let submission_tag = match node_tag.type_params().get(1) {
985        Some(TypeTag::Struct(value)) => value,
986        _ => return None,
987    };
988    if move_types::DealerSubmissionV1::matches(packages, submission_tag) {
989        decode::<
990            move_types::Field<
991                Address,
992                move_types::LinkedTableNode<Address, move_types::DealerSubmissionV1>,
993            >,
994        >(contents, id)
995        .map(|field| {
996            (
997                field.name,
998                move_types::LinkedTableNode {
999                    prev: field.value.prev,
1000                    next: field.value.next,
1001                    value: move_types::StampedDealerSubmissionV1 {
1002                        submission: field.value.value,
1003                        timestamp_ms: 0,
1004                    },
1005                },
1006            )
1007        })
1008    } else if move_types::StampedDealerSubmissionV1::matches(packages, submission_tag) {
1009        decode::<
1010            move_types::Field<
1011                Address,
1012                move_types::LinkedTableNode<Address, move_types::StampedDealerSubmissionV1>,
1013            >,
1014        >(contents, id)
1015        .map(|field| (field.name, field.value))
1016    } else {
1017        None
1018    }
1019}
