4//! Structural apply of one transaction's changed objects into the
5//! mirror.
6//!
7//! The unit of work is a [`TxView`]: the full post-state of every
8//! object a Hashi-touching transaction wrote, plus the ids it deleted.
9//! [`apply_transaction`] routes each object by ownership (see
10//! [`super::route`]) and updates the matching `types::Hashi` map, so
11//! the mirror no longer depends on events — a Move mutation with no
12//! `event::emit` still arrives here as an object write.
13//!
14//! Side effects that used to ride the event path (limiter advance,
15//! notifications) are returned as [`Effect`]s derived from state
16//! transitions; the caller owns acting on them.
17

41/// One successful transaction's object changes, decoded from the gRPC
42/// payload into `sui_sdk_types` values.
43#[derive(Debug)]
44pub(super) struct TxView {
45    pub checkpoint: u64,
46    /// Zero-based position within the checkpoint. Together with
47    /// `checkpoint` this is the transaction's total order on the chain;
48    /// the watcher ratchets over `(checkpoint, transaction_index)` so a
49    /// transaction delivered by both replay and the live stream applies
50    /// exactly once.
51    pub transaction_index: u64,
52    /// The transaction's checkpoint timestamp; drives the limiter
53    /// advance and the withdrawal duration metrics.
54    pub timestamp_ms: u64,
55    pub changes: Vec<TxChange>,
56    /// Changed objects whose `output_state` was missing, unrecognized,
57    /// or introduced by a newer proto than this build knows. The mirror
58    /// cannot know what happened to them; apply surfaces each through
59    /// the unrouted tripwire rather than dropping it silently.
60    pub unknown_states: Vec<(Address, &'static str)>,
61}
62
63impl TxView {
64    /// The transaction's position in the chain's total order.
65    pub(super) fn position(&self) -> (u64, u64) {
66        (self.checkpoint, self.transaction_index)
67    }
68}

70#[derive(Debug)]
71pub(super) enum TxChange {
72    /// A created or mutated object, at its post-transaction state.
73    Written(Box<Object>),
74    /// The object no longer exists after this transaction (deleted or
75    /// wrapped). Ids are never reused on Sui, so removal needs no
76    /// version guard: within each ordered source a write for this id
77    /// always precedes its deletion, and re-applying an overlap replays
78    /// the write and the deletion in order.
79    Deleted { id: Address },
80}
81
82/// Objects keyed by (id, version), BCS-decoded from a proto
83/// `ObjectSet`. The set carries both input and output versions of the
84/// same id, so the version is part of the key; each changed object is
85/// paired with its output version.
86type ObjectPool = HashMap<(Address, u64), Object>;
87
88/// Decode a transaction's proto `ObjectSet` into an [`ObjectPool`].
89/// Requires the `bcs` field in the read mask.
90fn decode_object_pool(set: Option<&proto::ObjectSet>) -> anyhow::Result<ObjectPool> {
91    let mut objects = ObjectPool::new();
92    let Some(set) = set else {
93        return Ok(objects);
94    };
95    for obj in &set.objects {
96        let Some(bytes) = obj.bcs.as_ref().and_then(|b| b.value.as_ref()) else {
97            continue;
98        };
99        let object: Object =
100            bcs::from_bytes(bytes).context("failed to BCS-decode an object from an object set")?;
101        objects.insert((object.object_id(), object.version()), object);
102    }
103    Ok(objects)
104}

106impl TxView {
107    /// Decode a proto `ExecutedTransaction` (as delivered by the
108    /// filtered transaction stream) into a `TxView`, sourcing
109    /// post-state objects from its per-transaction object set. Returns
110    /// `Ok(None)` for transactions that did not execute successfully —
111    /// their object changes are gas-only and carry no Hashi state.
112    pub(super) fn from_proto(tx: &proto::ExecutedTransaction) -> anyhow::Result<Option<Self>> {
113        let effects = tx
114            .effects
115            .as_ref()
116            .context("ExecutedTransaction is missing effects")?;
117        // Check success before decoding the object set: a failed
118        // transaction's changes are gas-only and discarded, so its
119        // objects are neither worth decoding nor worth failing on.
120        if !effects
121            .status
122            .as_ref()
123            .context("TransactionEffects is missing status")?
124            .success()
125        {
126            return Ok(None);
127        }
128        let mut pool = decode_object_pool(tx.objects.as_ref())?;
129
130        let mut changes = Vec::new();
131        let mut unknown_states = Vec::new();
132        for changed in &effects.changed_objects {
133            let id: Address = changed
134                .object_id
135                .as_deref()
136                .context("ChangedObject is missing object_id")?
137                .parse()
138                .context("invalid ChangedObject object_id")?;
139            let state = changed
140                .output_state
141                .and_then(|s| OutputObjectState::try_from(s).ok())
142                .unwrap_or(OutputObjectState::Unknown);
143            match state {
144                OutputObjectState::ObjectWrite => {
145                    let version = changed
146                        .output_version
147                        .context("written ChangedObject is missing output_version")?;
148                    let object = pool.remove(&(id, version)).with_context(|| {
149                        let mut available: Vec<String> = pool
150                            .keys()
151                            .map(|(id, version)| format!("{id}@v{version}"))
152                            .collect();
153                        available.sort();
154                        available.truncate(8);
155                        format!(
156                            "object {id} v{version} not in the object set ({} present: \
157                             [{}]) — the read mask needs objects.objects.bcs",
158                            pool.len(),
159                            available.join(", ")
160                        )
161                    })?;
162                    changes.push(TxChange::Written(Box::new(object)));
163                }
164                OutputObjectState::DoesNotExist => changes.push(TxChange::Deleted { id }),
165                // Package objects are not mirrored (the upgrade effect
166                // derives from the root's `UpgradeCap` change), so don't
167                // demand one from the object set — failing here would
168                // kill the stream on every upgrade if a server ever
169                // omitted the package object.
170                OutputObjectState::PackageWrite => {}
171                // Accumulator writes carry no object contents and no
172                // Hashi state.
173                OutputObjectState::AccumulatorWrite => {}
174                // Missing, unrecognized, or newer-than-this-build
175                // states: the one place a change could slip through
176                // undecoded, so it feeds the tripwire instead.
177                state => unknown_states.push((id, state.as_str_name())),
178            }
179        }
180
181        // The bare prost accessors default missing fields to zero; a
182        // zeroed position would fall below the delivery ratchet and the
183        // transaction would be dropped silently, and a zeroed timestamp
184        // would corrupt the limiter window. Fail the frame instead —
185        // the watcher reconnects and replays.
186        Ok(Some(Self {
187            checkpoint: tx
188                .checkpoint_opt()
189                .context("ExecutedTransaction is missing checkpoint")?,
190            transaction_index: tx
191                .transaction_index_opt()
192                .context("ExecutedTransaction is missing transaction_index")?,
193            timestamp_ms: proto_to_timestamp_ms(
194                tx.timestamp
195                    .context("ExecutedTransaction is missing timestamp")?,
196            )
197            .context("invalid ExecutedTransaction timestamp")?,
198            changes,
199            unknown_states,
200        }))
201    }
202}

204/// A side effect derived from a state transition during apply. The
205/// caller (the watcher) owns notifications, the limiter, and metrics.
206#[derive(Debug)]
207pub(super) enum Effect {
208    /// A member's `MemberInfo` entry was written.
209    ValidatorInfoUpdated(Address),
210    /// The root's `pending_epoch_change` became set for this epoch.
211    ReconfigStarted(u64),
212    /// The root's `pending_epoch_change` for this epoch was cleared without
213    /// activating: the reconfiguration was aborted.
214    ReconfigAborted(u64),
215    /// The root's `UpgradeCap` now points at a new package id. The
216    /// cap's counter starts at 1 on publish and increments in lockstep
217    /// with the package chain, so `version` is the new package's
218    /// version. Apply extends the version map itself (under the same
219    /// state guard); this effect is the caller's signal to log.
220    PackageUpgraded { package: Address, version: u64 },
221    /// An active deposit request was created or updated.
222    DepositRequestUpserted {
223        request_id: Address,
224        outpoint: bitcoin::OutPoint,
225    },
226    /// A deposit request left the active queue.
227    DepositRequestRemoved(Address),
228    /// A withdrawal transaction transitioned to fully signed in this
229    /// transaction (2-of-2 witness complete). Drives the local limiter
230    /// and the pick-to-sign metric.
231    WithdrawalTxnFullySigned(Box<types::WithdrawalTransaction>),
232    /// A withdrawal transaction's `confirmed_timestamp_ms` went from
233    /// `None` to `Some` in this transaction: the v2 in-place confirm,
234    /// which leaves the txn in the in-flight bag until a later archival
235    /// GC moves it out. Drives the sign-to-confirm and total duration
236    /// metrics in the deferred-archival world (under v1 bytecode the
237    /// confirm moves the txn instead, so this never fires there).
238    WithdrawalTxnConfirmed(Box<types::WithdrawalTransaction>),
239    /// A withdrawal transaction left the in-flight bag (confirmed and
240    /// moved to the historical record, or deleted).
241    WithdrawalTxnRemoved(Box<types::WithdrawalTransaction>),
242}

244#[derive(Debug, Default)]
245pub(super) struct ApplyOutcome {
246    pub effects: Vec<Effect>,
247    /// Objects owned by a UID we don't recognize, or carrying a type no
248    /// handler decodes. Every entry is a potential mirror gap; the
249    /// caller surfaces them as the lossless-coverage tripwire.
250    pub unrouted: Vec<(Address, String)>,
251}

253/// Apply one successful transaction's changes to the mirror.
254///
255/// Order matters within a transaction: the root and `BitcoinState` are
256/// applied first (they carry the container UIDs), then DOF wrappers
257/// (they extend the routing table), then remaining writes (committee
258/// handoffs last, because their certificate references a committee that
259/// may be written in the same transaction), then deletions, which route
260/// through the object index rather than the routing table so no
261/// pre-deletion contents are needed.
262pub(super) fn apply_transaction(
263    hashi: &mut types::Hashi,
264    packages: &mut move_types::PackageVersions,
265    routing: &mut RoutingTable,
266    index: &mut ObjectIndex,
267    tx: &TxView,
268) -> ApplyOutcome {
269    let mut out = ApplyOutcome::default();
270
271    for (id, state) in &tx.unknown_states {
272        out.unrouted.push((*id, format!("output_state {state}")));
273    }
274
275    // Pass 1: the Hashi root and the BitcoinState dynamic field.
276    for change in &tx.changes {
277        let TxChange::Written(obj) = change else {
278            continue;
279        };
280        if obj.object_id() == routing.hashi_id() {
281            apply_root(hashi, packages, routing, index, obj, &mut out);
282        } else if obj.object_id() == routing.bitcoin_state_field_id() {
283            apply_bitcoin_state(routing, index, obj, &mut out);
284        }
285    }
286
287    // Pass 2: DOF wrappers, so values created in the same transaction
288    // can resolve their container.
289    for change in &tx.changes {
290        let TxChange::Written(obj) = change else {
291            continue;
292        };
293        let Some(struct_) = obj.as_struct() else {
294            continue;
295        };
296        if is_dof_wrapper(struct_.object_type()) {
297            apply_wrapper(routing, index, obj, &mut out);
298        }
299    }
300
301    // Pass 3: all other writes, in two rounds. Writes that register an
302    // interior container go first (a tob entry or per-user request bag
303    // is rewritten in the same transaction that touches its children,
304    // because the child mutation bumps the embedded container's size),
305    // so those children route as known-ignored instead of unrouted.
306    // Committee handoffs are deferred past both rounds.
307    let mut handoffs: Vec<&Object> = Vec::new();
308    let mut deferred: Vec<&Object> = Vec::new();
309    for change in &tx.changes {
310        let TxChange::Written(obj) = change else {
311            continue;
312        };
313        let id = obj.object_id();
314        if id == routing.hashi_id() || id == routing.bitcoin_state_field_id() {
315            continue;
316        }
317        let Some(struct_) = obj.as_struct() else {
318            // Package writes: the upgrade effect is derived from the
319            // root's UpgradeCap change, not the package object.
320            continue;
321        };
322        if is_dof_wrapper(struct_.object_type()) {
323            continue;
324        }
325        if is_committee_handoff(struct_.object_type()) {
326            handoffs.push(obj);
327            continue;
328        }
329        let registers_interior = matches!(obj.owner(), Owner::Object(parent)
330        if matches!(
331            routing.resolve_owner(parent),
332            Some(Slot::Tob | Slot::UserRequests)
333        ));
334        if registers_interior {
335            apply_write(hashi, packages, routing, index, obj, &mut out);
336        } else {
337            deferred.push(obj);
338        }
339    }
340    for obj in deferred {
341        apply_write(hashi, packages, routing, index, obj, &mut out);
342    }
343
344    // Pass 4: committee handoffs, after any same-transaction committee.
345    for obj in handoffs {
346        apply_write(hashi, packages, routing, index, obj, &mut out);
347    }
348
349    // Pass 5: deletions, routed by what the mirror knows the id to be.
350    for change in &tx.changes {
351        let TxChange::Deleted { id } = change else {
352            continue;
353        };
354        let Some(entry) = index.remove(id) else {
355            continue;
356        };
357        retire(hashi, routing, id, &entry.kind, &mut out.effects);
358    }
359
360    out
361}

368fn apply_root(
369    hashi: &mut types::Hashi,
370    packages: &mut move_types::PackageVersions,
371    routing: &mut RoutingTable,
372    index: &mut ObjectIndex,
373    obj: &Object,
374    out: &mut ApplyOutcome,
375) {
376    let id = obj.object_id();
377    if index.is_stale_write(&id, obj.version()) {
378        return;
379    }
380    let Some(struct_) = obj.as_struct() else {
381        return;
382    };
383    let root: move_types::Hashi = match bcs::from_bytes(struct_.contents()) {
384        Ok(root) => root,
385        Err(e) => {
386            tracing::error!("failed to decode the Hashi root object: {e}");
387            return;
388        }
389    };
390    routing.set_root_containers(&root);
391
392    let move_types::Hashi {
393        id: _,
394        committees,
395        config,
396        epoch_config,
397        versioning,
398        treasury: _,
399        proposals: _,
400        tob: _,
401        num_consumed_presigs,
402    } = root;
403
404    let new_config = super::convert_move_config(config, versioning);
405    if let (Some(old), Some(new)) = (&hashi.config.upgrade_cap, &new_config.upgrade_cap)
406        && old.package != new.package
407    {
408        out.effects.push(Effect::PackageUpgraded {
409            package: new.package,
410            version: new.version,
411        });
412    }
413    // The cap always points at the latest package version; record it
414    // before pass 3 so objects of a just-upgraded package route (or
415    // trip the mirror-gap tripwire) within the same transaction.
416    if let Some(cap) = &new_config.upgrade_cap {
417        packages.insert(cap.version, cap.package);
418    }
419    hashi.config = new_config;
420    hashi.epoch_config = epoch_config;
421    hashi.num_consumed_presigs = num_consumed_presigs;
422
423    let old_pending = hashi.committees.pending_epoch_change();
424    let new_pending = committees.pending_epoch_change.as_ref().map(|p| p.epoch);
425    if let Some(epoch) = new_pending
426        && old_pending != new_pending
427    {
428        out.effects.push(Effect::ReconfigStarted(epoch));
429    }
430    // Pending cleared without activating. `end_reconfig` advances the epoch,
431    // or, for the genesis DKG on a fresh network where the epoch stays 0,
432    // sets the MPC key; an unchanged epoch and key mean `abort_reconfig`.
433    if let Some(epoch) = old_pending
434        && new_pending.is_none()
435        && committees.epoch == hashi.committees.epoch()
436        && committees.mpc_public_key.as_slice() == hashi.committees.mpc_public_key()
437    {
438        out.effects.push(Effect::ReconfigAborted(epoch));
439    }
440    hashi
441        .committees
442        .set_epoch(committees.epoch)
443        .set_pending_epoch_change(new_pending)
444        .set_mpc_public_key(committees.mpc_public_key);
445
446    index.record(id, obj.version(), TrackedKind::HashiRoot);
447}

449fn apply_bitcoin_state(
450    routing: &mut RoutingTable,
451    index: &mut ObjectIndex,
452    obj: &Object,
453    _out: &mut ApplyOutcome,
454) {
455    let id = obj.object_id();
456    if index.is_stale_write(&id, obj.version()) {
457        return;
458    }
459    let Some(struct_) = obj.as_struct() else {
460        return;
461    };
462    let field: move_types::Field<move_types::BitcoinStateKey, move_types::BitcoinState> =
463        match bcs::from_bytes(struct_.contents()) {
464            Ok(field) => field,
465            Err(e) => {
466                tracing::error!("failed to decode the BitcoinState dynamic field: {e}");
467                return;
468            }
469        };
470    routing.set_bitcoin_state_containers(&field.value);
471    index.record(id, obj.version(), TrackedKind::BitcoinStateField);
472}

474fn apply_wrapper(
475    routing: &mut RoutingTable,
476    index: &mut ObjectIndex,
477    obj: &Object,
478    out: &mut ApplyOutcome,
479) {
480    let id = obj.object_id();
481    let Owner::Object(container) = obj.owner() else {
482        out.unrouted
483            .push((id, "DOF wrapper not object-owned".into()));
484        return;
485    };
486    // Our container set is closed and known a priori (root and
487    // BitcoinState inner UIDs), so a wrapper under an unknown container
488    // belongs to some foreign object a root-touching transaction also
489    // mutated. Don't register or track it.
490    if routing.slot_of_container(container).is_none() {
491        tracing::debug!(wrapper = %id, container = %container,
492            "ignoring a DOF wrapper under a foreign container");
493        return;
494    }
495    // Registration is idempotent, so it runs even for stale replayed
496    // writes — the mapping must exist for value routing regardless.
497    routing.register_wrapper(id, *container);
498    if index.is_stale_write(&id, obj.version()) {
499        return;
500    }
501    index.record(
502        id,
503        obj.version(),
504        TrackedKind::DofWrapper {
505            container: *container,
506        },
507    );
508}

510fn apply_write(
511    hashi: &mut types::Hashi,
512    packages: &move_types::PackageVersions,
513    routing: &mut RoutingTable,
514    index: &mut ObjectIndex,
515    obj: &Object,
516    out: &mut ApplyOutcome,
517) {
518    let id = obj.object_id();
519    let Some(struct_) = obj.as_struct() else {
520        return;
521    };
522    let slot = match obj.owner() {
523        // Address-owned objects (gas coins, user funds) are not part
524        // of the mirrored state.
525        Owner::Address(_) => return,
526        // Immutable objects (packages) carry no mirrored state either.
527        Owner::Immutable => return,
528        Owner::Object(parent) => match routing.resolve_owner(parent) {
529            Some(slot) => Some(slot),
530            // A value whose wrapper isn't in this transaction and isn't
531            // registered: fall back to what the mirror already knows
532            // the id to be.
533            None => index.get(&id).and_then(|e| slot_for_kind(&e.kind)),
534        },
535        // The only hashi shared object is the root, handled in pass 1;
536        // other shared objects a root-touching transaction mutates
537        // (e.g. the system CoinRegistry) are foreign state.
538        _ => {
539            tracing::debug!(object = %id, "ignoring a foreign shared object");
540            return;
541        }
542    };
543    let Some(slot) = slot else {
544        // An unroutable object is a mirror gap only when it carries a
545        // hashi-package type (from any published version — the boot
546        // scrape seeds the full history, and upgrades extend it);
547        // foreign objects under foreign owners are expected companions
548        // of root-touching transactions.
549        if packages.contains(struct_.object_type().address()) {
550            out.unrouted.push((id, struct_.object_type().to_string()));
551        } else {
552            tracing::debug!(object = %id, r#type = %struct_.object_type(),
553                "ignoring a foreign object-owned write");
554        }
555        return;
556    };
557
558    if index.is_stale_write(&id, obj.version()) {
559        return;
560    }
561
562    let tag = struct_.object_type();
563    let contents = struct_.contents();
564    let new_kind = match slot {
565        Slot::Members => decode::<move_types::Field<Address, move_types::MemberInfo>>(
566            contents, &id,
567        )
568        .map(|field| {
569            let info = super::convert_move_member_info(field.value);
570            let validator = info.validator_address;
571            hashi.committees.update_validator(info);
572            out.effects.push(Effect::ValidatorInfoUpdated(validator));
573            TrackedKind::Member(validator)
574        }),
575        Slot::Committees => {
576            // The committees bag is heterogeneous: per-epoch committees
577            // and committee handoffs, distinguished by value type.
578            if is_committee_handoff(tag) {
579                decode::<
580                    move_types::Field<
581                        move_types::CommitteeHandoffKey,
582                        move_types::CommitteeHandoff,
583                    >,
584                >(contents, &id)
585                .and_then(|field| {
586                    let from_epoch = field.name.epoch;
587                    apply_committee_handoff(hashi, from_epoch, field.value)
588                        .then_some(TrackedKind::CommitteeHandoff(from_epoch))
589                })
590            } else if move_types::is_field_with_value(tag, "committee", "Committee") {
591                decode::<move_types::Field<u64, move_types::Committee>>(contents, &id).map(
592                    |field| {
593                        let epoch = field.name;
594                        hashi
595                            .committees
596                            .committees_mut()
597                            .insert(epoch, super::convert_move_committee(field.value.clone()));
598                        // Keep the verbatim on-chain committee too: it is
599                        // the only form a `CommitteeTransitionRequest`
600                        // may embed (see `CommitteeSet::raw_committees`).
601                        hashi
602                            .committees
603                            .raw_committees_mut()
604                            .insert(epoch, field.value);
605                        TrackedKind::Committee(epoch)
606                    },
607                )
608            } else {
609                None
610            }
611        }
612        Slot::UtxoRecords => decode::<move_types::Field<types::UtxoId, move_types::UtxoRecord>>(
613            contents, &id,
614        )
615        .map(|field| {
616            let utxo_id = field.value.utxo.id;
617            hashi
618                .bitcoin_mut()
619                .utxo_pool
620                .utxo_records
621                .insert(utxo_id, field.value);
622            TrackedKind::UtxoRecord(utxo_id)
623        }),
624        Slot::SpentUtxos => {
625            // A tombstone is created once and never rewritten, moved,
626            // or deleted, so there is no version to guard a replay
627            // with and nothing for a deletion to retire: recording it
628            // would only make the index grow with the bag, the very
629            // growth the mirror stopped tracking the bag to avoid.
630            return;
631        }
632        Slot::DepositRequests => {
633            decode::<move_types::DepositRequest>(contents, &id).map(|request| {
634                let request_id = request.id;
635                out.effects.push(Effect::DepositRequestUpserted {
636                    request_id,
637                    outpoint: request.utxo.id.into(),
638                });
639                hashi
640                    .bitcoin_mut()
641                    .deposit_queue
642                    .requests
643                    .insert(request_id, request);
644                TrackedKind::DepositRequest(request_id)
645            })
646        }
647        Slot::WithdrawalRequests => {
648            decode::<move_types::WithdrawalRequest>(contents, &id).map(|request| {
649                let request_id = request.id;
650                hashi
651                    .bitcoin_mut()
652                    .withdrawal_queue
653                    .requests
654                    .insert(request_id, request);
655                TrackedKind::WithdrawalRequest(request_id)
656            })
657        }
658        Slot::WithdrawalTxns => {
659            decode::<move_types::WithdrawalTransaction>(contents, &id).map(|txn| {
660                let txn_id = txn.id;
661                let withdrawal_queue = &mut hashi.bitcoin_mut().withdrawal_queue;
662                let old = withdrawal_queue.withdrawal_txns.get(&txn_id);
663                let was_fully_signed = old.is_some_and(|t| t.is_fully_signed());
664                let was_confirmed = old.is_some_and(|t| t.confirmed_timestamp_ms.is_some());
665                if !was_fully_signed && txn.is_fully_signed() {
666                    out.effects
667                        .push(Effect::WithdrawalTxnFullySigned(Box::new(txn.clone())));
668                }
669                if !was_confirmed && txn.is_confirmed() {
670                    out.effects
671                        .push(Effect::WithdrawalTxnConfirmed(Box::new(txn.clone())));
672                }
673                withdrawal_queue.withdrawal_txns.insert(txn_id, txn);
674                TrackedKind::WithdrawalTxn(txn_id)
675            })
676        }
677        Slot::Treasury => {
678            let type_tag = TypeTag::Struct(Box::new(tag.clone()));
679            if let Some(cap) = types::TreasuryCap::try_from_contents(&type_tag, contents) {
680                let coin_type = cap.coin_type.clone();
681                hashi.treasury.treasury_caps.insert(coin_type.clone(), cap);
682                Some(TrackedKind::TreasuryCap(coin_type))
683            } else if let Some(cap) = types::MetadataCap::try_from_contents(&type_tag, contents) {
684                let coin_type = cap.coin_type.clone();
685                hashi.treasury.metadata_caps.insert(coin_type.clone(), cap);
686                Some(TrackedKind::MetadataCap(coin_type))
687            } else {
688                None
689            }
690        }
691        Slot::ProposalsActive | Slot::ProposalsExecuted => {
692            let type_tag = TypeTag::Struct(Box::new(tag.clone()));
693            super::decode_proposal(&type_tag, contents).map(|proposal| {
694                let executed = slot == Slot::ProposalsExecuted;
695                let proposal_id = proposal.id;
696                let bucket = if executed {
697                    &mut hashi.proposals.executed
698                } else {
699                    &mut hashi.proposals.active
700                };
701                bucket.insert(proposal_id, proposal);
702                TrackedKind::Proposal {
703                    executed,
704                    id: proposal_id,
705                }
706            })
707        }
708        Slot::Tob => {
709            // A bucket Field is rewritten on every submission (the
710            // embedded LinkedTable's head/tail/size live in it); the
711            // nodes themselves arrive as TobCerts writes. The two
712            // bucket structs are BCS-identical, so one decode serves
713            // both; the Field's value-side type selects the node
714            // layout, and an unrecognized one falls through to the
715            // unrouted tripwire rather than misdecoding.
716            tob_bucket_layout(packages, tag).and_then(|layout| {
717                decode::<move_types::Field<move_types::TobKey, move_types::EpochCertsV1>>(
718                    contents, &id,
719                )
720                .map(|field| {
721                    let key = field.name;
722                    let certs_id = field.value.certs.id;
723                    routing.register_interior(certs_id, Slot::TobCerts);
724                    routing.register_tob_table(certs_id, key);
725                    let bucket = hashi
726                        .tob
727                        .buckets
728                        .entry(key)
729                        .or_insert_with(|| types::TobBucket {
730                            layout,
731                            certs_id,
732                            head: None,
733                            size: 0,
734                            nodes: std::collections::BTreeMap::new(),
735                        });
736                    bucket.layout = layout;
737                    bucket.certs_id = certs_id;
738                    bucket.head = field.value.certs.head;
739                    bucket.size = field.value.certs.size;
740                    TrackedKind::TobBucket(key)
741                })
742            })
743        }
744        Slot::TobCerts => {
745            // The owner is the bucket's LinkedTable UID; a node write is
746            // a new dealer submission, or a neighbor's prev/next link
747            // update from a same-transaction insertion. The node's own
748            // value type says whether the submission is bare or stamped;
749            // both are stored in the normalized stamped form.
750            let parent = match obj.owner() {
751                Owner::Object(parent) => *parent,
752                _ => return,
753            };
754            routing.tob_key_of_table(&parent).and_then(|key| {
755                decode_tob_node(packages, tag, contents, &id).map(|(dealer, node)| {
756                    if let Some(bucket) = hashi.tob.buckets.get_mut(&key) {
757                        bucket.nodes.insert(dealer, node);
758                    }
759                    TrackedKind::TobCert { key, dealer }
760                })
761            })
762        }
763        Slot::UserRequests => {
764            // Same shape: each per-user value is a Bag with its own UID.
765            decode::<move_types::Field<Address, move_types::Bag>>(contents, &id).map(|field| {
766                routing.register_interior(field.value.id, Slot::UserRequestBag);
767                TrackedKind::Ignored
768            })
769        }
770        Slot::DepositProcessed
771        | Slot::WithdrawalProcessed
772        | Slot::ConfirmedTxns
773        | Slot::UserRequestBag => Some(TrackedKind::Ignored),
774    };
775
776    let Some(new_kind) = new_kind else {
777        out.unrouted.push((id, struct_.object_type().to_string()));
778        return;
779    };
780
781    // A write whose meaning changed is a move between containers (e.g.
782    // a proposal executing, a withdrawal txn confirming): retire the
783    // old mirror entry before the new one takes over.
784    if let Some(old) = index.get(&id)
785        && old.kind != new_kind
786    {
787        let old_kind = old.kind.clone();
788        retire(hashi, routing, &id, &old_kind, &mut out.effects);
789    }
790
791    index.record(id, obj.version(), new_kind);
792}

794/// Insert a committee handoff, converting through the raw Move
795/// committee its certificate signs. Returns false when the referenced
796/// committee is unknown (the mirror can't validate the cert shape).
797///
798/// The committee embedded in the stored transition is the verbatim
799/// on-chain one: Move's `submit_committee_handoff` verified this cert
800/// over a `CommitteeTransitionRequest` rebuilt from the stored
801/// committee, so those are the only bytes the cert matches (see
802/// `CommitteeSet::raw_committees`).
803fn apply_committee_handoff(
804    hashi: &mut types::Hashi,
805    from_epoch: u64,
806    handoff: move_types::CommitteeHandoff,
807) -> bool {
808    let Some(raw) = hashi.committees.raw_committee(handoff.next_epoch).cloned() else {
809        tracing::error!(
810            from_epoch,
811            next_epoch = handoff.next_epoch,
812            "committee handoff references a committee the mirror doesn't hold"
813        );
814        return false;
815    };
816    match super::convert_move_committee_handoff(handoff, raw) {
817        Ok(signed) => {
818            hashi
819                .committees
820                .committee_handoffs_mut()
821                .insert(from_epoch, signed);
822            true
823        }
824        Err(e) => {
825            tracing::error!(from_epoch, "invalid committee handoff: {e}");
826            false
827        }
828    }
829}

831/// Remove whatever the mirror holds for a retired object id. Used both
832/// for deletions and for cross-container moves.
833fn retire(
834    hashi: &mut types::Hashi,
835    routing: &mut RoutingTable,
836    id: &Address,
837    kind: &TrackedKind,
838    effects: &mut Vec<Effect>,
839) {
840    match kind {
841        TrackedKind::HashiRoot | TrackedKind::BitcoinStateField => {
842            tracing::error!(?kind, "refusing to retire a root object");
843        }
844        TrackedKind::DofWrapper { .. } => {
845            routing.remove_wrapper(id);
846        }
847        TrackedKind::Member(validator) => {
848            hashi.committees.remove_validator(validator);
849        }
850        TrackedKind::Committee(epoch) => {
851            hashi.committees.committees_mut().remove(epoch);
852            hashi.committees.raw_committees_mut().remove(epoch);
853        }
854        TrackedKind::CommitteeHandoff(epoch) => {
855            hashi.committees.committee_handoffs_mut().remove(epoch);
856        }
857        TrackedKind::UtxoRecord(utxo_id) => {
858            hashi.bitcoin_mut().utxo_pool.utxo_records.remove(utxo_id);
859        }
860        TrackedKind::DepositRequest(id) => {
861            hashi.bitcoin_mut().deposit_queue.requests.remove(id);
862            effects.push(Effect::DepositRequestRemoved(*id));
863        }
864        TrackedKind::WithdrawalRequest(id) => {
865            hashi.bitcoin_mut().withdrawal_queue.requests.remove(id);
866        }
867        TrackedKind::WithdrawalTxn(id) => {
868            if let Some(txn) = hashi
869                .bitcoin_mut()
870                .withdrawal_queue
871                .withdrawal_txns
872                .remove(id)
873            {
874                effects.push(Effect::WithdrawalTxnRemoved(Box::new(txn)));
875            }
876        }
877        TrackedKind::TreasuryCap(coin_type) => {
878            hashi.treasury.treasury_caps.remove(coin_type);
879        }
880        TrackedKind::MetadataCap(coin_type) => {
881            hashi.treasury.metadata_caps.remove(coin_type);
882        }
883        TrackedKind::Proposal { executed, id } => {
884            let bucket = if *executed {
885                &mut hashi.proposals.executed
886            } else {
887                &mut hashi.proposals.active
888            };
889            bucket.remove(id);
890        }
891        TrackedKind::TobBucket(key) => {
892            // A destroyed bucket takes its LinkedTable's routing entries
893            // with it; the node deletions in the same transaction then
894            // resolve through the index alone.
895            if let Some(bucket) = hashi.tob.buckets.remove(key) {
896                routing.remove_tob_table(&bucket.certs_id);
897                routing.remove_interior(&bucket.certs_id);
898            }
899        }
900        TrackedKind::TobCert { key, dealer } => {
901            // The bucket may already be retired if its deletion was
902            // processed first within the same transaction.
903            if let Some(bucket) = hashi.tob.buckets.get_mut(key) {
904                bucket.nodes.remove(dealer);
905            }
906        }
907        TrackedKind::Ignored => {}
908    }
909}

911/// Slot equivalent of an already-tracked kind, used to route value-only
912/// mutations whose wrapper is neither in the transaction nor registered.
913fn slot_for_kind(kind: &TrackedKind) -> Option<Slot> {
914    match kind {
915        TrackedKind::HashiRoot
916        | TrackedKind::BitcoinStateField
917        | TrackedKind::DofWrapper { .. } => None,
918        TrackedKind::Member(_) => Some(Slot::Members),
919        TrackedKind::Committee(_) | TrackedKind::CommitteeHandoff(_) => Some(Slot::Committees),
920        TrackedKind::UtxoRecord(_) => Some(Slot::UtxoRecords),
921        TrackedKind::DepositRequest(_) => Some(Slot::DepositRequests),
922        TrackedKind::WithdrawalRequest(_) => Some(Slot::WithdrawalRequests),
923        TrackedKind::WithdrawalTxn(_) => Some(Slot::WithdrawalTxns),
924        TrackedKind::TreasuryCap(_) | TrackedKind::MetadataCap(_) => Some(Slot::Treasury),
925        TrackedKind::Proposal { executed, .. } => Some(if *executed {
926            Slot::ProposalsExecuted
927        } else {
928            Slot::ProposalsActive
929        }),
930        TrackedKind::TobBucket(_) => Some(Slot::Tob),
931        TrackedKind::TobCert { .. } => Some(Slot::TobCerts),
932        TrackedKind::Ignored => None,
933    }
934}

971/// Decode a TOB dealer submission node in whichever layout its own
972/// value type reports — the value side of a linked-table entry is
973/// `0x2::linked_table::Node<address, V>` and `V` is the submission
974/// type — normalized to the stamped form (`timestamp_ms: 0` for bare
975/// submissions). `None` surfaces the write as unrouted.
976fn decode_tob_node(
977    packages: &move_types::PackageVersions,
978    tag: &sui_sdk_types::StructTag,
979    contents: &[u8],
980    id: &Address,
981) -> Option<(
982    Address,
983    move_types::LinkedTableNode<Address, move_types::StampedDealerSubmissionV1>,
984)> {
985    let node_tag = dynamic_field_value_tag(tag)?;
986    if node_tag.address() != &Address::TWO
987        || node_tag.module() != "linked_table"
988        || node_tag.name() != "Node"
989    {
990        return None;
991    }
992    let submission_tag = match node_tag.type_params().get(1) {
993        Some(TypeTag::Struct(value)) => value,
994        _ => return None,
995    };
996    if move_types::DealerSubmissionV1::matches(packages, submission_tag) {
997        decode::<
998            move_types::Field<
999                Address,
1000                move_types::LinkedTableNode<Address, move_types::DealerSubmissionV1>,
1001            >,
1002        >(contents, id)
1003        .map(|field| {
1004            (
1005                field.name,
1006                move_types::LinkedTableNode {
1007                    prev: field.value.prev,
1008                    next: field.value.next,
1009                    value: move_types::StampedDealerSubmissionV1 {
1010                        submission: field.value.value,
1011                        timestamp_ms: 0,
1012                    },
1013                },
1014            )
1015        })
1016    } else if move_types::StampedDealerSubmissionV1::matches(packages, submission_tag) {
1017        decode::<
1018            move_types::Field<
1019                Address,
1020                move_types::LinkedTableNode<Address, move_types::StampedDealerSubmissionV1>,
1021            >,
1022        >(contents, id)
1023        .map(|field| (field.name, field.value))
1024    } else {
1025        None
1026    }
1027}
