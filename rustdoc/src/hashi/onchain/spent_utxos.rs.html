4//! Live membership lookups against the on-chain `spent_utxos` bag.
5//!
6//! The bag holds one `UtxoId -> spent_epoch` tombstone per UTXO the
7//! bridge has ever spent, kept permanently as replay protection. It
8//! only grows, and on a busy network it runs to millions of entries,
9//! so the object mirror deliberately does not track it (see
10//! `route::Slot::SpentUtxos`). The two replay checks that need
11//! membership — approving a deposit and confirming an approved one —
12//! ask the fullnode for the single entry instead.
13

28/// Whether the `spent_utxos` bag holds an entry for `utxo_id`.
29///
30/// A bag entry is a dynamic field whose object id derives from the bag
31/// id and the BCS-encoded key, so one `GetObject` decides membership:
32/// an object means spent, `NotFound` means not spent, and any other
33/// status is an error for the caller to retry. A wrong "not spent"
34/// costs one rejected transaction, never a double-spend — every entry
35/// function that inserts a UTXO re-checks the bag itself.
36pub(super) async fn lookup_spent_utxo(
37    mut client: Client,
38    spent_utxos_id: Address,
39    package_id: Address,
40    utxo_id: &UtxoId,
41) -> Result<bool> {
42    let field_id = spent_utxo_field_id(spent_utxos_id, package_id, utxo_id);
43    let response = client
44        .ledger_client()
45        .get_object(
46            GetObjectRequest::new(&field_id)
47                .with_read_mask(FieldMask::from_paths([Object::path_builder().object_id()])),
48        )
49        .await;
50    match response {
51        Ok(_) => Ok(true),
52        Err(status) if status.code() == tonic::Code::NotFound => Ok(false),
53        Err(status) => Err(anyhow::Error::new(status).context(format!(
54            "failed to look up the spent_utxos entry for {utxo_id:?}"
55        ))),
56    }
57}

59/// The derived object id of the `spent_utxos` bag entry keyed by
60/// `utxo_id`. The key type is `utxo::UtxoId` from the original package:
61/// a struct keeps its defining package's address through upgrades.
62fn spent_utxo_field_id(spent_utxos_id: Address, package_id: Address, utxo_id: &UtxoId) -> Address {
63    let key_type = TypeTag::Struct(Box::new(StructTag::new(
64        package_id,
65        Identifier::from_static("utxo"),
66        Identifier::from_static("UtxoId"),
67        vec![],
68    )));
69    spent_utxos_id.derive_dynamic_child_id(
70        &key_type,
71        &utxo_id
72            .to_bcs()
73            .expect("UtxoId BCS serialization cannot fail"),
74    )
75}

89    fn addr(byte: u8) -> Address {
90        Address::from_bytes([byte; 32]).unwrap()
91    }
92
93    fn utxo_id(byte: u8, vout: u32) -> UtxoId {
94        UtxoId {
95            txid: BitcoinTxid::from(addr(byte)),
96            vout,
97        }
98    }

100    /// A ledger that knows one object id and answers everything else
101    /// with the configured status, recording what it was asked for.
102    #[derive(Clone)]
103    struct OneObjectLedger {
104        known: Address,
105        miss: tonic::Code,
106        asked: Arc<Mutex<Vec<String>>>,
107    }
108
109    #[tonic::async_trait]
110    impl LedgerService for OneObjectLedger {
111        async fn get_object(
112            &self,
113            request: tonic::Request<GetObjectRequest>,
114        ) -> Result<tonic::Response<GetObjectResponse>, tonic::Status> {
115            let object_id = request.into_inner().object_id.unwrap_or_default();
116            self.asked.lock().unwrap().push(object_id.clone());
117            if object_id != self.known.to_string() {
118                return Err(tonic::Status::new(self.miss, "no such object"));
119            }
120            let mut object = Object::default();
121            object.object_id = Some(object_id);
122            Ok(tonic::Response::new(GetObjectResponse::new(object)))
123        }
124    }
125
126    async fn spawn_ledger(ledger: OneObjectLedger) -> Client {
127        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
128        let addr = listener.local_addr().unwrap();
129        let incoming = futures::stream::unfold(listener, |listener| async move {
130            let result = listener.accept().await.map(|(stream, _)| stream);
131            Some((result, listener))
132        });
133        tokio::spawn(
134            tonic::transport::Server::builder()
135                .add_service(LedgerServiceServer::new(ledger))
136                .serve_with_incoming(incoming),
137        );
138        Client::new(format!("http://{addr}").as_str()).unwrap()
139    }

141    #[test]
142    fn field_id_depends_on_bag_key_and_package() {
143        let base = spent_utxo_field_id(addr(0x10), addr(0xAA), &utxo_id(0x77, 0));
144        assert_ne!(
145            base,
146            spent_utxo_field_id(addr(0x11), addr(0xAA), &utxo_id(0x77, 0))
147        );
148        assert_ne!(
149            base,
150            spent_utxo_field_id(addr(0x10), addr(0xAB), &utxo_id(0x77, 0))
151        );
152        assert_ne!(
153            base,
154            spent_utxo_field_id(addr(0x10), addr(0xAA), &utxo_id(0x77, 1))
155        );
156        assert_eq!(
157            base,
158            spent_utxo_field_id(addr(0x10), addr(0xAA), &utxo_id(0x77, 0))
159        );
160    }

162    #[tokio::test]
163    async fn present_entry_is_spent_and_missing_entry_is_not() {
164        let spent_utxos_id = addr(0x10);
165        let package_id = addr(0xAA);
166        let spent = utxo_id(0x77, 0);
167        let unspent = utxo_id(0x77, 1);
168        let ledger = OneObjectLedger {
169            known: spent_utxo_field_id(spent_utxos_id, package_id, &spent),
170            miss: tonic::Code::NotFound,
171            asked: Arc::default(),
172        };
173        let client = spawn_ledger(ledger.clone()).await;
174
175        assert!(
176            lookup_spent_utxo(client.clone(), spent_utxos_id, package_id, &spent)
177                .await
178                .unwrap()
179        );
180        assert!(
181            !lookup_spent_utxo(client, spent_utxos_id, package_id, &unspent)
182                .await
183                .unwrap()
184        );
185
186        // Each answer came from exactly the derived entry id, not from
187        // a listing or a guess.
188        let asked = ledger.asked.lock().unwrap();
189        assert_eq!(
190            *asked,
191            vec![
192                spent_utxo_field_id(spent_utxos_id, package_id, &spent).to_string(),
193                spent_utxo_field_id(spent_utxos_id, package_id, &unspent).to_string(),
194            ]
195        );
196    }
