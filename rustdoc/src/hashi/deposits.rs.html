35    pub async fn validate_deposit_request(
36        &self,
37        deposit_request: &DepositRequest,
38    ) -> Result<(), UnapprovedDepositError> {
39        self.validate_deposit_request_on_sui(deposit_request)?;

42        self.screen_deposit(deposit_request).await?;

46    /// Run AML/Sanctions checks for the deposit request.
47    /// If no screener client is configured, checks are skipped.
48    #[tracing::instrument(level = "debug", skip_all, fields(deposit_id = %deposit_request.id))]
49    async fn screen_deposit(
50        &self,
51        deposit_request: &DepositRequest,
52    ) -> Result<(), UnapprovedDepositError> {
53        let Some(screener) = self.screener_client() else {
54            tracing::debug!("AML checks skipped: no screener configured");
55            return Ok(());
56        };
57
58        // bitcoin
59        let source_tx_hash = deposit_request.utxo.id.txid.to_string();
60        let bitcoin_chain_id = self.config.bitcoin_chain_id().to_string();
61
62        // sui
63        let destination_address = deposit_request.id.to_string();
64        let sui_chain_id = self.config.sui_chain_id().to_string();
65
66        let approved = screener
67            .approve_deposit(
68                &source_tx_hash,
69                &destination_address,
70                &bitcoin_chain_id,
71                &sui_chain_id,
72            )
73            .await
74            .map_err(|e| UnapprovedDepositError::AmlServiceError(anyhow!(e)))?;
75
76        if !approved {
77            return Err(UnapprovedDepositError::AmlRejected(anyhow!(
78                "AML checks failed for source tx {source_tx_hash}, destination {destination_address}, bitcoin chain {bitcoin_chain_id}, sui chain {sui_chain_id}"
79            )));
80        }
81
82        Ok(())
83    }

85    /// Validate that the deposit request exists on Sui
86    #[tracing::instrument(level = "debug", skip_all, fields(deposit_id = %deposit_request.id))]
87    fn validate_deposit_request_on_sui(
88        &self,
89        deposit_request: &DepositRequest,
90    ) -> Result<(), UnapprovedDepositError> {
91        let state = self.onchain_state().state();
92        let deposit_queue = &state.hashi().bitcoin().deposit_queue;
93        match deposit_queue.requests().get(&deposit_request.id) {
94            None => {
95                return Err(UnapprovedDepositError::InvalidOnchainRequest(anyhow!(
96                    "Deposit request not found on Sui"
97                )));
98            }
99            Some(onchain_request) => {
100                // Approval state lives on the on-chain request but is not
101                // carried by `DepositRequest` constructed from the proto, so
102                // compare the immutable core fields only.
103                let core_matches = onchain_request.id == deposit_request.id
104                    && onchain_request.sender == deposit_request.sender
105                    && onchain_request.created_timestamp_ms == deposit_request.created_timestamp_ms
106                    && onchain_request.sui_tx_digest == deposit_request.sui_tx_digest
107                    && onchain_request.utxo == deposit_request.utxo;
108                if !core_matches {
109                    return Err(UnapprovedDepositError::InvalidOnchainRequest(anyhow!(
110                        "Deposit request fields do not match on-chain state"
111                    )));
112                }
113
114                // Refuse to sign a re-approval if the on-chain request is
115                // already approved by the current committee. The on-chain
116                // `approve_deposit` would reject it anyway, so we don't
117                // want to waste a signature exchange or a transaction.
118                //
119                // Read from the held snapshot; `OnchainState::epoch()` would
120                // reacquire the same state lock.
121                let current_epoch = state.hashi().committees.epoch();
122                if onchain_request
123                    .approval_cert
124                    .as_ref()
125                    .is_some_and(|cert| cert.epoch == current_epoch)
126                {
127                    return Err(UnapprovedDepositError::AlreadyApprovedThisEpoch);
128                }
129            }
130        }
131
132        let utxo_pool = &state.hashi().bitcoin().utxo_pool;
133        if utxo_pool.is_active_or_spent(&deposit_request.utxo.id) {
134            return Err(UnapprovedDepositError::DuplicateOrSpentOnSui(anyhow!(
135                "UTXO {:?} is already active or spent",
136                deposit_request.utxo.id
137            )));
138        }
139
140        Ok(())
141    }

143    /// Validate that there is a txout on Bitcoin that matches the deposit request
144    #[tracing::instrument(
145        level = "debug",
146        skip_all,
147        fields(
148            deposit_id = %deposit_request.id,
149            bitcoin_txid = %deposit_request.utxo.id.txid,
150            vout = deposit_request.utxo.id.vout,
151        ),
152    )]
153    async fn validate_deposit_request_on_bitcoin(
154        &self,
155        deposit_request: &DepositRequest,
156    ) -> Result<(), UnapprovedDepositError> {
157        let outpoint = deposit_request.utxo.id.into();
158        // Read the threshold live from on-chain state so governance
159        // updates take effect for new deposits without a restart.
160        let confirmation_threshold = self.onchain_state().bitcoin_confirmation_threshold();
161        let check_result = self
162            .btc_monitor()
163            .confirm_deposit(outpoint, confirmation_threshold)
164            .await;
165        let txout = match check_result {
166            Ok(DepositConfirmation::Confirmed(txout)) => txout,
167            Ok(DepositConfirmation::NotFound) => {
168                return Err(UnapprovedDepositError::BitcoinNotConfirmed(anyhow!(
169                    "transaction {} was not found",
170                    outpoint.txid
171                )));
172            }
173            Ok(DepositConfirmation::InMempool) => {
174                return Err(UnapprovedDepositError::BitcoinNotConfirmed(anyhow!(
175                    "transaction {} is still in the mempool",
176                    outpoint.txid
177                )));
178            }
179            Ok(DepositConfirmation::InsufficientConfirmations {
180                confirmations,
181                required_confirmations,
182            }) => {
183                return Err(UnapprovedDepositError::BitcoinNotConfirmed(anyhow!(
184                    "transaction {} has {confirmations}/{required_confirmations} confirmations",
185                    outpoint.txid
186                )));
187            }
188            Err(e @ DepositConfirmError::UtxoSpent { .. }) => {
189                self.metrics.deposits_rejected_utxo_spent.inc();
190                return Err(UnapprovedDepositError::BitcoinUtxoSpent(anyhow!(e)));
191            }
192            Err(e @ DepositConfirmError::InvalidVout { .. }) => {
193                return Err(UnapprovedDepositError::DepositDataMismatch(anyhow!(e)));
194            }
195            Err(e @ DepositConfirmError::TimedOut) => {
196                return Err(UnapprovedDepositError::BitcoinConfirmFailed(anyhow!(e)));
197            }
198            Err(DepositConfirmError::Other(err)) => {
199                return Err(UnapprovedDepositError::BitcoinConfirmFailed(err));
200            }
201        };
202        if txout.value.to_sat() != deposit_request.utxo.amount {
203            return Err(UnapprovedDepositError::DepositDataMismatch(anyhow!(
204                "Bitcoin deposit amount mismatch: got {}, onchain is {}",
205                deposit_request.utxo.amount,
206                txout.value.to_sat(),
207            )));
208        }
209
210        self.validate_deposit_request_derivation_path(&txout.script_pubkey, deposit_request)
211            .await?;
212        Ok(())
213    }

215    async fn validate_deposit_request_derivation_path(
216        &self,
217        script_pubkey: &ScriptBuf,
218        deposit_request: &DepositRequest,
219    ) -> Result<(), UnapprovedDepositError> {
220        let deposit_address = hashi_bitcoin::BitcoinAddress::from_script(
221            script_pubkey,
222            self.config.bitcoin_network(),
223        )
224        .map_err(|e| {
225            UnapprovedDepositError::DepositDataMismatch(anyhow!(
226                "Failed to extract address from script_pubkey: {e}"
227            ))
228        })?;
229        let expected_address = self
230            .get_deposit_address(deposit_request.utxo.derivation_path.as_ref())
231            .map_err(UnapprovedDepositError::DepositDataMismatch)?;
232
233        if deposit_address != expected_address {
234            return Err(UnapprovedDepositError::DepositDataMismatch(anyhow!(
235                "Expected address {expected_address}, got address {deposit_address}",
236            )));
237        }
238
239        Ok(())
240    }

286    /// Hashi committee's child pubkey at `derivation_path`. `None` maps
287    /// to a zero-byte path (change outputs).
288    pub(crate) fn deposit_pubkey(
289        &self,
290        derivation_path: Option<&sui_sdk_types::Address>,
291    ) -> anyhow::Result<XOnlyPublicKey> {
292        // Prefer the local signing manager (available after DKG preparation).
293        // Fall back to the on-chain key, which is guaranteed present once the
294        // initial committee has formed and `end_reconfig` has been processed.
295        let verifying_key = self
296            .signing_verifying_key()
297            .map(Ok)
298            .unwrap_or_else(|| self.onchain_state().onchain_verifying_key_g())
299            .context("MPC public key not available yet")?;
300        let derivation_path = normalized_derivation_path(derivation_path).into_inner();
301        let derived = fastcrypto_tbls::threshold_schnorr::key_derivation::derive_verifying_key(
302            &verifying_key,
303            &derivation_path,
304        )
305        .context("derive child verifying key")?;
306        XOnlyPublicKey::from_slice(&derived.to_byte_array()).context("valid 32-byte x-only key")
307    }

315    fn sign_deposit_confirmation(
316        &self,
317        deposit_request: &DepositRequest,
318    ) -> anyhow::Result<MemberSignature> {
319        let epoch = self.onchain_state().epoch();
320        let validator_address = self
321            .config
322            .validator_address()
323            .map_err(|e| anyhow!("No validator address configured: {}", e))?;
324        let committee = self
325            .onchain_state()
326            .state()
327            .hashi()
328            .committees
329            .committees()
330            .get(&epoch)
331            .cloned()
332            .ok_or_else(|| anyhow!("no committee for epoch {epoch}"))?;
333        let private_key =
334            self.find_signing_key_for_committee(&committee, validator_address, epoch)?;
335        let public_key_bytes = private_key.public_key().as_bytes().to_vec().into();
336
337        let message = DepositConfirmationMessage {
338            request_id: deposit_request.id,
339            utxo: deposit_request.utxo.clone(),
340        };
341
342        let signature_bytes = private_key
343            .sign(
344                self.config.hashi_ids().hashi_object_id,
345                epoch,
346                validator_address,
347                &message,
348            )
349            .signature()
350            .as_bytes()
351            .to_vec()
352            .into();
353
354        Ok(MemberSignature {
355            epoch: Some(epoch),
356            address: Some(validator_address.to_string()),
357            public_key: Some(public_key_bytes),
358            signature: Some(signature_bytes),
359        })
360    }

363/// `tr(NUMS, {multi_a(2, guardian_btc_pubkey, h), and_v(v:older(delay), pk(h))})`
364/// where `h = derive(mpc_pubkey, path)`: an immediate guardian+MPC 2-of-2 leaf,
365/// plus an MPC-only recovery leaf spendable after
366/// `HASHI_MPC_RECOVERY_DELAY_SECONDS`.
367/// `path = None` (change address) maps to a zero-byte path so deposit
368/// and withdrawal sides agree on the leaf key without a special case.
369///
370/// `mpc_key` is the raw MPC verifying key (`G`). The derivation is taken
371/// directly against this point — using only the x-only projection would
372/// silently force the parent to even-y and produce a different child key
373/// for ~half of all MPC vks, breaking the 2-of-2 leaf script.
374pub fn derive_deposit_address(
375    mpc_key: &ProjectivePoint,
376    guardian_btc_pubkey: &XOnlyPublicKey,
377    derivation_path: Option<&sui_sdk_types::Address>,
378    btc_network: bitcoin::Network,
379) -> anyhow::Result<hashi_bitcoin::BitcoinAddress> {
380    Ok(hashi_bitcoin::taproot_address(
381        guardian_btc_pubkey,
382        mpc_key,
383        &normalized_derivation_path(derivation_path),
384        btc_network,
385    ))
386}

388pub(crate) fn normalized_derivation_path(
389    derivation_path: Option<&sui_sdk_types::Address>,
390) -> sui_sdk_types::Address {
391    derivation_path
392        .copied()
393        .unwrap_or(sui_sdk_types::Address::ZERO)
394}

402#[derive(Debug, Error)]
403pub enum UnapprovedDepositError {
404    #[error("Failed to confirm Bitcoin deposit: {0}")]
405    BitcoinConfirmFailed(#[source] anyhow::Error),
406
407    #[error("Bitcoin deposit is not yet confirmed: {0}")]
408    BitcoinNotConfirmed(#[source] anyhow::Error),
409
410    #[error("Screener service error: {0}")]
411    AmlServiceError(#[source] anyhow::Error),
412
413    #[error("Invalid on-chain deposit request: {0}")]
414    InvalidOnchainRequest(#[source] anyhow::Error),
415
416    #[error("UTXO is already active or spent on Sui: {0}")]
417    DuplicateOrSpentOnSui(#[source] anyhow::Error),
418
419    #[error("Deposit UTXO has already been spent on Bitcoin: {0}")]
420    BitcoinUtxoSpent(#[source] anyhow::Error),
421
422    #[error("Deposit data mismatch: {0}")]
423    DepositDataMismatch(#[source] anyhow::Error),
424
425    #[error("AML checks rejected deposit: {0}")]
426    AmlRejected(#[source] anyhow::Error),
427
428    #[error("Failed quorum: weight {weight} < {required_weight}")]
429    FailedQuorum { weight: u64, required_weight: u64 },
430
431    #[error("Failed to build deposit certificate: {0}")]
432    CertificateBuildFailed(#[source] anyhow::Error),
433
434    #[error("Failed to create Sui transaction executor: {0}")]
435    ExecutorInitFailed(#[source] anyhow::Error),
436
437    #[error("Failed to approve deposit on Sui: {0}")]
438    ApproveDepositFailed(#[source] anyhow::Error),
439
440    #[error("Failed to sign deposit confirmation: {0}")]
441    SignDepositFailed(#[source] anyhow::Error),
442
443    #[error("Deposit has already been approved by the current committee")]
444    AlreadyApprovedThisEpoch,
445
446    #[error("Deposit processing timed out after {0:?}")]
447    TimedOut(std::time::Duration),
448}
449
450impl UnapprovedDepositError {
451    pub(crate) fn kind(&self) -> UnapprovedDepositErrorKind {
452        match self {
453            Self::BitcoinConfirmFailed(_)
454            | Self::BitcoinNotConfirmed(_)
455            | Self::AmlServiceError(_)
456            | Self::FailedQuorum { .. }
457            | Self::CertificateBuildFailed(_)
458            | Self::ExecutorInitFailed(_)
459            | Self::ApproveDepositFailed(_)
460            | Self::SignDepositFailed(_)
461            | Self::AlreadyApprovedThisEpoch
462            | Self::TimedOut(_) => UnapprovedDepositErrorKind::RetryOnNextBlock,
463            Self::InvalidOnchainRequest(_)
464            | Self::DuplicateOrSpentOnSui(_)
465            | Self::BitcoinUtxoSpent(_)
466            | Self::DepositDataMismatch(_)
467            | Self::AmlRejected(_) => UnapprovedDepositErrorKind::NeverRetry,
468        }
469    }
470}

479#[derive(Debug, Error)]
480pub enum ApprovedDepositError {
481    #[error("Failed to create Sui transaction executor: {0}")]
482    ExecutorInitFailed(#[source] anyhow::Error),
483
484    #[error("Failed to confirm deposit on Sui: {0}")]
485    ConfirmDepositFailed(#[source] anyhow::Error),
486
487    #[error("Deposit processing timed out after {0:?}")]
488    TimedOut(std::time::Duration),
489
490    #[error("Deposit confirmation checkpoint wait timed out")]
491    CheckpointWaitTimedOut,
492}
493
494impl ApprovedDepositError {
495    pub(crate) fn kind(&self) -> ApprovedDepositErrorKind {
496        match self {
497            Self::ExecutorInitFailed(_) => ApprovedDepositErrorKind::ExecutorInitFailed,
498            Self::ConfirmDepositFailed(_) => ApprovedDepositErrorKind::ConfirmDepositFailed,
499            Self::TimedOut(_) | Self::CheckpointWaitTimedOut => ApprovedDepositErrorKind::TimedOut,
500        }
501    }
502}

504// This backoff only applies after an eligible approved deposit confirmation
505// actually fails. Deposits whose time-delay has not elapsed are skipped before
506// entering retry tracking.
507impl RetryPolicy for ApprovedDepositErrorKind {
508    fn retry_base_delay_ms(self) -> u64 {
509        // All approved-deposit failures are transient in practice (epoch
510        // mismatch, transient RPC errors, slow network). A short base delay
511        // with exponential backoff handles the common case while throttling
512        // genuinely broken deposits.
513        3_000
514    }
515
516    fn max_delay_ms(self) -> u64 {
517        30 * 60 * 1000
518    }
519
520    fn max_retries(self) -> u32 {
521        u32::MAX
522    }
