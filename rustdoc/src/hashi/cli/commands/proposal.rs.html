39/// Resolve and show the committee member the configured keypair acts for, so
40/// the operator sees which member the transaction is attributed to before
41/// confirming it. The signer may be a member's validator address or an
42/// operator address a member delegated to it; see
43/// [`HashiClient::resolve_validator_address`] for the rules.
44pub(crate) fn print_acting_validator(client: &HashiClient) -> Result<()> {
45    let validator_address = client.resolve_validator_address()?;
46    let via_operator = match client.acting_address() {
47        Some(acting) if acting != validator_address => {
48            format!(" (delegated operator {})", acting.to_hex().dimmed())
49        }
50        _ => String::new(),
51    };
52    print_detail(&format!(
53        "  Acting as validator: {}{via_operator}",
54        validator_address.to_hex().cyan()
55    ));
56    Ok(())
57}

71/// Return the proposal when it is open, and otherwise refuse with a message
72/// that names the proposal, its type and why the action cannot happen.
73///
74/// Executed proposals stay on chain in a second bag, so without this the
75/// command finds the proposal, prints its details, prompts, and only then
76/// fails in simulation with a framework abort (`dynamic_field` code 1) that
77/// never mentions the word "executed" and shares its code with the
78/// vote-already-counted error.
79pub fn open_proposal(
80    location: ProposalLocation,
81    proposal_id: &str,
82    action: ProposalAction,
83    sui_rpc_url: &str,
84) -> Result<Proposal> {
85    match location {
86        ProposalLocation::Active(proposal) => Ok(proposal),
87        ProposalLocation::Executed(proposal) => {
88            let kind = display::format_proposal_type(&proposal.proposal_type);
89            let consequence = match action {
90                ProposalAction::Vote => {
91                    "voting is closed and there is nothing to submit".to_owned()
92                }
93                ProposalAction::RemoveVote => {
94                    "its votes are final and cannot be removed".to_owned()
95                }
96                ProposalAction::Execute => {
97                    "nothing to execute; its effect is already on chain".to_owned()
98                }
99            };
100            anyhow::bail!(
101                "proposal {proposal_id} ({kind}) has already been executed; {consequence}. \
102                 Run `hashi proposal view {proposal_id}` for the final tally."
103            )
104        }
105        ProposalLocation::Missing => anyhow::bail!(
106            "proposal {proposal_id} was not found in the active or executed proposals on \
107             {sui_rpc_url}: either the id is wrong, or the proposal expired (7 days after \
108             creation) and was deleted. Run `hashi proposal list` to see the open proposals."
109        ),
110    }
111}

113/// One-line lifecycle status for `proposal view`, from bag membership and
114/// the seven-day expiry.
115pub fn proposal_status(location: &ProposalLocation, now_ms: u64) -> String {
116    match location {
117        ProposalLocation::Executed(_) => "Executed".to_owned(),
118        ProposalLocation::Active(proposal) => {
119            let expires_ms = proposal.timestamp_ms.saturating_add(PROPOSAL_MAX_AGE_MS);
120            if now_ms > expires_ms {
121                format!(
122                    "Expired on {} (no longer votable; awaiting deletion)",
123                    display::format_timestamp(expires_ms)
124                )
125            } else {
126                format!("Active (expires {})", display::format_timestamp(expires_ms))
127            }
128        }
129        ProposalLocation::Missing => "Not found".to_owned(),
130    }
131}

133/// Vote tally against the quorum threshold, computed once and printed the
134/// same way everywhere (view, list --votes, after a vote, before execute).
135#[derive(Clone, Debug, PartialEq, Eq)]
136pub struct QuorumProgress {
137    pub voters: usize,
138    pub voted_weight: u64,
139    pub threshold_weight: u64,
140    pub total_weight: u64,
141}
142
143impl QuorumProgress {
144    pub fn new(
145        votes: &[Address],
146        quorum_threshold_bps: u64,
147        committee: &hashi_types::committee::Committee,
148    ) -> Self {
149        let total_weight = committee.total_weight();
150        let voted_weight = votes
151            .iter()
152            .map(|voter| {
153                committee
154                    .members()
155                    .iter()
156                    .find(|m| m.validator_address() == *voter)
157                    .map(|m| m.weight())
158                    .unwrap_or(0)
159            })
160            .sum();
161        Self {
162            voters: votes.len(),
163            voted_weight,
164            threshold_weight: total_weight
165                .saturating_mul(quorum_threshold_bps)
166                .div_ceil(10_000),
167            total_weight,
168        }
169    }
170
171    pub fn met(&self) -> bool {
172        self.total_weight > 0 && self.voted_weight >= self.threshold_weight
173    }
174
175    pub fn missing_weight(&self) -> u64 {
176        self.threshold_weight.saturating_sub(self.voted_weight)
177    }
178
179    /// `3 voter(s), 120/200 weight (80 more needed)` or
180    /// `3 voter(s), 200/200 weight, quorum reached`.
181    pub fn describe(&self) -> String {
182        if self.met() {
183            format!(
184                "{} voter(s), {}/{} weight, quorum reached",
185                self.voters, self.voted_weight, self.threshold_weight
186            )
187        } else {
188            format!(
189                "{} voter(s), {}/{} weight ({} more needed)",
190                self.voters,
191                self.voted_weight,
192                self.threshold_weight,
193                self.missing_weight()
194            )
195        }
196    }
197}

199/// Refuse a proposal past its seven-day window: the chain would abort with
200/// `EProposalExpired` after the prompt, and the only remaining action is
201/// deletion.
202pub fn refuse_if_expired(proposal: &Proposal, proposal_id: &str, now_ms: u64) -> Result<()> {
203    let expires_ms = proposal.timestamp_ms.saturating_add(PROPOSAL_MAX_AGE_MS);
204    anyhow::ensure!(
205        now_ms <= expires_ms,
206        "proposal {proposal_id} ({}) expired on {}; it can no longer be voted on or executed, \
207         only deleted",
208        display::format_proposal_type(&proposal.proposal_type),
209        display::format_timestamp(expires_ms)
210    );
211    Ok(())
212}

215/// committee: only seated members vote (`ENotCommitteeMember` on chain).
216/// `seated` is the current committee's member addresses; `None` before the

219    if let Some((epoch, members)) = seated {
220        anyhow::ensure!(
221            members.contains(&validator),
222            "validator {} is registered but not seated in the current committee (epoch \
223             {epoch}); only seated members can vote",
224            validator.to_hex()
225        );
226    }
227    Ok(())

230/// Refuse a second vote, or the removal of a vote that was never cast, before
231/// anything is signed (`EVoteAlreadyCounted` / `ENoVoteFound` on chain).
232pub fn refuse_vote_state(
233    validator: Address,
234    votes: &[Address],
235    proposal_id: &str,
236    action: ProposalAction,
237) -> Result<()> {
238    let has_voted = votes.contains(&validator);
239    match action {
240        ProposalAction::Vote => anyhow::ensure!(
241            !has_voted,
242            "validator {} has already voted on proposal {proposal_id}; nothing to submit. \
243             Use `hashi proposal remove-vote {proposal_id}` to withdraw the vote.",
244            validator.to_hex()
245        ),
246        ProposalAction::RemoveVote => anyhow::ensure!(
247            has_voted,
248            "validator {} has no vote on proposal {proposal_id} to remove",
249            validator.to_hex()
250        ),
251        ProposalAction::Execute => {}
252    }
253    Ok(())
254}

308/// Refuse an update the chain would reject with `EInvalidConfigEntry`: the
309/// key must already exist in the targeted store, and its value type cannot
310/// change. When the key lives in the other store, name the right command.
311pub fn refuse_bad_config_update(
312    key: &str,
313    proposed: &hashi_types::move_types::ConfigValue,
314    target: ConfigStore,
315    current: Option<&hashi_types::move_types::ConfigValue>,
316    in_other_store: bool,
317) -> Result<()> {
318    let Some(current) = current else {
319        if in_other_store {
320            anyhow::bail!(
321                "key {key} is not in the {} config; it lives in the {} config. Use `{}` instead.",
322                target.name(),
323                target.other().name(),
324                target.other().update_command()
325            );
326        }
327        anyhow::bail!(
328            "key {key} does not exist in the {} config, and updates can only change existing \
329             keys. Use `hashi proposal create add-config` to introduce a new key, or \
330             `hashi config on-chain` to see the current keys.",
331            target.name()
332        );
333    };
334    anyhow::ensure!(
335        std::mem::discriminant(current) == std::mem::discriminant(proposed),
336        "key {key} holds a {} value on chain but the proposed value is {}; a key's value \
337         type cannot change",
338        config_value_type(current),
339        config_value_type(proposed)
340    );
341    Ok(())
342}

375/// A human explanation for a governance transaction that failed in
376/// simulation or on chain, or `None` when the failure is not a Move abort.
377///
378/// Two sources: a clever `#[error]` constant from the hashi package, which is
379/// named and given a one-line hint; and the one framework abort every
380/// governance entry can hit, `dynamic_field` code 1 when the proposal has
381/// left the active bag between the pre-flight and the simulate. Matching is
382/// on module and constant name, never on the bare code: `1` is also
383/// `EVoteAlreadyCounted`, and `0` is both `EVersionDisabled` and
384/// `EUnauthorizedCaller`.
385pub fn explain_execution_error(
386    error: &sui_rpc::proto::sui::rpc::v2::ExecutionError,
387) -> Option<String> {
388    use sui_rpc::proto::sui::rpc::v2::clever_error::Value;
389    use sui_rpc::proto::sui::rpc::v2::execution_error::ExecutionErrorKind;
390
391    if error
392        .kind
393        .and_then(|kind| ExecutionErrorKind::try_from(kind).ok())
394        != Some(ExecutionErrorKind::MoveAbort)
395    {
396        return None;
397    }
398    let abort = error.abort_opt()?;
399    let module = abort.location().module_opt();
400    let clever = abort.clever_error.as_ref();
401    let constant = clever.and_then(|c| c.constant_name.as_deref());
402    let rendered = clever.and_then(|c| match &c.value {
403        Some(Value::Rendered(text)) => Some(text.as_str()),
404        _ => None,
405    });
406
407    match (module, abort.abort_code, constant) {
408        (Some("dynamic_field"), Some(1), None) => Some(
409            "the proposal is no longer in the active proposal bag: it was executed, or \
410             deleted after expiring, while this command was running. Run \
411             `hashi proposal view <proposal-id>` to see its state."
412                .to_owned(),
413        ),
414        (_, _, Some(name)) => {
415            let hint = match name {
416                "EVoteAlreadyCounted" => Some("this validator has already voted on the proposal"),
417                "ENoVoteFound" => Some("this validator has no vote on the proposal to remove"),
418                "EQuorumNotReached" => Some(
419                    "the proposal has not reached quorum yet; `hashi proposal view` shows the tally",
420                ),
421                "EProposalExpired" => {
422                    Some("proposals expire 7 days after creation; this one can only be deleted")
423                }
424                "EProposalAlreadyExecuted" => Some("the proposal has already been executed"),
425                "ENotCommitteeMember" => {
426                    Some("the validator is registered but not seated in the current committee")
427                }
428                "EUnauthorizedCaller" => {
429                    Some("the signer is neither the validator's address nor its delegated operator")
430                }
431                "EVersionDisabled" => Some(
432                    "this binary targets a package version governance has disabled; upgrade hashi",
433                ),
434                _ => None,
435            };
436            let mut text = match rendered {
437                Some(rendered) => format!("{name}: {rendered}"),
438                None => name.to_owned(),
439            };
440            if let Some(hint) = hint {
441                text.push_str(" (");
442                text.push_str(hint);
443                text.push(')');
444            }
445            Some(text)
446        }
447        _ => None,
448    }
449}

451/// [`explain_execution_error`] for the error `finalize_tx` returns, whichever
452/// mode produced it: the executor wraps the execute path's build error in
453/// `TxFailure::NotSubmitted`, while dry-run and serialize-unsigned return the
454/// SDK build error bare, so both shapes are searched.
455pub(crate) fn explain_move_abort(err: &anyhow::Error) -> Option<String> {
456    let from_executor = crate::sui_tx_executor::transaction_execution_error(err);
457    let from_builder =
458        err.chain().find_map(
459            |e| match e.downcast_ref::<sui_transaction_builder::Error>() {
460                Some(sui_transaction_builder::Error::SimulationFailure(failure)) => {
461                    Some(failure.execution_error())
462                }
463                _ => None,
464            },
465        );
466    explain_execution_error(from_executor.or(from_builder)?)
467}

469/// Finalize a transaction according to `tx_opts`: serialize it unsigned,
470/// dry-run it, or sign and submit it.
471///
472/// Returns `Some(response)` when a real transaction was executed, and `None`
473/// for dry-run, serialize-unsigned, or when execution is requested but no
474/// keypair is configured.
475pub(crate) async fn execute_or_simulate(
476    client: &mut HashiClient,
477    tx: sui_transaction_builder::TransactionBuilder,
478    tx_opts: &TxOptions,
479) -> Result<Option<ExecuteTransactionResponse>> {
480    use crate::sui_tx_executor::TxMode;
481
482    // Only the execute path needs a keypair; serialize/dry-run build with just
483    // the sender address.
484    if tx_opts.mode() == TxMode::Execute && !client.can_execute() {
485        print_warning(
486            "Transaction execution requires a keypair (--keypair). Use \
487             --serialize-unsigned-transaction to emit an unsigned transaction, or --dry-run.",
488        );
489        return Ok(None);
490    }
491
492    match tx_opts.mode() {
493        TxMode::SerializeUnsigned => print_info("Building unsigned transaction..."),
494        TxMode::DryRun => print_info("Simulating transaction (dry-run)..."),
495        TxMode::Execute => print_info("Executing transaction..."),
496    }
497
498    // `finalize_tx` decodes Move aborts for every command that goes through it.
499    let outcome = client.finalize_tx(tx, tx_opts).await?;
500    Ok(crate::cli::print_tx_outcome(outcome, client.sui_rpc_url()).map(|response| *response))
501}

517/// List all active proposals
518pub async fn list_proposals(
519    config: &CliConfig,
520    type_filter: Option<String>,
521    detailed: bool,
522    executed: bool,
523    votes: bool,
524    json: bool,
525) -> Result<()> {
526    let client = HashiClient::new(config).await?;
527
528    print_info("Fetching proposals...");
529
530    let (proposals, bag) = if executed {
531        (client.fetch_executed_proposals(), "executed")
532    } else {
533        (client.fetch_proposals(), "active")
534    };
535
536    // Filter by type if specified
537    let proposals: Vec<_> = if let Some(ref filter) = type_filter {
538        let filter_lower = filter.to_lowercase();
539        proposals
540            .into_iter()
541            .filter(|p| {
542                display::format_proposal_type(&p.proposal_type)
543                    .to_lowercase()
544                    .contains(&filter_lower)
545            })
546            .collect()
547    } else {
548        proposals
549    };
550
551    // The optional per-row tally costs one live read per proposal.
552    let committee = client.fetch_current_committee();
553    let mut tallies: Vec<Option<QuorumProgress>> = Vec::with_capacity(proposals.len());
554    if votes {
555        for proposal in &proposals {
556            let tally = match (client.fetch_proposal_details(proposal.id).await, &committee) {
557                (Ok(details), Some(committee)) => Some(QuorumProgress::new(
558                    &details.votes,
559                    details.quorum_threshold_bps,
560                    committee,
561                )),
562                _ => None,
563            };
564            tallies.push(tally);
565        }
566    }
567
568    let now = now_ms();
569    let location = |p: &Proposal| {
570        if executed {
571            ProposalLocation::Executed(p.clone())
572        } else {
573            ProposalLocation::Active(p.clone())
574        }
575    };
576
577    if json {
578        let rows: Vec<serde_json::Value> = proposals
579            .iter()
580            .enumerate()
581            .map(|(i, p)| {
582                let mut row = serde_json::json!({
583                    "id": p.id.to_hex(),
584                    "type": display::format_proposal_type(&p.proposal_type),
585                    "created_ms": p.timestamp_ms,
586                    "status": proposal_status(&location(p), now),
587                });
588                if let Some(Some(tally)) = tallies.get(i) {
589                    row["votes"] = serde_json::json!({
590                        "voters": tally.voters,
591                        "voted_weight": tally.voted_weight,
592                        "threshold_weight": tally.threshold_weight,
593                        "total_weight": tally.total_weight,
594                        "quorum_reached": tally.met(),
595                    });
596                }
597                row
598            })
599            .collect();
600        println!("{}", serde_json::to_string_pretty(&rows)?);
601        return Ok(());
602    }
603
604    if proposals.is_empty() {
605        let what = match &type_filter {
606            Some(filter) => format!("No {bag} proposals found matching type filter: {filter}"),
607            None => format!("No {bag} proposals found."),
608        };
609        println!("\n{}", what.dimmed());
610        return Ok(());
611    }
612
613    println!(
614        "\n📋 {} Proposals:\n",
615        if executed { "Executed" } else { "Active" }
616    );
617
618    if detailed {
619        for proposal in &proposals {
620            let status = proposal_status(&location(proposal), now);
621            print_proposal_detailed(proposal, &status, None, None);
622            println!();
623        }
624    } else {
625        #[derive(Tabled)]
626        struct ProposalRow {
627            #[tabled(rename = "ID")]
628            id: String,
629            #[tabled(rename = "Type")]
630            proposal_type: String,
631            #[tabled(rename = "Created")]
632            timestamp: String,
633            #[tabled(rename = "Status")]
634            status: String,
635            #[tabled(rename = "Votes")]
636            votes: String,
637        }
638
639        let rows: Vec<ProposalRow> = proposals
640            .iter()
641            .enumerate()
642            .map(|(i, p)| ProposalRow {
643                // Full ids: this is what gets pasted into `vote` and `view`.
644                id: display::format_address_full(&p.id),
645                proposal_type: display::format_proposal_type(&p.proposal_type),
646                timestamp: display::format_timestamp(p.timestamp_ms),
647                status: proposal_status(&location(p), now),
648                votes: match tallies.get(i) {
649                    Some(Some(tally)) => tally.describe(),
650                    Some(None) => "unavailable".to_owned(),
651                    None => "(pass --votes)".to_owned(),
652                },
653            })
654            .collect();
655
656        let table = Table::new(rows).to_string();
657        println!("{}", table);
658    }
659
660    println!(
661        "\n{} {} {bag} proposal(s) found",
662        "ℹ".blue(),

701/// Vote on a proposal, optionally chaining an execute if this vote pushes
702/// the proposal over quorum.
703pub async fn vote(
704    config: &CliConfig,
705    proposal_id: &str,
706    execute: bool,
707    tx_opts: &TxOptions,
708) -> Result<()> {
709    let mut client = HashiClient::new(config).await?;
710
711    let proposal_addr = Address::from_hex(proposal_id)
712        .with_context(|| format!("Invalid proposal ID: {}", proposal_id))?;
713
714    print_info(&format!("Fetching proposal {}...", proposal_id));
715
716    let proposal = open_proposal(
717        client.locate_proposal(&proposal_addr),
718        proposal_id,
719        ProposalAction::Vote,
720        client.sui_rpc_url(),
721    )?;
722    refuse_if_expired(&proposal, proposal_id, now_ms())?;
723
724    // Everything the chain would refuse after the prompt is refused here
725    // first, from one live read of the proposal.
726    let details = client.fetch_proposal_details(proposal_addr).await?;
727    let committee = client.fetch_current_committee();
728    if client.acting_address().is_some() {
729        let validator = client.resolve_validator_address()?;
730        let seated = seated_members(committee.as_ref());
731        refuse_if_not_seated(validator, seated.as_ref().map(|(e, m)| (*e, m.as_slice())))?;
732        refuse_vote_state(validator, &details.votes, proposal_id, ProposalAction::Vote)?;
733    }
734
735    let proposal_type_str = display::format_proposal_type(&proposal.proposal_type);
736
737    print_detail(&format!("\n{}", "Proposal Details:".bold()));
738    print_detail(&format!("  Type: {}", proposal_type_str.cyan()));
739    if let Some(committee) = &committee {
740        let progress = QuorumProgress::new(&details.votes, details.quorum_threshold_bps, committee);
741        print_detail(&format!("  Votes so far: {}", progress.describe()));
742    }
743    print_acting_validator(&client)?;
744
745    if !prompt_continue("vote on this proposal", tx_opts).await? {
746        print_warning("Aborted.");
747        return Ok(());
748    }
749
750    print_info("Building vote transaction...");
751
752    // Infer the type tag from the on-chain proposal type
753    let type_arg = client.proposal_type_arg(&proposal.proposal_type)?;
754    let tx = client.build_vote_transaction(proposal_addr, type_arg)?;
755
756    print_info(&format!(
757        "Transaction: proposal::vote<{}> on {}",
758        proposal_type_str, proposal_id
759    ));
760
761    let vote_response = execute_or_simulate(&mut client, tx, tx_opts).await?;
762
763    // Nothing landed on chain (dry-run, serialize, or no keypair).
764    if vote_response.is_none() {
765        return Ok(());
766    }
767
768    // Re-fetch live vote state: `HashiClient`'s cached scrape is from CLI
769    // start, so this has to be the live `list_dynamic_fields` call.
770    let details = client
771        .fetch_proposal_details(proposal_addr)
772        .await
773        .context("failed to re-fetch proposal state after voting")?;
774    let Some(committee) = committee else {
775        print_info("Vote recorded; no committee is formed yet, so quorum cannot be computed.");
776        return Ok(());
777    };
778    let progress = QuorumProgress::new(&details.votes, details.quorum_threshold_bps, &committee);
779    print_info(&format!("Votes: {}", progress.describe()));
780
781    if !progress.met() {
782        if execute {
783            print_info("Quorum not reached yet; skipping --execute.");
784        }
785        return Ok(());
786    }
787
788    // Upgrade proposals require the dedicated upgrade flow: the generic
789    // `<module>::execute` path can't construct an UpgradeTicket.
790    if is_upgrade_proposal(&proposal.proposal_type) {
791        print_info(&format!(
792            "Quorum reached; run `hashi proposal execute-upgrade {proposal_id} \
793             --package-path <dir>` to execute it."
794        ));
795        return Ok(());
796    }
797    if !execute {
798        print_info(&format!(
799            "Quorum reached; run `hashi proposal execute {proposal_id}` to execute it \
800             (or pass --execute to vote next time)."
801        ));
802        return Ok(());
803    }
804
805    print_info("Quorum reached; executing...");
806    let execute_tx =
807        client.build_execute_proposal_transaction(proposal_addr, &proposal.proposal_type)?;
808    print_info(&format!(
809        "Transaction: {}::execute on {}",
810        proposal.proposal_type.as_str(),
811        proposal_id
812    ));
813    execute_or_simulate(&mut client, execute_tx, tx_opts).await?;
814    Ok(())
815}

952/// Execute an approved upgrade proposal.
953///
954/// One programmable transaction: `<module>::execute` consumes the proposal and
955/// returns the `UpgradeTicket`, the `Upgrade` command publishes the freshly
956/// built modules against that ticket, and `<module>::finalize_upgrade` commits
957/// the receipt (which also enables the new version). The ticket and receipt
958/// are hot potatoes, so the three steps cannot be split across transactions.
959///
960/// The package is built here, from `package_path`, and must be byte-identical
961/// to the build whose digest went into the proposal: same commit, same `sui`.
962/// The chain enforces that when it processes the `Upgrade` command, so a
963/// mismatch fails at simulation rather than costing gas.
964pub async fn execute_upgrade(
965    config: &CliConfig,
966    proposal_id: &str,
967    args: ExecuteUpgradeArgs<'_>,
968    tx_opts: &TxOptions,
969) -> Result<()> {
970    let ExecuteUpgradeArgs {
971        package_path,
972        sui_binary,
973        sui_client_config,
974    } = args;
975    let mut client = HashiClient::new(config).await?;
976
977    let proposal_addr = Address::from_hex(proposal_id)
978        .with_context(|| format!("Invalid proposal ID: {}", proposal_id))?;
979
980    print_info(&format!("Fetching proposal {}...", proposal_id));
981
982    let proposal = client
983        .fetch_proposal(&proposal_addr)
984        .ok_or_else(|| anyhow::anyhow!("Proposal not found: {}", proposal_id))?;
985
986    anyhow::ensure!(
987        is_upgrade_proposal(&proposal.proposal_type),
988        "{} is a {} proposal, not an upgrade; use `proposal execute` instead",
989        proposal_id,
990        display::format_proposal_type(&proposal.proposal_type)
991    );
992    let module = "upgrade";
993
994    let current_version = client.highest_package_version().context(
995        "could not determine current package version from on-chain state; \
996         is the package deployed?",
997    )?;
998    let expected_version = current_version + 1;
999    let current_package_id = client.latest_package_id()?;
1000
1001    print_detail(&format!("\n{}", "Execute Upgrade Proposal:".bold()));
1002    print_detail(&format!(
1003        "  Type:            {}",
1004        display::format_proposal_type(&proposal.proposal_type).cyan()
1005    ));
1006    print_detail(&format!("  ID:              {}", proposal_id));
1007    print_detail(&format!("  Module:          {}", module));
1008    print_detail(&format!(
1009        "  Current package: v{} at {}",
1010        current_version,
1011        current_package_id.to_hex()
1012    ));
1013    print_detail(&format!("  Package path:    {}", package_path.display()));
1014    print_detail(&format!(
1015        "  Expects:         PACKAGE_VERSION = {expected_version}"
1016    ));
1017
1018    print_info(&format!(
1019        "Building upgrade package at {} with {}",
1020        package_path.display(),
1021        sui_binary.display()
1022    ));
1023    let (compiled, digest) = build_upgrade_package(
1024        sui_binary,
1025        package_path,
1026        sui_client_config,
1027        expected_version,
1028    )
1029    .context("failed to build upgrade package")?;
1030    print_detail(&format!("  Built digest:    0x{}", hex::encode(&digest)));
1031    print_detail(
1032        "  The publish is accepted only if this digest matches the proposal's; \
1033         build from the same commit with the same `sui` that produced the proposal.",
1034    );
1035
1036    if !prompt_continue(
1037        "execute this upgrade (execute + publish + finalize in one transaction)",
1038        tx_opts,
1039    )
1040    .await?
1041    {
1042        crate::cli::print_warning("Aborted.");
1043        return Ok(());
1044    }
1045
1046    let hashi_ids = *client.hashi_ids();
1047    let tx =
1048        build_upgrade_execution_transaction(hashi_ids, current_package_id, proposal_addr, compiled);
1049
1050    print_info(&format!(
1051        "Transaction: {module}::execute + Upgrade + {module}::finalize_upgrade on {}",
1052        proposal_id
1053    ));
1054
1055    let Some(response) = execute_or_simulate(&mut client, tx, tx_opts).await? else {
1056        return Ok(());
1057    };
1058
1059    let new_package_id = extract_new_package_id_from_response(&response)?;
1060    println!(
1061        "  {} {}",
1062        "New package ID:".bold(),
1063        new_package_id.to_hex().cyan()
1064    );
1065
1066    // Re-read the chain so the operator sees what the commit enabled, rather
1067    // than the pre-upgrade snapshot this client was built from.
1068    let refreshed = HashiClient::new(config).await?;
1069    let mut enabled: Vec<u64> = refreshed
1070        .onchain_state()
1071        .state()
1072        .hashi()
1073        .config
1074        .enabled_versions
1075        .iter()
1076        .copied()
1077        .collect();
1078    enabled.sort_unstable();
1079    println!("  {} {:?}", "Enabled versions:".bold(), enabled);
1080    if let Some(latest) = refreshed.highest_package_version() {
1081        println!("  {} v{latest}", "Latest package:".bold());
1082    }
1083    if enabled.contains(&current_version) {
1084        print_warning(&format!(
1085            "v{current_version} stays enabled (non-exclusive upgrade): execute a \
1086             DisableVersion({current_version}) proposal through the new package when \
1087             the fleet is ready."
1088        ));
1089    }
1090    Ok(())
1091}

1093/// Create an upgrade proposal.
1094///
1095/// Exactly one of `digest` or `package_path` must be `Some`. When
1096/// `package_path` is provided, the CLI builds the package via `sui move build`
1097/// and verifies that its `PACKAGE_VERSION` constant is exactly +1 of the
1098/// currently published version (pre-flight check) before submitting the
1099/// proposal. The `--digest` path skips that check and is retained only for
1100/// callers with a pre-built package; combined with an exclusive upgrade it is
1101/// refused unless `allow_unverified_exclusive` acknowledges the skipped check.
1102pub struct CreateUpgradeProposalArgs<'a> {
1103    pub digest: Option<&'a str>,
1104    pub package_path: Option<&'a std::path::Path>,
1105    pub sui_binary: &'a std::path::Path,
1106    pub sui_client_config: Option<&'a std::path::Path>,
1107    pub exclusive: bool,
1108    pub allow_unverified_exclusive: bool,
1109    pub metadata: Vec<(String, String)>,
1110}
1111
1112/// Refuse the brick-capable flag combination: an exclusive upgrade proposed
1113/// from a pre-built `--digest`, whose `PACKAGE_VERSION` constant was therefore
1114/// never checked against the chain, unless the operator explicitly
1115/// acknowledged the bypass.
1116fn check_exclusive_digest_acknowledged(
1117    digest: Option<&str>,
1118    exclusive: bool,
1119    allow_unverified_exclusive: bool,
1120) -> Result<()> {
1121    if digest.is_some() && exclusive && !allow_unverified_exclusive {
1122        anyhow::bail!(
1123            "--digest skips the PACKAGE_VERSION pre-flight, and an exclusive upgrade \
1124             publishing a package whose PACKAGE_VERSION does not match the new \
1125             on-chain version permanently bricks the contract with no on-chain \
1126             recovery. Use --package-path so the constant is verified, or pass \
1127             --allow-unverified-exclusive after manually verifying that the \
1128             pre-built package declares PACKAGE_VERSION = current on-chain \
1129             version + 1."
1130        );
1131    }
1132    Ok(())
1133}
1134
1135pub async fn create_upgrade_proposal(
1136    config: &CliConfig,
1137    args: CreateUpgradeProposalArgs<'_>,
1138    tx_opts: &TxOptions,
1139) -> Result<()> {
1140    let CreateUpgradeProposalArgs {
1141        digest,
1142        package_path,
1143        sui_binary,
1144        sui_client_config,
1145        exclusive,
1146        allow_unverified_exclusive,
1147        metadata,
1148    } = args;
1149    check_exclusive_digest_acknowledged(digest, exclusive, allow_unverified_exclusive)?;
1150    let mut client = HashiClient::new(config).await?;
1151
1152    let digest_bytes = match (digest, package_path) {
1153        (Some(d), None) => {
1154            print_warning(
1155                "--digest skips pre-flight checks (PACKAGE_VERSION = current + 1). \
1156                 Prefer --package-path.",
1157            );
1158            hex::decode(d.trim_start_matches("0x")).context("Invalid digest hex")?
1159        }
1160        (None, Some(path)) => {
1161            let current_version = client.highest_package_version().context(
1162                "could not determine current package version from on-chain state; \
1163                 is the package deployed?",
1164            )?;
1165            let expected_version = current_version + 1;
1166            print_info(&format!(
1167                "Building upgrade package at {} (expecting PACKAGE_VERSION = {expected_version})",
1168                path.display()
1169            ));
1170            let (_compiled, digest) = crate::cli::upgrade::build_upgrade_package(
1171                sui_binary,
1172                path,
1173                sui_client_config,
1174                expected_version,
1175            )
1176            .context("failed to build upgrade package")?;
1177            digest
1178        }
1179        (None, None) => {
1180            anyhow::bail!("must provide either --digest or --package-path");
1181        }
1182        (Some(_), Some(_)) => unreachable!("clap enforces mutual exclusion"),
1183    };
1184
1185    print_detail(&format!("\n{}", "Creating Upgrade Proposal:".bold()));
1186    print_detail(&format!("  Digest: 0x{}", hex::encode(&digest_bytes)));
1187    print_detail(&format!("  Exclusive: {exclusive}"));
1188    print_metadata(&metadata);
1189    print_acting_validator(&client)?;
1190
1191    if !prompt_continue("create this upgrade proposal", tx_opts).await? {
1192        crate::cli::print_warning("Aborted.");
1193        return Ok(());
1194    }
1195
1196    let tx = client.build_create_proposal_transaction(CreateProposalParams::Upgrade {
1197        digest: digest_bytes,
1198        exclusive,
1199        metadata,
1200    })?;
1201    let module = "upgrade";
1202
1203    print_info(&format!("Transaction: {module}::propose"));
1204    let response = execute_or_simulate(&mut client, tx, tx_opts).await?;
1205    print_created_proposal_id(response.as_ref());
1206    Ok(())
1207}

1363pub async fn create_update_mpc_config_proposal(
1364    config: &CliConfig,
1365    max_faulty_bps: Option<u64>,
1366    weight_reduction_allowed_delta: Option<u64>,
1367    nonce_generation_protocol: Option<u64>,
1368    metadata: Vec<(String, String)>,
1369    tx_opts: &TxOptions,
1370) -> Result<()> {
1371    const MAX_BPS: u64 = 10_000;
1372    const MAX_FAULTY_BPS: u64 = 3_333;
1373    if let Some(f) = max_faulty_bps {
1374        anyhow::ensure!(
1375            (1..=MAX_FAULTY_BPS).contains(&f),
1376            "--max-faulty-bps must be in 1..={MAX_FAULTY_BPS}, got {f}"
1377        );
1378    }
1379    if let Some(d) = weight_reduction_allowed_delta {
1380        anyhow::ensure!(
1381            d <= MAX_BPS,
1382            "--weight-reduction-allowed-delta must be in 0..={MAX_BPS}, got {d}"
1383        );
1384    }
1385    if let (Some(f), Some(d)) = (max_faulty_bps, weight_reduction_allowed_delta) {
1386        anyhow::ensure!(
1387            d < f,
1388            "--weight-reduction-allowed-delta must be below --max-faulty-bps ({f}), got {d}; \
1389             pinning would silently clamp it to {}",
1390            f - 1
1391        );
1392    }
1393    if let Some(p) = nonce_generation_protocol {
1394        anyhow::ensure!(
1395            p <= 1,
1396            "--nonce-generation-protocol must be 0 (vanilla) or 1 (avid), got {p}"
1397        );
1398    }
1399
1400    let count = [
1401        max_faulty_bps,
1402        weight_reduction_allowed_delta,
1403        nonce_generation_protocol,
1404    ]
1405    .iter()
1406    .filter(|v| v.is_some())
1407    .count();
1408    if count == 0 {
1409        anyhow::bail!(
1410            "must provide at least one of --max-faulty-bps, --weight-reduction-allowed-delta, --nonce-generation-protocol"
1411        );
1412    }
1413
1414    let mut client = HashiClient::new(config).await?;
1415    print_acting_validator(&client)?;
1416
1417    if !prompt_continue("create this MPC config update proposal", tx_opts).await? {
1418        crate::cli::print_warning("Aborted.");
1419        return Ok(());
1420    }
1421
1422    let tx = client.build_create_proposal_transaction(CreateProposalParams::UpdateMpcConfig {
1423        max_faulty_bps,
1424        weight_reduction_allowed_delta,
1425        nonce_generation_protocol,
1426        metadata,
1427    })?;
1428
1429    let response = execute_or_simulate(&mut client, tx, tx_opts).await?;

1434/// Parse a CLI config value string like "u64:1000" or "bool:true" into a ConfigValueParam.
1435fn parse_config_value(s: &str) -> Result<hashi_types::move_types::ConfigValue> {
1436    use hashi_types::move_types::ConfigValue;
1437
1438    let (type_prefix, raw) = s
1439        .split_once(':')
1440        .ok_or_else(|| anyhow::anyhow!("expected type:value format (e.g. u64:1000)"))?;
1441
1442    match type_prefix {
1443        "u64" => Ok(ConfigValue::U64(raw.parse().context("invalid u64")?)),
1444        "bool" => Ok(ConfigValue::Bool(raw.parse().context("invalid bool")?)),
1445        "string" => Ok(ConfigValue::String(raw.to_string())),
1446        "address" => Ok(ConfigValue::Address(
1447            raw.parse().context("invalid address")?,
1448        )),
1449        other => anyhow::bail!(
1450            "unknown type prefix '{}' (expected u64, bool, string, address)",
1451            other
1452        ),
1453    }
1454}

1623/// Create a proposal to ignore (or re-admit) a registered committee member.
1624pub async fn create_ignore_member_proposal(
1625    config: &CliConfig,
1626    validator: &str,
1627    unignore: bool,
1628    metadata: Vec<(String, String)>,
1629    tx_opts: &TxOptions,
1630) -> Result<()> {
1631    let target: Address = validator
1632        .parse()
1633        .with_context(|| format!("invalid validator address: {validator}"))?;
1634    let action = if unignore { "Un-ignore" } else { "Ignore" };
1635    let title = format!("Creating {action} Member Proposal:");
1636    print_detail(&format!("\n{}", title.bold()));
1637    print_detail(&format!("  Target: {}", target));
1638    print_metadata(&metadata);
1639
1640    let client = HashiClient::new(config).await?;
1641
1642    // The flag is only read at the next committee formation. Tell the
1643    // operator when the change will actually bite.
1644    match client.onchain_state().pending_epoch_change() {
1645        Some(pending) => print_detail(&format!(
1646            "  Effect: a reconfiguration to epoch {pending} is already in flight — the change \
1647             takes effect one epoch later, at the formation after it completes",
1648        )),
1649        None => print_detail(
1650            "  Effect: at the next committee formation (start of the next reconfiguration)",
1651        ),
1652    }
1653
1654    if !unignore {
1655        // Warn when the target carries so much weight that excluding it
1656        // approaches the BFT bound (see ignore_member.move's module doc).
1657        if let Some(committee) = client.onchain_state().current_committee() {
1658            let weight = committee.weight_of(&target).unwrap_or(0);
1659            let total = committee.total_weight();
1660            if total > 0 && weight * 10_000 / total > 2_500 {
1661                print_detail(&format!(
1662                    "  WARNING: target holds {weight} of {total} committee weight (> 25%). \
1663                     Exclusion only works while non-participating weight stays at or below \
1664                     1/3 of the committee — beyond that, governance and reconfiguration are \
1665                     both blocked."
1666                ));
1667            }
1668        }
1669    }
1670
1671    print_acting_validator(&client)?;
1672
1673    if !prompt_continue(
1674        &format!("create this {} member proposal", action.to_lowercase()),
1675        tx_opts,
1676    )
1677    .await?
1678    {
1679        crate::cli::print_warning("Aborted.");
1680        return Ok(());
1681    }
1682
1683    let mut client = client;
1684    let tx = client.build_create_proposal_transaction(CreateProposalParams::IgnoreMember {
1685        target_validator_address: target,
1686        ignored: !unignore,
1687        metadata,
1688    })?;
1689
1690    print_info("Transaction: ignore_member::propose");
1691    let response = execute_or_simulate(&mut client, tx, tx_opts).await?;
1692    print_created_proposal_id(response.as_ref());
1693    Ok(())
1694}

1767/// Ask the operator to confirm before a real execution. Returns `true` to
1768/// proceed. Always `true` with `-y/--yes`, or in dry-run and
1769/// serialize-unsigned mode, which change no on-chain state. Requires an
1770/// explicit `y`, and refuses when stdin is not a terminal (see
1771/// [`crate::cli::confirm`]).
1772pub(crate) async fn prompt_continue(action: &str, tx_opts: &TxOptions) -> Result<bool> {
1773    use crate::sui_tx_executor::TxMode;
1774
1775    if tx_opts.skip_confirm || tx_opts.mode() != TxMode::Execute {
1776        return Ok(true);
1777    }
1778    eprintln!("\n{}", format!("About to {action}.").yellow());
1779    crate::cli::confirm()
1780}
