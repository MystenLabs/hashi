24pub async fn run(action: WithdrawCommands, config: &CliConfig, tx_opts: &TxOptions) -> Result<()> {
25    match action {
26        WithdrawCommands::Request {
27            amount,
28            btc_address,
29            count,
30        } => request(config, tx_opts, amount, &btc_address, count).await,
31        WithdrawCommands::Cancel { request_id } => cancel(config, tx_opts, &request_id).await,
32        WithdrawCommands::Status { request_id } => status(config, &request_id).await,
33        WithdrawCommands::List {
34            output_format,
35            json,
36        } => {
37            let output_format = if json {
38                OutputFormat::Json
39            } else {
40                output_format
41            };
42            list(config, output_format).await
43        }
44    }
45}

47async fn request(
48    config: &CliConfig,
49    tx_opts: &TxOptions,
50    amount: u64,
51    btc_address: &str,
52    count: usize,
53) -> Result<()> {
54    use crate::sui_tx_executor::TxMode;
55
56    config.validate()?;
57    anyhow::ensure!(count >= 1, "--count must be at least 1");
58
59    let hashi_ids = crate::config::HashiIds {
60        package_id: config.package_id(),
61        hashi_object_id: config.hashi_object_id(),
62    };
63
64    // A keypair is optional: serialize/dry-run only need the sender address.
65    let signer = config.load_keypair()?;
66    if tx_opts.mode() == TxMode::Execute && signer.is_none() {
67        anyhow::bail!(
68            "Keypair required to submit a withdrawal request (set keypair_path in config), \
69             or use --serialize-unsigned-transaction to emit an unsigned transaction."
70        );
71    }
72
73    // Sender: explicit --sender (e.g. a multisig), else the keypair's address.
74    // The BTC balance is drawn from this sender during the build.
75    let sender = tx_opts
76        .sender
77        .or_else(|| signer.as_ref().map(|s| s.verifying_key().derive_address()));
78
79    // Parse the BTC destination address and verify it matches the configured network
80    let btc_network = crate::btc_monitor::config::parse_btc_network(
81        config.bitcoin.as_ref().and_then(|b| b.network.as_deref()),
82    )?;
83    let btc_addr: BitcoinAddress<NetworkUnchecked> =
84        btc_address.parse().context("Invalid Bitcoin address")?;
85    let btc_addr = btc_addr
86        .require_network(btc_network)
87        .context("Withdrawal address does not match the configured Bitcoin network")?;
88    let destination_bytes = witness_program_from_address(&btc_addr)?;
89
90    // `request_withdrawal` gates on `assert_version_enabled`, so the call
91    // must target the active version's package (a retired version's entry
92    // aborts). The resolved state also routes the batch executor below.
93    let (onchain, call_package) = super::resolve_active_call_package(config, hashi_ids).await?;
94
95    let mut client = crate::sui_rpc_client::new_sui_rpc_client(&config.sui_rpc_url)?;
96
97    // A single request supports all tx modes (execute / dry-run /
98    // serialize-unsigned) via the builder + finalize path.
99    if count == 1 {
100        print_info(&format!("Withdrawal amount: {amount} sats"));
101        print_info(&format!("BTC destination: {btc_address}"));
102
103        let builder = crate::sui_tx_executor::build_create_withdrawal_request(
104            hashi_ids,
105            call_package,
106            amount,
107            destination_bytes,
108        );
109
110        match tx_opts.mode() {
111            TxMode::SerializeUnsigned => print_info("Building unsigned withdrawal request..."),
112            TxMode::DryRun => print_info("Simulating withdrawal request (dry-run)..."),
113            TxMode::Execute => print_info("Submitting withdrawal request on Sui..."),
114        }
115
116        let outcome = crate::sui_tx_executor::finalize(
117            &mut client,
118            signer.as_ref(),
119            builder,
120            sender,
121            &tx_opts.gas_overrides(),
122            tx_opts.mode(),
123            std::time::Duration::from_secs(10),
124        )
125        .await
126        .map_err(crate::cli::explain_tx_error)?;
127
128        if let Some(response) = crate::cli::print_tx_outcome(outcome, &config.sui_rpc_url) {
129            let request_id =
130                crate::sui_tx_executor::withdrawal_request_id_from_response(&response)?;
131            print_success(&format!("Withdrawal request created: {request_id}"));
132        }
133        return Ok(());
134    }
135
136    // `--count > 1` bulk-submits many requests across PTBs and only makes sense
137    // for direct execution: there is no batch builder to emit or simulate, so
138    // --serialize-unsigned-transaction / --dry-run apply to a single request.
139    anyhow::ensure!(
140        tx_opts.mode() == TxMode::Execute,
141        "--count > 1 only supports execute mode; --serialize-unsigned-transaction \
142         and --dry-run apply to a single withdrawal request"
143    );
144
145    // Execute mode guarantees a signer (rejected above otherwise). The
146    // attached reader state makes the executor route its withdrawal calls
147    // through the withdrawal-effective version's package.
148    let signer = signer.expect("execute mode requires a signer");
149    let mut executor = crate::sui_tx_executor::SuiTxExecutor::new(client, signer, hashi_ids)
150        .with_onchain_state(&onchain);
151
152    // ~3 PTB commands per request (split + call) vs the 1024 command cap.
153    const CHUNK_SIZE: usize = 250;
154
155    print_info(&format!(
156        "Submitting {count} withdrawal requests of {amount} sats to {btc_address} ({CHUNK_SIZE} per PTB)...",
157    ));
158
159    let total_chunks = count.div_ceil(CHUNK_SIZE);
160    let mut chunk_idx = 0usize;
161    let mut submitted = 0usize;
162    let mut remaining = count;
163    while remaining > 0 {
164        chunk_idx += 1;
165        let this_batch = remaining.min(CHUNK_SIZE);
166        print_info(&format!(
167            "Batch {chunk_idx}/{total_chunks} ({this_batch} requests)...",
168        ));
169        let ids = executor
170            .execute_create_withdrawal_requests_batch(amount, destination_bytes.clone(), this_batch)
171            .await
172            .map_err(crate::cli::explain_tx_error)?;
173        submitted += ids.len();
174        remaining -= this_batch;
175    }
176
177    print_success(&format!("Created {submitted} withdrawal requests"));
178
179    Ok(())
180}

182async fn cancel(config: &CliConfig, tx_opts: &TxOptions, request_id: &str) -> Result<()> {
183    use crate::sui_tx_executor::TxMode;
184
185    config.validate()?;
186
187    let req_addr = request_id
188        .parse::<sui_sdk_types::Address>()
189        .context("Invalid request ID")?;
190
191    let hashi_ids = crate::config::HashiIds {
192        package_id: config.package_id(),
193        hashi_object_id: config.hashi_object_id(),
194    };
195
196    let signer = config.load_keypair()?;
197    if tx_opts.mode() == TxMode::Execute && signer.is_none() {
198        anyhow::bail!(
199            "Keypair required to cancel a withdrawal, or use \
200             --serialize-unsigned-transaction to emit an unsigned transaction."
201        );
202    }
203
204    // The refunded Balance<BTC> is sent to `sender`, which must equal the
205    // transaction sender. Required up front so the PTB can address the refund.
206    let sender = tx_opts
207        .sender
208        .or_else(|| signer.as_ref().map(|s| s.verifying_key().derive_address()))
209        .context(
210            "No sender available: pass --sender (the refund recipient) or configure a keypair",
211        )?;
212
213    // Resolve the active version's package so the cancel runs the bytecode
214    // generation the flow is committing under. A resolution failure
215    // must abort the command, not fall back: a v1-routed cancel aimed at a
216    // v2-committed request would destroy a request its live withdrawal txn
217    // still references.
218    let (_onchain, call_package) = super::resolve_active_call_package(config, hashi_ids).await?;
219    let builder =
220        crate::sui_tx_executor::build_cancel_withdrawal(hashi_ids, call_package, &req_addr, sender);
221
222    match tx_opts.mode() {
223        TxMode::SerializeUnsigned => print_info("Building unsigned withdrawal cancellation..."),
224        TxMode::DryRun => print_info("Simulating withdrawal cancellation (dry-run)..."),
225        TxMode::Execute => print_info("Cancelling withdrawal..."),
226    }
227
228    let mut client = crate::sui_rpc_client::new_sui_rpc_client(&config.sui_rpc_url)?;
229    let outcome = crate::sui_tx_executor::finalize(
230        &mut client,
231        signer.as_ref(),
232        builder,
233        Some(sender),
234        &tx_opts.gas_overrides(),
235        tx_opts.mode(),
236        std::time::Duration::from_secs(10),
237    )
238    .await
239    .map_err(crate::cli::explain_tx_error)?;
240
241    if crate::cli::print_tx_outcome(outcome, &config.sui_rpc_url).is_some() {
242        print_success("Withdrawal cancelled.");
243    }
244
245    Ok(())
246}

248async fn status(config: &CliConfig, request_id: &str) -> Result<()> {
249    let client = HashiClient::new_with_bitcoin_state(config).await?;
250
251    let req_addr = request_id
252        .parse::<sui_sdk_types::Address>()
253        .context("Invalid request ID")?;
254
255    let withdrawal_requests = client.fetch_withdrawal_requests()?;
256    let withdrawal_txns = client.fetch_withdrawal_txns()?;
257
258    println!("\n{}", "Withdrawal Status".bold());
259    println!("{}", "━".repeat(60).dimmed());
260
261    // Check the mirrored request map first. With deferred archival a request
262    // stays here for its whole live lifecycle — awaiting approval, then
263    // commitment, then committed into a withdrawal txn until the archival GC
264    // moves it to the processed archive.
265    if let Some(wr) = withdrawal_requests.iter().find(|w| w.id == req_addr) {
266        println!(
267            "  {} {}",
268            "Request ID:".bold(),
269            display::format_address_full(&wr.id)
270        );
271        println!("  {} {} sats", "Amount:".bold(), wr.btc_amount);
272        println!(
273            "  {} {}",
274            "BTC Address:".bold(),
275            hex::encode(&wr.bitcoin_address)
276        );
277        println!(
278            "  {} {}",
279            "Requester:".bold(),
280            display::format_address(&wr.sender)
281        );
282        println!(
283            "  {} {}",
284            "Requested:".bold(),
285            display::format_timestamp(wr.created_timestamp_ms)
286        );
287
288        // Committed into a withdrawal txn (BTC drained): render the txn's
289        // signing progress, looked up by the request's withdrawal_txn_id.
290        if let Some(txn_id) = wr.withdrawal_txn_id {
291            if let Some(pw) = withdrawal_txns.iter().find(|p| p.id == txn_id) {
292                print_txn_progress(config, pw);
293            } else {
294                println!();
295                print_info(&format!(
296                    "Request is committed to withdrawal transaction {} but that \
297                     transaction was not found in the pending queues.",
298                    display::format_address_full(&txn_id)
299                ));
300            }
301        } else {
302            println!();
303            let status_label = if wr.is_approved() {
304                "Approved".green()
305            } else {
306                "Requested".yellow()
307            };
308
309            let step = if wr.is_approved() { 2 } else { 1 };
310            println!("  {} {} ({}/6)", "Progress:".bold(), status_label, step);
311            println!(
312                "    {} Requested",
313                if step >= 1 {
314                    "[done]".green()
315                } else {
316                    "[    ]".dimmed()
317                }
318            );
319            println!(
320                "    {} Approved",
321                if step >= 2 {
322                    "[done]".green()
323                } else {
324                    "[    ]".dimmed()
325                }
326            );
327            println!("    {} Committed", "[    ]".dimmed());
328            println!("    {} Signed", "[    ]".dimmed());
329            println!("    {} Broadcast", "[    ]".dimmed());
330            println!("    {} Confirmed", "[    ]".dimmed());
331        }
332    }
333    // Check committed/signed withdrawal transactions
334    else if let Some(pw) = withdrawal_txns
335        .iter()
336        .find(|p| p.request_ids.contains(&req_addr))
337    {
338        println!(
339            "  {} {}",
340            "Request ID:".bold(),
341            display::format_address_full(&req_addr)
342        );
343        print_txn_progress(config, pw);
344    } else {
345        print_info(
346            "Withdrawal request not found in pending queues (may be confirmed or cancelled).",
347        );
348    }
349
350    println!("{}", "━".repeat(60).dimmed());
351    Ok(())
352}

354/// Render the committed-phase progress checklist (steps 3-6) and Bitcoin-side
355/// context for a withdrawal transaction. Shared by the request-map lookup
356/// (committed requests point at their txn via `withdrawal_txn_id`) and the
357/// txn-map fallback lookup.
358fn print_txn_progress(config: &CliConfig, pw: &WithdrawalTransaction) {
359    let txid: bitcoin::Txid = pw.txid.into();
360    let is_confirmed = pw.is_confirmed();
361    let is_signed = pw.is_fully_signed();
362    let signed_inputs = pw.signing.signed_count();
363    let num_inputs = pw.signing.num_inputs();
364    let step = if is_confirmed {
365        6
366    } else if is_signed {
367        4
368    } else {
369        3
370    };
371    // Distinguish the multi-checkpoint signing window: an in-progress txn
372    // shows "Signing (X/N)" rather than a flat "Committed". A confirmed txn
373    // lingers in the pending map until the archival GC sweeps it, so it must
374    // render as complete rather than broadcast-ready.
375    let status_label = if is_confirmed {
376        "Confirmed (archival pending)".green()
377    } else if is_signed {
378        "Signed".green()
379    } else if signed_inputs > 0 {
380        format!("Signing ({signed_inputs}/{num_inputs})").cyan()
381    } else {
382        "Committed".cyan()
383    };
384
385    println!("  {} {}", "BTC txid:".bold(), txid);
386    println!();
387    println!("  {} {} ({}/6)", "Progress:".bold(), status_label, step);
388    println!("    {} Requested", "[done]".green());
389    println!("    {} Approved", "[done]".green());
390    println!("    {} Committed          txid: {}", "[done]".green(), txid);
391    println!(
392        "    {} Signed",
393        if is_signed {
394            "[done]".green()
395        } else {
396            "[    ]".dimmed()
397        }
398    );
399    println!(
400        "    {} Broadcast",
401        if is_confirmed {
402            "[done]".green()
403        } else {
404            "[    ]".dimmed()
405        }
406    );
407    println!(
408        "    {} Confirmed",
409        if is_confirmed {
410            "[done]".green()
411        } else {
412            "[    ]".dimmed()
413        }
414    );
415
416    // BTC context
417    if let Ok(Some(btc_rpc)) = config.btc_rpc_client() {
418        println!();
419        println!("  {}", "BTC Context:".bold());
420        match btc_rpc.get_raw_transaction_verbose(txid) {
421            Ok(info) => {
422                let confirmations = info.confirmations.unwrap_or(0) as u32;
423                let tx_status = if confirmations > 0 {
424                    "Confirmed".to_string()
425                } else {
426                    "In Mempool".to_string()
427                };
428                println!("    {} {}", "TX Status:".bold(), tx_status);
429                println!("    {} {}/6", "Confirmations:".bold(), confirmations);
430            }
431            Err(_) => {
432                println!("    {}", "(transaction not found on BTC node)".dimmed());
433            }
434        }
435    }
436}

438async fn list(config: &CliConfig, output_format: OutputFormat) -> Result<()> {
439    let client = HashiClient::new_with_bitcoin_state(config).await?;
440
441    let requests = client.fetch_withdrawal_requests()?;
442    let pending = client.fetch_withdrawal_txns()?;
443    // The v2 flow commits requests in place, so the mirrored map holds both
444    // the actionable queue and requests already committed into a withdrawal
445    // txn (those linger until the archival GC sweeps them and are counted
446    // through their txn's request_count).
447    let (queued, committed_in_place) = partition_queued(&requests);
448    // Confirmed txns linger in the pending map until the archival GC sweeps
449    // them; classify them first so they are never counted as actionable
450    // "signed" (they are also fully signed).
451    let confirmed_count = pending.iter().filter(|pw| pw.is_confirmed()).count();
452    let signed_count = pending
453        .iter()
454        .filter(|pw| !pw.is_confirmed() && pw.is_fully_signed())
455        .count();
456    let committed_count = pending.len() - confirmed_count - signed_count;
457
458    match output_format {
459        OutputFormat::Json => {
460            let queued_rows: Vec<_> = queued
461                .iter()
462                .map(|wr| {
463                    serde_json::json!({
464                        "request_id": wr.id.to_string(),
465                        "amount_sats": wr.btc_amount,
466                        "status": queued_status(wr),
467                        "caller": wr.sender.to_string(),
468                        "requested_ms": wr.created_timestamp_ms,
469                    })
470                })
471                .collect();
472
473            let withdrawal_txns: Vec<_> = pending.iter().map(withdrawal_txn_row).collect();
474
475            println!(
476                "{}",
477                serde_json::to_string_pretty(&serde_json::json!({
478                    "queued": queued_rows,
479                    "withdrawal_txns": withdrawal_txns,
480                    "queued_count": queued.len(),
481                    "committed_in_place_count": committed_in_place.len(),
482                    "committed_count": committed_count,
483                    "signed_count": signed_count,
484                    "confirmed_count": confirmed_count,
485                }))?
486            );
487        }
488        OutputFormat::HumanTable => {
489            println!("\n{}", "Withdrawal Requests".bold());
490            println!("{}", "━".repeat(100).dimmed());
491
492            if requests.is_empty() && pending.is_empty() {
493                print_info("No withdrawal requests found.");
494            } else {
495                if !queued.is_empty() {
496                    println!("  {}", "Queued:".bold().underline());
497                    println!(
498                        "  {:<20} {:<14} {:<10} {:<20} {}",
499                        "Request ID".bold(),
500                        "Amount (sats)".bold(),
501                        "Status".bold(),
502                        "Caller".bold(),
503                        "Requested".bold()
504                    );
505                    for wr in &queued {
506                        println!(
507                            "  {:<20} {:<14} {:<10} {:<20} {}",
508                            display::format_address_full(&wr.id),
509                            wr.btc_amount,
510                            queued_status(wr),
511                            display::format_address_full(&wr.sender),
512                            display::format_timestamp(wr.created_timestamp_ms)
513                        );
514                    }
515                }
516
517                if !pending.is_empty() {
518                    if !queued.is_empty() {
519                        println!();
520                    }
521                    println!("  {}", "Withdrawal Transactions:".bold().underline());
522                    for pw in &pending {
523                        let txid: bitcoin::Txid = pw.txid.into();
524                        let status = if pw.is_confirmed() {
525                            "Confirmed (archival pending)"
526                        } else if pw.is_fully_signed() {
527                            "Signed"
528                        } else {
529                            "Committed"
530                        };
531                        println!(
532                            "  txid: {}  status: {}  requests: {}",
533                            txid,
534                            status,
535                            pw.request_ids.len()
536                        );
537                    }
538                }
539
540                println!(
541                    "\n  {} queued, {} committed in place; txns: {} committed, {} signed, \
542                     {} confirmed awaiting archival",
543                    queued.len(),
544                    committed_in_place.len(),
545                    committed_count,
546                    signed_count,
547                    confirmed_count
548                );
549            }
550
551            println!("{}", "━".repeat(100).dimmed());
552        }
553    }
554
555    Ok(())
556}

558/// Split the mirrored request map into the actionable queue (awaiting
559/// approval or commitment) and the requests the v2 flow committed in place.
560/// The latter must not report as queued backlog: their BTC is already
561/// drained into a withdrawal txn, which the txn view counts.
562fn partition_queued(
563    requests: &[WithdrawalRequest],
564) -> (Vec<&WithdrawalRequest>, Vec<&WithdrawalRequest>) {
565    requests.iter().partition(|wr| !wr.is_committed())
566}
567
568/// Lowercase state of a request in the actionable queue, for JSON rows and
569/// the table. Only meaningful for uncommitted requests (see
570/// `partition_queued`).
571fn queued_status(wr: &WithdrawalRequest) -> &'static str {
572    if wr.is_approved() {
573        "approved"
574    } else {
575        "requested"
576    }
577}

579/// The JSON row for one in-flight withdrawal transaction.
580///
581/// The Bitcoin txid comes from the `txid` field; `pw.id` is the Sui object ID
582/// of the `WithdrawalTransaction` and is unrelated to the Bitcoin txid.
583fn withdrawal_txn_row(pw: &WithdrawalTransaction) -> serde_json::Value {
584    let txid: bitcoin::Txid = pw.txid.into();
585    // A confirmed txn is also fully signed, so check confirmation first: it
586    // lingers only until the archival GC sweeps it and must not be reported
587    // as an actionable "signed" txn.
588    let status = if pw.is_confirmed() {
589        "confirmed"
590    } else if pw.is_fully_signed() {
591        "signed"
592    } else {
593        "committed"
594    };
595    serde_json::json!({
596        "txid": txid.to_string(),
597        "status": status,
598        "request_count": pw.request_ids.len(),
599    })
600}

669    /// A request at one of the three mirrored lifecycle points: requested,
670    /// approved (cert recorded), or committed (linked to a withdrawal txn,
671    /// BTC drained).
672    fn request(id: u8, approved: bool, committed: bool) -> WithdrawalRequest {
673        WithdrawalRequest {
674            id: sui_sdk_types::Address::new([id; 32]),
675            sender: sui_sdk_types::Address::new([2; 32]),
676            btc_amount: 1,
677            bitcoin_address: vec![0; 20],
678            created_timestamp_ms: 0,
679            approval_cert: approved.then(|| CommitteeSignature {
680                epoch: 0,
681                signature: Vec::new(),
682                signers_bitmap: Vec::new(),
683            }),
684            approved_timestamp_ms: approved.then_some(1),
685            withdrawal_txn_id: committed.then(|| sui_sdk_types::Address::new([9; 32])),
686            sui_tx_digest: sui_sdk_types::Digest::new([0; 32]),
687            btc: if committed { 0 } else { 1 },
688        }
689    }

691    /// Requests the v2 flow committed in place linger in the mirrored map
692    /// until the archival GC sweeps them; the queued view must exclude them
693    /// (they are already counted through their withdrawal txn).
694    #[test]
695    fn queued_view_excludes_committed_in_place_requests() {
696        let requests = vec![
697            request(1, false, false),
698            request(2, true, false),
699            request(3, true, true),
700        ];
701
702        let (queued, committed_in_place) = partition_queued(&requests);
703        let ids = |group: &[&WithdrawalRequest]| group.iter().map(|wr| wr.id).collect::<Vec<_>>();
704        assert_eq!(ids(&queued), [requests[0].id, requests[1].id]);
705        assert_eq!(ids(&committed_in_place), [requests[2].id]);
706        assert_eq!(
707            queued
708                .iter()
709                .map(|wr| queued_status(wr))
710                .collect::<Vec<_>>(),
711            ["requested", "approved"]
712        );
