24#[derive(Clone, Debug)]
25pub struct RecipientInput {
26    path: PathBuf,
27    subkey: Option<Fingerprint>,
28}
29
30impl FromStr for RecipientInput {
31    type Err = anyhow::Error;
32
33    fn from_str(value: &str) -> Result<Self> {
34        let (path, subkey) = match value.rsplit_once('=') {
35            Some((path, fingerprint)) => (path, Some(parse_recipient_fingerprint(fingerprint)?)),
36            None => (value, None),
37        };
38        anyhow::ensure!(
39            !path.is_empty(),
40            "recipient certificate path must not be empty"
41        );
42        Ok(Self {
43            path: PathBuf::from(path),
44            subkey,
45        })
46    }
47}
48
49fn parse_recipient_fingerprint(value: &str) -> Result<Fingerprint> {
50    anyhow::ensure!(
51        matches!(value.len(), 40 | 64) && value.bytes().all(|byte| byte.is_ascii_hexdigit()),
52        "recipient subkey fingerprint must be a full 40- or 64-digit hexadecimal fingerprint"
53    );
54    value.parse()
55}

57#[derive(Debug)]
58struct RecipientAttestationBytes {
59    device_pem: Vec<u8>,
60    dec_pem: Vec<u8>,
61}
62
63fn load_recipient_attestations(
64    inputs: &[PathBuf],
65) -> Result<BTreeMap<Fingerprint, RecipientAttestationBytes>> {
66    let mut attestations = BTreeMap::new();
67    let mut triples = inputs.chunks_exact(3);
68    anyhow::ensure!(
69        triples.remainder().is_empty(),
70        "each recipient attestation requires SUBKEY_FINGERPRINT DEVICE_PEM DEC_PEM"
71    );
72    for triple in &mut triples {
73        let fingerprint = parse_recipient_fingerprint(
74            triple[0]
75                .to_str()
76                .context("recipient subkey fingerprint must be valid UTF-8")?,
77        )?;
78        anyhow::ensure!(
79            !attestations.contains_key(&fingerprint),
80            "duplicate recipient attestation for subkey {fingerprint}"
81        );
82        let device_pem = fs::read(&triple[1]).with_context(|| {
83            format!(
84                "Failed to read recipient device attestation PEM {}",
85                triple[1].display()
86            )
87        })?;
88        let dec_pem = fs::read(&triple[2]).with_context(|| {
89            format!(
90                "Failed to read recipient DEC attestation PEM {}",
91                triple[2].display()
92            )
93        })?;
94        attestations.insert(
95            fingerprint,
96            RecipientAttestationBytes {
97                device_pem,
98                dec_pem,
99            },
100        );
101    }
102    Ok(attestations)
103}

105pub fn run(
106    primary_cert: &Path,
107    recipients: &[RecipientInput],
108    recipient_attestations: &[PathBuf],
109    require_recipient_attestation: bool,
110    output: &Path,
111    gpg_homedir: Option<&Path>,
112) -> Result<()> {
113    let primary = Cert::from_file(primary_cert).with_context(|| {
114        format!(
115            "Failed to read primary public certificate {}",
116            primary_cert.display()
117        )
118    })?;
119    let recipient_certs = recipients
120        .iter()
121        .map(|recipient| {
122            Cert::from_file(&recipient.path).with_context(|| {
123                format!(
124                    "Failed to read recipient public certificate {}",
125                    recipient.path.display()
126                )
127            })
128        })
129        .collect::<Result<Vec<_>>>()?;
130    let recipients = recipients
131        .iter()
132        .zip(&recipient_certs)
133        .map(|(input, cert)| EncryptionRecipient {
134            cert,
135            subkey: input.subkey.as_ref(),
136        })
137        .collect::<Vec<_>>();
138    let attestation_bytes = load_recipient_attestations(recipient_attestations)?;
139    let attestations = attestation_bytes
140        .iter()
141        .map(|(fingerprint, evidence)| {
142            (
143                fingerprint.clone(),
144                RecipientAttestation {
145                    device_pem: &evidence.device_pem,
146                    dec_pem: &evidence.dec_pem,
147                },
148            )
149        })
150        .collect::<BTreeMap<_, _>>();
151    let context = match gpg_homedir {
152        Some(home) => sequoia_gpg_agent::Context::with_homedir(home),
153        None => sequoia_gpg_agent::Context::new(),
154    }
155    .context(
156        "Failed to initialize GnuPG context; gpgconf and an existing GnuPG home are required",
157    )?;
158    context
159        .start("gpg-agent")
160        .context("Failed to start the GnuPG agent")?;
161    let mut signer =
162        sequoia_gpg_agent::KeyPair::new_for_gnupg_context(&context, primary.primary_key().key())
163            .context("Failed to create the gpg-agent signer for the primary key")?;
164    let assembled = assemble_public_cert(
165        &primary,
166        &recipients,
167        &attestations,
168        require_recipient_attestation,
169        &mut signer,
170    )
171    .context("Failed to assemble backup certificate")?;
172    let cert = Cert::from_bytes(assembled.armored().as_bytes())
173        .context("Failed to parse assembled public certificate")?;
174    let policy = StandardPolicy::new();
175    let valid_cert = cert
176        .with_policy(&policy, None)
177        .context("Failed to validate assembled public certificate for expiry reporting")?;
178    let primary_expiry = valid_cert.primary_key().key_expiration_time();
179    let format_expiry = |expiry: Option<std::time::SystemTime>| -> Result<String> {
180        match expiry {
181            Some(expiry) => Ok(jiff::Timestamp::try_from(expiry)?.to_string()),
182            None => Ok("never".to_owned()),
183        }
184    };
185    let primary_expiry_display = format_expiry(primary_expiry)?;
186    let recipient_expiries = valid_cert
187        .keys()
188        .subkeys()
189        .map(|key| {
190            Ok((
191                key.key().fingerprint(),
192                format_expiry(key.key_expiration_time())?,
193            ))
194        })
195        .collect::<Result<Vec<_>>>()?;
196
197    let parent = output
198        .parent()
199        .filter(|parent| !parent.as_os_str().is_empty())
200        .unwrap_or_else(|| Path::new("."));
201    let mut staging = tempfile::Builder::new()
202        .prefix(".hashi-assemble-cert-")
203        .tempfile_in(parent)
204        .with_context(|| {
205            format!(
206                "Failed to create certificate staging file in {}",
207                parent.display()
208            )
209        })?;
210    staging
211        .write_all(assembled.armored().as_bytes())
212        .context("Failed to write assembled public certificate")?;
213    staging
214        .as_file()
215        .sync_all()
216        .context("Failed to sync assembled public certificate")?;
217    staging
218        .persist_noclobber(output)
219        .map_err(|error| error.error)
220        .with_context(|| {
221            format!(
222                "Failed to publish certificate {}; output must not exist",
223                output.display()
224            )
225        })?;
226
227    print_success(&format!(
228        "Assembled backup certificate written to {}",
229        output.display()
230    ));
231    print_info(&format!("Primary fingerprint: {}", cert.fingerprint()));
232    print_info(&format!(
233        "Certifier primary key expiry (UTC): {primary_expiry_display}"
234    ));
235    for (fingerprint, expiry) in recipient_expiries {
236        print_info(&format!(
237            "Recipient encryption subkey fingerprint: {fingerprint}; own key expiry (UTC): {expiry}"
238        ));
239    }
240    if primary_expiry.is_some() {
241        print_warning(&format!(
242            "Certifier key expires at {primary_expiry_display}. This limits the whole certificate, \
243             even recipients whose own key expiry is later or never. After certifier expiry, \
244             new backups fail and node startup rejects the expired configured certificate. \
245             Replace the certificate before expiry. Expiry does not make existing backups \
246             cryptographically undecryptable."
247        ));
248    }
249    Ok(())
250}

257    #[test]
258    fn recipient_input_parses_paths_and_full_fingerprints() {
259        let path = "certs/recipient public.asc";
260        let input: RecipientInput = path.parse().unwrap();
261        assert_eq!(input.path, Path::new(path));
262        assert!(input.subkey.is_none());
263
264        for fingerprint in ["a".repeat(40), "B".repeat(64)] {
265            let input: RecipientInput = format!("{path}={fingerprint}").parse().unwrap();
266            assert_eq!(input.path, Path::new(path));
267            assert_eq!(input.subkey, Some(fingerprint.parse().unwrap()));
268        }
269
270        for invalid in [
271            String::new(),
272            format!("={}", "A".repeat(40)),
273            format!("{path}="),
274            format!("{path}=0123456789ABCDEF"),
275            format!("{path}={}", "G".repeat(40)),
276        ] {
277            assert!(invalid.parse::<RecipientInput>().is_err(), "{invalid}");
278        }
279    }

281    #[test]
282    fn assemble_cert_cli_parses_recipient_attestation_triples() {
283        use clap::Parser as _;
284
285        #[derive(clap::Parser)]
286        struct TestCli {
287            #[command(subcommand)]
288            command: crate::cli::BackupCommands,
289        }
290
291        let first = "a".repeat(40);
292        let second = "B".repeat(64);
293        let mut args = vec![
294            "hashi",
295            "assemble-cert",
296            "--primary-cert",
297            "primary.asc",
298            "--recipient",
299            "recipient.asc",
300            "--output",
301            "assembled.asc",
302            "--require-recipient-attestation",
303            "--recipient-attestation",
304            &first,
305            "device one.pem",
306            "dec one.pem",
307            "--recipient-attestation",
308            &second,
309            "device two.pem",
310            "dec two.pem",
311        ];
312        let crate::cli::BackupCommands::AssembleCert {
313            recipient_attestation,
314            require_recipient_attestation,
315            ..
316        } = TestCli::try_parse_from(&args).unwrap().command
317        else {
318            panic!("expected assemble-cert");
319        };
320        assert!(require_recipient_attestation);
321        assert_eq!(
322            recipient_attestation,
323            [
324                first.as_str(),
325                "device one.pem",
326                "dec one.pem",
327                second.as_str(),
328                "device two.pem",
329                "dec two.pem",
330            ]
331            .map(PathBuf::from)
332        );
333        args.pop();
334        assert!(TestCli::try_parse_from(&args).is_err());
335    }

337    #[test]
338    fn recipient_attestation_files_reject_case_insensitive_duplicate_fingerprints() {
339        let dir = TempDir::new().unwrap();
340        let device = dir.path().join("device.pem");
341        let dec = dir.path().join("dec.pem");
342        fs::write(&device, b"device evidence").unwrap();
343        fs::write(&dec, b"DEC evidence").unwrap();
344        let fingerprint = "a".repeat(40);
345        let mut inputs = vec![PathBuf::from(&fingerprint), device.clone(), dec.clone()];
346        inputs.extend([PathBuf::from(fingerprint.to_uppercase()), device, dec]);
347        assert!(
348            load_recipient_attestations(&inputs)
349                .unwrap_err()
350                .to_string()
351                .contains("duplicate recipient attestation")
352        );
353    }
