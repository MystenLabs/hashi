23/// List committee members
24pub async fn list_members(config: &CliConfig, epoch: Option<u64>) -> Result<()> {
25    let client = HashiClient::new(config).await?;
26
27    let current_epoch = client.fetch_epoch();
28
29    // Note: Currently only the current epoch's members are available
30    if let Some(requested_epoch) = epoch
31        && requested_epoch != current_epoch
32    {
33        print_warning(&format!(
34            "Only current epoch ({}) data is available. Showing current epoch.",
35            current_epoch
36        ));
37    }
38
39    print_info(&format!(
40        "Fetching committee for epoch {}...",
41        current_epoch
42    ));
43
44    let members = client.fetch_committee_members();
45
46    if members.is_empty() {
47        println!("\n{}", "No committee members found.".dimmed());
48        print_warning("This may indicate the committee data could not be fetched.");
49        return Ok(());
50    }
51
52    println!("\n👥 Committee Members (Epoch {}):\n", current_epoch);
53
54    #[derive(Tabled)]
55    struct MemberRow {
56        #[tabled(rename = "Validator Address")]
57        validator: String,
58        #[tabled(rename = "Operator Address")]
59        operator: String,
60        #[tabled(rename = "Ignored")]
61        ignored: String,
62        #[tabled(rename = "Resigned")]
63        resigned: String,
64    }
65
66    let rows: Vec<MemberRow> = members
67        .iter()
68        .map(|m| MemberRow {
69            validator: display::format_address(&m.validator_address),
70            operator: display::format_address(&m.operator_address),
71            // Registry view: shows the governance flag as soon as it is set,
72            // before the epoch-boundary effect on the committee.
73            ignored: if m.ignored {
74                "yes".to_string()
75            } else {
76                String::new()
77            },
78            resigned: if m.resigned {
79                "yes".to_string()
80            } else {
81                String::new()
82            },
83        })
84        .collect();
85
86    let table = Table::new(rows).to_string();
87    println!("{}", table);
88
89    println!(
90        "\n  {} {} member(s)",
91        "ℹ".blue(),
92        members.len().to_string().bold()
93    );
94
95    Ok(())
96}

98/// View a specific committee member
99pub async fn view_member(config: &CliConfig, address: &str) -> Result<()> {
100    let client = HashiClient::new(config).await?;
101
102    let member_addr =
103        Address::from_hex(address).with_context(|| format!("Invalid address: {}", address))?;
104
105    print_info(&format!("Fetching member info for {}...", address));
106
107    let members = client.fetch_committee_members();
108
109    let member = members.iter().find(|m| m.validator_address == member_addr);
110
111    match member {
112        Some(m) => {
113            println!("\n{}", "Committee Member Details:".bold());
114            println!("{}", "━".repeat(60).dimmed());
115            println!(
116                "  {} {}",
117                "Validator:".bold(),
118                display::format_address_full(&m.validator_address).cyan()
119            );
120            println!(
121                "  {} {}",
122                "Operator:".bold(),
123                display::format_address_full(&m.operator_address)
124            );
125            if let Some(uri) = &m.endpoint_url {
126                println!("  {} {}", "Endpoint:".bold(), uri);
127            }
128            if m.ignored {
129                println!(
130                    "  {} yes (excluded from the next committee formation)",
131                    "Ignored:".bold()
132                );
133            }
134            if m.resigned {
135                println!(
136                    "  {} yes (removed at the next epoch transition)",
137                    "Resigned:".bold()
138                );
139            }
140            println!("{}", "━".repeat(60).dimmed());
141        }
142        None => {
143            print_warning(&format!(
144                "Address {} is not a member of the current committee.",
145                display::format_address(&member_addr)
146            ));
147        }
148    }
149
150    Ok(())
151}

178/// Refuse an abort the chain would reject, from the scraped pending epoch and
179/// Sui's current epoch: nothing pending (`reconfig::ENotReconfiguring`), or a
180/// pending epoch that is still Sui's current epoch
181/// (`committee_set::EPendingEpochStillCurrent`). Returns the epoch the abort
182/// would tear down.
183pub fn refuse_unabortable_reconfig(pending_epoch: Option<u64>, sui_epoch: u64) -> Result<u64> {
184    let Some(pending_epoch) = pending_epoch else {
185        anyhow::bail!("no reconfiguration is in progress; there is nothing to abort");
186    };
187    anyhow::ensure!(
188        pending_epoch != sui_epoch,
189        "the pending reconfiguration targets Sui's current epoch ({sui_epoch}), so it may still \
190         complete; the chain refuses the abort until Sui's epoch moves past it"
191    );
192    Ok(pending_epoch)
193}
194
195/// Abort a reconfiguration that has overrun its Sui epoch
196/// (`reconfig::abort_reconfig`). Permissionless: any funded signer may send
197/// it, and no vote is involved. Running nodes do this themselves at the
198/// epoch boundary, so this is the manual fallback.
199pub async fn abort_reconfig(config: &CliConfig, tx_opts: &TxOptions) -> Result<()> {
200    let mut client = HashiClient::new(config).await?;
201    let sui_epoch = client.fetch_sui_epoch().await?;
202    let pending_epoch =
203        refuse_unabortable_reconfig(client.onchain_state().pending_epoch_change(), sui_epoch)?;
204
205    print_detail(&format!(
206        "\n{}",
207        "Aborting the pending reconfiguration:".bold()
208    ));
209    print_detail(&format!("  Pending Hashi epoch: {pending_epoch}"));
210    print_detail(&format!("  Current Sui epoch:   {sui_epoch}"));
211    print_detail(&format!(
212        "  Effect: the pending committee is discarded and Hashi stays at epoch {} under its \
213         current committee. Running nodes abort an overrun reconfiguration themselves at the \
214         Sui epoch boundary, so this is the manual fallback.",
215        client.fetch_epoch()
216    ));
217    print_detail(&replacement_note(
218        client
219            .onchain_state()
220            .state()
221            .hashi()
222            .config
223            .reconfig_hold(),
224    ));
225
226    if !prompt_continue("abort the pending reconfiguration", tx_opts).await? {
227        print_warning("Aborted.");
228        return Ok(());
229    }
230
231    let tx = client.build_abort_reconfig_transaction(pending_epoch)?;
232    print_info("Transaction: reconfig::abort_reconfig");
233    execute_or_simulate(&mut client, tx, tx_opts).await?;
234    Ok(())
235}

237/// What happens to the replacement reconfiguration after an abort, given the
238/// on-chain `reconfig_hold` flag.
239fn replacement_note(reconfig_hold: bool) -> String {
240    if reconfig_hold {
241        "  Replacement: governance holds reconfiguration (reconfig_hold is set), so the chain \
242         refuses start_reconfig and no node submits one; the current committee keeps serving \
243         until an update-config proposal clears the flag."
244            .to_string()
245    } else {
246        "  Replacement: nodes that observe the abort submit a fresh start_reconfig right away. \
247         If no node is running, the replacement starts when one comes up. To stay on the \
248         current committee instead, have governance set reconfig_hold first (update-config)."
249            .to_string()
250    }
251}

253/// Refuse a `start_reconfig` the chain would reject, mirroring its asserts
254/// in order: governance holds reconfiguration (`reconfig::EReconfigHeld`),
255/// a reconfiguration is already pending, a committee already exists for
256/// Sui's current epoch, or Hashi is already on Sui's epoch (genesis at epoch
257/// 0 is the one case the chain lets through with equal epochs). The launch
258/// switch is not mirrored: a genesis start the chain refuses for that reason
259/// fails with `EGenesisNotAuthorized`, and `hashi launch --status` shows why.
260pub fn refuse_unstartable_reconfig(
261    reconfig_hold: bool,
262    pending_epoch: Option<u64>,
263    hashi_epoch: u64,
264    sui_epoch: u64,
265    committee_exists_for_sui_epoch: bool,
266) -> Result<()> {
267    anyhow::ensure!(
268        !reconfig_hold,
269        "governance holds reconfiguration (the reconfig_hold config flag is set); the chain \
270         refuses start_reconfig until an update-config proposal clears it"
271    );
272    if let Some(pending_epoch) = pending_epoch {
273        anyhow::bail!(
274            "a reconfiguration to epoch {pending_epoch} is already in progress; if it has \
275             overrun its Sui epoch, abort it first with `hashi committee abort-reconfig`"
276        );
277    }
278    anyhow::ensure!(
279        !committee_exists_for_sui_epoch,
280        "a committee already exists for Sui's current epoch ({sui_epoch}); the next \
281         reconfiguration can only start once Sui's epoch moves on"
282    );
283    anyhow::ensure!(
284        hashi_epoch == 0 || hashi_epoch != sui_epoch,
285        "Hashi is already on Sui's current epoch ({sui_epoch}); there is nothing to start \
286         until Sui's epoch moves on"
287    );
288    anyhow::ensure!(
289        hashi_epoch <= sui_epoch,
290        "Hashi's epoch ({hashi_epoch}) is ahead of the fullnode's Sui epoch ({sui_epoch}); \
291         the fullnode is behind, retry against one that has caught up"
292    );
293    Ok(())
294}

296/// Start a reconfiguration by hand (`reconfig::start_reconfig`).
297/// Permissionless: any funded signer may send it. Running nodes submit it
298/// themselves whenever Hashi lags Sui with nothing pending, so this is the
299/// fallback for when no node is doing so. Refused, like on chain, while
300/// governance holds reconfiguration with the `reconfig_hold` config flag.
301pub async fn start_reconfig(config: &CliConfig, tx_opts: &TxOptions) -> Result<()> {
302    let mut client = HashiClient::new(config).await?;
303    let sui_epoch = client.fetch_sui_epoch().await?;
304    let hashi_epoch = client.fetch_epoch();
305    let (reconfig_hold, pending_epoch, committee_exists_for_sui_epoch) = {
306        let state = client.onchain_state().state();
307        let hashi = state.hashi();
308        (
309            hashi.config.reconfig_hold(),
310            hashi.committees.pending_epoch_change(),
311            hashi.committees.committees().contains_key(&sui_epoch),
312        )
313    };
314    refuse_unstartable_reconfig(
315        reconfig_hold,
316        pending_epoch,
317        hashi_epoch,
318        sui_epoch,
319        committee_exists_for_sui_epoch,
320    )?;
321
322    print_detail(&format!("\n{}", "Starting a reconfiguration:".bold()));
323    print_detail(&format!("  Current Hashi epoch: {hashi_epoch}"));
324    print_detail(&format!("  Current Sui epoch:   {sui_epoch}"));
325    print_detail(&format!(
326        "  Effect: a committee for epoch {sui_epoch} is formed from Sui's current validator \
327         set and every running node drives DKG or key rotation for it. It must complete \
328         within Sui epoch {sui_epoch}, or it can only be aborted.",
329    ));
330
331    if !prompt_continue("start a reconfiguration", tx_opts).await? {
332        print_warning("Aborted.");
333        return Ok(());
334    }
335
336    let tx = client.build_start_reconfig_transaction()?;
337    print_info("Transaction: reconfig::start_reconfig");
338    execute_or_simulate(&mut client, tx, tx_opts).await?;
339    Ok(())
340}
