17/// Resolve, from a one-shot governance read, the package id CLI-built Hashi
18/// transactions must call: the active version's package. Resolution
19/// failure is a hard
20/// error with no fallback to the original package id, because every
21/// v1-targeted entry aborts once v1 is disabled. Returns the reader state
22/// alongside the id so batch paths can attach it to a
23/// [`crate::sui_tx_executor::SuiTxExecutor`] for self-routing.
24pub(crate) async fn resolve_active_call_package(
25    config: &super::config::CliConfig,
26    hashi_ids: crate::config::HashiIds,
27) -> anyhow::Result<(crate::onchain::OnchainState, sui_sdk_types::Address)> {
28    let state = crate::onchain::OnchainState::new_reader(
29        &config.sui_rpc_url,
30        hashi_ids,
31        None,
32        crate::onchain::ScrapeScope::GovernanceOnly,
33    )
34    .await
35    .context("failed to read on-chain governance state to resolve the active package")?;
36    let package = state.active_package().map(|(id, _version)| id).context(
37        "no supported active on-chain package version resolved; refusing to fall back to the \
38         original package",
39    )?;
40    Ok((state, package))
41}

43/// The package `hashi register` calls: the highest version that is both
44/// governance-enabled and published on-chain, resolved from a one-shot
45/// governance read and deliberately NOT intersected with this binary's
46/// [`crate::constants::SUPPORTED_PACKAGE_VERSIONS`]. That list gates what a
47/// node may decode and mutate autonomously; the registration transaction is a
48/// fixed set of `validator::*` entry calls whose only version dependency is
49/// the called package's `assert_version_enabled`, so the live package is the
50/// correct target even for a CLI built before the chain's latest upgrade. A
51/// chain with no enabled+published version is a hard error, never a fallback
52/// to the original package id (`hashi_ids.package_id`), whose entries abort
53/// once its version is retired.
54pub async fn resolve_latest_enabled_package(
55    sui_rpc_url: &str,
56    hashi_ids: crate::config::HashiIds,
57) -> anyhow::Result<sui_sdk_types::Address> {
58    let state = crate::onchain::OnchainState::new_reader(
59        sui_rpc_url,
60        hashi_ids,
61        None,
62        crate::onchain::ScrapeScope::GovernanceOnly,
63    )
64    .await
65    .context("failed to read on-chain governance state to resolve the enabled package")?;
66    let state = state.state();
67    let published: Vec<u64> = state
68        .package_versions()
69        .versions()
70        .keys()
71        .copied()
72        .collect();
73    let version = state
74        .version_support(&published)
75        .active_version()
76        .context("no on-chain package version is both enabled and published")?;
77    state
78        .package_versions()
79        .get(version)
80        .with_context(|| format!("package id for enabled version {version} is not known"))
81}
