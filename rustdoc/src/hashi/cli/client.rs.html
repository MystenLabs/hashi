32/// Parameters for creating different types of proposals

48        nonce_generation_protocol: Option<u64>,
49        metadata: Vec<(String, String)>,

52        version: u64,
53        metadata: Vec<(String, String)>,
54    },

56        version: u64,
57        metadata: Vec<(String, String)>,
58    },

62    },

64        url: String,

107/// Fetch a shared object's initial shared version from its owner field.
108///
109/// Needed to build fully-resolved shared inputs: pre-resolving a shared input's
110/// initial version + mutability keeps sui >= 1.76 fullnodes from having to
111/// inspect an upgrade-introduced module's signature at simulate time (that
112/// inspection fails with `INVALID_LINKAGE`).
113pub async fn fetch_initial_shared_version(
114    client: &mut sui_rpc::Client,
115    object_id: Address,
116) -> Result<u64> {
117    use sui_rpc::field::FieldMask;
118    use sui_rpc::field::FieldMaskUtil;
119    use sui_rpc::proto::sui::rpc::v2::GetObjectRequest;
120
121    let response = client
122        .ledger_client()
123        .get_object(
124            GetObjectRequest::new(&object_id).with_read_mask(FieldMask::from_paths(["owner"])),
125        )
126        .await
127        .with_context(|| format!("fetching owner of shared object {object_id}"))?
128        .into_inner();
129    Ok(response.object().owner().version())
130}

145    async fn with_scope(config: &CliConfig, scope: ScrapeScope) -> Result<Self> {
146        config.validate()?;
147
148        let hashi_ids = HashiIds {
149            package_id: config.package_id(),
150            hashi_object_id: config.hashi_object_id(),
151        };
152
153        let onchain_state = OnchainState::new_reader(
154            &config.sui_rpc_url,
155            hashi_ids,
156            Some(crate::config::DEFAULT_GRPC_MAX_DECODING_MESSAGE_SIZE),
157            scope,
158        )
159        .await
160        .context("Failed to initialize on-chain state")?;
161
162        // Try to create executor if keypair is available
163        let executor = match config.load_keypair()? {
164            Some(signer) => {
165                tracing::debug!("Keypair loaded, transaction execution enabled");
166                Some(SuiTxExecutor::new(
167                    onchain_state.client(),
168                    signer,
169                    hashi_ids,
170                ))
171            }
172            None => {
173                tracing::debug!("No keypair configured, transaction execution disabled");
174                None
175            }
176        };
177
178        Ok(Self {
179            onchain_state,
180            hashi_ids,
181            executor,
182        })
183    }

206    /// Finalize `builder` according to `tx_opts`: serialize it unsigned,
207    /// dry-run it, or sign and submit it.
208    ///
209    /// A keypair is required only for [`TxMode::Execute`](crate::sui_tx_executor::TxMode);
210    /// the serialize and dry-run paths build with just the sender address
211    /// (explicit `--sender`, or the keypair's address when one is configured).
212    pub async fn finalize_tx(
213        &self,
214        builder: TransactionBuilder,
215        tx_opts: &crate::cli::TxOptions,
216    ) -> Result<crate::sui_tx_executor::TxOutcome> {
217        let signer = self.executor.as_ref().map(|e| e.signer());
218        let mut client = self.onchain_state.client();
219        crate::sui_tx_executor::finalize(
220            &mut client,
221            signer,
222            builder,
223            tx_opts.sender,
224            &tx_opts.gas_overrides(),
225            tx_opts.mode(),
226            std::time::Duration::from_secs(10),
227        )
228        .await
229    }

280    /// Live-fetch the full on-chain `Proposal<T>` (votes + quorum threshold +
281    /// metadata) for a specific proposal via one `list_dynamic_fields` call on
282    /// the proposals bag. Separate from the cached `fetch_proposal` because
283    /// validators don't need these fields in their in-memory state.
284    ///
285    /// The proposal type is derived from the matched child object's
286    /// `object_type` — callers don't pass it, so they can't pass the wrong one.
287    pub async fn fetch_proposal_details(&self, proposal_id: Address) -> Result<ProposalDetails> {
288        use crate::onchain::parse_proposal_type;
289        use crate::onchain::types::ProposalType;
290        use futures::TryStreamExt;
291        use hashi_types::move_types;
292        use sui_rpc::field::FieldMask;
293        use sui_rpc::field::FieldMaskUtil;
294        use sui_rpc::proto::sui::rpc::v2::DynamicField;
295        use sui_rpc::proto::sui::rpc::v2::ListDynamicFieldsRequest;
296
297        // The proposal could live in either the active or executed bag.
298        // Walk active first (the common case), then executed.
299        let (active_id, executed_id) = {
300            let state = self.onchain_state.state();
301            let proposals = &state.hashi().proposals;
302            (proposals.active_id(), proposals.executed_id())
303        };
304        let client = self.onchain_state.client();
305
306        for bag_id in [active_id, executed_id] {
307            let mut stream = Box::pin(
308                client.list_dynamic_fields(
309                    ListDynamicFieldsRequest::default()
310                        .with_parent(bag_id)
311                        .with_page_size(u32::MAX)
312                        .with_read_mask(FieldMask::from_paths([
313                            DynamicField::path_builder().name().finish(),
314                            DynamicField::path_builder().child_object().object_type(),
315                            DynamicField::path_builder()
316                                .child_object()
317                                .contents()
318                                .finish(),
319                        ])),
320                ),
321            );
322
323            while let Some(field) = stream.try_next().await? {
324                // The bag key is BCS-encoded `ID`, which is equivalent to `Address`.
325                let Ok(key) = bcs::from_bytes::<Address>(field.name().value()) else {
326                    continue;
327                };
328                if key != proposal_id {
329                    continue;
330                }
331
332                let object_type_str = field.child_object().object_type();
333                let type_tag: TypeTag = object_type_str
334                    .parse()
335                    .with_context(|| format!("parse object_type {object_type_str:?}"))?;
336                let proposal_type = parse_proposal_type(&type_tag);
337
338                let value_bytes = field.child_object().contents().value();
339                let (creator, votes, quorum_threshold_bps, metadata) = match proposal_type {
340                    ProposalType::UpdateConfig => {
341                        let p: move_types::Proposal<move_types::UpdateConfig> =
342                            bcs::from_bytes(value_bytes).context("deserialize UpdateConfig")?;
343                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
344                    }
345                    ProposalType::EnableVersion => {
346                        let p: move_types::Proposal<move_types::EnableVersion> =
347                            bcs::from_bytes(value_bytes).context("deserialize EnableVersion")?;
348                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
349                    }
350                    ProposalType::DisableVersion => {
351                        let p: move_types::Proposal<move_types::DisableVersion> =
352                            bcs::from_bytes(value_bytes).context("deserialize DisableVersion")?;
353                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
354                    }
355                    ProposalType::Upgrade => {
356                        let p: move_types::Proposal<move_types::Upgrade> =
357                            bcs::from_bytes(value_bytes).context("deserialize Upgrade")?;
358                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
359                    }
360                    ProposalType::EmergencyPause => {
361                        let p: move_types::Proposal<move_types::EmergencyPause> =
362                            bcs::from_bytes(value_bytes).context("deserialize EmergencyPause")?;
363                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
364                    }
365                    ProposalType::AbortReconfig => {
366                        let p: move_types::Proposal<move_types::AbortReconfig> =
367                            bcs::from_bytes(value_bytes).context("deserialize AbortReconfig")?;
368                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
369                    }
370                    ProposalType::UpdateGuardian => {
371                        let p: move_types::Proposal<move_types::UpdateGuardian> =
372                            bcs::from_bytes(value_bytes).context("deserialize UpdateGuardian")?;
373                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
374                    }
375                    ProposalType::Unknown(s) => {
376                        anyhow::bail!("Cannot fetch details for unknown proposal type: {s}")
377                    }
378                };
379                return Ok(ProposalDetails {
380                    creator,
381                    votes,
382                    quorum_threshold_bps,
383                    metadata,
384                });
385            }
386        }
387
388        anyhow::bail!("Proposal {proposal_id} not found in proposals bags")
389    }

433    /// Resolve the committee member (validator address) the signer is acting for:
434    /// the signer's own address if it is a validator member, otherwise the member
435    /// that has delegated its operator address to the signer.
436    fn resolve_validator_address(&self) -> anyhow::Result<Address> {
437        let sender = self
438            .executor
439            .as_ref()
440            .ok_or_else(|| anyhow::anyhow!("Cannot resolve validator: no keypair configured"))?
441            .sender();
442        let members = self.fetch_committee_members();
443        members
444            .iter()
445            .find(|m| *m.validator_address() == sender || *m.operator_address() == sender)
446            .map(|m| *m.validator_address())
447            .ok_or_else(|| {
448                anyhow::anyhow!("signer {sender} is not a committee member or delegated operator")
449            })
450    }

470    /// Build a remove_vote transaction for a proposal.
471    ///
472    /// Calls: `proposal::remove_vote<T>(hashi, validator_address, proposal_id)`
473    pub fn build_remove_vote_transaction(
474        &self,
475        proposal_id: Address,
476        type_arg: TypeTag,
477    ) -> anyhow::Result<TransactionBuilder> {
478        let validator_address = self.resolve_validator_address()?;
479
480        let mut builder = TransactionBuilder::new();
481
482        let hashi_arg = builder.object(
483            ObjectInput::new(self.hashi_ids.hashi_object_id)
484                .as_shared()
485                .with_mutable(true),
486        );
487        let validator_address_arg = builder.pure(&validator_address);
488        let proposal_id_arg = builder.pure(&proposal_id);
489
490        builder.move_call(
491            Function::new(
492                self.call_package(),
493                Identifier::from_static("proposal"),
494                Identifier::from_static("remove_vote"),
495            )
496            .with_type_args(vec![type_arg]),
497            vec![hashi_arg, validator_address_arg, proposal_id_arg],
498        );
499
500        Ok(builder)
501    }

517    /// The package governance calls execute on. `disable_version` checks its
518    /// guard against the *executing* package's `PACKAGE_VERSION`, so running it
519    /// through the original id after an upgrade lets it disable the live one.
520    fn call_package(&self) -> Address {
521        self.onchain_state
522            .active_package()
523            .map_or(self.hashi_ids.package_id, |(id, _)| id)
524    }

526    /// Build a transaction to execute a proposal that has reached quorum.
527    ///
528    /// Each proposal type has its own `module::execute(hashi, proposal_id, clock)`
529    /// entry point. This method dispatches to the correct one based on the
530    /// on-chain proposal type.
531    pub fn build_execute_proposal_transaction(
532        &self,
533        proposal_id: Address,
534        proposal_type: &crate::onchain::types::ProposalType,
535    ) -> anyhow::Result<TransactionBuilder> {
536        use crate::onchain::types::ProposalType;
537
538        let module_name = match proposal_type {
539            ProposalType::UpdateConfig => "update_config",
540            ProposalType::EnableVersion => "enable_version",
541            ProposalType::DisableVersion => "disable_version",
542            ProposalType::EmergencyPause => "emergency_pause",
543            ProposalType::AbortReconfig => "abort_reconfig",
544            ProposalType::UpdateGuardian => "update_guardian",
545            ProposalType::Upgrade => {
546                anyhow::bail!(
547                    "Upgrade proposals require the full upgrade flow (execute + publish + finalize)"
548                );
549            }
550            ProposalType::Unknown(s) => {
551                anyhow::bail!("Cannot execute unknown proposal type: {s}");
552            }
553        };
554
555        let mut builder = TransactionBuilder::new();
556        let hashi_arg = builder.object(
557            ObjectInput::new(self.hashi_ids.hashi_object_id)
558                .as_shared()
559                .with_mutable(true),
560        );
561        let proposal_id_arg = builder.pure(&proposal_id);
562        let clock_arg = builder.object(
563            ObjectInput::new(SUI_CLOCK_OBJECT_ID)
564                .as_shared()
565                .with_mutable(false),
566        );
567
568        builder.move_call(
569            Function::new(
570                self.call_package(),
571                Identifier::new(module_name)?,
572                Identifier::from_static("execute"),
573            ),
574            vec![hashi_arg, proposal_id_arg, clock_arg],
575        );
576
577        Ok(builder)
578    }
579}

581/// Build a `TransactionBuilder` for creating a proposal, given `HashiIds` and params.
582///
583/// This is a standalone function so it can be reused outside `HashiClient` (e.g. in tests).
584pub fn build_create_proposal_transaction(
585    hashi_ids: HashiIds,
586    call_package: Address,
587    validator_address: Address,
588    params: CreateProposalParams,
589) -> TransactionBuilder {
590    let mut builder = TransactionBuilder::new();
591
592    let hashi_arg = builder.object(
593        ObjectInput::new(hashi_ids.hashi_object_id)
594            .as_shared()
595            .with_mutable(true),
596    );
597    let clock_arg = builder.object(
598        ObjectInput::new(SUI_CLOCK_OBJECT_ID)
599            .as_shared()
600            .with_mutable(false),
601    );
602    let validator_address_arg = builder.pure(&validator_address);
603
604    match params {
605        CreateProposalParams::Upgrade { digest, metadata } => {
606            let digest_arg = builder.pure(&digest);
607            let metadata_arg = build_metadata(&mut builder, &metadata);
608            builder.move_call(
609                Function::new(
610                    call_package,
611                    Identifier::from_static("upgrade"),
612                    Identifier::from_static("propose"),
613                ),
614                vec![
615                    hashi_arg,
616                    validator_address_arg,
617                    digest_arg,
618                    metadata_arg,
619                    clock_arg,
620                ],
621            );
622        }
623        CreateProposalParams::UpdateConfig {
624            key,
625            value,
626            metadata,
627        } => {
628            let entries_arg = build_config_entries(
629                &mut builder,
630                hashi_ids.package_id,
631                call_package,
632                &[(key, value)],
633            );
634            let metadata_arg = build_metadata(&mut builder, &metadata);
635            builder.move_call(
636                Function::new(
637                    call_package,
638                    Identifier::from_static("update_config"),
639                    Identifier::from_static("propose"),
640                ),
641                vec![
642                    hashi_arg,
643                    validator_address_arg,
644                    entries_arg,
645                    metadata_arg,
646                    clock_arg,
647                ],
648            );
649        }
650        CreateProposalParams::UpdateMpcConfig {
651            threshold_bps,
652            max_faulty_bps,
653            weight_reduction_allowed_delta,
654            nonce_generation_protocol,
655            metadata,
656        } => {
657            let entries: Vec<(String, ConfigValue)> = [
658                ("mpc_threshold_in_basis_points", threshold_bps),
659                ("mpc_max_faulty_in_basis_points", max_faulty_bps),
660                (
661                    "mpc_weight_reduction_allowed_delta",
662                    weight_reduction_allowed_delta,
663                ),
664                ("mpc_nonce_generation_protocol", nonce_generation_protocol),
665            ]
666            .into_iter()
667            .filter_map(|(k, v)| v.map(|v| (k.to_string(), ConfigValue::U64(v))))
668            .collect();
669            let entries_arg =
670                build_config_entries(&mut builder, hashi_ids.package_id, call_package, &entries);
671            let metadata_arg = build_metadata(&mut builder, &metadata);
672            builder.move_call(
673                Function::new(
674                    call_package,
675                    Identifier::from_static("update_config"),
676                    Identifier::from_static("propose"),
677                ),
678                vec![
679                    hashi_arg,
680                    validator_address_arg,
681                    entries_arg,
682                    metadata_arg,
683                    clock_arg,
684                ],
685            );
686        }
687        CreateProposalParams::EnableVersion { version, metadata } => {
688            let version_arg = builder.pure(&version);
689            let metadata_arg = build_metadata(&mut builder, &metadata);
690            builder.move_call(
691                Function::new(
692                    call_package,
693                    Identifier::from_static("enable_version"),
694                    Identifier::from_static("propose"),
695                ),
696                vec![
697                    hashi_arg,
698                    validator_address_arg,
699                    version_arg,
700                    metadata_arg,
701                    clock_arg,
702                ],
703            );
704        }
705        CreateProposalParams::DisableVersion { version, metadata } => {
706            let version_arg = builder.pure(&version);
707            let metadata_arg = build_metadata(&mut builder, &metadata);
708            builder.move_call(
709                Function::new(
710                    call_package,
711                    Identifier::from_static("disable_version"),
712                    Identifier::from_static("propose"),
713                ),
714                vec![
715                    hashi_arg,
716                    validator_address_arg,
717                    version_arg,
718                    metadata_arg,
719                    clock_arg,
720                ],
721            );
722        }
723        CreateProposalParams::AbortReconfig { epoch, metadata } => {
724            let epoch_arg = builder.pure(&epoch);
725            let metadata_arg = build_metadata(&mut builder, &metadata);
726            builder.move_call(
727                Function::new(
728                    call_package,
729                    Identifier::from_static("abort_reconfig"),
730                    Identifier::from_static("propose"),
731                ),
732                vec![
733                    hashi_arg,
734                    validator_address_arg,
735                    epoch_arg,
736                    metadata_arg,
737                    clock_arg,
738                ],
739            );
740        }
741        CreateProposalParams::UpdateGuardian { url, metadata } => {
742            let url_arg = builder.pure(&url);
743            let metadata_arg = build_metadata(&mut builder, &metadata);
744            builder.move_call(
745                Function::new(
746                    call_package,
747                    Identifier::from_static("update_guardian"),
748                    Identifier::from_static("propose"),
749                ),
750                vec![
751                    hashi_arg,
752                    validator_address_arg,
753                    url_arg,
754                    metadata_arg,
755                    clock_arg,
756                ],
757            );
758        }
759        CreateProposalParams::EmergencyPause { pause, metadata } => {
760            let pause_arg = builder.pure(&pause);
761            let metadata_arg = build_metadata(&mut builder, &metadata);
762            builder.move_call(
763                Function::new(
764                    call_package,
765                    Identifier::from_static("emergency_pause"),
766                    Identifier::from_static("propose"),
767                ),
768                vec![
769                    hashi_arg,
770                    validator_address_arg,
771                    pause_arg,
772                    metadata_arg,
773                    clock_arg,
774                ],
775            );
776        }
777    }
778
779    builder
780}

782/// Build a `config_value::Value` enum via a move call (e.g. `config_value::new_u64(v)`).
783/// Returns the `Argument` holding the constructed `Value`.
784fn build_config_value(
785    builder: &mut TransactionBuilder,
786    call_package: Address,
787    value: &hashi_types::move_types::ConfigValue,
788) -> sui_transaction_builder::Argument {
789    use hashi_types::move_types::ConfigValue;
790
791    let (func_name, arg) = match value {
792        ConfigValue::U64(v) => ("new_u64", builder.pure(v)),
793        ConfigValue::Address(v) => ("new_address", builder.pure(v)),
794        ConfigValue::String(v) => ("new_string", builder.pure(v)),
795        ConfigValue::Bool(v) => ("new_bool", builder.pure(v)),
796        ConfigValue::Bytes(v) => ("new_bytes", builder.pure(v)),
797        ConfigValue::U128(v) => ("new_u128", builder.pure(v)),
798        // The 32 LE bytes BCS-encode identically to a Move `u256` pure arg.
799        ConfigValue::U256(v) => ("new_u256", builder.pure(v)),
800    };
801
802    builder.move_call(
803        Function::new(
804            call_package,
805            Identifier::from_static("config_value"),
806            Identifier::new(func_name).unwrap(),
807        ),
808        vec![arg],
809    )
810}
811
812fn build_config_entries(
813    builder: &mut TransactionBuilder,
814    type_package: Address,
815    call_package: Address,
816    entries: &[(String, ConfigValue)],
817) -> sui_transaction_builder::Argument {
818    let sui_framework = Address::from_static("0x2");
819    let move_stdlib = Address::from_static("0x1");
820    let string_type = TypeTag::Struct(Box::new(StructTag::new(
821        move_stdlib,
822        Identifier::from_static("string"),
823        Identifier::from_static("String"),
824        vec![],
825    )));
826    let value_type = TypeTag::Struct(Box::new(StructTag::new(
827        type_package,
828        Identifier::from_static("config_value"),
829        Identifier::from_static("Value"),
830        vec![],
831    )));
832    let map = builder.move_call(
833        Function::new(
834            sui_framework,
835            Identifier::from_static("vec_map"),
836            Identifier::from_static("empty"),
837        )
838        .with_type_args(vec![string_type.clone(), value_type.clone()]),
839        vec![],
840    );
841    for (key, value) in entries {
842        let key_arg = builder.pure(key);
843        let value_arg = build_config_value(builder, call_package, value);
844        builder.move_call(
845            Function::new(
846                sui_framework,
847                Identifier::from_static("vec_map"),
848                Identifier::from_static("insert"),
849            )
850            .with_type_args(vec![string_type.clone(), value_type.clone()]),
851            vec![map, key_arg, value_arg],
852        );
853    }
854    map
855}

857/// Build a `VecMap<String, String>` for proposal metadata via move calls.
858///
859/// Move structs like `VecMap` cannot be passed as pure args in PTBs.
860/// Instead we construct one via `vec_map::empty()` + `vec_map::insert()`.
861fn build_metadata(
862    builder: &mut TransactionBuilder,
863    metadata: &[(String, String)],
864) -> sui_transaction_builder::Argument {
865    let sui_framework = Address::from_static("0x2");
866    let move_stdlib = Address::from_static("0x1");
867
868    let string_type = TypeTag::Struct(Box::new(StructTag::new(
869        move_stdlib,
870        Identifier::from_static("string"),
871        Identifier::from_static("String"),
872        vec![],
873    )));
874
875    // vec_map::empty<String, String>()
876    let map = builder.move_call(
877        Function::new(
878            sui_framework,
879            Identifier::from_static("vec_map"),
880            Identifier::from_static("empty"),
881        )
882        .with_type_args(vec![string_type.clone(), string_type.clone()]),
883        vec![],
884    );
885
886    // vec_map::insert(&mut map, key, value) for each entry
887    for (key, value) in metadata {
888        let key_arg = builder.pure(key);
889        let value_arg = builder.pure(value);
890        builder.move_call(
891            Function::new(
892                sui_framework,
893                Identifier::from_static("vec_map"),
894                Identifier::from_static("insert"),
895            )
896            .with_type_args(vec![string_type.clone(), string_type.clone()]),
897            vec![map, key_arg, value_arg],
898        );
899    }
900
901    map
902}

904/// Build a `proposal::vote<T>` transaction as a standalone. Reusable outside
905/// `HashiClient` — e2e test infra needs to build vote PTBs for every
906/// committee member.
907pub fn build_vote_transaction(
908    hashi_ids: HashiIds,
909    call_package: Address,
910    validator_address: Address,
911    proposal_id: Address,
912    type_arg: TypeTag,
913) -> TransactionBuilder {
914    let mut builder = TransactionBuilder::new();
915    let hashi_arg = builder.object(
916        ObjectInput::new(hashi_ids.hashi_object_id)
917            .as_shared()
918            .with_mutable(true),
919    );
920    let validator_address_arg = builder.pure(&validator_address);
921    let proposal_id_arg = builder.pure(&proposal_id);
922    let clock_arg = builder.object(
923        ObjectInput::new(SUI_CLOCK_OBJECT_ID)
924            .as_shared()
925            .with_mutable(false),
926    );
927
928    builder.move_call(
929        Function::new(
930            call_package,
931            Identifier::from_static("proposal"),
932            Identifier::from_static("vote"),
933        )
934        .with_type_args(vec![type_arg]),
935        vec![hashi_arg, validator_address_arg, proposal_id_arg, clock_arg],
936    );
937
938    builder
939}

941/// Get the TypeTag for a proposal type (from on-chain type)
942///
943/// Returns an error if the proposal type is `Unknown`.
944pub fn get_proposal_type_arg(
945    package_id: Address,
946    proposal_type: &crate::onchain::types::ProposalType,
947) -> Result<TypeTag> {
948    use crate::onchain::types::ProposalType;
949
950    let (module, name) = match proposal_type {
951        ProposalType::Upgrade => ("upgrade", "Upgrade"),
952        ProposalType::UpdateConfig => ("update_config", "UpdateConfig"),
953        ProposalType::EnableVersion => ("enable_version", "EnableVersion"),
954        ProposalType::DisableVersion => ("disable_version", "DisableVersion"),
955        ProposalType::EmergencyPause => ("emergency_pause", "EmergencyPause"),
956        ProposalType::AbortReconfig => ("abort_reconfig", "AbortReconfig"),
957        ProposalType::UpdateGuardian => ("update_guardian", "UpdateGuardian"),
958        ProposalType::Unknown(s) => {
959            anyhow::bail!(
960                "Cannot vote on unknown proposal type '{}'. \
961                 This may be a new proposal type not yet supported by this CLI version.",
962                s
963            );
964        }
965    };
966
967    Ok(TypeTag::Struct(Box::new(StructTag::new(
968        package_id,
969        Identifier::new(module).context("Invalid module name")?,
970        Identifier::new(name).context("Invalid type name")?,
971        vec![],
972    ))))
