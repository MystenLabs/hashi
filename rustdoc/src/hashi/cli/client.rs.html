113/// Client for reading Hashi on-chain state and building/executing transactions.
114///
115/// Uses `OnchainState` for reading on-chain data (committees, proposals, etc.)
116/// and `SuiTxExecutor` for transaction execution when a keypair is configured.
117pub struct HashiClient {
118    /// On-chain state reader from hashi crate
119    onchain_state: OnchainState,
120    /// Hashi package and object IDs
121    hashi_ids: HashiIds,
122    /// The Hashi shared object's initial shared version, fetched once at
123    /// construction. Immutable for the object's lifetime, and needed to
124    /// build fully-resolved shared inputs (see
125    /// [`build_create_proposal_transaction`]).
126    hashi_initial_shared_version: u64,
127    /// `--sender`, when given: the identity governance commands act as when
128    /// there is no keypair to derive one from.
129    acting_sender: Option<Address>,
130    /// Optional executor for signing and submitting transactions
131    executor: Option<SuiTxExecutor>,
132    /// The RPC endpoint this client talks to (used for explorer deep-links)
133    sui_rpc_url: String,
134}

136/// Fetch a shared object's initial shared version from its owner field.
137///
138/// Needed to build fully-resolved shared inputs: pre-resolving a shared input's
139/// initial version + mutability keeps sui >= 1.76 fullnodes from having to
140/// inspect an upgrade-introduced module's signature at simulate time (that
141/// inspection fails with `INVALID_LINKAGE`).
142pub async fn fetch_initial_shared_version(
143    client: &mut sui_rpc::Client,
144    object_id: Address,
145) -> Result<u64> {
146    use sui_rpc::field::FieldMask;
147    use sui_rpc::field::FieldMaskUtil;
148    use sui_rpc::proto::sui::rpc::v2::GetObjectRequest;
149
150    let response = client
151        .ledger_client()
152        .get_object(
153            GetObjectRequest::new(&object_id).with_read_mask(FieldMask::from_paths(["owner"])),
154        )
155        .await
156        .with_context(|| format!("fetching owner of shared object {object_id}"))?
157        .into_inner();
158    Ok(response.object().owner().version())
159}

161/// Which on-chain proposal bag holds a proposal. Executed proposals are not
162/// deleted: `proposal::execute` moves them from the active bag to the
163/// executed bag, where they stay for inspection, so a lookup that only asks
164/// "does it exist" cannot tell an open proposal from a finished one.
165#[derive(Clone, Debug)]
166pub enum ProposalLocation {
167    Active(Proposal),
168    Executed(Proposal),
169    /// In neither bag: the id is wrong, or the proposal expired and was
170    /// deleted.
171    Missing,
172}
173
174/// Locate `proposal_id` in the scraped active and executed bags.
175pub fn locate_proposal(
176    active: &[Proposal],
177    executed: &[Proposal],
178    proposal_id: &Address,
179) -> ProposalLocation {
180    if let Some(proposal) = active.iter().find(|p| p.id == *proposal_id) {
181        return ProposalLocation::Active(proposal.clone());
182    }
183    if let Some(proposal) = executed.iter().find(|p| p.id == *proposal_id) {
184        return ProposalLocation::Executed(proposal.clone());
185    }
186    ProposalLocation::Missing
187}

189/// A gRPC `NotFound` while loading the Hashi object almost always means the
190/// configured ids and the RPC URL come from different networks; say so instead
191/// of surfacing the bare status.
192pub fn explain_missing_object(
193    err: anyhow::Error,
194    hashi_ids: HashiIds,
195    sui_rpc_url: &str,
196) -> anyhow::Error {
197    let not_found = err.chain().any(|e| {
198        matches!(e.downcast_ref::<tonic::Status>(), Some(s) if s.code() == tonic::Code::NotFound)
199    });
200    if not_found {
201        err.context(format!(
202            "Hashi object {} (package {}) was not found on {sui_rpc_url}: check that package-id, \
203             hashi-object-id and the RPC URL all belong to the same network",
204            hashi_ids.hashi_object_id, hashi_ids.package_id
205        ))
206    } else {
207        err
208    }
209}

224    async fn with_scope(config: &CliConfig, scope: ScrapeScope) -> Result<Self> {
225        config.validate()?;
226
227        let hashi_ids = HashiIds {
228            package_id: config.package_id(),
229            hashi_object_id: config.hashi_object_id(),
230        };
231
232        // Say which network every command is talking to: the RPC URL defaults
233        // to mainnet when no config is found, and a wrong-network object id is
234        // otherwise indistinguishable from a typo.
235        crate::cli::print_info(&format!("Sui RPC: {}", config.sui_rpc_url));
236
237        let onchain_state = OnchainState::new_reader(
238            &config.sui_rpc_url,
239            hashi_ids,
240            Some(crate::config::DEFAULT_GRPC_MAX_DECODING_MESSAGE_SIZE),
241            scope,
242        )
243        .await
244        .map_err(|e| explain_missing_object(e, hashi_ids, &config.sui_rpc_url))
245        .context("Failed to initialize on-chain state")?;
246
247        let hashi_initial_shared_version =
248            fetch_initial_shared_version(&mut onchain_state.client(), hashi_ids.hashi_object_id)
249                .await
250                .map_err(|e| explain_missing_object(e, hashi_ids, &config.sui_rpc_url))?;
251
252        // Try to create executor if keypair is available
253        let executor = match config.load_keypair()? {
254            Some(signer) => {
255                tracing::debug!("Keypair loaded, transaction execution enabled");
256                Some(SuiTxExecutor::new(
257                    onchain_state.client(),
258                    signer,
259                    hashi_ids,
260                ))
261            }
262            None => {
263                tracing::debug!("No keypair configured, transaction execution disabled");
264                None
265            }
266        };
267
268        Ok(Self {
269            onchain_state,
270            hashi_ids,
271            hashi_initial_shared_version,
272            acting_sender: config.acting_sender,
273            executor,
274            sui_rpc_url: config.sui_rpc_url.clone(),
275        })
276    }

305    /// dry-run it, or sign and submit it.
306    ///
307    /// A keypair is required only for [`TxMode::Execute`](crate::sui_tx_executor::TxMode);

311        &self,
312        builder: TransactionBuilder,
313        tx_opts: &crate::cli::TxOptions,
314    ) -> Result<crate::sui_tx_executor::TxOutcome> {
315        let signer = self.executor.as_ref().map(|e| e.signer());

317        crate::sui_tx_executor::finalize(
318            &mut client,
319            signer,
320            builder,
321            tx_opts.sender,
322            &tx_opts.gas_overrides(),
323            tx_opts.mode(),
324            std::time::Duration::from_secs(10),
325        )
326        .await
327        .map_err(crate::cli::explain_tx_error)
328    }

394    /// Live-fetch the full on-chain `Proposal<T>` (votes + quorum threshold +
395    /// metadata) for a specific proposal via one `list_dynamic_fields` call on
396    /// the proposals bag. Separate from the cached `fetch_proposal` because
397    /// validators don't need these fields in their in-memory state.
398    ///
399    /// The proposal type is derived from the matched child object's
400    /// `object_type` — callers don't pass it, so they can't pass the wrong one.
401    pub async fn fetch_proposal_details(&self, proposal_id: Address) -> Result<ProposalDetails> {
402        use crate::onchain::parse_proposal_type;
403        use crate::onchain::types::ProposalType;
404        use futures::TryStreamExt;
405        use hashi_types::move_types;
406        use sui_rpc::field::FieldMask;
407        use sui_rpc::field::FieldMaskUtil;
408        use sui_rpc::proto::sui::rpc::v2::DynamicField;
409        use sui_rpc::proto::sui::rpc::v2::ListDynamicFieldsRequest;
410
411        // The proposal could live in either the active or executed bag.
412        // Walk active first (the common case), then executed.
413        let (active_id, executed_id) = {
414            let state = self.onchain_state.state();
415            let proposals = &state.hashi().proposals;
416            (proposals.active_id(), proposals.executed_id())
417        };
418        let client = self.onchain_state.client();
419
420        for bag_id in [active_id, executed_id] {
421            let mut stream = Box::pin(
422                client.list_dynamic_fields(
423                    ListDynamicFieldsRequest::default()
424                        .with_parent(bag_id)
425                        .with_page_size(u32::MAX)
426                        .with_read_mask(FieldMask::from_paths([
427                            DynamicField::path_builder().name().finish(),
428                            DynamicField::path_builder().child_object().object_type(),
429                            DynamicField::path_builder()
430                                .child_object()
431                                .contents()
432                                .finish(),
433                        ])),
434                ),
435            );
436
437            while let Some(field) = stream.try_next().await? {
438                // The bag key is BCS-encoded `ID`, which is equivalent to `Address`.
439                let Ok(key) = bcs::from_bytes::<Address>(field.name().value()) else {
440                    continue;
441                };
442                if key != proposal_id {
443                    continue;
444                }
445
446                let object_type_str = field.child_object().object_type();
447                let type_tag: TypeTag = object_type_str
448                    .parse()
449                    .with_context(|| format!("parse object_type {object_type_str:?}"))?;
450                let proposal_type = parse_proposal_type(&type_tag);
451
452                let value_bytes = field.child_object().contents().value();
453                let (creator, votes, quorum_threshold_bps, metadata) = match proposal_type {
454                    ProposalType::UpdateConfig => {
455                        let p: move_types::Proposal<move_types::UpdateConfig> =
456                            bcs::from_bytes(value_bytes).context("deserialize UpdateConfig")?;
457                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
458                    }
459                    ProposalType::UpdateEpochConfig => {
460                        let p: move_types::Proposal<move_types::UpdateEpochConfig> =
461                            bcs::from_bytes(value_bytes)
462                                .context("deserialize UpdateEpochConfig")?;
463                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
464                    }
465                    ProposalType::AddConfig => {
466                        let p: move_types::Proposal<move_types::AddConfig> =
467                            bcs::from_bytes(value_bytes).context("deserialize AddConfig")?;
468                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
469                    }
470                    ProposalType::EnableVersion => {
471                        let p: move_types::Proposal<move_types::EnableVersion> =
472                            bcs::from_bytes(value_bytes).context("deserialize EnableVersion")?;
473                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
474                    }
475                    ProposalType::DisableVersion => {
476                        let p: move_types::Proposal<move_types::DisableVersion> =
477                            bcs::from_bytes(value_bytes).context("deserialize DisableVersion")?;
478                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
479                    }
480                    ProposalType::Upgrade => {
481                        let p: move_types::Proposal<move_types::Upgrade> =
482                            bcs::from_bytes(value_bytes).context("deserialize Upgrade")?;
483                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
484                    }
485                    ProposalType::EmergencyPause => {
486                        let p: move_types::Proposal<move_types::EmergencyPause> =
487                            bcs::from_bytes(value_bytes).context("deserialize EmergencyPause")?;
488                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
489                    }
490                    ProposalType::AbortReconfig => {
491                        let p: move_types::Proposal<move_types::AbortReconfig> =
492                            bcs::from_bytes(value_bytes).context("deserialize AbortReconfig")?;
493                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
494                    }
495                    ProposalType::UpdateGuardian => {
496                        let p: move_types::Proposal<move_types::UpdateGuardian> =
497                            bcs::from_bytes(value_bytes).context("deserialize UpdateGuardian")?;
498                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
499                    }
500                    ProposalType::IgnoreMember => {
501                        let p: move_types::Proposal<move_types::IgnoreMember> =
502                            bcs::from_bytes(value_bytes).context("deserialize IgnoreMember")?;
503                        (p.creator, p.votes, p.quorum_threshold_bps, p.metadata)
504                    }
505                    ProposalType::Unknown(s) => {
506                        anyhow::bail!("Cannot fetch details for unknown proposal type: {s}")
507                    }
508                };
509                return Ok(ProposalDetails {
510                    creator,
511                    votes,
512                    quorum_threshold_bps,
513                    metadata,
514                });
515            }
516        }
517
518        anyhow::bail!("Proposal {proposal_id} not found in proposals bags")
519    }

578    /// Resolve the committee member (validator address) the acting address
579    /// acts for in governance calls: an exact validator match wins, otherwise
580    /// exactly one operator delegation; more than one is an error (see
581    /// `resolve_governance_identity`).
582    pub fn resolve_validator_address(&self) -> anyhow::Result<Address> {
583        let sender = self.acting_address().ok_or_else(|| {
584            anyhow::anyhow!("Cannot resolve validator: no keypair configured and no --sender given")
585        })?;
586        resolve_governance_identity(&self.fetch_committee_members(), sender)
587    }

708    /// Build a transaction to execute a proposal that has reached quorum.
709    ///
710    /// Each proposal type has its own `module::execute(hashi, proposal_id, clock)`
711    /// entry point. This method dispatches to the correct one based on the
712    /// on-chain proposal type.
713    pub fn build_execute_proposal_transaction(
714        &self,
715        proposal_id: Address,
716        proposal_type: &crate::onchain::types::ProposalType,
717    ) -> anyhow::Result<TransactionBuilder> {
718        use crate::onchain::types::ProposalType;
719
720        let module_name = match proposal_type {
721            ProposalType::UpdateConfig => "update_config",
722            ProposalType::UpdateEpochConfig => "update_epoch_config",
723            ProposalType::AddConfig => "add_config",
724            ProposalType::EnableVersion => "enable_version",
725            ProposalType::DisableVersion => "disable_version",
726            ProposalType::EmergencyPause => "emergency_pause",
727            ProposalType::AbortReconfig => "abort_reconfig",
728            ProposalType::UpdateGuardian => "update_guardian",
729            ProposalType::IgnoreMember => "ignore_member",
730            ProposalType::Upgrade => {
731                anyhow::bail!(
732                    "Upgrade proposals require the full upgrade flow (execute + publish + finalize)"
733                );
734            }
735            ProposalType::Unknown(s) => {
736                anyhow::bail!("Cannot execute unknown proposal type: {s}");
737            }
738        };
739
740        let mut builder = TransactionBuilder::new();
741        // Fully-resolved shared inputs: see build_create_proposal_transaction.
742        let hashi_arg = builder.object(
743            ObjectInput::new(self.hashi_ids.hashi_object_id)
744                .with_version(self.hashi_initial_shared_version)
745                .as_shared()
746                .with_mutable(true),
747        );
748        let proposal_id_arg = builder.pure(&proposal_id);
749        let clock_arg = builder.object(
750            ObjectInput::new(SUI_CLOCK_OBJECT_ID)
751                .with_version(1)
752                .as_shared()
753                .with_mutable(false),
754        );
755
756        builder.move_call(
757            Function::new(
758                self.call_package(),
759                Identifier::new(module_name)?,
760                Identifier::from_static("execute"),
761            ),
762            vec![hashi_arg, proposal_id_arg, clock_arg],
763        );
764
765        Ok(builder)
766    }

792/// Build a `TransactionBuilder` for creating a proposal, given `HashiIds` and params.
793///
794/// This is a standalone function so it can be reused outside `HashiClient` (e.g. in tests).
795///
796/// `call_package` is the active package id. Proposal modules introduced by an
797/// upgrade (such as `ignore_member`) only exist at an upgraded address, and
798/// all governance calls use one package generation consistently.
799pub fn build_create_proposal_transaction(
800    hashi_ids: HashiIds,
801    hashi_initial_shared_version: u64,
802    call_package: Address,
803    validator_address: Address,
804    params: CreateProposalParams,
805) -> TransactionBuilder {
806    let mut builder = TransactionBuilder::new();
807
808    // Shared inputs are fully resolved (initial shared version + mutability)
809    // so the fullnode's simulate-time resolver never has to inspect the
810    // called function's signature: that inspection fails with
811    // INVALID_LINKAGE on sui >= 1.76 fullnodes for modules introduced by a
812    // package upgrade (e.g. ignore_member).
813    let hashi_arg = builder.object(
814        ObjectInput::new(hashi_ids.hashi_object_id)
815            .with_version(hashi_initial_shared_version)
816            .as_shared()
817            .with_mutable(true),
818    );
819    let clock_arg = builder.object(
820        ObjectInput::new(SUI_CLOCK_OBJECT_ID)
821            .with_version(1)
822            .as_shared()
823            .with_mutable(false),
824    );
825    let validator_address_arg = builder.pure(&validator_address);
826
827    match params {
828        CreateProposalParams::Upgrade {
829            digest,
830            exclusive,
831            metadata,
832        } => {
833            let digest_arg = builder.pure(&digest);
834            let exclusive_arg = builder.pure(&exclusive);
835            let metadata_arg = build_metadata(&mut builder, &metadata);
836            builder.move_call(
837                Function::new(
838                    call_package,
839                    Identifier::from_static("upgrade"),
840                    Identifier::from_static("propose"),
841                ),
842                vec![
843                    hashi_arg,
844                    validator_address_arg,
845                    digest_arg,
846                    exclusive_arg,
847                    metadata_arg,
848                    clock_arg,
849                ],
850            );
851        }
852        CreateProposalParams::UpdateConfig {
853            key,
854            value,
855            metadata,
856        } => {
857            let entries_arg = build_config_entries(
858                &mut builder,
859                hashi_ids.package_id,
860                call_package,
861                &[(key, value)],
862            );
863            let metadata_arg = build_metadata(&mut builder, &metadata);
864            builder.move_call(
865                Function::new(
866                    call_package,
867                    Identifier::from_static("update_config"),
868                    Identifier::from_static("propose"),
869                ),
870                vec![
871                    hashi_arg,
872                    validator_address_arg,
873                    entries_arg,
874                    metadata_arg,
875                    clock_arg,
876                ],
877            );
878        }
879        CreateProposalParams::UpdateEpochConfig {
880            key,
881            value,
882            metadata,
883        } => {
884            let entries_arg = build_config_entries(
885                &mut builder,
886                hashi_ids.package_id,
887                call_package,
888                &[(key, value)],
889            );
890            let metadata_arg = build_metadata(&mut builder, &metadata);
891            builder.move_call(
892                Function::new(
893                    call_package,
894                    Identifier::from_static("update_epoch_config"),
895                    Identifier::from_static("propose"),
896                ),
897                vec![
898                    hashi_arg,
899                    validator_address_arg,
900                    entries_arg,
901                    metadata_arg,
902                    clock_arg,
903                ],
904            );
905        }
906        CreateProposalParams::AddConfig {
907            epoch,
908            key,
909            value,
910            metadata,
911        } => {
912            let epoch_arg = builder.pure(&epoch);
913            let entries_arg = build_config_entries(
914                &mut builder,
915                hashi_ids.package_id,
916                call_package,
917                &[(key, value)],
918            );
919            let metadata_arg = build_metadata(&mut builder, &metadata);
920            builder.move_call(
921                Function::new(
922                    call_package,
923                    Identifier::from_static("add_config"),
924                    Identifier::from_static("propose"),
925                ),
926                vec![
927                    hashi_arg,
928                    validator_address_arg,
929                    epoch_arg,
930                    entries_arg,
931                    metadata_arg,
932                    clock_arg,
933                ],
934            );
935        }
936        CreateProposalParams::UpdateMpcConfig {
937            max_faulty_bps,
938            weight_reduction_allowed_delta,
939            nonce_generation_protocol,
940            metadata,
941        } => {
942            let entries: Vec<(String, ConfigValue)> = [
943                ("mpc_max_faulty_in_basis_points", max_faulty_bps),
944                (
945                    "mpc_weight_reduction_allowed_delta",
946                    weight_reduction_allowed_delta,
947                ),
948                ("mpc_nonce_generation_protocol", nonce_generation_protocol),
949            ]
950            .into_iter()
951            .filter_map(|(k, v)| v.map(|v| (k.to_string(), ConfigValue::U64(v))))
952            .collect();
953            let entries_arg =
954                build_config_entries(&mut builder, hashi_ids.package_id, call_package, &entries);
955            let metadata_arg = build_metadata(&mut builder, &metadata);
956            // The MPC parameters live in the epoch config.
957            builder.move_call(
958                Function::new(
959                    call_package,
960                    Identifier::from_static("update_epoch_config"),
961                    Identifier::from_static("propose"),
962                ),
963                vec![
964                    hashi_arg,
965                    validator_address_arg,
966                    entries_arg,
967                    metadata_arg,
968                    clock_arg,
969                ],
970            );
971        }
972        CreateProposalParams::EnableVersion { version, metadata } => {
973            let version_arg = builder.pure(&version);
974            let metadata_arg = build_metadata(&mut builder, &metadata);
975            builder.move_call(
976                Function::new(
977                    call_package,
978                    Identifier::from_static("enable_version"),
979                    Identifier::from_static("propose"),
980                ),
981                vec![
982                    hashi_arg,
983                    validator_address_arg,
984                    version_arg,
985                    metadata_arg,
986                    clock_arg,
987                ],
988            );
989        }
990        CreateProposalParams::DisableVersion { version, metadata } => {
991            let version_arg = builder.pure(&version);
992            let metadata_arg = build_metadata(&mut builder, &metadata);
993            builder.move_call(
994                Function::new(
995                    call_package,
996                    Identifier::from_static("disable_version"),
997                    Identifier::from_static("propose"),
998                ),
999                vec![
1000                    hashi_arg,
1001                    validator_address_arg,
1002                    version_arg,
1003                    metadata_arg,
1004                    clock_arg,
1005                ],
1006            );
1007        }
1008        CreateProposalParams::AbortReconfig { epoch, metadata } => {
1009            let epoch_arg = builder.pure(&epoch);
1010            let metadata_arg = build_metadata(&mut builder, &metadata);
1011            builder.move_call(
1012                Function::new(
1013                    call_package,
1014                    Identifier::from_static("abort_reconfig"),
1015                    Identifier::from_static("propose"),
1016                ),
1017                vec![
1018                    hashi_arg,
1019                    validator_address_arg,
1020                    epoch_arg,
1021                    metadata_arg,
1022                    clock_arg,
1023                ],
1024            );
1025        }
1026        CreateProposalParams::UpdateGuardian { url, metadata } => {
1027            let url_arg = builder.pure(&url);
1028            let metadata_arg = build_metadata(&mut builder, &metadata);
1029            builder.move_call(
1030                Function::new(
1031                    call_package,
1032                    Identifier::from_static("update_guardian"),
1033                    Identifier::from_static("propose"),
1034                ),
1035                vec![
1036                    hashi_arg,
1037                    validator_address_arg,
1038                    url_arg,
1039                    metadata_arg,
1040                    clock_arg,
1041                ],
1042            );
1043        }
1044        CreateProposalParams::EmergencyPause { pause, metadata } => {
1045            let pause_arg = builder.pure(&pause);
1046            let metadata_arg = build_metadata(&mut builder, &metadata);
1047            builder.move_call(
1048                Function::new(
1049                    call_package,
1050                    Identifier::from_static("emergency_pause"),
1051                    Identifier::from_static("propose"),
1052                ),
1053                vec![
1054                    hashi_arg,
1055                    validator_address_arg,
1056                    pause_arg,
1057                    metadata_arg,
1058                    clock_arg,
1059                ],
1060            );
1061        }
1062        CreateProposalParams::IgnoreMember {
1063            target_validator_address,
1064            ignored,
1065            metadata,
1066        } => {
1067            let target_arg = builder.pure(&target_validator_address);
1068            let ignored_arg = builder.pure(&ignored);
1069            let metadata_arg = build_metadata(&mut builder, &metadata);
1070            builder.move_call(
1071                Function::new(
1072                    // Upgrade-introduced modules route through the active package.
1073                    call_package,
1074                    Identifier::from_static("ignore_member"),
1075                    Identifier::from_static("propose"),
1076                ),
1077                vec![
1078                    hashi_arg,
1079                    validator_address_arg,
1080                    target_arg,
1081                    ignored_arg,
1082                    metadata_arg,
1083                    clock_arg,
1084                ],
1085            );
1086        }
1087    }
1088
1089    builder
1090}

1092/// Build a `config_value::Value` enum via a move call (e.g. `config_value::new_u64(v)`).
1093/// Returns the `Argument` holding the constructed `Value`.
1094fn build_config_value(
1095    builder: &mut TransactionBuilder,
1096    call_package: Address,
1097    value: &hashi_types::move_types::ConfigValue,
1098) -> sui_transaction_builder::Argument {
1099    use hashi_types::move_types::ConfigValue;
1100
1101    let (func_name, arg) = match value {
1102        ConfigValue::U64(v) => ("new_u64", builder.pure(v)),
1103        ConfigValue::Address(v) => ("new_address", builder.pure(v)),
1104        ConfigValue::String(v) => ("new_string", builder.pure(v)),
1105        ConfigValue::Bool(v) => ("new_bool", builder.pure(v)),
1106        ConfigValue::Bytes(v) => ("new_bytes", builder.pure(v)),
1107        ConfigValue::U128(v) => ("new_u128", builder.pure(v)),
1108        // The 32 LE bytes BCS-encode identically to a Move `u256` pure arg.
1109        ConfigValue::U256(v) => ("new_u256", builder.pure(v)),
1110    };
1111
1112    builder.move_call(
1113        Function::new(
1114            call_package,
1115            Identifier::from_static("config_value"),
1116            Identifier::new(func_name).unwrap(),
1117        ),
1118        vec![arg],
1119    )
1120}

1122fn build_config_entries(
1123    builder: &mut TransactionBuilder,
1124    type_package: Address,
1125    call_package: Address,
1126    entries: &[(String, ConfigValue)],
1127) -> sui_transaction_builder::Argument {
1128    let sui_framework = Address::from_static("0x2");
1129    let move_stdlib = Address::from_static("0x1");
1130    let string_type = TypeTag::Struct(Box::new(StructTag::new(
1131        move_stdlib,
1132        Identifier::from_static("string"),
1133        Identifier::from_static("String"),
1134        vec![],
1135    )));
1136    let value_type = TypeTag::Struct(Box::new(StructTag::new(
1137        type_package,
1138        Identifier::from_static("config_value"),
1139        Identifier::from_static("Value"),
1140        vec![],
1141    )));
1142    let map = builder.move_call(
1143        Function::new(
1144            sui_framework,
1145            Identifier::from_static("vec_map"),
1146            Identifier::from_static("empty"),
1147        )
1148        .with_type_args(vec![string_type.clone(), value_type.clone()]),
1149        vec![],
1150    );
1151    for (key, value) in entries {
1152        let key_arg = builder.pure(key);
1153        let value_arg = build_config_value(builder, call_package, value);
1154        builder.move_call(
1155            Function::new(
1156                sui_framework,
1157                Identifier::from_static("vec_map"),
1158                Identifier::from_static("insert"),
1159            )
1160            .with_type_args(vec![string_type.clone(), value_type.clone()]),
1161            vec![map, key_arg, value_arg],
1162        );
1163    }
1164    map
1165}

1167/// Build a `VecMap<String, String>` for proposal metadata via move calls.
1168///
1169/// Move structs like `VecMap` cannot be passed as pure args in PTBs.
1170/// Instead we construct one via `vec_map::empty()` + `vec_map::insert()`.
1171fn build_metadata(
1172    builder: &mut TransactionBuilder,
1173    metadata: &[(String, String)],
1174) -> sui_transaction_builder::Argument {
1175    let sui_framework = Address::from_static("0x2");
1176    let move_stdlib = Address::from_static("0x1");
1177
1178    let string_type = TypeTag::Struct(Box::new(StructTag::new(
1179        move_stdlib,
1180        Identifier::from_static("string"),
1181        Identifier::from_static("String"),
1182        vec![],
1183    )));
1184
1185    // vec_map::empty<String, String>()
1186    let map = builder.move_call(
1187        Function::new(
1188            sui_framework,
1189            Identifier::from_static("vec_map"),
1190            Identifier::from_static("empty"),
1191        )
1192        .with_type_args(vec![string_type.clone(), string_type.clone()]),
1193        vec![],
1194    );
1195
1196    // vec_map::insert(&mut map, key, value) for each entry
1197    for (key, value) in metadata {
1198        let key_arg = builder.pure(key);
1199        let value_arg = builder.pure(value);
1200        builder.move_call(
1201            Function::new(
1202                sui_framework,
1203                Identifier::from_static("vec_map"),
1204                Identifier::from_static("insert"),
1205            )
1206            .with_type_args(vec![string_type.clone(), string_type.clone()]),
1207            vec![map, key_arg, value_arg],
1208        );
1209    }
1210
1211    map
1212}

1289/// Resolve the committee member `sender` acts for in governance calls
1290/// (propose, vote, remove_vote, resign, withdraw_resignation).
1291///
1292/// An exact validator-address match always wins. Otherwise `sender` may be
1293/// the delegated operator of exactly one member. On chain,
1294/// `validator::update_operator_address` only checks that the caller is
1295/// authorized for the member being updated, so any member can point its
1296/// operator address at another member's signing address; a first-match scan
1297/// would then let that stray or malicious delegation redirect the signer's
1298/// vote to the wrong member (SEC-546). Two or more delegations to the same
1299/// signer are therefore an error naming the candidates rather than a guess.
1300fn resolve_governance_identity(members: &[MemberInfo], sender: Address) -> Result<Address> {
1301    if let Some(member) = members.iter().find(|m| *m.validator_address() == sender) {
1302        return Ok(*member.validator_address());
1303    }
1304
1305    let delegating: Vec<Address> = members
1306        .iter()
1307        .filter(|m| *m.operator_address() == sender)
1308        .map(|m| *m.validator_address())
1309        .collect();
1310
1311    match delegating.as_slice() {
1312        [] => anyhow::bail!("signer {sender} is not a committee member or delegated operator"),
1313        [validator_address] => Ok(*validator_address),
1314        candidates => {
1315            let count = candidates.len();
1316            let candidates = candidates
1317                .iter()
1318                .map(Address::to_hex)
1319                .collect::<Vec<_>>()
1320                .join(", ");
1321            anyhow::bail!(
1322                "signer {sender} is the delegated operator of {count} committee members \
1323                 ({candidates}); the CLI cannot tell which one to act for. Sign with that \
1324                 member's validator key instead, or have the other members point their \
1325                 operator address elsewhere (validator::update_operator_address, \
1326                 `hashi register --operator-address`)."
1327            )
1328        }
1329    }
1330}

1332/// Build a `proposal::vote<T>` transaction as a standalone. Reusable outside
1333/// `HashiClient` — e2e test infra needs to build vote PTBs for every
1334/// committee member.
1335pub fn build_vote_transaction(
1336    hashi_ids: HashiIds,
1337    hashi_initial_shared_version: u64,
1338    call_package: Address,
1339    validator_address: Address,
1340    proposal_id: Address,
1341    type_arg: TypeTag,
1342) -> TransactionBuilder {
1343    let mut builder = TransactionBuilder::new();
1344    // Fully-resolved shared inputs: the type arg may name an
1345    // upgrade-introduced type, and the fullnode's simulate-time resolver
1346    // fails with INVALID_LINKAGE on those when it has to inspect the call
1347    // to infer input mutability (sui >= 1.76).
1348    let hashi_arg = builder.object(
1349        ObjectInput::new(hashi_ids.hashi_object_id)
1350            .with_version(hashi_initial_shared_version)
1351            .as_shared()
1352            .with_mutable(true),
1353    );
1354    let validator_address_arg = builder.pure(&validator_address);
1355    let proposal_id_arg = builder.pure(&proposal_id);
1356    let clock_arg = builder.object(
1357        ObjectInput::new(SUI_CLOCK_OBJECT_ID)
1358            .with_version(1)
1359            .as_shared()
1360            .with_mutable(false),
1361    );
1362
1363    // Call through the active package: a transaction's linkage cannot call
1364    // the original package while a type argument references an
1365    // upgrade-introduced type (exact-v1 vs at-least-v2 conflict). Every
1366    // module rides along in an upgrade, and v1-era type args unify fine
1367    // under a v2 call, so latest is always safe.
1368    builder.move_call(
1369        Function::new(
1370            call_package,
1371            Identifier::from_static("proposal"),
1372            Identifier::from_static("vote"),
1373        )
1374        .with_type_args(vec![type_arg]),
1375        vec![hashi_arg, validator_address_arg, proposal_id_arg, clock_arg],
1376    );
1377
1378    builder
1379}

1381/// Get the TypeTag for a proposal type (from on-chain type)
1382///
1383/// `package_id` must be the type's DEFINING package (a Move type's tag
1384/// carries that address forever) — resolve it via
1385/// [`HashiClient::proposal_type_arg`], which routes each proposal type to
1386/// the package version that introduced it.
1387///
1388/// Returns an error if the proposal type is `Unknown`.
1389pub fn get_proposal_type_arg(
1390    package_id: Address,
1391    proposal_type: &crate::onchain::types::ProposalType,
1392) -> Result<TypeTag> {
1393    use crate::onchain::types::ProposalType;
1394
1395    let (module, name) = match proposal_type {
1396        ProposalType::Upgrade => ("upgrade", "Upgrade"),
1397        ProposalType::UpdateConfig => ("update_config", "UpdateConfig"),
1398        ProposalType::UpdateEpochConfig => ("update_epoch_config", "UpdateEpochConfig"),
1399        ProposalType::AddConfig => ("add_config", "AddConfig"),
1400        ProposalType::EnableVersion => ("enable_version", "EnableVersion"),
1401        ProposalType::DisableVersion => ("disable_version", "DisableVersion"),
1402        ProposalType::EmergencyPause => ("emergency_pause", "EmergencyPause"),
1403        ProposalType::AbortReconfig => ("abort_reconfig", "AbortReconfig"),
1404        ProposalType::UpdateGuardian => ("update_guardian", "UpdateGuardian"),
1405        ProposalType::IgnoreMember => ("ignore_member", "IgnoreMember"),
1406        ProposalType::Unknown(s) => {
1407            anyhow::bail!(
1408                "Cannot vote on unknown proposal type '{}'. \
1409                 This may be a new proposal type not yet supported by this CLI version.",
1410                s
1411            );
1412        }
1413    };
1414
1415    Ok(TypeTag::Struct(Box::new(StructTag::new(
1416        package_id,
1417        Identifier::new(module).context("Invalid module name")?,
1418        Identifier::new(name).context("Invalid type name")?,
1419        vec![],
1420    ))))
1421}
