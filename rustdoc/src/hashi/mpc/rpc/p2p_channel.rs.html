24pub struct RpcP2PChannel {
25    onchain_state: OnchainState,
26    epoch: u64,
27    protocol_label: &'static str,
28}
29
30impl RpcP2PChannel {
31    pub fn new(onchain_state: OnchainState, epoch: u64, protocol_label: &'static str) -> Self {
32        Self {
33            onchain_state,
34            epoch,
35            protocol_label,
36        }
37    }
38
39    fn get_client(&self, address: &Address) -> ChannelResult<Client> {
40        self.onchain_state
41            .state()
42            .hashi()
43            .committees
44            .client(address)
45            .ok_or(ChannelError::ClientNotFound(*address))
46    }
47
48    /// Wrap a protobuf message in a `tonic::Request` tagged with the MPC
49    /// protocol label, so the server-side metrics layer can attribute
50    /// traffic to the originating protocol.
51    fn build_request<T>(&self, message: T) -> tonic::Request<T> {
52        let mut req = tonic::Request::new(message);
53        req.metadata_mut().insert(
54            MPC_PROTOCOL_METADATA_KEY,
55            MetadataValue::from_static(self.protocol_label),
56        );
57        req
58    }
59}
60
61fn map_status(status: tonic::Status) -> ChannelError {
62    if status.code() == tonic::Code::Unavailable
63        && status
64            .message()
65            .contains(crate::grpc::SIGNING_MANAGER_NOT_READY_MSG)
66    {
67        ChannelError::NotReady(status.to_string())
68    } else {
69        ChannelError::RequestFailed(status.to_string())
70    }
71}
72
73#[async_trait]
74impl P2PChannel for RpcP2PChannel {
75    async fn send_messages(
76        &self,
77        recipient: &Address,
78        request: &SendMessagesRequest,
79    ) -> ChannelResult<SendMessagesResponse> {
80        let client = self.get_client(recipient)?;
81        let proto_request = self.build_request(request.to_proto(self.epoch));
82        let response = client
83            .mpc_service_client()
84            .send_messages(proto_request)
85            .await
86            .map_err(map_status)?;
87        SendMessagesResponse::try_from(response.get_ref())
88            .map_err(|e| ChannelError::RequestFailed(e.to_string()))
89    }
90
91    async fn retrieve_messages(

156#[cfg(test)]
157mod tests {
158    use super::*;
159
160    #[test]
161    fn map_status_treats_only_the_not_ready_response_as_not_ready() {
162        let not_ready = tonic::Status::unavailable(format!(
163            "{} for epoch 7; retry",
164            crate::grpc::SIGNING_MANAGER_NOT_READY_MSG
165        ));
166        assert!(matches!(map_status(not_ready), ChannelError::NotReady(_)));
167
168        let refused = tonic::Status::unavailable("error trying to connect: connection refused");
169        assert!(matches!(
170            map_status(refused),
171            ChannelError::RequestFailed(_)
172        ));
173
174        let not_found = tonic::Status::not_found("WithdrawalTransaction not found on-chain");
175        assert!(matches!(
176            map_status(not_found),
177            ChannelError::RequestFailed(_)
178        ));
179    }
