23#[derive(Debug, thiserror::Error)]
24pub(crate) enum Refused {
25    #[error("count check: {0}")]
26    Count(#[from] CountCheckError),
27    #[error("{0}")]
28    Variant(String),
29    #[error("share index {0} is not a non-zero u16")]
30    Shape(u32),
31    #[error("responder not in the committee")]
32    Member,
33    #[error("bytes don't match the certificate's hash")]
34    Hash,
35    #[error("{0}")]
36    Decode(String),
37}
38
39impl Refused {
40    pub(crate) fn label(&self) -> &'static str {
41        match self {
42            Self::Count(_) => "count",
43            Self::Variant(_) => "variant",
44            Self::Shape(_) => "shape",
45            Self::Member => "member",
46            Self::Hash => "hash",
47            Self::Decode(_) => "decode",
48        }
49    }
50}

52pub(crate) fn certified_retrieval(
53    protocol_type: ProtocolTypeIndicator,
54    response: proto::RetrieveMessagesResponse,
55    expected: MessagesHash,
56) -> Result<PreparedMessages, Refused> {
57    use proto::retrieve_messages_response::Messages as Wire;
58    match (protocol_type, response.messages) {
59        (ProtocolTypeIndicator::Dkg, Some(Wire::DkgMessage(message))) => {
60            Cached::certified(bytes(&message), expected).map(PreparedMessages::Dkg)
61        }
62        (ProtocolTypeIndicator::KeyRotation, Some(Wire::RotationMessages(map))) => {
63            rotation_blob(map)
64                .and_then(|blob| Cached::certified(&blob, expected).map(PreparedMessages::Rotation))
65        }
66        (protocol_type, _) => Err(Refused::Variant(format!(
67            "not a {protocol_type:?} retrieval response"
68        ))),
69    }
70}

72pub(crate) fn avid_bundle(
73    signer: Address,
74    shape: &ShapeParams,
75    dealer: PartyId,
76    response: &proto::RetrieveMessagesResponse,
77) -> Result<AvidNonceRetrievalMessage, Refused> {
78    use proto::retrieve_messages_response::Messages as Wire;
79    let not_avid = || Refused::Variant("not an AVID retrieval response".into());
80    let Some(Wire::AvidNonceRetrievalMessage(bundle)) = &response.messages else {
81        return Err(not_avid());
82    };
83    let current = shape.current();
84    let responder = current.party_of(&signer).ok_or(Refused::Member)?;
85    bundle
86        .common
87        .as_ref()
88        .map_or(Ok(()), |common| {
89            count_check::check_avss_common_message(bytes(common), shape.avid_shape(), dealer)
90        })
91        .and_then(|()| {
92            bundle.echo.as_ref().map_or(Ok(()), |echo| {
93                count_check::check_echo(bytes(echo), current.weights(), responder)
94            })
95        })
96        .and_then(|()| {
97            bundle.avid_vote.as_ref().map_or(Ok(()), |vote| {
98                count_check::check_avid_vote(bytes(vote), current.weights())
99            })
100        })?;
101    match RetrieveMessagesResponse::try_from(response)
102        .map_err(|e| Refused::Decode(e.to_string()))?
103        .messages
104    {
105        Messages::AvidNonceRetrieval(bundle) => Ok(bundle),
106        _ => Err(not_avid()),
107    }
108}

110pub(crate) fn avss_complaint_response(
111    protocol_type: ProtocolTypeIndicator,
112    responder_weight: Option<u16>,
113    response: &proto::ComplainResponse,
114) -> Result<avss::ComplaintResponse, Refused> {
115    use proto::complain_response::Responses;
116    let shares = match (protocol_type, &response.responses) {
117        (ProtocolTypeIndicator::Dkg, Some(Responses::DkgResponse(shares)))
118        | (ProtocolTypeIndicator::KeyRotation, Some(Responses::RotationResponse(shares))) => {
119            bytes(shares)
120        }
121        (protocol_type, _) => {
122            return Err(Refused::Variant(format!(
123                "not a {protocol_type:?} complaint response"
124            )));
125        }
126    };
127    let weight = responder_weight.ok_or(Refused::Member)?;
128    count_check::check_avss_complaint_response(shares, weight)?;
129    bcs::from_bytes(shares).map_err(complaint_decode_refused)
130}

132pub(crate) fn avid_complaint_response(
133    response: &proto::ComplainResponse,
134) -> Result<batch_avss_avid::ComplaintResponse, Refused> {
135    use proto::complain_response::Responses;
136    let Some(Responses::AvidNonceResponse(avid)) = &response.responses else {
137        return Err(Refused::Variant("not an AVID complaint response".into()));
138    };
139    count_check::check_avid_complaint_response(bytes(avid))?;
140    bcs::from_bytes(bytes(avid)).map_err(complaint_decode_refused)
141}

143pub(crate) fn public_output(
144    response: &proto::GetPublicMpcOutputResponse,
145    total_weight: u16,
146) -> Result<GetPublicMpcOutputResponse, Refused> {
147    let max = usize::from(total_weight);
148    (response.commitments.len() <= max)
149        .then_some(())
150        .ok_or(CountCheckError::TooMany {
151            field: "commitments",
152            found: response.commitments.len(),
153            max,
154        })?;
155    GetPublicMpcOutputResponse::try_from(response).map_err(|e| Refused::Decode(e.to_string()))
156}

158pub(crate) fn complaint_response_limit(responder_weight: u16) -> usize {
159    2 * count_check::avss_complaint_response_size(responder_weight) + 1024
160}

162fn rotation_blob(map: proto::RotationMessages) -> Result<Vec<u8>, Refused> {
163    const FIELD: &str = "rotation_messages";
164    let max = usize::from(u16::MAX);
165    (map.len() <= max)
166        .then_some(())
167        .ok_or(CountCheckError::TooMany {
168            field: FIELD,
169            found: map.len(),
170            max,
171        })?;
172    let mut entries = BTreeMap::new();
173    for entry in map.into_entries() {
174        let (key, message) = entry.map_err(|e| Refused::Decode(format!("{FIELD}: {e}")))?;
175        let index = u16::try_from(key)
176            .ok()
177            .filter(|index| *index != 0)
178            .ok_or(Refused::Shape(key))?;
179        if entries
180            .insert(index.to_le_bytes(), message.value.unwrap_or_default())
181            .is_some()
182        {
183            return Err(Refused::Count(CountCheckError::DuplicateKey {
184                field: FIELD,
185                index,
186            }));
187        }
188    }
189    let mut blob = Vec::with_capacity(
190        5 + entries
191            .iter()
192            .map(|(key, value)| key.len() + value.len())
193            .sum::<usize>(),
194    );
195    let mut count = entries.len();
196    while count >= 0x80 {
197        blob.push(count as u8 | 0x80);
198        count >>= 7;
199    }
200    blob.push(count as u8);
201    for (key, value) in &entries {
202        blob.extend_from_slice(key);
203        blob.extend_from_slice(value);
204    }
205    Ok(blob)
206}

208fn complaint_decode_refused(e: bcs::Error) -> Refused {
209    Refused::Decode(format!("complaint response: {e}"))
210}
211
212fn bytes(bcs: &Bcs) -> &[u8] {
213    bcs.value.as_deref().unwrap_or(&[])
214}

240    #[test]
241    fn complaint_and_signature_reply_limits_admit_the_largest_honest_replies() {
242        for weight in [0u16, 1, 127, 128, 10_000] {
243            let response =
244                proto::ComplainResponse::from(&ComplaintResponse::Dkg(avss::ComplaintResponse {
245                    shares: avss::SharesForNode {
246                        shares: (1..=weight)
247                            .map(|index| Eval {
248                                index: ShareIndex::new(index).unwrap(),
249                                value: S::zero(),
250                            })
251                            .collect(),
252                    },
253                }));
254            assert!(response.encoded_len() <= complaint_response_limit(weight));
255            avss_complaint_response(ProtocolTypeIndicator::Dkg, Some(weight), &response).unwrap();
256        }
257        let reply = proto::SendMessagesResponse {
258            signature: Some(vec![0; 96].into()),
259        };
260        assert!(reply.encoded_len() <= crate::grpc::SIGNATURE_REPLY_LIMIT);
261    }

263    #[test]
264    fn a_held_rotation_map_is_refused_before_its_blob_is_rebuilt() {
265        let held = |entries: Vec<(u32, Bcs)>| {
266            let wire = proto::RetrieveMessagesResponse {
267                messages: Some(
268                    proto::retrieve_messages_response::Messages::RotationMessages(
269                        entries.into_iter().collect(),
270                    ),
271                ),
272            }
273            .encode_to_vec();
274            proto::RetrieveMessagesResponse::decode(wire.as_slice()).unwrap()
275        };
276        let refused = |response: proto::RetrieveMessagesResponse| {
277            certified_retrieval(
278                ProtocolTypeIndicator::KeyRotation,
279                response,
280                MessagesHash::from([7; 32]),
281            )
282            .err()
283            .unwrap()
284        };
285
286        assert!(matches!(
287            refused(held(
288                (0..=u32::from(u16::MAX))
289                    .map(|key| (key, Bcs::default()))
290                    .collect()
291            )),
292            Refused::Count(CountCheckError::TooMany { found: 65_536, .. })
293        ));
294        assert!(matches!(
295            refused(held(vec![(1, Bcs::default()), (1, Bcs::default())])),
296            Refused::Count(CountCheckError::DuplicateKey { index: 1, .. })
297        ));
298        assert!(matches!(
299            refused(
300                proto::RetrieveMessagesResponse::decode(&[0x12, 3, 0x0a, 1, 0x08][..]).unwrap()
301            ),
302            Refused::Decode(_)
303        ));
304    }
