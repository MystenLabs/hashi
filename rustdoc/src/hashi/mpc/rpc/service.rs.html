36    #[tracing::instrument(skip(self, request))]
37    async fn send_messages(
38        &self,
39        request: tonic::Request<SendMessagesRequest>,
40    ) -> Result<tonic::Response<SendMessagesResponse>, Status> {
41        let sender = authenticate_caller(&request)?;
42        let external_request = request.into_inner();
43        let mpc_manager = self.mpc_manager()?;
44        let service = self.clone();
45        let process_duration = self
46            .metrics()
47            .mpc_rpc_handler_process_duration_seconds
48            .clone();
49        let rotation_received = self.metrics().mpc_rotation_messages_received_total.clone();
50        let response = spawn_blocking(move || -> Result<_, Status> {
51            let shape = service.shape_params()?;
52            let epoch = external_request.epoch;
53            validate_epoch(shape.epoch(), epoch)?;
54            gates::check_send_messages(&shape, sender, &external_request)
55                .map_err(|e| send_messages_error(sender, e))?;
56            let internal_request = types::SendMessagesRequest::try_from(&external_request)
57                .map_err(|e| Status::invalid_argument(e.to_string()))?;
58            if let Some(wire_shape) = rotation_wire_shape(&external_request) {
59                rotation_received.with_label_values(&[wire_shape]).inc();
60                if wire_shape == "map" && shape.current().rotation_messages_bcs() {
61                    tracing::warn!(
62                        "send_messages from {sender:?}: epoch {} rotation messages arrived as a \
63                         proto map, but the epoch's committee selects one BCS blob",
64                        shape.epoch()
65                    );
66                }
67            }
68            drop(external_request);
69            let prepared = PreparedMessages::new(internal_request.messages);
70            let label = match &prepared {
71                PreparedMessages::Dkg(_) => crate::metrics::MPC_LABEL_DKG,
72                PreparedMessages::Rotation(_) => crate::metrics::MPC_LABEL_KEY_ROTATION,
73                PreparedMessages::NonceGenerationAvid(_) | PreparedMessages::AvidNonceRetrieval => {
74                    crate::metrics::MPC_LABEL_NONCE_GENERATION
75                }
76            };
77            let _timer = process_duration.with_label_values(&[label]).start_timer();
78            let signed = {
79                let mut mgr = mpc_manager.write().unwrap();
80                validate_epoch(mgr.mpc_config.epoch, epoch)?;
81                mgr.handle_prepared_messages(sender, prepared)
82                    .map_err(|e| send_messages_error(sender, e))?
83            };
84            Ok(SendMessagesResponse::from(&signed))
85        })
86        .await?;
87        Ok(tonic::Response::new(response))
88    }

90    #[tracing::instrument(skip(self, request))]
91    async fn retrieve_messages(
92        &self,
93        request: tonic::Request<RetrieveMessagesRequest>,
94    ) -> Result<tonic::Response<RetrieveMessagesResponse>, Status> {
95        let requester = authenticate_caller(&request)?;
96        let external_request = request.into_inner();
97        let internal_request = types::RetrieveMessagesRequest::try_from(&external_request)
98            .map_err(|e| Status::invalid_argument(e.to_string()))?;
99        let mpc_manager = self.mpc_manager()?;
100        let refused = self.metrics().mpc_rpc_caller_refused_total.clone();
101        let response = spawn_blocking(move || -> Result<_, Status> {
102            let to_status = |e: MpcError| {
103                match &e {
104                    MpcError::NotFound(_) => {
105                        tracing::debug!("retrieve_messages: {e}");
106                    }
107                    _ => {
108                        tracing::warn!("retrieve_messages failed: {e}");
109                    }
110                }
111                mpc_error_to_status(e)
112            };
113            let (outcome, store, shape) = {
114                let mgr = mpc_manager.read().unwrap();
115                validate_epoch_current_or_previous(
116                    mgr.mpc_config.epoch,
117                    mgr.previous_epoch,
118                    internal_request.epoch,
119                )?;
120                caller_membership::check_current_or_previous(
121                    &requester,
122                    internal_request.epoch,
123                    mgr.mpc_config.epoch,
124                    &mgr.committee,
125                    mgr.previous_committee.as_ref(),
126                )
127                .map_err(|refusal| {
128                    caller_membership::refuse(
129                        &refused,
130                        "retrieve_messages",
131                        &requester,
132                        &format!("epoch {}", internal_request.epoch),
133                        refusal,
134                    )
135                })?;
136                let outcome = mgr
137                    .begin_retrieve(requester, &internal_request)
138                    .map_err(to_status)?;
139                (
140                    outcome,
141                    std::sync::Arc::clone(&mgr.public_messages_store),
142                    mgr.shape_params(),
143                )
144            };
145            let reply = match outcome {
146                RetrieveOutcome::Ready(reply) => reply,
147                RetrieveOutcome::NeedsStore => {
148                    retrieve_from_store(&*store, &internal_request).map_err(to_status)?
149                }
150                RetrieveOutcome::NeedsAvidStore(pending) => RetrieveReply::Avid(
151                    finish_avid_retrieval(&*store, pending, &shape).map_err(to_status)?,
152                ),
153            };
154            reply.to_proto(&shape, &internal_request).map_err(to_status)
155        })
156        .await?;
157        Ok(tonic::Response::new(response))
158    }

160    #[tracing::instrument(skip(self, request))]
161    async fn complain(
162        &self,
163        request: tonic::Request<ComplainRequest>,
164    ) -> Result<tonic::Response<ComplainResponse>, Status> {
165        let caller = authenticate_caller(&request)?;
166        let external_request = request.into_inner();
167        let mpc_manager = self.mpc_manager()?;
168        let service = self.clone();
169        let refused = self.metrics().mpc_rpc_caller_refused_total.clone();
170        let result = spawn_blocking(move || -> Result<_, Status> {
171            let epoch = external_request
172                .epoch
173                .ok_or_else(|| Status::invalid_argument("epoch: missing required field"))?;
174            let shape = service.shape_params()?;
175            validate_epoch_current_or_previous(shape.epoch(), shape.previous_epoch(), epoch)?;
176            let committee =
177                gates::complain_committee(&shape, epoch, &caller).map_err(|refusal| {
178                    caller_membership::refuse(
179                        &refused,
180                        "complain",
181                        &caller,
182                        &format!("epoch {epoch}"),
183                        refusal,
184                    )
185                })?;
186            if let Err(e) =
187                gates::check_complain(&shape, committee, caller, epoch, &external_request)
188            {
189                return Ok(Err(e));
190            }
191            let internal_request = types::ComplainRequest::try_from(&external_request)
192                .map_err(|e| Status::invalid_argument(e.to_string()))?;
193            drop(external_request);
194            let mut mgr = mpc_manager.write().unwrap();
195            validate_epoch_current_or_previous(
196                mgr.mpc_config.epoch,
197                mgr.previous_epoch,
198                internal_request.epoch,
199            )?;
200            Ok(mgr.handle_complain_request(caller, &internal_request))
201        })
202        .await?;
203        let complaint = result.map_err(|e| {
204            if matches!(e, MpcError::ComplaintWithheld { .. }) {
205                self.metrics().mpc_complaints_withheld_total.inc();
206            } else {
207                tracing::warn!("complain failed: {e}");
208            }
209            mpc_error_to_status(e)
210        })?;
211        Ok(tonic::Response::new(ComplainResponse::from(&complaint)))
212    }

214    #[tracing::instrument(skip(self, request))]
215    async fn get_public_mpc_output(
216        &self,
217        request: tonic::Request<GetPublicMpcOutputRequest>,
218    ) -> Result<tonic::Response<GetPublicMpcOutputResponse>, Status> {
219        let caller = authenticate_caller(&request)?;
220        let external_request = request.into_inner();
221        let internal_request = types::GetPublicMpcOutputRequest::try_from(&external_request)
222            .map_err(|e| Status::invalid_argument(e.to_string()))?;
223        let mpc_manager = self.mpc_manager()?;
224        let onchain_state = self.onchain_state_opt();
225        let refused = self.metrics().mpc_rpc_caller_refused_total.clone();
226        let response = spawn_blocking(move || -> Result<_, Status> {
227            let epoch = internal_request.epoch;
228            // Resolved before the manager lock is taken, so the two locks never nest.
229            let membership = onchain_state.as_ref().map(|onchain_state| {
230                caller_membership::check_epoch_or_successor(
231                    onchain_state.state().hashi().committees.committees(),
232                    epoch,
233                    &caller,
234                )
235            });
236            let output = {
237                let mgr = mpc_manager.read().unwrap();
238                mgr.handle_get_public_mpc_output_request(&internal_request)
239                    .map_err(|e| {
240                        tracing::warn!("get_public_mpc_output failed: {e}");
241                        mpc_error_to_status(e)
242                    })?
243            };
244            if let Some(Err(refusal)) = membership {
245                return Err(caller_membership::refuse(
246                    &refused,
247                    "get_public_mpc_output",
248                    &caller,
249                    &format!("epoch {epoch} or the committee after it"),
250                    refusal,
251                ));
252            }
253            Ok(GetPublicMpcOutputResponse::from(&output))
254        })
255        .await?;
256        Ok(tonic::Response::new(response))
257    }

260    async fn get_reconfig_completion_signature(
261        &self,
262        request: tonic::Request<GetReconfigCompletionSignatureRequest>,
263    ) -> Result<tonic::Response<GetReconfigCompletionSignatureResponse>, Status> {
264        let caller = authenticate_caller(&request)?;
265        let external_request = request.into_inner();
266        let epoch = external_request
267            .epoch
268            .ok_or_else(|| Status::invalid_argument("epoch: missing required field"))?;
269        let signature = self.get_reconfig_signature(epoch);
270        if signature.is_some() {
271            check_caller_in_epoch(self, "get_reconfig_completion_signature", &caller, epoch)?;
272        }
273        Ok(tonic::Response::new(
274            GetReconfigCompletionSignatureResponse {
275                signature: signature.map(Into::into),
276            },
277        ))
278    }

299            signature: signature.map(Into::into),

303    #[tracing::instrument(skip(self, request))]
304    async fn get_partial_signatures(
305        &self,
306        request: tonic::Request<GetPartialSignaturesRequest>,
307    ) -> Result<tonic::Response<GetPartialSignaturesResponse>, Status> {
308        let caller = authenticate_caller(&request)?;
309        let external_request = request.into_inner();
310        let internal_request = types::GetPartialSignaturesRequest::try_from(&external_request)
311            .map_err(|e| Status::invalid_argument(e.to_string()))?;
312        let request_epoch = external_request
313            .epoch
314            .ok_or_else(|| Status::invalid_argument("epoch: missing required field"))?;
315        let response = {
316            let signing_manager = self.signing_manager_for(request_epoch)?;
317            caller_membership::check_member(signing_manager.committee(), &caller).map_err(
318                |refusal| {
319                    caller_membership::refuse(
320                        &self.metrics().mpc_rpc_caller_refused_total,
321                        "get_partial_signatures",
322                        &caller,
323                        &format!("epoch {request_epoch}"),
324                        refusal,
325                    )
326                },
327            )?;
328            signing_manager
329                .handle_get_partial_signatures_request(&internal_request)
330                .map_err(|e| {
331                    match &e {
332                        SigningError::NotFound(_) => {
333                            tracing::debug!("get_partial_signatures: {e}");
334                        }
335                        _ => {
336                            tracing::warn!("get_partial_signatures failed: {e}");
337                        }
338                    }
339                    signing_error_to_status(e)
340                })?
341        };
342        Ok(tonic::Response::new(GetPartialSignaturesResponse::from(
343            &response,
344        )))
345    }

348fn authenticate_caller<T>(request: &tonic::Request<T>) -> Result<Address, Status> {
349    request
350        .extensions()
351        .get::<Address>()
352        .copied()
353        .ok_or_else(|| Status::permission_denied("unknown validator"))
354}

356fn check_caller_in_epoch(
357    service: &HttpService,
358    handler: &str,
359    caller: &Address,
360    epoch: u64,
361) -> Result<(), Status> {
362    let Some(onchain_state) = service.onchain_state_opt() else {
363        return Ok(());
364    };
365    let membership = caller_membership::check_epoch(
366        onchain_state.state().hashi().committees.committees(),
367        epoch,
368        caller,
369    );
370    membership.map_err(|refusal| {
371        caller_membership::refuse(
372            &service.metrics().mpc_rpc_caller_refused_total,
373            handler,
374            caller,
375            &format!("epoch {epoch}"),
376            refusal,
377        )
378    })
379}

381fn send_messages_error(sender: Address, e: MpcError) -> Status {
382    if matches!(&e, MpcError::NotReady(_)) {
383        tracing::info!("send_messages from {sender:?}: {e}");
384    } else {
385        tracing::warn!("send_messages from {sender:?} failed: {e}");
386    }
387    mpc_error_to_status(e)
388}

390fn rotation_wire_shape(request: &SendMessagesRequest) -> Option<&'static str> {
391    use hashi_types::proto::send_messages_request::Messages;
392    match &request.messages {
393        Some(Messages::RotationMessages(_)) => Some("map"),
394        Some(Messages::RotationMessagesBcs(_)) => Some("blob"),
395        _ => None,
396    }
397}

399fn validate_epoch(expected: u64, request_epoch: Option<u64>) -> Result<(), Status> {
400    let epoch =
401        request_epoch.ok_or_else(|| Status::invalid_argument("epoch: missing required field"))?;
402    if epoch != expected {
403        return Err(Status::failed_precondition(format!(
404            "epoch mismatch: expected {expected}, got {epoch}"
405        )));
406    }
407    Ok(())
408}
409
410fn validate_epoch_current_or_previous(
411    current_epoch: u64,
412    previous_epoch: u64,
413    request_epoch: u64,
414) -> Result<(), Status> {
415    if request_epoch != current_epoch && request_epoch != previous_epoch {
416        return Err(Status::failed_precondition(format!(
417            "epoch mismatch: expected {current_epoch} or {previous_epoch}, got {request_epoch}"
418        )));
419    }
420    Ok(())
421}

423pub(crate) fn signing_error_to_status(err: SigningError) -> Status {
424    match &err {
425        SigningError::InvalidMessage { .. } => Status::invalid_argument(err.to_string()),
426        SigningError::NotFound(_) => Status::not_found(err.to_string()),
427        SigningError::CryptoError(_) => Status::internal(err.to_string()),
428        SigningError::Timeout { .. } => Status::deadline_exceeded(err.to_string()),
429        SigningError::TooManyInvalidSignatures { .. } => {
430            Status::failed_precondition(err.to_string())
431        }
432        SigningError::PoolExhausted => Status::resource_exhausted(err.to_string()),
433        SigningError::RequestChanged { .. } => Status::failed_precondition(err.to_string()),
434        SigningError::PresigBatchNotSealed { .. } => Status::unavailable(err.to_string()),
435        SigningError::SealDealerSetMismatch { .. } => Status::failed_precondition(err.to_string()),
436        SigningError::MalformedSealRandomness { .. } => {
437            Status::failed_precondition(err.to_string())
438        }
439    }
440}

442fn mpc_error_to_status(err: MpcError) -> Status {
443    use types::MpcError::*;
444    match &err {
445        InvalidThreshold(_) | InvalidMessage { .. } | InvalidCertificate(_) => {
446            Status::invalid_argument(err.to_string())
447        }
448        Timeout { .. } => Status::deadline_exceeded(err.to_string()),
449        NotEnoughParticipants { .. }
450        | NotEnoughApprovals { .. }
451        | InvalidConfig(_)
452        | NotReady(_) => Status::failed_precondition(err.to_string()),
453        NotFound(_) => Status::not_found(err.to_string()),
454        ComplaintWithheld { .. } => Status::permission_denied(err.to_string()),
455        _ => Status::internal(err.to_string()),
456    }
