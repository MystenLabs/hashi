20/// Get a required field from a proto message.
21#[allow(clippy::result_large_err)]
22fn required<T>(field: Option<T>, name: &str) -> Result<T, TryFromProtoError> {
23    field.ok_or_else(|| TryFromProtoError::missing(name))
24}

26/// Parse an address from a hex string.
27#[allow(clippy::result_large_err)]
28fn parse_address(s: &str, field: &str) -> Result<Address, TryFromProtoError> {
29    s.parse::<Address>()
30        .map_err(|e| TryFromProtoError::invalid(field, e))
31}

33pub(crate) fn partial_sigs_response_limit(requested_ids: usize, max_owned: usize) -> usize {
34    const EVAL_BYTES: usize = 34;
35    const PER_ID_OVERHEAD_BYTES: usize = 512;
36    const FLOOR_BYTES: usize = 64 * 1024;
37    requested_ids * (max_owned * EVAL_BYTES + PER_ID_OVERHEAD_BYTES) + FLOOR_BYTES
38}

40const MAX_PARTIAL_SIGS_BLOB_BYTES: usize = 3 + u16::MAX as usize * 34;

42/// Deserialize a BCS-encoded proto field.
43#[allow(clippy::result_large_err)]
44fn deserialize_bcs<'de, T: Deserialize<'de>>(
45    bcs: &'de Bcs,
46    field: &str,
47) -> Result<T, TryFromProtoError> {
48    bcs.deserialize()
49        .map_err(|e| TryFromProtoError::invalid(field, e))
50}

52/// Serialize a value to BCS with a type name.
53fn serialize_bcs<T: Serialize>(value: &T) -> Bcs {
54    Bcs::serialize(value)
55        .expect("serialization should succeed")
56        .with_name(std::any::type_name::<T>())
57}

59/// Parse a share index map from proto.
60#[allow(clippy::result_large_err)]
61fn parse_rotation_messages_map(
62    map: &std::collections::HashMap<u32, Bcs>,
63) -> Result<BTreeMap<ShareIndex, avss::Message>, TryFromProtoError> {
64    let mut messages = BTreeMap::new();
65    for (&index, bcs) in map {
66        let share_index = ShareIndex::new(index as u16).ok_or_else(|| {
67            TryFromProtoError::invalid("rotation_messages.key", "index must be non-zero")
68        })?;
69        let message: avss::Message = deserialize_bcs(bcs, "rotation_messages.value")?;
70        messages.insert(share_index, message);
71    }
72    Ok(messages)
73}

75/// Convert rotation messages map to proto format.
76fn rotation_messages_to_proto(
77    messages: &BTreeMap<ShareIndex, avss::Message>,
78) -> std::collections::HashMap<u32, Bcs> {
79    messages
80        .iter()
81        .map(|(idx, msg)| (idx.get() as u32, serialize_bcs(msg)))
82        .collect()
83}

85/// Convert a domain AVID nonce message to its grouped proto form.
86fn avid_nonce_message_to_proto(avid: &types::AvidNonceMessage) -> proto::AvidNonceMessage {
87    use proto::avid_nonce_message::Kind;
88    let kind = match &avid.kind {
89        types::AvidNonceMessageKind::Optimistic(message) => {
90            Kind::Optimistic(proto::AvidNonceOptimistic {
91                message: Some(serialize_bcs(message)),
92            })
93        }
94        types::AvidNonceMessageKind::Dispersal {
95            dispersal,
96            confirm_cert,
97        } => Kind::Dispersal(proto::AvidNonceDispersal {
98            dispersal: Some(serialize_bcs(dispersal)),
99            confirm_cert: Some(serialize_bcs(confirm_cert)),
100        }),
101        types::AvidNonceMessageKind::Echo { dealer, echo } => Kind::Echo(proto::AvidNonceEcho {
102            dealer: Some(dealer.to_string()),
103            echo: Some(serialize_bcs(echo)),
104        }),
105    };
106    proto::AvidNonceMessage {
107        batch_index: Some(avid.batch_index),
108        kind: Some(kind),
109    }
110}

112/// Parse a grouped proto AVID nonce message into its domain form.
113#[allow(clippy::result_large_err)]
114fn avid_nonce_message_from_proto(
115    avid: &proto::AvidNonceMessage,
116) -> Result<types::AvidNonceMessage, TryFromProtoError> {
117    use proto::avid_nonce_message::Kind;
118    let batch_index = required(avid.batch_index, "avid_nonce_message.batch_index")?;
119    let kind = match avid.kind.as_ref() {
120        Some(Kind::Optimistic(optimistic)) => {
121            let message: batch_avss_avid::AvssMessage = deserialize_bcs(
122                required(
123                    optimistic.message.as_ref(),
124                    "avid_nonce_message.optimistic.message",
125                )?,
126                "avid_nonce_message.optimistic.message",
127            )?;
128            types::AvidNonceMessageKind::Optimistic(message)
129        }
130        Some(Kind::Dispersal(dispersal_msg)) => {
131            let dispersal: batch_avss_avid::Dispersal = deserialize_bcs(
132                required(
133                    dispersal_msg.dispersal.as_ref(),
134                    "avid_nonce_message.dispersal.dispersal",
135                )?,
136                "avid_nonce_message.dispersal.dispersal",
137            )?;
138            let confirm_cert: types::AvidConfirmCertificate = deserialize_bcs(
139                required(
140                    dispersal_msg.confirm_cert.as_ref(),
141                    "avid_nonce_message.dispersal.confirm_cert",
142                )?,
143                "avid_nonce_message.dispersal.confirm_cert",
144            )?;
145            types::AvidNonceMessageKind::Dispersal {
146                dispersal,
147                confirm_cert,
148            }
149        }
150        Some(Kind::Echo(echo_msg)) => {
151            let dealer = parse_address(
152                required(echo_msg.dealer.as_ref(), "avid_nonce_message.echo.dealer")?,
153                "avid_nonce_message.echo.dealer",
154            )?;
155            let echo: batch_avss_avid::Echo = deserialize_bcs(
156                required(echo_msg.echo.as_ref(), "avid_nonce_message.echo.echo")?,
157                "avid_nonce_message.echo.echo",
158            )?;
159            types::AvidNonceMessageKind::Echo { dealer, echo }
160        }
161        None => return Err(TryFromProtoError::missing("avid_nonce_message.kind")),
162    };
163    Ok(types::AvidNonceMessage { batch_index, kind })
164}

170impl types::SendMessagesRequest {
171    pub fn to_proto(&self, epoch: u64) -> proto::SendMessagesRequest {
172        use proto::send_messages_request::Messages;
173        let messages = match &self.messages {
174            types::Messages::Dkg(message) => Messages::DkgMessage(serialize_bcs(message)),
175            types::Messages::Rotation(messages) => {
176                Messages::RotationMessages(proto::RotationMessages {
177                    messages: rotation_messages_to_proto(messages),
178                })
179            }
180            types::Messages::NonceGeneration(nonce) => {
181                Messages::NonceMessage(proto::NonceMessage {
182                    batch_index: Some(nonce.batch_index),
183                    message: Some(serialize_bcs(&nonce.message)),
184                })
185            }
186            types::Messages::NonceGenerationAvid(avid) => {
187                Messages::AvidNonceMessage(avid_nonce_message_to_proto(avid))
188            }
189            types::Messages::AvidNonceRetrieval(_) => {
190                unreachable!("retrieval messages are response-only")
191            }
192        };
193        proto::SendMessagesRequest {
194            epoch: Some(epoch),
195            messages: Some(messages),
196        }
197    }
198}

200impl TryFrom<&proto::SendMessagesRequest> for types::SendMessagesRequest {
201    type Error = TryFromProtoError;
202
203    fn try_from(value: &proto::SendMessagesRequest) -> Result<Self, Self::Error> {
204        use proto::send_messages_request::Messages;
205        let messages = match &value.messages {
206            Some(Messages::DkgMessage(dkg_message)) => {
207                let message: avss::Message = deserialize_bcs(dkg_message, "dkg_message")?;
208                types::Messages::Dkg(message)
209            }
210            Some(Messages::RotationMessages(rotation)) => {
211                types::Messages::Rotation(parse_rotation_messages_map(&rotation.messages)?)
212            }
213            Some(Messages::NonceMessage(nonce)) => {
214                let batch_index = required(nonce.batch_index, "nonce_message.batch_index")?;
215                let message: batch_avss::Message = deserialize_bcs(
216                    required(nonce.message.as_ref(), "nonce_message.message")?,
217                    "nonce_message.message",
218                )?;
219                types::Messages::NonceGeneration(types::NonceMessage {
220                    batch_index,
221                    message,
222                })
223            }
224            Some(Messages::AvidNonceMessage(avid)) => {
225                types::Messages::NonceGenerationAvid(avid_nonce_message_from_proto(avid)?)
226            }
227            None => {
228                return Err(TryFromProtoError::missing("messages"));
229            }
230        };
231        Ok(Self { messages })
232    }
233}

292fn mpc_protocol_type_to_proto(pt: types::ProtocolTypeIndicator) -> proto::MpcProtocolType {
293    match pt {
294        types::ProtocolTypeIndicator::Dkg => proto::MpcProtocolType::Dkg,
295        types::ProtocolTypeIndicator::KeyRotation => proto::MpcProtocolType::KeyRotation,
296        types::ProtocolTypeIndicator::NonceGeneration => proto::MpcProtocolType::NonceGeneration,
297    }
298}
299
300#[allow(clippy::result_large_err)]
301fn mpc_protocol_type_from_proto(
302    value: i32,
303) -> Result<types::ProtocolTypeIndicator, TryFromProtoError> {
304    match proto::MpcProtocolType::try_from(value) {
305        Ok(proto::MpcProtocolType::Dkg) => Ok(types::ProtocolTypeIndicator::Dkg),
306        Ok(proto::MpcProtocolType::KeyRotation) => Ok(types::ProtocolTypeIndicator::KeyRotation),
307        Ok(proto::MpcProtocolType::NonceGeneration) => {
308            Ok(types::ProtocolTypeIndicator::NonceGeneration)
309        }
310        _ => Err(TryFromProtoError::missing("valid protocol_type")),
311    }

419impl TryFrom<&proto::ComplainRequest> for types::ComplainRequest {
420    type Error = TryFromProtoError;
421
422    fn try_from(value: &proto::ComplainRequest) -> Result<Self, Self::Error> {
423        let epoch = *required(value.epoch.as_ref(), "epoch")?;
424        let dealer = parse_address(required(value.dealer.as_ref(), "dealer")?, "dealer")?;
425        let share_index = if let Some(idx) = value.share_index {
426            Some(
427                std::num::NonZeroU16::new(idx as u16)
428                    .ok_or_else(|| TryFromProtoError::invalid("share_index", "must be non-zero"))?,

430        } else {
431            None
432        };
433        let protocol_type =
434            mpc_protocol_type_from_proto(required(value.protocol_type, "protocol_type")?)?;
435        let complaint_bytes = required(value.complaint.as_ref(), "complaint")?;
436        let complaint = deserialize_bcs(complaint_bytes, "complaint")?;
437        Ok(Self {
438            dealer,
439            share_index,
440            batch_index: value.batch_index,
441            complaint,
442            protocol_type,
443            epoch,
444        })
445    }
446}

626impl TryFrom<&proto::GetPartialSignaturesResponse> for types::GetPartialSignaturesResponse {

629    fn try_from(value: &proto::GetPartialSignaturesResponse) -> Result<Self, Self::Error> {

634                let id = parse_address(id, "partial_sigs key")?;
635                let len = bcs.value.as_ref().map_or(0, |v| v.len());
636                if len > MAX_PARTIAL_SIGS_BLOB_BYTES {
637                    return Err(TryFromProtoError::invalid(
638                        "partial_sigs",
639                        format!("{len} bytes exceeds the {MAX_PARTIAL_SIGS_BLOB_BYTES}-byte bound"),
640                    ));
641                }
642                let sigs = deserialize_bcs(bcs, "partial_sigs")?;
643                Ok((id, sigs))
644            })
645            .collect::<Result<BTreeMap<_, _>, _>>()?;
646        let signing_nonces = value
647            .signing_nonces
648            .iter()
649            .map(|(id, bcs)| {
650                let id = parse_address(id, "signing_nonces key")?;
651                let nonce = bcs.value.clone().unwrap_or_default().to_vec();
652                Ok((id, nonce))
653            })
654            .collect::<Result<BTreeMap<_, _>, _>>()?;
655        Ok(Self {
656            partial_sigs,
657            signing_nonces,
658        })
659    }
660}

689        }

692        }

695        }

698    /// Run a minimal AVID dealer/receiver flow to mint the three opaque fastcrypto
699    /// payloads — their innards are private, so they can only come from the protocol.
700    /// Parties `0..=7` confirm; pending recipients `{8, 9}` have weight `2 < f = 3`.
701    fn avid_artifacts() -> (
702        batch_avss_avid::AvssMessage,
703        batch_avss_avid::Dispersal,
704        batch_avss_avid::Echo,
705    ) {
706        let (t, f, n, batch) = (3u16, 3u16, 10u16, 3u16);
707        let mut rng = rand::thread_rng();
708        let sks: Vec<_> = (0..n)
709            .map(|_| ecies_v1::PrivateKey::<types::EncryptionGroupElement>::new(&mut rng))
710            .collect();
711        let nodes = Nodes::new(
712            sks.iter()
713                .enumerate()
714                .map(|(id, sk)| Node {
715                    id: id as u16,
716                    pk: ecies_v1::PublicKey::from_private_key(sk),
717                    weight: 1,
718                })
719                .collect(),
720        )
721        .unwrap();
722        let sid = b"avid conversion test".to_vec();
723        let params = Parameters { t, f };
724        let dealer =
725            batch_avss_avid::Dealer::new(nodes.clone(), 0, params, sid.clone(), batch).unwrap();
726        let builder = dealer.create_avss_messages(&mut rng).unwrap();
727        let own_message = builder.message_for(0).unwrap();
728        let cert = StubAvssCert {
729            voters: (0u16..=7).collect(),
730            vote: batch_avss_avid::AvssVote {
731                common_message_hash: own_message.common.hash(),
732            },
733        };
734        let messages = dealer.create_avid_messages(&builder, cert).unwrap();
735        // Receiver 0 processes the message addressed to it (its shards verify against its
736        // own Merkle leaf), then echoes for a pending recipient.
737        let pending: PartyId = 8;
738        let avid_message = messages.message_for(0).unwrap();
739        let dispersal = avid_message.dispersal.clone();
740        let receiver =
741            batch_avss_avid::Receiver::new(nodes, 0, 0, params, sid, sks[0].clone(), batch)
742                .unwrap();
743        let (_, _, verified_common) = receiver.process_avss_message(&own_message).unwrap();
744        let (echo_builder, _) = receiver
745            .process_avid_message(&verified_common, avid_message)
746            .unwrap();
747        let echo = echo_builder.create_echo(pending).unwrap();
748        (own_message, dispersal, echo)
749    }

760    }

806#[cfg(test)]
807mod partial_sigs_limit_tests {
808    use super::*;
809    use fastcrypto::groups::GroupElement;
810    use fastcrypto_tbls::polynomial::Eval;
811    use fastcrypto_tbls::threshold_schnorr::S;
812    use prost::Message;
813    use sui_sdk_types::Address;
814
815    fn response(inputs: usize, owned: usize) -> proto::GetPartialSignaturesResponse {
816        let mut partial_sigs = BTreeMap::new();
817        let mut signing_nonces = BTreeMap::new();
818        for i in 0..inputs {
819            let mut id = [0u8; 32];
820            id[0] = i as u8;
821            id[1] = (i >> 8) as u8;
822            let id = Address::new(id);
823            let sigs: Vec<Eval<S>> = (1..=owned as u16)
824                .map(|index| Eval {
825                    index: ShareIndex::new(index).unwrap(),
826                    value: S::zero(),
827                })
828                .collect();
829            partial_sigs.insert(id, sigs);
830            signing_nonces.insert(id, vec![2u8; 33]);
831        }
832        (&types::GetPartialSignaturesResponse {
833            partial_sigs,
834            signing_nonces,
835        })
836            .into()
837    }
838
839    #[test]
840    fn partial_sigs_response_limit_covers_the_honest_maximum() {
841        for (ids, owned) in [(1, 1), (64, 240), (400, 60), (700, 1)] {
842            let encoded = response(ids, owned).encoded_len();
843            assert!(
844                encoded <= partial_sigs_response_limit(ids, owned),
845                "{ids} ids x {owned} shares encode to {encoded} bytes"
846            );
847            let overhead = encoded - ids * owned * 34;
848            assert!(
849                overhead <= ids * 512,
850                "{ids} ids x {owned} shares carry {overhead} bytes of non-eval overhead"
851            );
852        }
853    }
