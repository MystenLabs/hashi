23const POLL_INTERVAL: Duration = Duration::from_secs(1);
24const MAX_RETRY_INTERVAL: Duration = Duration::from_secs(30);
25const RPC_TIMEOUT: Duration = Duration::from_secs(5);
26const SUBMIT_STAGGER: Duration = Duration::from_secs(5);
27
28pub(crate) fn start(
29    inner: &Arc<Hashi>,
30    tasks: &Mutex<JoinSet<()>>,
31    epoch: u64,
32    batch_index: u32,
33    dealer_set_digest: [u8; 32],
34) {
35    let message = PresigDealerSetMessage {
36        epoch,
37        batch_index,
38        dealer_set_digest: dealer_set_digest.to_vec(),
39    };
40    let signature = match sign(inner, &message) {
41        Ok(signature) => signature,
42        Err(e) => {
43            warn!("Cannot sign PresigDealerSet for epoch {epoch} batch {batch_index}: {e:#}");
44            return;
45        }
46    };
47    if !inner.store_presig_dealer_set_signature_if_absent(
48        epoch,
49        batch_index,
50        signature.as_bytes().to_vec(),
51    ) {
52        return;
53    }
54    let inner = Arc::clone(inner);
55    let mut tasks = tasks.lock().unwrap();
56    while tasks.try_join_next().is_some() {}
57    tasks.spawn(async move { seal(&inner, &message, signature).await });
58}

60fn sign(inner: &Hashi, message: &PresigDealerSetMessage) -> anyhow::Result<BLS12381Signature> {
61    let committee = inner.committee_for_epoch(message.epoch)?;
62    let my_address = inner.config.validator_address()?;
63    let key = inner.find_signing_key_for_committee(&committee, my_address, message.epoch)?;
64    Ok(key
65        .sign(
66            inner.config.hashi_ids().hashi_object_id,
67            message.epoch,
68            my_address,
69            message,
70        )
71        .signature()
72        .clone())
73}

75async fn seal(inner: &Arc<Hashi>, message: &PresigDealerSetMessage, signature: BLS12381Signature) {
76    let (epoch, batch_index) = (message.epoch, message.batch_index);
77    let mut retry_interval = POLL_INTERVAL;
78    while let Err(e) = try_seal(inner, message, &signature).await {
79        warn!("Sealing presig batch {batch_index} of epoch {epoch} failed: {e:#}");
80        retry_interval = (retry_interval * 2).min(MAX_RETRY_INTERVAL);
81        tokio::time::sleep(retry_interval).await;
82    }
83}

85async fn try_seal(
86    inner: &Arc<Hashi>,
87    message: &PresigDealerSetMessage,
88    signature: &BLS12381Signature,
89) -> anyhow::Result<()> {
90    if done_or_not_relevant(inner, message) {
91        return Ok(());
92    }
93    let committee = inner.committee_for_epoch(message.epoch)?;
94    let my_address = inner.config.validator_address()?;
95    tokio::time::sleep(submit_delay(&committee, my_address, message.batch_index)).await;
96    let Some(cert) = collect(inner, &committee, message, signature).await? else {
97        return Ok(());
98    };
99    if done_or_not_relevant(inner, message) {
100        return Ok(());
101    }
102    let mut executor = crate::sui_tx_executor::SuiTxExecutor::from_hashi(Arc::clone(inner))?;
103    executor
104        .execute_submit_presig_dealer_set(message, cert.committee_signature())
105        .await?;
106    info!(
107        "Submitted PresigDealerSet for epoch {} batch {}",
108        message.epoch, message.batch_index
109    );
110    Ok(())
111}

113fn done_or_not_relevant(inner: &Hashi, message: &PresigDealerSetMessage) -> bool {
114    let onchain_state = inner.onchain_state();
115    if onchain_state.epoch() > message.epoch {
116        return true;
117    }
118    match onchain_state
119        .presig_seals(message.epoch)
120        .remove(&message.batch_index)
121    {
122        Some(seal) => {
123            if seal.dealer_set_digest != message.dealer_set_digest {
124                warn!(
125                    "Presig batch {} of epoch {} was sealed over a dealer set this node did \
126                     not build; this node will not sign from it",
127                    message.batch_index, message.epoch,
128                );
129            }
130            true
131        }
132        None => false,
133    }
134}

136async fn collect(
137    inner: &Hashi,
138    committee: &RuntimeCommittee,
139    message: &PresigDealerSetMessage,
140    signature: &BLS12381Signature,
141) -> anyhow::Result<Option<SignedMessage<PresigDealerSetMessage>>> {
142    let my_address = inner.config.validator_address()?;
143    let mut aggregator =
144        committee.signature_aggregator(inner.config.hashi_ids().hashi_object_id, message.clone());
145    aggregator
146        .add_signature_from(my_address, signature.clone())
147        .map_err(|e| anyhow::anyhow!("failed to add own signature: {e}"))?;
148    let mut missing: HashSet<Address> = committee
149        .members()
150        .iter()
151        .map(|m| m.validator_address())
152        .filter(|address| *address != my_address)
153        .collect();
154    let required_weight = certificate_threshold(committee.total_weight());
155    let mut rejected: HashSet<Address> = HashSet::new();
156    let mut poll_interval = POLL_INTERVAL;
157    while aggregator.weight() < required_weight {
158        if done_or_not_relevant(inner, message) {
159            return Ok(None);
160        }
161        let fetched = join_all(missing.iter().map(|&address| async move {
162            let signature = tokio::time::timeout(RPC_TIMEOUT, async {
163                let client = inner
164                    .onchain_state()
165                    .state()
166                    .hashi()
167                    .committees
168                    .client(&address)?;
169                client
170                    .get_presig_dealer_set_signature(message.epoch, message.batch_index)
171                    .await
172                    .ok()
173                    .flatten()
174            })
175            .await
176            .ok()
177            .flatten();
178            (address, signature)
179        }))
180        .await;
181        for (address, signature) in fetched {
182            let Some(bytes) = signature else {
183                continue;
184            };
185            match BLS12381Signature::from_bytes(&bytes)
186                .map_err(|e| e.to_string())
187                .and_then(|signature| {
188                    aggregator
189                        .add_signature_from(address, signature)
190                        .map_err(|e| e.to_string())
191                }) {
192                Ok(()) => {
193                    missing.remove(&address);
194                }
195                Err(e) => {
196                    if rejected.insert(address) {
197                        info!("PresigDealerSet signature from {address} rejected: {e}");
198                    }
199                }
200            }
201        }
202        if aggregator.weight() < required_weight {
203            tokio::time::sleep(poll_interval).await;
204            poll_interval = (poll_interval * 2).min(MAX_RETRY_INTERVAL);
205        }
206    }
207    aggregator
208        .finish()
209        .map(Some)
210        .map_err(|e| anyhow::anyhow!("failed to finalize PresigDealerSet certificate: {e}"))
211}

213fn submit_delay(committee: &RuntimeCommittee, my_address: Address, batch_index: u32) -> Duration {
214    let members = committee.members();
215    let Some(position) = members
216        .iter()
217        .position(|m| m.validator_address() == my_address)
218    else {
219        return SUBMIT_STAGGER;
220    };
221    let rank = (position + members.len() - batch_index as usize % members.len()) % members.len();
222    SUBMIT_STAGGER * rank as u32
223}
