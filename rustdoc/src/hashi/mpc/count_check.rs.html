22const POINT: usize = POINT_SIZE_IN_BYTES;
23const SCALAR: usize = SCALAR_SIZE_IN_BYTES;
24const EG_POINT: usize = RISTRETTO_POINT_BYTE_LENGTH;
25const DDH_NIZK: usize = 2 * EG_POINT + RISTRETTO_SCALAR_BYTE_LENGTH;
26const SHARE: usize = size_of::<ShareIndex>() + SCALAR;
27const HASH: usize = Blake2b256::OUTPUT_SIZE;
28const MERKLE_DIGEST: usize = DIGEST_LEN;
29const ADDRESS: usize = Address::LENGTH;
30const PARTY_ID: usize = size_of::<PartyId>();
31const BATCH_INDEX: usize = size_of::<u32>();
32const EPOCH: usize = size_of::<u64>();
33const AGGREGATE_SIGNATURE: usize = BLS_G2_LENGTH;
34const SHARED_COMPONENTS: usize = 2 * EG_POINT + DDH_NIZK;
35const RECOVERY_PACKAGE: usize = EG_POINT + DDH_NIZK;
36const MAX_MERKLE_PATH: usize = usize::BITS as usize - 1;
37const MESSAGES_NONCE_GENERATION_AVID: usize = Messages::NONCE_GENERATION_AVID_VARIANT as usize;
38const AVID_NONCE_MESSAGE_KIND_ECHO: usize = AvidNonceMessageKind::ECHO_VARIANT as usize;
39const PROTOCOL_COMPLAINT_AVSS: usize = ProtocolComplaint::AVSS_VARIANT as usize;
40const PROTOCOL_COMPLAINT_AVID_REVEAL: usize = ProtocolComplaint::AVID_REVEAL_VARIANT as usize;
41const PROTOCOL_COMPLAINT_AVID_BLAME: usize = ProtocolComplaint::AVID_BLAME_VARIANT as usize;
42const RS_LENGTH_PREFIX: usize = size_of::<u32>();
43const RS_ELEMENT: usize = 2;

45#[derive(Clone, Debug, PartialEq, Eq, thiserror::Error)]
46pub enum CountCheckError {
47    #[error("{field}: {found} entries, expected {expected}")]
48    WrongCount {
49        field: &'static str,
50        found: usize,
51        expected: usize,
52    },
53    #[error("{field}: {found} entries, at most {max} allowed")]
54    TooMany {
55        field: &'static str,
56        found: usize,
57        max: usize,
58    },
59    #[error("{field}: {found} bytes, expected {expected}")]
60    WrongBytes {
61        field: &'static str,
62        found: usize,
63        expected: usize,
64    },
65    #[error("{field}: {found} bytes, at most {max} allowed")]
66    TooManyBytes {
67        field: &'static str,
68        found: usize,
69        max: usize,
70    },
71    #[error("{field}: payload ends early")]
72    Truncated { field: &'static str },
73    #[error("{field}: invalid ULEB128 length or variant index")]
74    BadLength { field: &'static str },
75    #[error("{field}: keys are not in canonical BCS order")]
76    KeyOrder { field: &'static str },
77    #[error("{field}: share index {index} is zero or not owned by the dealer")]
78    UnownedKey { field: &'static str, index: u16 },
79    #[error("{field}: key {key} is not a u16 share index")]
80    KeyRange { field: &'static str, key: u32 },
81    #[error("{field}: share index {index} appears more than once")]
82    DuplicateKey { field: &'static str, index: u16 },
83    #[error("{field}: an entry doesn't parse: {reason}")]
84    BadEntry { field: &'static str, reason: String },
85    #[error("{field}: party {party} is not in the committee")]
86    UnknownParty { field: &'static str, party: u16 },
87    #[error("{field}: unknown variant {variant}")]
88    BadVariant { field: &'static str, variant: usize },
89    #[error("{0} bytes after the payload")]
90    Trailing(usize),
91}

101#[derive(Clone, Copy, Debug)]
102pub struct PartyWeights<'a>(pub &'a [u16]);
103
104impl PartyWeights<'_> {
105    fn num_nodes(self) -> usize {
106        self.0.len()
107    }
108
109    fn weight_of(self, field: &'static str, party: u16) -> CountCheckResult<usize> {
110        self.0
111            .get(usize::from(party))
112            .map(|&weight| usize::from(weight))
113            .ok_or(CountCheckError::UnknownParty { field, party })
114    }
115}

124#[derive(Clone, Copy, Debug)]
125pub struct DispersalShape<'a> {
126    pub weights: PartyWeights<'a>,
127    pub batch_size: usize,
128    pub data_shards: NonZeroUsize,
129}
130
131impl<'a> DispersalShape<'a> {
132    pub fn new(
133        weights: PartyWeights<'a>,
134        dealer: u16,
135        batch_size_per_weight: u16,
136        max_faulty: u16,
137    ) -> Option<Self> {
138        let total_weight: usize = weights.0.iter().map(|&weight| usize::from(weight)).sum();
139        let data_shards = total_weight
140            .checked_sub(2 * usize::from(max_faulty))
141            .and_then(NonZeroUsize::new)?;
142        let dealer_weight = weights.0.get(usize::from(dealer)).copied().unwrap_or(0);
143        Some(Self {
144            weights,
145            batch_size: usize::from(dealer_weight) * usize::from(batch_size_per_weight),
146            data_shards,
147        })
148    }
149}

158fn avid_shard_bytes(row_weight: usize, batch_size: usize, data_shards: NonZeroUsize) -> usize {
159    let share_batch = uleb128_len(batch_size) + (batch_size + 1) * SCALAR;
160    let payload = uleb128_len(row_weight) + row_weight * share_batch;
161    (RS_LENGTH_PREFIX + payload).div_ceil(RS_ELEMENT * data_shards.get()) * RS_ELEMENT
162}
163
164fn uleb128_len(value: usize) -> usize {
165    (usize::BITS - value.leading_zeros()).div_ceil(7).max(1) as usize
166}

174pub fn check_rotation_messages(
175    bytes: &[u8],
176    shape: AvssShape,
177    owned: &[ShareIndex],
178) -> CountCheckResult<BTreeMap<ShareIndex, Range<usize>>> {
179    const FIELD: &str = "rotation_messages";
180    let mut cursor = Cursor::new(bytes);
181    let mut entries = BTreeMap::new();
182    walk_u16_map(&mut cursor, FIELD, owned.len(), |cursor, index| {
183        let share_index = ShareIndex::new(index)
184            .filter(|share_index| owned.contains(share_index))
185            .ok_or(CountCheckError::UnownedKey {
186                field: FIELD,
187                index,
188            })?;
189        let start = cursor.position;
190        walk_avss_message(cursor, shape)?;
191        entries.insert(share_index, start..cursor.position);
192        Ok(())
193    })?;
194    cursor.finish()?;
195    Ok(entries)
196}

233pub fn check_dispersal(
234    bytes: &[u8],
235    shape: DispersalShape<'_>,
236    holder: u16,
237) -> CountCheckResult<()> {
238    const FIELD: &str = "dispersal";
239    let shards = shape.weights.weight_of(FIELD, holder)?;
240    let row_depth = merkle_depth(shape.weights.num_nodes());
241    let top_depth = merkle_depth(Cursor::new(bytes).length(FIELD)?);
242    let mut cursor = Cursor::new(bytes);
243    walk_u16_map(
244        &mut cursor,
245        FIELD,
246        shape.weights.num_nodes(),
247        |cursor, row| {
248            let row_weight = shape.weights.weight_of(FIELD, row)?;
249            let exact = ExactShards {
250                shard_bytes: avid_shard_bytes(row_weight, shape.batch_size, shape.data_shards),
251                row_depth,
252                top_depth,
253            };
254            walk_authenticated_shards(cursor, shards, Some(exact))
255        },
256    )?;
257    cursor.finish()
258}

279pub fn check_protocol_complaint(bytes: &[u8], weights: PartyWeights<'_>) -> CountCheckResult<()> {
280    const FIELD: &str = "complaint";
281    let mut cursor = Cursor::new(bytes);
282    match cursor.variant(FIELD)? {
283        PROTOCOL_COMPLAINT_AVSS => {
284            cursor.take(FIELD, RECOVERY_PACKAGE)?;
285        }
286        PROTOCOL_COMPLAINT_AVID_REVEAL => {
287            walk_bytes(&mut cursor, "complaint.ciphertext")?;
288            cursor.take("complaint.proof", RECOVERY_PACKAGE)?;
289        }
290        PROTOCOL_COMPLAINT_AVID_BLAME => {
291            walk_u16_map(
292                &mut cursor,
293                "complaint.shards",
294                weights.num_nodes(),
295                |cursor, disperser| {
296                    let shards = weights.weight_of("complaint.shards", disperser)?;
297                    walk_authenticated_shards(cursor, shards, None)
298                },
299            )?;
300            walk_vote_certificate(&mut cursor, None)?;
301        }
302        variant => {
303            return Err(CountCheckError::BadVariant {
304                field: FIELD,
305                variant,
306            });
307        }
308    }
309    cursor.finish()
310}

319pub fn check_held_avid_echoes(
320    bytes: &[u8],
321    weights: PartyWeights<'_>,
322    holder: u16,
323) -> CountCheckResult<()> {
324    const FIELD: &str = "held_echoes";
325    let shards = weights.weight_of(FIELD, holder)?;
326    let mut cursor = Cursor::new(bytes);
327    walk_avid_vote(&mut cursor, weights)?;
328    let echoes = cursor.length(FIELD)?;
329    at_most(FIELD, echoes, weights.num_nodes())?;
330    for _ in 0..echoes {
331        cursor.take(FIELD, ADDRESS)?;
332        cursor.expect_variant(FIELD, MESSAGES_NONCE_GENERATION_AVID)?;
333        cursor.take(FIELD, BATCH_INDEX)?;
334        cursor.expect_variant(FIELD, AVID_NONCE_MESSAGE_KIND_ECHO)?;
335        cursor.take(FIELD, ADDRESS)?;
336        walk_authenticated_shards(&mut cursor, shards, None)?;
337    }
338    walk_vote_certificate(&mut cursor, None)?;
339    cursor.finish()
340}

342fn walk_avss_message(cursor: &mut Cursor<'_>, shape: AvssShape) -> CountCheckResult<()> {
343    const COMMITMENT: &str = "feldman_commitment";
344    const ENCS: &str = "ciphertext.encs";
345    let points = cursor.length(COMMITMENT)?;
346    expect_count(COMMITMENT, points, shape.threshold.into())?;
347    cursor.skip_elements(COMMITMENT, points, POINT)?;
348    cursor.take("ciphertext", 2 * EG_POINT)?;
349    let encs = cursor.length(ENCS)?;
350    expect_count(ENCS, encs, shape.num_nodes.into())?;
351    for _ in 0..encs {
352        walk_bytes(cursor, ENCS)?;
353    }
354    cursor.take("ciphertext.proof", DDH_NIZK)?;
355    Ok(())
356}

358fn walk_avss_common_message(
359    cursor: &mut Cursor<'_>,
360    shape: AvidShape<'_>,
361    dealer: u16,
362) -> CountCheckResult<()> {
363    const KEYS: &str = "full_public_keys";
364    const HASHES: &str = "ciphertext_hashes";
365    const POLYNOMIAL: &str = "response_polynomial";
366    let batch_size =
367        shape.weights.weight_of(KEYS, dealer)? * usize::from(shape.batch_size_per_weight);
368    let keys = cursor.length(KEYS)?;
369    expect_count(KEYS, keys, batch_size)?;
370    cursor.skip_elements(KEYS, keys, POINT)?;
371    cursor.take("blinding_commit", POINT)?;
372    cursor.take("ciphertext_shared", SHARED_COMPONENTS)?;
373    let hashes = cursor.length(HASHES)?;
374    expect_count(HASHES, hashes, shape.weights.num_nodes())?;
375    cursor.skip_elements(HASHES, hashes, HASH)?;
376    let coefficients = cursor.length(POLYNOMIAL)?;
377    expect_count(POLYNOMIAL, coefficients, shape.threshold.into())?;
378    cursor.skip_elements(POLYNOMIAL, coefficients, SCALAR)
379}

381fn walk_authenticated_shards(
382    cursor: &mut Cursor<'_>,
383    expected_shards: usize,
384    exact: Option<ExactShards>,
385) -> CountCheckResult<()> {
386    const SHARDS: &str = "shards";
387    let shards = cursor.length(SHARDS)?;
388    expect_count(SHARDS, shards, expected_shards)?;
389    for _ in 0..shards {
390        let len = cursor.length(SHARDS)?;
391        exact.map_or(Ok(()), |exact| expect_bytes(SHARDS, len, exact.shard_bytes))?;
392        cursor.take(SHARDS, len)?;
393    }
394    walk_merkle_path(cursor, "row_proof", exact.map(|exact| exact.row_depth))?;
395    walk_merkle_path(cursor, "top_proof", exact.map(|exact| exact.top_depth))
396}

398fn walk_merkle_path(
399    cursor: &mut Cursor<'_>,
400    field: &'static str,
401    depth: Option<usize>,
402) -> CountCheckResult<()> {
403    let nodes = cursor.length(field)?;
404    depth.map_or_else(
405        || at_most(field, nodes, MAX_MERKLE_PATH),
406        |depth| expect_count(field, nodes, depth),
407    )?;
408    (0..nodes).try_for_each(|_| walk_merkle_node(cursor, field))
409}
410
411fn merkle_depth(leaves: usize) -> usize {
412    leaves.next_power_of_two().trailing_zeros() as usize
413}
414
415fn walk_merkle_node(cursor: &mut Cursor<'_>, field: &'static str) -> CountCheckResult<()> {
416    match cursor.variant(field)? {
417        0 => Ok(()),
418        1 => cursor.take(field, MERKLE_DIGEST).map(drop),
419        variant => Err(CountCheckError::BadVariant { field, variant }),
420    }
421}

423fn walk_avid_vote(cursor: &mut Cursor<'_>, weights: PartyWeights<'_>) -> CountCheckResult<()> {
424    const RECIPIENTS: &str = "vote.recipients";
425    walk_merkle_node(cursor, "vote.top_root")?;
426    let recipients = cursor.length(RECIPIENTS)?;
427    at_most(RECIPIENTS, recipients, weights.num_nodes())?;
428    cursor.skip_elements(RECIPIENTS, recipients, PARTY_ID)?;
429    cursor.take("common_message_hash", HASH).map(drop)
430}

432fn walk_vote_certificate(
433    cursor: &mut Cursor<'_>,
434    max_signers_bytes: Option<usize>,
435) -> CountCheckResult<()> {
436    const FIELD: &str = "certificate";
437    const SIGNERS: &str = "certificate.signers";
438    cursor.take(FIELD, EPOCH + AGGREGATE_SIGNATURE)?;
439    let signers = cursor.length(SIGNERS)?;
440    max_signers_bytes.map_or(Ok(()), |max| at_most_bytes(SIGNERS, signers, max))?;
441    cursor.take(SIGNERS, signers)?;
442    cursor.take(FIELD, ADDRESS)?;
443    walk_bytes(cursor, "certificate.messages_hash")?;
444    cursor.take(FIELD, BATCH_INDEX).map(drop)
445}

447fn walk_u16_map<'a>(
448    cursor: &mut Cursor<'a>,
449    field: &'static str,
450    max_entries: usize,
451    mut entry: impl FnMut(&mut Cursor<'a>, u16) -> CountCheckResult<()>,
452) -> CountCheckResult<()> {
453    let count = cursor.length(field)?;
454    at_most(field, count, max_entries)?;
455    let mut previous_key: Option<[u8; 2]> = None;
456    for _ in 0..count {
457        let key = cursor.array::<2>(field)?;
458        if previous_key.is_some_and(|previous| key <= previous) {
459            return Err(CountCheckError::KeyOrder { field });
460        }
461        previous_key = Some(key);
462        entry(cursor, u16::from_le_bytes(key))?;
463    }
464    Ok(())
465}

467fn walk_bytes(cursor: &mut Cursor<'_>, field: &'static str) -> CountCheckResult<()> {
468    let len = cursor.length(field)?;
469    cursor.take(field, len).map(drop)
470}

472fn expect_count(field: &'static str, found: usize, expected: usize) -> CountCheckResult<()> {
473    (found == expected)
474        .then_some(())
475        .ok_or(CountCheckError::WrongCount {
476            field,
477            found,
478            expected,
479        })
480}
481
482fn at_most(field: &'static str, found: usize, max: usize) -> CountCheckResult<()> {
483    (found <= max)
484        .then_some(())
485        .ok_or(CountCheckError::TooMany { field, found, max })
486}
487
488fn expect_bytes(field: &'static str, found: usize, expected: usize) -> CountCheckResult<()> {
489    (found == expected)
490        .then_some(())
491        .ok_or(CountCheckError::WrongBytes {
492            field,
493            found,
494            expected,
495        })
496}
497
498fn at_most_bytes(field: &'static str, found: usize, max: usize) -> CountCheckResult<()> {
499    (found <= max)
500        .then_some(())
501        .ok_or(CountCheckError::TooManyBytes { field, found, max })
502}

504struct Cursor<'a> {
505    bytes: &'a [u8],
506    position: usize,
507}
508
509impl<'a> Cursor<'a> {
510    fn new(bytes: &'a [u8]) -> Self {
511        Self { bytes, position: 0 }
512    }
513
514    fn take(&mut self, field: &'static str, len: usize) -> CountCheckResult<&'a [u8]> {
515        let bytes = self.bytes;
516        let taken = self
517            .position
518            .checked_add(len)
519            .and_then(|end| bytes.get(self.position..end))
520            .ok_or(CountCheckError::Truncated { field })?;
521        self.position += len;
522        Ok(taken)
523    }
524
525    fn array<const N: usize>(&mut self, field: &'static str) -> CountCheckResult<[u8; N]> {
526        self.take(field, N)?
527            .try_into()
528            .map_err(|_| CountCheckError::Truncated { field })
529    }
530
531    fn skip_elements(
532        &mut self,
533        field: &'static str,
534        count: usize,
535        size: usize,
536    ) -> CountCheckResult<()> {
537        let len = count
538            .checked_mul(size)
539            .ok_or(CountCheckError::Truncated { field })?;
540        self.take(field, len).map(drop)
541    }
542
543    fn length(&mut self, field: &'static str) -> CountCheckResult<usize> {
544        let mut value: u64 = 0;
545        for shift in (0..32).step_by(7) {
546            let [byte] = self.array::<1>(field)?;
547            let digit = byte & 0x7f;
548            value |= u64::from(digit) << shift;
549            if digit == byte {
550                if shift > 0 && digit == 0 {
551                    return Err(CountCheckError::BadLength { field });
552                }
553                return usize::try_from(value)
554                    .ok()
555                    .filter(|&len| len <= bcs::MAX_SEQUENCE_LENGTH)
556                    .ok_or(CountCheckError::BadLength { field });
557            }
558        }
559        Err(CountCheckError::BadLength { field })
560    }
561
562    fn variant(&mut self, field: &'static str) -> CountCheckResult<usize> {
563        self.length(field)
564    }
565
566    fn expect_variant(&mut self, field: &'static str, expected: usize) -> CountCheckResult<()> {
567        match self.variant(field)? {
568            variant if variant == expected => Ok(()),
569            variant => Err(CountCheckError::BadVariant { field, variant }),
570        }
571    }
572
573    fn finish(self) -> CountCheckResult<()> {
574        match self.bytes.len().saturating_sub(self.position) {
575            0 => Ok(()),
576            trailing => Err(CountCheckError::Trailing(trailing)),
577        }
578    }
579}
