4use std::collections::BTreeMap;
5use std::sync::Arc;
6use std::sync::RwLock;
7
8use hashi_types::proto;
9
10use crate::mpc::types::GetPublicMpcOutputResponse;
11use crate::mpc::types::MpcError;
12use crate::mpc::types::MpcOutput;
13use crate::mpc::types::MpcResult;
14use crate::mpc::types::PublicMpcOutput;
15
16#[derive(Default)]
17pub(crate) struct PublicOutputs(RwLock<BTreeMap<u64, Arc<proto::GetPublicMpcOutputResponse>>>);
18
19impl PublicOutputs {
20    pub(crate) fn lookup(
21        &self,
22        epoch: u64,
23        current_epoch: u64,
24        previous_epoch: u64,
25    ) -> MpcResult<Arc<proto::GetPublicMpcOutputResponse>> {
26        if epoch != current_epoch && epoch != previous_epoch {
27            return Err(MpcError::NotFound(format!(
28                "no DKG output for epoch {epoch} (current epoch is {current_epoch})"
29            )));
30        }
31        self.0.read().unwrap().get(&epoch).cloned().ok_or_else(|| {
32            MpcError::NotFound(format!("DKG output for epoch {epoch} not yet available"))
33        })
34    }
35
36    #[cfg(test)]
37    pub(crate) fn epochs(&self) -> Vec<u64> {
38        self.0.read().unwrap().keys().copied().collect()
39    }
40
41    fn publish(&self, epoch: u64, output: &MpcOutput) {
42        let response = Arc::new(proto::GetPublicMpcOutputResponse::from(
43            &GetPublicMpcOutputResponse {
44                output: PublicMpcOutput::from_mpc_output(output),
45            },
46        ));
47        let replaced = self.0.write().unwrap().insert(epoch, Arc::clone(&response));
48        if replaced.is_some_and(|replaced| replaced != response) {
49            tracing::warn!("Public MPC output for epoch {epoch} replaced by a different one");
50        }
51    }
52
53    fn retain(&self, epoch: u64, previous_epoch: Option<u64>) {
54        let _dropped = {
55            let mut published = self.0.write().unwrap();
56            let kept = [Some(epoch), previous_epoch]
57                .into_iter()
58                .flatten()
59                .filter_map(|wanted| published.remove_entry(&wanted))
60                .collect();
61            std::mem::replace(&mut *published, kept)
62        };
63    }
64}
65
66#[derive(Default)]
67pub(crate) struct MpcOutputs {
68    current: Option<MpcOutput>,
69    previous: Option<MpcOutput>,
70    public: Arc<PublicOutputs>,
71}
72
73impl MpcOutputs {
74    pub(crate) fn current(&self) -> Option<&MpcOutput> {
75        self.current.as_ref()
76    }
77
78    pub(crate) fn previous(&self) -> Option<&MpcOutput> {
79        self.previous.as_ref()
80    }
81
82    pub(crate) fn public(&self) -> &PublicOutputs {
83        &self.public
84    }
85
86    pub(crate) fn set_current(&mut self, epoch: u64, output: MpcOutput) {
87        self.public.publish(epoch, &output);
88        self.current = Some(output);
89    }
90
91    pub(crate) fn set_previous(&mut self, epoch: Option<u64>, output: MpcOutput) {
92        if let Some(epoch) = epoch {
93            self.public.publish(epoch, &output);
94        }
95        self.previous = Some(output);
96    }
97
98    pub(crate) fn adopt(
99        &mut self,
100        public: &Arc<PublicOutputs>,
101        epoch: u64,
102        previous_epoch: Option<u64>,
103    ) {
104        public.retain(epoch, previous_epoch);
105        self.public = Arc::clone(public);
106        if let Some(output) = &self.current {
107            self.public.publish(epoch, output);
108        }
109        if let Some((previous_epoch, output)) = previous_epoch.zip(self.previous.as_ref()) {
110            self.public.publish(previous_epoch, output);
111        }
112    }
113}
