4use std::sync::LockResult;
5use std::sync::RwLock;
6use std::sync::RwLockReadGuard;
7use std::sync::RwLockWriteGuard;
8use std::sync::TryLockError;
9use std::sync::TryLockResult;
10
11use parking_lot::MutexGuard;
12
13pub struct FairRwLock<T> {
14    turnstile: parking_lot::Mutex<()>,
15    lock: RwLock<T>,
16}
17
18impl<T> FairRwLock<T> {
19    pub fn new(value: T) -> Self {
20        Self {
21            turnstile: parking_lot::Mutex::new(()),
22            lock: RwLock::new(value),
23        }
24    }
25
26    pub fn into_inner(self) -> LockResult<T> {
27        self.lock.into_inner()
28    }
29
30    pub fn read(&self) -> LockResult<RwLockReadGuard<'_, T>> {
31        let turn = self.turnstile.lock();
32        let guard = self.lock.read();
33        MutexGuard::unlock_fair(turn);
34        guard
35    }
36
37    pub fn write(&self) -> LockResult<RwLockWriteGuard<'_, T>> {
38        let turn = self.turnstile.lock();
39        let guard = self.lock.write();
40        MutexGuard::unlock_fair(turn);
41        guard
42    }
43
44    pub fn try_write(&self) -> TryLockResult<RwLockWriteGuard<'_, T>> {
45        let turn = self.turnstile.try_lock().ok_or(TryLockError::WouldBlock)?;
46        let guard = self.lock.try_write();
47        MutexGuard::unlock_fair(turn);
48        guard
49    }
50
51    #[cfg(test)]
52    pub(crate) fn turnstile_held(&self) -> bool {
53        self.turnstile.is_locked()
54    }
55}
56
57#[cfg(test)]
58mod tests {
59    use std::sync::Arc;
60    use std::sync::Mutex;
61    use std::time::Duration;
62    use std::time::Instant;
63
64    use super::*;
65
66    #[test]
67    fn a_reader_queued_behind_a_writer_goes_before_a_later_writer() {
68        let lock = Arc::new(FairRwLock::new(()));
69        let order = Arc::new(Mutex::new(Vec::new()));
70        let first = lock.write().unwrap();
71        let reader = {
72            let (lock, order) = (lock.clone(), order.clone());
73            std::thread::spawn(move || {
74                let _guard = lock.read().unwrap();
75                order.lock().unwrap().push("reader");
76            })
77        };
78        let deadline = Instant::now() + Duration::from_secs(10);
79        while !lock.turnstile_held() {
80            assert!(Instant::now() < deadline, "the reader never queued");
81            std::thread::sleep(Duration::from_millis(1));
82        }
83        let writer = {
84            let (lock, order) = (lock.clone(), order.clone());
85            std::thread::spawn(move || {
86                let _guard = lock.write().unwrap();
87                order.lock().unwrap().push("writer");
88            })
89        };
90        std::thread::sleep(Duration::from_millis(50));
91        drop(first);
92        reader.join().unwrap();
93        writer.join().unwrap();
94
95        assert_eq!(*order.lock().unwrap(), ["reader", "writer"]);
96    }
97}
