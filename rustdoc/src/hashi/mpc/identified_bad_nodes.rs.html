4use std::collections::BTreeMap;
5use std::collections::BTreeSet;
6use std::sync::Mutex;
7
8use sui_sdk_types::Address;
9
10use crate::mpc::types::ComplaintRefusal;
11use crate::mpc::types::ProtocolTypeIndicator;
12
13#[derive(Default)]
14pub(crate) struct IdentifiedBadNodes(Mutex<BTreeMap<u64, BTreeSet<Address>>>);
15
16impl IdentifiedBadNodes {
17    pub(crate) fn complaint_refusal(
18        &self,
19        protocol: ProtocolTypeIndicator,
20        epoch: u64,
21        caller: &Address,
22        dealer: &Address,
23    ) -> Option<ComplaintRefusal> {
24        if protocol != ProtocolTypeIndicator::NonceGeneration {
25            return None;
26        }
27        if caller == dealer {
28            return Some(ComplaintRefusal::OwnDealing);
29        }
30        self.contains(epoch, caller)
31            .then_some(ComplaintRefusal::Flagged)
32    }
33
34    pub(crate) fn contains(&self, epoch: u64, caller: &Address) -> bool {
35        self.0
36            .lock()
37            .unwrap()
38            .get(&epoch)
39            .is_some_and(|flagged| flagged.contains(caller))
40    }
41
42    pub(crate) fn insert(&self, epoch: u64, caller: Address) {
43        let mut flagged = self.0.lock().unwrap();
44        flagged.retain(|&flagged_epoch, _| flagged_epoch >= epoch);
45        flagged.entry(epoch).or_default().insert(caller);
46    }
47}
