102        let mut rx = self.key_ready_rx.clone();
103        loop {
104            {
105                let value = rx.borrow();
106                if let Some(pk) = value.as_ref() {
107                    return *pk;
108                }
109            }
110            if rx.changed().await.is_err() {
111                std::future::pending().await
112            }
113        }

149fn walk_step(size: Option<usize>, batch_start: u64, num_consumed: u64) -> WalkStep {
150    match size {
151        Some(size) if num_consumed < batch_start + size as u64 => WalkStep::Last(size),

198    #[tracing::instrument(name = "mpc_service", skip_all)]
199    async fn run(mut self) {
200        let pending = self.get_pending_epoch_change();
201        let is_in_committee = self.inner.is_in_current_committee();
202        info!(
203            "MPC service starting: pending_epoch_change={pending:?}, \
204             is_in_current_committee={is_in_committee}",
205        );
206        self.run_major_compaction(self.inner.onchain_state().epoch())
207            .await;
208        let mut notifications = self.inner.onchain_state().subscribe();
209        if let Some(epoch) = pending {
210            self.drive_reconfig(epoch).await;
211        } else if self.is_awaiting_genesis() {
212            // No committee has been formed yet (epoch 0, no committee for epoch 0).
213            // Wait for enough validators to register then trigger genesis reconfig.
214            info!("No initial committee yet; waiting for enough validators to register...");
215            self.try_submit_genesis_reconfig().await;
216        } else if self.inner.is_in_current_committee() {
217            loop {
218                self.inner.metrics.task_heartbeat("mpc_service");
219                if let Some(epoch) = self.get_pending_epoch_change() {
220                    self.drive_reconfig(epoch).await;
221                    continue;
222                }
223                self.sync_if_stale().await;
224                let epoch = self.inner.onchain_state().epoch();
225                if self.inner.signing_manager_for(epoch).is_some() {
226                    break;
227                }
228                tokio::time::sleep(RETRY_INTERVAL).await;
229            }
230        } else {
231            info!("Node is not in the current committee, waiting for reconfig notification...");
232        }
233        // Hashi lagging Sui with nothing pending is what a manual abort, or a
234        // boundary missed while this node was down, leaves behind; catch up
235        // now rather than wait for the next boundary.
236        self.try_submit_start_reconfig(self.inner.onchain_state().latest_checkpoint_epoch())
237            .await;
238
239        let mut checkpoint_rx = self.inner.onchain_state().subscribe_checkpoint();
240        let mut reconcile_tick = tokio::time::interval(RECONCILE_TICK);
241        reconcile_tick.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
242        loop {
243            self.inner.metrics.task_heartbeat("mpc_service");
244            // Re-sample the presig pool every iteration (at least each
245            // RECONCILE_TICK): the signing-time update in
246            // `forward_signing_results` alone leaves the gauge frozen — at 0
247            // after a restart — whenever no withdrawals are flowing.
248            if let Some(manager) = self.inner.current_signing_manager() {
249                self.inner
250                    .metrics
251                    .presig_pool_remaining
252                    .set(manager.presignatures_remaining() as i64);
253            }
254            // Check for pending reconfig before blocking on `recv()`.
255            if let Some(epoch) = self.get_pending_epoch_change() {
256                self.drive_reconfig(epoch).await;
257                continue;
258            }
259            tokio::select! {
260                notification = notifications.recv() => {
261                    match notification {
262                        Ok(notification) => match notification {
263                            Notification::StartReconfig(epoch) => {
264                                self.drive_reconfig(epoch).await;
265                            }
266                            Notification::SuiEpochChanged(sui_epoch) => {
267                                self.try_submit_start_reconfig(sui_epoch).await;
268                            }
269                            Notification::ReconfigAborted(epoch) => {
270                                // Whoever tore it down, Hashi now lags Sui;
271                                // start the replacement rather than wait for
272                                // the next boundary.
273                                warn!("reconfiguration to epoch {epoch} was aborted on chain");
274                                let sui_epoch =
275                                    self.inner.onchain_state().latest_checkpoint_epoch();
276                                self.try_submit_start_reconfig(sui_epoch).await;
277                            }
278                            _ => {}
279                        },
280                        Err(e) => {
281                            error!("MPC notification recv error: {e:?}, resubscribing");
282                            notifications = self.inner.onchain_state().subscribe();
283                        }
284                    }
285                }
286                Ok(()) = checkpoint_rx.changed() => {
287                    self.sync_if_stale().await;
288                }
289                _ = reconcile_tick.tick() => {
290                    self.sync_if_stale().await;
291                }
292                Ok(()) = self.refill_rx.changed() => {
293                    let next_batch = *self.refill_rx.borrow();
294                    self.refill_with_retries(next_batch).await;
295                }
296            }
297        }
298    }

300    async fn refill_with_retries(&self, next_batch: u32) {
301        for attempt in 1..=MAX_PROTOCOL_ATTEMPTS {
302            if let Err(e) = self.bail_if_reconfig_pending() {
303                info!("presignature refill stopped: {e}");
304                return;
305            }
306            match self.refill_presignatures(next_batch).await {
307                Ok(()) => break,
308                Err(e) => {
309                    error!(
310                        "Presignature refill attempt {attempt}/{MAX_PROTOCOL_ATTEMPTS} failed: {e}"
311                    );
312                    if attempt < MAX_PROTOCOL_ATTEMPTS {
313                        tokio::time::sleep(RETRY_INTERVAL).await;
314                    }
315                }
316            }
317        }
318    }

320    async fn post_reconfig_housekeeping(
321        &self,
322        target_epoch: u64,
323        epoch_change_confirmed: bool,
324        backup: Backup,
325    ) {
326        let current_epoch = self.inner.onchain_state().epoch();
327        let tombstoned = if epoch_change_confirmed || current_epoch >= target_epoch {
328            let pruning_references = {
329                let state = self.inner.onchain_state().state();
330                build_pruning_references(&state.hashi().committees, target_epoch)
331            };
332            match self
333                .inner
334                .db
335                .prune_messages_below(target_epoch, &pruning_references)
336            {
337                Ok(deleted) => deleted > 0,
338                Err(e) => {
339                    error!("Failed to prune old MPC messages below epoch {target_epoch}: {e}");
340                    true
341                }
342            }
343        } else {
344            info!(
345                "handle_reconfig: still on epoch {current_epoch}; skipping the prune for \
346                 epoch {target_epoch}"
347            );
348            false
349        };
350        if tombstoned {
351            self.run_major_compaction(target_epoch).await;
352        }
353        self.backup_handle
354            .maintain_backups_after_epoch_change(target_epoch, backup == Backup::Write);
355    }

383    /// Advance the pending reconfiguration to `epoch`: run it while its Sui
384    /// epoch window is open, then reconcile with Sui's epoch, which aborts
385    /// it if the window closed without `end_reconfig` landing and starts a
386    /// replacement whenever Hashi lags Sui with nothing pending (the abort
387    /// path, a manual abort observed from inside `handle_reconfig`, or a
388    /// completion that itself overran the boundary).
389    async fn drive_reconfig(&self, epoch: u64) {
390        if !self.reconfig_window_closed(epoch) {
391            info!("Entering handle_reconfig for epoch {epoch}");
392            self.handle_reconfig(epoch).await;
393        }
394        let sui_epoch = self.inner.onchain_state().latest_checkpoint_epoch();
395        self.try_submit_start_reconfig(sui_epoch).await;
396    }

398    /// Submit `abort_reconfig` for a pending reconfiguration whose Sui epoch
399    /// window has closed, then wait (bounded) for the mirror to reflect it.
400    /// Any node may do this and several usually race to; the chain settles
401    /// the race, so "nothing pending" or "wrong epoch" back from it means
402    /// another node got there first.
403    async fn abort_stale_reconfig(&self, epoch: u64) {
404        warn!(
405            "reconfiguration to epoch {epoch} overran its Sui epoch window without completing; \
406             submitting abort_reconfig"
407        );
408        let result = async {
409            let mut executor =
410                crate::sui_tx_executor::SuiTxExecutor::from_hashi(self.inner.clone())?;
411            executor.execute_abort_reconfig(epoch).await
412        };
413        match result.await {
414            Ok(()) => info!("abort_reconfig for epoch {epoch} landed"),
415            Err(e) => match classify_abort_submission_error(&e) {
416                AbortSubmissionErrorKind::AlreadyResolved => {
417                    info!("abort_reconfig for epoch {epoch}: already resolved on chain: {e:#}");
418                }
419                AbortSubmissionErrorKind::StillCurrent => {
420                    // Our checkpoint view of Sui's epoch ran ahead of the
421                    // chain's, which should not happen; let the chain win.
422                    warn!(
423                        "abort_reconfig for epoch {epoch} refused: the chain still considers \
424                         its Sui epoch current: {e:#}"
425                    );
426                    tokio::time::sleep(RETRY_INTERVAL).await;
427                    return;
428                }
429                AbortSubmissionErrorKind::Other => {
430                    warn!("abort_reconfig for epoch {epoch} failed: {e:#}; retrying later");
431                    tokio::time::sleep(RETRY_INTERVAL).await;
432                    return;
433                }
434            },
435        }
436        self.wait_for_pending_clear_visibility(epoch).await;
437    }

439    /// Wait (bounded) for the object mirror to reflect that `epoch` is no
440    /// longer pending: another node's `end_reconfig` activated it, or an
441    /// abort tore it down. The lossless watcher applies the winning
442    /// transaction as a root object write within a few checkpoints; clock
443    /// ticks pace the re-checks.
444    async fn wait_for_pending_clear_visibility(&self, epoch: u64) {
445        const VISIBILITY_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(60);
446        let mut checkpoint_rx = self.inner.onchain_state().subscribe_checkpoint();
447        let _ = tokio::time::timeout(VISIBILITY_TIMEOUT, async {
448            while self.get_pending_epoch_change() == Some(epoch) {
449                if checkpoint_rx.changed().await.is_err() {
450                    break;
451                }
452            }
453        })
454        .await;
455    }

470    /// Wait for enough validators to register, then submit `start_reconfig`
471    /// to form the initial committee. Blocks until a pending epoch change
472    /// appears (either from our own submission or another node's).
473    async fn try_submit_genesis_reconfig(&self) {
474        loop {
475            if self.get_pending_epoch_change().is_some() {
476                return;
477            }
478            match self.inner.next_reconfig_epoch().await {
479                Ok(target) => {
480                    if let Err(e) = self.inner.prepare_and_register_keys(target).await {
481                        debug!(
482                            "Encryption/signing key registration for epoch {target} failed: {e}; \
483                             will retry on next genesis_reconfig iteration"
484                        );
485                    }
486                }
487                Err(e) => debug!("Failed to compute next_reconfig_epoch: {e}"),
488            }
489            // Attempt to submit start_reconfig. This will fail on-chain until
490            // the publisher sends finish_publish (the launch switch).
491            let result = async {
492                let mut executor =
493                    crate::sui_tx_executor::SuiTxExecutor::from_hashi(self.inner.clone())?;
494                executor.execute_start_reconfig().await
495            };
496            match result.await {
497                Ok(()) => {
498                    info!("Genesis start_reconfig submitted successfully");
499                    return;
500                }
501                Err(e) => {
502                    debug!("Genesis start_reconfig not yet possible: {e}");
503                    // Poll for pending epoch change while waiting, in case
504                    // another node submitted start_reconfig.
505                    let polls = (RETRY_INTERVAL.as_millis()
506                        / START_RECONFIG_POLL_INTERVAL.as_millis())
507                        as u32;
508                    for _ in 0..polls {
509                        if self.get_pending_epoch_change().is_some() {
510                            return;
511                        }
512                        tokio::time::sleep(START_RECONFIG_POLL_INTERVAL).await;
513                    }
514                }
515            }
516        }
517    }

519    async fn recover_mpc_state(&self) -> anyhow::Result<MpcOutput> {
520        let onchain_state = self.inner.onchain_state().clone();
521        let epoch = onchain_state.epoch();
522        let is_key_rotation = onchain_state.is_key_rotation_epoch(epoch);
523        let onchain_mpc_key = onchain_state.mpc_public_key();
524        info!(
525            "recover_mpc_state: epoch={epoch}, is_key_rotation={is_key_rotation}, \
526             onchain_mpc_key_len={}",
527            onchain_mpc_key.len(),
528        );
529        let output = if is_key_rotation {
530            self.recover_current_rotation(epoch, &onchain_mpc_key).await
531        } else {
532            self.recover_current_dkg(epoch, &onchain_mpc_key).await
533        }?;
534        info!(
535            "recover_mpc_state: recovered vk={}",
536            hex::encode(output.public_key.to_byte_array())
537        );
538        Ok(output)
539    }

541    async fn recover_current_dkg(
542        &self,
543        epoch: u64,
544        onchain_mpc_key: &[u8],
545    ) -> anyhow::Result<MpcOutput> {
546        self.setup_initial_dkg(epoch)?;
547        let onchain_state = self.inner.onchain_state().clone();
548        let certs: Vec<CertificateV1> =
549            tob_certificates(&onchain_state, epoch, None, move_types::ProtocolType::Dkg)
550                .map_err(|e| anyhow::anyhow!("failed to read DKG certs for epoch {epoch}: {e}"))?
551                .into_iter()
552                .map(|(_, cert)| cert)
553                .collect();
554        let mpc_manager = self
555            .inner
556            .mpc_manager()
557            .ok_or_else(|| anyhow::anyhow!("MpcManager not initialized for DKG recovery"))?;
558        let certs = verify_fetched_certificates(&mpc_manager, certs, &self.inner.metrics).await;
559        match MpcManager::reconstruct_current_dkg_output(&mpc_manager, &certs, onchain_mpc_key) {
560            MpcOutputRecoveryOutcome::Recovered(output) => {
561                info!(
562                    "recover_current_dkg: recovered current epoch {epoch} from local DKG \
563                     messages (no peers)"
564                );
565                Ok(output)
566            }
567            MpcOutputRecoveryOutcome::NotApplicable => match self.run_dkg(epoch).await? {
568                ReconfigOutcome::Output(output) => Ok(output),
569                other => Err(anyhow::anyhow!(
570                    "dkg for epoch {epoch} produced no output ({}): this node has no \
571                         share state to recover",
572                    other.label()
573                )),
574            },
575            MpcOutputRecoveryOutcome::Suspicious(reason) => {
576                error!(
577                    "recover_current_dkg: local DKG state for epoch {epoch} contradicts on-chain \
578                     truth ({reason}); observing this epoch, will recover at the next rotation"
579                );
580                self.inner.metrics.mpc_recovery_suspicious_total.inc();
581                Err(anyhow::anyhow!(
582                    "suspicious local DKG state for epoch {epoch}: {reason}"
583                ))
584            }
585        }
586    }

592    ) -> anyhow::Result<MpcOutput> {
593        self.setup_key_rotation(epoch)?;
594        let onchain_state = self.inner.onchain_state().clone();
595        let mpc_manager = self
596            .inner
597            .mpc_manager()
598            .ok_or_else(|| anyhow::anyhow!("MpcManager not initialized for rotation recovery"))?;
599        let previous_epoch = mpc_manager.read().unwrap().previous_epoch;
600        let current_certs: Vec<CertificateV1> = tob_certificates(
601            &onchain_state,
602            epoch,
603            None,
604            move_types::ProtocolType::KeyRotation,
605        )
606        .map_err(|e| anyhow::anyhow!("failed to read rotation certs for epoch {epoch}: {e}"))?
607        .into_iter()
608        .map(|(_, cert)| cert)
609        .collect();
610        let previous_certs: Vec<CertificateV1> =
611            key_generation_certificates(&onchain_state, previous_epoch)
612                .map_err(|e| {
613                    anyhow::anyhow!("failed to read certs for previous epoch {previous_epoch}: {e}")
614                })?
615                .into_iter()
616                .map(|(_, cert)| cert)
617                .collect();
618        let current_certs =
619            verify_fetched_certificates(&mpc_manager, current_certs, &self.inner.metrics).await;
620        let previous_certs =
621            verify_fetched_certificates(&mpc_manager, previous_certs, &self.inner.metrics).await;
622        match MpcManager::reconstruct_current_rotation_output(
623            &mpc_manager,
624            &current_certs,
625            &previous_certs,
626            onchain_mpc_key,
627        ) {
628            MpcOutputRecoveryOutcome::Recovered(output) => {
629                info!(
630                    "recover_current_rotation: recovered current epoch {epoch} and previous \
631                     {previous_epoch} from local rotation messages (no peers)"
632                );
633                Ok(output)
634            }
635            MpcOutputRecoveryOutcome::NotApplicable => match self.run_key_rotation(epoch).await? {
636                ReconfigOutcome::Output(output) => Ok(output),
637                other => Err(anyhow::anyhow!(
638                    "rotation for epoch {epoch} produced no output ({}): this node has no \
639                         share state to recover",
640                    other.label()
641                )),
642            },
643            MpcOutputRecoveryOutcome::Suspicious(reason) => {
644                error!(
645                    "recover_current_rotation: local rotation state for epoch {epoch} contradicts \
646                     on-chain truth ({reason}); observing this epoch, will recover at the next \
647                     rotation"
648                );
649                self.inner.metrics.mpc_recovery_suspicious_total.inc();
650                Err(anyhow::anyhow!(
651                    "suspicious local rotation state for epoch {epoch}: {reason}"
652                ))
653            }
654        }
655    }

690    async fn generate_presignatures(
691        &self,
692        epoch: u64,
693        batch_index: u32,
694    ) -> anyhow::Result<(Committee, Presignatures, u16)> {
695        let onchain_state = self.inner.onchain_state().clone();
696        let committee = onchain_state
697            .state()
698            .hashi()
699            .committees
700            .committees()
701            .get(&epoch)
702            .ok_or_else(|| anyhow::anyhow!("No committee found for epoch {}", epoch))?
703            .clone();
704        let mpc_manager = self
705            .inner
706            .mpc_manager()
707            .ok_or_else(|| anyhow::anyhow!("MpcManager not initialized"))?;
708        mpc_manager.read().unwrap().ensure_manager_epoch(epoch)?;
709        let signer = self.inner.config.operator_private_key()?;
710        let p2p_channel =
711            RpcP2PChannel::new(onchain_state.clone(), epoch, MPC_LABEL_NONCE_GENERATION);
712        let onchain_state_for_certs = onchain_state.clone();
713        let mut tob_channel = SuiTobSessionChannel::new(
714            self.inner.config.hashi_ids(),
715            onchain_state,
716            epoch,
717            Some(batch_index),
718            move_types::ProtocolType::NonceGeneration,
719            signer,
720        )
721        .with_idle_timeout(NONCE_RECEIVE_IDLE_TIMEOUT);
722        let metrics = &self.inner.metrics;
723        let _timer = metrics
724            .mpc_total_duration_seconds
725            .with_label_values(&[MPC_LABEL_NONCE_GENERATION])
726            .start_timer();
727        MpcManager::run_nonce_dealer_phase(
728            &mpc_manager,
729            batch_index,
730            &p2p_channel,
731            &mut tob_channel,
732            metrics,
733        )
734        .await;
735        let cert_wait_timer = metrics
736            .mpc_tob_poll_duration_seconds
737            .with_label_values(&[MPC_LABEL_NONCE_GENERATION])
738            .start_timer();
739        let (final_certs, settled_cutoff_ms) = Self::fetch_final_nonce_certs(
740            &onchain_state_for_certs,
741            &mpc_manager,
742            epoch,
743            batch_index,
744            true,
745            metrics,
746        )
747        .await?;
748        drop(cert_wait_timer);
749        let canonical = nonce_certificates(&final_certs, epoch, batch_index);
750        let admitted = {
751            let mgr = mpc_manager.read().unwrap();
752            mgr.avid_admitted_nonce_dealers(&canonical, settled_cutoff_ms)?
753        };
754        let served_weight = admitted.weight;
755        if !admitted.floor_reached() {
756            return Err(admitted.below_floor_error(batch_index, metrics).into());
757        }
758        let nonce_result = MpcManager::run_avid_nonce_party_phase(
759            &mpc_manager,
760            batch_index,
761            &p2p_channel,
762            &admitted,
763            Some((onchain_state_for_certs.clone(), epoch)),
764            metrics,
765        )
766        .await;
767        drop(_timer);
768        let outcome = nonce_result.map_err(|e| anyhow::anyhow!("Nonce generation failed: {e}"))?;
769        if outcome.local_skips > 0 {
770            metrics.mpc_nonce_local_skip_batches_total.inc();
771            anyhow::bail!(
772                "nonce batch {batch_index} for epoch {epoch} skipped {} dealer(s) for \
773                 node-local reasons; discarding so this node does not install a batch its peers did not build",
774                outcome.local_skips,
775            );
776        }
777        let dealer_count = outcome.outputs.len();
778        let (batch_size_per_weight, params) = {
779            let mgr = mpc_manager.read().unwrap();
780            (
781                mgr.batch_size_per_weight,
782                Parameters {
783                    t: mgr.mpc_config.threshold,
784                    f: mgr.mpc_config.max_faulty,
785                },
786            )
787        };
788        let _timer = metrics
789            .mpc_presig_conversion_duration_seconds
790            .with_label_values(&[MPC_LABEL_NONCE_GENERATION])
791            .start_timer();
792        let presignatures = Presignatures::new(
793            outcome.outputs,
794            batch_size_per_weight,
795            params,
796            USE_LEGACY_PRESIG_DERIVATION,
797        )
798        .map_err(|e| anyhow::anyhow!("Failed to create presignatures: {e}"))?;
799        drop(_timer);
800        let served_implies = presig_count(served_weight as usize, params, batch_size_per_weight);
801        if presignatures.len() != served_implies {
802            metrics.mpc_nonce_size_mismatch_total.inc();
803            anyhow::bail!(
804                "nonce batch {batch_index} for epoch {epoch}: built {} presigs but the \
805                 served certs size to {served_implies} (weight {served_weight}); a dealer \
806                 dealt for a weight other than its configured reduced weight, refusing \
807                 to install",
808                presignatures.len(),
809            );
810        }
811        metrics.mpc_nonce_batch_index.set(batch_index as i64);
812        metrics.mpc_nonce_batch_dealers.set(dealer_count as i64);
813        info!(
814            "nonce batch {batch_index} for epoch {epoch}: {} presigs from the admitted set",
815            presignatures.len(),
816        );
817        Ok((committee, presignatures, batch_size_per_weight))
818    }

881    async fn recover_presigning_state(&self, output: &MpcOutput) -> anyhow::Result<PresigRecovery> {
882        let (num_consumed, epoch, committee, pending) = {
883            let state = self.inner.onchain_state().state();
884            let hashi = state.hashi();
885            let num_consumed = hashi.num_consumed_presigs;
886            let epoch = hashi.committees.epoch();
887            let committee = hashi
888                .committees
889                .committees()
890                .get(&epoch)
891                .ok_or_else(|| anyhow::anyhow!("No committee found for epoch {epoch}"))?
892                .clone();
893            let mut pending: HashSet<u64> = HashSet::new();
894            for txn in hashi.bitcoin().withdrawal_queue.withdrawal_txns().values() {
895                if txn.signing_epoch() != epoch {
896                    continue;
897                }
898                for signature in &txn.signing.signatures {
899                    if let move_types::MpcSig::Pending(index) = signature {
900                        anyhow::ensure!(
901                            *index < num_consumed,
902                            "pending presig index {index} is not below allocation cursor {num_consumed}",
903                        );
904                        anyhow::ensure!(
905                            pending.insert(*index),
906                            "pending presig index {index} is assigned more than once",
907                        );
908                    }
909                }
910            }
911            (num_consumed, epoch, committee, pending)
912        };
913        let mpc_manager = self
914            .inner
915            .mpc_manager()
916            .ok_or_else(|| anyhow::anyhow!("MpcManager not initialized"))?;
917        let (batch_size_per_weight, params, floor) = {
918            let mgr = mpc_manager.read().unwrap();
919            (
920                mgr.batch_size_per_weight,
921                Parameters {
922                    t: mgr.mpc_config.threshold,
923                    f: mgr.mpc_config.max_faulty,
924                },
925                mgr.required_nonce_weight(),
926            )
927        };
928        mpc_manager.read().unwrap().ensure_manager_epoch(epoch)?;
929        let mut boundaries: Vec<(u32, u64, usize)> = Vec::new();
930        let mut batch_start = 0u64;
931        let mut batch_index = 0u32;
932        loop {
933            self.bail_if_reconfig_pending()?;
934            let size = self
935                .nonce_batch_size_from_certs(
936                    &mpc_manager,
937                    epoch,
938                    batch_index,
939                    params,
940                    batch_size_per_weight,
941                    floor,
942                )
943                .await?;
944            let (size, holds_cursor) = match walk_step(size, batch_start, num_consumed) {
945                WalkStep::Missing => {
946                    return Ok(PresigRecovery::MissingBatch {
947                        epoch,
948                        batch_index,
949                        batch_start,
950                    });
951                }
952                WalkStep::Stop => break,
953                WalkStep::Last(size) => (size, true),
954                WalkStep::Continue(size) => (size, false),
955            };
956            boundaries.push((batch_index, batch_start, size));
957            if holds_cursor {
958                break;
959            }
960            batch_start += size as u64;
961            batch_index += 1;
962        }
963        anyhow::ensure!(
964            !boundaries.is_empty(),
965            "no certified nonce batches for epoch {epoch}: batch 0 is not yet sizeable and \
966             the cursor has drawn nothing",
967        );
968        let recovered_end = boundaries
969            .last()
970            .map_or(0, |&(_, start, size)| start + size as u64);
971        let first_pending = boundaries
972            .iter()
973            .position(|&(_, start, size)| {
974                let end = start + size as u64;
975                pending.iter().any(|&p| p >= start && p < end)
976            })
977            .unwrap_or_else(|| boundaries.len().saturating_sub(1));
978        let mut retained: Vec<(Presignatures, u32, u64)> = Vec::new();
979        // TODO(IOP-529): Avoid the double cert-fetch in presig recovery.
980        for &(bidx, start, size) in boundaries.iter().skip(first_pending) {
981            self.bail_if_reconfig_pending()?;
982            let presigs = self
983                .recover_presignatures_from_certs(
984                    &mpc_manager,
985                    epoch,
986                    bidx,
987                    batch_size_per_weight,
988                    params,
989                )
990                .await?;
991            anyhow::ensure!(
992                presigs.len() == size,
993                "batch {bidx} boundary size {size} (Phase 1) != reconstructed len {} (Phase 2)",
994                presigs.len(),
995            );
996            retained.push((presigs, bidx, start));
997        }
998        anyhow::ensure!(
999            self.inner.onchain_state().epoch() == epoch,
1000            "epoch changed during presigning recovery",
1001        );
1002        let latest_cursor = self
1003            .inner
1004            .onchain_state()
1005            .state()
1006            .hashi()
1007            .num_consumed_presigs;
1008        anyhow::ensure!(
1009            latest_cursor <= recovered_end,
1010            "cursor {latest_cursor} advanced past recovered end {recovered_end} during recovery",
1011        );
1012        let address = self.inner.config.validator_address()?;
1013        let retained_count = retained.len();
1014        let share_owners = self.share_owners_for_epoch(epoch)?;
1015        let (signing_manager, _identities) = SigningManager::new_recovered(
1016            address,
1017            committee,
1018            output.threshold,
1019            output.key_shares.clone(),
1020            output.public_key,
1021            share_owners,
1022            retained,
1023            num_consumed,
1024            &pending,
1025            PRESIG_REFILL_DIVISOR,
1026            self.refill_tx.clone(),
1027            self.identity_inputs(epoch, batch_size_per_weight),
1028        )?;
1029        self.inner.store_signing_manager(signing_manager);
1030        info!(
1031            "Recovered presigning state: {retained_count} of {} batch(es) retained from \
1032             first_pending_batch={first_pending}, recovered_end={recovered_end}, \
1033             num_consumed_presigs={num_consumed}, pending={}",
1034            boundaries.len(),
1035            pending.len(),
1036        );
1037        Ok(PresigRecovery::Installed)
1038    }

1048        &self,
1049        mpc_manager: &Arc<std::sync::RwLock<MpcManager>>,
1050        epoch: u64,
1051        batch_index: u32,
1052        params: Parameters,

1055    ) -> anyhow::Result<Option<usize>> {

1088    async fn sync_if_stale(&self) {
1089        if self.is_awaiting_genesis() || !self.inner.is_in_current_committee() {
1090            return;
1091        }
1092        let epoch = self.inner.onchain_state().epoch();
1093        if self.inner.signing_manager_for(epoch).is_some() {
1094            return;
1095        }
1096        let _guard = match self.reconciling.try_lock() {
1097            Ok(g) => g,
1098            Err(_) => return,
1099        };
1100        let epoch = self.inner.onchain_state().epoch();
1101        if self.inner.signing_manager_for(epoch).is_some() {
1102            return;
1103        }
1104        info!("sync_if_stale: rebuilding SigningManager for epoch {epoch}");
1105        let output = match self.recover_mpc_state().await {
1106            Ok(output) => output,
1107            Err(e) => {
1108                if let Some(p) = self.get_pending_epoch_change() {
1109                    info!(
1110                        "sync_if_stale: recover_mpc_state for epoch {epoch} superseded by \
1111                         pending reconfig to {p} ({e})"
1112                    );
1113                    return;
1114                }
1115                error!("sync_if_stale: recover_mpc_state failed for epoch {epoch}: {e}");
1116                self.re_register_keys_if_lost().await;
1117                return;
1118            }
1119        };
1120        match self.recover_presigning_state(&output).await {
1121            Ok(PresigRecovery::Installed) => {
1122                info!("sync_if_stale: recovered epoch {epoch} via certs path");
1123            }
1124            Ok(PresigRecovery::MissingBatch {
1125                epoch: walk_epoch,
1126                batch_index,
1127                batch_start,
1128            }) => {
1129                self.repair_missing_batch(walk_epoch, batch_index, batch_start)
1130                    .await;
1131                return;
1132            }
1133            Err(certs_err) => {
1134                if let Some(p) = self.get_pending_epoch_change() {
1135                    info!(
1136                        "sync_if_stale: recovery for epoch {epoch} superseded by pending \
1137                         reconfig to {p} ({certs_err})"
1138                    );
1139                    return;
1140                }
1141                let num_consumed = self
1142                    .inner
1143                    .onchain_state()
1144                    .state()
1145                    .hashi()
1146                    .num_consumed_presigs;
1147                if num_consumed > 0 {
1148                    error!(
1149                        "sync_if_stale: recovery failed for epoch {epoch} at cursor \
1150                         {num_consumed} ({certs_err}); retrying on the next tick (no genesis fallback)"
1151                    );
1152                    return;
1153                }
1154                info!(
1155                    "sync_if_stale: certs path failed for epoch {epoch} ({certs_err}), \
1156                     falling back to protocol"
1157                );
1158                if let Err(e) = self.prepare_signing(epoch, &output).await {
1159                    error!("sync_if_stale: protocol fallback failed for epoch {epoch}: {e}");
1160                    return;
1161                }
1162                info!("sync_if_stale: recovered epoch {epoch} via protocol path");
1163            }
1164        }
1165        let _ = self.key_ready_tx.send(Some(output.public_key));
1166    }

1168    async fn repair_missing_batch(&self, epoch: u64, batch_index: u32, batch_start: u64) {
1169        let metrics = &self.inner.metrics;
1170        let repair_total = &metrics.mpc_presig_batch_repair_total;
1171        let skip = |outcome: &str| repair_total.with_label_values(&[outcome]).inc();
1172        let now = tokio::time::Instant::now;
1173        let rearm_soon = || {
1174            *self.next_batch_repair.lock().unwrap() =
1175                Some((epoch, batch_index, now() + RETRY_INTERVAL))
1176        };
1177        if self.inner.onchain_state().epoch() != epoch {
1178            skip("epoch_moved");
1179            return;
1180        }
1181        {
1182            let mut next = self.next_batch_repair.lock().unwrap();
1183            if repair_rate_limited(*next, epoch, batch_index, now()) {
1184                skip("rate_limited");
1185                return;
1186            }
1187            *next = Some((epoch, batch_index, now() + REPAIR_MIN_INTERVAL));
1188        }
1189        let certs = match tob_certificates(
1190            self.inner.onchain_state(),
1191            epoch,
1192            Some(batch_index),
1193            move_types::ProtocolType::NonceGeneration,
1194        ) {
1195            Ok(certs) => certs,
1196            Err(e) => {
1197                skip("read_failed");
1198                rearm_soon();
1199                warn!(
1200                    "repair_missing_batch: cert read failed for epoch {epoch} batch \
1201                     {batch_index}: {e}"
1202                );
1203                return;
1204            }
1205        };
1206
1207        let address = match self.inner.config.validator_address() {
1208            Ok(address) => address,
1209            Err(e) => {
1210                skip("no_address");
1211                error!("repair_missing_batch: no validator address: {e}");
1212                return;
1213            }
1214        };
1215        if certs.iter().any(|(dealer, _)| *dealer == address) {
1216            skip("already_dealt");
1217            warn!(
1218                "repair_missing_batch: epoch {epoch} batch {batch_index} already carries our \
1219                 cert and is still below floor; no withdrawal signing"
1220            );
1221            return;
1222        }
1223
1224        let Some(mpc_manager) = self.inner.mpc_manager() else {
1225            skip("no_manager");
1226            error!("repair_missing_batch: no MpcManager for epoch {epoch} batch {batch_index}");
1227            return;
1228        };
1229        {
1230            let mgr = mpc_manager.read().unwrap();
1231            if let Err(e) = mgr.ensure_manager_epoch(epoch) {
1232                skip("manager_epoch");
1233                rearm_soon();
1234                warn!("repair_missing_batch: {e}");
1235                return;
1236            }
1237            if mgr.this_node_deals_nothing() {
1238                skip("no_weight");
1239                warn!(
1240                    "repair_missing_batch: epoch {epoch} batch {batch_index} is below floor and \
1241                     this node has nothing to deal into it; no withdrawal signing"
1242                );
1243                return;
1244            }
1245        }
1246        if let Err(e) = self.bail_if_reconfig_pending() {
1247            skip("reconfig_pending");
1248            rearm_soon();
1249            info!("repair_missing_batch: {e}");
1250            return;
1251        }
1252        let signer = match self.inner.config.operator_private_key() {
1253            Ok(signer) => signer,
1254            Err(e) => {
1255                skip("no_signer");
1256                error!("repair_missing_batch: no operator key: {e}");
1257                return;
1258            }
1259        };
1260        warn!(
1261            "repair_missing_batch: epoch {epoch} batch {batch_index} below floor at {batch_start} \
1262             with {} cert(s), cursor has drawn from it; dealing into it, no withdrawal \
1263             signing",
1264            certs.len(),
1265        );
1266        let onchain_state = self.inner.onchain_state().clone();
1267        let p2p_channel =
1268            RpcP2PChannel::new(onchain_state.clone(), epoch, MPC_LABEL_NONCE_GENERATION);
1269        let mut tob_channel = SuiTobSessionChannel::new(
1270            self.inner.config.hashi_ids(),
1271            onchain_state,
1272            epoch,
1273            Some(batch_index),
1274            move_types::ProtocolType::NonceGeneration,
1275            signer,
1276        )
1277        .with_idle_timeout(NONCE_RECEIVE_IDLE_TIMEOUT);
1278        repair_total.with_label_values(&["attempted"]).inc();
1279        let started = now();
1280        let round = MpcManager::run_nonce_dealer_phase(
1281            &mpc_manager,
1282            batch_index,
1283            &p2p_channel,
1284            &mut tob_channel,
1285            metrics,
1286        );
1287        tokio::pin!(round);
1288        let mut heartbeat = tokio::time::interval(REPAIR_HEARTBEAT_TICK);
1289        loop {
1290            tokio::select! {
1291                () = &mut round => break,
1292                _ = heartbeat.tick() => metrics.task_heartbeat("mpc_service"),
1293            }
1294        }
1295        *self.next_batch_repair.lock().unwrap() =
1296            Some((epoch, batch_index, now() + REPAIR_MIN_INTERVAL));
1297        info!(
1298            "repair_missing_batch: epoch {epoch} batch {batch_index} round returned after {:?}",
1299            started.elapsed(),
1300        );
1301    }

1303    async fn re_register_keys_if_lost(&self) {
1304        let committee = self
1305            .inner
1306            .onchain_state()
1307            .state()
1308            .hashi()
1309            .committees
1310            .current_committee()
1311            .cloned();
1312        let Some(committee) = committee else { return };
1313        let Ok(me) = self.inner.config.validator_address() else {
1314            return;
1315        };
1316        if !self.inner.committee_key_lost(&committee, me) {
1317            return;
1318        }
1319        self.inner.metrics.mpc_committee_key_lost_total.inc();
1320        let mut target = match self.inner.next_reconfig_epoch().await {
1321            Ok(target) => target,
1322            Err(e) => {
1323                warn!("cannot determine next reconfig epoch for key re-registration: {e}");
1324                return;
1325            }
1326        };
1327        for _ in 0..MAX_KEY_REREGISTRATION_BUMPS {
1328            if self
1329                .replacement_keys_target_epoch
1330                .lock()
1331                .unwrap()
1332                .is_some_and(|recorded| recorded >= target)
1333            {
1334                return;
1335            }
1336            warn!(
1337                "no DB encryption or signing key matches the current committee record; \
1338                 registering fresh keys for epoch {target} so the node rejoins at that reconfig"
1339            );
1340            let landed_at = match self.inner.prepare_and_register_keys(target).await {
1341                Ok(landed_at) => landed_at,
1342                Err(e) => {
1343                    warn!(
1344                        "failed to register replacement keys for epoch {target}: {e}; will retry"
1345                    );
1346                    return;
1347                }
1348            };
1349            // The snapshot check below reads the mirror, so the mirror must
1350            // first reflect the registration that just landed: a lagging
1351            // view could show the epoch-`target` committee as not yet
1352            // snapshotted when it was in fact already frozen without the
1353            // replacement keys, ending the bump loop one epoch short.
1354            if let Some(landed_at) = landed_at {
1355                const VISIBILITY_TIMEOUT: Duration = Duration::from_secs(30);
1356                if tokio::time::timeout(
1357                    VISIBILITY_TIMEOUT,
1358                    self.inner.onchain_state().wait_until_checkpoint(landed_at),
1359                )
1360                .await
1361                .is_err()
1362                {
1363                    warn!(
1364                        "mirror did not reach the key-registration checkpoint {landed_at} within \
1365                         {VISIBILITY_TIMEOUT:?}; re-verifying next tick"
1366                    );
1367                    return;
1368                }
1369            }
1370            let frozen = self
1371                .inner
1372                .onchain_state()
1373                .state()
1374                .hashi()
1375                .committees
1376                .committees()
1377                .get(&target)
1378                .cloned();
1379            match frozen {
1380                Some(frozen) if self.inner.committee_key_lost(&frozen, me) => {
1381                    self.inner.metrics.mpc_key_reregistration_bumps_total.inc();
1382                    info!(
1383                        "epoch {target} committee already snapshotted without replacement keys; \
1384                         re-targeting {}",
1385                        target + 1
1386                    );
1387                    target += 1;
1388                }
1389                Some(frozen) => {
1390                    if frozen.members().iter().any(|m| m.validator_address() == me) {
1391                        info!("replacement keys frozen into the epoch {target} committee");
1392                    } else {
1393                        warn!(
1394                            "not a member of the epoch {target} committee; replacement keys \
1395                             registered for a future committee"
1396                        );
1397                    }
1398                    *self.replacement_keys_target_epoch.lock().unwrap() = Some(target);
1399                    return;
1400                }
1401                None => {
1402                    info!(
1403                        "replacement keys registered; the epoch {target} committee is not yet \
1404                         snapshotted and will include them"
1405                    );
1406                    *self.replacement_keys_target_epoch.lock().unwrap() = Some(target);
1407                    return;
1408                }
1409            }
1410        }
1411        warn!(
1412            "replacement keys still excluded after {MAX_KEY_REREGISTRATION_BUMPS} registration \
1413             attempts; will retry next tick"
1414        );
1415    }

1417    async fn refill_presignatures(&self, batch_index: u32) -> anyhow::Result<()> {
1418        let epoch = self.inner.onchain_state().epoch();
1419        let signing_manager = self
1420            .inner
1421            .signing_manager_for(epoch)
1422            .ok_or_else(|| anyhow::anyhow!("SigningManager not available for epoch {epoch}"))?;
1423        // Refill requests arrive via a coalescing channel that outlives
1424        // manager rebuilds and epoch changes, so a replayed value can name a
1425        // batch that is already installed, already staged, or not contiguous
1426        // with the installed range. Only the immediately-next batch is
1427        // actionable; anything else is skipped before spending generation
1428        // work, and the trigger paths re-request the right index on demand.
1429        let expected = signing_manager.batch_index() + 1;
1430        if batch_index != expected {
1431            info!(
1432                "Skipping presignature refill for batch {batch_index}: \
1433                 the next installable batch is {expected}"
1434            );
1435            return Ok(());
1436        }
1437        if signing_manager.prefetched_batch_index() == Some(batch_index) {
1438            info!(
1439                "Skipping presignature refill for batch {batch_index}: \
1440                 it is already staged for installation"
1441            );
1442            return Ok(());
1443        }
1444        let (_, presignatures, batch_size_per_weight) =
1445            self.generate_presignatures(epoch, batch_index).await?;
1446        if self.inner.onchain_state().epoch() != epoch {
1447            return Err(anyhow::anyhow!("Epoch changed during presignature refill"));
1448        }
1449        signing_manager.set_next_batch(
1450            batch_index,
1451            presignatures,
1452            self.identity_inputs(epoch, batch_size_per_weight),
1453        );
1454        Ok(())
1455    }

1457    async fn fetch_final_nonce_certs(
1458        onchain_state: &crate::onchain::OnchainState,
1459        mpc_manager: &Arc<std::sync::RwLock<MpcManager>>,
1460        epoch: u64,
1461        batch_index: u32,
1462        wait_for_floor: bool,
1463        metrics: &crate::metrics::Metrics,
1464    ) -> anyhow::Result<(
1465        VerifiedNonceCerts<move_types::StampedDealerSubmissionV1>,
1466        Option<u64>,
1467    )> {
1468        let mut wait_deadline = tokio::time::Instant::now() + NONCE_RECEIVE_IDLE_TIMEOUT;
1469        let overall_deadline = tokio::time::Instant::now() + NONCE_WAIT_TOTAL_BUDGET;
1470        let mut best_weight = 0u32;
1471        let mut cutoff_confirmed: Option<u64> = None;
1472        let mut adjudicated = HashMap::new();
1473        let bail_if_superseded = || -> anyhow::Result<()> {
1474            let (onchain_epoch, pending) = {
1475                let state = onchain_state.state();
1476                let committees = &state.hashi().committees;
1477                (committees.epoch(), committees.pending_epoch_change())
1478            };
1479            anyhow::ensure!(
1480                !crate::communication::sui_tob::tob_wait_superseded(
1481                    move_types::ProtocolType::NonceGeneration,
1482                    epoch,
1483                    onchain_epoch,
1484                    pending,
1485                ),
1486                "nonce cert wait for epoch {epoch} batch {batch_index} superseded \
1487                 (onchain epoch {onchain_epoch}, pending epoch change {pending:?})"
1488            );
1489            Ok(())
1490        };
1491        loop {
1492            bail_if_superseded()?;
1493            let certs = match onchain_state.tob_certs(
1494                epoch,
1495                Some(batch_index),
1496                move_types::ProtocolType::NonceGeneration,
1497            ) {
1498                Ok(certs) => certs.map(|(_, certs)| certs).unwrap_or_default(),
1499                Err(e) if crate::onchain::is_inconsistent_listing(&e) => {
1500                    if tokio::time::Instant::now() >= wait_deadline {
1501                        return Err(e);
1502                    }
1503                    tokio::time::sleep(NONCE_WINDOW_WAIT_POLL).await;
1504                    continue;
1505                }
1506                Err(e) => return Err(e),
1507            };
1508            let certs = MpcManager::verified_nonce_certs(
1509                mpc_manager,
1510                epoch,
1511                certs,
1512                batch_index,
1513                &mut adjudicated,
1514                metrics,
1515            )
1516            .await;
1517            bail_if_superseded()?;
1518            let (floor_reached, cutoff_ms, window_ms, weight) = {
1519                let mgr = mpc_manager.read().unwrap();
1520                let window = mgr.window_certified_nonce_dealers(&certs).1;
1521                (
1522                    window.floor_reached(),
1523                    window.cutoff_ms(),
1524                    mgr.mpc_config.nonce_accumulation_window_ms,
1525                    window.weight(),
1526                )
1527            };
1528            if weight > best_weight {
1529                best_weight = weight;
1530                wait_deadline = (tokio::time::Instant::now() + NONCE_RECEIVE_IDLE_TIMEOUT)
1531                    .min(overall_deadline);
1532            }
1533            if wait_for_floor && !floor_reached {
1534                if tokio::time::Instant::now() >= wait_deadline {
1535                    metrics.mpc_nonce_fetch_floor_unreached_total.inc();
1536                    anyhow::bail!(
1537                        "nonce certs never reached the floor for epoch {epoch} batch {batch_index}"
1538                    );
1539                }
1540                tokio::time::sleep(NONCE_WINDOW_WAIT_POLL).await;
1541                continue;
1542            }
1543            // The certs above came from the mirror, so the settling
1544            // condition is the mirror's own clock: once it has applied
1545            // everything through a checkpoint stamped past the cutoff,
1546            // every submission at or before the cutoff is either in the
1547            // read we just took or in the re-read the loop does next.
1548            let Some(cutoff_ms) = cutoff_ms else {
1549                return Ok((certs, None));
1550            };
1551            if cutoff_confirmed == Some(cutoff_ms) {
1552                return Ok((certs, Some(cutoff_ms)));
1553            }
1554            if tokio::time::Instant::now() >= wait_deadline {
1555                metrics.mpc_nonce_cutoff_unsettled_total.inc();
1556                anyhow::bail!(
1557                    "nonce cert cutoff never settled for epoch {epoch} batch {batch_index} \
1558                     (last confirmed {cutoff_confirmed:?}, now {cutoff_ms})"
1559                );
1560            }
1561            let deadline = (Duration::from_millis(window_ms) + NONCE_WINDOW_WAIT_SLACK)
1562                .min(overall_deadline.saturating_duration_since(tokio::time::Instant::now()));
1563            if tokio::time::timeout(
1564                deadline,
1565                onchain_state.wait_mirror_past_timestamp_ms(cutoff_ms),
1566            )
1567            .await
1568            .is_err()
1569            {
1570                metrics.mpc_nonce_window_cutoff_unreached_total.inc();
1571                warn!(
1572                    "fetch_final_nonce_certs: the mirror never advanced past window cutoff \
1573                     {cutoff_ms} for epoch {epoch} batch {batch_index} after {deadline:?} \
1574                     (stalled chain clock or a lagging mirror); failing this batch"
1575                );
1576                anyhow::bail!("nonce window did not close for epoch {epoch} batch {batch_index}");
1577            }
1578            if cutoff_confirmed.is_some() {
1579                tokio::time::sleep(NONCE_WINDOW_WAIT_POLL).await;
1580            }
1581            cutoff_confirmed = Some(cutoff_ms);
1582        }
1583    }

1585    async fn recover_presignatures_from_certs(
1586        &self,
1587        mpc_manager: &Arc<std::sync::RwLock<MpcManager>>,
1588        epoch: u64,
1589        batch_index: u32,
1590        batch_size_per_weight: u16,
1591        params: Parameters,
1592    ) -> anyhow::Result<Presignatures> {
1593        let onchain_state = self.inner.onchain_state().clone();
1594        let p2p_channel = RpcP2PChannel::new(
1595            self.inner.onchain_state().clone(),
1596            epoch,
1597            MPC_LABEL_NONCE_GENERATION,
1598        );
1599        let (certs, settled_cutoff_ms) = Self::fetch_final_nonce_certs(
1600            &onchain_state,
1601            mpc_manager,
1602            epoch,
1603            batch_index,
1604            false,
1605            &self.inner.metrics,
1606        )
1607        .await?;
1608        if certs.is_empty() {
1609            return Err(anyhow::anyhow!(
1610                "No nonce gen certificates on TOB for epoch {epoch} batch {batch_index}"
1611            ));
1612        }
1613        let (outputs, served_weight) = {
1614            let avid_certs = nonce_certificates(&certs, epoch, batch_index);
1615            let admitted = mpc_manager
1616                .read()
1617                .unwrap()
1618                .avid_admitted_nonce_dealers(&avid_certs, settled_cutoff_ms)?;
1619            if !admitted.floor_reached() {
1620                return Err(admitted
1621                    .below_floor_error(batch_index, &self.inner.metrics)
1622                    .into());
1623            }
1624            let outcome = MpcManager::run_avid_nonce_party_phase(
1625                mpc_manager,
1626                batch_index,
1627                &p2p_channel,
1628                &admitted,
1629                Some((onchain_state.clone(), epoch)),
1630                &self.inner.metrics,
1631            )
1632            .await
1633            .map_err(|e| anyhow::anyhow!("AVID nonce recovery from certs failed: {e}"))?;
1634            if outcome.local_skips > 0 {
1635                self.inner.metrics.mpc_nonce_local_skip_batches_total.inc();
1636                anyhow::bail!(
1637                    "AVID nonce recovery for epoch {epoch} batch {batch_index} skipped {} \
1638                     dealer(s) for node-local reasons; cannot rebuild the original batch",
1639                    outcome.local_skips,
1640                );
1641            }
1642            (outcome.outputs, admitted.weight)
1643        };
1644        if outputs.is_empty() {
1645            return Err(anyhow::anyhow!(
1646                "No valid nonce outputs after reconstruction for epoch {epoch} batch {batch_index}"
1647            ));
1648        }
1649        let dealer_count = outputs.len();
1650        let presignatures = Presignatures::new(
1651            outputs,
1652            batch_size_per_weight,
1653            params,
1654            USE_LEGACY_PRESIG_DERIVATION,
1655        )
1656        .map_err(|e| anyhow::anyhow!("Failed to create presignatures: {e}"))?;
1657        let metrics = &self.inner.metrics;
1658        let served_implies = presig_count(served_weight as usize, params, batch_size_per_weight);
1659        if presignatures.len() != served_implies {
1660            metrics.mpc_nonce_size_mismatch_total.inc();
1661            anyhow::bail!(
1662                "nonce batch {batch_index} for epoch {epoch} rebuilt from certs: built {} \
1663                 presigs but the admitted certs size to {served_implies} (weight \
1664                 {served_weight}); refusing to install",
1665                presignatures.len(),
1666            );
1667        }
1668        metrics.mpc_nonce_batch_index.set(batch_index as i64);
1669        metrics.mpc_nonce_batch_dealers.set(dealer_count as i64);
1670        Ok(presignatures)
1671    }

1673    async fn try_submit_start_reconfig(&self, sui_epoch: u64) {
1674        if let Some(pending) = self.get_pending_epoch_change() {
1675            if !reconfig_window_closed(sui_epoch, pending) {
1676                return;
1677            }
1678            // The pending reconfiguration overran its window, so the chain
1679            // will not let it complete; abort it (any node may) so the
1680            // replacement can form in the new epoch.
1681            self.abort_stale_reconfig(pending).await;
1682            if self.get_pending_epoch_change().is_some() {
1683                return;
1684            }
1685        }
1686        let hashi_epoch = self
1687            .inner
1688            .onchain_state()
1689            .state()
1690            .hashi()
1691            .committees
1692            .epoch();
1693        if hashi_epoch >= sui_epoch {
1694            return;
1695        }
1696        if self.is_awaiting_genesis() {
1697            // Pre-genesis (where an aborted genesis DKG lands) there is no
1698            // committee to serve, so block in the same retry-until-pending
1699            // loop startup uses rather than give up after a few attempts and
1700            // leave the replacement to the next Sui epoch boundary or a
1701            // restart.
1702            self.try_submit_genesis_reconfig().await;
1703            return;
1704        }
1705        if let Err(e) = self.inner.prepare_and_register_keys(sui_epoch).await {
1706            warn!(
1707                "Failed to prepare/register encryption+signing keys for epoch {sui_epoch}: {e}; \
1708                 will retry on next trigger"
1709            );
1710        }
1711        for attempt in 1..=MAX_PROTOCOL_ATTEMPTS {
1712            let result = async {
1713                let mut executor =
1714                    crate::sui_tx_executor::SuiTxExecutor::from_hashi(self.inner.clone())?;
1715                executor.execute_start_reconfig().await
1716            };
1717            match result.await {
1718                Ok(()) => {
1719                    return;
1720                }
1721                Err(e) => {
1722                    warn!("start_reconfig attempt {attempt}/{MAX_PROTOCOL_ATTEMPTS} failed: {e:#}");
1723                    if attempt < MAX_PROTOCOL_ATTEMPTS {
1724                        // Poll for pending epoch change while waiting, so we can
1725                        // return early if another node submitted start_reconfig.
1726                        let polls = (RETRY_INTERVAL.as_millis()
1727                            / START_RECONFIG_POLL_INTERVAL.as_millis())
1728                            as u32;
1729                        for _ in 0..polls {
1730                            if self.get_pending_epoch_change().is_some() {
1731                                return;
1732                            }
1733                            tokio::time::sleep(START_RECONFIG_POLL_INTERVAL).await;
1734                        }
1735                    }
1736                }
1737            }
1738        }
1739    }

1741    #[tracing::instrument(level = "info", skip_all, fields(target_epoch))]
1742    async fn handle_reconfig(&self, target_epoch: u64) {
1743        let run_dkg = self
1744            .inner
1745            .onchain_state()
1746            .state()
1747            .hashi()
1748            .committees
1749            .mpc_public_key()
1750            .is_empty();
1751        let protocol_label = if run_dkg {
1752            MPC_LABEL_DKG
1753        } else {
1754            MPC_LABEL_KEY_ROTATION
1755        };
1756        if !self.reconfig_target_pending(target_epoch) {
1757            info!(
1758                "handle_reconfig: epoch {target_epoch} no longer pending (or past its Sui epoch \
1759                 window) at entry, aborting before start",
1760            );
1761            return;
1762        }
1763        if let Err(e) = self.inner.verify_bitcoin_chain_id() {
1764            error!("refusing to participate in reconfig for epoch {target_epoch}: {e}");
1765            self.sleep_if_still_pending(target_epoch).await;
1766            return;
1767        }
1768        let metrics = &self.inner.metrics;
1769        let _reconfig_timer = metrics
1770            .mpc_reconfig_total_duration_seconds
1771            .with_label_values(&[protocol_label])
1772            .start_timer();
1773        info!("handle_reconfig: epoch={target_epoch}, run_dkg={run_dkg}, entering retry loop",);
1774        let output = loop {
1775            if !self.reconfig_target_pending(target_epoch) {
1776                info!(
1777                    "handle_reconfig: epoch {target_epoch} no longer pending or past its Sui \
1778                     epoch window, aborting"
1779                );
1780                return;
1781            }
1782            let in_committee = self.inner.is_in_committee_for(target_epoch);
1783            let needs_fresh_manager = match self.inner.mpc_manager() {
1784                None => true,
1785                Some(mgr) => {
1786                    let mgr = mgr.read().unwrap();
1787                    mgr.mpc_config.epoch != target_epoch
1788                        || mgr.is_committee_member() != in_committee
1789                }
1790            };
1791            if needs_fresh_manager {
1792                let setup_result = if run_dkg {
1793                    self.setup_initial_dkg(target_epoch)
1794                } else {
1795                    self.setup_key_rotation(target_epoch)
1796                };
1797                if let Err(e) = setup_result {
1798                    error!(
1799                        "Failed to set up MPC manager for epoch {}: {e}, retrying...",
1800                        target_epoch
1801                    );
1802                    self.sleep_if_still_pending(target_epoch).await;
1803                    continue;
1804                }
1805            }
1806            let _timer = metrics
1807                .mpc_total_duration_seconds
1808                .with_label_values(&[protocol_label])
1809                .start_timer();
1810            let result = if run_dkg {
1811                self.run_dkg(target_epoch).await
1812            } else {
1813                self.run_key_rotation(target_epoch).await
1814            };
1815            drop(_timer);
1816            match result {
1817                Ok(ReconfigOutcome::Output(output)) => break output,
1818                Ok(outcome) => {
1819                    let reason = outcome.label();
1820                    if matches!(outcome, ReconfigOutcome::NoShares) {
1821                        warn!(
1822                            "handle_reconfig: epoch {target_epoch} produced no output: this node \
1823                             holds no previous-epoch shares to reshare"
1824                        );
1825                    } else {
1826                        info!(
1827                            "handle_reconfig: epoch {target_epoch} produced no output for this \
1828                             node ({reason}); waiting out the pending window"
1829                        );
1830                    }
1831                    metrics
1832                        .mpc_reconfig_no_output_total
1833                        .with_label_values(&[protocol_label, reason])
1834                        .inc();
1835                    _reconfig_timer.stop_and_discard();
1836                    while self.reconfig_target_pending(target_epoch) {
1837                        metrics.task_heartbeat("mpc_service");
1838                        self.sleep_if_still_pending(target_epoch).await;
1839                    }
1840                    let backup = match outcome {
1841                        ReconfigOutcome::NoRole => Backup::Skip,
1842                        ReconfigOutcome::Output(_)
1843                        | ReconfigOutcome::Dealt
1844                        | ReconfigOutcome::NotNeeded
1845                        | ReconfigOutcome::NoShares => Backup::Write,
1846                    };
1847                    self.post_reconfig_housekeeping(target_epoch, false, backup)
1848                        .await;
1849                    return;
1850                }
1851                Err(e) => {
1852                    error!(
1853                        "MPC protocol for epoch {} failed: {e}, retrying...",
1854                        target_epoch
1855                    );
1856                    self.sleep_if_still_pending(target_epoch).await;
1857                }
1858            }
1859        };
1860        if !self.reconfig_target_live(target_epoch) {
1861            info!("handle_reconfig: epoch {target_epoch} aborted after protocol completion");
1862            return;
1863        }
1864        let _ = self.key_ready_tx.send(Some(output.public_key));
1865        info!("MPC key ready for epoch {target_epoch}, submitting end_reconfig");
1866        let _end_reconfig_timer = metrics
1867            .mpc_end_reconfig_duration_seconds
1868            .with_label_values(&[protocol_label])
1869            .start_timer();
1870        let mut end_reconfig_confirmed = false;
1871        loop {
1872            // Pending cleared: activated by another node (signing setup
1873            // below) or aborted (nothing to do). Checked before the window
1874            // so a completion that landed in the window's last moments is
1875            // not mistaken for an overrun.
1876            if self.get_pending_epoch_change() != Some(target_epoch) {
1877                break;
1878            }
1879            if self.reconfig_window_closed(target_epoch) {
1880                info!(
1881                    "handle_reconfig: epoch {target_epoch} overran its Sui epoch window before \
1882                     end_reconfig landed; it can only be aborted now"
1883                );
1884                _end_reconfig_timer.stop_and_discard();
1885                _reconfig_timer.stop_and_discard();
1886                return;
1887            }
1888            match self.submit_end_reconfig(target_epoch, &output).await {
1889                Ok(()) => {
1890                    end_reconfig_confirmed = true;
1891                    break;
1892                }
1893                Err(e) => match classify_reconfig_submission_error(&e) {
1894                    ReconfigSubmissionErrorKind::NonMoveAbort => {
1895                        warn!(
1896                            "submit_end_reconfig for epoch {} failed: {e:#}, retrying...",
1897                            target_epoch
1898                        );
1899                        self.sleep_if_still_pending(target_epoch).await;
1900                    }
1901                    ReconfigSubmissionErrorKind::ReconfigTargetDead => {
1902                        info!(
1903                            "handle_reconfig: epoch {target_epoch} can no longer complete \
1904                             (aborted, or past its Sui epoch window); giving up on it: {e:#}"
1905                        );
1906                        _end_reconfig_timer.stop_and_discard();
1907                        _reconfig_timer.stop_and_discard();
1908                        // The chain knows the target is dead before the
1909                        // mirror does. Until the mirror catches up the outer
1910                        // loop still sees it pending and would re-enter here
1911                        // immediately, re-collecting signatures and
1912                        // submitting another doomed transaction each pass.
1913                        self.sleep_if_still_pending(target_epoch).await;
1914                        return;
1915                    }
1916                    ReconfigSubmissionErrorKind::NonRetryableMoveAbort
1917                    | ReconfigSubmissionErrorKind::CommitteeHandoffAlreadySubmitted
1918                    | ReconfigSubmissionErrorKind::EndReconfigAlreadyCompleted => {
1919                        let msg = format!(
1920                            "submit_end_reconfig for epoch {target_epoch} failed with non-retryable error: {e:#}"
1921                        );
1922                        error!("{msg}");
1923                        panic!("{msg}");
1924                    }
1925                },
1926            }
1927        }
1928        drop(_end_reconfig_timer);
1929        let next_epoch = target_epoch + 1;
1930        if let Err(e) = self.inner.prepare_and_register_keys(next_epoch).await {
1931            warn!(
1932                "Failed to prepare/register encryption+signing keys for epoch {next_epoch}: {e}; \
1933                 will retry at next trigger"
1934            );
1935        }
1936        info!("end_reconfig complete for epoch {target_epoch}, running prepare_signing");
1937        self.post_reconfig_housekeeping(target_epoch, end_reconfig_confirmed, Backup::Write)
1938            .await;
1939        if !self.reconfig_target_live(target_epoch) {
1940            info!(
1941                "handle_reconfig: epoch {target_epoch} no longer pending nor current; \
1942                 skipping signing setup"
1943            );
1944            return;
1945        }
1946        let _prepare_signing_timer = metrics
1947            .mpc_prepare_signing_duration_seconds
1948            .with_label_values(&[protocol_label])
1949            .start_timer();
1950        for attempt in 1..=MAX_PROTOCOL_ATTEMPTS {
1951            match self.prepare_signing(target_epoch, &output).await {
1952                Ok(()) => break,
1953                Err(e) => {
1954                    error!(
1955                        "prepare_signing attempt {attempt}/{MAX_PROTOCOL_ATTEMPTS} \
1956                         for epoch {target_epoch}: {e}"
1957                    );
1958                    if attempt < MAX_PROTOCOL_ATTEMPTS {
1959                        tokio::time::sleep(RETRY_INTERVAL).await;
1960                    } else {
1961                        error!(
1962                            "All prepare_signing attempts exhausted for epoch {target_epoch}. \
1963                             Node cannot sign until next recovery trigger."
1964                        );
1965                    }
1966                }
1967            }
1968        }
1969        drop(_prepare_signing_timer);
1970        drop(_reconfig_timer);
1971    }

1973    /// Compact the database so rows tombstoned by pruning are actually unlinked.
1974    ///
1975    /// Awaited deliberately before `prepare_signing` refills the presig pool:
1976    /// the keyspaces are at their emptiest right after the prune, so the merge
1977    /// writes almost nothing.
1978    async fn run_major_compaction(&self, epoch: u64) {
1979        let db = self.inner.db.clone();
1980        let started = std::time::Instant::now();
1981        let compaction = match tokio::task::spawn_blocking(move || db.major_compact()).await {
1982            Ok(compaction) => compaction,
1983            Err(e) => {
1984                error!("major compaction for epoch {epoch} panicked: {e}");
1985                return;
1986            }
1987        };
1988        for keyspace in compaction.keyspaces {
1989            self.inner.metrics.record_major_compaction(&keyspace);
1990            match &keyspace.error {
1991                Some(e) => error!("compacting {} failed: {e}", keyspace.name),
1992                None => info!(
1993                    "compacted {} in {:?}: {} -> {} live bytes",
1994                    keyspace.name, keyspace.elapsed, keyspace.before, keyspace.after,
1995                ),
1996            }
1997        }
1998        // The unlink rotation is what actually releases the disk, so its
1999        // failure fails the run even when every keyspace compacted cleanly.
2000        match &compaction.unlink_error {
2001            Some(e) => {
2002                self.inner.metrics.record_major_compaction_unlink_failure();
2003                error!(
2004                    "major compaction for epoch {epoch} failed after {:?}: \
2005                     retired tables were not unlinked, the space is still held: {e}",
2006                    started.elapsed()
2007                );
2008            }
2009            None => info!(
2010                "major compaction for epoch {epoch} finished in {:?}",
2011                started.elapsed()
2012            ),
2013        }
2014    }

2101            .onchain_state()

2111            mpc_public_key: mpc_public_key.clone(),

2124            .store_reconfig_signature(epoch, my_sig.signature().as_bytes().to_vec());

2161                    .await

2171                        if self.get_pending_epoch_change() != Some(epoch) {
2172                            return Ok(());
2173                        }
2174                        warn!(
2175                            "end_reconfig for epoch {epoch} is complete on chain, but the watcher still reports it pending after the visibility wait; retrying"
2176                        );

2178                    }
2179                    ReconfigSubmissionErrorKind::CommitteeHandoffAlreadySubmitted => {

2181                            warn!(
2182                                "end_reconfig submission for epoch {epoch} found handoff already submitted; retrying without the handoff call: {e:#}"

2185                        }
2186                        Err(e).with_context(|| {
2187                            format!(
2188                                "end_reconfig submission for epoch {epoch} failed because a handoff was already submitted, but this transaction did not submit one"
2189                            )
2190                        })?;
2191                    }
2192                    ReconfigSubmissionErrorKind::NonRetryableMoveAbort => {
2193                        Err(e).with_context(|| {
2194                            format!(
2195                                "end_reconfig submission for epoch {epoch} failed with non-retryable error"
2196                            )
2197                        })?;
2198                    }
2199                    ReconfigSubmissionErrorKind::ReconfigTargetDead => {
2200                        Err(e).with_context(|| {
2201                            format!(
2202                                "end_reconfig submission for epoch {epoch} found the target dead: \
2203                                 aborted, or past its Sui epoch window"
2204                            )

2209                            "end_reconfig submission for epoch {} failed: {e:#}, retrying...",
2210                            epoch
2211                        );
2212                        self.sleep_if_still_pending(epoch).await;

2261    async fn collect_reconfig_signatures(
2262        &self,
2263        epoch: u64,
2264        mpc_public_key: &[u8],
2265        committee: &Committee,
2266    ) -> anyhow::Result<hashi_types::committee::SignedMessage<ReconfigCompletionMessage>> {
2267        let message = ReconfigCompletionMessage {
2268            epoch,
2269            mpc_public_key: mpc_public_key.to_vec(),
2270        };
2271        let my_address = self.inner.config.validator_address()?;
2272        let my_sig_bytes = self
2273            .inner
2274            .get_reconfig_signature(epoch)
2275            .expect("own signature must be stored before collecting");
2276        let my_sig =
2277            BLS12381Signature::from_bytes(&my_sig_bytes).expect("stored signature must be valid");
2278        let mut aggregator = BlsSignatureAggregator::new(
2279            self.inner.config.hashi_ids().hashi_object_id,
2280            committee,
2281            message.clone(),
2282        );
2283        aggregator
2284            .add_signature_from(my_address, my_sig)
2285            .map_err(|e| anyhow::anyhow!("failed to add own signature: {e}"))?;
2286        let required_weight = certificate_threshold(committee.total_weight());
2287        while aggregator.weight() < required_weight {
2288            if !self.reconfig_target_pending(epoch) {
2289                return Err(anyhow::anyhow!(
2290                    "epoch {epoch} no longer pending (or past its Sui epoch window) during \
2291                     signature collection"
2292                ));
2293            }
2294            let other_members: Vec<_> = committee
2295                .members()
2296                .iter()
2297                .filter(|m| m.validator_address() != my_address)
2298                .collect();
2299            let futures = other_members.iter().map(|member| {
2300                let address = member.validator_address();
2301                async move {
2302                    let result = tokio::time::timeout(RPC_TIMEOUT, async {
2303                        let client = self
2304                            .inner
2305                            .onchain_state()
2306                            .state()
2307                            .hashi()
2308                            .committees
2309                            .client(&address)
2310                            .ok_or_else(|| anyhow::anyhow!("client not found for {}", address))?;
2311                        client
2312                            .get_reconfig_completion_signature(epoch)
2313                            .await
2314                            .map_err(|e| anyhow::anyhow!("RPC failed: {e}"))
2315                    })
2316                    .await
2317                    .unwrap_or_else(|_| Err(anyhow::anyhow!("RPC timed out")));
2318                    (
2319                        address,
2320                        result.and_then(|opt| {
2321                            opt.ok_or_else(|| anyhow::anyhow!("signature not ready"))
2322                        }),
2323                    )
2324                }
2325            });
2326            let results = join_all(futures).await;
2327            for (address, result) in results {
2328                if let Ok(sig_bytes) = result {
2329                    match BLS12381Signature::from_bytes(&sig_bytes) {
2330                        Ok(sig) => {
2331                            if let Err(e) = aggregator.add_signature_from(address, sig) {
2332                                info!("Signature from {} rejected: {e}", address);
2333                            }
2334                        }
2335                        Err(e) => {
2336                            info!("Invalid signature bytes from {}: {e}", address);
2337                        }
2338                    }
2339                }
2340            }
2341            if aggregator.weight() < required_weight {
2342                tokio::time::sleep(RETRY_INTERVAL).await;
2343            }
2344        }
2345        aggregator
2346            .finish()
2347            .map_err(|e| anyhow::anyhow!("failed to finalize certificate: {e}"))
2348    }

2361pub(crate) async fn verify_fetched_certificates(
2362    mpc_manager: &Arc<std::sync::RwLock<MpcManager>>,
2363    certs: Vec<CertificateV1>,
2364    metrics: &Metrics,
2365) -> Vec<VerifiedCertificateV1> {
2366    let mgr = Arc::clone(mpc_manager);
2367    let (verified, rejected) = spawn_blocking(move || {
2368        let mgr = mgr.read().unwrap();
2369        let mut verified = Vec::with_capacity(certs.len());
2370        let mut rejected: Vec<(&'static str, &'static str)> = Vec::new();
2371        for cert in certs {
2372            let label = cert.protocol_label();
2373            match mgr.verify_certificate(cert.clone()) {
2374                Ok(cert) => verified.push(cert),
2375                Err(e) => {
2376                    warn!(
2377                        "dropping unverifiable TOB certificate from {:?}: {e}",
2378                        cert.dealer_address()
2379                    );
2380                    rejected.push((label, mgr.certificate_rejection_reason(&cert)));
2381                }
2382            }
2383        }
2384        (verified, rejected)
2385    })
2386    .await;

2439fn classify_abort_submission_error(err: &anyhow::Error) -> AbortSubmissionErrorKind {
2440    let Some(error) = crate::sui_tx_executor::transaction_execution_error(err) else {
2441        return AbortSubmissionErrorKind::Other;
2442    };
2443    classify_abort_execution_error(error)

2446fn classify_abort_execution_error(error: &ExecutionError) -> AbortSubmissionErrorKind {
2447    if error
2448        .kind
2449        .and_then(|kind| ExecutionErrorKind::try_from(kind).ok())
2450        != Some(ExecutionErrorKind::MoveAbort)
2451    {
2452        return AbortSubmissionErrorKind::Other;
2453    }
2454    let Some(abort) = error.abort_opt() else {
2455        return AbortSubmissionErrorKind::Other;
2456    };
2457    let abort_constant_name = abort
2458        .clever_error
2459        .as_ref()
2460        .and_then(|clever| clever.constant_name.as_deref());
2461    match (abort.location().module_opt(), abort_constant_name) {
2462        (Some("reconfig"), Some(RECONFIG_E_NOT_RECONFIGURING | RECONFIG_E_WRONG_EPOCH)) => {
2463            AbortSubmissionErrorKind::AlreadyResolved
2464        }
2465        (Some("committee_set"), Some(COMMITTEE_SET_E_PENDING_EPOCH_STILL_CURRENT)) => {
2466            AbortSubmissionErrorKind::StillCurrent
2467        }
2468        _ => AbortSubmissionErrorKind::Other,
2469    }
2470}

2472fn classify_reconfig_submission_error(err: &anyhow::Error) -> ReconfigSubmissionErrorKind {
2473    let Some(error) = crate::sui_tx_executor::transaction_execution_error(err) else {
2474        return ReconfigSubmissionErrorKind::NonMoveAbort;
2475    };
2476    classify_reconfig_execution_error(error)
2477}
2478
2479fn classify_reconfig_execution_error(error: &ExecutionError) -> ReconfigSubmissionErrorKind {
2480    if error
2481        .kind
2482        .and_then(|kind| ExecutionErrorKind::try_from(kind).ok())
2483        != Some(ExecutionErrorKind::MoveAbort)
2484    {
2485        return ReconfigSubmissionErrorKind::NonMoveAbort;
2486    }
2487
2488    let Some(abort) = error.abort_opt() else {
2489        return ReconfigSubmissionErrorKind::NonRetryableMoveAbort;
2490    };
2491    let location = abort.location();
2492
2493    let abort_constant_name = abort
2494        .clever_error
2495        .as_ref()
2496        .and_then(|clever| clever.constant_name.as_deref());
2497    match (
2498        location.module_opt(),
2499        location.function_name_opt(),
2500        abort_constant_name,
2501    ) {
2502        (Some("committee_set"), Some("set_pending_committee_handoff_cert"), _) => {
2503            ReconfigSubmissionErrorKind::CommitteeHandoffAlreadySubmitted
2504        }
2505        // `ENotReconfiguring` and the window check live in a private helper
2506        // shared by `submit_committee_handoff` and `end_reconfig`, while the
2507        // entries raise the already-completed and wrong-epoch aborts
2508        // themselves, so the location varies; the constant carries the
2509        // meaning.
2510        (Some("reconfig"), _, Some(RECONFIG_E_ALREADY_COMPLETED)) => {
2511            ReconfigSubmissionErrorKind::EndReconfigAlreadyCompleted
2512        }
2513        (
2514            Some("reconfig"),
2515            _,
2516            Some(RECONFIG_E_NOT_RECONFIGURING | RECONFIG_E_WINDOW_CLOSED | RECONFIG_E_WRONG_EPOCH),
2517        ) => ReconfigSubmissionErrorKind::ReconfigTargetDead,
2518        _ => ReconfigSubmissionErrorKind::NonRetryableMoveAbort,
2519    }

2596pub(crate) fn build_pruning_references(
2597    committee_set: &crate::onchain::types::CommitteeSet,
2598    target_epoch: u64,
2599) -> crate::db::PruningReferences {
2600    let mut referenced = crate::db::PruningReferences::default();
2601    let previous = committee_set.previous_committee_for_target(target_epoch);
2602    let preceding = committee_set.committees().range(..target_epoch).next_back();
2603    for committee in committee_set
2604        .committees()
2605        .get(&target_epoch)
2606        .into_iter()
2607        .chain(previous.map(|(_, committee)| committee))
2608        .chain(preceding.map(|(_, committee)| committee))
2609    {
2610        for member in committee.members() {
2611            referenced.add_member_pubkeys(member.encryption_public_key(), member.public_key());
2612        }
2613    }
2614    for (epoch, _) in previous.into_iter().chain(preceding.map(|(e, c)| (*e, c))) {
2615        referenced.add_committee_epoch(epoch);
2616    }
2617    for member_info in committee_set.members().values() {
2618        referenced.add_pending_registration(
2619            member_info.next_epoch_encryption_public_key(),
2620            member_info.next_epoch_public_key(),
2621        );
2622    }
2623    referenced
2624}

2667/// Whether `target_epoch` is still worth acting on after the protocol ran:
2668/// pending, or already the current epoch (another node's `end_reconfig`
2669/// activated it). "Current" needs a committee to exist for it: before
2670/// genesis the epoch is 0 with no committee, and the genesis committee on a
2671/// fresh network is pinned to Sui epoch 0 too, so an aborted genesis DKG
2672/// would otherwise look activated.
2673fn reconfig_target_live(
2674    pending: Option<u64>,
2675    current_epoch: u64,
2676    current_committee_exists: bool,
2677    target_epoch: u64,
2678) -> bool {
2679    pending == Some(target_epoch) || (current_epoch == target_epoch && current_committee_exists)
2680}
2681
2682/// `start_reconfig` pins the pending committee to Sui's epoch, and the chain
2683/// refuses to complete it once Sui's epoch has moved on. `latest_sui_epoch`
2684/// is the watcher's checkpoint view, which can lag the chain but never lead
2685/// it, so a closed window here is closed on chain too.
2686fn reconfig_window_closed(latest_sui_epoch: u64, target_epoch: u64) -> bool {
2687    latest_sui_epoch > target_epoch
2688}
2689
2690/// Pending on chain and inside its Sui epoch window.
2691fn reconfig_target_pending(pending: Option<u64>, latest_sui_epoch: u64, target_epoch: u64) -> bool {
2692    pending == Some(target_epoch) && !reconfig_window_closed(latest_sui_epoch, target_epoch)
2693}
