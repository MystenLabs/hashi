24pub(crate) fn dkg_message(
25    shape: &ShapeParams,
26    epoch: u64,
27    dealer: &Address,
28    bytes: &[u8],
29) -> MpcResult<avss::Message> {
30    check_dkg_message(shape, epoch, dealer, bytes)?;
31    bcs::from_bytes(bytes).map_err(|e| refused(dealer, e.to_string()))
32}

34pub(crate) fn check_dkg_message(
35    shape: &ShapeParams,
36    epoch: u64,
37    dealer: &Address,
38    bytes: &[u8],
39) -> MpcResult<()> {
40    count_check::check_avss_message(bytes, avss_shape(shape, epoch, dealer)?)
41        .map_err(|e| refused(dealer, format!("count check: {e}")))
42}

54pub(crate) fn rotation_entries(
55    shape: &ShapeParams,
56    epoch: u64,
57    dealer: &Address,
58    bytes: &[u8],
59) -> MpcResult<BTreeMap<ShareIndex, Range<usize>>> {
60    let avss = avss_shape(shape, epoch, dealer)?;
61    let owned = shape
62        .dealing_committee_for_epoch(epoch)
63        .ok_or_else(|| {
64            refused(
65                dealer,
66                format!("no dealing committee known for epoch {epoch}"),
67            )
68        })?
69        .share_ids_of(dealer)
70        .ok_or_else(|| {
71            refused(
72                dealer,
73                format!("no shares in epoch {epoch}'s dealing committee"),
74            )
75        })?;
76    count_check::check_rotation_messages(bytes, avss, &owned)
77        .map_err(|e| refused(dealer, format!("count check: {e}")))
78}

80pub(crate) fn avid_round_state(
81    shape: &ShapeParams,
82    epoch: u64,
83    dealer: &Address,
84    bytes: &[u8],
85) -> MpcResult<AvidRoundState> {
86    let dealer_party = avid_committee(shape, epoch, dealer)?
87        .party_of(dealer)
88        .ok_or_else(|| refused(dealer, "not in the committee".into()))?;
89    decode(
90        dealer,
91        bytes,
92        count_check::check_avid_avss_message(bytes, shape.avid_shape(), dealer_party),
93    )
94}

96pub(crate) fn held_avid_echoes(
97    shape: &ShapeParams,
98    epoch: u64,
99    dealer: &Address,
100    bytes: &[u8],
101) -> MpcResult<HeldAvidEchoes> {
102    let current = avid_committee(shape, epoch, dealer)?;
103    let holder = current
104        .own_party()
105        .ok_or_else(|| refused(dealer, "this node is not in the committee".into()))?;
106    decode(
107        dealer,
108        bytes,
109        count_check::check_held_avid_echoes(bytes, current.weights(), holder),
110    )
111}

121fn variant_hash(variant: u8, bytes: &[u8]) -> MessagesHash {
122    let mut hasher = Blake2b256::default();
123    hasher.update([variant]);
124    hasher.update(bytes);
125    MessagesHash::from(hasher.finalize().digest)
126}

128fn avss_shape(
129    shape: &ShapeParams,
130    epoch: u64,
131    dealer: &Address,
132) -> MpcResult<count_check::AvssShape> {
133    shape
134        .committee_for_epoch(epoch)
135        .map(CommitteeShape::avss_shape)
136        .ok_or_else(|| refused(dealer, format!("no committee known for epoch {epoch}")))
137}

139fn avid_committee<'a>(
140    shape: &'a ShapeParams,
141    epoch: u64,
142    dealer: &Address,
143) -> MpcResult<&'a CommitteeShape> {
144    (epoch == shape.epoch())
145        .then(|| shape.current())
146        .ok_or_else(|| refused(dealer, format!("no AVID parameters for epoch {epoch}")))
147}

149fn decode<T: DeserializeOwned>(
150    dealer: &Address,
151    bytes: &[u8],
152    checked: Result<(), CountCheckError>,
153) -> MpcResult<T> {
154    checked.map_err(|e| refused(dealer, format!("count check: {e}")))?;
155    bcs::from_bytes(bytes).map_err(|e| refused(dealer, e.to_string()))
156}
157
158fn refused(dealer: &Address, reason: String) -> MpcError {
159    MpcError::StoredRowRefused {
160        dealer: *dealer,
161        reason,
162    }
