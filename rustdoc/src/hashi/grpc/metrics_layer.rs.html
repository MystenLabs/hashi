4//! Tower callback layer plumbing for HTTP/gRPC metrics.
5//!
6//! This module owns every [`sui_http::middleware::callback`] handler the
7//! crate uses. A single [`RpcMetricsMakeCallbackHandler`] type drives
8//! both the server-side handler attached to the axum router in
9//! `crate::grpc::HttpService::start` and the client-side handler attached
10//! to outbound `tonic_rustls::Channel`s in `crate::grpc::client::Client`;
11//! the per-role behavior is selected via the [`Role`] passed at
12//! construction. The actual prometheus registration lives in
13//! [`crate::metrics::Metrics`]; this module is purely the wiring that
14//! observes traffic and writes into those metrics.
15

30/// Which RFC 9110 role this node is playing on the observed connection.
31///
32/// Selects the value written to the `role` label dimension and the
33/// direction each body travels relative to this node.
34#[derive(Clone, Copy, Debug)]
35pub enum Role {
36    Client,
37    Server,
38}
39
40impl Role {
41    /// Value for the `role` label dimension on every metric this module
42    /// writes.
43    fn label(self) -> &'static str {
44        match self {
45            Role::Client => "client",
46            Role::Server => "server",
47        }
48    }
49
50    /// Direction the request body travels from this node's perspective:
51    /// the client sends the request (outbound), the server receives it
52    /// (inbound).
53    fn request_direction(self) -> Direction {
54        match self {
55            Role::Client => Direction::Outbound,
56            Role::Server => Direction::Inbound,
57        }
58    }
59
60    /// Direction the response body travels from this node's perspective;
61    /// the inverse of [`Self::request_direction`].
62    fn response_direction(self) -> Direction {
63        match self {
64            Role::Client => Direction::Inbound,
65            Role::Server => Direction::Outbound,
66        }
67    }
68}
69
70/// Direction of HTTP body bytes from the perspective of this node.
71/// Selects which byte counter (`bytes_sent_total` vs
72/// `bytes_received_total`) the observed bytes are flushed to, and —
73/// together with [`Role`] — which body (request vs response) the tally
74/// was tracking.
75#[derive(Clone, Copy, Debug)]
76enum Direction {
77    Outbound,
78    Inbound,
79}
80
81/// Accumulates HTTP body bytes and, when the handler is dropped, flushes
82/// the total to both the per-path size histogram and the aggregate
83/// sent/received byte counter.
84///
85/// Flushing on `Drop` covers both the clean end-of-stream path and the
86/// partial-body path (cancellation, body error) in one place, so bytes
87/// that were actually transferred are always observed.
88struct SizeTally {
89    metrics: Arc<Metrics>,
90    path: Cow<'static, str>,
91    role: Role,
92    direction: Direction,
93    /// MPC protocol label
94    mpc_protocol: Option<&'static str>,
95    bytes: u64,
96}
97
98impl SizeTally {
99    fn new(
100        metrics: Arc<Metrics>,
101        path: Cow<'static, str>,
102        role: Role,
103        direction: Direction,
104        mpc_protocol: Option<&'static str>,
105    ) -> Self {
106        Self {
107            metrics,
108            path,
109            role,
110            direction,
111            mpc_protocol,
112            bytes: 0,
113        }
114    }
115
116    fn add(&mut self, n: usize) {
117        self.bytes += n as u64;
118    }
119}
120
121impl Drop for SizeTally {
122    fn drop(&mut self) {
123        let labels = &[self.path.as_ref(), self.role.label()];
124
125        // (role, direction) uniquely determines whether this tally was
126        // tracking the request or the response body:
127        //   client + outbound = request sent; server + inbound = request received
128        //   client + inbound  = response received; server + outbound = response sent
129        let is_request = matches!(
130            (self.role, self.direction),
131            (Role::Client, Direction::Outbound) | (Role::Server, Direction::Inbound)
132        );
133        let size_histogram = if is_request {
134            &self.metrics.request_size_bytes
135        } else {
136            &self.metrics.response_size_bytes
137        };
138        size_histogram
139            .with_label_values(labels)
140            .observe(self.bytes as f64);
141
142        let bytes_counter = match self.direction {
143            Direction::Outbound => &self.metrics.bytes_sent_total,
144            Direction::Inbound => &self.metrics.bytes_received_total,
145        };
146        bytes_counter.with_label_values(labels).inc_by(self.bytes);
147
148        if let Some(protocol) = self.mpc_protocol {
149            let mpc_labels = &[protocol];
150            let mpc_size_histogram = if is_request {
151                &self.metrics.mpc_request_size_bytes
152            } else {
153                &self.metrics.mpc_response_size_bytes
154            };
155            mpc_size_histogram
156                .with_label_values(mpc_labels)
157                .observe(self.bytes as f64);
158            let mpc_counter = match self.direction {
159                Direction::Outbound => &self.metrics.mpc_bytes_sent_total,
160                Direction::Inbound => &self.metrics.mpc_bytes_received_total,
161            };
162            mpc_counter.with_label_values(mpc_labels).inc_by(self.bytes);
163        }
164    }
165}

167/// Single factory for the per-request handler pair used on both ends of
168/// the wire. Construct with [`Self::server`] for the axum-side
169/// CallbackLayer or [`Self::client`] for the tonic-channel-side
170/// CallbackLayer.
171#[derive(Clone)]
172pub struct RpcMetricsMakeCallbackHandler {
173    metrics: Arc<Metrics>,
174    role: Role,
175}
176
177impl RpcMetricsMakeCallbackHandler {
178    pub fn new(metrics: Arc<Metrics>, role: Role) -> Self {
179        Self { metrics, role }
180    }
181
182    pub fn server(metrics: Arc<Metrics>) -> Self {
183        Self::new(metrics, Role::Server)
184    }
185
186    pub fn client(metrics: Arc<Metrics>) -> Self {
187        Self::new(metrics, Role::Client)
188    }
189}
190
191impl callback::MakeCallbackHandler for RpcMetricsMakeCallbackHandler {
192    type RequestHandler = RequestHandler;
193    type ResponseHandler = ResponseHandler;
194
195    fn make_handler(
196        &self,
197        request: &request::Parts,
198    ) -> (Self::RequestHandler, Self::ResponseHandler) {
199        let metrics = self.metrics.clone();
200        let label = self.role.label();
201        let path = extract_path(request, self.role);
202        let mpc_protocol = extract_mpc_protocol(request);
203
204        metrics
205            .inflight_requests
206            .with_label_values(&[path.as_ref(), label])
207            .inc();
208
209        let request_handler = RequestHandler {
210            tally: SizeTally::new(
211                metrics.clone(),
212                path.clone(),
213                self.role,
214                self.role.request_direction(),
215                mpc_protocol,
216            ),
217        };
218        let response_handler = ResponseHandler {
219            tally: SizeTally::new(
220                metrics.clone(),
221                path.clone(),
222                self.role,
223                self.role.response_direction(),
224                mpc_protocol,
225            ),
226            metrics,
227            role: self.role,
228            path,
229            start: Instant::now(),
230            counted_response: false,
231        };
232        (request_handler, response_handler)
233    }
234}
235
236fn extract_mpc_protocol(request: &request::Parts) -> Option<&'static str> {
237    let value = request
238        .headers
239        .get(crate::grpc::MPC_PROTOCOL_METADATA_KEY)?
240        .to_str()
241        .ok()?;
242    match value {
243        crate::metrics::MPC_LABEL_DKG => Some(crate::metrics::MPC_LABEL_DKG),
244        crate::metrics::MPC_LABEL_KEY_ROTATION => Some(crate::metrics::MPC_LABEL_KEY_ROTATION),
245        crate::metrics::MPC_LABEL_NONCE_GENERATION => {
246            Some(crate::metrics::MPC_LABEL_NONCE_GENERATION)
247        }
248        crate::metrics::MPC_LABEL_SIGNING => Some(crate::metrics::MPC_LABEL_SIGNING),
249        _ => None,
250    }
251}

253/// Request-side handler. Tallies request body bytes; path-level metrics
254/// live on [`ResponseHandler`].
255pub struct RequestHandler {
256    tally: SizeTally,
257}
258
259impl callback::RequestHandler for RequestHandler {
260    fn on_body_chunk<B: Buf>(&mut self, chunk: &B) {
261        self.tally.add(chunk.remaining());
262    }
263}

265/// Response-side handler. Records path-level RPC metrics
266/// (`inflight_requests`, `request_latency`, the per-status `requests`
267/// counter) labeled by `role`, and tallies response body bytes.
268pub struct ResponseHandler {
269    metrics: Arc<Metrics>,
270    role: Role,
271    path: Cow<'static, str>,
272    start: Instant,
273    // Indicates if we successfully counted the response. In some cases when a request is
274    // prematurely canceled this will remain false.
275    counted_response: bool,
276    tally: SizeTally,
277}
278
279impl callback::ResponseHandler for ResponseHandler {
280    fn on_response(&mut self, response: &response::Parts) {
281        const GRPC_STATUS: http::HeaderName = http::HeaderName::from_static("grpc-status");
282
283        let status = if super::is_grpc_content_type(&response.headers) {
284            let code = response
285                .headers
286                .get(&GRPC_STATUS)
287                .map(http::HeaderValue::as_bytes)
288                .map(tonic::Code::from_bytes)
289                .unwrap_or(tonic::Code::Ok);
290
291            code_as_str(code)
292        } else {
293            response.status.as_str()
294        };
295
296        self.metrics
297            .requests
298            .with_label_values(&[self.path.as_ref(), status, self.role.label()])
299            .inc();
300
301        self.counted_response = true;
302    }
303
304    fn on_service_error<E>(&mut self, _error: &E)
305    where
306        E: std::fmt::Display + 'static,
307    {
308        // Only reachable on the client role: axum requires server
309        // services to be infallible (Error = Infallible), so no service
310        // error can propagate to the server handler. Record under a
311        // distinct status so service-level failures are visible and
312        // aren't lumped in with canceled requests by the `Drop` fallback.
313        self.metrics
314            .requests
315            .with_label_values(&[self.path.as_ref(), "service-error", self.role.label()])
316            .inc();
317        self.counted_response = true;
318    }
319
320    fn on_body_chunk<B: Buf>(&mut self, chunk: &B) {
321        self.tally.add(chunk.remaining());
322    }
323}
324
325impl Drop for ResponseHandler {
326    fn drop(&mut self) {
327        let label = self.role.label();
328
329        self.metrics
330            .inflight_requests
331            .with_label_values(&[self.path.as_ref(), label])
332            .dec();
333
334        let latency = self.start.elapsed().as_secs_f64();
335        self.metrics
336            .request_latency
337            .with_label_values(&[self.path.as_ref(), label])
338            .observe(latency);
339
340        if !self.counted_response {
341            self.metrics
342                .requests
343                .with_label_values(&[self.path.as_ref(), "canceled", label])
344                .inc();
345        }
346    }
347}

349fn extract_path(request: &request::Parts, role: Role) -> Cow<'static, str> {
350    match role {
351        Role::Server => extract_server_path(request),
352        Role::Client => extract_client_path(request),
353    }
354}

356fn known_grpc_paths() -> &'static HashSet<Box<str>> {
357    static PATHS: OnceLock<HashSet<Box<str>>> = OnceLock::new();
358    PATHS.get_or_init(|| {
359        use prost::Message as _;
360
361        let mut paths = HashSet::new();
362        for encoded in [
363            hashi_types::proto::FILE_DESCRIPTOR_SET,
364            tonic_health::pb::FILE_DESCRIPTOR_SET,
365            tonic_reflection::pb::v1::FILE_DESCRIPTOR_SET,
366            tonic_reflection::pb::v1alpha::FILE_DESCRIPTOR_SET,
367        ] {
368            let Ok(set) = prost_types::FileDescriptorSet::decode(encoded) else {
369                continue;
370            };
371            for file in set.file {
372                let package = file.package();
373                for service in &file.service {
374                    for method in &service.method {
375                        paths.insert(
376                            format!("/{package}.{}/{}", service.name(), method.name())
377                                .into_boxed_str(),
378                        );
379                    }
380                }
381            }
382        }
383        paths
384    })
385}
386
387fn extract_server_path(request: &request::Parts) -> Cow<'static, str> {
388    if let Some(matched_path) = request.extensions.get::<axum::extract::MatchedPath>() {
389        if super::is_grpc_content_type(&request.headers) {
390            match known_grpc_paths().get(request.uri.path()) {
391                Some(path) => Cow::Borrowed(path),
392                None => Cow::Owned(matched_path.as_str().to_owned()),
393            }
394        } else {
395            Cow::Owned(matched_path.as_str().to_owned())
396        }
397    } else {
398        Cow::Borrowed("unknown")
399    }
400}

402/// Build the client-side path label from tonic's [`GrpcMethod`]
403/// extension. Tonic-generated clients populate it with the gRPC service
404/// and method name before the request reaches our tower stack, giving us
405/// a stable label that matches the URI path the server sees. Falls back
406/// to the raw URI path for non-tonic callers.
407///
408/// [`GrpcMethod`]: tonic::GrpcMethod
409fn extract_client_path(request: &request::Parts) -> Cow<'static, str> {
410    if let Some(method) = request.extensions.get::<tonic::GrpcMethod<'static>>() {
411        Cow::Owned(format!("/{}/{}", method.service(), method.method()))
412    } else {
413        Cow::Owned(request.uri.path().to_owned())
414    }
415}

417fn code_as_str(code: tonic::Code) -> &'static str {
418    match code {
419        tonic::Code::Ok => "ok",
420        tonic::Code::Cancelled => "canceled",
421        tonic::Code::Unknown => "unknown",
422        tonic::Code::InvalidArgument => "invalid-argument",
423        tonic::Code::DeadlineExceeded => "deadline-exceeded",
424        tonic::Code::NotFound => "not-found",
425        tonic::Code::AlreadyExists => "already-exists",
426        tonic::Code::PermissionDenied => "permission-denied",
427        tonic::Code::ResourceExhausted => "resource-exhausted",
428        tonic::Code::FailedPrecondition => "failed-precondition",
429        tonic::Code::Aborted => "aborted",
430        tonic::Code::OutOfRange => "out-of-range",
431        tonic::Code::Unimplemented => "unimplemented",
432        tonic::Code::Internal => "internal",
433        tonic::Code::Unavailable => "unavailable",
434        tonic::Code::DataLoss => "data-loss",
435        tonic::Code::Unauthenticated => "unauthenticated",
436    }
437}
