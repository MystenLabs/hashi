29/// Wrapper that triggers graceful HTTP server shutdown on drop.
30///
31/// The HTTP server is spawned on a detached tokio task by `sui_http::Builder::serve`.
32/// This guard ensures that `trigger_shutdown` is called when the owning `Service` is
33/// dropped (whether via explicit `Service::shutdown()` or implicit drop), so the server's
34/// accept loop breaks and in-flight connections are drained.
35struct ServerHandleGuard(Arc<ServerHandle>);
36
37impl Drop for ServerHandleGuard {
38    fn drop(&mut self) {
39        self.0.trigger_shutdown();
40    }
41}

62    pub async fn start(self) -> (std::net::SocketAddr, Service) {
63        let router = {
64            let max_decoding_message_size = self.inner.config.grpc_max_decoding_message_size();
65            let bridge_service =
66                hashi_types::proto::bridge_service_server::BridgeServiceServer::new(self.clone())
67                    .max_decoding_message_size(max_decoding_message_size);
68            let mpc_service =
69                hashi_types::proto::mpc_service_server::MpcServiceServer::new(self.clone())
70                    .max_decoding_message_size(max_decoding_message_size);
71
72            let (health_reporter, health_service) = tonic_health::server::health_reporter();
73
74            let mut reflection_v1 = tonic_reflection::server::Builder::configure();
75            let mut reflection_v1alpha = tonic_reflection::server::Builder::configure();
76
77            for file_descriptor_set in [
78                sui_rpc::proto::google::protobuf::FILE_DESCRIPTOR_SET,
79                sui_rpc::proto::google::rpc::FILE_DESCRIPTOR_SET,
80                tonic_health::pb::FILE_DESCRIPTOR_SET,
81                hashi_types::proto::FILE_DESCRIPTOR_SET,
82            ] {
83                reflection_v1 =
84                    reflection_v1.register_encoded_file_descriptor_set(file_descriptor_set);
85                reflection_v1alpha =
86                    reflection_v1alpha.register_encoded_file_descriptor_set(file_descriptor_set);
87            }
88
89            let reflection_v1 = reflection_v1.build_v1().unwrap();
90            let reflection_v1alpha = reflection_v1alpha.build_v1alpha().unwrap();
91
92            fn service_name<S: tonic::server::NamedService>(_service: &S) -> &'static str {
93                S::NAME
94            }
95
96            for service_name in [
97                service_name(&bridge_service),
98                service_name(&mpc_service),
99                service_name(&reflection_v1),
100                service_name(&reflection_v1alpha),
101            ] {
102                health_reporter
103                    .set_service_status(service_name, tonic_health::ServingStatus::Serving)
104                    .await;
105            }
106
107            axum::Router::new()
108                .add_grpc_service(bridge_service)
109                .add_grpc_service(mpc_service)
110                .add_grpc_service(reflection_v1)
111                .add_grpc_service(reflection_v1alpha)
112                .add_grpc_service(health_service)
113        };
114
115        let hashi_for_ready = self.inner.clone();
116        let health_endpoint = axum::Router::new()
117            .route("/health", axum::routing::get(health))
118            .route(
119                "/ready",
120                axum::routing::get(move || ready(hashi_for_ready.clone())),
121            );
122
123        let limiter = peer_limit::PeerInflightLimiter::new(
124            self.inner.config.grpc_per_peer_inflight_limit(),
125            self.inner.metrics.clone(),
126        );
127        let layers = ServiceBuilder::new()
128            .layer(axum::middleware::from_fn_with_state(
129                self.inner.clone(),
130                require_known_validator,
131            ))
132            .layer(axum::middleware::from_fn_with_state(
133                limiter,
134                peer_limit::limit_per_peer,
135            ))
136            .layer(sui_http::middleware::callback::CallbackLayer::new(
137                metrics_layer::RpcMetricsMakeCallbackHandler::server(self.inner.metrics.clone()),
138            ));
139
140        let router = router.merge(health_endpoint).layer(layers);
141
142        let tls_config =
143            crate::tls::make_server_config(self.inner.config.tls_private_key().unwrap());
144        // let tls_config =
145        //     crate::tls_rpk::make_server_config(self.inner.config.tls_private_key().unwrap());
146
147        let server_handle = Arc::new(
148            sui_http::Builder::new()
149                // Recycle peer connections via a graceful max-age drain: a
150                // long-lived multiplexed HTTP/2 connection accumulates state that
151                // trips a connection-level GoAway(PROTOCOL_ERROR) under sustained
152                // load, failing every in-flight request to that peer at once.
153                .config(
154                    sui_http::Config::default()
155                        .max_concurrent_streams(crate::config::DEFAULT_GRPC_PER_PEER_INFLIGHT_LIMIT)
156                        .max_connection_age(std::time::Duration::from_secs(120))
157                        .max_connection_age_grace(std::time::Duration::from_secs(120))
158                        .http2_keepalive_interval(Some(std::time::Duration::from_secs(30))),
159                )
160                .tls_config(tls_config)
161                .serve(self.inner.config.listen_address(), router)
162                .unwrap(),
163        );
164        let local_addr = *server_handle.local_addr();
165
166        (local_addr, supervise(server_handle))
167    }

202// A server that stops without a shutdown signal has crashed (one HTTP/1 handler
203// panic ends sui-http's accept loop), so fail rather than run on without it.
204fn supervise(server_handle: Arc<ServerHandle>) -> Service {
205    let guard = ServerHandleGuard(server_handle);
206    Service::new().spawn_aborting(async move {
207        guard.0.wait_for_shutdown().await;
208        anyhow::bail!("HTTP server stopped unexpectedly")
209    })
210}

216async fn ready(hashi: Arc<Hashi>) -> impl axum::response::IntoResponse {
217    let Some(onchain_state) = hashi.onchain_state_opt() else {
218        return (
219            http::StatusCode::SERVICE_UNAVAILABLE,
220            "on-chain state not yet initialized",
221        );
222    };
223    // If the chain has moved past what this binary supports, report not-ready so
224    // operators/orchestration surface it — the node has halted autonomous work
225    // (see leader gate) and needs a binary upgrade. Details are in the
226    // `hashi_package_version_unsupported` metric and the leader log.
227    if matches!(
228        onchain_state.autonomous_halt_reason(),
229        Some(crate::onchain::HaltReason::BinaryUnsupported { .. })
230    ) {
231        return (
232            http::StatusCode::SERVICE_UNAVAILABLE,
233            "binary does not support the enabled on-chain package version(s) — upgrade required",
234        );
235    }
236    let epoch = onchain_state.epoch();
237    // Treat genesis (no committee formed yet) as ready so OrderedReady can bring
238    // up the rest of the StatefulSet to register keys and run the initial DKG.
239    let awaiting_genesis = epoch == 0 && onchain_state.current_committee().is_none();
240    if awaiting_genesis {
241        (http::StatusCode::OK, "ready (awaiting genesis)")
242    } else if hashi.signing_manager_for(epoch).is_some() {
243        (http::StatusCode::OK, "ready")
244    } else {
245        (
246            http::StatusCode::SERVICE_UNAVAILABLE,
247            "SigningManager not yet initialized",
248        )
249    }
250}

285const ANONYMOUS_PATHS: &[&str] = &["/health", "/ready"];
286
287async fn require_known_validator(
288    axum::extract::State(hashi): axum::extract::State<Arc<Hashi>>,
289    mut request: axum::extract::Request,
290    next: axum::middleware::Next,
291) -> axum::response::Response {
292    match lookup_validator_address(&hashi, &request) {
293        Ok(validator_address) => {
294            request.extensions_mut().insert(validator_address);
295        }
296        Err(_) if ANONYMOUS_PATHS.contains(&request.uri().path()) => {}
297        Err(reason) => {
298            hashi
299                .metrics
300                .unknown_caller_refused_total
301                .with_label_values(&[reason.as_str()])
302                .inc();
303            return refuse(&request);
304        }
305    }
306    next.run(request).await
307}
308
309enum RefusalReason {
310    NoClientCert,
311    StateUnavailable,
312    NotRegistered,
313}
314
315impl RefusalReason {
316    fn as_str(&self) -> &'static str {
317        match self {
318            Self::NoClientCert => "no_client_cert",
319            Self::StateUnavailable => "state_unavailable",
320            Self::NotRegistered => "not_registered",
321        }
322    }
323}
324
325fn refuse<B>(request: &http::Request<B>) -> axum::response::Response {
326    if is_grpc_content_type(request.headers()) {
327        tonic::Status::permission_denied("unknown validator").into_http()
328    } else {
329        axum::response::IntoResponse::into_response((
330            http::StatusCode::FORBIDDEN,
331            "unknown validator",
332        ))
333    }
334}

336pub(super) fn is_grpc_content_type(headers: &http::HeaderMap) -> bool {
337    headers
338        .get(&http::header::CONTENT_TYPE)
339        .is_some_and(|header| {
340            header
341                .as_bytes()
342                .starts_with(tonic::metadata::GRPC_CONTENT_TYPE.as_bytes())
343        })
344}
345
346fn lookup_validator_address<B>(
347    hashi: &Hashi,
348    request: &http::Request<B>,
349) -> Result<sui_sdk_types::Address, RefusalReason> {
350    let tls_public_key = request
351        .extensions()
352        .get::<sui_http::PeerCertificates>()
353        .and_then(|peer_certs| peer_certs.peer_certs().first())
354        .and_then(|cert| crate::tls::public_key_from_certificate(cert).ok())
355        .ok_or(RefusalReason::NoClientCert)?;
356    hashi
357        .onchain_state_opt()
358        .ok_or(RefusalReason::StateUnavailable)?
359        .state()
360        .hashi()
361        .committees
362        .lookup_address_by_tls_public_key(&tls_public_key)
363        .ok_or(RefusalReason::NotRegistered)
364}
