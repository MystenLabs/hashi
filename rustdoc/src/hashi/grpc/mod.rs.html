39/// Wrapper that triggers graceful HTTP server shutdown on drop.
40///
41/// The HTTP server is spawned on a detached tokio task by `sui_http::Builder::serve`.
42/// This guard ensures that `trigger_shutdown` is called when the owning `Service` is
43/// dropped (whether via explicit `Service::shutdown()` or implicit drop), so the server's
44/// accept loop breaks and in-flight connections are drained.
45struct ServerHandleGuard(Arc<ServerHandle>);
46
47impl Drop for ServerHandleGuard {
48    fn drop(&mut self) {
49        self.0.trigger_shutdown();
50    }
51}

87    pub async fn start(self) -> (std::net::SocketAddr, Service) {
88        let mut routes = route_limits::RouteLimits::default();
89        let router = {
90            use hashi_types::proto::bridge_service_server;
91            use hashi_types::proto::mpc_service_server;
92
93            let max_decoding_message_size = self.inner.config.grpc_max_decoding_message_size();
94            let bridge_service = bridge_service_server::BridgeServiceServer::new(self.clone())
95                .max_decoding_message_size(routes.service(
96                    bridge_service_server::SERVICE_NAME,
97                    max_decoding_message_size.min(route_limits::BRIDGE_REQUEST_LIMIT),
98                ));
99            let mpc_service = mpc_service_server::MpcServiceServer::new(self.clone())
100                .max_decoding_message_size(
101                    routes.service(mpc_service_server::SERVICE_NAME, max_decoding_message_size),
102                );
103            for method in route_limits::SMALL_MPC_METHODS {
104                routes.method(
105                    mpc_service_server::SERVICE_NAME,
106                    method,
107                    route_limits::SMALL_MPC_REQUEST_LIMIT,
108                );
109            }
110            for method in route_limits::MPC_WORK_METHODS {
111                routes.mpc_work(mpc_service_server::SERVICE_NAME, method);
112            }
113
114            let (health_reporter, health_service) = tonic_health::server::health_reporter();
115
116            let mut reflection_v1 = tonic_reflection::server::Builder::configure();
117            let mut reflection_v1alpha = tonic_reflection::server::Builder::configure();
118
119            for file_descriptor_set in [
120                sui_rpc::proto::google::protobuf::FILE_DESCRIPTOR_SET,
121                sui_rpc::proto::google::rpc::FILE_DESCRIPTOR_SET,
122                tonic_health::pb::FILE_DESCRIPTOR_SET,
123                hashi_types::proto::FILE_DESCRIPTOR_SET,
124            ] {
125                reflection_v1 =
126                    reflection_v1.register_encoded_file_descriptor_set(file_descriptor_set);
127                reflection_v1alpha =
128                    reflection_v1alpha.register_encoded_file_descriptor_set(file_descriptor_set);
129            }
130
131            let reflection_v1 = reflection_v1.build_v1().unwrap();
132            let reflection_v1alpha = reflection_v1alpha.build_v1alpha().unwrap();
133
134            fn service_name<S: tonic::server::NamedService>(_service: &S) -> &'static str {
135                S::NAME
136            }
137
138            let limit = route_limits::HEALTH_DECODE_LIMIT;
139            let name = service_name(&health_service);
140            let health_service =
141                health_service.max_decoding_message_size(routes.service(name, limit));
142            let limit = route_limits::REFLECTION_REQUEST_LIMIT;
143            let name = service_name(&reflection_v1);
144            let reflection_v1 =
145                reflection_v1.max_decoding_message_size(routes.service(name, limit));
146            let name = service_name(&reflection_v1alpha);
147            let reflection_v1alpha =
148                reflection_v1alpha.max_decoding_message_size(routes.service(name, limit));
149
150            for service_name in [
151                service_name(&bridge_service),
152                service_name(&mpc_service),
153                service_name(&reflection_v1),
154                service_name(&reflection_v1alpha),
155            ] {
156                health_reporter
157                    .set_service_status(service_name, tonic_health::ServingStatus::Serving)
158                    .await;
159            }
160
161            axum::Router::new()
162                .add_grpc_service(bridge_service, &routes)
163                .add_grpc_service(mpc_service, &routes)
164                .add_grpc_service(reflection_v1, &routes)
165                .add_grpc_service(reflection_v1alpha, &routes)
166                .add_grpc_service(health_service, &routes)
167        };
168
169        let hashi_for_ready = self.inner.clone();
170        let health_endpoint = axum::Router::new()
171            .route("/health", axum::routing::get(health))
172            .route(
173                "/ready",
174                axum::routing::get(move || ready(hashi_for_ready.clone())),
175            );
176
177        let budget_bytes = per_peer_byte_budget(
178            self.inner.config.grpc_per_peer_inflight_bytes(),
179            routes.largest(),
180        );
181        self.inner
182            .metrics
183            .peer_inflight_limit_bytes
184            .set(i64::try_from(budget_bytes).unwrap_or(i64::MAX));
185        let limiter = peer_limit::PeerInflightLimiter::new(
186            self.inner.config.grpc_per_peer_inflight_limit(),
187            budget_bytes,
188            routes,
189            self.inner.metrics.clone(),
190        );
191        let connections = connection_limit::ConnectionLimiter::new(
192            per_peer_connection_limit(self.inner.config.grpc_per_peer_connection_limit()),
193            self.inner.metrics.clone(),
194        );
195        let layers = ServiceBuilder::new()
196            .layer(axum::middleware::from_fn_with_state(
197                self.inner.clone(),
198                require_known_validator,
199            ))
200            .layer(axum::middleware::from_fn_with_state(
201                connections.clone(),
202                connection_limit::limit_connections_per_peer,
203            ))
204            .layer(axum::middleware::from_fn_with_state(
205                limiter,
206                peer_limit::limit_per_peer,
207            ))
208            .layer(sui_http::middleware::callback::CallbackLayer::new(
209                metrics_layer::RpcMetricsMakeCallbackHandler::server(self.inner.metrics.clone()),
210            ));
211
212        let router = router.merge(health_endpoint).layer(layers);
213
214        let tls_config =
215            crate::tls::make_server_config(self.inner.config.tls_private_key().unwrap());
216        // let tls_config =
217        //     crate::tls_rpk::make_server_config(self.inner.config.tls_private_key().unwrap());
218
219        let server_handle = Arc::new(
220            sui_http::Builder::new()
221                // Recycle peer connections via a graceful max-age drain: a
222                // long-lived multiplexed HTTP/2 connection accumulates state that
223                // trips a connection-level GoAway(PROTOCOL_ERROR) under sustained
224                // load, failing every in-flight request to that peer at once.
225                .config(
226                    sui_http::Config::default()
227                        .max_concurrent_streams(crate::config::DEFAULT_GRPC_PER_PEER_INFLIGHT_LIMIT)
228                        .http2_max_pending_accept_reset_streams(Some(
229                            crate::config::DEFAULT_GRPC_PER_PEER_INFLIGHT_LIMIT as usize,
230                        ))
231                        .max_connection_age(std::time::Duration::from_secs(120))
232                        .max_connection_age_grace(std::time::Duration::from_secs(120))
233                        .http2_keepalive_interval(Some(std::time::Duration::from_secs(30))),
234                )
235                .tls_config(tls_config)
236                .serve(self.inner.config.listen_address(), router)
237                .unwrap(),
238        );
239        connections.set_server(server_handle.clone());
240        let local_addr = *server_handle.local_addr();
241
242        (local_addr, supervise(server_handle))
243    }

286fn per_peer_byte_budget(configured: Option<u64>, largest_decode_limit: usize) -> u64 {
287    let floor = u64::try_from(largest_decode_limit)
288        .unwrap_or(u64::MAX)
289        .saturating_add(request_body::PREFIX_LEN)
290        .saturating_mul(2);
291    match configured {
292        Some(bytes) if bytes < floor => {
293            tracing::warn!(
294                "grpc_per_peer_inflight_bytes {bytes} is below room for two maximum-size \
295                 requests; using {floor}"
296            );
297            floor
298        }
299        Some(bytes) => bytes,
300        None => floor,
301    }
302}
303
304fn per_peer_connection_limit(configured: Option<usize>) -> usize {
305    const FLOOR: usize = 3;
306    let limit = configured.unwrap_or(crate::config::DEFAULT_GRPC_PER_PEER_CONNECTION_LIMIT);
307    if limit < FLOOR {
308        tracing::warn!("grpc_per_peer_connection_limit {limit} is below {FLOOR}; using {FLOOR}");
309    }
310    limit.max(FLOOR)
311}

313fn unavailable<B>(request: &http::Request<B>, message: &'static str) -> axum::response::Response {
314    if is_grpc_content_type(request.headers()) {
315        tonic::Status::unavailable(message).into_http()
316    } else {
317        axum::response::IntoResponse::into_response((
318            http::StatusCode::SERVICE_UNAVAILABLE,
319            message,
320        ))
321    }
322}

324// A server that stops without a shutdown signal has crashed (one HTTP/1 handler
325// panic ends sui-http's accept loop), so fail rather than run on without it.
326fn supervise(server_handle: Arc<ServerHandle>) -> Service {
327    let guard = ServerHandleGuard(server_handle);
328    Service::new().spawn_aborting(async move {
329        guard.0.wait_for_shutdown().await;
330        anyhow::bail!("HTTP server stopped unexpectedly")
331    })
332}

338async fn ready(hashi: Arc<Hashi>) -> impl axum::response::IntoResponse {
339    let Some(onchain_state) = hashi.onchain_state_opt() else {
340        return (
341            http::StatusCode::SERVICE_UNAVAILABLE,
342            "on-chain state not yet initialized",
343        );
344    };
345    // If the chain has moved past what this binary supports, report not-ready so
346    // operators/orchestration surface it — the node has halted autonomous work
347    // (see leader gate) and needs a binary upgrade. Details are in the
348    // `hashi_package_version_unsupported` metric and the leader log.
349    if matches!(
350        onchain_state.autonomous_halt_reason(),
351        Some(crate::onchain::HaltReason::BinaryUnsupported { .. })
352    ) {
353        return (
354            http::StatusCode::SERVICE_UNAVAILABLE,
355            "binary does not support the enabled on-chain package version(s) — upgrade required",
356        );
357    }
358    let epoch = onchain_state.epoch();
359    // Treat genesis (no committee formed yet) as ready so OrderedReady can bring
360    // up the rest of the StatefulSet to register keys and run the initial DKG.
361    let awaiting_genesis = epoch == 0 && onchain_state.current_committee().is_none();
362    if awaiting_genesis {
363        (http::StatusCode::OK, "ready (awaiting genesis)")
364    } else if hashi.signing_manager_for(epoch).is_some() {
365        (http::StatusCode::OK, "ready")
366    } else {
367        (
368            http::StatusCode::SERVICE_UNAVAILABLE,
369            "SigningManager not yet initialized",
370        )
371    }
372}

389impl RouterExt for axum::Router {
390    fn add_grpc_service<S>(self, svc: S, routes: &route_limits::RouteLimits) -> Self
391    where
392        S: tower::Service<
393                axum::extract::Request,
394                Response: axum::response::IntoResponse,
395                Error = std::convert::Infallible,
396            > + tonic::server::NamedService
397            + Clone
398            + Send
399            + Sync
400            + 'static,
401        S::Future: Send + 'static,
402    {
403        assert!(
404            routes.has_service(S::NAME),
405            "{} is served without a decode limit",
406            S::NAME
407        );
408        self.route_service(&format!("/{}/{{*rest}}", S::NAME), svc)
409    }
410}

412const ANONYMOUS_PATHS: &[&str] = &["/health", "/ready"];
413
414async fn require_known_validator(
415    axum::extract::State(hashi): axum::extract::State<Arc<Hashi>>,
416    mut request: axum::extract::Request,
417    next: axum::middleware::Next,
418) -> axum::response::Response {
419    match lookup_validator_address(&hashi, &request) {
420        Ok(validator_address) => {
421            request.extensions_mut().insert(validator_address);
422        }
423        Err(_) if ANONYMOUS_PATHS.contains(&request.uri().path()) => {}
424        Err(reason) => {
425            hashi
426                .metrics
427                .unknown_caller_refused_total
428                .with_label_values(&[reason.as_str()])
429                .inc();
430            return refuse(&request);
431        }
432    }
433    next.run(request).await
434}

436enum RefusalReason {
437    NoClientCert,
438    StateUnavailable,
439    NotRegistered,
440}

442impl RefusalReason {
443    fn as_str(&self) -> &'static str {
444        match self {
445            Self::NoClientCert => "no_client_cert",
446            Self::StateUnavailable => "state_unavailable",
447            Self::NotRegistered => "not_registered",
448        }
449    }
450}

452fn refuse<B>(request: &http::Request<B>) -> axum::response::Response {
453    if is_grpc_content_type(request.headers()) {
454        tonic::Status::permission_denied("unknown validator").into_http()
455    } else {
456        axum::response::IntoResponse::into_response((
457            http::StatusCode::FORBIDDEN,
458            "unknown validator",
459        ))
460    }
461}

463pub(super) fn is_grpc_content_type(headers: &http::HeaderMap) -> bool {
464    headers
465        .get(&http::header::CONTENT_TYPE)
466        .is_some_and(|header| {
467            header
468                .as_bytes()
469                .starts_with(tonic::metadata::GRPC_CONTENT_TYPE.as_bytes())
470        })
471}
472
473fn lookup_validator_address<B>(
474    hashi: &Hashi,
475    request: &http::Request<B>,
476) -> Result<sui_sdk_types::Address, RefusalReason> {
477    let tls_public_key = request
478        .extensions()
479        .get::<sui_http::PeerCertificates>()
480        .and_then(|peer_certs| peer_certs.peer_certs().first())
481        .and_then(|cert| crate::tls::public_key_from_certificate(cert).ok())
482        .ok_or(RefusalReason::NoClientCert)?;
483    hashi
484        .onchain_state_opt()
485        .ok_or(RefusalReason::StateUnavailable)?
486        .state()
487        .hashi()
488        .committees
489        .lookup_address_by_tls_public_key(&tls_public_key)
490        .ok_or(RefusalReason::NotRegistered)
491}
