41/// Wrapper that triggers graceful HTTP server shutdown on drop.
42///
43/// The HTTP server is spawned on a detached tokio task by `sui_http::Builder::serve`.
44/// This guard ensures that `trigger_shutdown` is called when the owning `Service` is
45/// dropped (whether via explicit `Service::shutdown()` or implicit drop), so the server's
46/// accept loop breaks and in-flight connections are drained.
47struct ServerHandleGuard(Arc<ServerHandle>);
48
49impl Drop for ServerHandleGuard {
50    fn drop(&mut self) {
51        self.0.trigger_shutdown();
52    }
53}

92    pub async fn start(self) -> (std::net::SocketAddr, Service) {
93        let mut routes = route_limits::RouteLimits::default();
94        let router = {
95            use hashi_types::proto::bridge_service_server;
96            use hashi_types::proto::mpc_service_server;
97
98            let max_decoding_message_size = self.inner.config.grpc_max_decoding_message_size();
99            let bridge_service = bridge_service_server::BridgeServiceServer::new(self.clone())
100                .max_decoding_message_size(routes.service(
101                    bridge_service_server::SERVICE_NAME,
102                    max_decoding_message_size.min(route_limits::BRIDGE_REQUEST_LIMIT),
103                ));
104            let mpc_service = mpc_service_server::MpcServiceServer::new(self.clone())
105                .max_decoding_message_size(
106                    routes.service(mpc_service_server::SERVICE_NAME, max_decoding_message_size),
107                );
108            for method in route_limits::SMALL_MPC_METHODS {
109                routes.method(
110                    mpc_service_server::SERVICE_NAME,
111                    method,
112                    route_limits::SMALL_MPC_REQUEST_LIMIT,
113                );
114            }
115            for method in route_limits::MPC_WORK_METHODS {
116                routes.mpc_work(mpc_service_server::SERVICE_NAME, method);
117            }
118
119            let (health_reporter, health_service) = tonic_health::server::health_reporter();
120
121            let mut reflection_v1 = tonic_reflection::server::Builder::configure();
122            let mut reflection_v1alpha = tonic_reflection::server::Builder::configure();
123
124            for file_descriptor_set in [
125                sui_rpc::proto::google::protobuf::FILE_DESCRIPTOR_SET,
126                sui_rpc::proto::google::rpc::FILE_DESCRIPTOR_SET,
127                tonic_health::pb::FILE_DESCRIPTOR_SET,
128                hashi_types::proto::FILE_DESCRIPTOR_SET,
129            ] {
130                reflection_v1 =
131                    reflection_v1.register_encoded_file_descriptor_set(file_descriptor_set);
132                reflection_v1alpha =
133                    reflection_v1alpha.register_encoded_file_descriptor_set(file_descriptor_set);
134            }
135
136            let reflection_v1 = reflection_v1.build_v1().unwrap();
137            let reflection_v1alpha = reflection_v1alpha.build_v1alpha().unwrap();
138
139            fn service_name<S: tonic::server::NamedService>(_service: &S) -> &'static str {
140                S::NAME
141            }
142
143            let limit = route_limits::HEALTH_DECODE_LIMIT;
144            let name = service_name(&health_service);
145            let health_service =
146                health_service.max_decoding_message_size(routes.service(name, limit));
147            let limit = route_limits::REFLECTION_REQUEST_LIMIT;
148            let name = service_name(&reflection_v1);
149            let reflection_v1 =
150                reflection_v1.max_decoding_message_size(routes.service(name, limit));
151            let name = service_name(&reflection_v1alpha);
152            let reflection_v1alpha =
153                reflection_v1alpha.max_decoding_message_size(routes.service(name, limit));
154
155            for service_name in [
156                service_name(&bridge_service),
157                service_name(&mpc_service),
158                service_name(&reflection_v1),
159                service_name(&reflection_v1alpha),
160            ] {
161                health_reporter
162                    .set_service_status(service_name, tonic_health::ServingStatus::Serving)
163                    .await;
164            }
165
166            axum::Router::new()
167                .add_grpc_service(bridge_service, &routes)
168                .add_grpc_service(mpc_service, &routes)
169                .add_grpc_service(reflection_v1, &routes)
170                .add_grpc_service(reflection_v1alpha, &routes)
171                .add_grpc_service(health_service, &routes)
172        };
173
174        let hashi_for_ready = self.inner.clone();
175        let health_endpoint = axum::Router::new()
176            .route("/health", axum::routing::get(health))
177            .route(
178                "/ready",
179                axum::routing::get(move || ready(hashi_for_ready.clone())),
180            );
181
182        let budget_bytes = per_peer_byte_budget(
183            self.inner.config.grpc_per_peer_inflight_bytes(),
184            routes.largest(),
185        );
186        self.inner
187            .metrics
188            .peer_inflight_limit_bytes
189            .set(i64::try_from(budget_bytes).unwrap_or(i64::MAX));
190        let limiter = peer_limit::PeerInflightLimiter::new(
191            self.inner.config.grpc_per_peer_inflight_limit(),
192            budget_bytes,
193            routes,
194            self.inner.metrics.clone(),
195        );
196        let connections = connection_limit::ConnectionLimiter::new(
197            per_peer_connection_limit(self.inner.config.grpc_per_peer_connection_limit()),
198            self.inner.metrics.clone(),
199        );
200        let layers = ServiceBuilder::new()
201            .layer(axum::middleware::from_fn_with_state(
202                self.inner.clone(),
203                require_known_validator,
204            ))
205            .layer(axum::middleware::from_fn_with_state(
206                connections.clone(),
207                connection_limit::limit_connections_per_peer,
208            ))
209            .layer(axum::middleware::from_fn_with_state(
210                limiter,
211                peer_limit::limit_per_peer,
212            ))
213            .layer(sui_http::middleware::callback::CallbackLayer::new(
214                metrics_layer::RpcMetricsMakeCallbackHandler::server(self.inner.metrics.clone()),
215            ));
216
217        let router = router.merge(health_endpoint).layer(layers);
218
219        let tls_config =
220            crate::tls::make_server_config(self.inner.config.tls_private_key().unwrap());
221        // let tls_config =
222        //     crate::tls_rpk::make_server_config(self.inner.config.tls_private_key().unwrap());
223
224        let server_handle = Arc::new(
225            sui_http::Builder::new()
226                // Recycle peer connections via a graceful max-age drain: a
227                // long-lived multiplexed HTTP/2 connection accumulates state that
228                // trips a connection-level GoAway(PROTOCOL_ERROR) under sustained
229                // load, failing every in-flight request to that peer at once.
230                .config(
231                    sui_http::Config::default()
232                        .max_concurrent_streams(crate::config::DEFAULT_GRPC_PER_PEER_INFLIGHT_LIMIT)
233                        .http2_max_pending_accept_reset_streams(Some(
234                            crate::config::DEFAULT_GRPC_PER_PEER_INFLIGHT_LIMIT as usize,
235                        ))
236                        .max_connection_age(std::time::Duration::from_secs(120))
237                        .max_connection_age_grace(std::time::Duration::from_secs(120))
238                        .http2_keepalive_interval(Some(std::time::Duration::from_secs(30))),
239                )
240                .tls_config(tls_config)
241                .serve(self.inner.config.listen_address(), router)
242                .unwrap(),
243        );
244        connections.set_server(server_handle.clone());
245        let local_addr = *server_handle.local_addr();
246
247        (local_addr, supervise(server_handle))
248    }

295fn per_peer_byte_budget(configured: Option<u64>, largest_decode_limit: usize) -> u64 {
296    let floor = u64::try_from(largest_decode_limit)
297        .unwrap_or(u64::MAX)
298        .saturating_add(request_body::PREFIX_LEN)
299        .saturating_mul(2);
300    match configured {
301        Some(bytes) if bytes < floor => {
302            tracing::warn!(
303                "grpc_per_peer_inflight_bytes {bytes} is below room for two maximum-size \
304                 requests; using {floor}"
305            );
306            floor
307        }
308        Some(bytes) => bytes,
309        None => floor,
310    }
311}

313fn per_peer_connection_limit(configured: Option<usize>) -> usize {
314    const FLOOR: usize = 3;
315    let limit = configured.unwrap_or(crate::config::DEFAULT_GRPC_PER_PEER_CONNECTION_LIMIT);
316    if limit < FLOOR {
317        tracing::warn!("grpc_per_peer_connection_limit {limit} is below {FLOOR}; using {FLOOR}");
318    }
319    limit.max(FLOOR)
320}

322fn unavailable<B>(request: &http::Request<B>, message: &'static str) -> axum::response::Response {
323    if is_grpc_content_type(request.headers()) {
324        tonic::Status::unavailable(message).into_http()
325    } else {
326        axum::response::IntoResponse::into_response((
327            http::StatusCode::SERVICE_UNAVAILABLE,
328            message,
329        ))
330    }
331}

333// A server that stops without a shutdown signal has crashed (one HTTP/1 handler
334// panic ends sui-http's accept loop), so fail rather than run on without it.
335fn supervise(server_handle: Arc<ServerHandle>) -> Service {
336    let guard = ServerHandleGuard(server_handle);
337    Service::new().spawn_aborting(async move {
338        guard.0.wait_for_shutdown().await;
339        anyhow::bail!("HTTP server stopped unexpectedly")
340    })
341}

347async fn ready(hashi: Arc<Hashi>) -> impl axum::response::IntoResponse {
348    let Some(onchain_state) = hashi.onchain_state_opt() else {
349        return (
350            http::StatusCode::SERVICE_UNAVAILABLE,
351            "on-chain state not yet initialized",
352        );
353    };
354    // If the chain has moved past what this binary supports, report not-ready so
355    // operators/orchestration surface it — the node has halted autonomous work
356    // (see leader gate) and needs a binary upgrade. Details are in the
357    // `hashi_package_version_unsupported` metric and the leader log.
358    if matches!(
359        onchain_state.autonomous_halt_reason(),
360        Some(crate::onchain::HaltReason::BinaryUnsupported { .. })
361    ) {
362        return (
363            http::StatusCode::SERVICE_UNAVAILABLE,
364            "binary does not support the enabled on-chain package version(s) — upgrade required",
365        );
366    }
367    let epoch = onchain_state.epoch();
368    // Treat genesis (no committee formed yet) as ready so OrderedReady can bring
369    // up the rest of the StatefulSet to register keys and run the initial DKG.
370    let awaiting_genesis = epoch == 0 && onchain_state.current_committee().is_none();
371    if awaiting_genesis {
372        (http::StatusCode::OK, "ready (awaiting genesis)")
373    } else if hashi.signing_manager_for(epoch).is_some() {
374        (http::StatusCode::OK, "ready")
375    } else {
376        (
377            http::StatusCode::SERVICE_UNAVAILABLE,
378            "SigningManager not yet initialized",
379        )
380    }
381}

398impl RouterExt for axum::Router {
399    fn add_grpc_service<S>(self, svc: S, routes: &route_limits::RouteLimits) -> Self
400    where
401        S: tower::Service<
402                axum::extract::Request,
403                Response: axum::response::IntoResponse,
404                Error = std::convert::Infallible,
405            > + tonic::server::NamedService
406            + Clone
407            + Send
408            + Sync
409            + 'static,
410        S::Future: Send + 'static,
411    {
412        assert!(
413            routes.has_service(S::NAME),
414            "{} is served without a decode limit",
415            S::NAME
416        );
417        self.route_service(&format!("/{}/{{*rest}}", S::NAME), svc)
418    }
419}

421const ANONYMOUS_PATHS: &[&str] = &["/health", "/ready"];
422
423async fn require_known_validator(
424    axum::extract::State(hashi): axum::extract::State<Arc<Hashi>>,
425    mut request: axum::extract::Request,
426    next: axum::middleware::Next,
427) -> axum::response::Response {
428    match lookup_validator_address(&hashi, &request) {
429        Ok(validator_address) => {
430            request.extensions_mut().insert(validator_address);
431        }
432        Err(_) if ANONYMOUS_PATHS.contains(&request.uri().path()) => {}
433        Err(reason) => {
434            hashi
435                .metrics
436                .unknown_caller_refused_total
437                .with_label_values(&[reason.as_str()])
438                .inc();
439            return refuse(&request);
440        }
441    }
442    next.run(request).await
443}

445enum RefusalReason {
446    NoClientCert,
447    StateUnavailable,
448    NotRegistered,
449}
450
451impl RefusalReason {
452    fn as_str(&self) -> &'static str {
453        match self {
454            Self::NoClientCert => "no_client_cert",
455            Self::StateUnavailable => "state_unavailable",
456            Self::NotRegistered => "not_registered",
457        }
458    }
459}

461fn refuse<B>(request: &http::Request<B>) -> axum::response::Response {
462    if is_grpc_content_type(request.headers()) {
463        tonic::Status::permission_denied("unknown validator").into_http()
464    } else {
465        axum::response::IntoResponse::into_response((
466            http::StatusCode::FORBIDDEN,
467            "unknown validator",
468        ))
469    }
470}

472pub(super) fn is_grpc_content_type(headers: &http::HeaderMap) -> bool {
473    headers
474        .get(&http::header::CONTENT_TYPE)
475        .is_some_and(|header| {
476            header
477                .as_bytes()
478                .starts_with(tonic::metadata::GRPC_CONTENT_TYPE.as_bytes())
479        })
480}
481
482fn lookup_validator_address<B>(
483    hashi: &Hashi,
484    request: &http::Request<B>,
485) -> Result<sui_sdk_types::Address, RefusalReason> {
486    let tls_public_key = request
487        .extensions()
488        .get::<sui_http::PeerCertificates>()
489        .and_then(|peer_certs| peer_certs.peer_certs().first())
490        .and_then(|cert| crate::tls::public_key_from_certificate(cert).ok())
491        .ok_or(RefusalReason::NoClientCert)?;
492    hashi
493        .onchain_state_opt()
494        .ok_or(RefusalReason::StateUnavailable)?
495        .state()
496        .hashi()
497        .committees
498        .lookup_address_by_tls_public_key(&tls_public_key)
499        .ok_or(RefusalReason::NotRegistered)
500}
