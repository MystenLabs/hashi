36/// Wrapper that triggers graceful HTTP server shutdown on drop.
37///
38/// The HTTP server is spawned on a detached tokio task by `sui_http::Builder::serve`.
39/// This guard ensures that `trigger_shutdown` is called when the owning `Service` is
40/// dropped (whether via explicit `Service::shutdown()` or implicit drop), so the server's
41/// accept loop breaks and in-flight connections are drained.
42struct ServerHandleGuard(Arc<ServerHandle>);
43
44impl Drop for ServerHandleGuard {
45    fn drop(&mut self) {
46        self.0.trigger_shutdown();
47    }
48}

73    pub async fn start(self) -> (std::net::SocketAddr, Service) {
74        let mut routes = route_limits::RouteLimits::default();
75        let router = {
76            use hashi_types::proto::bridge_service_server;
77            use hashi_types::proto::mpc_service_server;
78
79            let max_decoding_message_size = self.inner.config.grpc_max_decoding_message_size();
80            let bridge_service = bridge_service_server::BridgeServiceServer::new(self.clone())
81                .max_decoding_message_size(routes.service(
82                    bridge_service_server::SERVICE_NAME,
83                    max_decoding_message_size.min(route_limits::BRIDGE_REQUEST_LIMIT),
84                ));
85            let mpc_service = mpc_service_server::MpcServiceServer::new(self.clone())
86                .max_decoding_message_size(
87                    routes.service(mpc_service_server::SERVICE_NAME, max_decoding_message_size),
88                );
89            for method in route_limits::SMALL_MPC_METHODS {
90                routes.method(
91                    mpc_service_server::SERVICE_NAME,
92                    method,
93                    route_limits::SMALL_MPC_REQUEST_LIMIT,
94                );
95            }
96            for method in route_limits::MPC_WORK_METHODS {
97                routes.mpc_work(mpc_service_server::SERVICE_NAME, method);
98            }
99
100            let (health_reporter, health_service) = tonic_health::server::health_reporter();
101
102            let mut reflection_v1 = tonic_reflection::server::Builder::configure();
103            let mut reflection_v1alpha = tonic_reflection::server::Builder::configure();
104
105            for file_descriptor_set in [
106                sui_rpc::proto::google::protobuf::FILE_DESCRIPTOR_SET,
107                sui_rpc::proto::google::rpc::FILE_DESCRIPTOR_SET,
108                tonic_health::pb::FILE_DESCRIPTOR_SET,
109                hashi_types::proto::FILE_DESCRIPTOR_SET,
110            ] {
111                reflection_v1 =
112                    reflection_v1.register_encoded_file_descriptor_set(file_descriptor_set);
113                reflection_v1alpha =
114                    reflection_v1alpha.register_encoded_file_descriptor_set(file_descriptor_set);
115            }
116
117            let reflection_v1 = reflection_v1.build_v1().unwrap();
118            let reflection_v1alpha = reflection_v1alpha.build_v1alpha().unwrap();
119
120            fn service_name<S: tonic::server::NamedService>(_service: &S) -> &'static str {
121                S::NAME
122            }
123
124            let limit = route_limits::HEALTH_DECODE_LIMIT;
125            let name = service_name(&health_service);
126            let health_service =
127                health_service.max_decoding_message_size(routes.service(name, limit));
128            let limit = route_limits::REFLECTION_REQUEST_LIMIT;
129            let name = service_name(&reflection_v1);
130            let reflection_v1 =
131                reflection_v1.max_decoding_message_size(routes.service(name, limit));
132            let name = service_name(&reflection_v1alpha);
133            let reflection_v1alpha =
134                reflection_v1alpha.max_decoding_message_size(routes.service(name, limit));
135
136            for service_name in [
137                service_name(&bridge_service),
138                service_name(&mpc_service),
139                service_name(&reflection_v1),
140                service_name(&reflection_v1alpha),
141            ] {
142                health_reporter
143                    .set_service_status(service_name, tonic_health::ServingStatus::Serving)
144                    .await;
145            }
146
147            axum::Router::new()
148                .add_grpc_service(bridge_service, &routes)
149                .add_grpc_service(mpc_service, &routes)
150                .add_grpc_service(reflection_v1, &routes)
151                .add_grpc_service(reflection_v1alpha, &routes)
152                .add_grpc_service(health_service, &routes)
153        };
154
155        let hashi_for_ready = self.inner.clone();
156        let health_endpoint = axum::Router::new()
157            .route("/health", axum::routing::get(health))
158            .route(
159                "/ready",
160                axum::routing::get(move || ready(hashi_for_ready.clone())),
161            );
162
163        let budget_bytes = per_peer_byte_budget(
164            self.inner.config.grpc_per_peer_inflight_bytes(),
165            routes.largest(),
166        );
167        self.inner
168            .metrics
169            .peer_inflight_limit_bytes
170            .set(i64::try_from(budget_bytes).unwrap_or(i64::MAX));
171        let limiter = peer_limit::PeerInflightLimiter::new(
172            self.inner.config.grpc_per_peer_inflight_limit(),
173            budget_bytes,
174            routes,
175            self.inner.metrics.clone(),
176        );
177        let connections = connection_limit::ConnectionLimiter::new(
178            per_peer_connection_limit(self.inner.config.grpc_per_peer_connection_limit()),
179            self.inner.metrics.clone(),
180        );
181        let layers = ServiceBuilder::new()
182            .layer(axum::middleware::from_fn_with_state(
183                self.inner.clone(),
184                require_known_validator,
185            ))
186            .layer(axum::middleware::from_fn_with_state(
187                connections.clone(),
188                connection_limit::limit_connections_per_peer,
189            ))
190            .layer(axum::middleware::from_fn_with_state(
191                limiter,
192                peer_limit::limit_per_peer,
193            ))
194            .layer(sui_http::middleware::callback::CallbackLayer::new(
195                metrics_layer::RpcMetricsMakeCallbackHandler::server(self.inner.metrics.clone()),
196            ));
197
198        let router = router.merge(health_endpoint).layer(layers);
199
200        let tls_config =
201            crate::tls::make_server_config(self.inner.config.tls_private_key().unwrap());
202        // let tls_config =
203        //     crate::tls_rpk::make_server_config(self.inner.config.tls_private_key().unwrap());
204
205        let server_handle = Arc::new(
206            sui_http::Builder::new()
207                // Recycle peer connections via a graceful max-age drain: a
208                // long-lived multiplexed HTTP/2 connection accumulates state that
209                // trips a connection-level GoAway(PROTOCOL_ERROR) under sustained
210                // load, failing every in-flight request to that peer at once.
211                .config(
212                    sui_http::Config::default()
213                        .max_concurrent_streams(crate::config::DEFAULT_GRPC_PER_PEER_INFLIGHT_LIMIT)
214                        .http2_max_pending_accept_reset_streams(Some(
215                            crate::config::DEFAULT_GRPC_PER_PEER_INFLIGHT_LIMIT as usize,
216                        ))
217                        .max_connection_age(std::time::Duration::from_secs(120))
218                        .max_connection_age_grace(std::time::Duration::from_secs(120))
219                        .http2_keepalive_interval(Some(std::time::Duration::from_secs(30))),
220                )
221                .tls_config(tls_config)
222                .serve(self.inner.config.listen_address(), router)
223                .unwrap(),
224        );
225        connections.set_server(server_handle.clone());
226        let local_addr = *server_handle.local_addr();
227
228        (local_addr, supervise(server_handle))
229    }

272fn per_peer_byte_budget(configured: Option<u64>, largest_decode_limit: usize) -> u64 {
273    let floor = u64::try_from(largest_decode_limit)
274        .unwrap_or(u64::MAX)
275        .saturating_add(request_body::PREFIX_LEN)
276        .saturating_mul(2);
277    match configured {
278        Some(bytes) if bytes < floor => {
279            tracing::warn!(
280                "grpc_per_peer_inflight_bytes {bytes} is below room for two maximum-size \
281                 requests; using {floor}"
282            );
283            floor
284        }
285        Some(bytes) => bytes,
286        None => floor,
287    }
288}

290fn per_peer_connection_limit(configured: Option<usize>) -> usize {
291    const FLOOR: usize = 3;
292    let limit = configured.unwrap_or(crate::config::DEFAULT_GRPC_PER_PEER_CONNECTION_LIMIT);
293    if limit < FLOOR {
294        tracing::warn!("grpc_per_peer_connection_limit {limit} is below {FLOOR}; using {FLOOR}");
295    }
296    limit.max(FLOOR)
297}

299fn unavailable<B>(request: &http::Request<B>, message: &'static str) -> axum::response::Response {
300    if is_grpc_content_type(request.headers()) {
301        tonic::Status::unavailable(message).into_http()
302    } else {
303        axum::response::IntoResponse::into_response((
304            http::StatusCode::SERVICE_UNAVAILABLE,
305            message,
306        ))
307    }
308}

310// A server that stops without a shutdown signal has crashed (one HTTP/1 handler
311// panic ends sui-http's accept loop), so fail rather than run on without it.
312fn supervise(server_handle: Arc<ServerHandle>) -> Service {
313    let guard = ServerHandleGuard(server_handle);
314    Service::new().spawn_aborting(async move {
315        guard.0.wait_for_shutdown().await;
316        anyhow::bail!("HTTP server stopped unexpectedly")
317    })
318}

324async fn ready(hashi: Arc<Hashi>) -> impl axum::response::IntoResponse {
325    let Some(onchain_state) = hashi.onchain_state_opt() else {
326        return (
327            http::StatusCode::SERVICE_UNAVAILABLE,
328            "on-chain state not yet initialized",
329        );
330    };
331    // If the chain has moved past what this binary supports, report not-ready so
332    // operators/orchestration surface it — the node has halted autonomous work
333    // (see leader gate) and needs a binary upgrade. Details are in the
334    // `hashi_package_version_unsupported` metric and the leader log.
335    if matches!(
336        onchain_state.autonomous_halt_reason(),
337        Some(crate::onchain::HaltReason::BinaryUnsupported { .. })
338    ) {
339        return (
340            http::StatusCode::SERVICE_UNAVAILABLE,
341            "binary does not support the enabled on-chain package version(s) — upgrade required",
342        );
343    }
344    let epoch = onchain_state.epoch();
345    // Treat genesis (no committee formed yet) as ready so OrderedReady can bring
346    // up the rest of the StatefulSet to register keys and run the initial DKG.
347    let awaiting_genesis = epoch == 0 && onchain_state.current_committee().is_none();
348    if awaiting_genesis {
349        (http::StatusCode::OK, "ready (awaiting genesis)")
350    } else if hashi.signing_manager_for(epoch).is_some() {
351        (http::StatusCode::OK, "ready")
352    } else {
353        (
354            http::StatusCode::SERVICE_UNAVAILABLE,
355            "SigningManager not yet initialized",
356        )
357    }
358}

375impl RouterExt for axum::Router {
376    fn add_grpc_service<S>(self, svc: S, routes: &route_limits::RouteLimits) -> Self
377    where
378        S: tower::Service<
379                axum::extract::Request,
380                Response: axum::response::IntoResponse,
381                Error = std::convert::Infallible,
382            > + tonic::server::NamedService
383            + Clone
384            + Send
385            + Sync
386            + 'static,
387        S::Future: Send + 'static,
388    {
389        assert!(
390            routes.has_service(S::NAME),
391            "{} is served without a decode limit",
392            S::NAME
393        );
394        self.route_service(&format!("/{}/{{*rest}}", S::NAME), svc)
395    }
396}

398const ANONYMOUS_PATHS: &[&str] = &["/health", "/ready"];
399
400async fn require_known_validator(
401    axum::extract::State(hashi): axum::extract::State<Arc<Hashi>>,
402    mut request: axum::extract::Request,
403    next: axum::middleware::Next,
404) -> axum::response::Response {
405    match lookup_validator_address(&hashi, &request) {
406        Ok(validator_address) => {
407            request.extensions_mut().insert(validator_address);
408        }
409        Err(_) if ANONYMOUS_PATHS.contains(&request.uri().path()) => {}
410        Err(reason) => {
411            hashi
412                .metrics
413                .unknown_caller_refused_total
414                .with_label_values(&[reason.as_str()])
415                .inc();
416            return refuse(&request);
417        }
418    }
419    next.run(request).await
420}

422enum RefusalReason {
423    NoClientCert,
424    StateUnavailable,
425    NotRegistered,
426}
427
428impl RefusalReason {
429    fn as_str(&self) -> &'static str {
430        match self {
431            Self::NoClientCert => "no_client_cert",
432            Self::StateUnavailable => "state_unavailable",
433            Self::NotRegistered => "not_registered",
434        }
435    }
436}

438fn refuse<B>(request: &http::Request<B>) -> axum::response::Response {
439    if is_grpc_content_type(request.headers()) {
440        tonic::Status::permission_denied("unknown validator").into_http()
441    } else {
442        axum::response::IntoResponse::into_response((
443            http::StatusCode::FORBIDDEN,
444            "unknown validator",
445        ))
446    }
447}

449pub(super) fn is_grpc_content_type(headers: &http::HeaderMap) -> bool {
450    headers
451        .get(&http::header::CONTENT_TYPE)
452        .is_some_and(|header| {
453            header
454                .as_bytes()
455                .starts_with(tonic::metadata::GRPC_CONTENT_TYPE.as_bytes())
456        })
457}
458
459fn lookup_validator_address<B>(
460    hashi: &Hashi,
461    request: &http::Request<B>,
462) -> Result<sui_sdk_types::Address, RefusalReason> {
463    let tls_public_key = request
464        .extensions()
465        .get::<sui_http::PeerCertificates>()
466        .and_then(|peer_certs| peer_certs.peer_certs().first())
467        .and_then(|cert| crate::tls::public_key_from_certificate(cert).ok())
468        .ok_or(RefusalReason::NoClientCert)?;
469    hashi
470        .onchain_state_opt()
471        .ok_or(RefusalReason::StateUnavailable)?
472        .state()
473        .hashi()
474        .committees
475        .lookup_address_by_tls_public_key(&tls_public_key)
476        .ok_or(RefusalReason::NotRegistered)
477}
