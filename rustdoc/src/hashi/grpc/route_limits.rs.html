31#[derive(Debug, Default)]
32pub(crate) struct RouteLimits {
33    services: HashMap<&'static str, usize>,
34    methods: HashMap<String, usize>,
35    mpc_work: HashSet<String>,
36}
37
38impl RouteLimits {
39    pub(crate) fn service(&mut self, name: &'static str, limit: usize) -> usize {
40        self.services.insert(name, limit);
41        limit
42    }
43
44    pub(crate) fn method(&mut self, service: &'static str, method: &str, limit: usize) {
45        self.methods.insert(format!("/{service}/{method}"), limit);
46    }
47
48    pub(crate) fn mpc_work(&mut self, service: &'static str, method: &str) {
49        self.mpc_work.insert(format!("/{service}/{method}"));
50    }
51
52    pub(crate) fn mpc_work_paths(&self) -> impl Iterator<Item = &str> {
53        self.mpc_work.iter().map(String::as_str)
54    }
55
56    pub(crate) fn has_service(&self, name: &str) -> bool {
57        self.services.contains_key(name)
58    }
59
60    pub(crate) fn limit(&self, path: &str) -> Option<usize> {
61        let service = self.services.get(service_of(path)?).copied()?;
62        Some(
63            self.methods
64                .get(path)
65                .map_or(service, |&method| method.min(service)),
66        )
67    }
68
69    pub(crate) fn largest(&self) -> usize {
70        self.services.values().copied().max().unwrap_or(0)
71    }
72}
73
74fn service_of(path: &str) -> Option<&str> {
75    let (service, _) = path.strip_prefix('/')?.split_once('/')?;
76    Some(service)
77}

79pub(crate) struct GrpcMethod {
80    pub(crate) path: Box<str>,
81    pub(crate) client_streaming: bool,
82}
83
84pub(crate) fn grpc_methods() -> &'static [GrpcMethod] {
85    static METHODS: OnceLock<Vec<GrpcMethod>> = OnceLock::new();
86    METHODS.get_or_init(|| {
87        use prost::Message as _;
88
89        let mut methods = Vec::new();
90        for encoded in [
91            hashi_types::proto::FILE_DESCRIPTOR_SET,
92            tonic_health::pb::FILE_DESCRIPTOR_SET,
93            tonic_reflection::pb::v1::FILE_DESCRIPTOR_SET,
94            tonic_reflection::pb::v1alpha::FILE_DESCRIPTOR_SET,
95        ] {
96            let Ok(set) = prost_types::FileDescriptorSet::decode(encoded) else {
97                continue;
98            };
99            for file in set.file {
100                let package = file.package();
101                for service in &file.service {
102                    for method in &service.method {
103                        methods.push(GrpcMethod {
104                            path: format!("/{package}.{}/{}", service.name(), method.name())
105                                .into_boxed_str(),
106                            client_streaming: method.client_streaming(),
107                        });
108                    }
109                }
110            }
111        }
112        methods
113    })
114}
115
116pub(crate) fn streams_requests(path: &str) -> bool {
117    static PATHS: OnceLock<HashSet<&'static str>> = OnceLock::new();
118    PATHS
119        .get_or_init(|| {
120            grpc_methods()
121                .iter()
122                .filter(|method| method.client_streaming)
123                .map(|method| &*method.path)
124                .collect()
125        })
126        .contains(path)
127}

129#[cfg(test)]
130mod tests {
131    use super::*;
132
133    #[test]
134    fn only_reflection_streams_requests_and_every_listed_mpc_method_exists() {
135        let streaming: Vec<_> = grpc_methods()
136            .iter()
137            .filter(|method| method.client_streaming)
138            .map(|method| &*method.path)
139            .collect();
140        assert_eq!(
141            streaming,
142            [
143                "/grpc.reflection.v1.ServerReflection/ServerReflectionInfo",
144                "/grpc.reflection.v1alpha.ServerReflection/ServerReflectionInfo",
145            ]
146        );
147
148        let mpc = hashi_types::proto::mpc_service_server::SERVICE_NAME;
149        for method in SMALL_MPC_METHODS.into_iter().chain(MPC_WORK_METHODS) {
150            let path = format!("/{mpc}/{method}");
151            assert!(
152                grpc_methods().iter().any(|known| *known.path == path),
153                "{path}"
154            );
155        }
156    }
157
158    #[test]
159    fn a_method_cap_never_exceeds_its_service_limit() {
160        let mut limits = RouteLimits::default();
161        limits.service("pkg.Big", 1024);
162        limits.service("pkg.Small", 16);
163        limits.method("pkg.Big", "Capped", 64);
164        limits.method("pkg.Small", "Capped", 64);
165
166        assert_eq!(limits.limit("/pkg.Big/Capped"), Some(64));
167        assert_eq!(limits.limit("/pkg.Big/Other"), Some(1024));
168        assert_eq!(limits.limit("/pkg.Small/Capped"), Some(16));
169        assert_eq!(limits.limit("/pkg.Unknown/Capped"), None);
170        assert_eq!(limits.limit("/health"), None);
171        assert_eq!(limits.largest(), 1024);
172    }
