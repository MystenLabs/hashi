21#[derive(Clone)]
22pub(crate) struct PeerInflightLimiter(Arc<Inner>);
23
24struct Inner {
25    limit: u32,
26    peers: RwLock<HashMap<Address, Arc<Peer>>>,
27    metrics: Arc<Metrics>,
28}
29
30struct Peer {
31    inflight: AtomicU32,
32    high_water: AtomicU32,
33    high_water_publish: Mutex<()>,
34    at_admission: Histogram,
35    max: IntGauge,
36    shed: IntCounter,
37}
38
39impl PeerInflightLimiter {
40    pub(crate) fn new(limit: u32, metrics: Arc<Metrics>) -> Self {
41        Self(Arc::new(Inner {
42            limit,
43            peers: RwLock::new(HashMap::new()),
44            metrics,
45        }))
46    }
47
48    fn peer(&self, address: Address) -> Arc<Peer> {
49        if let Some(peer) = self.0.peers.read().unwrap().get(&address) {
50            return peer.clone();
51        }
52        self.0
53            .peers
54            .write()
55            .unwrap()
56            .entry(address)
57            .or_insert_with(|| {
58                let label = address.to_string();
59                let metrics = &self.0.metrics;
60                Arc::new(Peer {
61                    inflight: AtomicU32::new(0),
62                    high_water: AtomicU32::new(0),
63                    high_water_publish: Mutex::new(()),
64                    at_admission: metrics
65                        .peer_inflight_at_admission
66                        .with_label_values(&[&label]),
67                    max: metrics.peer_inflight_max.with_label_values(&[&label]),
68                    shed: metrics
69                        .peer_requests_shed_total
70                        .with_label_values(&[&label]),
71                })
72            })
73            .clone()
74    }
75
76    fn try_admit(&self, address: Address) -> Option<Slot> {
77        let peer = self.peer(address);
78        let limit = self.0.limit;
79        let Ok(before) = peer
80            .inflight
81            .fetch_update(Ordering::AcqRel, Ordering::Acquire, |n| {
82                (n < limit).then_some(n + 1)
83            })
84        else {
85            peer.shed.inc();
86            return None;
87        };
88        peer.at_admission.observe(f64::from(before));
89        let now = before + 1;
90        if peer.high_water.fetch_max(now, Ordering::AcqRel) < now {
91            let _publish = peer.high_water_publish.lock().unwrap();
92            peer.max
93                .set(i64::from(peer.high_water.load(Ordering::Acquire)));
94        }
95        Some(Slot(peer))
96    }
97
98    #[cfg(test)]
99    fn inflight(&self, address: Address) -> u32 {
100        self.peer(address).inflight.load(Ordering::Acquire)
101    }
102}
103
104struct Slot(Arc<Peer>);
105
106impl Drop for Slot {
107    fn drop(&mut self) {
108        self.0.inflight.fetch_sub(1, Ordering::AcqRel);
109    }
110}

112pub(crate) async fn limit_per_peer(
113    axum::extract::State(limiter): axum::extract::State<PeerInflightLimiter>,
114    request: axum::extract::Request,
115    next: axum::middleware::Next,
116) -> axum::response::Response {
117    let Some(address) = request.extensions().get::<Address>().copied() else {
118        return next.run(request).await;
119    };
120    let Some(slot) = limiter.try_admit(address) else {
121        return shed(&request);
122    };
123    let response = next.run(request).await;
124    response.map(|body| axum::body::Body::new(Guarded { body, _slot: slot }))
125}
126
127fn shed<B>(request: &http::Request<B>) -> axum::response::Response {
128    if super::is_grpc_content_type(request.headers()) {
129        tonic::Status::unavailable(super::PEER_INFLIGHT_LIMIT_MSG).into_http()
130    } else {
131        axum::response::IntoResponse::into_response((
132            http::StatusCode::SERVICE_UNAVAILABLE,
133            super::PEER_INFLIGHT_LIMIT_MSG,
134        ))
135    }
136}

138struct Guarded<B> {
139    body: B,
140    _slot: Slot,
141}
142
143impl<B: http_body::Body + Unpin> http_body::Body for Guarded<B> {
144    type Data = B::Data;
145    type Error = B::Error;
146
147    fn poll_frame(
148        mut self: Pin<&mut Self>,
149        cx: &mut Context<'_>,
150    ) -> Poll<Option<Result<http_body::Frame<Self::Data>, Self::Error>>> {
151        Pin::new(&mut self.body).poll_frame(cx)
152    }
153
154    fn is_end_stream(&self) -> bool {
155        self.body.is_end_stream()
156    }
157
158    fn size_hint(&self) -> http_body::SizeHint {
159        self.body.size_hint()
160    }
161}

171    fn peer(id: u8) -> Address {
172        Address::new([id; 32])
173    }
174
175    async fn gate(
176        mut request: axum::extract::Request,
177        next: axum::middleware::Next,
178    ) -> axum::response::Response {
179        let id = request.headers().get("x-peer").map(|v| v.as_bytes()[0]);
180        if let Some(id) = id {
181            request.extensions_mut().insert(peer(id));
182        }
183        next.run(request).await
184    }
185
186    async fn handler(request: axum::extract::Request) -> &'static str {
187        if request.headers().contains_key("x-hold") {
188            std::future::pending::<()>().await;
189        }
190        "ok"
191    }
192
193    fn app(limiter: PeerInflightLimiter) -> Router {
194        Router::new().route("/", get(handler)).layer(
195            tower::ServiceBuilder::new()
196                .layer(axum::middleware::from_fn(gate))
197                .layer(axum::middleware::from_fn_with_state(
198                    limiter,
199                    limit_per_peer,
200                )),
201        )
202    }
203
204    fn request(id: u8, hold: bool, grpc: bool) -> axum::extract::Request {
205        let mut builder = http::Request::builder()
206            .uri("/")
207            .header("x-peer", http::HeaderValue::from_bytes(&[id]).unwrap());
208        if hold {
209            builder = builder.header("x-hold", "1");
210        }
211        if grpc {
212            builder = builder.header(http::header::CONTENT_TYPE, "application/grpc");
213        }
214        builder.body(axum::body::Body::empty()).unwrap()
215    }
216
217    async fn call(app: &Router, request: axum::extract::Request) -> axum::response::Response {
218        app.clone().call(request).await.unwrap()
219    }

221    #[tokio::test]
222    async fn a_full_peer_is_shed_others_are_served_and_a_cancelled_request_frees_its_slot() {
223        let limit = 3;
224        let registry = prometheus::Registry::new();
225        let limiter = PeerInflightLimiter::new(limit, Arc::new(Metrics::new(&registry)));
226        let app = app(limiter.clone());
227
228        let mut held: Vec<_> = (0..limit)
229            .map(|_| {
230                let app = app.clone();
231                tokio::spawn(async move { call(&app, request(b'a', true, false)).await })
232            })
233            .collect();
234        while limiter.inflight(peer(b'a')) < limit {
235            tokio::task::yield_now().await;
236        }
237
238        let shed = call(&app, request(b'a', false, false)).await;
239        assert_eq!(shed.status(), http::StatusCode::SERVICE_UNAVAILABLE);
240        let shed = call(&app, request(b'a', false, true)).await;
241        assert_eq!(shed.status(), http::StatusCode::OK);
242        assert_eq!(shed.headers().get("grpc-status").unwrap(), "14");
243        assert_eq!(limiter.peer(peer(b'a')).shed.get(), 2);
244
245        let served = call(&app, request(b'b', false, false)).await;
246        assert_eq!(served.status(), http::StatusCode::OK);
247        let anonymous = http::Request::builder()
248            .uri("/")
249            .body(axum::body::Body::empty())
250            .unwrap();
251        assert_eq!(call(&app, anonymous).await.status(), http::StatusCode::OK);
252        assert_eq!(limiter.inflight(peer(b'a')), limit);
253
254        let cancelled = held.pop().unwrap();
255        cancelled.abort();
256        let _ = cancelled.await;
257        assert_eq!(limiter.inflight(peer(b'a')), limit - 1);
258
259        let admitted = call(&app, request(b'a', false, false)).await;
260        assert_eq!(admitted.status(), http::StatusCode::OK);
261        assert_eq!(limiter.inflight(peer(b'a')), limit);
262        drop(admitted);
263        assert_eq!(limiter.inflight(peer(b'a')), limit - 1);
264        assert_eq!(limiter.peer(peer(b'a')).max.get(), i64::from(limit));
265
266        for task in held {
267            task.abort();
268        }
269    }
