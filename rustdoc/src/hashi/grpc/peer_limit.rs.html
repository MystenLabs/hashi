39pub(crate) fn guard_blocking_mpc_work() -> Option<Arc<MpcWorkGuard>> {
40    MPC_WORK
41        .try_with(|work| match work {
42            MpcWork::Guarded(guard) => Some(guard.clone()),
43            MpcWork::Unguarded(unguarded) => {
44                unguarded.inc();
45                None
46            }
47        })
48        .ok()
49        .flatten()
50}

79struct ByteBudget {
80    reserved: AtomicU64,
81    high_water: AtomicU64,
82    publish: Mutex<()>,
83    max: IntGauge,
84    over: IntCounter,
85}
86
87impl ByteBudget {
88    fn new(metrics: &Metrics, peer: &str, budget: &str) -> Self {
89        Self {
90            reserved: AtomicU64::new(0),
91            high_water: AtomicU64::new(0),
92            publish: Mutex::new(()),
93            max: metrics
94                .peer_inflight_max_bytes
95                .with_label_values(&[peer, budget]),
96            over: metrics
97                .peer_requests_over_byte_budget_total
98                .with_label_values(&[peer, budget]),
99        }
100    }
101
102    fn try_reserve(&self, bytes: u64, limit: u64) -> bool {
103        let Ok(before) =
104            self.reserved
105                .fetch_update(Ordering::AcqRel, Ordering::Acquire, |reserved| {
106                    reserved.checked_add(bytes).filter(|&now| now <= limit)
107                })
108        else {
109            return false;
110        };
111        let now = before + bytes;
112        if self.high_water.fetch_max(now, Ordering::AcqRel) < now {
113            let _publish = self.publish.lock().unwrap();
114            self.max
115                .set(i64::try_from(self.high_water.load(Ordering::Acquire)).unwrap_or(i64::MAX));
116        }
117        true
118    }
119
120    fn release(&self, bytes: u64) {
121        self.reserved.fetch_sub(bytes, Ordering::AcqRel);
122    }
123}

125pub(super) struct RequestCharge {
126    peer: Arc<Peer>,
127    budget: u64,
128    reserved: AtomicU64,
129    work: Option<Arc<MpcWorkGuard>>,
130}
131
132impl RequestCharge {
133    pub(super) fn reserve(&self, bytes: u64) -> bool {
134        if !self.peer.request_bytes.try_reserve(bytes, self.budget) {
135            self.peer.request_bytes.over.inc();
136            return false;
137        }
138        self.reserved.fetch_add(bytes, Ordering::AcqRel);
139        self.work.as_ref().is_none_or(|work| work.reserve(bytes))
140    }
141
142    pub(super) fn release(&self, bytes: u64) {
143        self.reserved.fetch_sub(bytes, Ordering::AcqRel);
144        self.peer.request_bytes.release(bytes);
145    }
146
147    pub(super) fn refuse_too_large(&self) {
148        self.peer.too_large.inc();
149    }
150}
151
152impl Drop for RequestCharge {
153    fn drop(&mut self) {
154        self.peer.request_bytes.release(*self.reserved.get_mut());
155    }
156}

164impl MpcWorkGuard {
165    fn reserve(&self, bytes: u64) -> bool {
166        if !self.peer.work_bytes.try_reserve(bytes, self.budget) {
167            self.peer.work_bytes.over.inc();
168            return false;
169        }
170        self.reserved.fetch_add(bytes, Ordering::AcqRel);
171        true
172    }
173}

175impl Drop for MpcWorkGuard {
176    fn drop(&mut self) {
177        self.peer.work_bytes.release(*self.reserved.get_mut());
178        self.peer.work_inflight.fetch_sub(1, Ordering::AcqRel);
179    }

182impl PeerInflightLimiter {
183    pub(crate) fn new(
184        limit: u32,
185        budget_bytes: u64,
186        routes: RouteLimits,
187        metrics: Arc<Metrics>,
188    ) -> Self {
189        let mpc_work = routes
190            .mpc_work_paths()
191            .map(|path| {
192                let at_admission = metrics
193                    .peer_mpc_work_at_admission
194                    .with_label_values(&[path]);
195                (path.to_owned(), at_admission)
196            })
197            .collect();
198        Self(Arc::new(Inner {
199            limit,
200            budget_bytes,
201            routes,
202            mpc_work,
203            peers: RwLock::new(HashMap::new()),
204            metrics,
205        }))
206    }
207
208    fn peer(&self, address: Address) -> Arc<Peer> {
209        if let Some(peer) = self.0.peers.read().unwrap().get(&address) {
210            return peer.clone();
211        }
212        self.0
213            .peers
214            .write()
215            .unwrap()
216            .entry(address)
217            .or_insert_with(|| {
218                let label = address.to_string();
219                let metrics = &self.0.metrics;
220                metrics.peer_first_seen(&label);
221                Arc::new(Peer {
222                    inflight: AtomicU32::new(0),
223                    high_water: AtomicU32::new(0),
224                    high_water_publish: Mutex::new(()),
225                    at_admission: metrics
226                        .peer_inflight_at_admission
227                        .with_label_values(&[&label]),
228                    max: metrics.peer_inflight_max.with_label_values(&[&label]),
229                    shed: metrics
230                        .peer_requests_shed_total
231                        .with_label_values(&[&label]),
232                    request_bytes: ByteBudget::new(metrics, &label, REQUEST_BUDGET),
233                    work_inflight: AtomicU32::new(0),
234                    work_shed: metrics
235                        .peer_mpc_work_shed_total
236                        .with_label_values(&[&label]),
237                    work_bytes: ByteBudget::new(metrics, &label, MPC_WORK_BUDGET),
238                    too_large: metrics
239                        .peer_requests_too_large_total
240                        .with_label_values(&[&label]),
241                    responses_shed: metrics
242                        .peer_responses_shed_total
243                        .with_label_values(&[&label]),
244                })
245            })
246            .clone()
247    }
248
249    fn try_admit(&self, address: Address) -> Option<Slot> {
250        let peer = self.peer(address);
251        let limit = self.0.limit;
252        let Ok(before) = peer
253            .inflight
254            .fetch_update(Ordering::AcqRel, Ordering::Acquire, |n| {
255                (n < limit).then_some(n + 1)
256            })
257        else {
258            peer.shed.inc();
259            return None;
260        };
261        peer.at_admission.observe(f64::from(before));
262        let now = before + 1;
263        if peer.high_water.fetch_max(now, Ordering::AcqRel) < now {
264            let _publish = peer.high_water_publish.lock().unwrap();
265            peer.max
266                .set(i64::from(peer.high_water.load(Ordering::Acquire)));
267        }
268        Some(Slot(peer))
269    }
270
271    fn try_admit_work(&self, peer: &Arc<Peer>, at_admission: &Histogram) -> Option<MpcWorkGuard> {
272        let limit = self.0.limit;
273        let Ok(before) =
274            peer.work_inflight
275                .fetch_update(Ordering::AcqRel, Ordering::Acquire, |n| {
276                    (n < limit).then_some(n + 1)
277                })
278        else {
279            peer.work_shed.inc();
280            return None;
281        };
282        at_admission.observe(f64::from(before));
283        Some(MpcWorkGuard {
284            peer: peer.clone(),
285            budget: self.0.budget_bytes,
286            reserved: AtomicU64::new(0),
287        })
288    }
289

293    }
294
295    #[cfg(test)]
296    fn reserved(&self, address: Address) -> u64 {

302
303    #[cfg(test)]

306    }

311            .work_bytes
312            .reserved
313            .load(Ordering::Acquire)
314    }

325pub(crate) async fn limit_per_peer(
326    axum::extract::State(limiter): axum::extract::State<PeerInflightLimiter>,
327    request: axum::extract::Request,
328    next: axum::middleware::Next,
329) -> axum::response::Response {
330    let unguarded =
331        MpcWork::Unguarded(limiter.0.metrics.mpc_unguarded_spawn_blocking_total.clone());
332    let Some(address) = request.extensions().get::<Address>().copied() else {
333        return MPC_WORK.scope(unguarded, next.run(request)).await;
334    };
335    let Some(slot) = limiter.try_admit(address) else {
336        return shed(&request);
337    };
338    let path = request.uri().path();
339    let work = match limiter.0.mpc_work.get(path) {
340        Some(at_admission) => match limiter.try_admit_work(&slot.0, at_admission) {
341            Some(guard) => Some(Arc::new(guard)),
342            None => return shed(&request),
343        },
344        None => None,
345    };
346    let charge = Arc::new(RequestCharge {
347        peer: slot.0.clone(),
348        budget: limiter.0.budget_bytes,
349        reserved: AtomicU64::new(0),
350        work: work.clone(),
351    });
352    let request = match limiter.0.routes.limit(path) {
353        Some(limit) => {
354            let single_message = !super::route_limits::streams_requests(path);
355            let charge = charge.clone();
356            request.map(|body| {
357                axum::body::Body::new(LimitedBody::new(body, charge, limit, single_message))
358            })
359        }
360        None => request,
361    };
362    let work = work.map_or(unguarded, MpcWork::Guarded);
363    // Held across the handler, since tonic drops a unary body before calling it.
364    let response = MPC_WORK.scope(work, next.run(request)).await;
365    drop(charge);
366    let budget = super::is_grpc_content_type(response.headers()).then_some(limiter.0.budget_bytes);
367    response.map(|body| axum::body::Body::new(Guarded::new(body, slot, budget)))
368}

374struct Guarded {
375    body: axum::body::Body,
376    slot: Slot,
377    budget: Option<u64>,
378    pending: Option<(Bytes, usize)>,
379    done: bool,
380}
381
382impl Guarded {
383    fn new(body: axum::body::Body, slot: Slot, budget: Option<u64>) -> Self {
384        Self {
385            body,
386            slot,
387            budget,
388            pending: None,
389            done: false,
390        }
391    }
392
393    fn release(&mut self) {
394        if let Some((frame, _)) = self.pending.take() {
395            self.slot
396                .0
397                .request_bytes
398                .release(u64::try_from(frame.len()).unwrap_or(u64::MAX));
399        }
400    }
401}
402
403impl http_body::Body for Guarded {
404    type Data = Bytes;
405    type Error = axum::Error;
406
407    fn poll_frame(
408        self: Pin<&mut Self>,
409        cx: &mut Context<'_>,
410    ) -> Poll<Option<Result<http_body::Frame<Bytes>, axum::Error>>> {
411        let this = self.get_mut();
412        loop {
413            if let Some((frame, sent)) = &mut this.pending
414                && *sent < frame.len()
415            {
416                let end = frame.len().min(*sent + RESPONSE_PIECE);
417                let piece = Bytes::copy_from_slice(&frame[*sent..end]);
418                *sent = end;
419                return Poll::Ready(Some(Ok(http_body::Frame::data(piece))));
420            }
421            this.release();
422            if this.done {
423                return Poll::Ready(None);
424            }
425            let frame = match ready!(Pin::new(&mut this.body).poll_frame(cx)) {
426                Some(Ok(frame)) => frame,
427                other => {
428                    this.done = true;
429                    return Poll::Ready(other);
430                }
431            };
432            let Some(budget) = this.budget else {
433                return Poll::Ready(Some(Ok(frame)));
434            };
435            let data = match frame.into_data() {
436                Ok(data) => data,
437                Err(frame) => return Poll::Ready(Some(Ok(frame))),
438            };
439            let bytes = u64::try_from(data.len()).unwrap_or(u64::MAX);
440            if !this.slot.0.request_bytes.try_reserve(bytes, budget) {
441                this.slot.0.responses_shed.inc();
442                this.done = true;
443                let mut trailers = http::HeaderMap::new();
444                let _ = tonic::Status::unavailable(super::PEER_INFLIGHT_LIMIT_MSG)
445                    .add_header(&mut trailers);
446                return Poll::Ready(Some(Ok(http_body::Frame::trailers(trailers))));
447            }
448            this.pending = Some((data, 0));
449        }
450    }
451
452    fn is_end_stream(&self) -> bool {
453        let pieces_left = self
454            .pending
455            .as_ref()
456            .is_some_and(|(frame, sent)| *sent < frame.len());
457        !pieces_left && (self.done || self.body.is_end_stream())
458    }
459
460    fn size_hint(&self) -> http_body::SizeHint {
461        match self.budget {
462            Some(_) => http_body::SizeHint::default(),
463            None => self.body.size_hint(),
464        }
465    }
466}

468impl Drop for Guarded {
469    fn drop(&mut self) {
470        self.release();
471    }

476    tasks: Arc<Mutex<HashMap<Address, CallerTasks>>>,
477    metrics: Arc<Metrics>,

480#[derive(Default)]
481struct CallerTasks {
482    active: usize,
483    high_water: usize,
484}
485
486pub(crate) struct CallerTaskSlot {
487    tasks: Arc<Mutex<HashMap<Address, CallerTasks>>>,
488    caller: Address,
489}
490
491impl CallerTaskLimiter {
492    pub(crate) fn new(metrics: Arc<Metrics>) -> Self {
493        Self {
494            tasks: Arc::new(Mutex::new(HashMap::new())),
495            metrics,
496        }
497    }
498
499    pub(crate) fn try_admit(&self, caller: Address, limit: usize) -> Option<CallerTaskSlot> {
500        let mut tasks = self.tasks.lock().unwrap();
501        let entry = tasks.entry(caller).or_default();
502        if entry.active >= limit {
503            return None;
504        }
505        entry.active += 1;
506        if entry.active > entry.high_water {
507            entry.high_water = entry.active;
508            self.metrics
509                .withdrawal_signing_tasks_max
510                .with_label_values(&[&caller.to_string()])
511                .set(entry.high_water as i64);
512        }
513        Some(CallerTaskSlot {
514            tasks: self.tasks.clone(),
515            caller,
516        })
517    }
518}
519
520impl Drop for CallerTaskSlot {
521    fn drop(&mut self) {
522        if let Some(entry) = self.tasks.lock().unwrap().get_mut(&self.caller) {
523            entry.active -= 1;
524        }
525    }
526}

698        open: Arc<(Mutex<bool>, Condvar)>,
699    }
700
701    struct Release(Blocked);
702
703    impl Drop for Release {
704        fn drop(&mut self) {
705            let (open, opened) = &*self.0.open;
706            *open.lock().unwrap() = true;
707            opened.notify_all();
708        }
709    }

711    async fn blocking_work(
712        axum::extract::State(blocked): axum::extract::State<Blocked>,
713        request: axum::extract::Request,
714    ) -> &'static str {
715        let _ = axum::body::to_bytes(request.into_body(), usize::MAX).await;
716        crate::mpc::spawn_blocking(move || {
717            blocked.started.fetch_add(1, Ordering::AcqRel);
718            let (open, opened) = &*blocked.open;
719            drop(opened.wait_while(open.lock().unwrap(), |open| !*open));
720        })
721        .await;
722        "ok"
723    }

734        path: &str,
735        id: u8,
736        grpc: bool,
737    ) -> (h2::client::ResponseFuture, h2::SendStream<bytes::Bytes>) {
738        let mut request = http::Request::post(format!("http://localhost{path}"))

743        }

746            .send_request(request.body(()).unwrap(), false)
747            .unwrap()
748    }

756    async fn grpc_status(response: h2::client::ResponseFuture) -> tonic::Status {
757        tokio::time::timeout(std::time::Duration::from_secs(10), async {
758            let response = response.await.unwrap();
759            if let Some(status) = tonic::Status::from_header_map(response.headers()) {
760                return status;
761            }
762            let mut body = response.into_body();
763            while let Some(chunk) = body.data().await {
764                let chunk = chunk.unwrap();
765                body.flow_control().release_capacity(chunk.len()).unwrap();
766            }
767            tonic::Status::from_header_map(&body.trailers().await.unwrap().unwrap()).unwrap()
768        })
769        .await
770        .unwrap()
771    }

773    async fn until(mut condition: impl FnMut() -> bool) {
774        tokio::time::timeout(std::time::Duration::from_secs(10), async {
775            while !condition() {
776                tokio::task::yield_now().await;
777            }
778        })
779        .await
780        .unwrap();
781    }

908    struct Frames(std::collections::VecDeque<http_body::Frame<Bytes>>);
909
910    impl http_body::Body for Frames {
911        type Data = Bytes;
912        type Error = std::convert::Infallible;
913
914        fn poll_frame(
915            mut self: Pin<&mut Self>,
916            _: &mut Context<'_>,
917        ) -> Poll<Option<Result<http_body::Frame<Bytes>, Self::Error>>> {
918            Poll::Ready(self.0.pop_front().map(Ok))
919        }
920    }
921
922    fn guarded_response(limiter: &PeerInflightLimiter, frame: Bytes) -> Guarded {
923        let body = Frames(
924            [
925                http_body::Frame::data(frame),
926                http_body::Frame::trailers(http::HeaderMap::new()),
927            ]
928            .into(),
929        );
930        Guarded::new(
931            axum::body::Body::new(body),
932            limiter.try_admit(peer(b'a')).unwrap(),
933            Some(limiter.0.budget_bytes),
934        )
935    }
936
937    async fn next_frame(body: &mut Guarded) -> Option<http_body::Frame<Bytes>> {
938        std::future::poll_fn(|cx| Pin::new(&mut *body).poll_frame(cx))
939            .await
940            .map(Result::unwrap)
941    }
