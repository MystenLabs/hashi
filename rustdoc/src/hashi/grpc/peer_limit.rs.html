43pub(crate) fn guard_blocking_mpc_work() -> Option<Arc<MpcWorkGuard>> {
44    MPC_WORK
45        .try_with(|work| match work {
46            MpcWork::Guarded(guard) => Some(guard.clone()),
47            MpcWork::Unguarded(unguarded) => {
48                unguarded.inc();
49                None
50            }
51        })
52        .ok()
53        .flatten()
54}

83struct ByteBudget {
84    reserved: AtomicU64,
85    high_water: AtomicU64,
86    publish: Mutex<()>,
87    max: IntGauge,
88    over: IntCounter,
89}
90
91impl ByteBudget {
92    fn new(metrics: &Metrics, peer: &str, budget: &str) -> Self {
93        Self {
94            reserved: AtomicU64::new(0),
95            high_water: AtomicU64::new(0),
96            publish: Mutex::new(()),
97            max: metrics
98                .peer_inflight_max_bytes
99                .with_label_values(&[peer, budget]),
100            over: metrics
101                .peer_requests_over_byte_budget_total
102                .with_label_values(&[peer, budget]),
103        }
104    }
105
106    fn try_reserve(&self, bytes: u64, limit: u64) -> bool {
107        let Ok(before) =
108            self.reserved
109                .fetch_update(Ordering::AcqRel, Ordering::Acquire, |reserved| {
110                    reserved.checked_add(bytes).filter(|&now| now <= limit)
111                })
112        else {
113            return false;
114        };
115        let now = before + bytes;
116        if self.high_water.fetch_max(now, Ordering::AcqRel) < now {
117            let _publish = self.publish.lock().unwrap();
118            self.max
119                .set(i64::try_from(self.high_water.load(Ordering::Acquire)).unwrap_or(i64::MAX));
120        }
121        true
122    }
123
124    fn release(&self, bytes: u64) {
125        self.reserved.fetch_sub(bytes, Ordering::AcqRel);
126    }
127}

129pub(super) struct RequestCharge {
130    peer: Arc<Peer>,
131    budget: u64,
132    reserved: AtomicU64,
133    work: Option<Arc<MpcWorkGuard>>,
134}

136impl RequestCharge {
137    pub(super) fn reserve(&self, bytes: u64) -> bool {
138        if !self.peer.request_bytes.try_reserve(bytes, self.budget) {
139            self.peer.request_bytes.over.inc();
140            return false;
141        }
142        self.reserved.fetch_add(bytes, Ordering::AcqRel);
143        self.work.as_ref().is_none_or(|work| work.reserve(bytes))
144    }
145
146    pub(super) fn release(&self, bytes: u64) {
147        self.reserved.fetch_sub(bytes, Ordering::AcqRel);
148        self.peer.request_bytes.release(bytes);
149    }
150
151    pub(super) fn refuse_too_large(&self) {
152        self.peer.too_large.inc();
153    }
154}
155
156impl Drop for RequestCharge {
157    fn drop(&mut self) {
158        self.peer.request_bytes.release(*self.reserved.get_mut());
159    }
160}

168impl MpcWorkGuard {
169    fn reserve(&self, bytes: u64) -> bool {
170        if !self.peer.work_bytes.try_reserve(bytes, self.budget) {
171            self.peer.work_bytes.over.inc();
172            return false;
173        }
174        self.reserved.fetch_add(bytes, Ordering::AcqRel);
175        true
176    }

183    }

186impl PeerInflightLimiter {
187    pub(crate) fn new(
188        limit: u32,
189        budget_bytes: u64,
190        routes: RouteLimits,
191        metrics: Arc<Metrics>,
192    ) -> Self {
193        let mpc_work = routes
194            .mpc_work_paths()
195            .map(|path| {
196                let at_admission = metrics
197                    .peer_mpc_work_at_admission
198                    .with_label_values(&[path]);
199                (path.to_owned(), at_admission)
200            })
201            .collect();
202        Self(Arc::new(Inner {
203            limit,
204            budget_bytes,
205            routes,
206            mpc_work,
207            peers: RwLock::new(HashMap::new()),
208            metrics,
209        }))
210    }
211
212    fn peer(&self, address: Address) -> Arc<Peer> {
213        if let Some(peer) = self.0.peers.read().unwrap().get(&address) {
214            return peer.clone();
215        }
216        self.0
217            .peers
218            .write()
219            .unwrap()
220            .entry(address)
221            .or_insert_with(|| {
222                let label = address.to_string();
223                let metrics = &self.0.metrics;
224                metrics.peer_first_seen(&label);
225                for reason in [
226                    crate::metrics::SIGNING_REFUSAL_BUSY,
227                    crate::metrics::SIGNING_REFUSAL_SHAPE,
228                ] {
229                    metrics
230                        .withdrawal_signing_refused_total
231                        .with_label_values(&[&label, reason]);
232                }
233                metrics
234                    .peer_bridge_work_seconds_total
235                    .with_label_values(&[&label]);
236                metrics.peer_store_reads_max.with_label_values(&[&label]);
237                metrics
238                    .peer_store_reads_shed_total
239                    .with_label_values(&[&label]);
240                Arc::new(Peer {
241                    inflight: AtomicU32::new(0),
242                    high_water: AtomicU32::new(0),
243                    high_water_publish: Mutex::new(()),
244                    at_admission: metrics
245                        .peer_inflight_at_admission
246                        .with_label_values(&[&label]),
247                    max: metrics.peer_inflight_max.with_label_values(&[&label]),
248                    shed: metrics
249                        .peer_requests_shed_total
250                        .with_label_values(&[&label]),
251                    request_bytes: ByteBudget::new(metrics, &label, REQUEST_BUDGET),
252                    work_inflight: AtomicU32::new(0),
253                    work_shed: metrics
254                        .peer_mpc_work_shed_total
255                        .with_label_values(&[&label]),
256                    work_bytes: ByteBudget::new(metrics, &label, MPC_WORK_BUDGET),
257                    too_large: metrics
258                        .peer_requests_too_large_total
259                        .with_label_values(&[&label]),
260                    responses_shed: metrics
261                        .peer_responses_shed_total
262                        .with_label_values(&[&label]),
263                })
264            })
265            .clone()
266    }
267
268    fn try_admit(&self, address: Address) -> Option<Slot> {
269        let peer = self.peer(address);
270        let limit = self.0.limit;
271        let Ok(before) = peer
272            .inflight
273            .fetch_update(Ordering::AcqRel, Ordering::Acquire, |n| {
274                (n < limit).then_some(n + 1)
275            })
276        else {
277            peer.shed.inc();
278            return None;
279        };
280        peer.at_admission.observe(f64::from(before));
281        let now = before + 1;
282        if peer.high_water.fetch_max(now, Ordering::AcqRel) < now {
283            let _publish = peer.high_water_publish.lock().unwrap();
284            peer.max
285                .set(i64::from(peer.high_water.load(Ordering::Acquire)));
286        }
287        Some(Slot(peer))
288    }
289
290    fn try_admit_work(&self, peer: &Arc<Peer>, at_admission: &Histogram) -> Option<MpcWorkGuard> {
291        let limit = self.0.limit;
292        let Ok(before) =
293            peer.work_inflight
294                .fetch_update(Ordering::AcqRel, Ordering::Acquire, |n| {
295                    (n < limit).then_some(n + 1)
296                })
297        else {
298            peer.work_shed.inc();
299            return None;
300        };
301        at_admission.observe(f64::from(before));
302        Some(MpcWorkGuard {
303            peer: peer.clone(),
304            budget: self.0.budget_bytes,
305            reserved: AtomicU64::new(0),
306        })
307    }
308
309    #[cfg(test)]
310    fn inflight(&self, address: Address) -> u32 {
311        self.peer(address).inflight.load(Ordering::Acquire)
312    }
313
314    #[cfg(test)]
315    fn reserved(&self, address: Address) -> u64 {
316        self.peer(address)
317            .request_bytes
318            .reserved
319            .load(Ordering::Acquire)
320    }
321
322    #[cfg(test)]
323    fn work_inflight(&self, address: Address) -> u32 {
324        self.peer(address).work_inflight.load(Ordering::Acquire)
325    }
326
327    #[cfg(test)]
328    fn work_reserved(&self, address: Address) -> u64 {
329        self.peer(address)
330            .work_bytes
331            .reserved
332            .load(Ordering::Acquire)
333    }
334}

344pub(crate) async fn limit_per_peer(
345    axum::extract::State(limiter): axum::extract::State<PeerInflightLimiter>,
346    request: axum::extract::Request,
347    next: axum::middleware::Next,
348) -> axum::response::Response {
349    let unguarded =
350        MpcWork::Unguarded(limiter.0.metrics.mpc_unguarded_spawn_blocking_total.clone());
351    let Some(address) = request.extensions().get::<Address>().copied() else {
352        return MPC_WORK.scope(unguarded, next.run(request)).await;
353    };
354    let Some(slot) = limiter.try_admit(address) else {
355        return shed(&request);
356    };
357    let path = request.uri().path();
358    let work = match limiter.0.mpc_work.get(path) {
359        Some(at_admission) => match limiter.try_admit_work(&slot.0, at_admission) {
360            Some(guard) => Some(Arc::new(guard)),
361            None => return shed(&request),
362        },
363        None => None,
364    };
365    let charge = Arc::new(RequestCharge {
366        peer: slot.0.clone(),
367        budget: limiter.0.budget_bytes,
368        reserved: AtomicU64::new(0),
369        work: work.clone(),
370    });
371    let request = match limiter.0.routes.limit(path) {
372        Some(limit) => {
373            let single_message = !super::route_limits::streams_requests(path);
374            let charge = charge.clone();
375            request.map(|body| {
376                axum::body::Body::new(LimitedBody::new(body, charge, limit, single_message))
377            })
378        }
379        None => request,
380    };
381    let work = work.map_or(unguarded, MpcWork::Guarded);
382    // Held across the handler, since tonic drops a unary body before calling it.
383    let response = MPC_WORK.scope(work, next.run(request)).await;
384    drop(charge);
385    let budget = super::is_grpc_content_type(response.headers()).then_some(limiter.0.budget_bytes);
386    response.map(|body| axum::body::Body::new(Guarded::new(body, slot, budget)))
387}

393struct Guarded {
394    body: axum::body::Body,
395    slot: Slot,
396    budget: Option<u64>,
397    pending: Option<(Bytes, usize)>,
398    done: bool,
399}
400
401impl Guarded {
402    fn new(body: axum::body::Body, slot: Slot, budget: Option<u64>) -> Self {
403        Self {
404            body,
405            slot,
406            budget,
407            pending: None,
408            done: false,
409        }
410    }
411
412    fn release(&mut self) {
413        if let Some((frame, _)) = self.pending.take() {
414            self.slot
415                .0
416                .request_bytes
417                .release(u64::try_from(frame.len()).unwrap_or(u64::MAX));
418        }
419    }
420}
421
422impl http_body::Body for Guarded {
423    type Data = Bytes;
424    type Error = axum::Error;
425
426    fn poll_frame(
427        self: Pin<&mut Self>,
428        cx: &mut Context<'_>,
429    ) -> Poll<Option<Result<http_body::Frame<Bytes>, axum::Error>>> {
430        let this = self.get_mut();
431        loop {
432            if let Some((frame, sent)) = &mut this.pending
433                && *sent < frame.len()
434            {
435                let end = frame.len().min(*sent + RESPONSE_PIECE);
436                let piece = Bytes::copy_from_slice(&frame[*sent..end]);
437                *sent = end;
438                return Poll::Ready(Some(Ok(http_body::Frame::data(piece))));
439            }
440            this.release();
441            if this.done {
442                return Poll::Ready(None);
443            }
444            let frame = match ready!(Pin::new(&mut this.body).poll_frame(cx)) {
445                Some(Ok(frame)) => frame,
446                other => {
447                    this.done = true;
448                    return Poll::Ready(other);
449                }
450            };
451            let Some(budget) = this.budget else {
452                return Poll::Ready(Some(Ok(frame)));
453            };
454            let data = match frame.into_data() {
455                Ok(data) => data,
456                Err(frame) => return Poll::Ready(Some(Ok(frame))),
457            };
458            let bytes = u64::try_from(data.len()).unwrap_or(u64::MAX);
459            if !this.slot.0.request_bytes.try_reserve(bytes, budget) {
460                this.slot.0.responses_shed.inc();
461                this.done = true;
462                let mut trailers = http::HeaderMap::new();
463                let _ = tonic::Status::unavailable(super::PEER_INFLIGHT_LIMIT_MSG)
464                    .add_header(&mut trailers);
465                return Poll::Ready(Some(Ok(http_body::Frame::trailers(trailers))));
466            }
467            this.pending = Some((data, 0));
468        }
469    }
470
471    fn is_end_stream(&self) -> bool {
472        let pieces_left = self
473            .pending
474            .as_ref()
475            .is_some_and(|(frame, sent)| *sent < frame.len());
476        !pieces_left && (self.done || self.body.is_end_stream())
477    }
478
479    fn size_hint(&self) -> http_body::SizeHint {
480        match self.budget {
481            Some(_) => http_body::SizeHint::default(),
482            None => self.body.size_hint(),
483        }
484    }
485}

487impl Drop for Guarded {
488    fn drop(&mut self) {
489        self.release();
490    }

496    high_water: IntGaugeVec,

499#[derive(Default)]
500struct CallerTasks {
501    active: usize,
502    high_water: usize,
503}
504
505pub(crate) struct CallerTaskSlot {
506    tasks: Arc<Mutex<HashMap<Address, CallerTasks>>>,
507    caller: Address,
508}
509
510impl CallerTaskLimiter {
511    pub(crate) fn new(high_water: IntGaugeVec) -> Self {
512        Self {
513            tasks: Arc::new(Mutex::new(HashMap::new())),
514            high_water,
515        }
516    }
517
518    pub(crate) fn try_admit(&self, caller: Address, limit: usize) -> Option<CallerTaskSlot> {
519        let mut tasks = self.tasks.lock().unwrap();
520        let entry = tasks.entry(caller).or_default();
521        if entry.active >= limit {
522            return None;
523        }
524        entry.active += 1;
525        if entry.active > entry.high_water {
526            entry.high_water = entry.active;
527            self.high_water
528                .with_label_values(&[&caller.to_string()])
529                .set(entry.high_water as i64);
530        }
531        Some(CallerTaskSlot {
532            tasks: self.tasks.clone(),
533            caller,
534        })
535    }
536}
537
538impl Drop for CallerTaskSlot {
539    fn drop(&mut self) {
540        if let Some(entry) = self.tasks.lock().unwrap().get_mut(&self.caller) {
541            entry.active -= 1;
542        }
543    }
544}

640pub(crate) enum GatedRun {
641    Unbounded,
642    Caller {
643        run: OwnedSemaphorePermit,

648impl RunGate {
649    pub(crate) async fn admit(&self, deadline: tokio::time::Instant) -> Option<GatedRun> {
650        match self {
651            RunGate::Unbounded => Some(GatedRun::Unbounded),
652            RunGate::Caller { runs, work_seconds } => {
653                let run = tokio::time::timeout_at(deadline, runs.clone().acquire_owned())
654                    .await
655                    .ok()?
656                    .expect("signing run semaphores are never closed");
657                Some(GatedRun::Caller {
658                    run,
659                    work_seconds: work_seconds.clone(),
660                })
661            }
662        }
663    }
664}
665
666impl GatedRun {
667    pub(crate) async fn run_blocking<T: Send + 'static>(
668        self,
669        work: impl FnOnce() -> T + Send + 'static,
670    ) -> T {
671        match self {
672            GatedRun::Unbounded => {
673                let span = tracing::Span::current();
674                crate::mpc::spawn_blocking(move || span.in_scope(work)).await
675            }
676            GatedRun::Caller { run, work_seconds } => run_holding(run, work_seconds, work).await,
677        }
678    }
679}
680
681async fn run_holding<T: Send + 'static>(
682    run: OwnedSemaphorePermit,
683    work_seconds: Counter,
684    work: impl FnOnce() -> T + Send + 'static,
685) -> T {
686    let span = tracing::Span::current();
687    let job = tokio::task::spawn_blocking(move || {
688        let _span = span.entered();
689        let started = std::time::Instant::now();
690        let output = work();
691        work_seconds.inc_by(started.elapsed().as_secs_f64());
692        drop(run);
693        output
694    });
695    match job.await {
696        Ok(output) => output,
697        Err(e) if e.is_cancelled() => std::future::pending().await,
698        Err(e) => std::panic::resume_unwind(e.into_panic()),
699    }
700}

930    async fn grpc_status(response: h2::client::ResponseFuture) -> tonic::Status {
931        tokio::time::timeout(std::time::Duration::from_secs(10), async {
932            let response = response.await.unwrap();
933            if let Some(status) = tonic::Status::from_header_map(response.headers()) {
934                return status;
935            }
936            let mut body = response.into_body();
937            while let Some(chunk) = body.data().await {
938                let chunk = chunk.unwrap();
939                body.flow_control().release_capacity(chunk.len()).unwrap();
940            }
941            tonic::Status::from_header_map(&body.trailers().await.unwrap().unwrap()).unwrap()
942        })
943        .await
944        .unwrap()
945    }
