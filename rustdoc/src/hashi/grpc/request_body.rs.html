17pub(super) const PREFIX_LEN: u64 = 5;
18
19const END_STREAM_WAIT: Duration = Duration::from_millis(100);
20
21pub(super) struct LimitedBody {
22    inner: axum::body::Body,
23    charge: Arc<RequestCharge>,
24    limit: usize,
25    single_message: bool,
26    current: u64,
27    completed: u64,
28    state: State,
29}
30
31enum State {
32    Prefix {
33        bytes: [u8; PREFIX_LEN as usize],
34        filled: usize,
35    },
36    Message {
37        remaining: u64,
38    },
39    AwaitingEnd(Pin<Box<tokio::time::Sleep>>),
40    Done,
41}
42
43impl State {
44    fn prefix() -> Self {
45        Self::Prefix {
46            bytes: [0; PREFIX_LEN as usize],
47            filled: 0,
48        }
49    }
50}
51
52impl LimitedBody {
53    pub(super) fn new(
54        inner: axum::body::Body,
55        charge: Arc<RequestCharge>,
56        limit: usize,
57        single_message: bool,
58    ) -> Self {
59        Self {
60            inner,
61            charge,
62            limit,
63            single_message,
64            current: 0,
65            completed: 0,
66            state: State::prefix(),
67        }
68    }
69
70    fn forwardable(&mut self, data: &[u8]) -> Result<usize, tonic::Status> {
71        let mut consumed = 0;
72        while consumed < data.len() {
73            match &mut self.state {
74                State::Prefix { bytes, filled } => {
75                    let take = (bytes.len() - *filled).min(data.len() - consumed);
76                    bytes[*filled..*filled + take]
77                        .copy_from_slice(&data[consumed..consumed + take]);
78                    *filled += take;
79                    consumed += take;
80                    if *filled == bytes.len() {
81                        let len = u32::from_be_bytes([bytes[1], bytes[2], bytes[3], bytes[4]]);
82                        self.admit(len)?;
83                    }
84                }
85                State::Message { remaining } => {
86                    let take = u64::try_from(data.len() - consumed)
87                        .unwrap_or(u64::MAX)
88                        .min(*remaining);
89                    *remaining -= take;
90                    consumed += take as usize;
91                    if *remaining == 0 {
92                        self.message_complete();
93                    }
94                }
95                State::AwaitingEnd(_) | State::Done => break,
96            }
97        }
98        Ok(consumed)
99    }
100
101    fn admit(&mut self, len: u32) -> Result<(), tonic::Status> {
102        let limit = self.limit;
103        if usize::try_from(len).is_ok_and(|len| len > limit) {
104            self.charge.refuse_too_large();
105            return Err(tonic::Status::out_of_range(format!(
106                "declared message length {len} exceeds this route's limit of {limit} bytes"
107            )));
108        }
109        let bytes = PREFIX_LEN + u64::from(len);
110        if !self.charge.reserve(bytes) {
111            return Err(tonic::Status::unavailable(super::PEER_INFLIGHT_LIMIT_MSG));
112        }
113        self.current = bytes;
114        self.state = State::Message {
115            remaining: u64::from(len),
116        };
117        if len == 0 {
118            self.message_complete();
119        }
120        Ok(())
121    }
122
123    fn message_complete(&mut self) {
124        self.state = if self.single_message {
125            State::AwaitingEnd(Box::pin(tokio::time::sleep(END_STREAM_WAIT)))
126        } else {
127            self.completed += std::mem::take(&mut self.current);
128            State::prefix()
129        };
130    }
131}
132
133impl http_body::Body for LimitedBody {
134    type Data = Bytes;
135    type Error = axum::Error;
136
137    fn poll_frame(
138        self: Pin<&mut Self>,
139        cx: &mut Context<'_>,
140    ) -> Poll<Option<Result<Frame<Bytes>, axum::Error>>> {
141        let this = self.get_mut();
142        this.charge.release(std::mem::take(&mut this.completed));
143        loop {
144            if let State::AwaitingEnd(deadline) = &mut this.state {
145                let polled = Pin::new(&mut this.inner).poll_frame(cx);
146                if let Poll::Ready(Some(Ok(frame))) = &polled
147                    && frame.data_ref().is_some_and(Bytes::is_empty)
148                {
149                    continue;
150                }
151                if polled.is_pending() && deadline.as_mut().poll(cx).is_pending() {
152                    return Poll::Pending;
153                }
154                this.state = State::Done;
155                return match polled {
156                    Poll::Ready(Some(Ok(frame))) if frame.is_trailers() => {
157                        Poll::Ready(Some(Ok(frame)))
158                    }
159                    Poll::Ready(Some(Err(error))) => Poll::Ready(Some(Err(error))),
160                    _ => Poll::Ready(None),
161                };
162            }
163            if matches!(this.state, State::Done) {
164                return Poll::Ready(None);
165            }
166            let frame = match ready!(Pin::new(&mut this.inner).poll_frame(cx)) {
167                Some(Ok(frame)) => frame,
168                other => {
169                    this.state = State::Done;
170                    return Poll::Ready(other);
171                }
172            };
173            let data = match frame.into_data() {
174                Ok(data) => data,
175                Err(frame) => return Poll::Ready(Some(Ok(frame))),
176            };
177            match this.forwardable(&data) {
178                Ok(0) => continue,
179                Ok(forward) => return Poll::Ready(Some(Ok(Frame::data(data.slice(..forward))))),
180                Err(status) => {
181                    this.state = State::Done;
182                    return Poll::Ready(Some(Err(axum::Error::new(status))));
183                }
184            }
185        }
186    }
187
188    fn is_end_stream(&self) -> bool {
189        matches!(self.state, State::Done)
190    }
191}
