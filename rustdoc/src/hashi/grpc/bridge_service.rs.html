65    #[tracing::instrument(
66        level = "info",
67        skip_all,
68        fields(deposit_id = tracing::field::Empty, caller = tracing::field::Empty),
69    )]
70    async fn sign_deposit_confirmation(
71        &self,
72        request: Request<SignDepositConfirmationRequest>,
73    ) -> Result<Response<SignDepositConfirmationResponse>, Status> {
74        let caller = authenticate_caller(&request)?;
75        tracing::Span::current().record("caller", tracing::field::display(&caller));
76        let deposit_request = parse_deposit_request(request.get_ref())
77            .map_err(|e| Status::invalid_argument(e.to_string()))?;
78        tracing::Span::current().record("deposit_id", tracing::field::display(&deposit_request.id));
79        let member_signature = self
80            .inner
81            .validate_and_sign_deposit_confirmation(&deposit_request)
82            .await
83            .map_err(deposit_refusal_status)?;
84        tracing::info!(
85            utxo_txid = %deposit_request.utxo.id.txid,
86            utxo_vout = deposit_request.utxo.id.vout,
87            amount = deposit_request.utxo.amount,
88            "Signed deposit confirmation",
89        );
90        Ok(Response::new(SignDepositConfirmationResponse {
91            member_signature: Some(member_signature),
92        }))
93    }

212    #[tracing::instrument(
213        level = "info",
214        skip_all,
215        fields(withdrawal_txn_id = tracing::field::Empty, caller = tracing::field::Empty),
216    )]
217    async fn sign_withdrawal_transaction(
218        &self,
219        request: Request<SignWithdrawalTransactionRequest>,
220    ) -> Result<Response<Self::SignWithdrawalTransactionStream>, Status> {
221        let caller = authenticate_caller(&request)?;
222        tracing::Span::current().record("caller", tracing::field::display(&caller));
223        self.admit_committee_member(caller)?;
224        let req = request.get_ref();
225        let withdrawal_txn_id = Address::from_bytes(&req.withdrawal_txn_id)
226            .map_err(|e| Status::invalid_argument(format!("invalid withdrawal_txn_id: {e}")))?;
227        tracing::Span::current().record(
228            "withdrawal_txn_id",
229            tracing::field::display(&withdrawal_txn_id),
230        );
231        self.inner
232            .check_withdrawal_signing_request(&withdrawal_txn_id, &req.input_indices)
233            .inspect_err(|e| {
234                if !e.is::<MalformedSigningRequest>() {
235                    tracing::error!("sign_withdrawal_transaction failed: {e}");
236                }
237            })
238            .map_err(|e| self.signing_refusal(caller, e))?;
239        let slot = self.admit_withdrawal_signing(caller)?;
240        let requested_input_indices = req.input_indices.clone();
241        tracing::info!("sign_withdrawal_transaction called");
242        let concurrency = self.inner.config.withdrawal_signing_concurrency();
243        let (tx, rx) = tokio::sync::mpsc::channel(concurrency);
244        let inner = self.inner.clone();
245        tokio::spawn(async move {
246            let _slot = slot;
247            if let Err(e) = inner
248                .validate_and_sign_withdrawal_tx(
249                    &withdrawal_txn_id,
250                    &requested_input_indices,
251                    tx.clone(),
252                )
253                .await
254            {
255                tracing::error!("sign_withdrawal_transaction failed: {e}");
256                let _ = tx
257                    .send(Err(Status::failed_precondition(e.to_string())))
258                    .await;
259            }
260        });
261        Ok(Response::new(tokio_stream::wrappers::ReceiverStream::new(
262            rx,
263        )))
264    }

356const SIGNING_TASK_LIMIT_MSG: &str = "per-caller withdrawal signing limit reached";
357
358impl HttpService {
359    fn count_signing_refusal(&self, caller: Address, reason: &str) {
360        self.inner
361            .metrics
362            .withdrawal_signing_refused_total
363            .with_label_values(&[caller.to_string().as_str(), reason])
364            .inc();
365    }
366
367    fn admit_committee_member(&self, caller: Address) -> Result<(), Status> {
368        let refusal = {
369            let state = self.inner.onchain_state().state();
370            let committees = &state.hashi().committees;
371            let epoch = committees.epoch();
372            match committees.current_committee() {
373                Some(committee) if committee.index_of(&caller).is_some() => return Ok(()),
374                Some(_) => format!("caller is not in the epoch {epoch} committee"),
375                None => format!("this node does not know the epoch {epoch} committee"),
376            }
377        };
378        self.count_signing_refusal(caller, "committee");
379        Err(Status::permission_denied(refusal))
380    }
381
382    fn admit_withdrawal_signing(&self, caller: Address) -> Result<CallerTaskSlot, Status> {
383        let limit = self.inner.config.withdrawal_signing_per_caller_limit();
384        self.signing_tasks.try_admit(caller, limit).ok_or_else(|| {
385            self.count_signing_refusal(caller, "cap");
386            Status::unavailable(SIGNING_TASK_LIMIT_MSG)
387        })
388    }
389
390    fn signing_refusal(&self, caller: Address, err: anyhow::Error) -> Status {
391        if err.is::<MalformedSigningRequest>() {
392            self.count_signing_refusal(caller, "shape");
393        }
394        withdrawal_signing_refusal_status(err)
395    }
396}

398/// `AlreadyExists` tells the leader another leader already landed this epoch's
399/// approval, so it can drop the deposit instead of retrying it.
400pub(crate) fn deposit_refusal_status(err: UnapprovedDepositError) -> Status {
401    match err {
402        UnapprovedDepositError::AlreadyApprovedThisEpoch => Status::already_exists(err.to_string()),
403        err => Status::failed_precondition(err.to_string()),
404    }
405}
406
407/// `AlreadyExists` tells the leader the request is already approved or
408/// committed, so it can stop collecting signatures for it.
409pub(crate) fn withdrawal_approval_refusal_status(err: WithdrawalApprovalError) -> Status {
410    match err {
411        WithdrawalApprovalError::AlreadyApproved(_) => Status::already_exists(err.to_string()),
412        err => Status::failed_precondition(err.to_string()),
413    }
414}
415
416/// `AlreadyExists` tells the leader the withdrawal is already finalized, so it
417/// can stop collecting signatures for it.
418pub(crate) fn withdrawal_signing_refusal_status(err: anyhow::Error) -> Status {
419    if err.is::<WithdrawalAlreadyFinalized>() {
420        Status::already_exists(err.to_string())
421    } else {
422        Status::failed_precondition(err.to_string())
423    }
424}

426fn authenticate_caller<T>(request: &Request<T>) -> Result<Address, Status> {
427    request
428        .extensions()
429        .get::<Address>()
430        .copied()
431        .ok_or_else(|| Status::permission_denied("unknown validator"))
432}

434fn parse_deposit_request(
435    request: &SignDepositConfirmationRequest,
436) -> anyhow::Result<DepositRequest> {
437    let id = parse_address(&request.id)?;
438    let txid = parse_address(&request.txid)?.into();
439    let derivation_path = request
440        .derivation_path
441        .as_ref()
442        .map(|bytes| parse_address(bytes))
443        .transpose()?;
444    let requester_address = parse_address(&request.requester_address)?;
445    let sui_tx_digest = sui_sdk_types::Digest::new(
446        request
447            .sui_tx_digest
448            .as_ref()
449            .try_into()
450            .map_err(|_| anyhow::anyhow!("sui_tx_digest must be 32 bytes"))?,
451    );
452
453    Ok(DepositRequest {
454        id,
455        sender: requester_address,
456        created_timestamp_ms: request.timestamp_ms,
457        sui_tx_digest,
458        utxo: Utxo {
459            id: UtxoId {
460                txid,
461                vout: request.vout,
462            },
463            amount: request.amount,
464            derivation_path,
465        },
466        // Approval state isn't carried by the proto. Validators look up
467        // the on-chain version separately when verifying.
468        approval_cert: None,
469        approved_timestamp_ms: None,
470        confirmed_timestamp_ms: None,
471    })
472}

481fn parse_withdrawal_tx_commitment(
482    request: &SignWithdrawalTxConstructionRequest,
483) -> anyhow::Result<WithdrawalTxCommitment> {
484    let request_ids: Vec<Address> = request
485        .request_ids
486        .iter()
487        .map(|bytes| parse_address(bytes))
488        .collect::<anyhow::Result<_>>()?;
489    let selected_utxos: Vec<UtxoId> = request
490        .selected_utxos
491        .iter()
492        .map(|utxo_id| {
493            let txid: BitcoinTxid = utxo_id
494                .txid
495                .as_ref()
496                .map(|bytes| parse_address(bytes))
497                .context("missing utxo txid")??
498                .into();
499            let vout = utxo_id.vout.context("missing utxo vout")?;
500            Ok(UtxoId { txid, vout })
501        })
502        .collect::<anyhow::Result<_>>()?;
503    let outputs = request
504        .outputs
505        .iter()
506        .map(|output| OutputUtxo {
507            amount: output.amount,
508            bitcoin_address: output.bitcoin_address.to_vec(),
509        })
510        .collect();
511    let txid = parse_address(&request.txid)?.into();
512
513    Ok(WithdrawalTxCommitment {
514        request_ids,
515        selected_utxos,
516        outputs,
517        txid,
518    })
519}

521fn parse_withdrawal_tx_signing(
522    request: &SignWithdrawalTxSigningRequest,
523) -> anyhow::Result<WithdrawalTxSigning> {
524    let withdrawal_id = parse_address(&request.withdrawal_id)?;
525    let signatures: Vec<Vec<u8>> = request
526        .signatures
527        .iter()
528        .map(|bytes| bytes.to_vec())
529        .collect();
530    let guardian_signatures: Vec<Vec<u8>> = request
531        .guardian_signatures
532        .iter()
533        .map(|bytes| bytes.to_vec())
534        .collect();
535    Ok(WithdrawalTxSigning {
536        withdrawal_id,
537        signatures,
538        guardian_signatures,
539    })
540}
