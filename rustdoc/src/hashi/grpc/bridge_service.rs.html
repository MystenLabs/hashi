156    #[tracing::instrument(
157        level = "info",
158        skip_all,
159        fields(bitcoin_txid = tracing::field::Empty, caller = tracing::field::Empty),
160    )]
161    async fn sign_withdrawal_tx_construction(
162        &self,
163        request: Request<SignWithdrawalTxConstructionRequest>,
164    ) -> Result<Response<SignWithdrawalTxConstructionResponse>, Status> {
165        let admit_deadline = tokio::time::Instant::now() + ADMIT_WINDOW;
166        let caller = authenticate_caller(&request)?;
167        tracing::Span::current().record("caller", tracing::field::display(&caller));
168        let member = self.admit_committee_member(caller)?;
169        self.admit_construction(&member)?;
170        let approval = parse_withdrawal_tx_commitment(request.get_ref())
171            .map_err(|e| Status::invalid_argument(e.to_string()))?;
172        tracing::Span::current().record("bitcoin_txid", tracing::field::display(&approval.txid));
173        check_commitment_shape(&approval)
174            .map_err(|e| Status::failed_precondition(e.to_string()))?;
175        let approval = Arc::new(approval);
176        let place = self.take_signing_place(&member)?;
177        let work_seconds = self.work_seconds(caller);
178        let gate = place.run_gate(work_seconds.clone());
179        let run = place.run().await;
180        let inner = self.inner.clone();
181        let job_approval = approval.clone();
182        let (prepared, _place) = run
183            .run_blocking(work_seconds, move || {
184                inner.prepare_withdrawal_tx_commitment(&job_approval)
185            })
186            .await;
187        let prepared = prepared.map_err(|e| Status::failed_precondition(e.to_string()))?;
188        let charge = self
189            .bitcoind
190            .gate(caller, admit_deadline)
191            .take_place()
192            .await
193            .map_err(|e| Status::unavailable(e.to_string()))?;
194        let fee_rate = self
195            .inner
196            .withdrawal_commitment_fee_rate(charge)
197            .await
198            .map_err(|e| Status::failed_precondition(e.to_string()))?;
199        let Some(run) = gate.admit(admit_deadline).await else {
200            self.count_signing_refusal(caller, SIGNING_REFUSAL_BUSY);
201            return Err(Status::unavailable(SIGNING_BUSY_MSG));
202        };
203        let inner = self.inner.clone();
204        let job_approval = approval.clone();
205        let member_signature = run
206            .run_blocking(move || {
207                inner.finish_withdrawal_tx_commitment(&job_approval, &prepared, fee_rate)?;
208                inner.sign_withdrawal_tx_commitment(&job_approval)
209            })
210            .await
211            .map_err(|e| self.construction_refusal(caller, e))?;
212        tracing::info!(
213            requests = approval.request_ids.len(),
214            "Signed withdrawal tx construction",
215        );
216        Ok(Response::new(SignWithdrawalTxConstructionResponse {
217            member_signature: Some(member_signature),
218        }))
219    }

288    async fn sign_withdrawal_transaction(
289        &self,
290        request: Request<SignWithdrawalTransactionRequest>,
291    ) -> Result<Response<Self::SignWithdrawalTransactionStream>, Status> {
292        let caller = authenticate_caller(&request)?;
293        tracing::Span::current().record("caller", tracing::field::display(&caller));
294        let member = self.admit_committee_member(caller)?;
295        let req = request.get_ref();
296        let withdrawal_txn_id = Address::from_bytes(&req.withdrawal_txn_id).map_err(|e| {
297            self.malformed_request(caller, format!("invalid withdrawal_txn_id: {e}"))
298        })?;
299        tracing::Span::current().record(
300            "withdrawal_txn_id",
301            tracing::field::display(&withdrawal_txn_id),
302        );
303        self.inner
304            .check_withdrawal_signing_request(&withdrawal_txn_id, &req.input_indices)
305            .inspect_err(|e| {
306                if !e.is::<MalformedSigningRequest>() {
307                    tracing::error!("sign_withdrawal_transaction failed: {e}");
308                }
309            })
310            .map_err(|e| self.signing_refusal(caller, e))?;
311        let slot = self.admit_withdrawal_signing(&member)?;
312        let place = self.take_signing_place(&member)?;
313        let requested_input_indices = req.input_indices.clone();
314        tracing::info!("sign_withdrawal_transaction called");
315        let concurrency = self.inner.config.withdrawal_signing_concurrency();
316        let (tx, rx) = tokio::sync::mpsc::channel(concurrency);
317        let inner = self.inner.clone();
318        let work_seconds = self.work_seconds(caller);
319        let service = self.clone();
320        tokio::spawn(
321            async move {
322                let _slot = slot;
323                let gate = place.run_gate(work_seconds.clone());
324                let run = tokio::select! {
325                    biased;
326                    () = tx.closed() => return,
327                    run = place.run() => run,
328                };
329                if let Err(refusal) = service.ensure_signing_available(caller) {
330                    tracing::info!("Refusing sign_withdrawal_transaction: {refusal}");
331                    let _ = tx.send(Err(refusal)).await;
332                    return;
333                }
334                let job_inner = inner.clone();
335                let (signing, place) = run
336                    .run_blocking(work_seconds, move || {
337                        job_inner
338                            .prepare_withdrawal_signing(
339                                &withdrawal_txn_id,
340                                &requested_input_indices,
341                            )
342                            .map(|prepared| job_inner.sign_withdrawal_locally(prepared))
343                    })
344                    .await;
345                drop(place);
346                let signed = match signing {
347                    Ok(signing) => {
348                        inner
349                            .mpc_sign_withdrawal_tx(signing, &gate, tx.clone())
350                            .await
351                    }
352                    Err(e) => Err(e),
353                };
354                if let Err(e) = signed {
355                    tracing::error!("sign_withdrawal_transaction failed: {e}");
356                    let _ = tx
357                        .send(Err(Status::failed_precondition(e.to_string())))
358                        .await;
359                }
360            }
361            .instrument(tracing::Span::current()),
362        );
363        Ok(Response::new(tokio_stream::wrappers::ReceiverStream::new(
364            rx,
365        )))
366    }

487const SIGNING_TASK_LIMIT_MSG: &str = "per-caller withdrawal signing limit reached";
488const SIGNING_BUSY_MSG: &str = "per-caller withdrawal signing queue is full";

490const ADMIT_WINDOW: Duration = Duration::from_secs(super::client::REQUEST_TIMEOUT.as_secs() + 5);

492struct Member(Address);

494impl HttpService {
495    pub(crate) fn count_signing_refusal(&self, caller: Address, reason: &str) {
496        self.signing_refusals(caller, reason).inc();
497    }
498
499    fn signing_refusals(&self, caller: Address, reason: &str) -> prometheus::IntCounter {
500        self.inner
501            .metrics
502            .withdrawal_signing_refused_total
503            .with_label_values(&[caller.to_string().as_str(), reason])
504    }
505
506    fn admit_committee_member(&self, caller: Address) -> Result<Member, Status> {
507        let refusal = {
508            let state = self.inner.onchain_state().state();
509            let hashi = state.hashi();
510            let epoch = hashi.committees.epoch();
511            match hashi.committees.current_committee() {
512                Some(committee) if committee.index_of(&caller).is_some() => None,
513                Some(_) => Some(format!("caller is not in the epoch {epoch} committee")),
514                None => Some(format!(
515                    "this node does not know the epoch {epoch} committee"
516                )),
517            }
518        };
519        if let Some(refusal) = refusal {
520            self.count_signing_refusal(caller, SIGNING_REFUSAL_COMMITTEE);
521            return Err(Status::permission_denied(refusal));
522        }
523        self.ensure_signing_available(caller)?;
524        Ok(Member(caller))
525    }
526
527    pub(crate) fn ensure_signing_available(&self, caller: Address) -> Result<(), Status> {
528        let onchain_state = self
529            .onchain_state_opt()
530            .ok_or_else(|| Status::unavailable(SIGNING_MANAGER_NOT_READY_MSG))?;
531        let Some(reason) = onchain_state.autonomous_halt_reason() else {
532            return Ok(());
533        };
534        let (metric_reason, message) = match reason {
535            HaltReason::Paused => (SIGNING_REFUSAL_PAUSED, PAUSED_REFUSAL_MSG),
536            HaltReason::BinaryUnsupported { .. } => (
537                SIGNING_REFUSAL_BINARY_UNSUPPORTED,
538                BINARY_UNSUPPORTED_REFUSAL_MSG,
539            ),
540        };
541        self.count_signing_refusal(caller, metric_reason);
542        Err(Status::failed_precondition(message))
543    }
544
545    fn admit_construction(&self, member: &Member) -> Result<(), Status> {
546        if !self.inner.onchain_state().caught_up_with_scrape() {
547            self.count_signing_refusal(member.0, SIGNING_REFUSAL_CATCHING_UP);
548            return Err(Status::failed_precondition(CATCHING_UP_REFUSAL_MSG));
549        }
550        if self.inner.onchain_state().has_unsigned_withdrawal_txn() {
551            self.count_signing_refusal(member.0, SIGNING_REFUSAL_UNSIGNED);
552            return Err(Status::failed_precondition(UNSIGNED_REFUSAL_MSG));
553        }
554        Ok(())
555    }
556
557    fn admit_withdrawal_signing(&self, member: &Member) -> Result<CallerTaskSlot, Status> {
558        let limit = self.inner.config.withdrawal_signing_per_caller_limit();
559        self.signing_tasks
560            .try_admit(member.0, limit)
561            .ok_or_else(|| {
562                self.count_signing_refusal(member.0, SIGNING_REFUSAL_CAP);
563                Status::unavailable(SIGNING_TASK_LIMIT_MSG)
564            })
565    }
566
567    fn take_signing_place(&self, member: &Member) -> Result<SigningPlace, Status> {
568        self.signing_work.try_take_place(member.0).ok_or_else(|| {
569            self.count_signing_refusal(member.0, SIGNING_REFUSAL_BUSY);
570            Status::unavailable(SIGNING_BUSY_MSG)
571        })
572    }
573
574    fn work_seconds(&self, caller: Address) -> prometheus::Counter {
575        self.inner
576            .metrics
577            .peer_bridge_work_seconds_total
578            .with_label_values(&[caller.to_string().as_str()])
579    }
580
581    fn malformed_request(&self, caller: Address, err: impl std::fmt::Display) -> Status {
582        self.count_signing_refusal(caller, SIGNING_REFUSAL_SHAPE);
583        Status::invalid_argument(err.to_string())
584    }
585
586    fn signing_refusal(&self, caller: Address, err: anyhow::Error) -> Status {
587        if err.is::<MalformedSigningRequest>() {
588            self.count_signing_refusal(caller, SIGNING_REFUSAL_SHAPE);
589        }
590        withdrawal_signing_refusal_status(err)
591    }
592
593    fn construction_refusal(&self, caller: Address, err: anyhow::Error) -> Status {
594        if err.is::<MirrorCatchingUp>() {
595            self.count_signing_refusal(caller, SIGNING_REFUSAL_CATCHING_UP);
596        } else if err.is::<WithdrawalStillUnsigned>() {
597            self.count_signing_refusal(caller, SIGNING_REFUSAL_UNSIGNED);
598        } else if err.is::<OldestRequestMissing>() {
599            self.count_signing_refusal(caller, SIGNING_REFUSAL_NOT_OLDEST);
600        } else if err.is::<AboveLimiterCapacity>() {
601            self.count_signing_refusal(caller, SIGNING_REFUSAL_OVER_CAPACITY);
602        }
603        Status::failed_precondition(err.to_string())
604    }
605}

607/// All a peer learns when this node's sanctions check has not approved a request:
608/// its details are the operator's own.
609pub(crate) const SANCTIONS_CHECK_REFUSAL_MSG: &str =
610    "Sanctions check has not approved this request";
611
612/// `AlreadyExists` tells the leader another leader already landed this epoch's
613/// approval, so it can drop the deposit instead of retrying it.
614pub(crate) fn deposit_refusal_status(err: UnapprovedDepositError) -> Status {
615    match err {
616        UnapprovedDepositError::AlreadyApprovedThisEpoch => Status::already_exists(err.to_string()),
617        UnapprovedDepositError::BitcoindBusy(_) => Status::unavailable(err.to_string()),
618        UnapprovedDepositError::AmlRejected(_) | UnapprovedDepositError::AmlServiceError(_) => {
619            Status::failed_precondition(SANCTIONS_CHECK_REFUSAL_MSG)
620        }
621        err => Status::failed_precondition(err.to_string()),
622    }
623}
624
625/// `AlreadyExists` tells the leader the request is already approved or
626/// committed, so it can stop collecting signatures for it.
627pub(crate) fn withdrawal_approval_refusal_status(err: WithdrawalApprovalError) -> Status {
628    match err {
629        WithdrawalApprovalError::AlreadyApproved(_) => Status::already_exists(err.to_string()),
630        WithdrawalApprovalError::SanctionsCheckRejected(_)
631        | WithdrawalApprovalError::AmlServiceError(_) => {
632            Status::failed_precondition(SANCTIONS_CHECK_REFUSAL_MSG)
633        }
634        WithdrawalApprovalError::NeverRetry(_) => Status::failed_precondition(err.to_string()),
635    }
636}
637
638/// `AlreadyExists` tells the leader the withdrawal is already finalized, so it
639/// can stop collecting signatures for it.
640pub(crate) fn withdrawal_signing_refusal_status(err: anyhow::Error) -> Status {
641    if err.is::<WithdrawalAlreadyFinalized>() {
642        Status::already_exists(err.to_string())
643    } else {
644        Status::failed_precondition(err.to_string())
645    }
646}

656fn parse_deposit_request(
657    request: &SignDepositConfirmationRequest,
658) -> anyhow::Result<DepositRequest> {
659    let id = parse_address(&request.id)?;
660    let txid = parse_address(&request.txid)?.into();
661    let derivation_path = request
662        .derivation_path
663        .as_ref()
664        .map(|bytes| parse_address(bytes))
665        .transpose()?;
666    let requester_address = parse_address(&request.requester_address)?;
667    let sui_tx_digest = sui_sdk_types::Digest::new(
668        request
669            .sui_tx_digest
670            .as_ref()
671            .try_into()
672            .map_err(|_| anyhow::anyhow!("sui_tx_digest must be 32 bytes"))?,
673    );
674
675    Ok(DepositRequest {
676        id,
677        sender: requester_address,
678        created_timestamp_ms: request.timestamp_ms,
679        sui_tx_digest,
680        utxo: Utxo {
681            id: UtxoId {
682                txid,
683                vout: request.vout,
684            },
685            amount: request.amount,
686            derivation_path,
687        },
688        // Approval state isn't carried by the proto. Validators look up
689        // the on-chain version separately when verifying.
690        approval_cert: None,
691        approved_timestamp_ms: None,
692        confirmed_timestamp_ms: None,
693    })
694}

703fn parse_withdrawal_tx_commitment(
704    request: &SignWithdrawalTxConstructionRequest,
705) -> anyhow::Result<WithdrawalTxCommitment> {
706    let request_ids: Vec<Address> = request
707        .request_ids
708        .iter()
709        .map(|bytes| parse_address(bytes))
710        .collect::<anyhow::Result<_>>()?;
711    let selected_utxos: Vec<UtxoId> = request
712        .selected_utxos
713        .iter()
714        .map(|utxo_id| {
715            let txid: BitcoinTxid = utxo_id
716                .txid
717                .as_ref()
718                .map(|bytes| parse_address(bytes))
719                .context("missing utxo txid")??
720                .into();
721            let vout = utxo_id.vout.context("missing utxo vout")?;
722            Ok(UtxoId { txid, vout })
723        })
724        .collect::<anyhow::Result<_>>()?;
725    let outputs = request
726        .outputs
727        .iter()
728        .map(|output| OutputUtxo {
729            amount: output.amount,
730            bitcoin_address: output.bitcoin_address.to_vec(),
731        })
732        .collect();
733    let txid = parse_address(&request.txid)?.into();
734
735    Ok(WithdrawalTxCommitment {
736        request_ids,
737        selected_utxos,
738        outputs,
739        txid,
740    })
741}
