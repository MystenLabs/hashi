63    /// Validate and sign a confirmation of a bitcoin deposit request.
64    #[tracing::instrument(
65        level = "info",
66        skip_all,
67        fields(deposit_id = tracing::field::Empty, caller = tracing::field::Empty),
68    )]
69    async fn sign_deposit_confirmation(
70        &self,
71        request: Request<SignDepositConfirmationRequest>,
72    ) -> Result<Response<SignDepositConfirmationResponse>, Status> {
73        let caller = authenticate_caller(&request)?;
74        tracing::Span::current().record("caller", tracing::field::display(&caller));
75        let deposit_request = parse_deposit_request(request.get_ref())
76            .map_err(|e| Status::invalid_argument(e.to_string()))?;
77        tracing::Span::current().record("deposit_id", tracing::field::display(&deposit_request.id));
78        let member_signature = self
79            .inner
80            .validate_and_sign_deposit_confirmation(&deposit_request)
81            .await
82            .map_err(deposit_refusal_status)?;
83        tracing::info!(
84            utxo_txid = %deposit_request.utxo.id.txid,
85            utxo_vout = deposit_request.utxo.id.vout,
86            amount = deposit_request.utxo.amount,
87            "Signed deposit confirmation",
88        );
89        Ok(Response::new(SignDepositConfirmationResponse {
90            member_signature: Some(member_signature),
91        }))
92    }

211    #[tracing::instrument(
212        level = "info",
213        skip_all,
214        fields(withdrawal_txn_id = tracing::field::Empty, caller = tracing::field::Empty),
215    )]
216    async fn sign_withdrawal_transaction(
217        &self,
218        request: Request<SignWithdrawalTransactionRequest>,
219    ) -> Result<Response<Self::SignWithdrawalTransactionStream>, Status> {
220        let caller = authenticate_caller(&request)?;
221        tracing::Span::current().record("caller", tracing::field::display(&caller));
222        let slot = self.admit_withdrawal_signing(caller)?;
223        let req = request.get_ref();
224        let withdrawal_txn_id = Address::from_bytes(&req.withdrawal_txn_id)
225            .map_err(|e| Status::invalid_argument(format!("invalid withdrawal_txn_id: {e}")))?;
226        tracing::Span::current().record(
227            "withdrawal_txn_id",
228            tracing::field::display(&withdrawal_txn_id),
229        );
230        let requested_input_indices = req.input_indices.clone();
231        tracing::info!("sign_withdrawal_transaction called");
232        let concurrency = self.inner.config.withdrawal_signing_concurrency();
233        let (tx, rx) = tokio::sync::mpsc::channel(concurrency);
234        let inner = self.inner.clone();
235        tokio::spawn(async move {
236            let _slot = slot;
237            if let Err(e) = inner
238                .validate_and_sign_withdrawal_tx(
239                    &withdrawal_txn_id,
240                    &requested_input_indices,
241                    tx.clone(),
242                )
243                .await
244            {
245                tracing::error!("sign_withdrawal_transaction failed: {e}");
246                let _ = tx
247                    .send(Err(Status::failed_precondition(e.to_string())))
248                    .await;
249            }
250        });
251        Ok(Response::new(tokio_stream::wrappers::ReceiverStream::new(
252            rx,
253        )))
254    }

344const SIGNING_TASK_LIMIT_MSG: &str = "per-caller withdrawal signing limit reached";
345
346impl HttpService {
347    fn admit_withdrawal_signing(&self, caller: Address) -> Result<CallerTaskSlot, Status> {
348        let in_committee = self
349            .inner
350            .onchain_state()
351            .state()
352            .hashi()
353            .committees
354            .current_committee()
355            .is_some_and(|committee| committee.index_of(&caller).is_some());
356        let caller_label = caller.to_string();
357        let refused = |reason: &str| {
358            self.inner
359                .metrics
360                .withdrawal_signing_refused_total
361                .with_label_values(&[caller_label.as_str(), reason])
362                .inc()
363        };
364        if !in_committee {
365            refused("committee");
366            return Err(Status::permission_denied(
367                "caller is not in the current committee",
368            ));
369        }
370        let limit = self.inner.config.withdrawal_signing_per_caller_limit();
371        self.signing_tasks.try_admit(caller, limit).ok_or_else(|| {
372            refused("cap");
373            Status::unavailable(SIGNING_TASK_LIMIT_MSG)
374        })
375    }
376}

378/// `AlreadyExists` tells the leader another leader already landed this epoch's
379/// approval, so it can drop the deposit instead of retrying it.
380pub(crate) fn deposit_refusal_status(err: UnapprovedDepositError) -> Status {
381    match err {
382        UnapprovedDepositError::AlreadyApprovedThisEpoch => Status::already_exists(err.to_string()),
383        err => Status::failed_precondition(err.to_string()),
384    }
385}
386
387/// `AlreadyExists` tells the leader the request is already approved or
388/// committed, so it can stop collecting signatures for it.
389pub(crate) fn withdrawal_approval_refusal_status(err: WithdrawalApprovalError) -> Status {
390    match err {
391        WithdrawalApprovalError::AlreadyApproved(_) => Status::already_exists(err.to_string()),
392        err => Status::failed_precondition(err.to_string()),
393    }
394}
395
396/// `AlreadyExists` tells the leader the withdrawal is already finalized, so it
397/// can stop collecting signatures for it.
398pub(crate) fn withdrawal_signing_refusal_status(err: anyhow::Error) -> Status {
399    if err.is::<WithdrawalAlreadyFinalized>() {
400        Status::already_exists(err.to_string())
401    } else {
402        Status::failed_precondition(err.to_string())
403    }
404}

406fn authenticate_caller<T>(request: &Request<T>) -> Result<Address, Status> {
407    request
408        .extensions()
409        .get::<Address>()
410        .copied()
411        .ok_or_else(|| Status::permission_denied("unknown validator"))
412}

414fn parse_deposit_request(
415    request: &SignDepositConfirmationRequest,
416) -> anyhow::Result<DepositRequest> {
417    let id = parse_address(&request.id)?;
418    let txid = parse_address(&request.txid)?.into();
419    let derivation_path = request
420        .derivation_path
421        .as_ref()
422        .map(|bytes| parse_address(bytes))
423        .transpose()?;
424    let requester_address = parse_address(&request.requester_address)?;
425    let sui_tx_digest = sui_sdk_types::Digest::new(
426        request
427            .sui_tx_digest
428            .as_ref()
429            .try_into()
430            .map_err(|_| anyhow::anyhow!("sui_tx_digest must be 32 bytes"))?,
431    );
432
433    Ok(DepositRequest {
434        id,
435        sender: requester_address,
436        created_timestamp_ms: request.timestamp_ms,
437        sui_tx_digest,
438        utxo: Utxo {
439            id: UtxoId {
440                txid,
441                vout: request.vout,
442            },
443            amount: request.amount,
444            derivation_path,
445        },
446        // Approval state isn't carried by the proto. Validators look up
447        // the on-chain version separately when verifying.
448        approval_cert: None,
449        approved_timestamp_ms: None,
450        confirmed_timestamp_ms: None,
451    })
452}

461fn parse_withdrawal_tx_commitment(
462    request: &SignWithdrawalTxConstructionRequest,
463) -> anyhow::Result<WithdrawalTxCommitment> {
464    let request_ids: Vec<Address> = request
465        .request_ids
466        .iter()
467        .map(|bytes| parse_address(bytes))
468        .collect::<anyhow::Result<_>>()?;
469    let selected_utxos: Vec<UtxoId> = request
470        .selected_utxos
471        .iter()
472        .map(|utxo_id| {
473            let txid: BitcoinTxid = utxo_id
474                .txid
475                .as_ref()
476                .map(|bytes| parse_address(bytes))
477                .context("missing utxo txid")??
478                .into();
479            let vout = utxo_id.vout.context("missing utxo vout")?;
480            Ok(UtxoId { txid, vout })
481        })
482        .collect::<anyhow::Result<_>>()?;
483    let outputs = request
484        .outputs
485        .iter()
486        .map(|output| OutputUtxo {
487            amount: output.amount,
488            bitcoin_address: output.bitcoin_address.to_vec(),
489        })
490        .collect();
491    let txid = parse_address(&request.txid)?.into();
492
493    Ok(WithdrawalTxCommitment {
494        request_ids,
495        selected_utxos,
496        outputs,
497        txid,
498    })
499}

501fn parse_withdrawal_tx_signing(
502    request: &SignWithdrawalTxSigningRequest,
503) -> anyhow::Result<WithdrawalTxSigning> {
504    let withdrawal_id = parse_address(&request.withdrawal_id)?;
505    let signatures: Vec<Vec<u8>> = request
506        .signatures
507        .iter()
508        .map(|bytes| bytes.to_vec())
509        .collect();
510    let guardian_signatures: Vec<Vec<u8>> = request
511        .guardian_signatures
512        .iter()
513        .map(|bytes| bytes.to_vec())
514        .collect();
515    Ok(WithdrawalTxSigning {
516        withdrawal_id,
517        signatures,
518        guardian_signatures,
519    })
520}

525    let withdrawal_id = parse_address(&request.withdrawal_id)?;
