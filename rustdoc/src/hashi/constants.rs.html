51/// The `hashi::versioning` package versions whose on-chain semantics this
52/// binary implements.
53///
54/// The node computes an *active* version as
55/// `max(enabled_versions ∩ published_versions ∩ SUPPORTED_PACKAGE_VERSIONS)`
56/// and gates autonomous work on it (see `onchain::version`). If every
57/// enabled+published on-chain version is beyond this set — the chain has moved
58/// ahead of this binary — the node halts autonomous mutations and signals for
59/// an upgrade rather than acting on data it cannot interpret.
60///
61/// Add the next version here in the same change that teaches this binary to
62/// run it (e.g. a package upgrade); never list a version whose semantics this
63/// build does not implement.
64///
65/// Version-dependent calls (e.g. `start_reconfig`) execute the active
66/// version's code through `SuiTxExecutor::active_call_package_id` instead of
67/// the original publish id, and shared inputs are pre-resolved so
68/// upgrade-introduced modules never trip the fullnode's simulate-time linkage
69/// check. The testnet squash restarted the sequence: the whole feature set
70/// (stamped nonce certs, deferred archival, TOB pruning) ships as v1.
71pub const SUPPORTED_PACKAGE_VERSIONS: &[u64] = &[1];

77/// Refuse a Sui chain / Bitcoin chain pairing the protocol never deploys:
78/// Bitcoin mainnet if and only if Sui mainnet. On Sui mainnet anything but
79/// Bitcoin mainnet would mint a real `hashi::btc` against free signet or
80/// regtest deposits; on any other Sui network Bitcoin mainnet would lock real
81/// Bitcoin behind a non-mainnet package (SEC-510, audit D-13).
82///
83/// Move cannot enforce this (`finish_publish` stores whichever id it is
84/// handed and the Sui framework exposes no chain identifier to Move), so it
85/// runs in every Rust path that produces or consumes the pair. The Bitcoin
86/// side is decided by the network the hash resolves to, as every consumer of
87/// the id does, not by string equality, so letter case cannot dodge it. An
88/// empty or unrecognised id fails closed.
89pub fn check_sui_bitcoin_chain_pairing(
90    sui_chain_id: &str,
91    bitcoin_chain_id: &str,
92) -> anyhow::Result<()> {
93    use crate::btc_monitor::config::Network;
94    use crate::btc_monitor::config::network_from_chain_id;
95
96    anyhow::ensure!(
97        !sui_chain_id.is_empty(),
98        "Sui chain ID is empty; refusing to pair it with Bitcoin chain {bitcoin_chain_id}"
99    );
100    let network = network_from_chain_id(bitcoin_chain_id)
101        .ok_or_else(|| anyhow::anyhow!("unrecognized bitcoin chain id: {bitcoin_chain_id}"))?;
102    let sui_is_mainnet = sui_chain_id == SUI_MAINNET_CHAIN_ID;
103    let bitcoin_is_mainnet = network == Network::Bitcoin;
104    match (sui_is_mainnet, bitcoin_is_mainnet) {
105        (true, true) | (false, false) => Ok(()),
106        (true, false) => anyhow::bail!(
107            "refusing Bitcoin {network:?} ({bitcoin_chain_id}) on Sui mainnet: \
108             Sui mainnet requires Bitcoin mainnet"
109        ),
110        (false, true) => anyhow::bail!(
111            "refusing Bitcoin mainnet ({bitcoin_chain_id}) on Sui chain {sui_chain_id}: \
112             Bitcoin mainnet is allowed on Sui mainnet only; every other Sui network \
113             needs a non-mainnet bitcoin_chain_id, which defaults to mainnet when unset"
114        ),
115    }
116}

118/// Refuse Hashi ids that do not belong to the Sui chain: the pinned mainnet ids
119/// if and only if Sui mainnet. On Sui mainnet, any other id would point the
120/// node's keys and signatures at a package that is not the deployed Hashi. On
121/// any other Sui network, the mainnet ids name no object, and they are what an
122/// unset `hashi-ids` defaults to, so this names the cause where the fullnode
123/// would only report a missing object.
124///
125/// This applies to a node joining a deployment. `hashi publish` and
126/// `hashi launch` create the ids, so they check the chain pairing alone, and
127/// nodes will refuse a new mainnet publish until these constants name it.
128pub fn check_hashi_ids_for_sui_chain(sui_chain_id: &str, ids: HashiIds) -> anyhow::Result<()> {
129    if sui_chain_id == SUI_MAINNET_CHAIN_ID {
130        anyhow::ensure!(
131            ids.package_id == HASHI_MAINNET_PACKAGE_ID,
132            "refusing package-id {} on Sui mainnet: the mainnet package-id is \
133             {HASHI_MAINNET_PACKAGE_ID}; correct it or remove hashi-ids to use the default",
134            ids.package_id
135        );
136        anyhow::ensure!(
137            ids.hashi_object_id == HASHI_MAINNET_OBJECT_ID,
138            "refusing hashi-object-id {} on Sui mainnet: the mainnet hashi-object-id is \
139             {HASHI_MAINNET_OBJECT_ID}; correct it or remove hashi-ids to use the default",
140            ids.hashi_object_id
141        );
142    } else {
143        anyhow::ensure!(
144            ids.package_id != HASHI_MAINNET_PACKAGE_ID
145                && ids.hashi_object_id != HASHI_MAINNET_OBJECT_ID,
146            "refusing the mainnet hashi-ids (package-id {}, hashi-object-id {}) on Sui chain \
147             {sui_chain_id}: they are allowed on Sui mainnet only; every other Sui network \
148             must set its own hashi-ids, which default to mainnet when unset",
149            ids.package_id,
150            ids.hashi_object_id
151        );
152    }
153    Ok(())
154}

172        );

175    #[test]
176    fn hbtc_mainnet_type_is_btc_in_the_mainnet_package() {
177        let tag: StructTag = HBTC_MAINNET_TYPE.parse().unwrap();
178        assert_eq!(*tag.address(), HASHI_MAINNET_PACKAGE_ID);
179        assert_eq!(tag.module().as_str(), "btc");
180        assert_eq!(tag.name().as_str(), "BTC");
181        assert!(tag.type_params().is_empty());
182    }
