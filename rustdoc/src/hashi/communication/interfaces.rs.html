20/// Result type for channel operations
21pub type ChannelResult<T> = Result<T, ChannelError>;
22
23/// Error type for channel operations
24#[derive(Debug, Error)]
25pub enum ChannelError {
26    #[error("Request failed: {0}")]
27    RequestFailed(String),
28
29    /// The peer explicitly answered that it is up but not ready to serve
30    /// yet (e.g. still reconciling its signing manager right after an epoch
31    /// change). Distinguished from [`ChannelError::RequestFailed`] so
32    /// callers can treat it as "retry shortly" rather than as peer failure.
33    /// Deliberately narrow: transport failures that gRPC also reports as
34    /// `Unavailable` do NOT map here.
35    #[error("Peer not ready: {0}")]
36    NotReady(String),
37
38    #[error("Client not found for address {0}")]
39    ClientNotFound(Address),
40
41    #[error("Receive timeout")]
42    Timeout,
43
44    #[error("Wait superseded: {0}")]
45    Superseded(String),
46
47    #[error("Channel closed")]
48    Closed,
49
50    #[error("Replay stream exhausted")]
51    Exhausted,
52
53    #[error("Channel error: {0}")]
54    Other(String),
55}

57/// Point-to-point channel for direct validator-to-validator messaging
58#[async_trait]
59pub trait P2PChannel: Send + Sync {
60    async fn send_messages(
61        &self,
62        recipient: &Address,
63        request: &SendMessagesRequest,
64    ) -> ChannelResult<SendMessagesResponse>;
65
66    async fn retrieve_messages(
67        &self,
68        party: &Address,
69        request: &RetrieveMessagesRequest,
70    ) -> ChannelResult<RetrieveMessagesResponse>;
71
72    async fn complain(
73        &self,
74        party: &Address,
75        request: &ComplainRequest,
76    ) -> ChannelResult<ComplaintResponse>;
77
78    async fn get_public_mpc_output(
79        &self,
80        party: &Address,
81        request: &GetPublicMpcOutputRequest,
82    ) -> ChannelResult<GetPublicMpcOutputResponse>;
83
84    async fn get_partial_signatures(
85        &self,
86        party: &Address,
87        request: &GetPartialSignaturesRequest,
88    ) -> ChannelResult<GetPartialSignaturesResponse>;
89}

91#[must_use]
92#[derive(Debug, Clone, Copy, PartialEq, Eq)]
93pub enum PublishOutcome {
94    /// This call's transaction created the entry.
95    Landed,
96    /// A message with the same hash was already present when this call checked. An earlier
97    /// attempt of this same call may have been the one that submitted it.
98    AlreadyPresent,
99    /// This call created nothing and a re-read did not find our message in the slot: either a
100    /// different one from this sender, or nothing at all.
101    Diverged,
102}

104/// Ordered broadcast channel for consensus-critical messages
105///
106/// This is a generic interface that provides total ordering guarantees:
107/// all validators see messages in the same order.
108#[async_trait]
109pub trait OrderedBroadcastChannel<M>: Send + Sync
110where
111    M: Clone + Send + Sync + 'static,
112{
113    /// Broadcast a message with guaranteed ordering across all validators
114    async fn publish(&self, message: M) -> ChannelResult<PublishOutcome>;
115
116    /// Receive the next message in the total order
117    async fn receive(&mut self) -> ChannelResult<M>;
118
119    /// Fetch existing certificates, paired with their dealer. Callers must
120    /// verify them before acting on their weight.
121    async fn certified_dealers(&mut self) -> Vec<(Address, M)>;
122}
