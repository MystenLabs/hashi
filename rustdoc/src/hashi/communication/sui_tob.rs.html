28/// Pace of the receive loop's mirror polls. The read itself is a local
29/// lookup; the interval just paces the wait for new submissions to
30/// stream into the mirror.
31const POLL_INTERVAL: Duration = Duration::from_millis(500);
32const TX_CONFIRMATION_TIMEOUT: Duration = Duration::from_secs(30);
33/// Bound for the post-race settled read: how long publish waits for the
34/// mirror to show the certificate that beat ours before classifying the
35/// race from whatever it holds.
36const SETTLED_READ_TIMEOUT: Duration = Duration::from_secs(10);
37// Publish runs inside `with_timeout_and_retry`, whose `CALL_TIMEOUT`
38// covers the submit and the settled read together, so the bound must
39// leave the deadline arm reachable at all. It is not a guarantee that
40// the arm fires: a submit that runs close to `TX_CONFIRMATION_TIMEOUT`
41// still eats the budget, and the outer timeout then cancels the read
42// and the retry resubmits a paid on-chain no-op.
43const _: () = assert!(
44    SETTLED_READ_TIMEOUT.as_millis() < super::timeout_and_retry::CALL_TIMEOUT.as_millis(),
45    "a settled-read bound at or above CALL_TIMEOUT can never fire: the outer \
46     with_timeout_and_retry deadline cancels publish first",
47);

49#[derive(Debug, Error)]
50pub enum TobError {
51    #[error("Invalid certificate data: {0}")]
52    InvalidCertificate(String),
53
54    #[error("Invalid state: {0}")]
55    InvalidState(String),
56
57    #[error("incomplete cert table read: {0}")]
58    IncompleteRead(String),
59
60    #[error("unordered cert table read: {0}")]
61    UnorderedRead(String),
62}
63
64impl From<TobError> for ChannelError {
65    fn from(e: TobError) -> Self {
66        ChannelError::Other(e.to_string())
67    }

124fn tob_read_error(e: anyhow::Error) -> TobError {
125    if crate::onchain::is_inconsistent_listing(&e) {
126        TobError::IncompleteRead(e.to_string())
127    } else if e
128        .downcast_ref::<crate::onchain::UnorderedCertTableRead>()
129        .is_some()
130    {
131        TobError::UnorderedRead(e.to_string())
132    } else {
133        TobError::InvalidState(e.to_string())
134    }
135}

137/// Read one TOB bucket's certificates from the object mirror, in TOB
138/// order. Returns an empty vector when the bucket does not exist (yet):
139/// a lagging mirror shows up as certificates arriving slightly later,
140/// and every caller sits in a retry or poll loop, so freshness needs no
141/// extra gating here.
142pub fn tob_certificates(
143    onchain_state: &OnchainState,
144    epoch: u64,
145    batch_index: Option<u32>,
146    protocol_type: ProtocolType,
147) -> Result<Vec<(Address, CertificateV1)>, TobError> {
148    Ok(
149        tob_certificates_if_present(onchain_state, epoch, batch_index, protocol_type)?
150            .unwrap_or_default(),
151    )
152}
153
154fn tob_certificates_if_present(
155    onchain_state: &OnchainState,
156    epoch: u64,
157    batch_index: Option<u32>,
158    protocol_type: ProtocolType,
159) -> Result<Option<Vec<(Address, CertificateV1)>>, TobError> {
160    let Some(raw) = onchain_state
161        .tob_certs(epoch, batch_index, protocol_type)
162        .map_err(tob_read_error)?
163    else {
164        return Ok(None);
165    };
166    let mut certificates = Vec::with_capacity(raw.len());
167    for (dealer, submission) in raw {
168        let inner_cert = match DealerMessagesHash::from_onchain_cert(&submission, epoch) {
169            Ok(inner_cert) => inner_cert,
170            Err(e) => {
171                tracing::warn!(
172                    "Skipping malformed {protocol_type:?} dealer cert from {dealer} for epoch \
173                     {epoch} batch {batch_index:?}: {e}"
174                );
175                continue;
176            }
177        };
178        // Only a nonce certificate keeps the timestamp; DKG and rotation
179        // certificates drop it.
180        let cert = CertificateV1::new(
181            protocol_type,
182            batch_index,
183            inner_cert,
184            submission.timestamp_ms,
185        );
186        certificates.push((dealer, cert));
187    }
188    Ok(Some(certificates))
189}

221pub fn key_generation_certificates(
222    onchain_state: &OnchainState,
223    epoch: u64,
224) -> Result<Vec<(Address, CertificateV1)>, TobError> {
225    let earliest_committee_epoch = onchain_state.earliest_committee_epoch();
226    let protocol_type = key_generation_protocol(earliest_committee_epoch, epoch);
227    let certificates = tob_certificates_if_present(onchain_state, epoch, None, protocol_type)?
228        .ok_or_else(|| {
229            TobError::InvalidState(format!(
230                "epoch {epoch}: {protocol_type:?} certificate bucket not found — either absent or \
231                 the tob listing was incomplete (earliest committee epoch \
232                 {earliest_committee_epoch:?})"
233            ))
234        })?;
235    if certificates.is_empty() {
236        return Err(TobError::InvalidState(format!(
237            "epoch {epoch}: {protocol_type:?} certificate bucket exists but is empty \
238             (earliest committee epoch {earliest_committee_epoch:?})"
239        )));
240    }
241    Ok(certificates)
242}
243
244fn key_generation_protocol(earliest_committee_epoch: Option<u64>, epoch: u64) -> ProtocolType {
245    if OnchainState::epoch_after_first_committee(earliest_committee_epoch, epoch) {
246        ProtocolType::KeyRotation
247    } else {
248        ProtocolType::Dkg
249    }
250}

252#[async_trait]
253impl OrderedBroadcastChannel<CertificateV1> for SuiTobSessionChannel {
254    async fn publish(&self, cert: CertificateV1) -> ChannelResult<PublishOutcome> {
255        let ours = cert.message();
256        // The dedup read is a local mirror lookup; a lagging mirror at
257        // worst resubmits, which the on-chain first-submission-wins rule
258        // turns into a paid no-op.
259        let existing = match tob_certificates(
260            &self.onchain_state,
261            self.epoch,
262            self.batch_index,
263            self.protocol_type,
264        ) {
265            Ok(existing) => existing,
266            Err(e @ (TobError::IncompleteRead(_) | TobError::UnorderedRead(_))) => {
267                tracing::warn!(
268                    "{:?} dedup read for epoch {} batch {:?} unusable ({e}); submitting anyway",
269                    self.protocol_type,
270                    self.epoch,
271                    self.batch_index,
272                );
273                Vec::new()
274            }
275            Err(e) => return Err(ChannelError::from(e)),
276        };
277        let published: Vec<(Address, MessagesHash)> = existing
278            .iter()
279            .map(|(d, c)| (*d, c.message().messages_hash))
280            .collect();
281        let classified =
282            classify_published_cert(&published, &ours.dealer_address, ours.messages_hash);
283        if let PublishedCert::Diverged { on_chain } = &classified {
284            tracing::warn!(
285                "{:?} epoch {} batch {:?}: dealer {} already has a certificate over \
286                 different messages (on chain {}, ours {}); regenerated messages or an \
287                 AVID optimistic/pessimistic flip",
288                self.protocol_type,
289                self.epoch,
290                self.batch_index,
291                ours.dealer_address,
292                hex::encode(<MessagesHash as AsRef<[u8; 32]>>::as_ref(on_chain)),
293                hex::encode(<MessagesHash as AsRef<[u8; 32]>>::as_ref(
294                    &ours.messages_hash
295                )),
296            );
297        }
298        if let Some(outcome) = short_circuit_outcome(&classified) {
299            return Ok(outcome);
300        }
301        let mut executor = self.create_executor();
302        let inserted = executor
303            .execute_submit_certificate(&cert)
304            .await
305            // Only `Rejected` is the chain's answer; the rest are failures reaching it or reading
306            // the result back, and must not share a bucket with a conclusive rejection.
307            .map_err(|e| match &e {
308                SubmitCertError::Rejected(_) => ChannelError::Other(e.to_string()),
309                SubmitCertError::NotSubmitted(_)
310                | SubmitCertError::SubmitFailed(_)
311                | SubmitCertError::Unconfirmed(_) => ChannelError::RequestFailed(e.to_string()),
312            })?;
313        if inserted {
314            return Ok(PublishOutcome::Landed);
315        }
316        // Nothing inserted means a certificate for this dealer already
317        // holds the slot on-chain, but the mirror may not have applied
318        // the transaction that beat ours yet — poll briefly for it
319        // rather than misreading the lag as divergence.
320        let settled_deadline = tokio::time::Instant::now() + SETTLED_READ_TIMEOUT;
321        let classified = loop {
322            match tob_certificates(
323                &self.onchain_state,
324                self.epoch,
325                self.batch_index,
326                self.protocol_type,
327            ) {
328                Ok(settled) => {
329                    let settled: Vec<(Address, MessagesHash)> = settled
330                        .iter()
331                        .map(|(d, c)| (*d, c.message().messages_hash))
332                        .collect();
333                    let classified =
334                        classify_published_cert(&settled, &ours.dealer_address, ours.messages_hash);
335                    if !matches!(classified, PublishedCert::Absent)
336                        || tokio::time::Instant::now() >= settled_deadline
337                    {
338                        break classified;
339                    }
340                }
341                Err(e @ (TobError::IncompleteRead(_) | TobError::UnorderedRead(_)))
342                    if tokio::time::Instant::now() < settled_deadline =>
343                {
344                    tracing::debug!(
345                        "{:?} settled read for epoch {} batch {:?} unusable ({e}); retrying",
346                        self.protocol_type,
347                        self.epoch,
348                        self.batch_index,
349                    );
350                }
351                Err(e) => return Err(ChannelError::from(e)),
352            }
353            tokio::time::sleep(POLL_INTERVAL).await;
354        };
355        let outcome = raced_outcome(&classified);
356        tracing::warn!(
357            "{:?} epoch {} batch {:?}: this submission for dealer {} inserted nothing; a \
358             certificate already held the slot (ours {}), outcome {:?}",
359            self.protocol_type,
360            self.epoch,
361            self.batch_index,
362            ours.dealer_address,
363            hex::encode(<MessagesHash as AsRef<[u8; 32]>>::as_ref(
364                &ours.messages_hash
365            )),
366            outcome,
367        );
368        Ok(outcome)
369    }
370
371    async fn receive(&mut self) -> ChannelResult<CertificateV1> {
372        let wait_started = *self
373            .wait_started
374            .get_or_insert_with(tokio::time::Instant::now);
375        loop {
376            if let Some(cert) = self.pending_certs.pop_front() {
377                self.wait_started = None;
378                return Ok(cert);
379            }
380            let all_certs = match tob_certificates(
381                &self.onchain_state,
382                self.epoch,
383                self.batch_index,
384                self.protocol_type,
385            ) {
386                Ok(certs) => certs,
387                Err(TobError::IncompleteRead(msg)) => {
388                    tracing::debug!(
389                        "{:?} TOB cert read for epoch {} raced the mirror's replay ({msg}); \
390                         retrying",
391                        self.protocol_type,
392                        self.epoch,
393                    );
394                    Vec::new()
395                }
396                Err(e) => {
397                    self.wait_started = None;
398                    return Err(ChannelError::from(e));
399                }
400            };
401            for (dealer, cert) in all_certs {
402                if !self.seen_dealers.contains(&dealer) {
403                    self.seen_dealers.insert(dealer);
404                    self.pending_certs.push_back(cert);
405                }
406            }
407            if self.pending_certs.is_empty() {
408                let (onchain_epoch, pending) = {
409                    let state = self.onchain_state.state();
410                    let committees = &state.hashi().committees;
411                    (committees.epoch(), committees.pending_epoch_change())
412                };
413                if tob_wait_superseded(self.protocol_type, self.epoch, onchain_epoch, pending) {
414                    tracing::info!(
415                        "aborting {:?} TOB wait for epoch {}: superseded (onchain epoch \
416                         {onchain_epoch}, pending epoch change {pending:?})",
417                        self.protocol_type,
418                        self.epoch,
419                    );
420                    self.wait_started = None;
421                    return Err(ChannelError::Superseded(format!(
422                        "{:?} TOB wait for epoch {} (onchain epoch {onchain_epoch}, \
423                         pending epoch change {pending:?})",
424                        self.protocol_type, self.epoch,
425                    )));
426                }
427                if let Some(idle_timeout) = self.idle_timeout
428                    && wait_started.elapsed() >= idle_timeout
429                {
430                    tracing::info!(
431                        "aborting {:?} TOB wait for epoch {}: no certificate in {:?} \
432                         ({} dealers seen)",
433                        self.protocol_type,
434                        self.epoch,
435                        idle_timeout,
436                        self.seen_dealers.len(),
437                    );
438                    self.wait_started = None;
439                    return Err(ChannelError::Timeout);
440                }
441                tokio::time::sleep(POLL_INTERVAL).await;
442            }
443        }
444    }
445
446    async fn certified_dealers(&mut self) -> Vec<(Address, CertificateV1)> {
447        match tob_certificates(
448            &self.onchain_state,
449            self.epoch,
450            self.batch_index,
451            self.protocol_type,
452        ) {
453            Ok(all_certs) => {
454                for (dealer, cert) in &all_certs {
455                    if !self.seen_dealers.contains(dealer) {
456                        self.seen_dealers.insert(*dealer);
457                        self.pending_certs.push_back(cert.clone());
458                    }
459                }
460                all_certs
461            }
462            Err(e) => {
463                tracing::warn!(
464                    "{:?} certified_dealers read for epoch {} batch {:?} failed: {e}; \
465                     reporting none certified",
466                    self.protocol_type,
467                    self.epoch,
468                    self.batch_index,
469                );
470                vec![]
471            }
472        }
473    }

476#[derive(Debug, PartialEq, Eq)]
477enum PublishedCert {
478    Absent,
479    Same,
480    Diverged { on_chain: MessagesHash },
481}
482
483fn raced_outcome(classified: &PublishedCert) -> PublishOutcome {
484    match classified {
485        PublishedCert::Same => PublishOutcome::AlreadyPresent,
486        PublishedCert::Diverged { .. } | PublishedCert::Absent => PublishOutcome::Diverged,
487    }
488}
489
490fn short_circuit_outcome(classified: &PublishedCert) -> Option<PublishOutcome> {
491    match classified {
492        PublishedCert::Same => Some(PublishOutcome::AlreadyPresent),
493        PublishedCert::Diverged { .. } => Some(PublishOutcome::Diverged),
494        PublishedCert::Absent => None,
495    }
496}
497
498fn classify_published_cert(
499    existing: &[(Address, MessagesHash)],
500    dealer: &Address,
501    ours: MessagesHash,
502) -> PublishedCert {
503    let Some((_, on_chain)) = existing.iter().find(|(d, _)| d == dealer) else {
504        return PublishedCert::Absent;
505    };
506    if *on_chain == ours {
507        PublishedCert::Same
508    } else {
509        PublishedCert::Diverged {
510            on_chain: *on_chain,
511        }
512    }
513}

515pub(crate) fn tob_wait_superseded(
516    protocol_type: ProtocolType,
517    channel_epoch: u64,
518    onchain_epoch: u64,
519    pending_epoch_change: Option<u64>,
520) -> bool {
521    match protocol_type {
522        ProtocolType::NonceGeneration => {
523            matches!(pending_epoch_change, Some(p) if p != channel_epoch)
524                || onchain_epoch > channel_epoch
525        }
526        ProtocolType::Dkg | ProtocolType::KeyRotation => {
527            matches!(pending_epoch_change, Some(p) if p != channel_epoch)
528                || (pending_epoch_change.is_none() && onchain_epoch != channel_epoch)
529        }
530    }
531}

692    #[tokio::test]
693    async fn tokio_joinhandle_survives_cancelled_awaits() {
694        let polls = std::sync::Arc::new(std::sync::atomic::AtomicUsize::new(0));
695        let started = std::sync::Arc::clone(&polls);
696        let mut fetch: tokio::task::JoinHandle<usize> = tokio::spawn(async move {
697            started.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
698            tokio::time::sleep(Duration::from_millis(300)).await;
699            42
700        });
701
702        for _ in 0..3 {
703            assert!(
704                tokio::time::timeout(Duration::from_millis(20), &mut fetch)
705                    .await
706                    .is_err(),
707                "expected the short poll to elapse"
708            );
709        }
710
711        let out = tokio::time::timeout(Duration::from_millis(500), &mut fetch)
712            .await
713            .expect("fetch must still be progressing, not restarted")
714            .expect("task must not have been cancelled by the dropped awaits");
715        assert_eq!(out, 42);
716        assert_eq!(
717            polls.load(std::sync::atomic::Ordering::SeqCst),
718            1,
719            "the fetch must have run once, not restarted per poll"
720        );
721    }
