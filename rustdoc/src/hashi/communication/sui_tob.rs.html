32#[derive(Debug, Error)]
33pub enum TobError {
34    #[error("Sui RPC error: {0}")]
35    RpcError(String),
36
37    #[error("Invalid certificate data: {0}")]
38    InvalidCertificate(String),
39
40    #[error("Invalid state: {0}")]
41    InvalidState(String),
42}
43
44impl From<TobError> for ChannelError {
45    fn from(e: TobError) -> Self {
46        match e {
47            TobError::RpcError(msg) => ChannelError::RequestFailed(msg),
48            _ => ChannelError::Other(e.to_string()),
49        }
50    }
51}

104pub async fn fetch_certificates(
105    onchain_state: &OnchainState,
106    epoch: u64,
107    batch_index: Option<u32>,
108    protocol_type: ProtocolType,
109) -> Result<Vec<(Address, CertificateV1)>, TobError> {
110    Ok(
111        fetch_certificates_if_present(onchain_state, epoch, batch_index, protocol_type)
112            .await?
113            .unwrap_or_default(),
114    )
115}
116
117async fn fetch_certificates_if_present(
118    onchain_state: &OnchainState,
119    epoch: u64,
120    batch_index: Option<u32>,
121    protocol_type: ProtocolType,
122) -> Result<Option<Vec<(Address, CertificateV1)>>, TobError> {
123    let Some(raw_certs) = onchain_state
124        .fetch_certs(epoch, batch_index, protocol_type)
125        .await
126        .map_err(|e| TobError::RpcError(format!("{e:#}")))?
127    else {
128        return Ok(None);
129    };
130    let mut certificates = Vec::with_capacity(raw_certs.len());
131    for (dealer, cert) in raw_certs {
132        let inner_cert = DealerMessagesHash::from_onchain_cert(&cert, epoch)
133            .map_err(|e| TobError::InvalidCertificate(e.to_string()))?;
134        let cert = CertificateV1::new(protocol_type, batch_index, inner_cert);
135        certificates.push((dealer, cert));
136    }
137    Ok(Some(certificates))
138}

148    }

151#[async_trait]
152impl OrderedBroadcastChannel<CertificateV1> for PrefetchedTobChannel {
153    async fn publish(&self, _cert: CertificateV1) -> ChannelResult<PublishOutcome> {
154        Err(ChannelError::Other(
155            "replayed certificate stream is receive-only".into(),
156        ))
157    }
158
159    async fn receive(&mut self) -> ChannelResult<CertificateV1> {
160        let (_, cert) = self
161            .certs
162            .get(self.next)
163            .ok_or_else(|| ChannelError::Other("replayed certificate stream exhausted".into()))?;
164        self.next += 1;
165        Ok(cert.clone())
166    }
167
168    async fn certified_dealers(&mut self) -> Vec<(Address, CertificateV1)> {
169        self.certs.clone()
170    }
171}

173pub async fn fetch_key_generation_certificates(
174    onchain_state: &OnchainState,
175    epoch: u64,
176) -> Result<Vec<(Address, CertificateV1)>, TobError> {
177    let earliest_committee_epoch = onchain_state.earliest_committee_epoch();
178    let protocol_type = key_generation_protocol(earliest_committee_epoch, epoch);
179    let certificates = fetch_certificates_if_present(onchain_state, epoch, None, protocol_type)
180        .await?
181        .ok_or_else(|| {
182            TobError::InvalidState(format!(
183                "epoch {epoch}: {protocol_type:?} certificate bucket not found — either absent or \
184                 the tob listing was incomplete (earliest committee epoch \
185                 {earliest_committee_epoch:?})"
186            ))
187        })?;
188    if certificates.is_empty() {
189        return Err(TobError::InvalidState(format!(
190            "epoch {epoch}: {protocol_type:?} certificate bucket exists but is empty \
191             (earliest committee epoch {earliest_committee_epoch:?})"
192        )));
193    }
194    Ok(certificates)
195}
196
197fn key_generation_protocol(earliest_committee_epoch: Option<u64>, epoch: u64) -> ProtocolType {
198    if OnchainState::epoch_after_first_committee(earliest_committee_epoch, epoch) {
199        ProtocolType::KeyRotation
200    } else {
201        ProtocolType::Dkg
202    }
203}

205#[async_trait]
206impl OrderedBroadcastChannel<CertificateV1> for SuiTobSessionChannel {
207    async fn publish(&self, cert: CertificateV1) -> ChannelResult<PublishOutcome> {
208        let ours = cert.message();
209        let existing = fetch_certificates(
210            &self.onchain_state,
211            self.epoch,
212            self.batch_index,
213            self.protocol_type,
214        )
215        .await
216        .map_err(ChannelError::from)?;
217        let published: Vec<(Address, MessagesHash)> = existing
218            .iter()
219            .map(|(d, c)| (*d, c.message().messages_hash))
220            .collect();
221        let classified =
222            classify_published_cert(&published, &ours.dealer_address, ours.messages_hash);
223        if let PublishedCert::Diverged { on_chain } = &classified {
224            tracing::warn!(
225                "{:?} epoch {} batch {:?}: dealer {} already has a certificate over \
226                 different messages (on chain {}, ours {}); regenerated messages or an \
227                 AVID optimistic/pessimistic flip",
228                self.protocol_type,
229                self.epoch,
230                self.batch_index,
231                ours.dealer_address,
232                hex::encode(<MessagesHash as AsRef<[u8; 32]>>::as_ref(on_chain)),
233                hex::encode(<MessagesHash as AsRef<[u8; 32]>>::as_ref(
234                    &ours.messages_hash
235                )),
236            );
237        }
238        if let Some(outcome) = short_circuit_outcome(&classified) {
239            return Ok(outcome);
240        }
241        let mut executor = self.create_executor();
242        let inserted = executor
243            .execute_submit_certificate(&cert)
244            .await
245            // Only `Rejected` is the chain's answer; the rest are failures reaching it or reading
246            // the result back, and must not share a bucket with a conclusive rejection.
247            .map_err(|e| match &e {
248                SubmitCertError::Rejected(_) => ChannelError::Other(e.to_string()),
249                SubmitCertError::NotSubmitted(_)
250                | SubmitCertError::SubmitFailed(_)
251                | SubmitCertError::Unconfirmed(_) => ChannelError::RequestFailed(e.to_string()),
252            })?;
253        if inserted {
254            return Ok(PublishOutcome::Landed);
255        }
256        let settled = fetch_certificates(
257            &self.onchain_state,
258            self.epoch,
259            self.batch_index,
260            self.protocol_type,
261        )
262        .await
263        .map_err(ChannelError::from)?;
264        let settled: Vec<(Address, MessagesHash)> = settled
265            .iter()
266            .map(|(d, c)| (*d, c.message().messages_hash))
267            .collect();
268        let outcome = raced_outcome(&classify_published_cert(
269            &settled,
270            &ours.dealer_address,
271            ours.messages_hash,
272        ));
273        tracing::warn!(
274            "{:?} epoch {} batch {:?}: this submission for dealer {} inserted nothing; a \
275             certificate already held the slot (ours {}), outcome {:?}",
276            self.protocol_type,
277            self.epoch,
278            self.batch_index,
279            ours.dealer_address,
280            hex::encode(<MessagesHash as AsRef<[u8; 32]>>::as_ref(
281                &ours.messages_hash
282            )),
283            outcome,
284        );
285        Ok(outcome)
286    }
287
288    async fn receive(&mut self) -> ChannelResult<CertificateV1> {
289        let wait_started = tokio::time::Instant::now();
290        loop {
291            if let Some(cert) = self.pending_certs.pop_front() {
292                return Ok(cert);
293            }
294            // TODO: Optimize by checking table size first to avoid redundant fetches.
295            let all_certs = match tokio::time::timeout(
296                FETCH_STALL_TIMEOUT,
297                fetch_certificates(
298                    &self.onchain_state,
299                    self.epoch,
300                    self.batch_index,
301                    self.protocol_type,
302                ),
303            )
304            .await
305            {
306                Ok(result) => result.map_err(ChannelError::from)?,
307                Err(_) => {
308                    tracing::warn!(
309                        "{:?} TOB cert fetch for epoch {} stalled >{:?}; retrying",
310                        self.protocol_type,
311                        self.epoch,
312                        FETCH_STALL_TIMEOUT,
313                    );
314                    Vec::new()
315                }
316            };
317            for (dealer, cert) in all_certs {
318                if !self.seen_dealers.contains(&dealer) {
319                    self.seen_dealers.insert(dealer);
320                    self.pending_certs.push_back(cert);
321                }
322            }
323            if self.pending_certs.is_empty() {
324                let (onchain_epoch, pending) = {
325                    let state = self.onchain_state.state();
326                    let committees = &state.hashi().committees;
327                    (committees.epoch(), committees.pending_epoch_change())
328                };
329                if tob_wait_superseded(self.protocol_type, self.epoch, onchain_epoch, pending) {
330                    tracing::info!(
331                        "aborting {:?} TOB wait for epoch {}: superseded (onchain epoch \
332                         {onchain_epoch}, pending epoch change {pending:?})",
333                        self.protocol_type,
334                        self.epoch,
335                    );
336                    return Err(ChannelError::Superseded(format!(
337                        "{:?} TOB wait for epoch {} (onchain epoch {onchain_epoch}, \
338                         pending epoch change {pending:?})",
339                        self.protocol_type, self.epoch,
340                    )));
341                }
342                if let Some(idle_timeout) = self.idle_timeout
343                    && wait_started.elapsed() >= idle_timeout
344                {
345                    tracing::info!(
346                        "aborting {:?} TOB wait for epoch {}: no certificate in {:?} \
347                         ({} dealers seen)",
348                        self.protocol_type,
349                        self.epoch,
350                        idle_timeout,
351                        self.seen_dealers.len(),
352                    );
353                    return Err(ChannelError::Timeout);
354                }
355                tokio::time::sleep(POLL_INTERVAL).await;
356            }
357        }
358    }
359
360    async fn certified_dealers(&mut self) -> Vec<(Address, CertificateV1)> {
361        let Ok(Ok(all_certs)) = tokio::time::timeout(
362            FETCH_STALL_TIMEOUT,
363            fetch_certificates(
364                &self.onchain_state,
365                self.epoch,
366                self.batch_index,
367                self.protocol_type,
368            ),
369        )
370        .await
371        else {
372            return vec![];
373        };
374        for (dealer, cert) in &all_certs {
375            if !self.seen_dealers.contains(dealer) {
376                self.seen_dealers.insert(*dealer);
377                self.pending_certs.push_back(cert.clone());
378            }
379        }
380        all_certs
381    }
382}

384#[derive(Debug, PartialEq, Eq)]
385enum PublishedCert {
386    Absent,
387    Same,
388    Diverged { on_chain: MessagesHash },
389}
390
391fn raced_outcome(classified: &PublishedCert) -> PublishOutcome {
392    match classified {
393        PublishedCert::Same => PublishOutcome::AlreadyPresent,
394        PublishedCert::Diverged { .. } | PublishedCert::Absent => PublishOutcome::Diverged,
395    }
396}
397
398fn short_circuit_outcome(classified: &PublishedCert) -> Option<PublishOutcome> {
399    match classified {
400        PublishedCert::Same => Some(PublishOutcome::AlreadyPresent),
401        PublishedCert::Diverged { .. } => Some(PublishOutcome::Diverged),
402        PublishedCert::Absent => None,
403    }
404}
405
406fn classify_published_cert(
407    existing: &[(Address, MessagesHash)],
408    dealer: &Address,
409    ours: MessagesHash,
410) -> PublishedCert {
411    let Some((_, on_chain)) = existing.iter().find(|(d, _)| d == dealer) else {
412        return PublishedCert::Absent;
413    };
414    if *on_chain == ours {
415        PublishedCert::Same
416    } else {
417        PublishedCert::Diverged {
418            on_chain: *on_chain,
419        }
420    }
421}
422
423fn tob_wait_superseded(
424    protocol_type: ProtocolType,
425    channel_epoch: u64,
426    onchain_epoch: u64,
427    pending_epoch_change: Option<u64>,
428) -> bool {
429    match protocol_type {
430        ProtocolType::NonceGeneration => {
431            matches!(pending_epoch_change, Some(p) if p != channel_epoch)
432                || onchain_epoch > channel_epoch
433        }
434        ProtocolType::Dkg | ProtocolType::KeyRotation => {
435            matches!(pending_epoch_change, Some(p) if p != channel_epoch)
436                || (pending_epoch_change.is_none() && onchain_epoch != channel_epoch)
437        }
438    }
