34fn deserialize_backup_pgp_cert<'de, D>(
35    deserializer: D,
36) -> Result<hashi_types::pgp::PgpPublicCert, D::Error>
37where
38    D: serde::Deserializer<'de>,
39{
40    let value = <String as serde::Deserialize>::deserialize(deserializer)?;
41
42    let path = Path::new(&value);
43    let armored = if path.is_file() {
44        std::fs::read_to_string(path).map_err(serde::de::Error::custom)?
45    } else {
46        value
47    };
48
49    hashi_types::pgp::PgpPublicCert::new(armored).map_err(serde::de::Error::custom)
50}

420    Remote(RemoteBackupLocation<'a>),

425    provider: BackupProvider,
426    bucket: &'a str,
427    key: &'a str,
428}

430impl<'a> BackupLocation<'a> {
431    pub(crate) fn parse(source: &'a Path) -> anyhow::Result<Self> {
432        let Some((scheme, rest)) = source.to_str().and_then(|s| s.split_once("://")) else {
433            return Ok(Self::Local(source));
434        };
435        if !scheme.starts_with(|c: char| c.is_ascii_alphabetic())
436            || !scheme
437                .bytes()
438                .all(|c| c.is_ascii_alphanumeric() || matches!(c, b'+' | b'-' | b'.'))
439        {
440            return Ok(Self::Local(source));
441        }
442        let provider = match scheme {
443            "s3" => BackupProvider::S3,
444            "gs" => BackupProvider::Gcs,
445            _ => anyhow::bail!("unsupported backup URI scheme: {scheme}"),
446        };
447        let (bucket, key) = rest.split_once('/').unwrap_or((rest, ""));
448        match provider {
449            BackupProvider::S3 => S3BackupConfig::validate_bucket(bucket)?,
450            BackupProvider::Gcs => GcsBackupConfig::validate_bucket(bucket)?,
451        }
452        anyhow::ensure!(
453            !key.contains(['?', '#']) && !key.chars().any(char::is_control),
454            "backup URI key must not contain a query, fragment, or control characters"
455        );
456        Ok(Self::Remote(RemoteBackupLocation {
457            provider,
458            bucket,
459            key,
460        }))
461    }
462}

464impl<'a> RemoteBackupLocation<'a> {

473    pub(crate) fn into_config(self, region: Option<String>) -> anyhow::Result<BackupRemoteConfig> {
474        match self.provider {
475            BackupProvider::S3 => {
476                let region = region
477                    .ok_or_else(|| anyhow::anyhow!("backup-remote.region is required for S3"))?;
478                S3BackupConfig::validate_region(&region)?;
479                Ok(BackupRemoteConfig::S3(S3BackupConfig {
480                    bucket: self.bucket.to_owned(),
481                    region,
482                }))
483            }
484            BackupProvider::Gcs => {
485                anyhow::ensure!(
486                    region.is_none(),
487                    "backup-remote.region is forbidden for GCS"
488                );
489                Ok(BackupRemoteConfig::Gcs(GcsBackupConfig {
490                    bucket: self.bucket.to_owned(),
491                }))
492            }
493        }
494    }

556    pub fn new(destination: &str, region: Option<String>) -> anyhow::Result<Self> {
557        let BackupLocation::Remote(location) = BackupLocation::parse(Path::new(destination))?
558        else {

566    }

606        impl serde::Serialize for Destination<'_> {
607            fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
608                serializer.collect_str(&format_args!("{}://{}/", self.0.scheme(), self.0.bucket()))

610        }

620                let mut output = serializer.serialize_struct("BackupRemoteConfig", 1)?;

1019/// Return an ephemeral, available port. On unix systems, the port returned will be in the
1020/// TIME_WAIT state ensuring that the OS won't hand out this port for some grace period.
1021/// Callers should be able to bind to this port given they use SO_REUSEADDR.
1022pub fn get_available_port() -> u16 {
1023    const MAX_PORT_RETRIES: u32 = 1000;
1024
1025    for _ in 0..MAX_PORT_RETRIES {
1026        if let Ok(port) = get_ephemeral_port() {
1027            return port;
1028        }
1029    }
1030
1031    panic!("Error: could not find an available port on localhost");
1032}
1033
1034fn get_ephemeral_port() -> std::io::Result<u16> {
1035    use std::net::TcpListener;
1036    use std::net::TcpStream;
1037
1038    // Request a random available port from the OS
1039    let listener = TcpListener::bind(SocketAddr::from(([127, 0, 0, 1], 0)))?;
1040    let addr = listener.local_addr()?;
1041
1042    // Create and accept a connection (which we'll promptly drop) in order to force the port
1043    // into the TIME_WAIT state, ensuring that the port will be reserved from some limited
1044    // amount of time (roughly 60s on some Linux systems)
1045    let _sender = TcpStream::connect(addr)?;
1046    let _incoming = listener.accept()?;
1047
1048    Ok(addr.port())
1049}
