4//! Parse `packages/hashi/Published.toml` — the Move tooling's record of
5//! where (and at what version) the package is deployed per network.
6//!
7//! This file is the source of truth both the upgrade-compatibility CI gate
8//! (`crates/hashi/tests/move_upgrade_compat.rs`) and the e2e bytecode-snapshot
9//! tests (`crates/e2e-tests/src/snapshot.rs`) derive their snapshot locations
10//! from, so a version bump there without a matching snapshot capture fails
11//! those checks instead of silently exercising an obsolete package.
12
13use std::collections::BTreeMap;
14use std::path::Path;
15
16use anyhow::Context;
17use anyhow::Result;
18
19/// One `[published.<network>]` entry of `Published.toml`.
20#[derive(serde_derive::Deserialize)]
21#[serde(rename_all = "kebab-case")]
22pub struct PublishedEntry {
23    /// Storage id of the deployed package object (the id a snapshot capture
24    /// is fetched from).
25    pub published_at: String,
26    /// Runtime/original id the deployed modules are self-addressed at.
27    pub original_id: String,
28    /// Deployed package version (`v<version>` names the snapshot directory).
29    pub version: u64,
30}
31
32#[derive(serde_derive::Deserialize)]
33struct PublishedFile {
34    published: BTreeMap<String, PublishedEntry>,
35}
36
37/// Parse the `Published.toml` at `path` into its per-network entries.
38/// Fails on a missing/unparsable file or one declaring no environments.
39pub fn published_entries(path: &Path) -> Result<BTreeMap<String, PublishedEntry>> {
40    let text =
41        std::fs::read_to_string(path).with_context(|| format!("reading {}", path.display()))?;
42    let parsed: PublishedFile =
43        toml::from_str(&text).with_context(|| format!("parsing {}", path.display()))?;
44    anyhow::ensure!(
45        !parsed.published.is_empty(),
46        "{} declares no published environments",
47        path.display()
48    );
49    Ok(parsed.published)
50}
