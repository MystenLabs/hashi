37const MEMBER_WAIT: Duration = Duration::from_secs(10);
38const FAILURE_TTL: Duration = Duration::from_secs(5 * 60);

40#[derive(Clone, Debug, PartialEq, Eq)]
41pub enum Verdict {
42    Accepted,
43    ScriptFailure {
44        reason: String,
45        details: Option<String>,
46    },
47    RejectedOther {
48        reason: String,
49    },
50    Unavailable(String),
51}
52
53impl Verdict {
54    pub fn label(&self) -> &'static str {
55        match self {
56            Verdict::Accepted => "accepted",
57            Verdict::ScriptFailure { .. } => "script_failure",
58            Verdict::RejectedOther { .. } => "rejected_other",
59            Verdict::Unavailable(_) => "unavailable",
60        }
61    }
62
63    pub(crate) fn ensure_signable(
64        self,
65        withdrawal_id: Address,
66        txid: bitcoin::Txid,
67    ) -> anyhow::Result<()> {
68        if let Verdict::ScriptFailure { reason, details } = self {
69            anyhow::bail!(
70                "bitcoind fails withdrawal {withdrawal_id}'s signed transaction {txid}: {reason}{}",
71                details.map(|d| format!(" ({d})")).unwrap_or_default()
72            );
73        }
74        Ok(())
75    }
76}
77
78pub fn classify(results: &[MempoolAcceptance]) -> Verdict {
79    let [result] = results else {
80        return Verdict::Unavailable(format!(
81            "testmempoolaccept returned {} results, expected 1",
82            results.len()
83        ));
84    };
85    if result.allowed {
86        return Verdict::Accepted;
87    }
88    let reason = result.reject_reason.clone().unwrap_or_default();
89    if is_script_failure(&reason) {
90        Verdict::ScriptFailure {
91            reason,
92            details: result.reject_details.clone(),
93        }
94    } else {
95        Verdict::RejectedOther { reason }
96    }
97}

99fn is_script_failure(reason: &str) -> bool {
100    fails_script_check(reason)
101        || reason == "scriptsig-not-pushonly"
102        || reason == "scriptsig-size"
103        || reason == "bad-witness-nonstandard"
104}
105
106fn fails_script_check(reason: &str) -> bool {
107    reason.contains("script-verify-flag")
108}
109
110fn spend_probe_result(results: &[MempoolAcceptance]) -> Probed {
111    match classify(results) {
112        Verdict::ScriptFailure { reason, .. } if fails_script_check(&reason) => Probed::Refused,
113        Verdict::ScriptFailure { reason, .. } => Probed::Missed(format!(
114            "bitcoind rejected the probe's standard spend as {reason}, so it refuses every withdrawal"
115        )),
116        Verdict::RejectedOther { reason } if reason == "max-fee-exceeded" => {
117            Probed::Missed(format!(
118                "bitcoind passed the scripts of a spend with invalid signatures, then answered {reason}"
119            ))
120        }
121        Verdict::RejectedOther { reason } => Probed::Missed(format!(
122            "bitcoind answered {reason} before checking the scripts of a spend with invalid signatures"
123        )),
124        Verdict::Accepted => {
125            Probed::Missed("bitcoind accepted a spend with invalid signatures".into())
126        }
127        Verdict::Unavailable(error) => Probed::Unanswered(error),
128    }
129}

131pub(crate) struct FinalizeBitcoinCheck {
132    metrics: Arc<Metrics>,
133    failures: Mutex<HashMap<Wtxid, (Instant, Verdict)>>,
134    in_flight: Mutex<HashMap<Wtxid, watch::Receiver<Option<Verdict>>>>,
135}
136
137impl FinalizeBitcoinCheck {
138    pub(crate) fn new(metrics: Arc<Metrics>) -> Self {
139        Self {
140            metrics,
141            failures: Mutex::default(),
142            in_flight: Mutex::default(),
143        }
144    }
145
146    pub(crate) async fn check<F, Fut>(
147        self: &Arc<Self>,
148        withdrawal_id: Address,
149        tx: bitcoin::Transaction,
150        test_mempool_accept: F,
151    ) -> Verdict
152    where
153        F: FnOnce(bitcoin::Transaction) -> Fut + Send + 'static,
154        Fut: Future<Output = anyhow::Result<Vec<MempoolAcceptance>>> + Send + 'static,
155    {
156        let verdict = self
157            .check_uncounted(withdrawal_id, tx, test_mempool_accept)
158            .await;
159        self.record(&verdict);
160        verdict
161    }
162
163    pub(crate) fn unavailable(&self, withdrawal_id: Address, error: String) -> Verdict {
164        debug!(%withdrawal_id, %error, "could not assemble the signed withdrawal to check it");
165        let verdict = Verdict::Unavailable(error);
166        self.record(&verdict);
167        verdict
168    }
169
170    fn record(&self, verdict: &Verdict) {
171        self.metrics
172            .withdrawal_bitcoin_check_total
173            .with_label_values(&[verdict.label()])
174            .inc();
175    }
176
177    async fn check_uncounted<F, Fut>(
178        self: &Arc<Self>,
179        withdrawal_id: Address,
180        tx: bitcoin::Transaction,
181        test_mempool_accept: F,
182    ) -> Verdict
183    where
184        F: FnOnce(bitcoin::Transaction) -> Fut + Send + 'static,
185        Fut: Future<Output = anyhow::Result<Vec<MempoolAcceptance>>> + Send + 'static,
186    {
187        let wtxid = tx.compute_wtxid();
188        let mut receiver = {
189            let mut in_flight = self.in_flight.lock().unwrap();
190            if let Some(verdict) = self.cached_failure(&wtxid) {
191                return verdict;
192            }
193            match in_flight.get(&wtxid) {
194                Some(receiver) => receiver.clone(),
195                None => {
196                    let (sender, receiver) = watch::channel(None);
197                    in_flight.insert(wtxid, receiver.clone());
198                    tokio::spawn(self.clone().run(
199                        withdrawal_id,
200                        wtxid,
201                        tx,
202                        test_mempool_accept,
203                        sender,
204                    ));
205                    receiver
206                }
207            }
208        };
209        match tokio::time::timeout(MEMBER_WAIT, receiver.wait_for(Option::is_some)).await {
210            Ok(Ok(verdict)) => verdict.clone().expect("waited for a verdict"),
211            Ok(Err(_)) => Verdict::Unavailable("the bitcoind check ended without an answer".into()),
212            Err(_) => Verdict::Unavailable(format!(
213                "testmempoolaccept did not answer within {}s",
214                MEMBER_WAIT.as_secs()
215            )),
216        }
217    }
218
219    async fn run<F, Fut>(
220        self: Arc<Self>,
221        withdrawal_id: Address,
222        wtxid: Wtxid,
223        tx: bitcoin::Transaction,
224        test_mempool_accept: F,
225        sender: watch::Sender<Option<Verdict>>,
226    ) where
227        F: FnOnce(bitcoin::Transaction) -> Fut,
228        Fut: Future<Output = anyhow::Result<Vec<MempoolAcceptance>>>,
229    {
230        let _entry = InFlightEntry {
231            check: self.clone(),
232            wtxid,
233        };
234        let txid = tx.compute_txid();
235        let started = Instant::now();
236        let answer = test_mempool_accept(tx).await;
237        self.metrics
238            .withdrawal_bitcoin_check_latency_seconds
239            .observe(started.elapsed().as_secs_f64());
240        let verdict = match answer {
241            Ok(results) => classify(&results),
242            Err(e) => Verdict::Unavailable(format!("{e:#}")),
243        };
244        match &verdict {
245            Verdict::ScriptFailure { reason, details } => {
246                warn!(
247                    %withdrawal_id,
248                    %txid,
249                    %reason,
250                    details = details.as_deref().unwrap_or_default(),
251                    "bitcoind fails the signed withdrawal's scripts"
252                );
253                self.metrics
254                    .withdrawal_bitcoin_check_script_failures_total
255                    .inc();
256                let mut failures = self.failures.lock().unwrap();
257                failures.retain(|_, (at, _)| at.elapsed() < FAILURE_TTL);
258                failures.insert(wtxid, (Instant::now(), verdict.clone()));
259            }
260            other => debug!(%withdrawal_id, %txid, verdict = ?other, "bitcoind check answered"),
261        }
262        sender.send_replace(Some(verdict));
263    }
264
265    fn cached_failure(&self, wtxid: &Wtxid) -> Option<Verdict> {
266        self.failures
267            .lock()
268            .unwrap()
269            .get(wtxid)
270            .filter(|(at, _)| at.elapsed() < FAILURE_TTL)
271            .map(|(_, verdict)| verdict.clone())
272    }
273}
274
275struct InFlightEntry {
276    check: Arc<FinalizeBitcoinCheck>,
277    wtxid: Wtxid,
278}
279
280impl Drop for InFlightEntry {
281    fn drop(&mut self) {
282        self.check
283            .in_flight
284            .lock()
285            .unwrap_or_else(PoisonError::into_inner)
286            .remove(&self.wtxid);
287    }
288}

290const PROBE_INTERVAL: Duration = Duration::from_secs(60 * 60);
291const PROBE_POLL: Duration = Duration::from_secs(60);
292const PROBE_OUTPUT_SATS: u64 = 1_000;
293const PROBE_MIN_SATS: u64 = 30_000;
294
295#[derive(Debug)]
296enum Probed {
297    Refused,
298    Answered,
299    Missed(String),
300    Unanswered(String),
301}
302
303struct Probes {
304    answers: Option<bool>,
305    refuses_bad_spend: Option<bool>,
306    blind: i64,
307    logged: Option<(String, Instant)>,
308}
309
310impl Default for Probes {
311    fn default() -> Self {
312        Self {
313            answers: None,
314            refuses_bad_spend: None,
315            blind: -1,
316            logged: None,
317        }
318    }
319}
320
321impl Probes {
322    fn record(&mut self, gauge: &IntGauge, probed: Probed) {
323        let error = match probed {
324            Probed::Refused => {
325                self.answers = Some(true);
326                self.refuses_bad_spend = Some(true);
327                None
328            }
329            Probed::Answered => {
330                self.answers = Some(true);
331                None
332            }
333            Probed::Missed(error) => {
334                self.answers = Some(true);
335                self.refuses_bad_spend = Some(false);
336                Some(error)
337            }
338            Probed::Unanswered(error) => {
339                self.answers = Some(false);
340                Some(error)
341            }
342        };
343        let blind = match (self.answers, self.refuses_bad_spend) {
344            (Some(false), _) | (_, Some(false)) => 1,
345            (_, Some(true)) => 0,
346            _ => -1,
347        };
348        if let Some(error) = error {
349            if self
350                .logged
351                .as_ref()
352                .is_none_or(|(last, at)| *last != error || at.elapsed() >= PROBE_INTERVAL)
353            {
354                error!("The withdrawal finalize check is blind on this node: {error}");
355                self.logged = Some((error, Instant::now()));
356            } else {
357                debug!("The withdrawal finalize check is still blind on this node: {error}");
358            }
359        } else if blind == 1 {
360            debug!(
361                "testmempoolaccept answers, but the finalize check stays blind until a spend probe passes"
362            );
363        } else if self.blind == 1 {
364            info!(
365                blind,
366                "testmempoolaccept answered the finalize check's probe after a failure"
367            );
368            self.logged = None;
369        } else {
370            debug!(blind, "The withdrawal finalize check's probe passed");
371        }
372        self.blind = blind;
373        gauge.set(blind);
374    }
375}

377impl crate::Hashi {
378    pub(crate) fn start_finalize_bitcoin_check_probe(self: Arc<Self>) -> Service {
379        Service::new().spawn_aborting(async move {
380            let mut probes = Probes::default();
381            loop {
382                let probed = match self.probe_utxo().map(|utxo| self.probe_spend(&utxo)) {
383                    Some(Ok(spend)) => self.probe_spend_answer(spend).await,
384                    Some(Err(e)) => {
385                        debug!("Cannot build the finalize check's probe spend yet: {e:#}");
386                        tokio::time::sleep(PROBE_POLL).await;
387                        continue;
388                    }
389                    None => self.probe_answers().await,
390                };
391                let refused = matches!(probed, Probed::Refused);
392                let answered = matches!(probed, Probed::Answered);
393                probes.record(&self.metrics.withdrawal_bitcoin_check_blind, probed);
394                if refused {
395                    tokio::time::sleep(PROBE_INTERVAL).await;
396                } else if answered {
397                    let deadline = Instant::now() + PROBE_INTERVAL;
398                    while self.probe_utxo().is_none() && Instant::now() < deadline {
399                        tokio::time::sleep(PROBE_POLL).await;
400                    }
401                } else {
402                    tokio::time::sleep(PROBE_POLL).await;
403                }
404            }
405        })
406    }
407
408    fn probe_utxo(&self) -> Option<Utxo> {
409        let state = self.onchain_state().state();
410        let bitcoin = state.hashi().bitcoin();
411        let records = bitcoin.utxo_pool.utxo_records();
412        confirmed_unlocked_utxo_ids(records, bitcoin.withdrawal_queue.withdrawal_txns())
413            .iter()
414            .filter_map(|id| records.get(id))
415            .filter(|record| record.utxo.amount >= PROBE_MIN_SATS)
416            .max_by_key(|record| record.utxo.amount)
417            .map(|record| record.utxo.clone())
418    }
419
420    pub async fn probe_withdrawal_script_check(&self) -> anyhow::Result<()> {
421        let utxo = self
422            .probe_utxo()
423            .ok_or_else(|| anyhow::anyhow!("the pool holds no free confirmed UTXO to probe"))?;
424        match self.probe_spend_answer(self.probe_spend(&utxo)?).await {
425            Probed::Refused => Ok(()),
426            other => anyhow::bail!("{other:?}"),
427        }
428    }
429
430    fn probe_spend(&self, utxo: &Utxo) -> anyhow::Result<bitcoin::Transaction> {
431        let output = OutputUtxo {
432            amount: PROBE_OUTPUT_SATS,
433            bitcoin_address: hashi_types::bitcoin::witness_program_from_address(
434                &self.get_deposit_address(None)?,
435            )?,
436        };
437        let mut tx = self.build_unsigned_withdrawal_tx(std::slice::from_ref(utxo), &[output])?;
438        tx.input[0].witness =
439            self.withdrawal_input_witness(utxo.derivation_path.as_ref(), &[1; 64], &[2; 64])?;
440        Ok(tx)
441    }
442
443    async fn probe_spend_answer(&self, spend: bitcoin::Transaction) -> Probed {
444        match self.probe_test_mempool_accept(spend).await {
445            Ok(results) => spend_probe_result(&results),
446            Err(e) => Probed::Unanswered(format!("{e:#}")),
447        }
448    }
449
450    async fn probe_answers(&self) -> Probed {
451        let tx = bitcoin::Transaction {
452            version: Version::TWO,
453            lock_time: LockTime::ZERO,
454            input: vec![TxIn {
455                previous_output: OutPoint::null(),
456                script_sig: ScriptBuf::new(),
457                sequence: Sequence::MAX,
458                witness: Witness::new(),
459            }],
460            output: vec![TxOut {
461                value: Amount::from_sat(1_000),
462                script_pubkey: ScriptBuf::new(),
463            }],
464        };
465        match self
466            .probe_test_mempool_accept(tx)
467            .await
468            .map(|results| classify(&results))
469        {
470            Ok(Verdict::Unavailable(error)) => Probed::Unanswered(error),
471            Ok(_) => Probed::Answered,
472            Err(e) => Probed::Unanswered(format!("{e:#}")),
473        }
474    }
475
476    async fn probe_test_mempool_accept(
477        &self,
478        tx: bitcoin::Transaction,
479    ) -> anyhow::Result<Vec<MempoolAcceptance>> {
480        tokio::time::timeout(MEMBER_WAIT, self.btc_monitor().test_mempool_accept(tx))
481            .await
482            .map_err(|_| {
483                anyhow::anyhow!(
484                    "testmempoolaccept did not answer within {}s",
485                    MEMBER_WAIT.as_secs()
486                )
487            })?
488    }
