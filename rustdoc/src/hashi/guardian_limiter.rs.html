4//! Local emulator of the guardian's token-bucket limiter. Bootstrapped
5//! from the guardian at startup and advanced by the watcher on every
6//! `WithdrawalSigned`. MPC signing only `validate_consume`s.
7
8use hashi_types::guardian::LimiterConfig;
9use hashi_types::guardian::LimiterState;
10use std::collections::VecDeque;
11use std::fmt;
12use std::sync::RwLock;
13
14/// The mirrored policy and the position under it. Capacity is a function of
15/// both, so they are read as a pair.
16#[derive(Debug, Clone, Copy, PartialEq)]
17pub struct LimiterView {
18    pub config: LimiterConfig,
19    pub state: LimiterState,
20}
21
22impl LimiterView {
23    pub fn capacity_at(&self, timestamp_secs: u64) -> u64 {
24        project_capacity(&self.config, &self.state, timestamp_secs)
25    }
26}
27
28pub struct LocalLimiter {
29    view: RwLock<LimiterView>,
30}
31
32#[derive(Debug, thiserror::Error)]
33pub enum LocalLimiterError {
34    #[error("stale timestamp: local last_updated_at={local_last}, incoming={incoming}")]
35    StaleTimestamp { local_last: u64, incoming: u64 },
36
37    #[error("insufficient capacity: needed {needed}, available {available}")]
38    InsufficientCapacity { needed: u64, available: u64 },
39
40    #[error("seq mismatch: local={local}, incoming={incoming}")]
41    SeqMismatch { local: u64, incoming: u64 },
42}
43
44impl fmt::Debug for LocalLimiter {
45    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
46        // `try_read` like std's own `RwLock` impl: never block to format.
47        let config = self.view.try_read().ok().map(|view| view.config);
48        f.debug_struct("LocalLimiter")
49            .field("config", &config)
50            .finish_non_exhaustive()
51    }
52}
53
54impl LocalLimiter {
55    pub fn new(config: LimiterConfig, state: LimiterState) -> Self {
56        Self {
57            view: RwLock::new(LimiterView { config, state }),
58        }
59    }
60
61    /// Both halves under one lock; two accessors could straddle an
62    /// [`Self::adopt_config`].
63    pub fn view(&self) -> LimiterView {
64        *self.view.read().unwrap()
65    }
66
67    pub fn config(&self) -> LimiterConfig {
68        self.view.read().unwrap().config
69    }
70
71    pub fn snapshot(&self) -> LimiterState {
72        self.view.read().unwrap().state
73    }
74
75    pub fn capacity_at(&self, timestamp_secs: u64) -> u64 {
76        self.view().capacity_at(timestamp_secs)
77    }
78
79    pub fn next_seq(&self) -> u64 {
80        self.view.read().unwrap().state.next_seq
81    }
82
83    /// Validate a consume; does not mutate state.
84    pub fn validate_consume(
85        &self,
86        expected_seq: u64,
87        timestamp_secs: u64,
88        amount_sats: u64,
89    ) -> Result<(), LocalLimiterError> {
90        let view = *self.view.read().unwrap();
91        if expected_seq != view.state.next_seq {
92            return Err(LocalLimiterError::SeqMismatch {
93                local: view.state.next_seq,
94                incoming: expected_seq,
95            });
96        }
97        if timestamp_secs < view.state.last_updated_at {
98            return Err(LocalLimiterError::StaleTimestamp {
99                local_last: view.state.last_updated_at,
100                incoming: timestamp_secs,
101            });
102        }
103        let capacity = view.capacity_at(timestamp_secs);
104        if capacity < amount_sats {
105            return Err(LocalLimiterError::InsufficientCapacity {
106                needed: amount_sats,
107                available: capacity,
108            });
109        }
110        Ok(())
111    }
112
113    /// Advance local state to match an accepted consume. State is left
114    /// untouched on error.
115    pub fn apply_consume(
116        &self,
117        applied_seq: u64,
118        timestamp_secs: u64,
119        amount_sats: u64,
120    ) -> Result<(), LocalLimiterError> {
121        let mut guard = self.view.write().unwrap();
122        if applied_seq != guard.state.next_seq {
123            return Err(LocalLimiterError::SeqMismatch {
124                local: guard.state.next_seq,
125                incoming: applied_seq,
126            });
127        }
128        if timestamp_secs < guard.state.last_updated_at {
129            return Err(LocalLimiterError::StaleTimestamp {
130                local_last: guard.state.last_updated_at,
131                incoming: timestamp_secs,
132            });
133        }
134        let capacity = guard.capacity_at(timestamp_secs);
135        if capacity < amount_sats {
136            return Err(LocalLimiterError::InsufficientCapacity {
137                needed: amount_sats,
138                available: capacity,
139            });
140        }
141        guard.state.num_tokens_available = capacity - amount_sats;
142        guard.state.last_updated_at = timestamp_secs;
143        guard.state.next_seq += 1;
144        Ok(())
145    }
146
147    /// Take the guardian's current policy, capping tokens to the new ceiling as
148    /// the guardian's own activation recovery does. Returns the replaced policy
149    /// if it changed.
150    ///
151    /// A rotation re-provisions the enclave, so the policy can move under a
152    /// running mirror; reconciling state cannot correct that, since both sides
153    /// are then projected through a capacity function that no longer matches.
154    /// No seq guard is needed: `next_seq` is untouched and tokens only move
155    /// down, so this can neither double-count an in-flight withdrawal nor
156    /// invent capacity.
157    pub fn adopt_config(&self, config: LimiterConfig) -> Option<LimiterConfig> {
158        let mut guard = self.view.write().unwrap();
159        let previous = guard.config;
160        if previous == config {
161            return None;
162        }
163        guard.config = config;
164        guard.state.num_tokens_available = guard
165            .state
166            .num_tokens_available
167            .min(config.max_bucket_capacity);
168        Some(previous)
169    }
170
171    /// Overwrite local state with the guardian's authoritative `state`.
172    pub fn reconcile_to(&self, state: LimiterState) {
173        self.view.write().unwrap().state = state;
174    }
175
176    /// Snap the mirror to the guardian's `state` on genuine drift, only at a matching
177    /// `next_seq` (so a racing `apply_consume` isn't clobbered). A pure refill-timing
178    /// difference — same bucket line, different snapshot instant — isn't drift (equal
179    /// projected capacity at a common time); clobbering it would stall the refill.
180    /// Callers must [`Self::adopt_config`] first: both sides project through
181    /// the mirror's own policy, so a stale ceiling saturates them into a
182    /// false match.
183    pub fn reconcile_token_drift(&self, state: LimiterState) -> bool {
184        let mut guard = self.view.write().unwrap();
185        if guard.state.next_seq != state.next_seq {
186            return false;
187        }
188        let common = guard.state.last_updated_at.max(state.last_updated_at);
189        if guard.capacity_at(common) == project_capacity(&guard.config, &state, common) {
190            return false;
191        }
192        guard.state = state;
193        true
194    }
195}
196
197fn project_capacity(config: &LimiterConfig, state: &LimiterState, timestamp_secs: u64) -> u64 {
198    let elapsed = timestamp_secs.saturating_sub(state.last_updated_at);
199    let refilled = elapsed.saturating_mul(config.refill_rate);
200    state
201        .num_tokens_available
202        .saturating_add(refilled)
203        .min(config.max_bucket_capacity)
204}

206/// Defer when the local limiter is behind a guardian-consumed seq for a
207/// *different* withdrawal; same-wid retries are served idempotently.
208pub(crate) fn should_defer_guardian_finalize(
209    next_seq: u64,
210    last_finalized: Option<(u64, sui_sdk_types::Address)>,
211    wid: sui_sdk_types::Address,
212) -> bool {
213    last_finalized.is_some_and(|(last_seq, last_wid)| next_seq <= last_seq && wid != last_wid)
214}

216/// 20 ticks ≈ 5 min at the reconcile cadence.
217pub(crate) const STALL_RECONCILE_TICKS: usize = 20;
218
219/// Flags a local mirror that has fallen out of lockstep with the guardian.
220///
221/// The guardian bumps `next_seq` at finalize-request time but a node only
222/// advances on the matching on-chain `WithdrawalSigned`, so `local <
223/// guardian` is ordinary in-flight lag and must not, on its own, trip a
224/// reconcile (forward-snapping a healthy mirror would double-count the event
225/// still in flight). We fire only when the mirror has failed to reach the
226/// seq the guardian held a full [`STALL_RECONCILE_TICKS`] window ago — by
227/// now those withdrawals have certainly emitted their events, so a mirror
228/// still short has genuinely dropped one — or when it runs ahead of the
229/// guardian, which can only happen once lockstep is already lost.
230#[derive(Default)]
231pub(crate) struct LimiterStallTracker {
232    /// Guardian `next_seq` seen on each of the last [`STALL_RECONCILE_TICKS`] ticks.
233    guardian_seq_window: VecDeque<u64>,
234}
235
236impl LimiterStallTracker {
237    pub(crate) fn observe(&mut self, local_seq: u64, guardian_seq: u64) -> bool {
238        // Once the window is full its front is the guardian seq from a full
239        // window ago; the mirror should have reached it by now.
240        let stalled = self.guardian_seq_window.len() == STALL_RECONCILE_TICKS
241            && self
242                .guardian_seq_window
243                .front()
244                .is_some_and(|&windowed_seq| local_seq < windowed_seq || local_seq > guardian_seq);
245        if stalled {
246            self.guardian_seq_window.clear();
247            return true;
248        }
249        self.guardian_seq_window.push_back(guardian_seq);
250        if self.guardian_seq_window.len() > STALL_RECONCILE_TICKS {
251            self.guardian_seq_window.pop_front();
252        }
253        false
254    }
255}
