14#[derive(Clone)]
15pub struct MonitorConfig {
16    /// Bitcoin network to connect to
17    pub network: Network,
18
19    /// Peers for P2P connections, identified by hostname (or IP) and port.
20    /// Hostnames are resolved at each connection attempt, so IP changes
21    /// (e.g. Kubernetes pod rotation) are followed.
22    pub trusted_peers: Vec<kyoto::TrustedPeer>,
23
24    /// Starting block height for synchronization
25    pub start_height: u32,
26
27    /// bitcoind JSON-RPC server URL to connect to
28    pub bitcoind_rpc_url: String,
29
30    /// bitcoind JSON-RPC server auth config
31    pub bitcoind_rpc_auth: corepc_client::client_sync::Auth,
32}
33
34fn redacted_auth(auth: &corepc_client::client_sync::Auth) -> String {
35    match auth {
36        corepc_client::client_sync::Auth::None => "None".to_string(),
37        corepc_client::client_sync::Auth::UserPass(user, _) => {
38            format!("UserPass({user}, <redacted>)")
39        }
40        corepc_client::client_sync::Auth::CookieFile(path) => {
41            format!("CookieFile({})", path.display())
42        }
43    }
44}
45
46impl std::fmt::Debug for MonitorConfig {
47    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
48        f.debug_struct("MonitorConfig")
49            .field("network", &self.network)
50            .field("trusted_peers", &self.trusted_peers)
51            .field("start_height", &self.start_height)
52            .field("bitcoind_rpc_url", &self.bitcoind_rpc_url)
53            .field("bitcoind_rpc_auth", &redacted_auth(&self.bitcoind_rpc_auth))
54            .finish()
55    }
56}

64            bitcoind_rpc_url: "http://localhost:8332".to_string(),

77/// Builder for constructing monitor configuration.
78#[derive(Default)]
79pub struct MonitorConfigBuilder {
80    network: Option<Network>,
81    trusted_peers: Vec<kyoto::TrustedPeer>,
82    start_height: u32,
83    bitcoind_rpc_url: Option<String>,
84    bitcoind_rpc_auth: Option<corepc_client::client_sync::Auth>,
85}
86
87impl std::fmt::Debug for MonitorConfigBuilder {
88    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
89        f.debug_struct("MonitorConfigBuilder")
90            .field("network", &self.network)
91            .field("trusted_peers", &self.trusted_peers)
92            .field("start_height", &self.start_height)
93            .field("bitcoind_rpc_url", &self.bitcoind_rpc_url)
94            .field(
95                "bitcoind_rpc_auth",
96                &self.bitcoind_rpc_auth.as_ref().map(redacted_auth),
97            )
98            .finish()
99    }
100}

102impl MonitorConfigBuilder {
103    /// Set the Bitcoin network.
104    pub fn network(mut self, network: Network) -> Self {
105        self.network = Some(network);
106        self
107    }
108
109    /// Set peers for P2P connections. Accepts hostnames or IPs with port.
110    /// Hostnames are resolved at connection time.
111    pub fn trusted_peers(mut self, peers: Vec<kyoto::TrustedPeer>) -> Self {
112        self.trusted_peers = peers;
113        self
114    }
115
116    /// Set the starting block height for synchronization.
117    pub fn start_height(mut self, height: u32) -> Self {
118        self.start_height = height;
119        self
120    }
121
122    /// Set the bitcoind JSON-RPC server config.
123    pub fn bitcoind_rpc_config(
124        mut self,
125        url: String,
126        auth: corepc_client::client_sync::Auth,
127    ) -> Self {
128        self.bitcoind_rpc_url = Some(url);
129        self.bitcoind_rpc_auth = Some(auth);
130        self
131    }
132
133    pub fn build(self) -> MonitorConfig {
134        let default = MonitorConfig::default();
135
136        MonitorConfig {
137            network: self.network.unwrap_or(default.network),
138            trusted_peers: self.trusted_peers,
139            start_height: self.start_height,
140            bitcoind_rpc_url: self.bitcoind_rpc_url.unwrap_or(default.bitcoind_rpc_url),
141            bitcoind_rpc_auth: self.bitcoind_rpc_auth.unwrap_or(default.bitcoind_rpc_auth),
142        }
143    }
144}

146/// Wrapper around corepc_client::client_sync::Auth that we can serialize/deserialize from configs
147#[derive(Clone, serde_derive::Deserialize, serde_derive::Serialize)]
148pub enum BtcRpcAuth {
149    None,
150    UserPass(String, String),
151    CookieFile(PathBuf),
152}
153
154impl std::fmt::Debug for BtcRpcAuth {
155    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
156        match self {
157            Self::None => f.write_str("None"),
158            Self::UserPass(user, _) => f
159                .debug_tuple("UserPass")
160                .field(user)
161                .field(&"<redacted>")
162                .finish(),
163            Self::CookieFile(path) => f.debug_tuple("CookieFile").field(path).finish(),
164        }
165    }
166}
167
168impl BtcRpcAuth {
169    pub fn to_corepc_auth(&self) -> corepc_client::client_sync::Auth {
170        match self {
171            BtcRpcAuth::None => corepc_client::client_sync::Auth::None,
172            BtcRpcAuth::UserPass(user, pass) => {
173                corepc_client::client_sync::Auth::UserPass(user.clone(), pass.clone())
174            }
175            BtcRpcAuth::CookieFile(path) => {
176                corepc_client::client_sync::Auth::CookieFile(path.clone())
177            }
178        }
179    }
180}

182/// Create a `corepc_client` RPC client, handling the `Auth::None` case
183/// (where `Client::new_with_auth` would error with `MissingUserPassword`).
184pub fn new_rpc_client(
185    url: &str,
186    auth: corepc_client::client_sync::Auth,
187) -> corepc_client::client_sync::Result<corepc_client::client_sync::v29::Client> {
188    match auth {
189        corepc_client::client_sync::Auth::None => {
190            Ok(corepc_client::client_sync::v29::Client::new(url))
191        }
192        auth => corepc_client::client_sync::v29::Client::new_with_auth(url, auth),
193    }
194}

196/// Parse a human-readable network name into a [`Network`].
197///
198/// Recognised values: `"mainnet"`, `"testnet4"`, `"regtest"`.
199/// Returns [`Network::Regtest`] when `name` is `None`.
200/// Returns an error for unrecognised network names.
201pub fn parse_btc_network(name: Option<&str>) -> anyhow::Result<Network> {
202    match name {
203        Some("mainnet") => Ok(Network::Bitcoin),
204        Some("testnet4") => Ok(Network::Testnet4),
205        Some("signet") => Ok(Network::Signet),
206        Some("regtest") | None => Ok(Network::Regtest),
207        Some(other) => anyhow::bail!(
208            "Unknown BTC network '{}'. Use mainnet, testnet4, signet, or regtest",
209            other
210        ),
211    }
212}

214pub fn network_from_chain_id(chain_id: &str) -> Option<Network> {
215    let hash = BlockHash::from_str(chain_id).ok()?;
216
217    [
218        Network::Bitcoin,
219        Network::Testnet4,
220        Network::Signet,
221        Network::Regtest,
222    ]
223    .into_iter()
224    .find(|&net| genesis_block(net).block_hash() == hash)
225}

227/// Fallback start height when `bitcoin-start-height` is unset. On Mainnet any
228/// height past Taproot activation works; it maps to the Taproot checkpoint.
229const DEFAULT_START_HEIGHT: u32 = 800_000;
230
231/// Anchors Signet nodes below the Hashi Testnet deployment (~height 313343).
232/// Only ever lower this; deposits below the anchor are never seen.
233const SIGNET_DEFAULT_START_HEIGHT: u32 = 300_000;
234
235/// Networks without a deployment anchor start at genesis, which is always valid
236/// and cheap on the short chains (regtest, testnet4) that reach this arm.
237pub fn default_start_height(network: Network) -> u32 {
238    match network {
239        Network::Bitcoin => DEFAULT_START_HEIGHT,
240        Network::Signet => SIGNET_DEFAULT_START_HEIGHT,
241        _ => 0,
242    }
243}
