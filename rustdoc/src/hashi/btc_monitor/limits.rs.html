16const PLACES: usize = 192;
17const CALLS: usize = 16;
18const DISCOVERIES: usize = 64;
19const FETCH_SLOTS: usize = 1;
20
21#[derive(Debug, thiserror::Error)]
22#[error("per-caller Bitcoin monitor queue is full")]
23pub struct BitcoindBusy;
24
25#[derive(Clone)]
26pub(crate) struct BitcoindLimiter {
27    callers: Arc<Mutex<HashMap<Address, Arc<CallerLimits>>>>,
28    metrics: Arc<Metrics>,
29}
30
31struct CallerLimits {
32    caller: Address,
33    places: Arc<Semaphore>,
34    calls: Arc<Semaphore>,
35    discoveries: Arc<Semaphore>,
36    fetch_slots: Arc<Semaphore>,
37    places_in_use: IntGauge,
38    calls_in_use: IntGauge,
39    discoveries_in_use: IntGauge,
40    shed: IntCounter,
41    discoveries_refused: IntCounter,
42}
43
44impl BitcoindLimiter {
45    pub(crate) fn new(metrics: Arc<Metrics>) -> Self {
46        Self {
47            callers: Default::default(),
48            metrics,
49        }
50    }
51
52    pub(crate) fn gate(&self, caller: Address, deadline: tokio::time::Instant) -> BitcoindGate {
53        let limits = self
54            .callers
55            .lock()
56            .unwrap()
57            .entry(caller)
58            .or_insert_with(|| {
59                let label = caller.to_string();
60                let metrics = &self.metrics;
61                Arc::new(CallerLimits {
62                    caller,
63                    places: Arc::new(Semaphore::new(PLACES)),
64                    calls: Arc::new(Semaphore::new(CALLS)),
65                    discoveries: Arc::new(Semaphore::new(DISCOVERIES)),
66                    fetch_slots: Arc::new(Semaphore::new(FETCH_SLOTS)),
67                    places_in_use: metrics
68                        .peer_bitcoind_places_in_use
69                        .with_label_values(&[&label]),
70                    calls_in_use: metrics
71                        .peer_bitcoind_calls_in_use
72                        .with_label_values(&[&label]),
73                    discoveries_in_use: metrics
74                        .peer_bitcoind_discoveries_in_use
75                        .with_label_values(&[&label]),
76                    shed: metrics
77                        .peer_bitcoind_requests_shed_total
78                        .with_label_values(&[&label]),
79                    discoveries_refused: metrics
80                        .peer_bitcoind_discoveries_refused_total
81                        .with_label_values(&[&label]),
82                })
83            })
84            .clone();
85        BitcoindGate(Some((limits, deadline)))
86    }
87}

89pub struct BitcoindGate(Option<(Arc<CallerLimits>, tokio::time::Instant)>);
90
91impl BitcoindGate {
92    pub fn local() -> Self {
93        Self(None)
94    }
95
96    pub(crate) fn caller(&self) -> Option<Address> {
97        self.0.as_ref().map(|(limits, _)| limits.caller)
98    }
99
100    pub(crate) async fn take_place(&self) -> Result<BitcoindCharge, BitcoindBusy> {
101        let Some((limits, deadline)) = &self.0 else {
102            return Ok(None);
103        };
104        let place = tokio::time::timeout_at(*deadline, limits.places.clone().acquire_owned())
105            .await
106            .map_err(|_| {
107                limits.shed.inc();
108                BitcoindBusy
109            })?
110            .expect("bitcoind place semaphores are never closed");
111        Ok(Some(Arc::new(CallerCharge {
112            limits: limits.clone(),
113            _place: Held::new(place, &limits.places_in_use),
114        })))
115    }
116}

118pub type BitcoindCharge = Option<Arc<CallerCharge>>;
119
120pub struct CallerCharge {
121    limits: Arc<CallerLimits>,
122    _place: Held,
123}
124
125impl CallerCharge {
126    pub(crate) fn caller(&self) -> Address {
127        self.limits.caller
128    }
129
130    pub(crate) async fn call(&self) -> Held {
131        let call = self
132            .limits
133            .calls
134            .clone()
135            .acquire_owned()
136            .await
137            .expect("bitcoind call semaphores are never closed");
138        Held::new(call, &self.limits.calls_in_use)
139    }
140
141    pub(crate) async fn fetch_slot(&self) -> OwnedSemaphorePermit {
142        self.limits
143            .fetch_slots
144            .clone()
145            .acquire_owned()
146            .await
147            .expect("Kyoto fetch slot semaphores are never closed")
148    }
149
150    pub(crate) fn try_discovery(&self) -> Option<Held> {
151        match self.limits.discoveries.clone().try_acquire_owned() {
152            Ok(discovery) => Some(Held::new(discovery, &self.limits.discoveries_in_use)),
153            Err(_) => {
154                self.limits.discoveries_refused.inc();
155                None
156            }
157        }
158    }
159}
160
161impl std::fmt::Debug for CallerCharge {
162    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
163        f.debug_struct("CallerCharge")
164            .field("caller", &self.limits.caller)
165            .finish()
166    }
167}

169pub(crate) struct Held {
170    _permit: OwnedSemaphorePermit,
171    in_use: IntGauge,
172}
173
174impl Held {
175    fn new(permit: OwnedSemaphorePermit, in_use: &IntGauge) -> Self {
176        in_use.inc();
177        Self {
178            _permit: permit,
179            in_use: in_use.clone(),
180        }
181    }
182}
183
184impl Drop for Held {
185    fn drop(&mut self) {
186        self.in_use.dec();
187    }
