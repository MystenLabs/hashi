4//! TLS for the node listener, which a TCP-passthrough load balancer fronts:
5//! nodes present their registered TLS key as a client certificate, and an
6//! ALB can't verify Ed25519 keys. The listener serves a certificate for the
7//! node hostname: an exportable ACM certificate, re-exported as ACM renews it,
8//! or PEM files.

38/// ACM renews a certificate 45 days before it expires; the proxy picks the
39/// renewal up on its next reload.
40const RELOAD_INTERVAL: Duration = Duration::from_secs(6 * 60 * 60);
41/// The AWS SDK sets no read timeout, so a stalled export would otherwise hold
42/// up startup or every later reload.
43const EXPORT_TIMEOUT: Duration = Duration::from_secs(30);
44
45#[derive(Clone, Debug)]
46pub enum CertSource {
47    /// An exportable ACM certificate, exported with the task role.
48    Acm { arn: String },
49    /// A PEM certificate chain, leaf first, and its PEM private key.
50    Files { cert: PathBuf, key: PathBuf },
51}
52
53/// The certificate the proxy serves, replaced in place by a successful reload.
54#[derive(Debug)]
55pub struct ServerCert {
56    current: RwLock<Arc<CertifiedKey>>,
57}
58
59impl ServerCert {
60    pub async fn load(source: &CertSource, metrics: &ProxyMetrics) -> Result<Arc<Self>> {
61        let key = load_certified_key(source).await?;
62        record_expiry(&key, metrics)?;
63        Ok(Arc::new(Self {
64            current: RwLock::new(Arc::new(key)),
65        }))
66    }
67
68    /// Keeps the current certificate when the load fails.
69    async fn reload(&self, source: &CertSource, metrics: &ProxyMetrics) -> Result<()> {
70        let key = load_certified_key(source).await?;
71        record_expiry(&key, metrics)?;
72        *self.current.write().expect("certificate lock poisoned") = Arc::new(key);
73        Ok(())
74    }
75
76    pub async fn reload_forever(self: Arc<Self>, source: CertSource, metrics: Arc<ProxyMetrics>) {
77        loop {
78            tokio::time::sleep(RELOAD_INTERVAL).await;
79            match self.reload(&source, &metrics).await {
80                Ok(()) => info!("Reloaded the TLS certificate."),
81                Err(e) => {
82                    metrics.tls_cert_reload_failures.inc();
83                    warn!(
84                        error = %format!("{e:#}"),
85                        "TLS certificate reload failed; serving the current one."
86                    );
87                }
88            }
89        }
90    }
91}

93impl ResolvesServerCert for ServerCert {
94    fn resolve(&self, _: ClientHello<'_>) -> Option<Arc<CertifiedKey>> {
95        Some(
96            self.current
97                .read()
98                .expect("certificate lock poisoned")
99                .clone(),
100        )
101    }
102}

104pub fn server_config(cert: Arc<ServerCert>) -> Result<rustls::ServerConfig> {
105    Ok(rustls::ServerConfig::builder_with_provider(Arc::new(
106        rustls::crypto::ring::default_provider(),
107    ))
108    .with_protocol_versions(&[&rustls::version::TLS13])?
109    .with_client_cert_verifier(Arc::new(NodeCertVerifier::new()))
110    .with_cert_resolver(cert))
111}

113/// The Ed25519 key of a node's self-signed TLS certificate.
114pub fn node_tls_key(cert: &CertificateDer<'_>) -> Option<[u8; 32]> {
115    use ed25519_dalek::pkcs8::DecodePublicKey;
116    use x509_parser::prelude::FromDer;
117
118    let (_, cert) = x509_parser::certificate::X509Certificate::from_der(cert).ok()?;
119    let key = ed25519_dalek::VerifyingKey::from_public_key_der(cert.public_key().raw).ok()?;
120    Some(key.to_bytes())
121}

123/// Requires a client certificate with an Ed25519 key, proven by the client's
124/// TLS 1.3 signature. Nodes present self-signed certificates, so there is no
125/// chain to check; which keys may call node RPCs is decided per request.
126#[derive(Debug)]
127struct NodeCertVerifier {
128    supported_algs: WebPkiSupportedAlgorithms,
129}

131impl NodeCertVerifier {
132    fn new() -> Self {
133        Self {
134            supported_algs: rustls::crypto::ring::default_provider()
135                .signature_verification_algorithms,
136        }
137    }
138}

140impl ClientCertVerifier for NodeCertVerifier {
141    fn offer_client_auth(&self) -> bool {
142        true
143    }
144
145    fn client_auth_mandatory(&self) -> bool {
146        true
147    }
148
149    fn root_hint_subjects(&self) -> &[DistinguishedName] {
150        &[]
151    }
152
153    fn verify_client_cert(
154        &self,
155        end_entity: &CertificateDer<'_>,
156        _intermediates: &[CertificateDer<'_>],
157        _now: UnixTime,
158    ) -> Result<ClientCertVerified, rustls::Error> {
159        node_tls_key(end_entity).ok_or(rustls::Error::InvalidCertificate(
160            CertificateError::BadEncoding,
161        ))?;
162        Ok(ClientCertVerified::assertion())
163    }
164
165    fn verify_tls12_signature(
166        &self,
167        _message: &[u8],
168        _cert: &CertificateDer<'_>,
169        _dss: &DigitallySignedStruct,
170    ) -> Result<HandshakeSignatureValid, rustls::Error> {
171        Err(rustls::Error::PeerIncompatible(
172            PeerIncompatible::Tls12NotOffered,
173        ))
174    }
175
176    fn verify_tls13_signature(
177        &self,
178        message: &[u8],
179        cert: &CertificateDer<'_>,
180        dss: &DigitallySignedStruct,
181    ) -> Result<HandshakeSignatureValid, rustls::Error> {
182        rustls::crypto::verify_tls13_signature(message, cert, dss, &self.supported_algs)
183    }
184
185    fn supported_verify_schemes(&self) -> Vec<SignatureScheme> {
186        vec![SignatureScheme::ED25519]
187    }
188}

190async fn load_certified_key(source: &CertSource) -> Result<CertifiedKey> {
191    let (chain, key) = match source {
192        CertSource::Acm { arn } => tokio::time::timeout(EXPORT_TIMEOUT, export_from_acm(arn))
193            .await
194            .context("the ACM certificate export timed out")??,
195        CertSource::Files { cert, key } => {
196            let chain = CertificateDer::pem_file_iter(cert)
197                .and_then(|certs| certs.collect::<Result<Vec<_>, _>>())
198                .with_context(|| format!("read TLS certificate chain {}", cert.display()))?;
199            let key = PrivateKeyDer::from_pem_file(key)
200                .with_context(|| format!("read TLS private key {}", key.display()))?;
201            (chain, key)
202        }
203    };
204    anyhow::ensure!(!chain.is_empty(), "the TLS certificate chain is empty");
205    CertifiedKey::from_der(chain, key, &rustls::crypto::ring::default_provider())
206        .context("the TLS private key does not load or does not match the certificate")
207}

209async fn export_from_acm(
210    arn: &str,
211) -> Result<(Vec<CertificateDer<'static>>, PrivateKeyDer<'static>)> {
212    // arn:aws:acm:<region>:<account>:certificate/<id>
213    let region = arn
214        .split(':')
215        .nth(3)
216        .filter(|region| !region.is_empty())
217        .with_context(|| format!("no region in certificate ARN {arn}"))?;
218    let aws_config = aws_config::defaults(aws_config::BehaviorVersion::latest())
219        .region(aws_config::Region::new(region.to_string()))
220        .load()
221        .await;
222    let passphrase = hex::encode(rand::random::<[u8; 32]>());
223    let exported = aws_sdk_acm::Client::new(&aws_config)
224        .export_certificate()
225        .certificate_arn(arn)
226        .passphrase(aws_sdk_acm::primitives::Blob::new(passphrase.as_bytes()))
227        .send()
228        .await
229        .with_context(|| format!("export ACM certificate {arn}"))?;
230
231    let chain = exported_chain(
232        exported
233            .certificate()
234            .context("ACM export returned no certificate")?,
235        exported.certificate_chain(),
236    )?;
237    let key = decrypt_private_key(
238        exported
239            .private_key()
240            .context("ACM export returned no private key")?,
241        passphrase.as_bytes(),
242    )?;
243    Ok((chain, key))
244}

246/// The leaf followed by the export's chain, which excludes it.
247fn exported_chain(leaf: &str, chain: Option<&str>) -> Result<Vec<CertificateDer<'static>>> {
248    let pem = format!("{leaf}\n{}", chain.unwrap_or_default());
249    let chain = CertificateDer::pem_slice_iter(pem.as_bytes())
250        .collect::<Result<Vec<_>, _>>()
251        .context("parse the exported ACM certificate chain")?;
252    // Clients don't fetch missing intermediates, so a leaf alone fails every handshake.
253    anyhow::ensure!(
254        chain.len() > 1,
255        "ACM export returned no intermediate certificates"
256    );
257    Ok(chain)
258}

260fn decrypt_private_key(pem: &str, passphrase: &[u8]) -> Result<PrivateKeyDer<'static>> {
261    use pkcs8::der::Decode;
262
263    let (label, der) =
264        pkcs8::der::pem::decode_vec(pem.as_bytes()).context("parse the private key PEM")?;
265    let key_der = match label {
266        "ENCRYPTED PRIVATE KEY" => pkcs8::EncryptedPrivateKeyInfo::from_der(&der)
267            .context("parse the encrypted private key")?
268            .decrypt(passphrase)
269            .context("decrypt the private key")?
270            .as_bytes()
271            .to_vec(),
272        "PRIVATE KEY" => der,
273        other => anyhow::bail!("unexpected private key PEM label {other:?}"),
274    };
275    Ok(PrivateKeyDer::Pkcs8(key_der.into()))
276}
277
278fn record_expiry(key: &CertifiedKey, metrics: &ProxyMetrics) -> Result<()> {
279    use x509_parser::prelude::FromDer;
280
281    let leaf = key.end_entity_cert()?;
282    let (_, cert) = x509_parser::certificate::X509Certificate::from_der(leaf)
283        .context("parse the TLS leaf certificate")?;
284    metrics
285        .tls_cert_not_after
286        .set(cert.validity().not_after.timestamp());
287    Ok(())
288}

301    pub(crate) fn test_cert() -> TestCert {
302        let key = rcgen::KeyPair::generate().unwrap();
303        let cert = rcgen::CertificateParams::new(vec!["localhost".to_string()])
304            .unwrap()
305            .self_signed(&key)
306            .unwrap();
307        let dir = tempfile::tempdir().unwrap();
308        let (cert_path, key_path) = (dir.path().join("cert.pem"), dir.path().join("key.pem"));
309        std::fs::write(&cert_path, cert.pem()).unwrap();
310        std::fs::write(&key_path, key.serialize_pem()).unwrap();
311        TestCert {
312            cert_pem: cert.pem(),
313            source: CertSource::Files {
314                cert: cert_path,
315                key: key_path,
316            },
317            _dir: dir,
318        }
319    }

321    /// The self-signed certificate a node presents with its TLS key.
322    pub(crate) fn node_identity(key: &ed25519_dalek::SigningKey) -> tonic::transport::Identity {
323        use ed25519_dalek::pkcs8::EncodePrivateKey;
324
325        let pkcs8 = key.to_pkcs8_der().unwrap();
326        let key_pair = rcgen::KeyPair::from_der_and_sign_algo(
327            &PrivateKeyDer::Pkcs8(pkcs8.as_bytes().to_vec().into()),
328            &rcgen::PKCS_ED25519,
329        )
330        .unwrap();
331        let cert = rcgen::CertificateParams::new(vec!["hashi".to_string()])
332            .unwrap()
333            .self_signed(&key_pair)
334            .unwrap();
335        tonic::transport::Identity::from_pem(cert.pem(), key_pair.serialize_pem())
336    }
