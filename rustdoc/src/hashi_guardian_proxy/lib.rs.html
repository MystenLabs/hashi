4//! Out-of-enclave gRPC proxy for the hashi guardian. It fronts the enclave with
5//! a stable, hardenable surface. [`forward`] forwards the `GuardianService` RPCs
6//! to the enclave, rejecting operator/ceremony RPCs, and [`guardian_info`]
7//! caches ordinary `GetGuardianInfo` for every caller. `GetAttestedGuardianInfo`
8//! always forwards to the enclave without caching. The rest is grouped by who calls it:
9//!
10//! - [`node`]: [`node::cache`] makes `StandardWithdrawal` responses idempotent
11//!   by `wid`. It answers from an index over the guardian's own S3 withdrawal
12//!   log ([`node::widlog`]), which the proxy only reads.
13//!   [`node::member_auth`] gates every route: node RPCs are served only to
14//!   current or pending committee members ([`node::members`]), who present
15//!   their registered TLS key as a client certificate. [`node::handoffs`]
16//!   forwards a committee handoff only once the chain stores one between the
17//!   same two epochs.
18//! - [`kp`]: [`kp::relay`] serves `GuardianRelayService`: key provisioners
19//!   submit one share each — authenticated against the ceremony's committed
20//!   roster read from the S3 share log ([`kp::roster`]) — and the relay batches
21//!   a threshold-many into the guardian's `ProvisionerInit`.
22//! - [`public`]: [`public::info`] serves a read-only HTTP `/info` + `/health`
23//!   JSON surface (a curated limiter/identity view, with CORS) so browser /
24//!   `fetch` clients can read limiter status the gRPC surface only exposes to
25//!   nodes — on the same port as gRPC, so the guardian exposes one interface.
26//!
27//! Nodes call it on a second listener, where the proxy terminates TLS itself,
28//! requires their registered TLS key as a client certificate ([`tls`]), and
29//! serves only node RPCs and `GetGuardianInfo`.
30//!
31//! The proxy is liveness-only in the trust model: it can stall but never forge a
32//! withdrawal or read a KP share (shares are end-to-end encrypted to the enclave).

60/// Everything the proxy serves: gRPC (forwarder, relay, health) and the HTTP
61/// `/info` + `/health`. Both listeners mount it, and the member gate picks what
62/// each one serves. `Router::layer` only wraps the routes added before it, so
63/// the member gate goes last.
64pub fn router<L: LogStore, H: Health>(
65    guardian: CachingGuardianGrpc<Forwarding<L>, L>,
66    relay: Relay<L>,
67    health: HealthServer<H>,
68    info: public::info::InfoState,
69    gate: Arc<MemberGate>,
70) -> axum::Router {
71    axum::Router::new()
72        .add_grpc_service(health)
73        .add_grpc_service(GuardianServiceServer::new(guardian))
74        .add_grpc_service(GuardianRelayServiceServer::new(relay))
75        .merge(public::info::router(info))
76        .layer(axum::middleware::from_fn_with_state(
77            gate,
78            node::member_auth::require_committee_member,
79        ))
80}

82/// Mount a tonic gRPC service as an axum route-service at `/{ServiceName}/*`, so
83/// gRPC and plain-HTTP routes share one router. Mirrors `crates/hashi/src/grpc/mod.rs`.
84trait RouterExt {
85    fn add_grpc_service<S>(self, svc: S) -> Self
86    where
87        S: tower::Service<
88                axum::extract::Request,
89                Response: axum::response::IntoResponse,
90                Error = std::convert::Infallible,
91            > + tonic::server::NamedService
92            + Clone
93            + Send
94            + Sync
95            + 'static,
96        S::Future: Send + 'static;
97}
98
99impl RouterExt for axum::Router {
100    fn add_grpc_service<S>(self, svc: S) -> Self
101    where
102        S: tower::Service<
103                axum::extract::Request,
104                Response: axum::response::IntoResponse,
105                Error = std::convert::Infallible,
106            > + tonic::server::NamedService
107            + Clone
108            + Send
109            + Sync
110            + 'static,
111        S::Future: Send + 'static,
112    {
113        self.route_service(&format!("/{}/{{*rest}}", S::NAME), svc)
114    }
115}

174    impl Proxy {
175        /// The plaintext listener the load balancer forwards to.
176        fn public(&self) -> Channel {
177            Endpoint::from_shared(format!("http://{}", self.public_addr))
178                .unwrap()
179                .connect_lazy()
180        }
181
182        /// The node listener, presenting `key`'s certificate.
183        fn node_channel(&self, key: &ed25519_dalek::SigningKey) -> Channel {
184            let tls = ClientTlsConfig::new()
185                .ca_certificate(Certificate::from_pem(&self.cert.cert_pem))
186                .domain_name("localhost")
187                .identity(node_identity(key));
188            Endpoint::from_shared(format!("https://{}", self.node_server.local_addr()))
189                .unwrap()
190                .tls_config(tls)
191                .unwrap()
192                .connect_lazy()
193        }
194
195        fn node(&self, key: &ed25519_dalek::SigningKey) -> GuardianServiceClient<Channel> {
196            GuardianServiceClient::new(self.node_channel(key))
197        }
198    }

200    /// The real router over a stub guardian, on both listeners, with
201    /// `member_key` on the allowlist.
202    async fn spawn_proxy() -> Proxy {
203        let (stub, backend) = spawn_stub().await;
204        let metrics = Arc::new(ProxyMetrics::new());
205        let roster = Arc::new(RosterCache::new(MemStore::default()));
206        let guardian = CachingGuardianGrpc::new(
207            Forwarding::new(
208                backend.clone(),
209                backend.clone(),
210                roster.clone(),
211                gate_over(&[]),
212            ),
213            WidLogIndex::ready_for_tests(MemStore::default(), metrics.clone()).await,
214            metrics.clone(),
215        );
216        let relay = Relay::new(backend.clone(), roster);
217        let (reporter, health) = tonic_health::server::health_reporter();
218        reporter
219            .set_service_status("", tonic_health::ServingStatus::Serving)
220            .await;
221        let info = public::info::InfoState::new(
222            GuardianServiceClient::new(backend),
223            Duration::from_secs(1),
224        );
225        let allowlist = Arc::new(MemberAllowlist::new(metrics.clone()));
226        allowlist.store(snapshot(&[&member_key()]));
227        let app = router(
228            guardian,
229            relay,
230            health,
231            info,
232            Arc::new(MemberGate::new(allowlist, metrics.clone())),
233        );
234
235        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
236        let public_addr = listener.local_addr().unwrap();
237        tokio::spawn({
238            let app = app.clone();
239            async move { axum::serve(listener, app).await.unwrap() }
240        });
241        let cert = test_cert();
242        let served = tls::ServerCert::load(&cert.source, &metrics).await.unwrap();
243        let node_server = sui_http::Builder::new()
244            .tls_config(tls::server_config(served).unwrap())
245            .serve("127.0.0.1:0", app.clone())
246            .unwrap();
247        Proxy {
248            stub,
249            app,
250            public_addr,
251            node_server,
252            cert,
253        }
254    }

256    #[tokio::test]
257    async fn node_rpcs_need_the_node_listener_even_for_a_cached_wid() {
258        let proxy = spawn_proxy().await;
259        let mut public = GuardianServiceClient::new(proxy.public());
260
261        let refused = public
262            .standard_withdrawal(mock_request([0x11; 32], 0))
263            .await
264            .unwrap_err();
265        assert_eq!(refused.code(), Code::Unauthenticated);
266        assert_eq!(
267            proxy.stub.standard_withdrawal_calls.load(Ordering::SeqCst),
268            0
269        );
270
271        // A member's call on the node listener reaches the guardian and fills
272        // the wid cache.
273        proxy
274            .node(&member_key())
275            .standard_withdrawal(mock_request([0x11; 32], 0))
276            .await
277            .unwrap();
278        assert_eq!(
279            proxy.stub.standard_withdrawal_calls.load(Ordering::SeqCst),
280            1
281        );
282
283        // The gate runs before the cache, so the cached wid is no way in.
284        let replay = public
285            .standard_withdrawal(mock_request([0x11; 32], 1))
286            .await
287            .unwrap_err();
288        assert_eq!(replay.code(), Code::Unauthenticated);
289
290        let update = public
291            .update_committee_chain(proto::UpdateCommitteeChainRequest::default())
292            .await
293            .unwrap_err();
294        assert_eq!(update.code(), Code::Unauthenticated);
295        let update = public
296            .update_committee(proto::SignedCommitteeTransition::default())
297            .await
298            .unwrap_err();
299        assert_eq!(update.code(), Code::Unauthenticated);
300    }

393    async fn public_routes_stay_open_on_the_public_listener() {
394        let proxy = spawn_proxy().await;
395        let channel = proxy.public();
396        let mut client = GuardianServiceClient::new(channel.clone());
397
398        client
399            .get_guardian_info(proto::GetGuardianInfoRequest {})
400            .await
401            .unwrap();
402        assert_eq!(proxy.stub.get_guardian_info_calls.load(Ordering::SeqCst), 1);
403
404        // Reaches the KP signature check, which rejects the empty request.
405        let unsigned = client
406            .confirm_ceremony(proto::SignedCeremonyConfirmationRequest::default())
407            .await
408            .unwrap_err();
409        assert_eq!(unsigned.code(), Code::InvalidArgument);
410
411        client
412            .get_attested_guardian_info(attested(proto::GetAttestedGuardianInfoRequest {}))
413            .await
414            .unwrap();
415        GuardianRelayServiceClient::new(channel.clone())
416            .get_provisioning_target_info(attested(proto::GetProvisioningTargetInfoRequest {}))
417            .await
418            .unwrap();
419        assert_eq!(

421                .stub
422                .get_attested_guardian_info_calls
423                .load(Ordering::SeqCst),
424            2
425        );

427        let health = HealthClient::new(channel)
428            .check(HealthCheckRequest {
429                service: String::new(),

431            .await
432            .unwrap()
433            .into_inner();

443    async fn http_status_routes_stay_public_and_the_rest_is_refused() {
444        let proxy = spawn_proxy().await;
445        let status = |method: Method, uri: &str| {
446            let request = axum::http::Request::builder()
447                .method(method)
448                .uri(uri)
449                .header("origin", "https://hashi.example")
450                .header("access-control-request-method", "GET")
451                .body(Body::empty())
452                .unwrap();
453            let app = proxy.app.clone();
454            async move { app.oneshot(request).await.unwrap().status() }
455        };
456
457        assert_eq!(status(Method::GET, "/health").await, StatusCode::OK);
458        assert_eq!(status(Method::OPTIONS, "/info").await, StatusCode::OK);
459        assert_ne!(status(Method::GET, "/info").await, StatusCode::FORBIDDEN);
460        assert_eq!(status(Method::GET, "/metrics").await, StatusCode::FORBIDDEN);
461    }
