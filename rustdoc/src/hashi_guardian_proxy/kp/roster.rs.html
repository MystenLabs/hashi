4//! The proxy's KP roster, read from the guardian's S3 share log. A ceremony
5//! commits who holds shares — every encrypted share is labeled with its
6//! recipient's PGP fingerprint — so the latest share log IS the authorization
7//! roster. The read is deliberately unverified: the bucket only admits enclave
8//! writes and this gate is DoS-tier, with the enclave still verifying every
9//! KP-signed request against its own roster.
10//!
11//! Shares are stored at
12//! `kp-shares/{sharing_seq:020}/{cert_seq:020}.json` (`KpShareState`).
13//! The reader parses only the fields it needs.
14//!
15//! Which `sharing_seq` is current comes from `ceremony/`, not from the newest
16//! `kp-shares/` dir: a ceremony publishes its shares before its `ceremony/`
17//! record, so an aborted one leaves an orphan dir that was never authorized.
18//! This is the resolution the enclave performs
19//! (`hashi-guardian::s3_reader::read_latest_ceremony_state`), and the relay has
20//! to agree with it or it rejects the very KPs the enclave would accept.
21//!
22//! A ceremony commits only once every KP has confirmed it, so the KPs
23//! confirming one are named by its session's proposal instead,
24//! `kp-shares/proposed/{session_id}.json` (`CeremonyProposal`).
25

43/// The committed roster only changes at a ceremony, a re-deal, or a cert
44/// rotation — and rotation invalidates explicitly — so a minute of staleness
45/// costs nothing and bounds S3 reads under submission spam.
46const ROSTER_TTL: Duration = Duration::from_secs(60);
47/// A miss — no roster yet, or a signer the cached roster does not name — may
48/// be a share set the guardian committed since the last read: `SetupNewKey`
49/// and `RotateKpSet` run over the operator's tunnel, so nothing invalidates
50/// this cache for them. A miss re-reads at most this often, so a new roster
51/// admits its KPs on their first call while unrostered callers cannot force
52/// an S3 read each.
53const MISS_REFRESH_INTERVAL: Duration = Duration::from_secs(5);
54
55struct Cached {
56    at: Instant,
57    roster: Option<Arc<Vec<Fingerprint>>>,
58}
59
60#[derive(Default)]
61struct State {
62    cached: Option<Cached>,
63    miss_refreshed_at: Option<Instant>,
64}
65
66impl State {
67    fn miss_refresh_due(&self) -> bool {
68        self.miss_refreshed_at
69            .is_none_or(|at| at.elapsed() >= MISS_REFRESH_INTERVAL)
70    }
71}

73/// TTL-cached view of [`latest_kp_roster`]. The mutex is held across the fetch,
74/// so concurrent misses collapse into one S3 read.
75pub struct RosterCache<L> {
76    store: L,
77    state: Mutex<State>,
78}
79
80impl<L: LogStore> RosterCache<L> {
81    pub fn new(store: L) -> Self {
82        Self {
83            store,
84            state: Mutex::new(State::default()),
85        }
86    }
87
88    /// Drop the cached roster so the next read observes a just-committed change.
89    pub async fn invalidate(&self) {
90        self.state.lock().await.cached = None;
91    }
92
93    /// Admit `signer` only if the latest committed share set names it. No share
94    /// log yet is a definitive "not ready"; a read error is transient.
95    pub async fn authorize(&self, signer: &Fingerprint) -> Result<(), Status> {
96        let mut state = self.state.lock().await;
97        let fresh = state
98            .cached
99            .as_ref()
100            .filter(|cached| cached.at.elapsed() < ROSTER_TTL)
101            .map(|cached| cached.roster.clone());
102        let (mut roster, just_read) = match fresh {
103            Some(roster) => (roster, false),
104            None => (self.read(&mut state).await?, true),
105        };
106        let names_signer = |roster: &Option<Arc<Vec<Fingerprint>>>| {
107            roster
108                .as_deref()
109                .is_some_and(|roster| roster.contains(signer))
110        };
111        if !names_signer(&roster) && !just_read && state.miss_refresh_due() {
112            state.miss_refreshed_at = Some(Instant::now());
113            roster = self.read(&mut state).await?;
114        }
115        match roster {
116            Some(roster) if roster.contains(signer) => Ok(()),
117            Some(_) => Err(Status::permission_denied(format!(
118                "signer {signer} is not in the ceremony's committed KP roster"
119            ))),
120            None => Err(Status::failed_precondition(
121                "no KP share log in the guardian bucket; run the key ceremony first",
122            )),
123        }
124    }
125
126    /// Admit `signer` to confirm the ceremony `session_id` proposed, only if
127    /// that proposal names it. Read on every call: a ceremony takes one
128    /// confirmation per KP.
129    pub async fn authorize_confirmation(
130        &self,
131        session_id: &SessionID,
132        signer: &Fingerprint,
133    ) -> Result<(), Status> {
134        let roster = proposed_kp_roster(&self.store, session_id)
135            .await
136            .map_err(|e| {
137                warn!(error = %format!("{e:#}"), "KP ceremony proposal read failed");
138                Status::unavailable("KP roster unavailable; retry")
139            })?;
140        match roster {
141            Some(roster) if roster.contains(signer) => Ok(()),
142            Some(_) => Err(Status::permission_denied(format!(
143                "signer {signer} is not in the KP roster guardian session {session_id} proposed"
144            ))),
145            None => Err(Status::failed_precondition(format!(
146                "guardian session {session_id} has proposed no ceremony; run the operator's \
147                 ceremony step first"
148            ))),
149        }
150    }
151
152    /// One read of the share log, cached whatever it finds.
153    async fn read(&self, state: &mut State) -> Result<Option<Arc<Vec<Fingerprint>>>, Status> {
154        let roster = latest_kp_roster(&self.store)
155            .await
156            .map_err(|e| {
157                warn!(error = %format!("{e:#}"), "KP roster read failed");
158                Status::unavailable("KP roster unavailable; retry")
159            })?
160            .map(Arc::new);
161        state.cached = Some(Cached {
162            at: Instant::now(),
163            roster: roster.clone(),
164        });
165        Ok(roster)
166    }
167}

226/// Recipient fingerprints of the latest committed share set. `Ok(None)` means
227/// no share log exists anywhere (no ceremony yet) — a definitive miss; any
228/// `Err` is indeterminate and the caller must fail closed.
229pub async fn latest_kp_roster<L: LogStore>(log: &L) -> anyhow::Result<Option<Vec<Fingerprint>>> {
230    let Some(key) = latest_share_log_key(log).await? else {
231        return Ok(None);
232    };
233    let bytes = log.get(&key).await?;
234    let roster = parse_roster(&bytes).with_context(|| format!("parse share log {key}"))?;
235    Ok(Some(roster))
236}
237
238/// Recipient fingerprints of the ceremony `session_id` proposed. `Ok(None)`
239/// means that session proposed none; any `Err` is indeterminate and the caller
240/// must fail closed.
241pub async fn proposed_kp_roster<L: LogStore>(
242    log: &L,
243    session_id: &SessionID,
244) -> anyhow::Result<Option<Vec<Fingerprint>>> {
245    let key = CeremonyProposalLogMessage::object_key(session_id);
246    if !log.list_keys(&key).await?.contains(&key) {
247        return Ok(None);
248    }
249    let bytes = log.get(&key).await?;
250    let roster =
251        parse_proposed_roster(&bytes).with_context(|| format!("parse ceremony proposal {key}"))?;
252    Ok(Some(roster))
253}

255/// Key of the share-state record the latest completed ceremony committed: the
256/// lex-greatest `cert_seq` under that ceremony's `sharing_seq` dir. Zero-padded
257/// seqs make lex order the seq order throughout.
258async fn latest_share_log_key<L: LogStore>(log: &L) -> anyhow::Result<Option<String>> {
259    let Some(ceremony_key) = log
260        .list_keys(&CeremonyLogMessage::object_key_dir())
261        .await?
262        .into_iter()
263        .max()
264    else {
265        return Ok(None);
266    };
267    let sharing_seq = ceremony_sharing_seq(&ceremony_key)?;
268    log.list_keys(&KpShareStateLogMessage::object_key_dir(sharing_seq))
269        .await?
270        .into_iter()
271        .max()
272        .map(Some)
273        // Shares are written first, so a ceremony without them is a corrupt
274        // log, not a not-ready one. Fail closed rather than fall back to a
275        // roster this ceremony did not authorize.
276        .with_context(|| format!("ceremony {ceremony_key} has no kp-shares log"))
277}

279/// The `sharing_seq` a `ceremony/{sharing_seq:020}.json` key records.
280/// The canonical padding is required, not just parsed: lex order over these
281/// keys is the seq order only while every one of them pads to the same width.
282fn ceremony_sharing_seq(key: &str) -> anyhow::Result<u64> {
283    key.strip_prefix(&CeremonyLogMessage::object_key_dir())
284        .and_then(|name| name.strip_suffix(".json"))
285        .filter(|seq| seq.len() == 20 && seq.bytes().all(|b| b.is_ascii_digit()))
286        .with_context(|| format!("ceremony key {key:?} has no sharing_seq"))?
287        .parse()
288        .with_context(|| format!("ceremony key {key:?} has an out-of-range sharing_seq"))
289}

291/// Just the fields the roster needs, tolerant of everything else. Any record
292/// under the share prefix carries a `KpShareState` message; anything else
293/// is a poisoned log and fails closed upstream.
294#[derive(Deserialize)]
295struct ShareLogRecord {
296    message: ShareLogMessage,
297}
298
299#[derive(Deserialize)]
300enum ShareLogMessage {
301    KpShareState { encrypted_shares: Vec<LabeledShare> },
302}
303
304/// Each encrypted share contains exactly one required recipient fingerprint.
305/// Removed map-shaped payloads intentionally fail deserialization.
306#[derive(Deserialize)]
307struct LabeledShare {
308    recipient_fingerprint: String,
309}
310
311/// Like [`ShareLogRecord`], for the one message a proposal key carries.
312#[derive(Deserialize)]
313struct ProposalLogRecord {
314    message: ProposalLogMessage,
315}
316
317#[derive(Deserialize)]
318enum ProposalLogMessage {
319    CeremonyProposal { encrypted_shares: Vec<LabeledShare> },
320}
321
322fn parse_roster(bytes: &[u8]) -> anyhow::Result<Vec<Fingerprint>> {
323    let record: ShareLogRecord = serde_json::from_slice(bytes)?;
324    let ShareLogMessage::KpShareState { encrypted_shares } = record.message;
325    recipient_fingerprints(&encrypted_shares)
326}
327
328fn parse_proposed_roster(bytes: &[u8]) -> anyhow::Result<Vec<Fingerprint>> {
329    let record: ProposalLogRecord = serde_json::from_slice(bytes)?;
330    let ProposalLogMessage::CeremonyProposal { encrypted_shares } = record.message;
331    recipient_fingerprints(&encrypted_shares)
332}
333
334fn recipient_fingerprints(shares: &[LabeledShare]) -> anyhow::Result<Vec<Fingerprint>> {
335    shares
336        .iter()
337        .map(|share| parse_recipient_fingerprint(&share.recipient_fingerprint))
338        .collect()
339}
340
341fn parse_recipient_fingerprint(label: &str) -> anyhow::Result<Fingerprint> {
342    label
343        .parse::<Fingerprint>()
344        .ok()
345        // Sequoia parses odd-sized hex into `Fingerprint::Unknown` rather than
346        // failing; only real v4/v6 shapes can name a KP cert.
347        .filter(|fp| matches!(fp, Fingerprint::V4(_) | Fingerprint::V6(_)))
348        .with_context(|| format!("share label {label:?} is not a PGP fingerprint"))
349}

581    /// `SetupNewKey` and `RotateKpSet` commit a new roster over the operator's
582    /// tunnel, never through the proxy, so nothing invalidates the cache: a KP

608    #[tokio::test(start_paused = true)]
609    async fn miss_refreshes_are_bounded() {
610        let store = MemStore::default();
611        seed_roster(&store, 0, &[FP_A]);
612        let cache = RosterCache::new(store);
613        cache.authorize(&fp(FP_A)).await.unwrap();
614        let reads = || cache.store.list_calls.load(Ordering::SeqCst);
615
616        // The first unrostered signer costs one re-read; the next does not.
617        let before = reads();
618        let err = cache.authorize(&fp(FP_B)).await.unwrap_err();
619        assert_eq!(err.code(), tonic::Code::PermissionDenied);
620        assert!(reads() > before);
621        let before = reads();
622        let err = cache.authorize(&fp(FP_B)).await.unwrap_err();
623        assert_eq!(err.code(), tonic::Code::PermissionDenied);
624        assert_eq!(reads(), before);
625
626        // The budget renews after the interval.
627        tokio::time::advance(MISS_REFRESH_INTERVAL).await;
628        let err = cache.authorize(&fp(FP_B)).await.unwrap_err();
629        assert_eq!(err.code(), tonic::Code::PermissionDenied);
630        assert!(reads() > before);
631    }
