4//! What key provisioners call: the provisioning [`relay`], and the KP-signed
5//! `ConfirmCeremony` and `ProvisionerRotateCert`. Each is checked against the
6//! ceremony's committed [`roster`] before it reaches the enclave, except
7//! `ConfirmCeremony`, which is what commits one and is checked against its
8//! proposal.
9

11pub mod roster;

22pub fn parse<T, P>(request: &P) -> Result<KpSigned<T>, Status>
23where
24    T: KpSigningIntent,
25    P: Clone,
26    KpSigned<T>: TryFrom<P, Error = GuardianError>,
27{
28    KpSigned::<T>::try_from(request.clone())
29        .map_err(|e| Status::invalid_argument(format!("malformed request: {e}")))
30}
31
32/// Admission control only: the enclave repeats both checks. Signature first
33/// because it needs no roster read.
34pub async fn admit<'a, T, L>(
35    roster: &RosterCache<L>,
36    signed: &'a KpSigned<T>,
37) -> Result<&'a T, Status>
38where
39    T: KpSigningIntent,
40    L: LogStore,
41{
42    let payload = signed
43        .verify_signature()
44        .map_err(|e| Status::unauthenticated(e.to_string()))?;
45    roster.authorize(&signed.signer_fingerprint()).await?;
46    Ok(payload)
47}
48
49/// [`admit`] for a ceremony confirmation, whose signer the ceremony's own
50/// session proposed and no roster has committed yet.
51pub async fn admit_confirmation<L: LogStore>(
52    roster: &RosterCache<L>,
53    signed: &KpSigned<CeremonyConfirmationRequest>,
54) -> Result<(), Status> {
55    let confirmation = signed
56        .verify_signature()
57        .map_err(|e| Status::unauthenticated(e.to_string()))?;
58    roster
59        .authorize_confirmation(
60            confirmation.expected_session_id(),
61            &signed.signer_fingerprint(),
62        )
63        .await
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69    use crate::kp::roster::test_utils::seed_proposal;
70    use crate::log_store::test_store::MemStore;
71    use hashi_types::guardian::test_utils::mock_attested_kp_keypair;
72    use hashi_types::pgp::test_utils::sign_detached_in_process;
73
74    /// On a first deploy no roster is committed while the KPs confirm, so
75    /// `admit` refuses them all: the proposal is what lets the ceremony finish.
76    #[tokio::test]
77    async fn a_confirmation_is_admitted_on_its_sessions_proposal() {
78        let (cert, secret_armored) = mock_attested_kp_keypair();
79        let fingerprint = cert.fingerprint().to_hex();
80        let confirmation = CeremonyConfirmationRequest::new("sess-a".into(), [7u8; 32]);
81        let signature =
82            sign_detached_in_process(&secret_armored, &KpSigned::signed_bytes(&confirmation));
83        let signed = KpSigned::from_parts(confirmation, cert.clone(), signature.clone());
84
85        let roster = RosterCache::new(MemStore::default());
86        let err = admit(&roster, &signed).await.unwrap_err();
87        assert_eq!(err.code(), tonic::Code::FailedPrecondition);
88        let err = admit_confirmation(&roster, &signed).await.unwrap_err();
89        assert_eq!(err.code(), tonic::Code::FailedPrecondition);
90
91        let store = MemStore::default();
92        seed_proposal(&store, "sess-a", &[&fingerprint]);
93        let roster = RosterCache::new(store);
94        admit_confirmation(&roster, &signed).await.unwrap();
95
96        // The signature is checked before the proposal is read.
97        let other_session = CeremonyConfirmationRequest::new("sess-b".into(), [7u8; 32]);
98        let forged = KpSigned::from_parts(other_session, cert, signature);
99        let err = admit_confirmation(&roster, &forged).await.unwrap_err();
100        assert_eq!(err.code(), tonic::Code::Unauthenticated);
101    }
102}
