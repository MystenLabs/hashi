4//! The proxy's KP roster, read from the guardian's S3 share log. A ceremony
5//! commits who holds shares — every encrypted share is labeled with its
6//! recipient's PGP fingerprint — so the latest share log IS the authorization
7//! roster. The read is deliberately unverified: the bucket only admits enclave
8//! writes and this gate is DoS-tier, with the enclave still verifying every
9//! KP-signed request against its own roster.
10//!
11//! Two layouts are in flight (#779 migrates the first to the second):
12//!   `shares/{sharing_seq:020}-{session}.json` (message `Shares`)
13//!   `kp-shares/{sharing_seq:020}/{cert_seq:020}-{session}.json` (`KpShareState`)
14//! The reader prefers `kp-shares/` and parses only the fields it needs while
15//! retaining the already-scalar legacy `shares/` fallback.
16//!
17//! Which `sharing_seq` is current comes from `ceremony/`, not from the newest
18//! `kp-shares/` dir: a ceremony publishes its shares before its `ceremony/`
19//! record, so an aborted one leaves an orphan dir that was never authorized.
20//! This is the resolution the enclave performs
21//! (`hashi-guardian::s3_reader::read_latest_ceremony_state`), and the relay has
22//! to agree with it or it rejects the very KPs the enclave would accept.
23

39/// The pre-#779 layout, read-only: hashi-types no longer models it, so unlike
40/// the current prefixes this one has no constructor to borrow.
41const LEGACY_SHARES_PREFIX: &str = "shares/";
42
43/// The committed roster only changes at a ceremony, a re-deal, or a cert
44/// rotation — and rotation invalidates explicitly — so a minute of staleness
45/// costs nothing and bounds S3 reads under submission spam.
46const ROSTER_TTL: Duration = Duration::from_secs(60);
47/// A miss — no roster yet, or a signer the cached roster does not name — may
48/// be a share set the guardian committed since the last read: `SetupNewKey`
49/// and `RotateKpSet` run over the operator's tunnel, so nothing invalidates
50/// this cache for them. A miss re-reads at most this often, so a new roster
51/// admits its KPs on their first call while unrostered callers cannot force
52/// an S3 read each.
53const MISS_REFRESH_INTERVAL: Duration = Duration::from_secs(5);
54
55struct Cached {
56    at: Instant,
57    roster: Option<Arc<Vec<Fingerprint>>>,
58}
59
60#[derive(Default)]
61struct State {
62    cached: Option<Cached>,
63    miss_refreshed_at: Option<Instant>,
64}
65
66impl State {
67    fn miss_refresh_due(&self) -> bool {
68        self.miss_refreshed_at
69            .is_none_or(|at| at.elapsed() >= MISS_REFRESH_INTERVAL)
70    }
71}
72
73/// TTL-cached view of [`latest_kp_roster`]. The mutex is held across the fetch,
74/// so concurrent misses collapse into one S3 read.
75pub struct RosterCache<L> {
76    store: L,
77    state: Mutex<State>,
78}
79
80impl<L: LogStore> RosterCache<L> {
81    pub fn new(store: L) -> Self {
82        Self {
83            store,
84            state: Mutex::new(State::default()),
85        }
86    }
87
88    /// Drop the cached roster so the next read observes a just-committed change.
89    pub async fn invalidate(&self) {
90        self.state.lock().await.cached = None;
91    }
92
93    /// Admit `signer` only if the latest committed share set names it. No share
94    /// log yet is a definitive "not ready"; a read error is transient.
95    pub async fn authorize(&self, signer: &Fingerprint) -> Result<(), Status> {
96        let mut state = self.state.lock().await;
97        let fresh = state
98            .cached
99            .as_ref()
100            .filter(|cached| cached.at.elapsed() < ROSTER_TTL)
101            .map(|cached| cached.roster.clone());
102        let (mut roster, just_read) = match fresh {
103            Some(roster) => (roster, false),
104            None => (self.read(&mut state).await?, true),
105        };
106        let names_signer = |roster: &Option<Arc<Vec<Fingerprint>>>| {
107            roster
108                .as_deref()
109                .is_some_and(|roster| roster.contains(signer))
110        };
111        if !names_signer(&roster) && !just_read && state.miss_refresh_due() {
112            state.miss_refreshed_at = Some(Instant::now());
113            roster = self.read(&mut state).await?;
114        }
115        match roster {
116            Some(roster) if roster.contains(signer) => Ok(()),
117            Some(_) => Err(Status::permission_denied(format!(
118                "signer {signer} is not in the ceremony's committed KP roster"
119            ))),
120            None => Err(Status::failed_precondition(
121                "no KP share log in the guardian bucket; run the key ceremony first",
122            )),
123        }
124    }
125
126    /// One read of the share log, cached whatever it finds.
127    async fn read(&self, state: &mut State) -> Result<Option<Arc<Vec<Fingerprint>>>, Status> {
128        let roster = latest_kp_roster(&self.store)
129            .await
130            .map_err(|e| {
131                warn!(error = %format!("{e:#}"), "KP roster read failed");
132                Status::unavailable("KP roster unavailable; retry")
133            })?
134            .map(Arc::new);
135        state.cached = Some(Cached {
136            at: Instant::now(),
137            roster: roster.clone(),
138        });
139        Ok(roster)
140    }
141}

182/// Recipient fingerprints of the latest committed share set. `Ok(None)` means
183/// no share log exists anywhere (no ceremony yet) — a definitive miss; any
184/// `Err` is indeterminate and the caller must fail closed.
185pub async fn latest_kp_roster<L: LogStore>(log: &L) -> anyhow::Result<Option<Vec<Fingerprint>>> {
186    let Some(key) = latest_share_log_key(log).await? else {
187        return Ok(None);
188    };
189    let bytes = log.get(&key).await?;
190    let roster = parse_roster(&bytes).with_context(|| format!("parse share log {key}"))?;
191    Ok(Some(roster))
192}
193
194/// Key of the share-state record the latest completed ceremony committed: the
195/// lex-greatest `cert_seq` under that ceremony's `sharing_seq` dir. Zero-padded
196/// seqs make lex order the seq order throughout.
197///
198/// Pre-#779 buckets have no `ceremony/` records at all; there the lex-greatest
199/// flat `shares/` key is the whole story.
200async fn latest_share_log_key<L: LogStore>(log: &L) -> anyhow::Result<Option<String>> {
201    let Some(ceremony_key) = log
202        .list_keys(&CeremonyLogMessage::object_key_dir())
203        .await?
204        .into_iter()
205        .max()
206    else {
207        return Ok(log.list_keys(LEGACY_SHARES_PREFIX).await?.into_iter().max());
208    };
209    let sharing_seq = ceremony_sharing_seq(&ceremony_key)?;
210    log.list_keys(&KpShareStateLogMessage::object_key_dir(sharing_seq))
211        .await?
212        .into_iter()
213        .max()
214        .map(Some)
215        // Shares are written first, so a ceremony without them is a corrupt
216        // log, not a not-ready one. Fail closed rather than fall back to a
217        // roster this ceremony did not authorize.
218        .with_context(|| format!("ceremony {ceremony_key} has no kp-shares log"))
219}
220
221/// The `sharing_seq` a `ceremony/{sharing_seq:020}-{session}.json` key records.
222/// The canonical padding is required, not just parsed: lex order over these
223/// keys is the seq order only while every one of them pads to the same width.
224fn ceremony_sharing_seq(key: &str) -> anyhow::Result<u64> {
225    key.strip_prefix(&CeremonyLogMessage::object_key_dir())
226        .and_then(|name| name.split('-').next())
227        .filter(|seq| seq.len() == 20 && seq.bytes().all(|b| b.is_ascii_digit()))
228        .with_context(|| format!("ceremony key {key:?} has no sharing_seq"))?
229        .parse()
230        .with_context(|| format!("ceremony key {key:?} has an out-of-range sharing_seq"))
231}

233/// Just the fields the roster needs, tolerant of everything else. Any record
234/// under a share prefix carries one of these two message shapes; anything else
235/// is a poisoned log and fails closed upstream.
236#[derive(Deserialize)]
237struct ShareLogRecord {
238    message: ShareLogMessage,
239}
240
241#[derive(Deserialize)]
242enum ShareLogMessage {
243    Shares(ShareState),
244    KpShareState(ShareState),
245}
246
247#[derive(Deserialize)]
248struct ShareState {
249    encrypted_shares: Vec<LabeledShare>,
250}
251
252/// Both the legacy `Shares` envelope and versioned `KpShareState` envelope
253/// contain exactly one required recipient fingerprint per encrypted share.
254/// Removed map-shaped V2 payloads intentionally fail deserialization.
255#[derive(Deserialize)]
256struct LabeledShare {
257    recipient_fingerprint: String,
258}
259
260fn parse_roster(bytes: &[u8]) -> anyhow::Result<Vec<Fingerprint>> {
261    let record: ShareLogRecord = serde_json::from_slice(bytes)?;
262    let (ShareLogMessage::Shares(state) | ShareLogMessage::KpShareState(state)) = record.message;
263    state
264        .encrypted_shares
265        .iter()
266        .map(|share| parse_recipient_fingerprint(&share.recipient_fingerprint))
267        .collect()
268}
269
270fn parse_recipient_fingerprint(label: &str) -> anyhow::Result<Fingerprint> {
271    label
272        .parse::<Fingerprint>()
273        .ok()
274        // Sequoia parses odd-sized hex into `Fingerprint::Unknown` rather than
275        // failing; only real v4/v6 shapes can name a KP cert.
276        .filter(|fp| matches!(fp, Fingerprint::V4(_) | Fingerprint::V6(_)))
277        .with_context(|| format!("share label {label:?} is not a PGP fingerprint"))
278}

637    #[tokio::test(start_paused = true)]
638    async fn miss_refreshes_are_bounded() {
639        let store = MemStore::default();
640        seed_roster(&store, 0, &[FP_A]);
641        let cache = RosterCache::new(store);
642        cache.authorize(&fp(FP_A)).await.unwrap();
643        let reads = || cache.store.list_calls.load(Ordering::SeqCst);
644
645        // The first unrostered signer costs one re-read; the next does not.
646        let before = reads();
647        let err = cache.authorize(&fp(FP_B)).await.unwrap_err();
648        assert_eq!(err.code(), tonic::Code::PermissionDenied);
649        assert!(reads() > before);
650        let before = reads();
651        let err = cache.authorize(&fp(FP_B)).await.unwrap_err();
652        assert_eq!(err.code(), tonic::Code::PermissionDenied);
653        assert_eq!(reads(), before);
654
655        // The budget renews after the interval.
656        tokio::time::advance(MISS_REFRESH_INTERVAL).await;
657        let err = cache.authorize(&fp(FP_B)).await.unwrap_err();
658        assert_eq!(err.code(), tonic::Code::PermissionDenied);
659        assert!(reads() > before);
660    }
