4//! Environment-driven proxy configuration (matches the guardian's minimal,
5//! env-only config style). TLS for node traffic terminates at the fronting
6//! load balancer, so the proxy itself serves plaintext h2c.
7
8use std::net::SocketAddr;
9use std::time::Duration;
10
11use anyhow::Context;
12use anyhow::Result;
13use bitcoin::Network;
14
15use crate::remote_write;
16use crate::remote_write::RemoteWriteConfig;
17
18pub struct Config {
19    /// gRPC endpoint of the enclave guardian to forward to, e.g.
20    /// `http://10.0.1.20:3000` (`GUARDIAN_BACKEND_URL`, required).
21    pub backend_url: String,
22    /// gRPC endpoint of a standby guardian being armed (`STANDBY_GUARDIAN_URL`,
23    /// optional). When set, the relay — share submissions and
24    /// `GetProvisioningTargetInfo` — targets it while node-facing forwarding
25    /// stays on the active backend; unset, the relay provisions the active
26    /// backend (first deploy).
27    pub standby_backend_url: Option<String>,
28    /// Address the proxy serves everything on — gRPC (forwarder + relay + health)
29    /// and the HTTP `/info` + `/health` (`PROXY_LISTEN_ADDR`, default `0.0.0.0:3000`).
30    pub listen_addr: SocketAddr,
31    /// Address the prometheus `/metrics` endpoint listens on
32    /// (`METRICS_LISTEN_ADDR`, default `0.0.0.0:9184`).
33    pub metrics_listen_addr: SocketAddr,
34    /// TTL for the single-slot `/info` response cache
35    /// (`INFO_CACHE_TTL_MS`, default 1000).
36    pub info_cache_ttl: Duration,
37    /// TCP connect timeout to the backend
38    /// (`GUARDIAN_CONNECT_TIMEOUT_SECS`, default 5).
39    pub connect_timeout: Duration,
40    /// HTTP/2 keepalive ping interval to the backend
41    /// (`GUARDIAN_KEEPALIVE_SECS`, default 5).
42    pub keepalive_interval: Duration,
43    /// The guardian's S3 log bucket, read as the wid cache's durable tier
44    /// (`GUARDIAN_LOG_BUCKET` + `GUARDIAN_LOG_REGION`, required). Credentials
45    /// come from the AWS default provider chain.
46    pub log_bucket: String,
47    pub log_region: String,
48    /// BTC network the guardian signs for (`BTC_NETWORK`, required:
49    /// bitcoin|testnet|signet|regtest). Must match the guardian's config; used
50    /// to recompute sighashes when verifying a log replay.
51    pub btc_network: Network,
52    /// Push metrics to a Prometheus remote-write endpoint; `None` leaves them
53    /// on `/metrics`, which nothing can scrape (`MIMIR_URL`, `MIMIR_USERNAME`
54    /// default `incoming_metrics`, `MIMIR_PASSWORD`, `MIMIR_PUSH_INTERVAL_SECS`
55    /// default 60, `MIMIR_EXTERNAL_LABELS` comma-separated `k=v`).
56    pub remote_write: Option<RemoteWriteConfig>,
57}
58
59impl Config {
60    pub fn from_env() -> Result<Self> {
61        let backend_url = std::env::var("GUARDIAN_BACKEND_URL")
62            .context("GUARDIAN_BACKEND_URL must be set (gRPC endpoint of the enclave guardian)")?;
63        let standby_backend_url = std::env::var("STANDBY_GUARDIAN_URL")
64            .ok()
65            .filter(|url| !url.is_empty());
66        let listen_addr = std::env::var("PROXY_LISTEN_ADDR")
67            .unwrap_or_else(|_| "0.0.0.0:3000".to_string())
68            .parse()
69            .context("PROXY_LISTEN_ADDR must be a valid socket address")?;
70        let metrics_listen_addr = std::env::var("METRICS_LISTEN_ADDR")
71            .unwrap_or_else(|_| "0.0.0.0:9184".to_string())
72            .parse()
73            .context("METRICS_LISTEN_ADDR must be a valid socket address")?;
74        let info_cache_ttl = Duration::from_millis(parse_env_u64("INFO_CACHE_TTL_MS", 1000)?);
75        let connect_timeout =
76            Duration::from_secs(parse_env_u64("GUARDIAN_CONNECT_TIMEOUT_SECS", 5)?);
77        let keepalive_interval = Duration::from_secs(parse_env_u64("GUARDIAN_KEEPALIVE_SECS", 5)?);
78        let log_bucket = std::env::var("GUARDIAN_LOG_BUCKET")
79            .context("GUARDIAN_LOG_BUCKET must be set (the guardian's S3 log bucket)")?;
80        let log_region = std::env::var("GUARDIAN_LOG_REGION")
81            .context("GUARDIAN_LOG_REGION must be set (region of the guardian's S3 log bucket)")?;
82        let btc_network = std::env::var("BTC_NETWORK")
83            .context("BTC_NETWORK must be set (bitcoin|testnet|signet|regtest)")?
84            .parse()
85            .context("BTC_NETWORK must be one of bitcoin|testnet|signet|regtest")?;
86        let remote_write = match std::env::var("MIMIR_URL").ok().filter(|u| !u.is_empty()) {
87            None => None,
88            Some(url) => {
89                let username = std::env::var("MIMIR_USERNAME")
90                    .unwrap_or_else(|_| "incoming_metrics".to_string());
91                let password = std::env::var("MIMIR_PASSWORD")
92                    .context("MIMIR_PASSWORD must be set when MIMIR_URL is")?;
93                anyhow::ensure!(
94                    !username.is_empty() && !username.contains(':') && !password.is_empty(),
95                    "MIMIR_USERNAME and MIMIR_PASSWORD must be non-empty (no ':' in the name)"
96                );
97                // Zero panics `tokio::time::interval` (a panic aborts this binary);
98                // huge values overflow the deadline arithmetic.
99                let interval = Duration::from_secs(parse_env_u64("MIMIR_PUSH_INTERVAL_SECS", 60)?);
100                anyhow::ensure!(
101                    interval >= Duration::from_secs(1) && interval <= remote_write::MAX_INTERVAL,
102                    "MIMIR_PUSH_INTERVAL_SECS must be between 1 and {}",
103                    remote_write::MAX_INTERVAL.as_secs()
104                );
105                Some(RemoteWriteConfig {
106                    url: remote_write::parse_url(&url).context("MIMIR_URL")?,
107                    username,
108                    password,
109                    interval,
110                    external_labels: remote_write::parse_external_labels(
111                        &std::env::var("MIMIR_EXTERNAL_LABELS").unwrap_or_default(),
112                    )
113                    .context("MIMIR_EXTERNAL_LABELS")?,
114                })
115            }
116        };
117        Ok(Self {
118            backend_url,
119            standby_backend_url,
120            listen_addr,
121            metrics_listen_addr,
122            info_cache_ttl,
123            connect_timeout,
124            keepalive_interval,
125            log_bucket,
126            log_region,
127            btc_network,
128            remote_write,
129        })
130    }
131}

133fn parse_env_u64(key: &str, default: u64) -> Result<u64> {
134    match std::env::var(key) {
135        Ok(v) => v
136            .parse()
137            .with_context(|| format!("{key} must be a non-negative integer")),
138        Err(_) => Ok(default),
139    }
140}
