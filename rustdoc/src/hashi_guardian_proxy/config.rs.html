4//! Environment-driven proxy configuration (matches the guardian's minimal,
5//! env-only config style). The public listener serves plaintext behind a
6//! TLS-terminating load balancer; the node listener terminates TLS itself.
7
8use std::net::SocketAddr;
9use std::time::Duration;
10
11use anyhow::Context;
12use anyhow::Result;
13use bitcoin::Network;
14
15use crate::remote_write;
16use crate::remote_write::RemoteWriteConfig;
17use crate::tls::CertSource;
18
19pub struct Config {
20    /// gRPC endpoint of the enclave guardian to forward to, e.g.
21    /// `http://10.0.1.20:3000` (`GUARDIAN_BACKEND_URL`, required).
22    pub backend_url: String,
23    /// gRPC endpoint of a standby guardian being armed (`STANDBY_GUARDIAN_URL`,
24    /// optional). When set, the relay — share submissions and
25    /// `GetProvisioningTargetInfo` — targets it while node-facing forwarding
26    /// stays on the active backend; unset, the relay provisions the active
27    /// backend (first deploy).
28    pub standby_backend_url: Option<String>,
29    /// Address the public listener serves on — gRPC (forwarder + relay + health)
30    /// and the HTTP `/info` + `/health` (`PROXY_LISTEN_ADDR`, default `0.0.0.0:3000`).
31    pub listen_addr: SocketAddr,
32    /// Address the node listener serves on (`NODE_LISTEN_ADDR`, default
33    /// `0.0.0.0:3443`).
34    pub node_listen_addr: SocketAddr,
35    /// Address the prometheus `/metrics` endpoint listens on
36    /// (`METRICS_LISTEN_ADDR`, default `0.0.0.0:9184`).
37    pub metrics_listen_addr: SocketAddr,
38    /// TTL for the single-slot `/info` response cache
39    /// (`INFO_CACHE_TTL_MS`, default 30000).
40    pub info_cache_ttl: Duration,
41    /// TCP connect timeout to the backend
42    /// (`GUARDIAN_CONNECT_TIMEOUT_SECS`, default 5).
43    pub connect_timeout: Duration,
44    /// HTTP/2 keepalive ping interval to the backend
45    /// (`GUARDIAN_KEEPALIVE_SECS`, default 5).
46    pub keepalive_interval: Duration,
47    /// The guardian's S3 log bucket, read as the wid cache's durable tier
48    /// (`GUARDIAN_LOG_BUCKET` + `GUARDIAN_LOG_REGION`, required). Credentials
49    /// come from the AWS default provider chain.
50    pub log_bucket: String,
51    pub log_region: String,
52    /// BTC network the guardian signs for (`BTC_NETWORK`, required:
53    /// bitcoin|testnet|signet|regtest). Must match the guardian's config; used
54    /// to recompute sighashes when verifying a log replay.
55    pub btc_network: Network,
56    /// Sui fullnode gRPC endpoint the committee member allowlist and stored
57    /// committee handoffs are read from (`SUI_RPC_URL`, required), on the chain
58    /// of the guardian's Hashi object.
59    pub sui_rpc_url: String,
60    /// Push metrics to a Prometheus remote-write endpoint; `None` leaves them
61    /// on `/metrics`, which nothing can scrape (`MIMIR_URL`, `MIMIR_USERNAME`
62    /// default `incoming_metrics`, `MIMIR_PASSWORD`, `MIMIR_PUSH_INTERVAL_SECS`
63    /// default 60, `MIMIR_EXTERNAL_LABELS` comma-separated `k=v`).
64    pub remote_write: Option<RemoteWriteConfig>,
65    /// The certificate the node listener serves: an exportable ACM certificate
66    /// (`TLS_CERT_ARN`), or PEM files (`TLS_CERT_FILE` + `TLS_KEY_FILE`). With
67    /// neither there is no node listener, for local setups.
68    pub node_tls: Option<CertSource>,
69}
70
71impl Config {
72    pub fn from_env() -> Result<Self> {
73        let backend_url = std::env::var("GUARDIAN_BACKEND_URL")
74            .context("GUARDIAN_BACKEND_URL must be set (gRPC endpoint of the enclave guardian)")?;
75        let standby_backend_url = std::env::var("STANDBY_GUARDIAN_URL")
76            .ok()
77            .filter(|url| !url.is_empty());
78        let listen_addr = std::env::var("PROXY_LISTEN_ADDR")
79            .unwrap_or_else(|_| "0.0.0.0:3000".to_string())
80            .parse()
81            .context("PROXY_LISTEN_ADDR must be a valid socket address")?;
82        let node_listen_addr = std::env::var("NODE_LISTEN_ADDR")
83            .unwrap_or_else(|_| "0.0.0.0:3443".to_string())
84            .parse()
85            .context("NODE_LISTEN_ADDR must be a valid socket address")?;
86        let metrics_listen_addr = std::env::var("METRICS_LISTEN_ADDR")
87            .unwrap_or_else(|_| "0.0.0.0:9184".to_string())
88            .parse()
89            .context("METRICS_LISTEN_ADDR must be a valid socket address")?;
90        let info_cache_ttl = Duration::from_millis(parse_env_u64("INFO_CACHE_TTL_MS", 30_000)?);
91        let connect_timeout =
92            Duration::from_secs(parse_env_u64("GUARDIAN_CONNECT_TIMEOUT_SECS", 5)?);
93        let keepalive_interval = Duration::from_secs(parse_env_u64("GUARDIAN_KEEPALIVE_SECS", 5)?);
94        let log_bucket = std::env::var("GUARDIAN_LOG_BUCKET")
95            .context("GUARDIAN_LOG_BUCKET must be set (the guardian's S3 log bucket)")?;
96        let log_region = std::env::var("GUARDIAN_LOG_REGION")
97            .context("GUARDIAN_LOG_REGION must be set (region of the guardian's S3 log bucket)")?;
98        let btc_network = std::env::var("BTC_NETWORK")
99            .context("BTC_NETWORK must be set (bitcoin|testnet|signet|regtest)")?
100            .parse()
101            .context("BTC_NETWORK must be one of bitcoin|testnet|signet|regtest")?;
102        let sui_rpc_url = std::env::var("SUI_RPC_URL")
103            .context("SUI_RPC_URL must be set (Sui fullnode for the committee allowlist)")?;
104        let remote_write = match std::env::var("MIMIR_URL").ok().filter(|u| !u.is_empty()) {
105            None => None,
106            Some(url) => {
107                let username = std::env::var("MIMIR_USERNAME")
108                    .unwrap_or_else(|_| "incoming_metrics".to_string());
109                let password = std::env::var("MIMIR_PASSWORD")
110                    .context("MIMIR_PASSWORD must be set when MIMIR_URL is")?;
111                anyhow::ensure!(
112                    !username.is_empty() && !username.contains(':') && !password.is_empty(),
113                    "MIMIR_USERNAME and MIMIR_PASSWORD must be non-empty (no ':' in the name)"
114                );
115                // Zero panics `tokio::time::interval` (a panic aborts this binary);
116                // huge values overflow the deadline arithmetic.
117                let interval = Duration::from_secs(parse_env_u64("MIMIR_PUSH_INTERVAL_SECS", 60)?);
118                anyhow::ensure!(
119                    interval >= Duration::from_secs(1) && interval <= remote_write::MAX_INTERVAL,
120                    "MIMIR_PUSH_INTERVAL_SECS must be between 1 and {}",
121                    remote_write::MAX_INTERVAL.as_secs()
122                );
123                Some(RemoteWriteConfig {
124                    url: remote_write::parse_url(&url).context("MIMIR_URL")?,
125                    username,
126                    password,
127                    interval,
128                    external_labels: remote_write::parse_external_labels(
129                        &std::env::var("MIMIR_EXTERNAL_LABELS").unwrap_or_default(),
130                    )
131                    .context("MIMIR_EXTERNAL_LABELS")?,
132                })
133            }
134        };
135        let node_tls = match (
136            non_empty_env("TLS_CERT_ARN"),
137            non_empty_env("TLS_CERT_FILE"),
138            non_empty_env("TLS_KEY_FILE"),
139        ) {
140            (None, None, None) => None,
141            (Some(arn), None, None) => Some(CertSource::Acm { arn }),
142            (None, Some(cert), Some(key)) => Some(CertSource::Files {
143                cert: cert.into(),
144                key: key.into(),
145            }),
146            _ => anyhow::bail!(
147                "set either TLS_CERT_ARN or both TLS_CERT_FILE and TLS_KEY_FILE, not a mix"
148            ),
149        };
150        Ok(Self {
151            backend_url,
152            standby_backend_url,
153            listen_addr,
154            node_listen_addr,
155            metrics_listen_addr,
156            info_cache_ttl,
157            connect_timeout,
158            keepalive_interval,
159            log_bucket,
160            log_region,
161            btc_network,
162            sui_rpc_url,
163            remote_write,
164            node_tls,
165        })
166    }
167}

169fn non_empty_env(key: &str) -> Option<String> {
170    std::env::var(key).ok().filter(|value| !value.is_empty())
171}
172
173fn parse_env_u64(key: &str, default: u64) -> Result<u64> {
174    match std::env::var(key) {
175        Ok(v) => v
176            .parse()
177            .with_context(|| format!("{key} must be a non-negative integer")),
178        Err(_) => Ok(default),
179    }
