4//! Forwards the node/KP-facing `GuardianService` RPCs to the enclave guardian
5//! and rejects the operator surface with `PERMISSION_DENIED`: the proxy is
6//! internet-facing and `OperatorInit` is one-shot and unauthenticated, so
7//! exposing it would let anyone wedge the guardian. KP-signed RPCs are
8//! forwarded after a signature and roster check; `ConfirmCeremony` goes to the
9//! ceremony guardian, which is the relay's backend. Wrapped by
10//! [`crate::cache::CachingGuardianGrpc`] to cache `StandardWithdrawal`.
11

30/// Holds a plain [`Channel`] rather than the node's boxed transport: the generated
31/// server trait requires `Send + Sync + 'static`, and `BoxCloneService` is not `Sync`.
32#[derive(Clone)]
33pub struct Forwarding<L> {
34    client: GuardianServiceClient<Channel>,
35    /// The guardian KPs are provisioning, same as the relay's: the standby
36    /// when one is configured, else the active guardian.
37    ceremony_client: GuardianServiceClient<Channel>,
38    /// Shared with the relay: one gate admits every KP-signed RPC, and a cert
39    /// rotation drops the cached roster for both.
40    roster: Arc<RosterCache<L>>,
41}
42
43impl<L: LogStore> Forwarding<L> {
44    pub fn new(channel: Channel, ceremony_channel: Channel, roster: Arc<RosterCache<L>>) -> Self {
45        Self {
46            client: GuardianServiceClient::new(channel),
47            ceremony_client: GuardianServiceClient::new(ceremony_channel),
48            roster,
49        }
50    }
51
52    /// Admission control only: the enclave repeats both checks. Signature
53    /// first because it needs no roster read.
54    async fn admit<T, P>(&self, request: &P) -> Result<(), Status>
55    where
56        T: KpSigningIntent,
57        P: Clone,
58        KpSigned<T>: TryFrom<P, Error = GuardianError>,
59    {
60        let signer = verify_kp_signature::<T, P>(request)?.signer_fingerprint();
61        self.roster.authorize(&signer).await
62    }
63}
64
65fn denied(rpc: &str) -> Status {
66    Status::permission_denied(format!(
67        "{rpc} is not served by the guardian proxy; operator calls reach the \
68         guardian directly and KP shares use SingleProvisionerInit"
69    ))
70}
71
72fn verify_kp_signature<T, P>(request: &P) -> Result<KpSigned<T>, Status>
73where
74    T: KpSigningIntent,
75    P: Clone,
76    KpSigned<T>: TryFrom<P, Error = GuardianError>,
77{
78    let signed_request = KpSigned::<T>::try_from(request.clone())
79        .map_err(|e| Status::invalid_argument(format!("malformed request: {e}")))?;
80    signed_request
81        .verify_signature()
82        .map_err(|e| Status::unauthenticated(e.to_string()))?;
83    Ok(signed_request)
84}

86// Each method clones the cheap channel-backed client and forwards the whole
87// `Request<T>` so client deadlines/metadata propagate.
88#[tonic::async_trait]
89impl<L: LogStore> GuardianService for Forwarding<L> {
90    async fn get_guardian_info(
91        &self,
92        request: Request<proto::GetGuardianInfoRequest>,
93    ) -> Result<Response<proto::GetGuardianInfoResponse>, Status> {
94        self.client.clone().get_guardian_info(request).await
95    }
96
97    async fn standard_withdrawal(
98        &self,
99        request: Request<proto::SignedStandardWithdrawalRequest>,
100    ) -> Result<Response<proto::SignedStandardWithdrawalResponse>, Status> {
101        self.client.clone().standard_withdrawal(request).await
102    }
103
104    async fn update_committee(
105        &self,
106        request: Request<proto::SignedCommitteeTransition>,
107    ) -> Result<Response<proto::UpdateCommitteeResponse>, Status> {
108        self.client.clone().update_committee(request).await
109    }
110
111    async fn update_committee_chain(
112        &self,
113        request: Request<proto::UpdateCommitteeChainRequest>,
114    ) -> Result<Response<proto::UpdateCommitteeResponse>, Status> {
115        self.client.clone().update_committee_chain(request).await
116    }
117
118    async fn provisioner_rotate_cert(
119        &self,
120        request: Request<proto::SignedProvisionerRotateCertRequest>,
121    ) -> Result<Response<proto::SignedProvisionerRotateCertResponse>, Status> {
122        self.admit::<ProvisionerRotateCertRequest, _>(request.get_ref())
123            .await?;
124        let response = self.client.clone().provisioner_rotate_cert(request).await?;
125        // The enclave has committed the replacement cert to the share log, so
126        // drop the cached roster: otherwise the new cert is rejected until the
127        // TTL lapses.
128        self.roster.invalidate().await;
129        Ok(response)
130    }
131
132    // Safe to expose: the enclave binds each confirmation to its session, the
133    // ceremony digest and the dealt roster.
134    async fn confirm_ceremony(
135        &self,
136        request: Request<proto::SignedCeremonyConfirmationRequest>,
137    ) -> Result<Response<proto::CeremonyConfirmationResponse>, Status> {
138        self.admit::<CeremonyConfirmationRequest, _>(request.get_ref())
139            .await?;
140        self.ceremony_client.clone().confirm_ceremony(request).await
141    }
142
143    // --- Rejected: operator surface ---
144
145    async fn operator_init(
146        &self,
147        _request: Request<proto::OperatorInitRequest>,
148    ) -> Result<Response<proto::OperatorInitResponse>, Status> {
149        Err(denied("OperatorInit"))
150    }
151
152    async fn setup_new_key(
153        &self,
154        _request: Request<proto::SetupNewKeyRequest>,
155    ) -> Result<Response<proto::SignedSetupNewKeyResponse>, Status> {
156        Err(denied("SetupNewKey"))
157    }
158
159    async fn provisioner_init(
160        &self,
161        _request: Request<proto::BatchProvisionerInitRequest>,
162    ) -> Result<Response<proto::ProvisionerInitResponse>, Status> {
163        Err(denied("ProvisionerInit (use SingleProvisionerInit)"))
164    }
165
166    async fn operator_activate(
167        &self,
168        _request: Request<proto::OperatorActivateRequest>,
169    ) -> Result<Response<proto::OperatorActivateResponse>, Status> {
170        Err(denied("OperatorActivate"))
171    }
172
173    async fn rotate_kp_set(
174        &self,
175        _request: Request<proto::BatchProvisionerRotateKpSetRequest>,
176    ) -> Result<Response<proto::SignedRotateKpSetResponse>, Status> {
177        Err(denied("RotateKpSet"))
178    }
179}

301    async fn spawn_stub() -> (StubGuardian, tonic::transport::Channel) {
302        let stub = StubGuardian::default();
303        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
304        let addr = listener.local_addr().unwrap();
305        let server = GuardianServiceServer::new(stub.clone());
306        tokio::spawn(async move {
307            Server::builder()
308                .add_service(server)
309                .serve_with_incoming(TcpListenerStream::new(listener))
310                .await
311                .unwrap();
312        });
313        // Let the spawned server start serving HTTP/2 before the first call.
314        tokio::time::sleep(Duration::from_millis(100)).await;
315
316        let channel = tonic::transport::Endpoint::from_shared(format!("http://{addr}"))
317            .unwrap()
318            .connect_lazy();
319        (stub, channel)
320    }

335    /// A proxy whose active and ceremony guardian are the same stub.
336    async fn spawn_stub_proxy(
337        store: StubStore,
338    ) -> (
339        StubGuardian,

346    #[tokio::test]
347    async fn forwards_and_caches_over_real_grpc() {
348        let (stub, proxy) = spawn_stub_proxy(StubStore::default()).await;
349
350        // First withdrawal forwards to the stub; a same-wid retry at a bumped
351        // seq replays the cached response without re-calling the stub.
352        let r1 = proxy
353            .standard_withdrawal(mock_request([0x11; 32], 0))
354            .await
355            .unwrap()
356            .into_inner();
357        let r2 = proxy
358            .standard_withdrawal(mock_request([0x11; 32], 1))
359            .await
360            .unwrap()
361            .into_inner();
362        assert_eq!(stub.standard_withdrawal_calls.load(Ordering::SeqCst), 1);
363        assert_eq!(r1, r2);
364
365        // A non-withdrawal node RPC passes through to the stub.
366        proxy
367            .get_guardian_info(Request::new(proto::GetGuardianInfoRequest {}))
368            .await
369            .unwrap();
370        assert_eq!(stub.get_guardian_info_calls.load(Ordering::SeqCst), 1);
371    }
