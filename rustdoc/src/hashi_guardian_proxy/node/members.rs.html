39const REFRESH_INTERVAL: Duration = Duration::from_secs(30);
40const RETRY_INTERVAL: Duration = Duration::from_secs(5);
41/// Once the chain state it was read from is this old, a snapshot admits no one,
42/// so a Sui RPC outage or a stalled fullnode fails closed.
43const MAX_SNAPSHOT_AGE: Duration = Duration::from_secs(10 * 60);
44const FETCH_TIMEOUT: Duration = Duration::from_secs(30);
45const PAGE_SIZE: u32 = 1000;
46
47#[derive(Debug)]
48pub struct MemberSnapshot {
49    /// The members' registered TLS public keys.
50    pub members: HashSet<[u8; 32]>,
51    /// Checkpoint time of the stalest read it was built from.
52    pub checkpoint_timestamp_ms: u64,
53}
54
55#[tonic::async_trait]
56pub trait MemberSource: Send + Sync + 'static {
57    async fn fetch(&self) -> anyhow::Result<MemberSnapshot>;
58}

60pub struct MemberAllowlist {
61    latest: RwLock<Option<Arc<MemberSnapshot>>>,
62    metrics: Arc<ProxyMetrics>,
63}
64
65impl MemberAllowlist {
66    pub fn new(metrics: Arc<ProxyMetrics>) -> Self {
67        Self {
68            latest: RwLock::new(None),
69            metrics,
70        }
71    }
72
73    /// The latest snapshot, unless it is too old to trust.
74    pub fn current(&self) -> Option<Arc<MemberSnapshot>> {
75        let snapshot = self
76            .latest
77            .read()
78            .expect("allowlist lock poisoned")
79            .clone()?;
80        let age_ms = now_timestamp_ms().saturating_sub(snapshot.checkpoint_timestamp_ms);
81        (u128::from(age_ms) <= MAX_SNAPSHOT_AGE.as_millis()).then_some(snapshot)
82    }
83
84    /// Refresh from `source` forever. A failed read keeps the last snapshot
85    /// until it ages out.
86    pub async fn refresh_forever(self: Arc<Self>, source: impl MemberSource) {
87        loop {
88            let fetched = tokio::time::timeout(FETCH_TIMEOUT, source.fetch())
89                .await
90                .unwrap_or_else(|_| Err(anyhow::anyhow!("timed out after {FETCH_TIMEOUT:?}")));
91            let delay = match fetched {
92                Ok(snapshot) => {
93                    self.store(snapshot);
94                    REFRESH_INTERVAL
95                }
96                Err(e) => {
97                    self.metrics.member_refresh_failures.inc();
98                    warn!(error = %format!("{e:#}"), "Committee member refresh failed.");
99                    RETRY_INTERVAL
100                }
101            };
102            tokio::time::sleep(delay).await;
103        }
104    }
105
106    pub(crate) fn store(&self, snapshot: MemberSnapshot) {
107        let mut latest = self.latest.write().expect("allowlist lock poisoned");
108        if let Some(previous) = latest.as_ref() {
109            // A lagging fullnode behind a load balancer can answer with older state.
110            if snapshot.checkpoint_timestamp_ms < previous.checkpoint_timestamp_ms {
111                warn!(
112                    read_ms = snapshot.checkpoint_timestamp_ms,
113                    current_ms = previous.checkpoint_timestamp_ms,
114                    "Ignoring a committee member read older than the current snapshot."
115                );
116                return;
117            }
118        }
119        self.metrics
120            .member_allowlist_size
121            .set(snapshot.members.len() as i64);
122        self.metrics
123            .member_snapshot_timestamp_seconds
124            .set(unix_millis_to_seconds(snapshot.checkpoint_timestamp_ms) as i64);
125        if latest
126            .as_ref()
127            .is_none_or(|previous| previous.members != snapshot.members)
128        {
129            info!(
130                members = snapshot.members.len(),
131                "Committee member allowlist changed."
132            );
133        }
134        *latest = Some(Arc::new(snapshot));
135    }
136}

138/// Reads the Hashi object id from the active guardian and its committees from
139/// Sui.
140pub struct ChainMemberSource {
141    guardian: GuardianServiceClient<Channel>,
142    sui: sui_rpc::Client,
143    /// Read once: it can't change under a running proxy, and later refreshes
144    /// then never queue on the enclave's control lock.
145    hashi_object_id: OnceLock<Address>,
146}
147
148impl ChainMemberSource {
149    pub fn new(guardian: Channel, sui_rpc_url: &str) -> anyhow::Result<Self> {
150        Ok(Self {
151            guardian: GuardianServiceClient::new(guardian),
152            sui: sui_rpc::Client::new(sui_rpc_url).context("SUI_RPC_URL")?,
153            hashi_object_id: OnceLock::new(),
154        })
155    }
156
157    async fn hashi_object_id(&self) -> anyhow::Result<Address> {
158        if let Some(id) = self.hashi_object_id.get() {
159            return Ok(*id);
160        }
161        let raw = self
162            .guardian
163            .clone()
164            .get_guardian_info(proto::GetGuardianInfoRequest {})
165            .await
166            .context("GetGuardianInfo")?
167            .into_inner();
168        let info = GuardianResponse::<GuardianInfo>::try_from(raw)
169            .map_err(|e| anyhow::anyhow!("decode GetGuardianInfo: {e:?}"))?
170            .response;
171        let id = info
172            .hashi_object_id
173            .context("the guardian has no Hashi object id yet")?;
174        Ok(*self.hashi_object_id.get_or_init(|| {
175            info!(hashi_object_id = %id, "Following this Hashi object's committees.");
176            id
177        }))
178    }
179}

184        read_snapshot(self.sui.clone(), self.hashi_object_id().await?).await

188/// Registered TLS keys of the members of the current committee and, during a
189/// reconfig, the pending one.
190pub async fn read_snapshot(
191    mut sui: sui_rpc::Client,
192    hashi_object_id: Address,
193) -> anyhow::Result<MemberSnapshot> {
194    let response = sui
195        .ledger_client()
196        .get_object(contents_request(&hashi_object_id))
197        .await
198        .with_context(|| format!("get Hashi object {hashi_object_id}"))?;
199    let root_timestamp_ms = response
200        .timestamp_ms()
201        .context("the Hashi object read has no checkpoint timestamp")?;
202    let root: move_types::Hashi = response
203        .into_inner()
204        .object()
205        .contents()
206        .deserialize()
207        .context("decode the Hashi object")?;
208    let committee_set = root.committees;
209
210    let mut committee = HashSet::new();
211    match get_committee(&mut sui, committee_set.committees.id, committee_set.epoch).await? {
212        Some(current) => committee.extend(current.members.iter().map(|m| m.validator_address)),
213        // No committee exists before the first reconfig.
214        None if committee_set.epoch == 0 => {}
215        None => anyhow::bail!("no committee for current epoch {}", committee_set.epoch),
216    }
217    // An aborted reconfig can remove the pending committee under us.
218    if let Some(pending) = &committee_set.pending_epoch_change {
219        if let Some(next) =
220            get_committee(&mut sui, committee_set.committees.id, pending.epoch).await?
221        {
222            committee.extend(next.members.iter().map(|m| m.validator_address));
223        }
224    }
225
226    let (members, members_timestamp_ms) =
227        read_member_keys(&mut sui, committee_set.members.id, &committee).await?;
228    Ok(MemberSnapshot {
229        members,
230        // Each read can reach a different fullnode behind a load balancer.
231        checkpoint_timestamp_ms: root_timestamp_ms.min(members_timestamp_ms),
232    })
233}

235/// The registered TLS keys of `committee`'s members, and the checkpoint time of
236/// the stalest page they were read from.
237async fn read_member_keys(
238    sui: &mut sui_rpc::Client,
239    members_bag: Address,
240    committee: &HashSet<Address>,
241) -> anyhow::Result<(HashSet<[u8; 32]>, u64)> {
242    // The fullnode returns empty entries for a mask of `value` alone.
243    let mask = FieldMask::from_paths([
244        DynamicField::path_builder().name().finish(),
245        DynamicField::path_builder().value().finish(),
246    ]);
247    let mut members = HashSet::new();
248    let mut checkpoint_timestamp_ms = u64::MAX;
249    let mut page_token = None;
250    loop {
251        let mut request = ListDynamicFieldsRequest::default()
252            .with_parent(members_bag)
253            .with_page_size(PAGE_SIZE)
254            .with_read_mask(mask.clone());
255        if let Some(token) = page_token.take() {
256            request = request.with_page_token(token);
257        }
258        let response = sui
259            .state_client()
260            .list_dynamic_fields(request)
261            .await
262            .context("list committee members")?;
263        checkpoint_timestamp_ms = checkpoint_timestamp_ms.min(
264            response
265                .timestamp_ms()
266                .context("a committee member page has no checkpoint timestamp")?,
267        );
268        let page = response.into_inner();
269        for field in &page.dynamic_fields {
270            let info: move_types::MemberInfo =
271                field.value().deserialize().context("decode MemberInfo")?;
272            if !committee.contains(&info.validator_address) {
273                continue;
274            }
275            if let Ok(tls_public_key) = <[u8; 32]>::try_from(info.tls_public_key.as_slice()) {
276                members.insert(tls_public_key);
277            }
278        }
279        match page.next_page_token {
280            Some(token) => page_token = Some(token),
281            None => break,
282        }
283    }
284    Ok((members, checkpoint_timestamp_ms))
285}

287async fn get_committee(
288    sui: &mut sui_rpc::Client,
289    committees: Address,
290    epoch: u64,
291) -> anyhow::Result<Option<move_types::Committee>> {
292    let field_id = committees.derive_dynamic_child_id(&TypeTag::U64, &epoch.to_bcs()?);
293    let field: Option<move_types::Field<u64, move_types::Committee>> =
294        get_object(sui, field_id).await?;
295    Ok(field.map(|field| field.value))
296}
297
298async fn get_object<T: serde::de::DeserializeOwned>(
299    sui: &mut sui_rpc::Client,
300    id: Address,
301) -> anyhow::Result<Option<T>> {
302    match sui.ledger_client().get_object(contents_request(&id)).await {
303        Ok(response) => {
304            let object = response
305                .into_inner()
306                .object()
307                .contents()
308                .deserialize()
309                .with_context(|| format!("decode object {id}"))?;
310            Ok(Some(object))
311        }
312        Err(status) if status.code() == tonic::Code::NotFound => Ok(None),
313        Err(status) => Err(anyhow::Error::new(status).context(format!("get object {id}"))),
314    }
315}
316
317fn contents_request(id: &Address) -> GetObjectRequest {
318    GetObjectRequest::new(id).with_read_mask(FieldMask::from_paths([Object::path_builder()
319        .contents()
320        .finish()]))
321}

354    /// Serves scripted results in order, then errors.

356    struct ScriptedSource {
357        results: Arc<Mutex<VecDeque<anyhow::Result<MemberSnapshot>>>>,
358        calls: Arc<Mutex<usize>>,
359    }
360
361    impl ScriptedSource {
362        fn new(results: Vec<anyhow::Result<MemberSnapshot>>) -> Self {
363            Self {
364                results: Arc::new(Mutex::new(results.into())),
365                calls: Arc::default(),
366            }
367        }
368
369        fn calls(&self) -> usize {
370            *self.calls.lock().unwrap()
371        }
372    }

374    #[tonic::async_trait]
375    impl MemberSource for ScriptedSource {
376        async fn fetch(&self) -> anyhow::Result<MemberSnapshot> {
377            *self.calls.lock().unwrap() += 1;
378            self.results
379                .lock()
380                .unwrap()
381                .pop_front()
382                .unwrap_or_else(|| Err(anyhow::anyhow!("sui unavailable")))
383        }
384    }

538    /// Answers each member page from a fullnode at the given checkpoint time, or
539    /// without one when `None`. The pages hold no members.
540    struct StampedPages(Vec<Option<u64>>);
541
542    #[tonic::async_trait]
543    impl StateService for StampedPages {
544        async fn list_dynamic_fields(
545            &self,
546            request: tonic::Request<ListDynamicFieldsRequest>,
547        ) -> Result<tonic::Response<ListDynamicFieldsResponse>, tonic::Status> {
548            let index = request
549                .into_inner()
550                .page_token
551                .map_or(0, |token| usize::from(token[0]));
552            let mut page = ListDynamicFieldsResponse::default();
553            if index + 1 < self.0.len() {
554                page.next_page_token = Some(vec![index as u8 + 1].into());
555            }
556            let mut response = tonic::Response::new(page);
557            if let Some(timestamp_ms) = self.0[index] {
558                response.metadata_mut().insert(
559                    sui_rpc::headers::X_SUI_TIMESTAMP_MS,
560                    timestamp_ms.to_string().parse().unwrap(),
561                );
562            }
563            Ok(response)
564        }
565    }

567    async fn read_stamped_pages(pages: Vec<Option<u64>>) -> anyhow::Result<u64> {
568        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
569        let addr = listener.local_addr().unwrap();
570        tokio::spawn(
571            tonic::transport::Server::builder()
572                .add_service(StateServiceServer::new(StampedPages(pages)))
573                .serve_with_incoming(tokio_stream::wrappers::TcpListenerStream::new(listener)),
574        );
575        let mut sui = sui_rpc::Client::new(format!("http://{addr}").as_str()).unwrap();
576        let (members, timestamp_ms) =
577            read_member_keys(&mut sui, Address::ZERO, &HashSet::new()).await?;
578        assert!(members.is_empty());
579        Ok(timestamp_ms)
580    }
