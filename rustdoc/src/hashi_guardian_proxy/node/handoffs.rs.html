4//! The proxy forwards a committee handoff only once the chain stores one
5//! between the same two epochs. The outgoing committee certifies a handoff
6//! while its reconfig is still pending, and a pending reconfig can abort: a
7//! guardian that took the certificate early would follow a committee the chain
8//! never activated, with no way back. The chain stores a handoff only when its
9//! reconfig completes (`end_reconfig`), and never changes or removes one, so a
10//! stored handoff stays stored however stale the read that found it.
11//!
12//! A handoff is matched only by the epochs it leaves and reaches, which is how
13//! `submit_committee_handoff` tells a completed one. Its certificate and
14//! committee are not compared with the chain's: an epoch only ever forms one
15//! committee (it is the Sui epoch of its `start_reconfig`, and an abort needs
16//! that epoch to have passed), and the enclave verifies the certificate over
17//! the committee it is sent.
18

44/// Nodes set no deadline on a committee update, so a stalled fullnode would
45/// otherwise hold the request open.
46const LOOKUP_TIMEOUT: Duration = Duration::from_secs(10);
47
48#[tonic::async_trait]
49pub trait HandoffSource: Send + Sync + 'static {
50    /// The epoch activated by the handoff the chain stores out of `from_epoch`,
51    /// if it stores one.
52    async fn next_epoch(&self, from_epoch: u64) -> anyhow::Result<Option<u64>>;
53}
54
55pub struct HandoffGate {
56    source: Box<dyn HandoffSource>,
57    /// The stored handoffs read so far: the epoch each leaves, to the one it
58    /// activates.
59    stored: Mutex<HashMap<u64, u64>>,
60    metrics: Arc<ProxyMetrics>,
61}
62
63impl HandoffGate {
64    pub fn new(source: impl HandoffSource, metrics: Arc<ProxyMetrics>) -> Self {
65        Self {
66            source: Box::new(source),
67            stored: Mutex::default(),
68            metrics,
69        }
70    }
71
72    /// Admit `transitions` only if each matches a stored handoff by its two
73    /// epochs alone, and each leaves the epoch the one before it reached.
74    pub async fn admit(
75        &self,
76        transitions: &[proto::SignedCommitteeTransition],
77    ) -> Result<(), Status> {
78        self.check(transitions).await.map_err(|refusal| {
79            self.metrics
80                .handoff_refused
81                .with_label_values(&[refusal.reason()])
82                .inc();
83            refusal.status()
84        })
85    }
86
87    async fn check(&self, transitions: &[proto::SignedCommitteeTransition]) -> Result<(), Refusal> {
88        let mut reached = None;
89        for transition in transitions {
90            let (from_epoch, to_epoch) = epochs(transition).ok_or(Refusal::Malformed)?;
91            // A handoff reaches a later epoch, so consecutive ones are distinct
92            // stored ones, which bounds how many a request can send the enclave.
93            if reached.is_some_and(|reached| reached != from_epoch) {
94                return Err(Refusal::NotConsecutive);
95            }
96            let stored = self.stored_next_epoch(from_epoch).await?;
97            if stored != Some(to_epoch) {
98                warn!(
99                    from_epoch,
100                    to_epoch,
101                    ?stored,
102                    "Refusing a committee handoff the chain does not store."
103                );
104                return Err(match stored {
105                    Some(stored) => Refusal::Superseded {
106                        from_epoch,
107                        to_epoch,
108                        stored,
109                    },
110                    None => Refusal::NotOnChain {
111                        from_epoch,
112                        to_epoch,
113                    },
114                });
115            }
116            reached = Some(to_epoch);
117        }
118        Ok(())
119    }
120
121    /// Stored handoffs are remembered, so replaying the chain's history costs no
122    /// reads, and each request at most one lookup that finds nothing.
123    async fn stored_next_epoch(&self, from_epoch: u64) -> Result<Option<u64>, Refusal> {
124        let known = self.stored().get(&from_epoch).copied();
125        if known.is_some() {
126            return Ok(known);
127        }
128        let read = tokio::time::timeout(LOOKUP_TIMEOUT, self.source.next_epoch(from_epoch))
129            .await
130            .unwrap_or_else(|_| Err(anyhow::anyhow!("timed out after {LOOKUP_TIMEOUT:?}")))
131            .map_err(|e| {
132                warn!(from_epoch, error = %format!("{e:#}"), "Committee handoff read failed.");
133                Refusal::ChainUnavailable
134            })?;
135        if let Some(next_epoch) = read {
136            self.stored().insert(from_epoch, next_epoch);
137        }
138        Ok(read)
139    }
140
141    fn stored(&self) -> MutexGuard<'_, HashMap<u64, u64>> {
142        self.stored.lock().expect("stored handoffs mutex poisoned")
143    }
144}
145
146/// The epochs a handoff leaves and reaches: the one its certificate was signed
147/// in and its new committee's. The enclave applies it by the same two fields.
148fn epochs(transition: &proto::SignedCommitteeTransition) -> Option<(u64, u64)> {
149    let from_epoch = transition.committee_signature.as_ref()?.epoch?;
150    let to_epoch = transition.data.as_ref()?.new_committee.as_ref()?.epoch?;
151    Some((from_epoch, to_epoch))
152}
153
154#[derive(Clone, Copy, Debug)]
155enum Refusal {
156    Malformed,
157    NotConsecutive,
158    /// Nothing is stored out of this epoch yet: its reconfig is pending, or
159    /// the read is stale.
160    NotOnChain {
161        from_epoch: u64,
162        to_epoch: u64,
163    },
164    /// Another handoff is stored out of this epoch, so this one's reconfig
165    /// aborted. No stale read can show that.
166    Superseded {
167        from_epoch: u64,
168        to_epoch: u64,
169        stored: u64,
170    },
171    ChainUnavailable,
172}
173
174impl Refusal {
175    fn reason(self) -> &'static str {
176        match self {
177            Self::Malformed => "malformed",
178            Self::NotConsecutive => "not_consecutive",
179            Self::NotOnChain { .. } => "not_on_chain",
180            Self::Superseded { .. } => "superseded",
181            Self::ChainUnavailable => "chain_unavailable",
182        }
183    }
184
185    fn status(self) -> Status {
186        match self {
187            Self::Malformed => {
188                Status::invalid_argument("malformed committee transition: missing an epoch")
189            }
190            Self::NotConsecutive => Status::invalid_argument(
191                "committee transitions are not consecutive: each must leave the epoch the one \
192                 before it reached",
193            ),
194            Self::NotOnChain {
195                from_epoch,
196                to_epoch,
197            } => Status::unavailable(format!(
198                "no handoff from epoch {from_epoch} to epoch {to_epoch} is stored on chain yet; \
199                 retry"
200            )),
201            Self::Superseded {
202                from_epoch,
203                to_epoch,
204                stored,
205            } => Status::failed_precondition(format!(
206                "the chain stores a handoff from epoch {from_epoch} to epoch {stored}, not to \
207                 epoch {to_epoch}"
208            )),
209            Self::ChainUnavailable => Status::unavailable("committee handoffs unavailable; retry"),
210        }
211    }
212}

214#[tonic::async_trait]
215impl HandoffSource for ChainSource {
216    async fn next_epoch(&self, from_epoch: u64) -> anyhow::Result<Option<u64>> {
217        read_next_epoch(self.sui.clone(), self.hashi_object_id().await?, from_epoch).await
218    }
219}
220
221/// The epoch activated by the handoff out of `from_epoch`, which `end_reconfig`
222/// stores in the committee set's `committees` bag.
223async fn read_next_epoch(
224    mut sui: sui_rpc::Client,
225    hashi_object_id: Address,
226    from_epoch: u64,
227) -> anyhow::Result<Option<u64>> {
228    let response = sui
229        .ledger_client()
230        .get_object(
231            GetObjectRequest::new(&hashi_object_id).with_read_mask(FieldMask::from_paths([
232                Object::path_builder().object_type(),
233                Object::path_builder().contents().finish(),
234            ])),
235        )
236        .await
237        .with_context(|| format!("get Hashi object {hashi_object_id}"))?
238        .into_inner();
239    let object = response.object();
240    let hashi_type: StructTag = object
241        .object_type_opt()
242        .context("the Hashi object read has no type")?
243        .parse()
244        .context("parse the Hashi object's type")?;
245    let root: move_types::Hashi = object
246        .contents()
247        .deserialize()
248        .context("decode the Hashi object")?;
249
250    // The key's address is the `Hashi` type's only because both are defined in
251    // the original package: a type added by an upgrade keeps that upgrade's.
252    let key_type = TypeTag::Struct(Box::new(StructTag::new(
253        *hashi_type.address(),
254        Identifier::from_static("committee_set"),
255        Identifier::from_static("CommitteeHandoffKey"),
256        vec![],
257    )));
258    let key = move_types::CommitteeHandoffKey { epoch: from_epoch };
259    let field_id = root
260        .committees
261        .committees
262        .id
263        .derive_dynamic_child_id(&key_type, &key.to_bcs()?);
264    let field: Option<
265        move_types::Field<move_types::CommitteeHandoffKey, move_types::CommitteeHandoff>,
266    > = get_object(&mut sui, field_id).await?;
267    Ok(field.map(|field| field.value.next_epoch))
268}

270#[cfg(test)]
271pub(crate) mod test_utils {
272    use super::*;
273    use std::sync::atomic::AtomicUsize;
274    use std::sync::atomic::Ordering;
275
276    /// A chain storing these handoffs, by the epoch each leaves.
277    pub(super) struct StoredHandoffs {
278        next_epochs: HashMap<u64, u64>,
279        pub(super) lookups: Arc<AtomicUsize>,
280    }
281
282    impl StoredHandoffs {
283        pub(super) fn new(handoffs: &[(u64, u64)]) -> Self {
284            Self {
285                next_epochs: handoffs.iter().copied().collect(),
286                lookups: Arc::default(),
287            }
288        }
289    }
290
291    #[tonic::async_trait]
292    impl HandoffSource for StoredHandoffs {
293        async fn next_epoch(&self, from_epoch: u64) -> anyhow::Result<Option<u64>> {
294            self.lookups.fetch_add(1, Ordering::SeqCst);
295            Ok(self.next_epochs.get(&from_epoch).copied())
296        }
297    }
298
299    /// A gate over a chain storing `handoffs`, each as `(from_epoch, next_epoch)`.
300    pub(crate) fn gate_over(handoffs: &[(u64, u64)]) -> Arc<HandoffGate> {
301        Arc::new(HandoffGate::new(
302            StoredHandoffs::new(handoffs),
303            Arc::new(ProxyMetrics::new()),
304        ))
305    }
306
307    /// A transition carrying only the epochs the gate reads.
308    pub(crate) fn transition(from_epoch: u64, to_epoch: u64) -> proto::SignedCommitteeTransition {
309        proto::SignedCommitteeTransition {
310            data: Some(proto::CommitteeTransition {
311                new_committee: Some(proto::Committee {
312                    epoch: Some(to_epoch),
313                    ..Default::default()
314                }),
315            }),
316            committee_signature: Some(proto::CommitteeSignature {
317                epoch: Some(from_epoch),
318                ..Default::default()
319            }),
320        }
321    }
322}

352    fn gate(source: impl HandoffSource) -> (HandoffGate, Arc<ProxyMetrics>) {
353        let metrics = Arc::new(ProxyMetrics::new());
354        (HandoffGate::new(source, metrics.clone()), metrics)
355    }

357    fn refused(metrics: &ProxyMetrics, reason: &str) -> u64 {
358        metrics.handoff_refused.with_label_values(&[reason]).get()
359    }

361    #[tokio::test]
362    async fn admits_only_the_handoff_the_chain_stores() {
363        // Epoch 7 formed out of 5 and aborted; 8 replaced it and activated.
364        let (gate, metrics) = gate(StoredHandoffs::new(&[(5, 8)]));
365        gate.admit(&[transition(5, 8)]).await.unwrap();
366
367        let aborted = gate.admit(&[transition(5, 7)]).await.unwrap_err();
368        assert_eq!(aborted.code(), Code::FailedPrecondition);
369        assert_eq!(refused(&metrics, "superseded"), 1);
370        // A reconfig out of 8 is pending at most: nothing is stored for it.
371        let pending = gate.admit(&[transition(8, 9)]).await.unwrap_err();
372        assert_eq!(pending.code(), Code::Unavailable);
373        assert_eq!(refused(&metrics, "not_on_chain"), 1);
374    }

376    #[tokio::test]
377    async fn admits_a_chain_only_if_each_handoff_is_stored() {
378        let (gate, _) = gate(StoredHandoffs::new(&[(5, 7), (7, 9)]));
379        gate.admit(&[]).await.unwrap();
380        gate.admit(&[transition(5, 7), transition(7, 9)])
381            .await
382            .unwrap();
383
384        let unstored = gate
385            .admit(&[transition(5, 7), transition(7, 10)])
386            .await
387            .unwrap_err();
388        assert_eq!(unstored.code(), Code::FailedPrecondition);
389    }

391    #[tokio::test]
392    async fn refuses_a_chain_that_is_not_consecutive() {
393        let (gate, metrics) = gate(StoredHandoffs::new(&[(5, 7), (7, 9)]));
394
395        // Every handoff here is stored, but none follows the one before it.
396        let repeated = vec![transition(5, 7); 100];
397        let reordered = vec![transition(7, 9), transition(5, 7)];
398        for chain in [repeated, reordered] {
399            let err = gate.admit(&chain).await.unwrap_err();
400            assert_eq!(err.code(), Code::InvalidArgument);
401        }
402        assert_eq!(refused(&metrics, "not_consecutive"), 2);
403    }

405    #[tokio::test]
406    async fn reads_a_stored_handoff_once() {
407        let chain = StoredHandoffs::new(&[(5, 7)]);
408        let lookups = chain.lookups.clone();
409        let (gate, _) = gate(chain);
410
411        gate.admit(&[transition(5, 7)]).await.unwrap();
412        gate.admit(&[transition(5, 7)]).await.unwrap();
413        gate.admit(&[transition(5, 8)]).await.unwrap_err();
414        assert_eq!(lookups.load(Ordering::SeqCst), 1);
415
416        // One the chain may yet store is read again each time.
417        gate.admit(&[transition(7, 9)]).await.unwrap_err();
418        gate.admit(&[transition(7, 9)]).await.unwrap_err();
419        assert_eq!(lookups.load(Ordering::SeqCst), 3);
420    }
