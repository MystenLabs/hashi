4//! The provisioning relay: the out-of-enclave half of `single_provisioner_init`.
5//!
6//! Key provisioners submit signed, HPKE-encrypted shares one at a time; the relay
7//! pre-verifies and accumulates distinct submissions for the guardian's current
8//! session and, once it holds a threshold-many, forwards them unchanged in one
9//! batch `ProvisionerInit`. The enclave re-verifies every signature, so the relay
10//! is liveness-only: it can stall provisioning but cannot read a share or forge
11//! a key.
12//!
13//! The relay's backend is the guardian KPs are provisioning: the proxy's
14//! standby when one is configured, else the active guardian.
15//! `GetProvisioningTargetInfo` exposes that backend's `GetGuardianInfo` so KP
16//! tooling pins the session it is actually submitting to (the node-facing
17//! `GetGuardianInfo` always answers for the ACTIVE guardian).
18//!
19//! `Accumulator` holds the (pure, unit-tested) accumulation logic; a `tokio`
20//! mutex serializes it and keeps at most one `ProvisionerInit` in flight.
21

45/// Distinct-share accumulator for one guardian session. Reset whenever the
46/// backend session changes (buffered shares are encrypted to the old session's
47/// key and are useless against the new one).
48#[derive(Default)]
49struct Accumulator {
50    session_id: Option<String>,
51    submissions: BTreeMap<u32, proto::SignedSingleProvisionerInitRequest>,
52    completed: bool,
53}
54
55impl Accumulator {
56    /// Adopt `live` as the current session, clearing any stale buffer.
57    fn sync_session(&mut self, live: &str) {
58        if self.session_id.as_deref() != Some(live) {
59            self.session_id = Some(live.to_string());
60            self.submissions.clear();
61            self.completed = false;
62        }
63    }
64
65    fn insert(&mut self, id: u32, submission: proto::SignedSingleProvisionerInitRequest) {
66        self.submissions.insert(id, submission);
67    }
68
69    fn have(&self) -> usize {
70        self.submissions.len()
71    }
72
73    fn batch(&self) -> Vec<proto::SignedSingleProvisionerInitRequest> {
74        self.submissions.values().cloned().collect()
75    }
76
77    fn clear_submissions(&mut self) {
78        self.submissions.clear();
79    }
80}

82/// Live backend state the relay needs, read from `GetGuardianInfo`.
83struct BackendStatus {
84    session_id: String,
85    /// Installed by `operator provision` and cleared as a unit at activation,
86    /// so it is readable exactly while a KP could still be submitting.
87    arming: Option<BackendArming>,
88    provisioned: bool,
89}
90
91/// What a KP's submission pins itself to, as the backend reports it.
92#[derive(Debug)]
93struct BackendArming {
94    num_shares: usize,
95    threshold: usize,
96    config_hash: [u8; 32],
97    genesis_state_hash: Option<[u8; 32]>,
98}
99
100/// `GetProvisioningTargetInfo` is public and unauthenticated, and the enclave
101/// mints a fresh Nitro attestation for every `GetGuardianInfo`, so a flood could
102/// crowd out provisioning. The request carries no nonce, so a seconds-old
103/// response is as good as a fresh one; cache it briefly to bound the backend's
104/// exposure. The only effect of staleness is that a KP may pin a session that
105/// has since changed, and `single_provisioner_init` reads the session fresh and
106/// rejects that loudly.
107const TARGET_INFO_TTL: Duration = Duration::from_secs(5);
108
109struct CachedTargetInfo {
110    at: Instant,
111    response: proto::GetGuardianInfoResponse,
112}

114#[derive(Clone)]
115pub struct Relay<L> {
116    client: GuardianServiceClient<Channel>,
117    accumulator: Arc<Mutex<Accumulator>>,
118    roster: Arc<RosterCache<L>>,
119    target_info: Arc<Mutex<Option<CachedTargetInfo>>>,
120}
121
122impl<L: LogStore> Relay<L> {
123    pub fn new(channel: Channel, roster: Arc<RosterCache<L>>) -> Self {
124        Self {
125            client: GuardianServiceClient::new(channel),
126            accumulator: Arc::new(Mutex::new(Accumulator::default())),
127            roster,
128            target_info: Arc::new(Mutex::new(None)),
129        }
130    }
131
132    /// Pre-authenticate a submission: its detached signature must cover these
133    /// exact (session, config, share) bytes, and the signer's cert must be in
134    /// the ceremony's committed roster. Signature first because it needs no
135    /// I/O — a submission that isn't internally consistent never costs an S3
136    /// roster read. DoS guard only; the enclave re-verifies authoritatively.
137    async fn verify_kp_submission<'a>(
138        &self,
139        signed_request: &'a KpSigned<SingleProvisionerInitRequest>,
140    ) -> Result<&'a SingleProvisionerInitRequest, Status> {
141        let request = signed_request
142            .verify_signature()
143            .map_err(|error| Status::unauthenticated(error.to_string()))?;
144        // The ceremony's committed roster, not deploy config: a rotation
145        // re-deals shares without a proxy redeploy.
146        check_rostered(
147            &signed_request.signer_fingerprint(),
148            &self.authorized_kps().await?,
149        )?;
150        Ok(request)
151    }
152
153    /// The ceremony's committed roster, mapped onto the relay's failure modes:
154    /// no share log yet is a definitive "not ready", a read error is transient.
155    async fn authorized_kps(&self) -> Result<Arc<Vec<Fingerprint>>, Status> {
156        match self.roster.get().await {
157            Ok(Some(roster)) => Ok(roster),
158            Ok(None) => Err(Status::failed_precondition(
159                "no KP share log in the guardian bucket; run the key ceremony first",
160            )),
161            Err(e) => {
162                warn!(error = %format!("{e:#}"), "KP roster read failed");
163                Err(Status::unavailable("KP roster unavailable; retry"))
164            }
165        }
166    }
167
168    /// Backend's self-reported session, provisioning threshold, and provisioned flag.
169    /// The relay is liveness-only, so it does not verify the signature or attestation.
170    async fn backend_status(&self) -> Result<BackendStatus, Status> {
171        let pb = self
172            .client
173            .clone()
174            .get_guardian_info(proto::GetGuardianInfoRequest {})
175            .await?
176            .into_inner();
177        let resp = GetGuardianInfoResponse::try_from(pb)
178            .map_err(|e| Status::internal(format!("decode backend GuardianInfo: {e:?}")))?;
179        let (info, signing_pub_key) = resp.into_info_unchecked();
180        let session_id = SessionID::from_signing_pubkey(&signing_pub_key);
181        let arming = info
182            .secret_sharing_instance
183            .as_ref()
184            .zip(info.config_hash)
185            .map(|(sharing, config_hash)| BackendArming {
186                num_shares: sharing.num_shares(),
187                threshold: sharing.threshold(),
188                config_hash,
189                genesis_state_hash: info.genesis_state_hash,
190            });
191        Ok(BackendStatus {
192            session_id: session_id.into(),
193            arming,
194            provisioned: info.enclave_btc_pubkey.is_some(),
195        })
196    }
197}

199/// What the backend a submission reached turned out to be.
200#[derive(Debug)]
201enum Matched<'a> {
202    /// Provisioned under exactly the pins the KP submitted against.
203    Provisioned,
204    /// Armed and still collecting shares under those pins.
205    Armed(&'a BackendArming),
206}
207
208/// Confirm the backend this submission reached is the one the KP pinned.
209///
210/// A KP reads the session it pins from `GetProvisioningTargetInfo`, and during
211/// a proxy rollout that read and the submission can land on instances with
212/// different relay backends. Checking the session before anything else is what
213/// stops an already-provisioned ACTIVE guardian from reporting a standby
214/// submission complete. (The share is HPKE-encrypted to the pinned session too,
215/// so a restarted backend could not use it either.)
216fn match_backend<'a>(
217    request: &SingleProvisionerInitRequest,
218    status: &'a BackendStatus,
219) -> Result<Matched<'a>, Status> {
220    let expected_session_id = request.expected_session_id();
221    if expected_session_id != status.session_id {
222        return Err(Status::failed_precondition(format!(
223            "session mismatch: KP pinned {}, backend live session is {} \
224             (guardian restarted? re-run the provision flow)",
225            expected_session_id, status.session_id
226        )));
227    }
228
229    // Activation clears the arming, leaving nothing further to compare; the
230    // session match already establishes this is the backend the KP pinned.
231    let Some(arming) = &status.arming else {
232        return if status.provisioned {
233            Ok(Matched::Provisioned)
234        } else {
235            Err(Status::failed_precondition(
236                "guardian is not armed yet; run `operator provision` first",
237            ))
238        };
239    };
240    let expected_config_hash = *request.expected_config_hash();
241    if expected_config_hash != arming.config_hash {
242        return Err(Status::failed_precondition(format!(
243            "config hash mismatch: KP pinned {}, backend live config is {}",
244            hex::encode(expected_config_hash),
245            hex::encode(arming.config_hash),
246        )));
247    }
248    let expected_genesis_state_hash = request.expected_genesis_state_hash();
249    if expected_genesis_state_hash != arming.genesis_state_hash {
250        return Err(Status::failed_precondition(format!(
251            "genesis state hash mismatch: KP pinned {:?}, backend live genesis state is {:?}",
252            expected_genesis_state_hash.map(hex::encode),
253            arming.genesis_state_hash.map(hex::encode),
254        )));
255    }
256
257    // Already provisioned (by us, a prior relay, or out-of-band): the submission
258    // is unnecessary, now that the arming is confirmed to be the pinned one.
259    if status.provisioned {
260        return Ok(Matched::Provisioned);
261    }
262    Ok(Matched::Armed(arming))
263}
264
265fn check_rostered(fingerprint: &Fingerprint, roster: &[Fingerprint]) -> Result<(), Status> {
266    if roster.contains(fingerprint) {
267        return Ok(());
268    }
269    Err(Status::permission_denied(format!(
270        "signer {fingerprint} is not in the relay's authorized KP roster"
271    )))
272}

274fn done() -> Response<proto::SingleProvisionerInitResponse> {
275    Response::new(proto::SingleProvisionerInitResponse {
276        have: 0,
277        need: 0,
278        completed: true,
279    })
280}

282fn progress(have: usize, need: usize) -> Response<proto::SingleProvisionerInitResponse> {
283    Response::new(proto::SingleProvisionerInitResponse {
284        have: have as u32,
285        need: need as u32,
286        completed: false,
287    })
288}

290// Cheap input hygiene — the guardian re-verifies each share. A real KP's id is
291// 1-indexed, so anything outside [1, num_shares] is malformed.
292fn check_share_id(id: u32, num_shares: usize) -> Result<(), Status> {
293    if id == 0 || id as usize > num_shares {
294        return Err(Status::invalid_argument(format!(
295            "share id {id} out of range [1, {num_shares}]"
296        )));
297    }
298    Ok(())
299}

301#[tonic::async_trait]
302impl<L: LogStore> GuardianRelayService for Relay<L> {
303    /// The relay backend's `GetGuardianInfo`, verbatim. The lock is held across
304    /// the fetch, so a burst collapses into one backend call.
305    async fn get_provisioning_target_info(
306        &self,
307        _request: Request<proto::GetProvisioningTargetInfoRequest>,
308    ) -> Result<Response<proto::GetGuardianInfoResponse>, Status> {
309        let mut cached = self.target_info.lock().await;
310        if let Some(entry) = cached.as_ref() {
311            if entry.at.elapsed() < TARGET_INFO_TTL {
312                return Ok(Response::new(entry.response.clone()));
313            }
314        }
315        let response = self
316            .client
317            .clone()
318            .get_guardian_info(proto::GetGuardianInfoRequest {})
319            .await?
320            .into_inner();
321        *cached = Some(CachedTargetInfo {
322            at: Instant::now(),
323            response: response.clone(),
324        });
325        Ok(Response::new(response))
326    }
327
328    async fn single_provisioner_init(
329        &self,
330        request: Request<proto::SignedSingleProvisionerInitRequest>,
331    ) -> Result<Response<proto::SingleProvisionerInitResponse>, Status> {
332        let submission = request.into_inner();
333        let signed_request = KpSigned::<SingleProvisionerInitRequest>::try_from(submission.clone())
334            .map_err(|e| Status::invalid_argument(format!("malformed request: {e}")))?;
335
336        // Authenticate before the lock or any backend read: junk submissions
337        // can't poison the batch, hold the mutex, or cost enclave round-trips.
338        let verified_request = self.verify_kp_submission(&signed_request).await?;
339        let id = u32::from(verified_request.encrypted_share().id.get());
340
341        // Hold the accumulator across the status read + batch submit so a racing
342        // session change can't wipe a half-filled buffer, and only one runs at a time.
343        let mut acc = self.accumulator.lock().await;
344
345        let status = self.backend_status().await?;
346        let threshold = match match_backend(verified_request, &status)? {
347            Matched::Provisioned => return Ok(done()),
348            Matched::Armed(arming) => {
349                check_share_id(id, arming.num_shares)?;
350                arming.threshold
351            }
352        };
353
354        acc.sync_session(&status.session_id);
355        if acc.completed {
356            return Ok(done());
357        }
358        acc.insert(id, submission);
359        let have = acc.have();
360        info!(
361            share_id = id,
362            have,
363            threshold,
364            session = %status.session_id,
365            "relay accepted a provisioner share",
366        );
367        if have < threshold {
368            return Ok(progress(have, threshold));
369        }
370
371        // Threshold reached: submit every buffered share in one batch.
372        let submissions = acc.batch();
373        match self
374            .client
375            .clone()
376            .provisioner_init(proto::ProvisionerInitRequest { submissions })
377            .await
378        {
379            Ok(_) => {
380                acc.completed = true;
381                info!(
382                    session = %status.session_id,
383                    shares = have,
384                    "relay submitted batch ProvisionerInit; guardian provisioned",
385                );
386                Ok(done())
387            }
388            Err(e) => {
389                // A racing batch or out-of-band ProvisionerInit may have provisioned
390                // the guardian since our status read; re-check before erroring —
391                // against the same pins, so this stays the only way to reach
392                // `done()` and a backend that moved under us can't turn a failed
393                // batch into a success.
394                match self.backend_status().await {
395                    Ok(s)
396                        if matches!(
397                            match_backend(verified_request, &s),
398                            Ok(Matched::Provisioned)
399                        ) =>
400                    {
401                        acc.completed = true;
402                        Ok(done())
403                    }
404                    _ => {
405                        // Genuine failure (e.g. a share won't decrypt). The batch is
406                        // all-or-nothing and we can't tell which share is bad, so drop
407                        // the whole buffer and let the KPs resubmit a clean set.
408                        warn!(
409                            error = %e,
410                            "batch ProvisionerInit failed; clearing the submission buffer for resubmission",
411                        );
412                        acc.clear_submissions();
413                        Err(Status::internal(format!(
414                            "guardian ProvisionerInit failed: {e}"
415                        )))
416                    }
417                }
418            }
419        }
420    }
421}

732    /// A relay whose backend is a live `TaggedGuardian` over real gRPC.
733    async fn relay_fronting(guardian: TaggedGuardian) -> Relay<MemStore> {
734        use hashi_types::proto::guardian_service_server::GuardianServiceServer;
735        use tokio_stream::wrappers::TcpListenerStream;
736
737        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
738        let addr = listener.local_addr().unwrap();
739        tokio::spawn(async move {
740            tonic::transport::Server::builder()
741                .add_service(GuardianServiceServer::new(guardian))
742                .serve_with_incoming(TcpListenerStream::new(listener))
743                .await
744                .unwrap();
745        });
746        tokio::time::sleep(Duration::from_millis(100)).await;
747
748        let channel = tonic::transport::Endpoint::from_shared(format!("http://{addr}"))
749            .unwrap()
750            .connect_lazy();
751        Relay::new(channel, Arc::new(RosterCache::new(MemStore::default())))
752    }
