4//! `key-provisioner rotate-kp-set`: one current KP's half of a KP-set rotation.
5//! Verifies the fresh ceremony guardian, decrypts this KP's share and signs a
6//! submission binding it (re-encrypted to the guardian) to the proposed roster
7//! and params. The file holds nothing secret; the operator batches them.
8
9use std::path::Path;
10
11use anyhow::Context;
12use anyhow::Result;
13use anyhow::anyhow;
14use anyhow::ensure;
15use hashi_guardian::s3_reader::GuardianReader;
16use hashi_types::guardian::CeremonyStage;
17use hashi_types::guardian::EncPubKey;
18use hashi_types::guardian::KpSigned;
19use hashi_types::guardian::ProvisionerRotateKpSetRequest;
20use hpke::Deserializable;
21use rand::thread_rng;
22use tracing::info;
23
24use crate::config::Config;
25use crate::guardian_info::verified_ceremony_guardian_info;
26use crate::kp_roster::decrypt_kp_share;
27use crate::kp_roster::load_kp_cert;
28use crate::submission;
29
30pub async fn run(cfg: Config, submission_path: &Path) -> Result<()> {
31    cfg.kp_roster.validate()?;
32    let new_kp_set = cfg.require_new_kp_roster("key-provisioner rotate-kp-set")?;
33    new_kp_set.validate()?;
34    let guardian_s3 = hashi_guardian::resolve_s3_config(&cfg.guardian_s3).await?;
35    let allowlist = cfg.kp_roster.pcr_allowlist();
36
37    info!(
38        phase = "setup",
39        bucket = guardian_s3.bucket_name(),
40        region = guardian_s3.region(),
41        num_shares = cfg.kp_roster.num_shares,
42        threshold = cfg.kp_roster.threshold,
43        new_num_shares = new_kp_set.num_shares,
44        new_threshold = new_kp_set.threshold,
45        endpoint = %cfg.guardian_endpoint,
46        "signing a KP-set rotation submission",
47    );
48
49    let certs_roster = cfg.kp_roster.load_certs_roster()?;
50    let new_certs_roster = new_kp_set.load_certs_roster()?;
51    let new_params = new_kp_set.params()?;
52    let kp_cert = load_kp_cert(cfg.require_kp_pgp_cert_path("key-provisioner rotate-kp-set")?)?;
53    ensure!(
54        certs_roster
55            .cert_for_fingerprint(&kp_cert.fingerprint())
56            .is_some(),
57        "this KP's cert (fingerprint {}) is not among the current kp_roster.kp_pgp_cert_paths",
58        kp_cert.fingerprint()
59    );
60    // What this KP is about to authorize. Compare it with the operator's.
61    for (index, fingerprint) in new_certs_roster.fingerprints().iter().enumerate() {
62        info!(
63            phase = "proposal",
64            share_id = index + 1,
65            recipient_fingerprint = %fingerprint,
66            "proposed new KP set entry",
67        );
68    }
69
70    // 1. The ceremony guardian: attested current build, operator-initialized
71    //    on the expected bucket, session attestation in S3.
72    let target =
73        verified_ceremony_guardian_info(&cfg.guardian_endpoint, allowlist.current_build()).await?;
74    ensure!(
75        target.info.lifecycle == CeremonyStage::OperatorInitialized.into(),
76        "guardian lifecycle is {:?}; expected ceremony/operator_initialized (run `operator rotate-kp-set init`)",
77        target.info.lifecycle
78    );
79    ensure!(
80        target.info.bucket_info.as_ref() == Some(&guardian_s3.bucket_info),
81        "guardian bucket info mismatch: expected {:?}, got {:?}",
82        guardian_s3.bucket_info,
83        target.info.bucket_info
84    );
85    let guardian_pub_key =
86        EncPubKey::from_bytes(&target.info.encryption_pubkey).map_err(anyhow::Error::msg)?;
87    let session_id = target.session_id;
88    let mut reader = GuardianReader::new(&guardian_s3, allowlist.clone())
89        .await
90        .context("connect to guardian log bucket")?;
91    let verified_session = reader.get_current_session_info(&session_id).await?;
92    ensure!(
93        verified_session.signing_pubkey() == &target.signing_pub_key,
94        "guardian S3 attestation signing pubkey differs from gRPC signing pubkey"
95    );
96    info!(
97        phase = "guardian info",
98        session_id = %session_id,
99        enc_pubkey = hex::encode(&target.info.encryption_pubkey),
100        "ceremony guardian verified; session pinned",
101    );
102
103    // 2. This KP's share of the dealt set, from the latest attested logs.
104    let state = reader.read_latest_ceremony_state().await?;
105    state.validate_sharing_params(cfg.kp_roster.num_shares, cfg.kp_roster.threshold)?;
106    state.encrypted_shares.verify_recipient_set(&certs_roster)?;
107    let sharing_seq = state.secret_sharing_instance.sharing_seq();
108    info!(
109        phase = "share read",
110        sharing_seq,
111        cert_seq = state.cert_seq,
112        "latest ceremony and kp-shares logs verified against the current roster",
113    );
114    let decrypted = decrypt_kp_share(&state, &kp_cert)?;
115
116    // 3. Bind the re-encrypted share to the proposal and sign.
117    let share_id = decrypted.id;
118    let request = ProvisionerRotateKpSetRequest::build_from_share(
119        session_id.clone(),
120        allowlist,
121        &decrypted,
122        &guardian_pub_key,
123        new_certs_roster,
124        new_params,
125        &mut thread_rng(),
126    )?;
127    drop(decrypted);
128    let signed = KpSigned::sign(request, kp_cert.clone(), None)
129        .context("sign the rotation submission with the KP key")?;
130    signed
131        .verify_signature()
132        .map_err(|e| anyhow!("re-verify the signed submission: {e}"))?;
133    submission::write(submission_path, signed)?;
134
135    info!(
136        phase = "summary",
137        path = %submission_path.display(),
138        session_id = %session_id,
139        share_id = share_id.get(),
140        signer_fingerprint = %kp_cert.fingerprint(),
141        sharing_seq,
142        new_sharing_seq = sharing_seq + 1,
143        new_num_shares = new_params.num_shares(),
144        new_threshold = new_params.threshold(),
145        "rotation submission written; send it to the operator",
146    );
147    println!(
148        "KP-set rotation submission written to {}",
149        submission_path.display()
150    );
151    println!("  session_id:     {session_id}");
152    println!("  share_id:       {}", share_id.get());
153    println!("  signer:         {}", kp_cert.fingerprint());
154    println!("  sharing_seq:    {sharing_seq} -> {}", sharing_seq + 1);
155    println!(
156        "  new set:        {}-of-{}",
157        new_params.threshold(),
158        new_params.num_shares()
159    );
160    Ok(())
161}
