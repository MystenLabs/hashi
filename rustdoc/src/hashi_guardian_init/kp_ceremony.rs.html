4//! `key-provisioner ceremony` verifies, decrypts, saves, and confirms this KP's ceremony share.
5
6use anyhow::Context;
7use anyhow::Result;
8use anyhow::anyhow;
9use anyhow::ensure;
10use hashi_guardian::s3_reader::GuardianReader;
11use hashi_types::guardian::CeremonyConfirmationRequest;
12use hashi_types::guardian::CeremonyConfirmationResponse;
13use hashi_types::guardian::CeremonyStage;
14use hashi_types::guardian::KpSigned;
15use hashi_types::guardian::proto_conversions::signed_ceremony_confirmation_request_to_pb;
16use hashi_types::proto::guardian_service_client::GuardianServiceClient;
17use std::path::Path;
18use tracing::info;
19
20use crate::config::Config;
21use crate::guardian_info::verified_live_guardian_info;
22use crate::kp_roster::decrypt_kp_share;
23use crate::kp_roster::load_kp_cert;

25/// Verify this KP can fetch and decrypt its ceremony share, then submit a
26/// signed confirmation to the live ceremony guardian.
27///
28/// The share state is anchored to the guardian's S3 attestation log. The live
29/// confirmation endpoint is independently attestation-verified and
30/// authoritatively matches the signed ceremony digest against its pending state.
31///
32/// Security: the ceremony state containing the encrypted shares is saved to the
33/// requested path. Each ciphertext is piped into `gpg --decrypt` over stdin and
34/// the plaintext streams back over stdout; neither ciphertext nor plaintext is
35/// separately written to disk by this flow.
36pub async fn run(cfg: Config, encrypted_shares_path: &Path) -> Result<()> {
37    cfg.kp_roster.validate()?;
38    let guardian_s3 = hashi_guardian::resolve_s3_config(&cfg.guardian_s3).await?;
39
40    info!(
41        phase = "setup",
42        bucket = guardian_s3.bucket_name(),
43        region = guardian_s3.region(),
44        num_shares = cfg.kp_roster.num_shares,
45        threshold = cfg.kp_roster.threshold,
46        sui_rpc = %cfg.hashi.sui_rpc,
47        package_id = %cfg.hashi.hashi_ids.package_id,
48        hashi_object_id = %cfg.hashi.hashi_ids.hashi_object_id,
49        "verifying ceremony share",
50    );
51
52    info!(
53        phase = "roster load",
54        share_count = cfg.kp_roster.kp_pgp_cert_paths.len(),
55        "loading + validating full KP certificate roster",
56    );
57    let certs_roster = cfg.kp_roster.load_certs_roster()?;
58    info!(
59        phase = "roster load",
60        share_count = certs_roster.num_kps(),
61        "KP certificate roster loaded"
62    );
63
64    // The selected cert identifies this KP's roster entry.
65    let kp_pgp_cert_path = cfg.require_kp_pgp_cert_path("key-provisioner ceremony")?;
66    let kp_cert = load_kp_cert(kp_pgp_cert_path)?;
67    certs_roster
68        .cert_for_fingerprint(&kp_cert.fingerprint())
69        .with_context(|| {
70            format!(
71                "this KP's cert (fingerprint {}) is not among the configured \
72                 kp_roster.kp_pgp_cert_paths",
73                kp_cert.fingerprint()
74            )
75        })?;
76    info!(
77        phase = "setup",
78        fingerprint = %kp_cert.fingerprint(),
79        "identified this KP's configured certificate",
80    );
81
82    // 1. Discover and verify the latest ceremony from the immutable log
83    //    (attestation-verified once via the reader's session-key cache).
84    info!(
85        phase = "s3 connect",
86        bucket = guardian_s3.bucket_name(),
87        region = guardian_s3.region(),
88        current_pcr0 = hex::encode(cfg.kp_roster.pcr_allowlist.current_build().pcr0()),
89        "connecting to guardian log bucket",
90    );
91    let mut reader = GuardianReader::new(&guardian_s3, cfg.kp_roster.pcr_allowlist())
92        .await
93        .context("connect to guardian log bucket")?;
94    info!(phase = "s3 connect", "connected to guardian log bucket");
95
96    info!(
97        phase = "ceremony scrape",
98        "scraping latest ceremony/ + kp-shares/ logs (attestation-anchored)",
99    );
100    let state = reader
101        .read_latest_ceremony_state_from_current_build()
102        .await?;
103    state.validate_sharing_params(cfg.kp_roster.num_shares, cfg.kp_roster.threshold)?;
104    info!(
105        phase = "ceremony scrape",
106        sharing_seq = state.secret_sharing_instance.sharing_seq(),
107        cert_seq = state.cert_seq,
108        n = state.secret_sharing_instance.num_shares(),
109        t = state.secret_sharing_instance.threshold(),
110        share_count = state.encrypted_shares.share_count(),
111        "discovered + validated latest ceremony state",
112    );
113
114    // 2. Confirm every PGP-encrypted share is addressed to the expected KP cert.
115    info!(
116        phase = "roster verify",
117        share_count = state.encrypted_shares.share_count(),
118        "verifying every PGP-encrypted share against the expected KP certs (without decrypting)",
119    );
120    state.encrypted_shares.verify_recipients(&certs_roster)?;
121    info!(
122        phase = "roster verify",
123        "ceremony/ and kp-shares/ logs verified against expected params and KP certs",
124    );
125
126    // 3. Decrypt and commitment-check this KP's ciphertext.
127    let reconstructed = decrypt_kp_share(&state, &kp_cert)?;
128    let share_id = reconstructed.id;
129    let expected_commitment = state
130        .secret_sharing_instance
131        .commitments()
132        .iter()
133        .find(|c| c.id == share_id)
134        .ok_or_else(|| {
135            anyhow!(
136                "commitment for share id {} missing despite verify_share success",
137                share_id
138            )
139        })?;
140    info!(
141        phase = "commitment verify",
142        share_id = share_id.get(),
143        commitment = hex::encode(&expected_commitment.digest),
144        "decrypted share matches its commitment",
145    );
146
147    // 4. Save the ceremony state only after every verification step succeeds.
148    let ceremony_state_bytes =
149        serde_json::to_vec(&state).context("serialize ceremony state with encrypted shares")?;
150    std::fs::write(encrypted_shares_path, ceremony_state_bytes).with_context(|| {
151        format!(
152            "write ceremony state with encrypted shares to {}",
153            encrypted_shares_path.display()
154        )
155    })?;
156    info!(
157        phase = "share save",
158        path = %encrypted_shares_path.display(),
159        share_count = state.encrypted_shares.share_count(),
160        "saved ceremony state with encrypted shares",
161    );
162
163    // 5. Submit a signed confirmation only after the verified recovery artifact
164    //    is safely stored locally.
165    info!(
166        phase = "confirmation",
167        endpoint = %cfg.guardian_endpoint,
168        "connecting to live ceremony guardian",
169    );
170    let mut client = GuardianServiceClient::connect(cfg.guardian_endpoint.clone())
171        .await
172        .with_context(|| format!("connect to ceremony guardian at {}", cfg.guardian_endpoint))?;
173    let verified =
174        verified_live_guardian_info(&mut client, cfg.kp_roster.pcr_allowlist.current_build())
175            .await?;
176    ensure!(
177        verified.info.lifecycle == CeremonyStage::AwaitingKeyProvisionerConfirmations.into()
178            || verified.info.lifecycle == CeremonyStage::Completed.into(),
179        "guardian is not accepting key provisioner ceremony confirmations"
180    );
181    let kp_fingerprint = kp_cert.fingerprint();
182    let confirmation = CeremonyConfirmationRequest::new(verified.session_id, state.digest());
183    let signed = KpSigned::sign(confirmation, kp_cert, None)
184        .map_err(anyhow::Error::msg)
185        .context("sign ceremony confirmation with the KP key")?;
186    let status = CeremonyConfirmationResponse::try_from(
187        client
188            .confirm_ceremony(signed_ceremony_confirmation_request_to_pb(signed))
189            .await
190            .context("ConfirmCeremony RPC failed")?
191            .into_inner(),
192    )
193    .map_err(|error| anyhow!("decode CeremonyConfirmationResponse: {error:?}"))?;
194    info!(
195        phase = "confirmation",
196        share_id = share_id.get(),
197        have = status.have,
198        need = status.need,
199        completed = status.completed,
200        "ceremony confirmation progress",
201    );
202
203    info!(
204        phase = "summary",
205        share_id = share_id.get(),
206        sharing_seq = state.secret_sharing_instance.sharing_seq(),
207        cert_seq = state.cert_seq,
208        fingerprint = %kp_fingerprint,
209        commitment = hex::encode(&expected_commitment.digest),
210        "ceremony share verified through this KP's configured certificate",
211    );
212    Ok(())
213}
