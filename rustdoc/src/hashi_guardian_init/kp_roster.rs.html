4//! Shared KP-roster config and ceremony-state verification for key-provisioner
5//! commands.
6//!
7//! These commands need to:
8//! - load a roster of KP OpenPGP certs
9//! - discover the latest attested ceremony from S3
10//! - validate the ceremony's `secret_sharing_instance` against expected params
11//! - confirm every PGP-encrypted share in `kp-shares/` matches the
12//!   expected certificate for that KP/share id (without decrypting)
13//!
14//! The decryption helper lives here so all commands share the same
15//! gpg-streaming pattern.
16

40/// Common KP-roster config: the sharing params, the full KP cert roster, and the
41/// PCR allowlist. Shared by every command that needs to discover and verify a
42/// ceremony against an expected KP set.
43#[derive(Deserialize)]
44pub struct KpRosterConfig {
45    /// Total number of shares/KPs. Must equal `kp_pgp_cert_paths.len()`.
46    pub num_shares: usize,
47    /// Reconstruction threshold. Must satisfy `2 <= threshold <= num_shares`.
48    pub threshold: usize,
49    /// Ordered paths to each KP's armored OpenPGP public certificate. The path
50    /// at index `i` is assigned share id `i + 1`; read-only commands match
51    /// shares by fingerprint.
52    pub kp_pgp_cert_paths: Vec<PathBuf>,
53    #[serde(flatten)]
54    pub pcr_allowlist: PcrAllowlist,
55}
56
57impl KpRosterConfig {
58    pub fn validate(&self) -> Result<()> {
59        SecretSharingParams::new(self.num_shares, self.threshold)
60            .map_err(|e| anyhow!("invalid sharing params: {e:?}"))?;
61
62        let cert_path_count = self.kp_pgp_cert_paths.len();
63        anyhow::ensure!(
64            self.num_shares == cert_path_count,
65            "num_shares ({}) must equal the number of KP cert paths ({cert_path_count})",
66            self.num_shares
67        );
68        Ok(())
69    }
70
71    pub fn load_certs_roster(&self) -> Result<KpCertRoster> {
72        let certs = self
73            .kp_pgp_cert_paths
74            .iter()
75            .enumerate()
76            .map(|(idx, cert_path)| {
77                load_cert(cert_path)
78                    .with_context(|| format!("invalid KP cert at roster position {}", idx + 1))
79            })
80            .collect::<Result<Vec<_>>>()?;
81        KpCertRoster::new(certs).context("invalid KP certificate roster")
82    }
83
84    /// The PCR allowlist decoded from `current_build` + `prev_builds`.
85    pub fn pcr_allowlist(&self) -> PcrAllowlist {
86        self.pcr_allowlist.clone()
87    }
88}

90fn load_cert(path: &PathBuf) -> Result<PgpPublicCert> {
91    let armored = std::fs::read_to_string(path)
92        .with_context(|| format!("failed to read PGP cert at {}", path.display()))?;
93    let cert = PgpPublicCert::new(armored)
94        .with_context(|| format!("invalid PGP cert at {}", path.display()))?;
95    info!(fingerprint = %cert, path = %path.display(), "loaded PGP cert");
96    Ok(cert)
97}

99/// Find, decrypt, and commitment-check the share addressed to `kp_cert`.
100pub fn decrypt_kp_share(state: &CeremonyState, kp_cert: &PgpPublicCert) -> Result<DecryptedShare> {
101    decrypt_kp_share_with(state, kp_cert, decrypt_pgp_ciphertext)
102}
103
104fn decrypt_kp_share_with(
105    state: &CeremonyState,
106    kp_cert: &PgpPublicCert,
107    decrypt: impl FnOnce(ShareID, &str) -> Result<DecryptedShare>,
108) -> Result<DecryptedShare> {
109    let fingerprint = kp_cert.fingerprint().to_hex();
110    let encrypted_share = state
111        .encrypted_shares
112        .find_by_fingerprint(&fingerprint)
113        .ok_or_else(|| {
114            anyhow!(
115                "no share in the kp-shares log is addressed to this KP's \
116                 fingerprint {fingerprint} (recipients by share: {:?})",
117                state.encrypted_shares.recipient_fingerprints()
118            )
119        })?;
120    let share_id = encrypted_share.id;
121
122    info!(
123        phase = "share decrypt",
124        share_id = share_id.get(),
125        fingerprint = %fingerprint,
126        "decrypting encrypted share via yubikey"
127    );
128    let decrypted = decrypt(share_id, &encrypted_share.armored_ciphertext).with_context(|| {
129        format!(
130            "decrypt share id {} for fingerprint {fingerprint}",
131            share_id.get()
132        )
133    })?;
134    state
135        .secret_sharing_instance
136        .commitments()
137        .verify_share(&decrypted)
138        .with_context(|| {
139            format!(
140                "decrypted share id {} for fingerprint {fingerprint} does not match its commitment",
141                share_id.get()
142            )
143        })?;
144    info!(
145        phase = "share decrypt",
146        share_id = share_id.get(),
147        fingerprint = %fingerprint,
148        "decrypted and verified encrypted share"
149    );
150
151    Ok(decrypted)
152}

154/// Load the cert selected by this KP for the current command.
155pub fn load_kp_cert(path: &Path) -> Result<PgpPublicCert> {
156    let cert = PgpPublicCert::new(
157        std::fs::read_to_string(path)
158            .with_context(|| format!("read KP cert at {}", path.display()))?,
159    )
160    .with_context(|| format!("invalid PGP cert at {}", path.display()))?;
161    info!(
162        fingerprint = %cert.fingerprint(),
163        path = %path.display(),
164        "loaded this KP's cert"
165    );
166    Ok(cert)
167}

169/// Decrypt a KP's selected PGP-encrypted share via the yubikey-backed gpg
170/// agent, returning the share wrapped in a [`DecryptedShare`] that wipes its
171/// scalar on drop. Nothing touches disk: gpg reads the ciphertext from a piped
172/// stdin and streams the plaintext over its stdout pipe into memory.
173///
174/// **Zeroization scope:** the gpg plaintext bytes, the intermediate scalar
175/// byte array, and the returned wrapper's inner [`Scalar`] are zeroed on drop.
176/// `k256::Scalar` is `Copy`, so the compiler may produce transient stack
177/// copies (e.g. inside `verify_share` / `build_from_share`) that this can't
178/// reach — those are wiped only when the process exits. The named locations
179/// this code owns are wiped deterministically.
180pub fn decrypt_pgp_ciphertext(share_id: ShareID, ciphertext: &str) -> Result<DecryptedShare> {
181    let plaintext = Zeroizing::new(decrypt_armored_via_gpg(ciphertext, None)?);
182    let scalar = scalar_from_decrypted_plaintext(&plaintext)?;
183    Ok(DecryptedShare(Share {
184        id: share_id,
185        value: scalar,
186    }))
187}
188
189/// Owning wrapper around a decrypted [`Share`] that wipes the scalar value on
190/// drop. Use `&*share` to access the inner [`Share`] for commitment
191/// verification / re-encryption. See [`decrypt_pgp_ciphertext`] for the
192/// zeroization scope.
193pub struct DecryptedShare(Share);
194
195impl Deref for DecryptedShare {
196    type Target = Share;
197
198    fn deref(&self) -> &Share {
199        &self.0
200    }
201}
202
203impl Drop for DecryptedShare {
204    fn drop(&mut self) {
205        self.0.value.zeroize();
206    }
207}

209/// Parse the decrypted plaintext bytes into a secp256k1 scalar. Extracted from
210/// [`decrypt_pgp_ciphertext`] so the byte-length and canonical-scalar checks are
211/// unit-testable without invoking gpg.
212fn scalar_from_decrypted_plaintext(plaintext: &[u8]) -> Result<Scalar> {
213    let src: &[u8; 32] = plaintext
214        .try_into()
215        .map_err(|_| anyhow!("decrypted share is {} bytes, expected 32", plaintext.len()))?;
216    let mut scalar_bytes = Zeroizing::new([0u8; 32]);
217    scalar_bytes.copy_from_slice(src);
218    Option::<Scalar>::from(Scalar::from_repr(FieldBytes::from(*scalar_bytes)))
219        .ok_or_else(|| anyhow!("decrypted share is not a valid secp256k1 scalar"))
220}

244    fn decryptable_state(certs_by_share: &[&PgpPublicCert]) -> (CeremonyState, Vec<Share>) {
245        let shares = certs_by_share
246            .iter()
247            .enumerate()
248            .map(|(i, _)| Share {
249                id: NonZeroU16::new((i + 1) as u16).unwrap(),
250                value: Scalar::from((i + 11) as u64),
251            })
252            .collect::<Vec<_>>();
253        let encrypted_shares = certs_by_share
254            .iter()
255            .zip(&shares)
256            .map(|(cert, share)| KpEncryptedShare {
257                id: share.id,
258                recipient_fingerprint: cert.fingerprint().to_hex(),
259                armored_ciphertext: format!("ciphertext-for-share-{}", share.id.get()),
260            })
261            .collect();
262        let response = SetupNewKeyResponse {
263            encrypted_shares: KpEncryptedShareRoster::new(encrypted_shares).unwrap(),
264            secret_sharing_instance: SecretSharingInstance::new(
265                ShareCommitments::from_shares(&shares).unwrap(),
266                shares.len(),
267                2,
268                0,
269            )
270            .unwrap(),
271            btc_master_pubkey: dummy_btc_pubkey(),
272        };
273        (CeremonyState::from(response), shares)
274    }

385    #[test]
386    fn scalar_from_decrypted_plaintext_rejects_non_canonical() {
387        // secp256k1 order n = 0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEBAAEDCE6AF48A03BBFD25E8CD0364141.
388        // A 32-byte value >= n is non-canonical and must be rejected.
389        let mut over_order = [0xFFu8; 32];
390        over_order[31] = 0x42; // > 0x41 (low byte of n)
391        assert!(scalar_from_decrypted_plaintext(&over_order).is_err());
392    }
