34// The retry window is long enough for the next scheduled heartbeat to start a
35// complete S3 attempt: one heartbeat interval plus five minutes. A not-live
36// session may therefore still recover before the ten-minute window expires.
37const CURRENT_SESSION_HEARTBEAT_RETRY_WINDOW: Duration = Duration::from_mins(10);
38const _: () = assert!(
39    HEARTBEAT_INTERVAL.as_secs() + S3_WRITE_ATTEMPT_TIMEOUT.as_secs()
40        < CURRENT_SESSION_HEARTBEAT_RETRY_WINDOW.as_secs()
41);
42
43// Allow a prior guardian time to stop while retaining enough time to observe
44// the complete quiet period afterward.
45const ACTIVATION_HEARTBEAT_WAIT_BUFFER: Duration = Duration::from_mins(5);

47/// Activate a provisioner-initialized standby guardian.
48pub async fn run(cfg: Config) -> anyhow::Result<()> {
49    cfg.kp_roster.validate()?;
50    let guardian_s3 = hashi_guardian::resolve_s3_config(&cfg.guardian_s3).await?;
51    let allowlist = cfg.kp_roster.pcr_allowlist();
52
53    info!(
54        phase = "setup",
55        bucket = guardian_s3.bucket_name(),
56        region = guardian_s3.region(),
57        endpoint = %cfg.guardian_endpoint,
58        bitcoin_network = ?cfg.bitcoin_network,
59        limiter_refill_rate = cfg.limiter_config.refill_rate,
60        limiter_max_capacity = cfg.limiter_config.max_bucket_capacity,
61        "running operator activate flow",
62    );
63
64    info!(
65        phase = "s3 connect",
66        bucket = guardian_s3.bucket_name(),
67        region = guardian_s3.region(),
68        current_git_revision = %allowlist.current_build().git_revision(),
69        current_pcr0 = hex::encode(allowlist.current_build().pcr0()),
70        prev_build_count = allowlist.prev_builds().len(),
71        "connecting to guardian log bucket",
72    );
73    let mut reader = GuardianReader::new(&guardian_s3, allowlist.clone())
74        .await
75        .context("connect to guardian log bucket")?;
76    info!(phase = "s3 connect", "connected to guardian log bucket");
77
78    info!(
79        phase = "sui connect",
80        sui_rpc = %cfg.hashi.sui_rpc,
81        package_id = %cfg.hashi.hashi_ids.package_id,
82        hashi_object_id = %cfg.hashi.hashi_ids.hashi_object_id,
83        "connecting to Sui RPC for Hashi on-chain state",
84    );
85    let onchain_state = cfg.hashi.onchain_state().await?;
86    info!(phase = "sui connect", "connected to Sui RPC");
87
88    let master_g = onchain_state.onchain_verifying_key_g()?;
89    info!(phase = "setup", master_g = ?master_g, "fetched on-chain MPC master G");
90
91    info!(
92        phase = "guardian connect",
93        endpoint = %cfg.guardian_endpoint,
94        "connecting to withdraw-mode guardian",
95    );
96    let mut client = GuardianServiceClient::connect(cfg.guardian_endpoint.clone())
97        .await
98        .with_context(|| format!("connect to guardian at {}", cfg.guardian_endpoint))?;
99    info!(phase = "guardian connect", endpoint = %cfg.guardian_endpoint, "connected to guardian");
100
101    info!(
102        phase = "guardian preflight",
103        "fetching + verifying provisioned standby GuardianInfo"
104    );
105    let preflight = verified_live_guardian_info(&mut client, allowlist.current_build()).await?;
106    let session_id = preflight.session_id.clone();
107    let signing_pub_key = preflight.signing_pub_key;
108    let pre_info = preflight.info.clone();
109    let standby =
110        verify_provisioned_standby_info(&pre_info, &guardian_s3, &cfg, &allowlist, &master_g)?;
111    info!(
112        phase = "guardian preflight",
113        session_id = %session_id,
114        signing_pubkey = hex::encode(signing_pub_key.as_bytes()),
115        config_hash = hex::encode(standby.config_hash),
116        enclave_btc_pubkey = ?standby.enclave_btc_pubkey,
117        "guardian is current-build, provisioned, and not yet active",
118    );
119
120    info!(
121        phase = "attestation pin",
122        session_id = %session_id,
123        "verifying OI GuardianInfo matches the live guardian identity/config",
124    );
125    let verified_session = reader.get_current_session_info(&session_id).await?;
126    ensure!(
127        verified_session.signing_pubkey() == &signing_pub_key,
128        "guardian S3 attestation signing pubkey differs from gRPC signing pubkey"
129    );
130    verify_oi_info_matches_provisioned_standby(verified_session.info(), &pre_info)?;
131    info!(
132        phase = "attestation pin",
133        session_id = %session_id,
134        "OI GuardianInfo matches live standby identity/config",
135    );
136
137    info!(
138        phase = "heartbeat check",
139        session_id = %session_id,
140        "checking that the standby session is live and all other sessions are quiet",
141    );
142    wait_for_activation_heartbeat_conditions(&mut reader, &session_id)
143        .await
144        .context("heartbeat activation check failed")?;
145
146    info!(
147        phase = "activation state",
148        "reading latest committee and recovering limiter state",
149    );
150    let move_committee = reader
151        .read_latest_committee()
152        .await?
153        .context("no committee-update or genesis record found")?;
154    let committee_epoch = move_committee.epoch;
155    let committee: HashiCommittee = move_committee
156        .try_into()
157        .context("invalid serving committee")?;
158    let limiter_state = reader
159        .recover_limiter_state(standby.init_config.limiter_config())
160        .await
161        .context("recover limiter state")?;
162    let activation_state = ActivationState::new(
163        standby.config_hash,
164        standby.secret_sharing_instance.clone(),
165        committee,
166        limiter_state,
167    );
168    let state_hash = activation_state.digest();
169    info!(
170        phase = "activation state",
171        committee_epoch,
172        limiter_next_seq = limiter_state.next_seq,
173        limiter_tokens_available = limiter_state.num_tokens_available,
174        limiter_last_updated_at = limiter_state.last_updated_at,
175        state_hash = hex::encode(state_hash),
176        "computed expected ActivationState hash",
177    );
178
179    info!(
180        phase = "operator_activate",
181        session_id = %session_id,
182        committee_epoch,
183        limiter_next_seq = limiter_state.next_seq,
184        state_hash = hex::encode(state_hash),
185        "calling OperatorActivate",
186    );
187    let activate_req = operator_activate_request_to_pb(OperatorActivateRequest::new(state_hash));
188    client
189        .operator_activate(activate_req)
190        .await
191        .context("OperatorActivate RPC failed")?;
192    info!(
193        phase = "operator_activate",
194        session_id = %session_id,
195        "operator_activate RPC returned; verifying activated GuardianInfo",
196    );
197
198    info!(
199        phase = "guardian postcheck",
200        session_id = %session_id,
201        "fetching + verifying activated GuardianInfo"
202    );
203    let post = verified_live_guardian_info(&mut client, allowlist.current_build()).await?;
204    verify_activated_info(
205        post,
206        &session_id,
207        signing_pub_key,
208        standby.enclave_btc_pubkey,
209        committee_epoch,
210        limiter_state,
211    )?;
212    info!(
213        phase = "guardian postcheck",
214        session_id = %session_id,
215        committee_epoch,
216        limiter_next_seq = limiter_state.next_seq,
217        "activated GuardianInfo matches expected state",
218    );
219
220    info!(
221        phase = "summary",
222        session_id = %session_id,
223        committee_epoch,
224        config_hash = hex::encode(standby.config_hash),
225        state_hash = hex::encode(state_hash),
226        "operator activate complete",
227    );
228    println!("Guardian operator activate complete.");
229    println!("  session_id:       {session_id}");
230    println!("  config_hash:      {}", hex::encode(standby.config_hash));
231    println!("  state_hash:       {}", hex::encode(state_hash));
232    println!("  committee_epoch:  {committee_epoch}");
233    println!("  limiter_next_seq: {}", limiter_state.next_seq);
234    println!("  bitcoin_network:  {}", cfg.bitcoin_network);
235    println!("  bucket:           {}", guardian_s3.bucket_name());
236    println!("  region:           {}", guardian_s3.region());
237
238    Ok(())
239}

241async fn wait_for_activation_heartbeat_conditions(
242    reader: &mut GuardianReader,
243    session_id: &str,
244) -> GuardianResult<()> {
245    let now = Instant::now();
246    let current_session_deadline = now + CURRENT_SESSION_HEARTBEAT_RETRY_WINDOW;
247    let prior_session_deadline =
248        now + OTHER_SESSION_QUIET_PERIOD + ACTIVATION_HEARTBEAT_WAIT_BUFFER;
249    loop {
250        match reader
251            .ensure_session_live_and_others_quiet(session_id)
252            .await
253        {
254            Ok(()) => return Ok(()),
255            Err(error) => {
256                let deadline = match &error {
257                    GuardianError::CurrentSessionHeartbeatNotLive { .. } => {
258                        current_session_deadline
259                    }
260                    GuardianError::PriorSessionHeartbeatStillRecent { .. } => {
261                        prior_session_deadline
262                    }
263                    _ => return Err(error),
264                };
265                let Some(retry_after_secs) = error.retry_after_secs() else {
266                    return Err(error);
267                };
268                let retry_delay = Duration::from_secs(retry_after_secs);
269
270                let now = Instant::now();
271                if now >= deadline {
272                    return Err(error);
273                }
274
275                let retry_at = (now + retry_delay).min(deadline);
276                warn!(
277                    error = %error,
278                    retry_in_secs = retry_at.duration_since(now).as_secs(),
279                    "activation heartbeat conditions are not ready; retrying"
280                );
281                sleep_until(retry_at).await;
282            }
283        }
284    }
285}

287struct StandbyChecks {
288    init_config: InitConfig,
289    config_hash: [u8; 32],
290    secret_sharing_instance: hashi_types::guardian::SecretSharingInstance,
291    enclave_btc_pubkey: hashi_types::bitcoin::BitcoinPubkey,
292}

294fn verify_provisioned_standby_info(
295    info: &GuardianInfo,
296    guardian_s3: &ResolvedS3Config,
297    cfg: &Config,
298    allowlist: &hashi_types::guardian::PcrAllowlist,
299    master_g: &hashi_types::bitcoin::HashiMasterG,
300) -> anyhow::Result<StandbyChecks> {
301    ensure!(
302        info.lifecycle == WithdrawStage::ProvisionerInitialized.into(),
303        "guardian is not a provisioner-initialized withdraw enclave"
304    );
305    let instance = info
306        .secret_sharing_instance
307        .clone()
308        .context("Guardian info missing secret-sharing instance")?;
309    let bucket_info = info
310        .bucket_info
311        .as_ref()
312        .context("Guardian info missing bucket info")?;
313    let config_hash = info
314        .config_hash
315        .context("Guardian info missing config_hash")?;
316    let enclave_btc_pubkey = info
317        .enclave_btc_pubkey
318        .context("Guardian info missing enclave BTC pubkey; provisioner_init is not complete")?;
319    let limiter_config = info
320        .limiter_config
321        .context("Guardian info missing limiter config")?;
322    let mpc_master_g = info
323        .mpc_master_g
324        .context("Guardian info missing MPC master G")?;
325    ensure!(
326        info.limiter_state.is_none(),
327        "Guardian has limiter_state => operator activation already ran"
328    );
329    ensure!(
330        info.current_committee_epoch.is_none(),
331        "Guardian has current_committee_epoch => operator activation already ran"
332    );
333    ensure!(
334        &guardian_s3.bucket_info == bucket_info,
335        "Guardian bucket info mismatch: expected {:?}, got {:?}",
336        guardian_s3.bucket_info,
337        bucket_info
338    );
339    ensure!(
340        cfg.limiter_config == limiter_config,
341        "Guardian limiter config mismatch: expected {:?}, got {:?}",
342        cfg.limiter_config,
343        limiter_config
344    );
345    ensure!(
346        master_g == &mpc_master_g,
347        "Guardian MPC master G mismatch: expected {:?}, got {:?}",
348        master_g,
349        mpc_master_g
350    );
351    let init_config = InitConfig::new(
352        cfg.limiter_config,
353        *master_g,
354        allowlist.clone(),
355        guardian_s3.bucket_info.clone(),
356        guardian_s3.retention_environment,
357        cfg.bitcoin_network,
358        cfg.hashi.hashi_ids.hashi_object_id,
359    )?;
360    let expected_config_hash = init_config.digest();
361    ensure!(
362        expected_config_hash == config_hash,
363        "Guardian config_hash mismatch: expected {}, got {}",
364        hex::encode(expected_config_hash),
365        hex::encode(config_hash)
366    );
367
368    Ok(StandbyChecks {
369        init_config,
370        config_hash,
371        secret_sharing_instance: instance,
372        enclave_btc_pubkey,
373    })
374}

376fn verify_oi_info_matches_provisioned_standby(
377    oi_info: &GuardianInfo,
378    live_info: &GuardianInfo,
379) -> anyhow::Result<()> {
380    ensure!(
381        oi_info.lifecycle == WithdrawStage::Uninitialized.into(),
382        "OI GuardianInfo has an unexpected lifecycle stage"
383    );
384    ensure!(
385        oi_info.lifecycle.mode() == live_info.lifecycle.mode(),
386        "OI GuardianInfo enclave mode differs from live standby GuardianInfo"
387    );
388    ensure!(
389        oi_info.enclave_btc_pubkey.is_none(),
390        "OI GuardianInfo unexpectedly has a BTC pubkey"
391    );
392    ensure!(
393        oi_info.limiter_state.is_none(),
394        "OI GuardianInfo unexpectedly has limiter_state"
395    );
396    ensure!(
397        oi_info.current_committee_epoch.is_none(),
398        "OI GuardianInfo unexpectedly has current_committee_epoch"
399    );
400    ensure!(
401        oi_info.secret_sharing_instance == live_info.secret_sharing_instance,
402        "OI GuardianInfo secret-sharing instance differs from live standby GuardianInfo"
403    );
404    ensure!(
405        oi_info.bucket_info == live_info.bucket_info,
406        "OI GuardianInfo bucket info differs from live standby GuardianInfo"
407    );
408    ensure!(
409        oi_info.encryption_pubkey == live_info.encryption_pubkey,
410        "OI GuardianInfo encryption pubkey differs from live standby GuardianInfo"
411    );
412    ensure!(
413        oi_info.config_hash == live_info.config_hash,
414        "OI GuardianInfo config_hash differs from live standby GuardianInfo"
415    );
416    ensure!(
417        oi_info.genesis_state_hash == live_info.genesis_state_hash,
418        "OI GuardianInfo genesis_state_hash differs from live standby GuardianInfo"
419    );
420    ensure!(
421        oi_info.untrusted_git_revision == live_info.untrusted_git_revision,
422        "OI GuardianInfo git revision differs from live standby GuardianInfo"
423    );
424    ensure!(
425        oi_info.limiter_config == live_info.limiter_config,
426        "OI GuardianInfo limiter config differs from live standby GuardianInfo"
427    );
428    ensure!(
429        oi_info.mpc_master_g == live_info.mpc_master_g,
430        "OI GuardianInfo MPC master G differs from live standby GuardianInfo"
431    );
432    Ok(())
433}

435fn verify_activated_info(
436    post: VerifiedGuardianInfo,
437    expected_session_id: &str,
438    expected_signing_key: hashi_types::guardian::GuardianPubKey,
439    expected_enclave_btc_pubkey: hashi_types::bitcoin::BitcoinPubkey,
440    expected_committee_epoch: u64,
441    expected_limiter_state: hashi_types::guardian::LimiterState,
442) -> anyhow::Result<()> {
443    ensure!(
444        post.session_id.as_str() == expected_session_id,
445        "guardian session changed during operator activation: started {}, now {}",
446        expected_session_id,
447        post.session_id
448    );
449    ensure!(
450        post.signing_pub_key == expected_signing_key,
451        "guardian signing key changed during operator activation"
452    );
453    ensure!(
454        post.info.lifecycle == WithdrawStage::Activated.into(),
455        "guardian is not an activated withdraw enclave"
456    );
457    ensure!(
458        post.info.enclave_btc_pubkey == Some(expected_enclave_btc_pubkey),
459        "Guardian BTC pubkey changed during operator activation"
460    );
461    ensure!(
462        post.info.current_committee_epoch == Some(expected_committee_epoch),
463        "Guardian committee epoch mismatch: expected {}, got {:?}",
464        expected_committee_epoch,
465        post.info.current_committee_epoch
466    );
467    ensure!(
468        post.info.limiter_state == Some(expected_limiter_state),
469        "Guardian limiter state mismatch: expected {:?}, got {:?}",
470        expected_limiter_state,
471        post.info.limiter_state
472    );
473    Ok(())
