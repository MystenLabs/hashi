4//! UTXO and transaction types for Hashi Bitcoin withdrawals.
5//!
6//! Two classes of types exist here:
7//! - Domain types with checked addresses that implement Serialize but not Deserialize
8//! - `*Wire` types with unchecked addresses that implement both Serialize and Deserialize
9//!
10//! Untrusted input arrives as the `*Wire` form; `TxUTXOs::new` is the single gate
11//! that validates it (address-network, amounts, duplicates, fees) and converts it
12//! into the checked domain types.
13//!
14//! Internal-output addresses are derived via `super::taproot`.
15

128/// Represents an input to be spent.
129///
130/// All inputs are expected to be P2TR (Pay-to-Taproot) since spending is done via taproot script path.
131impl InputUTXO {
132    pub fn new(outpoint: OutPoint, amount: Amount, derivation_path: DerivationPath) -> Self {
133        Self {
134            outpoint,
135            amount,
136            derivation_path,
137        }
138    }
139
140    /// Returns a `TxIn` for this UTXO with placeholder witness data.
141    ///
142    /// The witness will be populated later after signing.
143    pub fn txin(&self) -> TxIn {
144        TxIn {
145            previous_output: self.outpoint,
146            // No script sig needed for taproot
147            script_sig: ScriptBuf::default(),
148            // Enables RBF, disables relative lock time, allows absolute lock time
149            sequence: Sequence::ENABLE_RBF_NO_LOCKTIME,
150            // Witness will be set later
151            witness: Witness::default(),
152        }
153    }
154
155    /// Prevout `TxOut` and tap leaf hash for sighash, derived from the path.
156    fn prevout_and_leaf_hash(
157        &self,
158        enclave_pubkey: &BitcoinPubkey,
159        hashi_master_g: &HashiMasterG,
160    ) -> (TxOut, TapLeafHash) {
161        let (script_pubkey, leaf_hash) = taproot_script_pubkey_and_leaf_hash(
162            enclave_pubkey,
163            hashi_master_g,
164            &self.derivation_path,
165        );
166        (
167            TxOut {
168                value: self.amount,
169                script_pubkey,
170            },
171            leaf_hash,
172        )
173    }
174}

211/// Represents an output destination for a withdrawal.
212///
213/// Outputs can be **external** (to a user-provided address) or **internal** (change, derived inside enclave).
214impl OutputUTXO {
215    /// Returns the output amount in satoshis.
216    fn amount(&self) -> Amount {
217        match self {
218            OutputUTXO::External(ExternalOutputUTXO { amount, .. }) => *amount,
219            OutputUTXO::Internal(InternalOutputUTXO { amount, .. }) => *amount,
220        }
221    }
222
223    /// Builds the `TxOut`: external uses the checked address, internal derives
224    /// its taproot script via `super::taproot`.
225    fn to_txout(&self, enclave_pubkey: &BitcoinPubkey, hashi_master_g: &HashiMasterG) -> TxOut {
226        match self {
227            OutputUTXO::External(ExternalOutputUTXO { address, amount }) => TxOut {
228                value: *amount,
229                script_pubkey: address.script_pubkey(),
230            },
231            OutputUTXO::Internal(InternalOutputUTXO {
232                derivation_path,
233                amount,
234            }) => {
235                let scripts = taproot_script_pubkey_and_leaf_hash(
236                    enclave_pubkey,
237                    hashi_master_g,
238                    derivation_path,
239                );
240                TxOut {
241                    value: *amount,
242                    script_pubkey: scripts.0,
243                }
244            }
245        }
246    }
247}

249/// Validates that an unchecked address is valid for `network` and returns a checked address.
250fn validate_address_for_network(
251    address: &BitcoinAddress<NetworkUnchecked>,
252    network: Network,
253) -> anyhow::Result<BitcoinAddress<NetworkChecked>> {
254    // Prefer the library's checked conversion to avoid accidentally assuming correctness.
255    address
256        .clone()
257        .require_network(network)
258        .map_err(|_| anyhow!("invalid address {:?} for network {}", address, network))
259}

261impl TxUTXOs {
262    /// Constructs a `TxUTXOs`, validating every invariant in one place: external
263    /// output addresses must be valid for `network`, amounts must be non-zero,
264    /// input and output totals must not overflow,
265    /// inputs must be unique, and fees must be positive. The single gate for both
266    /// locally-built and wire-parsed UTXO sets.
267    pub fn new(
268        inputs: Vec<InputUTXO>,
269        outputs: Vec<OutputUTXOWire>,
270        network: Network,
271    ) -> anyhow::Result<Self> {
272        if inputs.is_empty() {
273            anyhow::bail!("input utxos must not be empty");
274        }
275        if outputs.is_empty() {
276            anyhow::bail!("output utxos must not be empty");
277        }
278
279        // Validate each external address against the network, turning untrusted
280        // unchecked addresses into checked ones. Internal outputs carry no address.
281        let mut checked = Vec::with_capacity(outputs.len());
282        for output in outputs {
283            checked.push(match output {
284                OutputUTXOWire::External(ext) => OutputUTXO::External(ExternalOutputUTXO {
285                    address: validate_address_for_network(&ext.address, network)?,
286                    amount: ext.amount,
287                }),
288                OutputUTXOWire::Internal(int) => OutputUTXO::Internal(int),
289            });
290        }
291        let outputs = checked;
292
293        // Reject zero amounts on both sides: a 0-value input is meaningless and a
294        // 0-value output is invalid on Bitcoin.
295        for utxo in &inputs {
296            if utxo.amount == Amount::ZERO {
297                anyhow::bail!("input amount must be > 0");
298            }
299        }
300        for utxo in &outputs {
301            if utxo.amount() == Amount::ZERO {
302                anyhow::bail!("output amount must be > 0");
303            }
304        }
305
306        // Disallow duplicate inputs (same txid,vout), which would result in an invalid transaction.
307        let mut seen_inputs: HashSet<OutPoint> = HashSet::with_capacity(inputs.len());
308        for utxo in &inputs {
309            if !seen_inputs.insert(utxo.outpoint) {
310                anyhow::bail!("duplicate input outpoint: {}", utxo.outpoint);
311            }
312        }
313
314        let tx_info = Self { inputs, outputs };
315
316        // Enforce the intended invariant: fees > 0.
317        tx_info.assert_positive_fees()?;
318
319        Ok(tx_info)
320    }
321
322    /// Constructs all outputs (both external and internal).
323    ///
324    /// For `External` outputs, uses the user-provided address. For `Internal` outputs,
325    /// derives a taproot address using the enclave and hashi keys.
326    pub fn compute_all_outputs(
327        &self,
328        enclave_pubkey: &BitcoinPubkey,
329        hashi_master_g: &HashiMasterG,
330    ) -> Vec<TxOut> {
331        self.outputs
332            .iter()
333            .map(|utxo| utxo.to_txout(enclave_pubkey, hashi_master_g))
334            .collect()
335    }
336
337    /// BTC that leaves the pool when this txn broadcasts: `inputs - change`,
338    /// equivalent to `external outputs + miner_fee`. The amount that consumes
339    /// the rate-limiter (miner fee leaves the pool too; change flows back).
340    pub fn gross_outflow_amount(&self) -> Amount {
341        let inputs: Amount = self.inputs.iter().map(|i| i.amount).sum();
342        let internal: Amount = self
343            .outputs
344            .iter()
345            .filter_map(|utxo| match utxo {
346                OutputUTXO::Internal(x) => Some(x.amount),
347                OutputUTXO::External(_) => None,
348            })
349            .sum();
350        inputs - internal
351    }
352
353    /// Constructs an unsigned Bitcoin transaction for these UTXOs.
354    fn unsigned_tx(
355        &self,
356        enclave_pubkey: &BitcoinPubkey,
357        hashi_master_g: &HashiMasterG,
358    ) -> Transaction {
359        let all_outputs = self.compute_all_outputs(enclave_pubkey, hashi_master_g);
360        construct_tx(
361            self.inputs.iter().map(|input| input.txin()).collect(),
362            all_outputs,
363        )
364    }
365
366    /// Constructs sighash messages for each input, ready for signing.
367    ///
368    /// Uses `taproot_script_spend_signature_hash` for script-path spending.
369    pub fn signing_messages_and_txid(
370        &self,
371        enclave_pubkey: &BitcoinPubkey,
372        hashi_master_g: &HashiMasterG,
373    ) -> (Vec<Message>, Txid) {
374        let tx = self.unsigned_tx(enclave_pubkey, hashi_master_g);
375        // Derive each input's prevout + tap leaf hash from its path.
376        let artifacts: Vec<(TxOut, TapLeafHash)> = self
377            .inputs
378            .iter()
379            .map(|input| input.prevout_and_leaf_hash(enclave_pubkey, hashi_master_g))
380            .collect();
381        let prevouts: Vec<TxOut> = artifacts.iter().map(|(p, _)| p.clone()).collect();
382        let leaf_hashes: Vec<TapLeafHash> = artifacts.iter().map(|(_, h)| *h).collect();
383
384        let messages = taproot_script_spend_sighashes(&tx, &prevouts, &leaf_hashes)
385            .into_iter()
386            .map(Message::from_digest)
387            .collect();
388        (messages, tx.compute_txid())
389    }
390
391    fn assert_positive_fees(&self) -> anyhow::Result<()> {
392        let input_sum = self
393            .inputs
394            .iter()
395            .map(|utxo| utxo.amount)
396            .checked_sum()
397            .context("total input amount overflows")?;
398        let output_sum = self
399            .outputs
400            .iter()
401            .map(|utxo| utxo.amount())
402            .checked_sum()
403            .context("total output amount overflows")?;
404        if input_sum <= output_sum {
405            anyhow::bail!(
406                "fees must be greater than zero: input_sum={} output_sum={}",
407                input_sum,
408                output_sum
409            );
410        }
411        Ok(())
412    }
413}

419/// Signs messages using Schnorr signatures (suitable for taproot script-spend).
420///
421/// Each message is signed and wrapped in a `Signature` with `TapSighashType::Default`.
422pub fn sign_btc_tx(messages: &[Message], kp: &BitcoinKeypair) -> Vec<BitcoinSignature> {
423    messages
424        .iter()
425        // Fresh auxiliary randomness provides additional side-channel protection.
426        .map(|m| BTC_LIB.sign_schnorr_with_aux_rand(m, kp, &rand::random()))
427        .map(|s| Signature {
428            signature: s,
429            sighash_type: TapSighashType::Default,
430        })
431        .collect()
432}

434/// Constructs a Bitcoin transaction with the given inputs and outputs.
435///
436/// Uses BTC tx version 2 and disables lock time.
437pub fn construct_tx(inputs: Vec<TxIn>, outputs: Vec<TxOut>) -> Transaction {
438    Transaction {
439        // The latest BTC tx version
440        version: Version::TWO,
441        // Disable absolute lock time (i.e., can be mined immediately)
442        lock_time: LockTime::ZERO,
443        input: inputs,
444        output: outputs,
445    }
446}

479#[cfg(test)]
480mod tests {
481    use super::*;
482
483    fn utxos(inputs: &[u64], outputs: &[u64]) -> anyhow::Result<TxUTXOs> {
484        TxUTXOs::new(
485            inputs
486                .iter()
487                .enumerate()
488                .map(|(vout, amount)| {
489                    let mut outpoint = OutPoint::null();
490                    outpoint.vout = vout as u32;
491                    InputUTXO::new(outpoint, Amount::from_sat(*amount), DerivationPath::ZERO)
492                })
493                .collect(),
494            outputs
495                .iter()
496                .map(|amount| {
497                    OutputUTXOWire::internal(DerivationPath::ZERO, Amount::from_sat(*amount))
498                })
499                .collect(),
500            Network::Regtest,
501        )
502    }
503
504    #[test]
505    fn rejects_overflowing_amounts() {
506        assert!(utxos(&[u64::MAX, 2], &[1]).is_err());
507        // An unchecked output sum wraps to 1, passing the positive-fee check.
508        assert!(utxos(&[100], &[u64::MAX, 2]).is_err());
509    }
510
511    #[test]
512    fn accepts_non_overflowing_amounts_with_positive_fees() {
513        let max = u64::MAX;
514        let tx = utxos(&[max - 1, 1], &[max - 1]).unwrap();
515        assert_eq!(tx.gross_outflow_amount(), Amount::from_sat(1));
516    }
