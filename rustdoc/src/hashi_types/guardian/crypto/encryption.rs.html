26/// A key-provisioner's PGP fingerprint as bare uppercase hex — the string
27/// form persisted in ceremony artifacts. For comparing fingerprints, prefer
28/// the canonical [`crate::pgp::Fingerprint`].
29pub type KPFingerprint = String;
30
31/// A key-provisioner's PGP certificate and its verified YubiKey attestations.
32///
33/// Construction checks the certificate and all three PEM artifacts with
34/// [`crate::pgp::verify_yubikey_attestations`]. This records a successful check
35/// at construction, not ongoing freshness or a time-validity policy: the
36/// verifier does not enforce X.509 dates, revocation, touch, or freshness, or
37/// establish current possession of either private key.
38///
39/// Fields are immutable; deserialization repeats the pinned attestation checks.
40#[derive(Serialize, Debug, Clone, PartialEq)]
41pub struct AttestedKpCert {
42    cert: PgpPublicCert,
43    device_pem: Vec<u8>,
44    sig_pem: Vec<u8>,
45    dec_pem: Vec<u8>,
46    #[serde(skip)]
47    keys: AttestedPgpKeys,
48}

50impl AttestedKpCert {
51    /// Validate the attestation bundle using the pinned production trust policy.
52    pub fn new(
53        cert: PgpPublicCert,
54        device_pem: Vec<u8>,
55        sig_pem: Vec<u8>,
56        dec_pem: Vec<u8>,
57    ) -> GuardianResult<Self> {
58        let keys = crate::pgp::verify_yubikey_attestations_and_keys(
59            &cert,
60            &device_pem,
61            &sig_pem,
62            &dec_pem,
63        )
64        .map_err(|err| InvalidInputs(format!("invalid KP certificate attestations: {err:#}")))?;
65        Ok(Self {
66            cert,
67            device_pem,
68            sig_pem,
69            dec_pem,
70            keys,
71        })
72    }
73
74    pub fn cert(&self) -> &PgpPublicCert {
75        &self.cert
76    }
77
78    pub fn fingerprint(&self) -> Fingerprint {
79        self.cert.fingerprint()
80    }
81
82    pub(crate) fn signing_fingerprint(&self) -> &Fingerprint {
83        &self.keys.signing
84    }
85
86    pub(crate) fn encryption_fingerprint(&self) -> &Fingerprint {
87        &self.keys.encryption
88    }
89
90    pub fn device_pem(&self) -> &[u8] {
91        &self.device_pem
92    }
93
94    pub fn sig_pem(&self) -> &[u8] {
95        &self.sig_pem
96    }
97
98    pub fn dec_pem(&self) -> &[u8] {
99        &self.dec_pem
100    }
101
102    pub fn into_parts(self) -> (PgpPublicCert, Vec<u8>, Vec<u8>, Vec<u8>) {
103        (self.cert, self.device_pem, self.sig_pem, self.dec_pem)
104    }
105}

157impl KpCertRoster {
158    pub fn new(kp_certs: Vec<AttestedKpCert>) -> GuardianResult<Self> {
159        let mut seen = HashSet::with_capacity(kp_certs.len());
160        for cert in &kp_certs {
161            let fingerprint = cert.fingerprint();
162            if !seen.insert(fingerprint.clone()) {
163                return Err(InvalidInputs(format!(
164                    "duplicate OpenPGP certificate fingerprint {fingerprint}"
165                )));
166            }
167        }
168
169        Ok(Self(kp_certs))
170    }
171
172    pub fn num_kps(&self) -> usize {
173        self.0.len()
174    }
175
176    pub fn iter(&self) -> impl Iterator<Item = &AttestedKpCert> {
177        self.0.iter()
178    }
179
180    pub fn cert_for_share(&self, share_id: ShareID) -> Option<&AttestedKpCert> {
181        self.0.get(usize::from(share_id.get()) - 1)
182    }
183
184    pub fn cert_for_fingerprint(&self, fingerprint: &Fingerprint) -> Option<&AttestedKpCert> {
185        self.0
186            .iter()
187            .find(|cert| cert.fingerprint() == *fingerprint)
188    }
189
190    pub fn fingerprints(&self) -> Vec<KPFingerprint> {
191        self.0
192            .iter()
193            .map(|cert| cert.fingerprint().to_hex())
194            .collect()
195    }
196
197    pub fn into_vec(self) -> Vec<AttestedKpCert> {
198        self.0
199    }
200
201    /// Replace one certificate while preserving the KP/share ordering and
202    /// global fingerprint-uniqueness invariant.
203    pub fn replace_cert(
204        &self,
205        current_fingerprint: &Fingerprint,
206        new_cert: AttestedKpCert,
207    ) -> GuardianResult<Self> {
208        let new_fingerprint = new_cert.fingerprint();
209        if new_fingerprint == *current_fingerprint {
210            return Err(InvalidInputs(format!(
211                "replacement OpenPGP certificate fingerprint {new_fingerprint} must differ from \
212                 the current fingerprint"
213            )));
214        }
215
216        let mut kp_certs = self.0.clone();
217        let cert = kp_certs
218            .iter_mut()
219            .find(|cert| cert.fingerprint() == *current_fingerprint)
220            .ok_or_else(|| {
221                InvalidInputs(format!(
222                    "OpenPGP certificate fingerprint {current_fingerprint} is not in the KP \
223                     certificate roster"
224                ))
225            })?;
226        *cert = new_cert;
227        Self::new(kp_certs)
228    }
229}

231impl KpEncryptedShare {
232    /// Verify the recorded certificate identity and require every OpenPGP
233    /// recipient to identify its attested encryption key.
234    pub fn verify_recipient(&self, cert: &AttestedKpCert) -> GuardianResult<()> {
235        let expected_fingerprint = cert.fingerprint().to_hex();
236        if self.recipient_fingerprint != expected_fingerprint {
237            return Err(InvalidInputs(format!(
238                "encrypted share recipient differs for share id {}: expected {}, got {}",
239                self.id.get(),
240                expected_fingerprint,
241                self.recipient_fingerprint
242            )));
243        }
244        verify_pgp_ciphertext_recipient(
245            self.id,
246            &self.recipient_fingerprint,
247            &self.armored_ciphertext,
248            cert,
249        )
250    }
251}

253impl KpEncryptedShareRoster {
254    pub fn new(mut shares: Vec<KpEncryptedShare>) -> GuardianResult<Self> {
255        shares.sort_by_key(|s| s.id);
256
257        if shares.len() > MAX_NUM_SHARES {
258            return Err(InvalidInputs(format!(
259                "{} shares must be at most u16::MAX",
260                shares.len()
261            )));
262        }
263        if shares
264            .iter()
265            .enumerate()
266            .any(|(index, share)| usize::from(share.id.get()) != index + 1)
267        {
268            let ids = shares
269                .iter()
270                .map(|share| share.id.get())
271                .collect::<Vec<_>>();
272            return Err(InvalidInputs(format!(
273                "encrypted share ids are not exactly 1..={}: got {ids:?}",
274                shares.len()
275            )));
276        }
277
278        let mut seen_fingerprints: HashSet<&str> = HashSet::with_capacity(shares.len());
279        for share in &shares {
280            if !seen_fingerprints.insert(share.recipient_fingerprint.as_str()) {
281                return Err(InvalidInputs(format!(
282                    "duplicate encrypted share recipient fingerprint {}",
283                    share.recipient_fingerprint
284                )));
285            }
286        }
287
288        Ok(Self(shares))
289    }
290
291    pub fn share_count(&self) -> usize {
292        self.0.len()
293    }
294
295    pub fn iter(&self) -> impl Iterator<Item = &KpEncryptedShare> {
296        self.0.iter()
297    }
298
299    pub fn into_vec(self) -> Vec<KpEncryptedShare> {
300        self.0
301    }
302
303    pub fn find_by_fingerprint(&self, fingerprint: &str) -> Option<&KpEncryptedShare> {
304        self.iter()
305            .find(|share| share.recipient_fingerprint == fingerprint)
306    }

308    /// Require `fingerprint` to be assigned to `submitted_share_id` in this
309    /// encrypted-share roster.
310    pub fn validate_share_assignment(
311        &self,
312        fingerprint: &str,
313        submitted_share_id: ShareID,
314    ) -> GuardianResult<()> {
315        let assigned_share = self.find_by_fingerprint(fingerprint).ok_or_else(|| {
316            InvalidInputs(format!(
317                "KP fingerprint {fingerprint} is not present in the encrypted-share roster"
318            ))
319        })?;
320        if assigned_share.id != submitted_share_id {
321            return Err(InvalidInputs(format!(
322                "KP fingerprint {fingerprint} is assigned share id {}, not submitted share id {}",
323                assigned_share.id.get(),
324                submitted_share_id.get()
325            )));
326        }
327        Ok(())
328    }
329
330    /// Replace one recipient and ciphertext while preserving every other
331    /// KP/share entry and global fingerprint uniqueness.
332    pub fn replace_recipient(
333        self,
334        current_fingerprint: &str,
335        new_fingerprint: KPFingerprint,
336        new_ciphertext: String,
337    ) -> GuardianResult<(Self, KpEncryptedShare)> {
338        if new_fingerprint == current_fingerprint {
339            return Err(InvalidInputs(format!(
340                "replacement KP fingerprint {new_fingerprint} must differ from the current \
341                 fingerprint"
342            )));
343        }
344        if self.find_by_fingerprint(&new_fingerprint).is_some() {
345            return Err(InvalidInputs(format!(
346                "new KP fingerprint {new_fingerprint} is already present in the encrypted share \
347                 roster"
348            )));
349        }
350
351        let mut shares = self.0;
352        let share = shares
353            .iter_mut()
354            .find(|share| share.recipient_fingerprint == current_fingerprint)
355            .ok_or_else(|| {
356                InvalidInputs(format!(
357                    "current KP fingerprint {current_fingerprint} is not present in the \
358                     encrypted share roster"
359                ))
360            })?;
361        share.recipient_fingerprint = new_fingerprint;
362        share.armored_ciphertext = new_ciphertext;
363        let changed_share = share.clone();
364        Ok((Self(shares), changed_share))
365    }
366
367    pub fn recipient_fingerprints(&self) -> Vec<KPFingerprint> {
368        self.0
369            .iter()
370            .map(|share| share.recipient_fingerprint.clone())
371            .collect()
372    }
373
374    /// Verify every encrypted share against the cert assigned to the same
375    /// ordered share position.
376    pub fn verify_recipients(&self, certs_roster: &KpCertRoster) -> GuardianResult<()> {
377        if self.share_count() != certs_roster.num_kps() {
378            return Err(InvalidInputs(format!(
379                "expected {} KP cert roster entries, got {} encrypted shares",
380                certs_roster.num_kps(),
381                self.share_count()
382            )));
383        }
384
385        for (share, cert) in self.iter().zip(certs_roster.iter()) {
386            share.verify_recipient(cert)?;
387        }
388        Ok(())
389    }
390
391    /// Verify the exact configured certificate set and ciphertext recipients,
392    /// leaving share assignments to the signed encrypted-share roster.
393    pub fn verify_recipient_set(&self, certs_roster: &KpCertRoster) -> GuardianResult<()> {
394        if self.share_count() != certs_roster.num_kps() {
395            return Err(InvalidInputs(format!(
396                "expected {} KP cert roster entries, got {} encrypted shares",
397                certs_roster.num_kps(),
398                self.share_count()
399            )));
400        }
401
402        // Both rosters enforce uniqueness, so equal counts and membership of
403        // every configured fingerprint prove exact set equality.
404        for cert in certs_roster.iter() {
405            let fingerprint = cert.fingerprint().to_hex();
406            let share = self.find_by_fingerprint(&fingerprint).ok_or_else(|| {
407                InvalidInputs(format!(
408                    "KP fingerprint {fingerprint} is not present in the encrypted-share roster"
409                ))
410            })?;
411            share.verify_recipient(cert)?;
412        }
413        Ok(())
414    }
415}

427fn verify_pgp_ciphertext_recipient(
428    share_id: ShareID,
429    recipient_fingerprint: &str,
430    ciphertext: &str,
431    expected_cert: &AttestedKpCert,
432) -> GuardianResult<()> {
433    let recipients = pgp_message_recipients(ciphertext).map_err(|e| {
434        InvalidInputs(format!(
435            "failed to parse PGP recipients for share id {}: {e}",
436            share_id.get()
437        ))
438    })?;
439    if recipients.is_empty() {
440        return Err(InvalidInputs(format!(
441            "share id {} has no PGP recipients",
442            share_id.get()
443        )));
444    }
445    let expected_key =
446        sequoia_openpgp::KeyHandle::from(expected_cert.encryption_fingerprint().clone());
447    for handle in &recipients {
448        if !expected_key.aliases(handle) {
449            return Err(InvalidInputs(format!(
450                "share id {} (keyed by {}) is encrypted to key {handle}, which is not the \
451                 attested encryption key",
452                share_id.get(),
453                recipient_fingerprint
454            )));
455        }
456    }
457    info!(
458        share_id = share_id.get(),
459        fingerprint = %recipient_fingerprint,
460        recipient_count = recipients.len(),
461        "verified encrypted share targets only its keyed recipient cert"
462    );
463    Ok(())
464}

466/// Encrypt a share with optional AAD
467pub fn encrypt_share<R: CryptoRng + RngCore>(
468    share: &Share,
469    pk: &EncPubKey,
470    aad: Option<&[u8; 32]>,
471    rng: &mut R,
472) -> GuardianEncryptedShare {
473    GuardianEncryptedShare {
474        id: share.id,
475        ciphertext: encrypt(&share.value.to_bytes(), pk, aad, rng)
476            .expect("neither plaintext nor aad are long"),
477    }
478}

480/// Split `sk` into `params.num_shares()` shares with reconstruction threshold
481/// `params.threshold()`, encrypt each share to its matching KP certificate, and
482/// compute one commitment per share. The roster assigns its certificate at
483/// position `i` to share ID `i + 1`.
484///
485/// # Panics
486///
487/// Panics if `kp_certs_roster.num_kps() != params.num_shares()`.
488pub fn split_and_encrypt_for_kps<R: CryptoRng + RngCore>(
489    sk: &k256::SecretKey,
490    kp_certs_roster: &KpCertRoster,
491    params: &SecretSharingParams,
492    rng: &mut R,
493) -> (KpEncryptedShareRoster, ShareCommitments) {
494    assert_eq!(
495        kp_certs_roster.num_kps(),
496        params.num_shares(),
497        "request validation ensures one KP certificate per share",
498    );
499    let shares = split_secret(sk, params, rng);
500    let n = params.num_shares();
501    let mut encrypted_shares = Vec::with_capacity(n);
502    let mut commitments = Vec::with_capacity(n);
503    for (share, cert) in shares.iter().zip(kp_certs_roster.iter()) {
504        encrypted_shares.push(KpEncryptedShare {
505            id: share.id,
506            recipient_fingerprint: cert.fingerprint().to_hex(),
507            armored_ciphertext: encrypt_share_for_provisioner(share, cert),
508        });
509        commitments.push(commit_share(share));
510    }
511    let encrypted_shares = KpEncryptedShareRoster::new(encrypted_shares)
512        .expect("split_secret produces share ids exactly 1..=n");
513    let commitments =
514        ShareCommitments::new(commitments).expect("share IDs 1..=n are unique by construction");
515    (encrypted_shares, commitments)
516}

528/// Decrypt an encrypted share with optional AAD
529pub fn decrypt_share(
530    encrypted_share: &GuardianEncryptedShare,
531    sk: &EncSecKey,
532    aad: Option<&[u8; 32]>,
533) -> GuardianResult<Share> {
534    let serialized_share = decrypt(&encrypted_share.ciphertext, sk, aad)?;
535    let value = ScalarPrimitive::<Secp256k1>::from_slice(&serialized_share)
536        .map(Scalar::from)
537        .map_err(|_| InvalidInputs("Failed to deserialize share".into()))?;
538    Ok(Share {
539        id: encrypted_share.id,
540        value,
541    })
542}

544/// Decrypt each signed submission, verify it against the sharing instance, and
545/// reject duplicate share ids. The KP signature authenticates each ciphertext,
546/// so no additional HPKE AAD is needed.
547pub fn decrypt_verify_shares(
548    encrypted: &[GuardianEncryptedShare],
549    sk: &EncSecKey,
550    instance: &SecretSharingInstance,
551) -> GuardianResult<Vec<Share>> {
552    let share_count = encrypted.len();
553    let threshold = instance.threshold();
554    let num_shares = instance.num_shares();
555    if share_count < threshold {
556        return Err(InvalidInputs(format!(
557            "need at least {threshold} signed share submissions, got {share_count}"
558        )));
559    }
560    if share_count > num_shares {
561        return Err(InvalidInputs(format!(
562            "at most {num_shares} signed share submissions are allowed, got {share_count}"
563        )));
564    }
565
566    let mut shares: Vec<Share> = Vec::with_capacity(encrypted.len());
567    for enc in encrypted {
568        let share = decrypt_share(enc, sk, None)?;
569        instance.commitments().verify_share(&share)?;
570        if shares.iter().any(|s| s.id == share.id) {
571            return Err(InvalidInputs("Duplicate share ID".into()));
572        }
573        shares.push(share);
574    }
575    Ok(shares)
576}

993    fn split_and_encrypt_n5_t3_assigns_one_decryptable_ciphertext_per_cert() {
994        let keypairs = (0..5).map(|_| cert_and_secret()).collect::<Vec<_>>();
995        let roster =
996            KpCertRoster::new(keypairs.iter().map(|(cert, _)| cert.clone()).collect()).unwrap();
997        let secret_key = SecretKey::random(&mut rand::thread_rng());
998        let params = SecretSharingParams::new(5, 3).unwrap();
999
1000        let (encrypted_shares, commitments) =
1001            split_and_encrypt_for_kps(&secret_key, &roster, &params, &mut rand::thread_rng());
1002        assert_eq!(commitments.len(), 5);
1003
1004        assert_eq!(
1005            encrypted_shares.recipient_fingerprints(),
1006            roster.fingerprints(),
1007            "ordered cert position must determine the corresponding share recipient"
1008        );
1009
1010        let mut decrypted_shares = Vec::with_capacity(5);
1011        for (index, encrypted_share) in encrypted_shares.iter().enumerate() {
1012            assert_eq!(encrypted_share.id.get() as usize, index + 1);
1013
1014            let mut decryptor = decrypt_with_secret_key(
1015                Cursor::new(encrypted_share.armored_ciphertext.clone().into_bytes()),
1016                keypairs[index].1.as_bytes(),
1017            )
1018            .expect("the matching secret key must open its one ciphertext");
1019            let mut plaintext = Vec::new();
1020            decryptor.read_to_end(&mut plaintext).unwrap();
1021            let share = Share {
1022                id: encrypted_share.id,
1023                value: ScalarPrimitive::<Secp256k1>::from_slice(&plaintext)
1024                    .map(Scalar::from)
1025                    .expect("decrypted share must be one valid scalar"),
1026            };
1027            commitments.verify_share(&share).unwrap();
1028            decrypted_shares.push(share);
1029
1030            let wrong_secret = &keypairs[(index + 1) % keypairs.len()].1;
1031            let wrong_decryption = decrypt_with_secret_key(
1032                Cursor::new(encrypted_share.armored_ciphertext.clone().into_bytes()),
1033                wrong_secret.as_bytes(),
1034            )
1035            .and_then(|mut decryptor| {
1036                let mut plaintext = Vec::new();
1037                decryptor.read_to_end(&mut plaintext)?;
1038                Ok(plaintext)
1039            });
1040            assert!(
1041                wrong_decryption.is_err(),
1042                "share {} ciphertext must reject a nonmatching secret key",
1043                encrypted_share.id
1044            );
1045        }
1046
1047        let reconstructed = combine_shares(&decrypted_shares[..3], params.threshold()).unwrap();
1048        assert_eq!(reconstructed.to_bytes(), secret_key.to_bytes());
1049    }
