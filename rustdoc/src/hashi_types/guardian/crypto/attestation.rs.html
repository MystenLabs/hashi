22// Nitro Enclave PCR0 uses SHA-384 (384 bits / 8 = 48 bytes).
23// https://github.com/aws/aws-nitro-enclaves-image-format#eif-measurements
24pub(crate) const NITRO_PCR0_LEN: usize = 48;

26/// Raw AWS Nitro attestation document bytes.
27#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
28pub struct NitroAttestation(#[serde(with = "crate::guardian::serde::base64_bytes")] Vec<u8>);

30impl NitroAttestation {

32        Self(bytes)
33    }

35    pub fn into_bytes(self) -> Vec<u8> {
36        self.0
37    }

39    /// Verify a LIVE attestation document (COSE signature + AWS cert chain to the
40    /// Nitro root, chain validity checked at the current time), that it commits to
41    /// `signing_pubkey`, and that its PCR0 matches `build_pcrs`.
42    ///
43    /// In non-enclave dev/test builds the enclave emits a mock document, so the
44    /// attestation check is a no-op, mirroring `get_attestation` `non-enclave-dev`
45    /// stub behavior. Real verification runs only in enclave builds.
46    pub fn verify_live(
47        &self,
48        signing_pubkey: &GuardianPubKey,
49        build_pcrs: &BuildPcrs,
50    ) -> CryptoVerificationResult<()> {
51        self.verify_at(signing_pubkey, build_pcrs, VerifyTime::Now)
52    }
53
54    /// Verify a REPLAYED attestation document read back from the S3 audit log.
55    ///
56    /// The Nitro leaf certificate lives only a few hours, so a replayed document
57    /// can never chain-validate at the current time. Anchor the chain check at the
58    /// document's own timestamp instead: it is inside the COSE-signed payload (so
59    /// it cannot be altered without breaking the signature just verified), and a
60    /// genuine document's chain was valid at the instant NSM produced it.
61    pub fn verify_replay(
62        &self,
63        signing_pubkey: &GuardianPubKey,
64        build_pcrs: &BuildPcrs,
65    ) -> CryptoVerificationResult<()> {
66        self.verify_at(signing_pubkey, build_pcrs, VerifyTime::DocumentTimestamp)
67    }
68
69    fn verify_at(
70        &self,
71        signing_pubkey: &GuardianPubKey,
72        build_pcrs: &BuildPcrs,
73        verify_time: VerifyTime,
74    ) -> CryptoVerificationResult<()> {
75        #[cfg(any(test, feature = "non-enclave-dev"))]
76        {
77            let _ = (signing_pubkey, build_pcrs, verify_time);
78            // Real attestation is stubbed here — announce it loudly (once) so a
79            // `non-enclave-dev` binary can't be mistaken for a real enclave.
80            // Gated off `test` so unit tests stay quiet.
81            #[cfg(all(feature = "non-enclave-dev", not(test)))]
82            {
83                static WARNED: std::sync::Once = std::sync::Once::new();
84                WARNED.call_once(|| {
85                    tracing::warn!(
86                        "Nitro attestation verification DISABLED (non-enclave-dev build)"
87                    );
88                });
89            }
90            Ok(())
91        }
92        #[cfg(not(any(test, feature = "non-enclave-dev")))]
93        {
94            use fastcrypto::nitro_attestation::parse_nitro_attestation;
95            use fastcrypto::nitro_attestation::verify_nitro_attestation;
96
97            // Bools: (is_upgraded_parsing, include_all_nonzero_pcrs,
98            // always_include_required_pcrs). The last keeps PCR0 in `pcr_map` even if
99            // zero, so the pin below can't be bypassed by a missing entry.
100            let (signature, signed_message, doc) =
101                parse_nitro_attestation(&self.0, true, true, true).map_err(|e| {
102                    CryptoVerificationError::new(format!("attestation parse failed: {e}"))
103                })?;
104            let timestamp_ms = match verify_time {
105                VerifyTime::Now => now_timestamp_ms(),
106                VerifyTime::DocumentTimestamp => doc.timestamp,
107            };
108            // Fastcrypto uses this time to validate certificate dates; it does not
109            // check document freshness, which we enforce separately for live RPCs.
110            verify_nitro_attestation(&signature, &signed_message, &doc, timestamp_ms).map_err(
111                |e| CryptoVerificationError::new(format!("attestation verification failed: {e}")),
112            )?;
113            if matches!(verify_time, VerifyTime::Now) {
114                verify_live_timestamp(doc.timestamp, timestamp_ms)?;
115            }
116
117            let attested = doc
118                .public_key
119                .ok_or_else(|| CryptoVerificationError::new("attestation has no public_key"))?;
120            if attested != signing_pubkey.to_bytes() {
121                return Err(CryptoVerificationError::new(
122                    "attestation public_key does not match the session signing pubkey",
123                ));
124            }
125
126            // Pin PCR0 (the whole EIF image hash).
127            if doc.pcr_map.get(&0).map(Vec::as_slice) != Some(build_pcrs.pcr0()) {
128                return Err(CryptoVerificationError::new(
129                    "attestation PCR0 does not match the expected enclave image",
130                ));
131            }
132            Ok(())
133        }
134    }
135}

137/// Live RPCs generate an uncached attestation; allow for latency and clock skew.
138/// Historical S3 attestations deliberately do not use this check.
139/// The document must be at most 60 seconds old or 5 seconds in the future.
140///
141/// Pure hardening: KPs must pin the latest approved PCR and reject known-buggy
142/// builds. Replaying an accepted build's attestation exposes no private keys;
143/// a still-running session can attest afresh anyway. This does not establish
144/// freshness of separately signed response data.
145#[cfg(any(test, not(feature = "non-enclave-dev")))]
146fn verify_live_timestamp(document_ms: u64, now_ms: u64) -> CryptoVerificationResult<()> {
147    const MAX_AGE_MS: u64 = 60_000;
148    const MAX_FUTURE_SKEW_MS: u64 = 5_000;
149
150    if now_ms.saturating_sub(document_ms) > MAX_AGE_MS {
151        return Err(CryptoVerificationError::new(
152            "live attestation is more than 60 seconds old",
153        ));
154    }
155    if document_ms.saturating_sub(now_ms) > MAX_FUTURE_SKEW_MS {
156        return Err(CryptoVerificationError::new(
157            "live attestation is more than 5 seconds in the future; check clock synchronization",
158        ));
159    }
160    Ok(())
161}

163/// When the attestation's cert chain validity is checked: at the current time
164/// (live RPC responses) or at the document's own COSE-signed timestamp
165/// (replays from the S3 audit log, where the short-lived leaf has expired).
166#[derive(Clone, Copy)]
167enum VerifyTime {
168    Now,
169    DocumentTimestamp,
170}

172/// One enclave build: its git revision and known-good Nitro measurement. A build is
173/// identified by its revision and pinned by PCR0 - the hash of the whole enclave
174/// image (EIF), which uniquely identifies the build.
175///
176/// We record only PCR0: in a StageX (reproducible, single-binary) build it is
177/// the only measurement that carries signal - the others (kernel, bootloader, IAM
178/// role) are constant or irrelevant for our pinning.
179#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
180pub struct BuildPcrs {
181    git_revision: GitRevision,
182    #[serde(serialize_with = "serialize_pcr0")]
183    pcr0: Vec<u8>,
184}
185
186// Config files accept hex PCRs. Emit the same readable form in logged policies,
187// while preserving the binary representation used by existing config digests.
188fn serialize_pcr0<S: serde::Serializer>(pcr0: &[u8], serializer: S) -> Result<S::Ok, S::Error> {
189    if serializer.is_human_readable() {
190        serializer.serialize_str(&hex::encode(pcr0))
191    } else {
192        pcr0.serialize(serializer)
193    }
194}

196impl BuildPcrs {
197    pub fn new(git_revision: &str, pcr0: Vec<u8>) -> GuardianResult<Self> {
198        if pcr0.len() != NITRO_PCR0_LEN {
199            return Err(InvalidInputs(format!(
200                "build '{git_revision}' PCR0 must be {NITRO_PCR0_LEN} bytes, got {}",
201                pcr0.len()
202            )));
203        }
204        if pcr0.iter().all(|byte| *byte == 0) {
205            return Err(InvalidInputs(format!(
206                "build '{git_revision}' PCR0 must not be all zeros (Nitro debug mode)"
207            )));
208        }
209        Ok(Self {
210            git_revision: git_revision.to_string(),
211            pcr0,
212        })
213    }
214
215    pub fn git_revision(&self) -> &str {
216        &self.git_revision
217    }
218
219    pub fn pcr0(&self) -> &[u8] {
220        &self.pcr0
221    }
222}

224/// PCR pins for enclave builds that may appear in guardian attestations.
225/// Used by the guardian S3 reader.
226///
227/// `current_build` is the current/live build. `prev_builds` contains older
228/// builds that may still appear in persisted logs during an upgrade or replay.
229/// The reported deployment revision selects an entry; Nitro verification checks
230/// its PCR0 and signing key before log signatures are trusted. Callers use the
231/// resolved `BuildPcrs` to enforce the policy for their context.
232#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
233pub struct PcrAllowlist {
234    current_build: BuildPcrs,
235    prev_builds: Vec<BuildPcrs>,
236}
237
238impl PcrAllowlist {
239    pub fn new(
240        current_build: BuildPcrs,
241        prev_builds: impl IntoIterator<Item = BuildPcrs>,
242    ) -> GuardianResult<Self> {
243        let prev_builds = prev_builds.into_iter().collect::<Vec<_>>();
244        let mut seen = BTreeSet::new();
245        for build in std::iter::once(&current_build).chain(prev_builds.iter()) {
246            if !seen.insert(build.git_revision.clone()) {
247                return Err(InvalidInputs(format!(
248                    "duplicate PCR allowlist entry for build '{}'",
249                    build.git_revision
250                )));
251            }
252        }
253
254        Ok(Self {
255            current_build,
256            prev_builds,
257        })
258    }
259
260    pub fn current_build(&self) -> &BuildPcrs {
261        &self.current_build
262    }
263
264    pub fn prev_builds(&self) -> &[BuildPcrs] {
265        &self.prev_builds
266    }
267
268    /// The `BuildPcrs` whose revision is `git_revision`.
269    ///
270    /// A missing revision may indicate either incomplete configured build
271    /// history or a build that was never approved.
272    pub fn resolve(&self, git_revision: &str) -> GuardianResult<&BuildPcrs> {
273        if self.current_build.git_revision == git_revision {
274            return Ok(&self.current_build);
275        }
276        if let Some(prev_build) = self
277            .prev_builds
278            .iter()
279            .find(|build| build.git_revision == git_revision)
280        {
281            return Ok(prev_build);
282        }
283        Err(BuildNotAllowlisted(format!(
284            "guardian reports build '{git_revision}', which is not present in the PCR allowlist"
285        )))
286    }
287
288    pub fn require_current_build(&self, build_pcrs: &BuildPcrs) -> GuardianResult<()> {
289        if self.current_build == *build_pcrs {
290            Ok(())
291        } else {
292            Err(BuildNotCurrent(format!(
293                "guardian build '{}' does not match the required current build '{}'",
294                build_pcrs.git_revision(),
295                self.current_build.git_revision()
296            )))
297        }
298    }
299}

301impl<'de> Deserialize<'de> for BuildPcrs {
302    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
303        #[derive(Deserialize)]
304        struct BuildPcrsWire {
305            git_revision: GitRevision,
306            pcr0: String,
307        }
308
309        let wire = BuildPcrsWire::deserialize(deserializer)?;
310        let pcr0 = hex::decode(wire.pcr0.trim_start_matches("0x")).map_err(|e| {
311            serde::de::Error::custom(format!(
312                "build '{}' pcr0 is not valid hex: {e}",
313                wire.git_revision
314            ))
315        })?;
316        BuildPcrs::new(&wire.git_revision, pcr0).map_err(serde::de::Error::custom)
317    }
318}
319
320impl<'de> Deserialize<'de> for PcrAllowlist {
321    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
322        #[derive(Deserialize)]
323        struct PcrAllowlistWire {
324            current_build: BuildPcrs,
325            #[serde(default)]
326            prev_builds: Vec<BuildPcrs>,
327        }
328
329        let wire = PcrAllowlistWire::deserialize(deserializer)?;
330        PcrAllowlist::new(wire.current_build, wire.prev_builds).map_err(serde::de::Error::custom)
331    }
332}

338    #[test]
339    fn live_attestation_timestamp_window() {
340        let now_ms = 100_000;
341        for document_ms in [now_ms - 60_000, now_ms, now_ms + 5_000] {
342            assert!(verify_live_timestamp(document_ms, now_ms).is_ok());
343        }
344        assert!(verify_live_timestamp(now_ms - 60_001, now_ms).is_err());
345        assert!(verify_live_timestamp(now_ms + 5_001, now_ms).is_err());
346    }

348    #[test]
349    fn live_attestation_timestamp_handles_integer_limits() {
350        assert!(verify_live_timestamp(0, 0).is_ok());
351        assert!(verify_live_timestamp(u64::MAX, u64::MAX).is_ok());
352        assert!(verify_live_timestamp(0, u64::MAX).is_err());
353        assert!(verify_live_timestamp(u64::MAX, 0).is_err());
354    }

358        let build = BuildPcrs::new("current", [0, 255].repeat(24)).unwrap();
359        let json = serde_json::to_value(&build).unwrap();
360        assert_eq!(json["pcr0"], "00ff".repeat(24));
361        assert_eq!(serde_json::from_value::<BuildPcrs>(json).unwrap(), build);
362        // This is the original derived struct's BCS field order and encoding.
363        assert_eq!(
364            bcs::to_bytes(&build).unwrap(),
365            bcs::to_bytes(&(build.git_revision(), build.pcr0())).unwrap(),
366        );
