4//! Guardian- and KP-signed envelopes with intent-based domain separation.
5//!
6//! [`GuardianSigned`] uses Ed25519 signatures produced by the Guardian, while
7//! [`KpSigned`] uses detached OpenPGP signatures produced by key provisioners.
8//! Both serialize a payload together with its signing intent so a signature for
9//! one payload type cannot be replayed as another.
10

37/// All possible signing intent types.
38/// Using an enum ensures no two types can accidentally share the same intent value.
39#[repr(u8)]
40#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
41pub enum GuardianSigningIntentType {
42    /// Intent for LogEntry.
43    LogEntry = 0,
44    /// Intent for SetupNewKeyResponse.
45    SetupNewKeyResponse = 1,
46    /// Intent for StandardWithdrawalResponse.
47    StandardWithdrawalResponse = 2,
48    /// Intent for GuardianInfo.
49    GuardianInfo = 3,
50    /// Intent for RotateKpSetResponse.
51    RotateKpSetResponse = 4,
52    /// Intent for ProvisionerRotateCertResponse.
53    ProvisionerRotateCertResponse = 5,
54}
55
56/// Guardian-signed payloads and their intent-based domain separation.
57pub trait GuardianSigningIntent: Serialize {
58    const INTENT: GuardianSigningIntentType;
59}
60
61/// All possible KP signing intent types.
62///
63/// These signatures are detached OpenPGP signatures produced by KPs, not
64/// enclave ed25519 signatures. Each KP-submitted request type gets a stable
65/// intent so a signature for one request cannot be replayed as another request
66/// with the same BCS shape.
67#[repr(u8)]
68#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
69pub enum KpSigningIntentType {
70    /// Intent for ProvisionerInitRequest.
71    ProvisionerInitRequest = 0,
72    /// Intent for ProvisionerRotateCertRequest.
73    ProvisionerRotateCertRequest = 1,
74    /// Intent for ProvisionerRotateKpSetRequest.
75    ProvisionerRotateKpSetRequest = 2,
76}
77
78/// KP-signed payloads and their intent-based domain separation.
79pub trait KpSigningIntent: Serialize + SessionBoundRequest {
80    const INTENT: KpSigningIntentType;
81}

86pub struct KpSigned<T> {
87    data: T,
88    pub signer_cert: PgpPublicCert,
89    pub signature: String,
90}

92/// A timestamped response produced by the Guardian.
93///
94/// The timestamp is response metadata rather than a property of every
95/// Guardian-signed payload. Wrapping this value in [`GuardianSigned`] keeps the
96/// timestamp authenticated.
97#[derive(Serialize, Deserialize, Debug, Clone, PartialEq)]
98pub struct GuardianResponse<T> {
99    pub response: T,
100    /// Milliseconds since Unix epoch.
101    pub timestamp_ms: UnixMillis,
102}
103
104/// A signed, timestamped response produced by the Guardian.
105pub type GuardianSignedResponse<T> = GuardianSigned<GuardianResponse<T>>;

107/// Guardian-signed wrapper - adds a signature to any signable payload.
108#[derive(Serialize, Deserialize, Debug, Clone, PartialEq)]
109pub struct GuardianSigned<T> {
110    data: T,
111    pub signature: GuardianSignature,
112}

151impl<T> GuardianResponse<T> {
152    pub fn new(response: T, timestamp_ms: UnixMillis) -> Self {
153        Self {
154            response,
155            timestamp_ms,
156        }
157    }
158}

160// Guardian unchecked access is intentionally narrow: LogRecord's custom wire
161// handling and node/proxy/CLI paths that establish trust independently.
162// KpSigned has no unchecked extraction; production KP payloads are always
163// verified before access.
164impl<T> GuardianSigned<T> {
165    pub fn from_parts(data: T, signature: GuardianSignature) -> Self {
166        Self { data, signature }
167    }
168
169    fn signed_bytes(data: &T) -> Vec<u8>
170    where
171        T: GuardianSigningIntent,
172    {
173        bcs::to_bytes(&(T::INTENT, data)).expect("serialization should not fail")
174    }
175
176    /// Sign a payload with intent-based domain separation.
177    pub fn sign(data: T, signing_key: &SigningKey) -> Self
178    where
179        T: GuardianSigningIntent,
180    {
181        let signature = signing_key.sign(&Self::signed_bytes(&data));
182        Self { data, signature }
183    }
184
185    /// Verify the Guardian signature and borrow the authenticated payload.
186    pub fn verify_signature(&self, pub_key: &VerificationKey) -> CryptoVerificationResult<&T>
187    where
188        T: GuardianSigningIntent,
189    {
190        pub_key
191            .verify(&self.signature, &Self::signed_bytes(&self.data))
192            .map_err(|_| CryptoVerificationError::new("signature invalid"))?;
193        Ok(&self.data)
194    }
195
196    /// Verify the Guardian signature and move out the authenticated payload.
197    pub fn verify_into_data(self, pub_key: &VerificationKey) -> CryptoVerificationResult<T>
198    where
199        T: GuardianSigningIntent,
200    {
201        self.verify_signature(pub_key)?;
202        Ok(self.data)
203    }
204
205    /// Borrow the payload WITHOUT verifying the signature.
206    pub(crate) fn data_unchecked(&self) -> &T {
207        &self.data
208    }
209
210    #[cfg(test)]
211    pub(crate) fn data_unchecked_mut(&mut self) -> &mut T {
212        &mut self.data
213    }
214
215    /// Move out the payload WITHOUT verifying the signature.
216    /// The caller must establish trust in the payload independently.
217    pub fn into_data_unchecked(self) -> T {
218        self.data
219    }
220
221    pub(crate) fn into_parts(self) -> (T, GuardianSignature) {
222        (self.data, self.signature)
223    }
224}
225
226impl<T: KpSigningIntent> KpSigned<T> {
227    pub fn from_parts(data: T, signer_cert: PgpPublicCert, signature: String) -> Self {
228        Self {
229            data,
230            signer_cert,
231            signature,
232        }
233    }
234
235    /// Sign a KP payload by invoking `gpg --detach-sign` for the
236    /// signer certificate's fingerprint. Includes the KP intent in the signed
237    /// bytes; payload types carry any request-specific replay-binding fields.
238    pub fn sign(
239        data: T,
240        signer_cert: PgpPublicCert,
241        gpg_home: Option<&Path>,
242    ) -> GuardianResult<Self> {
243        let signing_payload = Self::signed_bytes(&data);
244        let signature =
245            sign_detached_via_gpg(&signing_payload, &signer_cert.fingerprint(), gpg_home)
246                .map_err(|e| InternalError(format!("KP signing failed: {e}")))?;
247        Ok(Self {
248            data,
249            signer_cert,
250            signature,
251        })
252    }
253
254    /// The exact bytes a key provisioner detached-signs for a typed guardian
255    /// request. Binds the request intent and request payload.
256    pub fn signed_bytes(data: &T) -> Vec<u8> {
257        let tuple = (T::INTENT, data);
258        bcs::to_bytes(&tuple).expect("serialization should not fail")
259    }
260
261    /// Verify the signature and borrow the authenticated request.
262    /// Checks the intent byte to ensure the signature is for this request type.
263    pub fn verify_signature(&self) -> CryptoVerificationResult<&T> {
264        let msg_bytes = Self::signed_bytes(&self.data);
265        verify_detached_signature(&msg_bytes, &self.signature, &self.signer_cert).map_err(|e| {
266            CryptoVerificationError::new(format!("KP signature verification failed: {e}"))
267        })?;
268        Ok(&self.data)
269    }
270
271    /// Verify the KP signature and move out the authenticated payload.
272    pub fn verify_into_data(self) -> CryptoVerificationResult<T> {
273        self.verify_signature()?;
274        Ok(self.data)
275    }
276
277    pub(crate) fn into_parts(self) -> (T, PgpPublicCert, String) {
278        (self.data, self.signer_cert, self.signature)
279    }
280
281    pub fn signer_fingerprint(&self) -> Fingerprint {
282        self.signer_cert.fingerprint()
283    }
284}
