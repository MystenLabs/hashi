27struct RawPublicKey(Vec<u8>, &'static SignatureAlgorithm);
28
29impl PublicKeyData for RawPublicKey {
30    fn der_bytes(&self) -> &[u8] {
31        &self.0
32    }
33
34    fn algorithm(&self) -> &SignatureAlgorithm {
35        self.1
36    }
37}

39fn params(name: &str) -> CertificateParams {
40    let mut distinguished_name = DistinguishedName::new();
41    distinguished_name.push(DnType::CommonName, name);
42    let mut params = CertificateParams::default();
43    params.distinguished_name = distinguished_name;
44    params
45}
46
47fn pem(der: &[u8]) -> Vec<u8> {
48    format!(
49        "-----BEGIN CERTIFICATE-----\n{}\n-----END CERTIFICATE-----\n",
50        Base64::encode_string(der),
51    )
52    .into_bytes()
53}

55/// Generate an immutable KP bundle verified against a synthetic test authority,
56/// together with its armored signing/decryption secret. The public constructor,
57/// serde decoder, and protobuf decoder do not trust this authority.
58///
59/// This helper only generates its own inputs: it cannot bless caller-supplied
60/// certificates, attestations, keys, or trust roots.
61pub fn mock_attested_kp_keypair() -> (AttestedKpCert, String) {
62    use sequoia_openpgp::cert::CertBuilder;
63    use sequoia_openpgp::cert::CipherSuite;
64    use sequoia_openpgp::serialize::Serialize;
65    use sequoia_openpgp::types::Features;
66
67    let (cert, _) = CertBuilder::general_purpose(["backup@example.com"])
68        .set_profile(sequoia_openpgp::Profile::RFC4880)
69        .unwrap()
70        .set_cipher_suite(CipherSuite::P256)
71        .set_features(Features::empty().set_seipdv1())
72        .unwrap()
73        .generate()
74        .unwrap();
75    let mut public = Vec::new();
76    cert.armored().export(&mut public).unwrap();
77    let mut secret = Vec::new();
78    cert.as_tsk().armored().serialize(&mut secret).unwrap();
79    (
80        attest_generated_cert(String::from_utf8(public).unwrap()),
81        String::from_utf8(secret).unwrap(),
82    )
83}

85/// Keep the expired software key in the same certificate as the attested keys.
86#[cfg(test)]
87pub(in crate::guardian::crypto) fn mock_attested_kp_keypair_with_expired_signer()
88-> (AttestedKpCert, Cert, std::time::SystemTime) {
89    use sequoia_openpgp::cert::CertBuilder;
90    use sequoia_openpgp::serialize::Serialize;
91    use sequoia_openpgp::types::KeyFlags;
92    use std::time::Duration;
93    use std::time::SystemTime;
94
95    let creation_time = SystemTime::now() - Duration::from_secs(7 * 24 * 60 * 60);
96    let signature_time = creation_time + Duration::from_secs(60);
97    let (pgp_cert, _) = CertBuilder::new()
98        .set_profile(sequoia_openpgp::Profile::RFC4880)
99        .unwrap()
100        .set_creation_time(creation_time)
101        .set_primary_key_flags(KeyFlags::empty().set_certification())
102        .add_subkey(
103            KeyFlags::empty().set_signing(),
104            Duration::from_secs(24 * 60 * 60),
105            None,
106        )
107        .add_signing_subkey()
108        .add_transport_encryption_subkey()
109        .generate()
110        .unwrap();
111    let mut public = Vec::new();
112    pgp_cert.armored().export(&mut public).unwrap();
113    let attested = attest_generated_cert(String::from_utf8(public).unwrap());
114    (attested, pgp_cert, signature_time)
115}

117fn attest_generated_cert(public: String) -> AttestedKpCert {
118    let pgp_cert = Cert::from_bytes(public.as_bytes()).unwrap();
119    let cert = PgpPublicCert::new(public).unwrap();
120
121    let issuer_key = KeyPair::generate_for(&PKCS_ECDSA_P256_SHA256).unwrap();
122    let mut issuer_params = params("test pinned issuer");
123    issuer_params.is_ca = IsCa::Ca(BasicConstraints::Unconstrained);
124    let issuer = issuer_params.self_signed(&issuer_key).unwrap();
125    let signing_key = SigningKey::new(rand::thread_rng());
126    let mut pkcs8 = vec![
127        0x30, 0x2e, 2, 1, 0, 0x30, 5, 6, 3, 0x2b, 0x65, 0x70, 4, 0x22, 4, 0x20,
128    ];
129    pkcs8.extend_from_slice(&signing_key.to_bytes());
130    let device_key = KeyPair::try_from(pkcs8).unwrap();
131    let device = params("test attestation device")
132        .signed_by(&device_key, &issuer, &issuer_key)
133        .unwrap();
134
135    let policy = StandardPolicy::new();
136    let statements = [true, false].map(|signing| {
137        let candidates = pgp_cert
138            .keys()
139            .with_policy(&policy, None)
140            .supported()
141            .alive()
142            .revoked(false);
143        let key = if signing {
144            candidates.for_signing().next().unwrap()
145        } else {
146            candidates.for_transport_encryption().next().unwrap()
147        };
148        let (bytes, algorithm, x25519) = match key.key().mpis() {
149            mpi::PublicKey::ECDSA {
150                curve: Curve::NistP256,
151                q,
152            }
153            | mpi::PublicKey::ECDH {
154                curve: Curve::NistP256,
155                q,
156                ..
157            } => {
158                q.decode_point(&Curve::NistP256).unwrap();
159                (q.value().to_vec(), &PKCS_ECDSA_P256_SHA256, false)
160            }
161            mpi::PublicKey::EdDSA {
162                curve: Curve::Ed25519,
163                q,
164            } => (
165                q.decode_point(&Curve::Ed25519).unwrap().0.to_vec(),
166                &PKCS_ED25519,
167                false,
168            ),
169            mpi::PublicKey::ECDH {
170                curve: Curve::Cv25519,
171                q,
172                ..
173            } => (
174                q.decode_point(&Curve::Cv25519).unwrap().0.to_vec(),
175                &PKCS_ED25519,
176                true,
177            ),
178            other => panic!("unexpected fixture key: {other:?}"),
179        };
180        let mut statement_params = params(if signing {
181            "YubiKey OPGP Attestation SIG"
182        } else {
183            "YubiKey OPGP Attestation DEC"
184        });
185        statement_params
186            .custom_extensions
187            .push(CustomExtension::from_oid_content(
188                &[1, 3, 6, 1, 4, 1, 41482, 5, 2],
189                vec![2, 1, 1], // Canonical ASN.1 INTEGER: generated on device.
190            ));
191        let mut der = statement_params
192            .signed_by(&RawPublicKey(bytes, algorithm), &device, &device_key)
193            .unwrap()
194            .der()
195            .to_vec();
196        if x25519 {
197            // rcgen lacks X25519 SPKI. Change only the Ed25519 SPKI OID,
198            // then sign the resulting TBS with the actual device key.
199            let (_, parsed) = parse_x509_certificate(&der).unwrap();
200            let spki_offset = parsed.public_key().raw.as_ptr() as usize - der.as_ptr() as usize;
201            der[spki_offset + 8] = 110; // id-X25519
202            let (_, parsed) = parse_x509_certificate(&der).unwrap();
203            let signature = signing_key.sign(parsed.tbs_certificate.as_ref());
204            let signature_offset = der.len() - 64;
205            der[signature_offset..].copy_from_slice(&signature.to_bytes());
206        }
207        der
208    });
209    let keys = crate::pgp::verify_yubikey_attestations_with_issuers(
210        &cert,
211        device.der(),
212        &statements[0],
213        &statements[1],
214        &[issuer.der().as_ref()],
215    )
216    .expect("generated KP fixture must pass all attestation checks");
217    AttestedKpCert {
218        cert,
219        device_pem: pem(device.der()),
220        sig_pem: pem(&statements[0]),
221        dec_pem: pem(&statements[1]),
222        keys,
223    }
224}
