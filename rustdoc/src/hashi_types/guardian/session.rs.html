13/// Guardian session identifier. Canonical IDs are short prefixes of the
14/// hex-encoded signing public key and tag per-session S3 objects.
15#[derive(Serialize, Deserialize, Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
16#[serde(transparent)]
17pub struct SessionID(String);
18
19/// A KP-signed request bound to one guardian session.
20pub trait SessionBoundRequest {
21    const REQUEST_CONTEXT: &'static str;
22
23    fn expected_session(&self) -> &SessionID;
24
25    fn validate_session(&self, live_session: &SessionID) -> GuardianResult<()> {
26        if self.expected_session() != live_session {
27            return Err(InvalidInputs(format!(
28                "{} expected guardian session {}, live session is {}",
29                Self::REQUEST_CONTEXT,
30                self.expected_session(),
31                live_session,
32            )));
33        }
34        Ok(())
35    }
36}
37
38impl SessionID {
39    /// Length of the signing-public-key prefix used for canonical session IDs.
40    pub const HEX_LEN: usize = 16;
41
42    pub fn from_signing_pubkey(signing_pub_key: &GuardianPubKey) -> Self {
43        let mut session_id = ::hex::encode(signing_pub_key.as_bytes());
44        session_id.truncate(Self::HEX_LEN);
45        Self(session_id)
46    }
47
48    pub fn as_str(&self) -> &str {
49        &self.0
50    }
51}
52
53impl From<String> for SessionID {
54    fn from(value: String) -> Self {
55        Self(value)
56    }
57}
58
59impl From<&str> for SessionID {
60    fn from(value: &str) -> Self {
61        Self(value.to_owned())
62    }
63}
64
65impl From<SessionID> for String {
66    fn from(value: SessionID) -> Self {
67        value.0
68    }
69}

105        request
106            .validate_session(&SessionID::from("mock-session"))
107            .unwrap();
108

114            InvalidInputs(message)
