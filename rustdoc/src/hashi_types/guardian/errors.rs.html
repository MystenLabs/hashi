5use super::lifecycle::EnclaveLifecycle;
6use super::time::UnixSeconds;

8#[derive(Debug, Clone, PartialEq)]
9pub enum GuardianError {
10    // ========================================================================
11    // Errors requiring corrected input/data or investigation
12    // ========================================================================
13    InternalError(String),
14    /// S3 API, object-lock, or version-history failures.
15    S3Error(String),
16    /// Persisted S3 guardian log data failed structural or authenticity validation.
17    InvalidS3Log(String),
18    InvalidInputs(String),
19    Unauthenticated(String),
20
21    // ========================================================================
22    // Errors that may succeed after configuration, lifecycle, or transient
23    // state changes
24    // ========================================================================
25    /// The guardian build is absent from the configured PCR allowlist.
26    BuildNotAllowlisted(String),
27    /// The guardian build does not match the configured current build.
28    BuildNotCurrent(String),
29    LifecycleMismatch {
30        expected: EnclaveLifecycle,
31        actual: EnclaveLifecycle,
32    },
33    LimiterSequenceMismatch {
34        expected: u64,
35        actual: u64,
36    },
37    /// The selected session has no heartbeat recent enough to be considered live.
38    CurrentSessionHeartbeatNotLive {
39        session_id: SessionID,
40        /// `None` means no heartbeat was found in the recent scan.
41        heartbeat_age_secs: Option<UnixSeconds>,
42        retry_after_secs: UnixSeconds,
43    },
44    /// A prior session remains inside the activation quiet period.
45    PriorSessionHeartbeatStillRecent {
46        session_id: SessionID,
47        heartbeat_age_secs: UnixSeconds,
48        required_quiet_secs: UnixSeconds,
49    },
50    RateLimitExceeded,
51    /// A service condition known to be temporary and safe for callers to retry.
52    Unavailable(String),
53}
54
55pub type GuardianResult<T> = Result<T, GuardianError>;
56
57impl GuardianError {
58    /// Returns how long a caller should wait before retrying, when the error
59    /// describes a condition that becomes ready on a known schedule.
60    pub fn retry_after_secs(&self) -> Option<UnixSeconds> {
61        match self {
62            Self::CurrentSessionHeartbeatNotLive {
63                retry_after_secs, ..
64            } => Some(*retry_after_secs),
65            Self::PriorSessionHeartbeatStillRecent {
66                heartbeat_age_secs,
67                required_quiet_secs,
68                ..
69            } => Some(required_quiet_secs.saturating_sub(*heartbeat_age_secs)),
70            _ => None,
71        }
72    }
73}

75/// Low-level error returned by signature and attestation verifiers.
76///
77/// This remains source-neutral so higher-level layers can convert it to
78/// [`GuardianError::Unauthenticated`] for inbound RPC evidence or
79/// [`GuardianError::InvalidS3Log`] for persisted S3 evidence. Client tooling
80/// may instead wrap it with its own context.
81///
82/// It is intended to be converted rather than exposed directly as the final
83/// RPC, API, or application error.
84#[derive(Debug, Clone, PartialEq, Eq)]
85pub struct CryptoVerificationError(String);
86
87impl CryptoVerificationError {
88    pub(super) fn new(message: impl Into<String>) -> Self {
89        Self(message.into())
90    }
91}
92
93pub type CryptoVerificationResult<T> = Result<T, CryptoVerificationError>;
94
95impl std::fmt::Display for CryptoVerificationError {
96    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
97        f.write_str(&self.0)
98    }
99}
100
101impl std::error::Error for CryptoVerificationError {}

103impl std::fmt::Display for GuardianError {
104    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
105        match self {
106            GuardianError::InternalError(e) => write!(f, "InternalError: {}", e),
107            GuardianError::S3Error(e) => write!(f, "S3Error: {}", e),
108            GuardianError::InvalidS3Log(e) => write!(f, "InvalidS3Log: {}", e),
109            GuardianError::InvalidInputs(e) => write!(f, "InvalidInputs: {}", e),
110            GuardianError::Unauthenticated(e) => write!(f, "Unauthenticated: {}", e),
111            GuardianError::BuildNotAllowlisted(e) => write!(f, "BuildNotAllowlisted: {}", e),
112            GuardianError::BuildNotCurrent(e) => write!(f, "BuildNotCurrent: {}", e),
113            GuardianError::LifecycleMismatch { expected, actual } => write!(
114                f,
115                "LifecycleMismatch: expected {expected:?}, but enclave is {actual:?}"
116            ),
117            GuardianError::LimiterSequenceMismatch { expected, actual } => write!(
118                f,
119                "LimiterSequenceMismatch: expected {expected}, got {actual}"
120            ),
121            GuardianError::CurrentSessionHeartbeatNotLive {
122                session_id,
123                heartbeat_age_secs,
124                retry_after_secs,
125            } => match heartbeat_age_secs {
126                Some(heartbeat_age_secs) => write!(
127                    f,
128                    "CurrentSessionHeartbeatNotLive: session {session_id} last heartbeated \
129                     {heartbeat_age_secs}s ago; retry in {retry_after_secs}s"
130                ),
131                None => write!(
132                    f,
133                    "CurrentSessionHeartbeatNotLive: no recent heartbeat found for session \
134                     {session_id}; retry in {retry_after_secs}s"
135                ),
136            },
137            GuardianError::PriorSessionHeartbeatStillRecent {
138                session_id,
139                heartbeat_age_secs,
140                required_quiet_secs,
141            } => write!(
142                f,
143                "PriorSessionHeartbeatStillRecent: session {session_id} heartbeated \
144                 {heartbeat_age_secs}s ago; required quiet period is {required_quiet_secs}s; retry \
145                 in {}s",
146                required_quiet_secs.saturating_sub(*heartbeat_age_secs)
147            ),
148            GuardianError::RateLimitExceeded => write!(f, "Rate limit exceeded"),
149            GuardianError::Unavailable(e) => write!(f, "Unavailable: {}", e),
150        }
151    }
152}
