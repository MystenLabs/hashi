358
359    pub fn mock_for_testing(kp: Option<BitcoinKeypair>) -> Self {
360        let kp = kp.unwrap_or(create_btc_keypair_for_test(&[1u8; 32]));
361        let max_capacity = 1000;
362
363        // Convert the bitcoin-lib keypair's x-only pubkey to a raw `G` point.
364        // The bitcoin Keypair always signs against its even-y projection, so
365        // Building `G` from the x-only bytes with even-y gives the parent that
366        // matches what the keypair signs against.
367        let hashi_btc_master_pubkey =
368            hashi_master_g_from_btc_xonly_for_test(&kp.x_only_public_key().0);
369
370        InitConfig::new(
371            LimiterConfig {
372                refill_rate: 10,
373                max_bucket_capacity: max_capacity,
374            },
375            hashi_btc_master_pubkey,
376            mock_pcr_allowlist(),
377            S3BucketInfo::mock_for_testing(),
378            super::S3RetentionEnvironment::Testnet,
379            super::Network::Regtest,
380            TEST_HASHI_OBJECT_ID,
381        )
382        .expect("valid InitConfig")
383    }
384}

395impl StandardWithdrawalRequest {
396    fn mock_for_testing(network: Network, wid: WithdrawalID) -> Self {
397        let kp = create_btc_keypair_for_test(&[2u8; 32]);
398        let (internal_key, _) = UntweakedPublicKey::from_keypair(&kp);
399        let addr_unchecked =
400            BitcoinAddress::p2tr(&BTC_LIB, internal_key, None, network).into_unchecked();
401
402        let txid = bitcoin::Txid::from_slice(&[9u8; 32]).expect("valid txid bytes");
403        let outpoint = bitcoin::OutPoint { txid, vout: 0 };
404
405        let input = InputUTXO::new(outpoint, Amount::from_sat(10_000), [7u8; 32].into());
406
407        let output_external = OutputUTXOWire::external(addr_unchecked, Amount::from_sat(9_000));
408        let output_internal = OutputUTXOWire::internal([42u8; 32].into(), Amount::from_sat(500));
409
410        let utxos = TxUTXOs::new(vec![input], vec![output_external, output_internal], network)
411            .expect("valid TxUTXOs");
412
413        StandardWithdrawalRequest::new(wid, utxos, 1_000_000, 0)
414    }

428    fn sign_for_request(
429        req: StandardWithdrawalRequest,
430    ) -> (HashiSigned<StandardWithdrawalRequest>, HashiCommittee) {
431        let epoch = 0u64;
432        let committee = mock_committee_with_one_member(epoch);
433
434        let sk = mock_hashi_bls_sk();
435        let address = TEST_SIGNER_ADDRESS;
436        let mut agg = BlsSignatureAggregator::new(TEST_HASHI_OBJECT_ID, &committee, req.clone());
437        agg.add_signature(sk.sign(TEST_HASHI_OBJECT_ID, epoch, address, &req))
438            .expect("member signature should verify");
439
440        (agg.finish().expect("finish aggregator"), committee)
441    }

444        network: Network,
445        wid: WithdrawalID,
446    ) -> (HashiSigned<StandardWithdrawalRequest>, HashiCommittee) {
447        Self::sign_for_request(Self::mock_for_testing(network, wid))
448    }
