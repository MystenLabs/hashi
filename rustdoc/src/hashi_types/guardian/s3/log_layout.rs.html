14pub enum ObjectKeyPattern {
15    Fixed(String),
16    /// Complete key prefix before the random suffix; finalize() appends the suffix.
17    RandomSuffix(String),
18}
19
20impl ObjectKeyPattern {
21    /// Finalizes the pattern into the complete S3 object key.
22    pub fn finalize(self) -> String {
23        match self {
24            Self::Fixed(key) => key,
25            Self::RandomSuffix(prefix) => {
26                format!("{prefix}{:032x}.json", rand::random::<u128>())
27            }
28        }
29    }
30}

32/// S3 sub-prefixes used for guardian log streams.
33/// See `crates/hashi-guardian/README.md` for canonical key layout.
34pub const S3_DIR_INIT: &str = "init";
35pub const S3_DIR_WITHDRAW: &str = "withdraw";
36pub const S3_DIR_HEARTBEAT: &str = "heartbeat";
37pub const S3_DIR_CEREMONY: &str = "ceremony";
38pub const S3_DIR_KP_SHARES: &str = "kp-shares";
39pub const S3_DIR_COMMITTEE_UPDATE: &str = "committee-update";
40pub const S3_DIR_GENESIS: &str = "genesis";
41
42pub const SECONDS_PER_HOUR: UnixSeconds = 60 * 60;
43const DIR_WRITES_COMPLETION_DELAY: Duration = Duration::from_mins(10);
44
45type Year = i32;
46type Month = u8;
47type Day = u8;
48type Hour = u8;
49
50/// An S3 directory: prefix/YYYY/MM/DD/HH.
51/// All logs emitted within an hour are stored in the same directory, e.g., logs emitted between 12-1 PM are in `<prefix>`/12 directory.
52#[derive(Clone, Debug, Eq, PartialEq)]
53pub struct S3HourScopedDirectory {
54    prefix: String,
55    year: Year,
56    month: Month,
57    day: Day,
58    hour: Hour,
59}
60
61impl S3HourScopedDirectory {
62    pub fn new(prefix: &str, t: UnixSeconds) -> Self {
63        let unix_seconds = i64::try_from(t).expect("timestamp should fit i64");
64        let datetime =
65            OffsetDateTime::from_unix_timestamp(unix_seconds).expect("timestamp should be valid");
66        Self {
67            prefix: prefix.to_string(),
68            year: datetime.year(),
69            month: u8::from(datetime.month()),
70            day: datetime.day(),
71            hour: datetime.hour(),
72        }
73    }
74
75    /// Construct the withdrawal-log directory containing `t`.
76    pub fn withdraw(t: UnixSeconds) -> Self {
77        Self::new(S3_DIR_WITHDRAW, t)
78    }
79
80    /// Construct the heartbeat-log directory containing `t`.
81    pub fn heartbeat(t: UnixSeconds) -> Self {
82        Self::new(S3_DIR_HEARTBEAT, t)
83    }
84
85    pub fn next_dir(&self) -> Self {
86        Self::new(
87            &self.prefix,
88            self.to_unix_seconds().saturating_add(SECONDS_PER_HOUR),
89        )
90    }
91
92    /// Returns the directory for the previous hour. Saturates at the Unix epoch.
93    pub fn prev_dir(&self) -> Self {
94        Self::new(
95            &self.prefix,
96            self.to_unix_seconds().saturating_sub(SECONDS_PER_HOUR),
97        )
98    }
99
100    /// The time at which writes to current S3 directory finish.
101    /// DIR_WRITES_COMPLETION_DELAY accounts for any in-flight retries and clock skew.
102    pub fn write_completion_time(&self) -> UnixSeconds {
103        self.next_dir()
104            .to_unix_seconds()
105            .saturating_add(DIR_WRITES_COMPLETION_DELAY.as_secs())
106    }
107
108    pub fn to_unix_seconds(&self) -> UnixSeconds {
109        let (date, time) = parse_calendar(self.year, self.month, self.day, self.hour)
110            .expect("invariants validated at construction");
111        let ts = PrimitiveDateTime::new(date, time)
112            .assume_utc()
113            .unix_timestamp();
114        UnixSeconds::try_from(ts).expect("timestamp should be non-negative")
115    }
116
117    /// Parses a directory path of the form `{prefix}/{yyyy}/{mm}/{dd}/{hh}/`
118    /// (with or without the trailing slash) back into a directory value.
119    /// Inverse of the `Display` impl.
120    pub fn from_path(path: &str) -> anyhow::Result<Self> {
121        let parts: Vec<&str> = path.trim_end_matches('/').split('/').collect();
122        anyhow::ensure!(
123            parts.len() == 5,
124            "expected `{{prefix}}/YYYY/MM/DD/HH/` in {path}"
125        );
126        let prefix = parts[0];
127        let year: Year = parts[1]
128            .parse()
129            .with_context(|| format!("invalid year in {path}"))?;
130        let month: Month = parts[2]
131            .parse()
132            .with_context(|| format!("invalid month in {path}"))?;
133        let day: Day = parts[3]
134            .parse()
135            .with_context(|| format!("invalid day in {path}"))?;
136        let hour: Hour = parts[4]
137            .parse()
138            .with_context(|| format!("invalid hour in {path}"))?;
139        parse_calendar(year, month, day, hour).with_context(|| format!("invalid path {path}"))?;
140        Ok(Self {
141            prefix: prefix.to_string(),
142            year,
143            month,
144            day,
145            hour,
146        })
147    }
148}
149
150/// Validates the (year, month, day, hour) tuple and returns the corresponding
151/// `(Date, Time)` if every component is in range. Shared by [`S3HourScopedDirectory::from_path`]
152/// (which uses it to validate before construction) and
153/// [`S3HourScopedDirectory::to_unix_seconds`] (which is infallible because the
154/// struct invariant guarantees validity).
155fn parse_calendar(year: Year, month: Month, day: Day, hour: Hour) -> anyhow::Result<(Date, Time)> {
156    let month_enum =
157        time::Month::try_from(month).map_err(|e| anyhow::anyhow!("invalid month {month}: {e}"))?;
158    let date = Date::from_calendar_date(year, month_enum, day)
159        .map_err(|e| anyhow::anyhow!("invalid date {year}-{month:02}-{day:02}: {e}"))?;
160    let time =
161        Time::from_hms(hour, 0, 0).map_err(|e| anyhow::anyhow!("invalid hour {hour}: {e}"))?;
162    Ok((date, time))
163}
164
165impl fmt::Display for S3HourScopedDirectory {
166    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
167        write!(
168            f,
169            "{}/{:04}/{:02}/{:02}/{:02}/",
170            self.prefix, self.year, self.month, self.day, self.hour
171        )
172    }
173}

175#[cfg(test)]
176mod tests {
177    use super::DIR_WRITES_COMPLETION_DELAY;
178    use super::S3HourScopedDirectory;
179
180    #[test]
181    fn test_epoch_directory_format() {
182        let dir = S3HourScopedDirectory::new("heartbeat", 0);
183        assert_eq!(dir.to_string(), "heartbeat/1970/01/01/00/");
184    }
185
186    #[test]
187    fn test_hour_and_day_rollover_format() {
188        let before_hour_boundary = S3HourScopedDirectory::new("withdraw", 3_599);
189        assert_eq!(before_hour_boundary.to_string(), "withdraw/1970/01/01/00/");
190
191        let next_hour = S3HourScopedDirectory::new("withdraw", 3_600);
192        assert_eq!(next_hour.to_string(), "withdraw/1970/01/01/01/");
193
194        let next_day = S3HourScopedDirectory::new("withdraw", 86_400);
195        assert_eq!(next_day.to_string(), "withdraw/1970/01/02/00/");
196    }
197
198    #[test]
199    fn test_prev_dir_walks_back_and_saturates_at_epoch() {
200        let mut dir = S3HourScopedDirectory::new("withdraw", 86_400 + 3_600);
201        assert_eq!(dir.to_string(), "withdraw/1970/01/02/01/");
202        dir = dir.prev_dir();
203        assert_eq!(dir.to_string(), "withdraw/1970/01/02/00/");
204        dir = dir.prev_dir();
205        assert_eq!(dir.to_string(), "withdraw/1970/01/01/23/");
206
207        // Saturates at epoch.
208        let epoch = S3HourScopedDirectory::new("withdraw", 0);
209        assert_eq!(epoch.prev_dir(), epoch);
210    }
211
212    #[test]
213    fn test_from_path_roundtrips_with_display() {
214        let dir = S3HourScopedDirectory::new("withdraw", 1_700_000_000);
215        let displayed = dir.to_string();
216        let parsed = S3HourScopedDirectory::from_path(&displayed).expect("roundtrip");
217        assert_eq!(parsed, dir);
218        // Also accept the trailing-slash-stripped form.
219        let parsed_nopfx = S3HourScopedDirectory::from_path(displayed.trim_end_matches('/'))
220            .expect("roundtrip without trailing slash");
221        assert_eq!(parsed_nopfx, dir);
222    }
223
224    #[test]
225    fn test_from_path_rejects_wrong_shape() {
226        assert!(S3HourScopedDirectory::from_path("withdraw/2024/03/15/").is_err()); // missing hour
227        assert!(S3HourScopedDirectory::from_path("withdraw/2024/03/15/14/extra/").is_err()); // too many parts
228        assert!(S3HourScopedDirectory::from_path("withdraw/2024/13/15/14/").is_err()); // invalid month
229        assert!(S3HourScopedDirectory::from_path("withdraw/2024/02/30/14/").is_err()); // invalid day
230        assert!(S3HourScopedDirectory::from_path("withdraw/2024/02/15/24/").is_err()); // invalid hour
231        assert!(S3HourScopedDirectory::from_path("withdraw/notayear/03/15/14/").is_err()); // non-numeric
232    }
233
234    #[test]
235    fn test_next_dir_and_completion_time() {
236        let mut dir = S3HourScopedDirectory::new("withdraw", 3_599);
237        assert_eq!(dir.to_string(), "withdraw/1970/01/01/00/");
238        assert_eq!(dir.to_unix_seconds(), 0);
239        assert_eq!(
240            dir.write_completion_time(),
241            3_600 + DIR_WRITES_COMPLETION_DELAY.as_secs()
242        );
243
244        for i in 0..24 {
245            assert_eq!(dir.to_string(), format!("withdraw/1970/01/01/{:02}/", i));
246            dir = dir.next_dir();
247        }
248        assert_eq!(dir.to_string(), "withdraw/1970/01/02/00/");
249    }
