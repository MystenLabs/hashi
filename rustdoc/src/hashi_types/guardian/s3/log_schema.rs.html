21/// The wire message stored in a [`crate::guardian::log::LogRecord`]. Its version is serialized
22/// as the record's sibling `schema_version` field rather than as an additional
23/// JSON enum layer.
24///
25/// Readers match these variants exhaustively at their consumption boundary so
26/// adding a schema version requires each reader to opt in explicitly.
27#[derive(Debug)]
28pub enum VersionedLogMessage {
29    V1(LogMessageV1),
30    V2(LogMessageV2),
31}
32
33impl From<LogMessageV1> for VersionedLogMessage {
34    fn from(message: LogMessageV1) -> Self {
35        Self::V1(message)
36    }
37}
38
39impl From<LogMessageV2> for VersionedLogMessage {
40    fn from(message: LogMessageV2) -> Self {
41        Self::V2(message)
42    }
43}
44
45impl Serialize for VersionedLogMessage {
46    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
47    where
48        S: serde::Serializer,
49    {
50        match self {
51            Self::V1(message) => message.serialize(serializer),
52            Self::V2(message) => message.serialize(serializer),
53        }
54    }
55}

57/// Schema-version-1 log messages.
58///
59/// V1 and V2 intentionally have the same message shape: the signed sibling
60/// `schema_version` distinguishes records, while separate types force readers
61/// to handle each deployed version explicitly.
62#[derive(Debug, Serialize, Deserialize)]
63pub enum LogMessageV1 {
64    Heartbeat(HeartbeatLogMessage),
65    Init(Box<InitLogMessage>),
66    Withdrawal(Box<WithdrawalLogMessage>),
67    Ceremony(Box<CeremonyLogMessage>),
68    KpShareState(Box<KpShareStateLogMessage>),
69    CommitteeUpdate(Box<CommitteeUpdateLogMessage>),
70    Genesis(Box<GenesisLogMessage>),
71}
72
73/// Schema-version-2 log messages emitted by the guardian enclave.
74/// Uses an enum discriminator for automatic domain separation between variants.
75// TODO(testnet-wipe): Collapse the V1/V2 compatibility layer into a single log
76// schema once existing testnet records no longer need to be read.
77#[derive(Debug, Serialize, Deserialize)]
78pub enum LogMessageV2 {
79    Heartbeat(HeartbeatLogMessage),
80    Init(Box<InitLogMessage>),
81    Withdrawal(Box<WithdrawalLogMessage>),
82    Ceremony(Box<CeremonyLogMessage>),
83    KpShareState(Box<KpShareStateLogMessage>),
84    CommitteeUpdate(Box<CommitteeUpdateLogMessage>),
85    Genesis(Box<GenesisLogMessage>),
86}
87
88/// Writer-facing alias for the log-message schema emitted by guardians.
89pub type LogMessage = LogMessageV2;
90
91/// Schema-independent category of a Guardian log payload.
92#[derive(Clone, Copy, Debug, Eq, PartialEq)]
93pub enum LogType {
94    Heartbeat,
95    Init,
96    Withdrawal,
97    Ceremony,
98    KpShareState,
99    CommitteeUpdate,
100    Genesis,
101}
102
103impl LogType {
104    pub(super) const fn object_lock_duration(self, policy: S3ObjectLockPolicy) -> Duration {
105        match self {
106            Self::Heartbeat | Self::KpShareState => policy.short_lived,
107            Self::Init
108            | Self::Withdrawal
109            | Self::Ceremony
110            | Self::CommitteeUpdate
111            | Self::Genesis => policy.long_lived,
112        }
113    }
114}

116trait LogMessageSchema {
117    fn log_type(&self) -> LogType;
118
119    fn object_key_pattern(&self, session_id: &str, timestamp_ms: UnixMillis) -> ObjectKeyPattern;
120}
121
122macro_rules! impl_log_message_schema {
123    ($schema:ty) => {
124        impl LogMessageSchema for $schema {
125            fn log_type(&self) -> LogType {
126                match self {
127                    Self::Heartbeat(..) => LogType::Heartbeat,
128                    Self::Init(..) => LogType::Init,
129                    Self::Withdrawal(..) => LogType::Withdrawal,
130                    Self::Ceremony(..) => LogType::Ceremony,
131                    Self::KpShareState(..) => LogType::KpShareState,
132                    Self::CommitteeUpdate(..) => LogType::CommitteeUpdate,
133                    Self::Genesis(..) => LogType::Genesis,
134                }
135            }
136
137            fn object_key_pattern(
138                &self,
139                session_id: &str,
140                timestamp_ms: UnixMillis,
141            ) -> ObjectKeyPattern {
142                match self {
143                    Self::Heartbeat(message) => {
144                        message.object_key_pattern(session_id, timestamp_ms)
145                    }
146                    Self::Init(message) => message.object_key_pattern(session_id),
147                    Self::Withdrawal(message) => {
148                        message.object_key_pattern(session_id, timestamp_ms)
149                    }
150                    Self::Ceremony(message) => message.object_key_pattern(session_id),
151                    Self::KpShareState(message) => message.object_key_pattern(session_id),
152                    Self::CommitteeUpdate(message) => message.object_key_pattern(session_id),
153                    Self::Genesis(message) => message.object_key_pattern(),
154                }
155            }
156        }
157    };
158}

160impl_log_message_schema!(LogMessageV1);
161impl_log_message_schema!(LogMessageV2);

163impl VersionedLogMessage {
164    pub const SCHEMA_VERSION_V1: u64 = 1;
165    pub const SCHEMA_VERSION_V2: u64 = 2;
166
167    pub fn schema_version(&self) -> u64 {
168        match self {
169            Self::V1(_) => Self::SCHEMA_VERSION_V1,
170            Self::V2(_) => Self::SCHEMA_VERSION_V2,
171        }
172    }
173
174    pub fn as_attestation_log(&self) -> Option<&InitLogMessage> {
175        let init = match self {
176            Self::V1(LogMessageV1::Init(init)) | Self::V2(LogMessageV2::Init(init)) => {
177                init.as_ref()
178            }
179            _ => return None,
180        };
181        matches!(init, InitLogMessage::OIAttestationUnsigned { .. }).then_some(init)
182    }
183
184    pub fn is_unsigned(&self) -> bool {
185        self.as_attestation_log().is_some()
186    }
187
188    pub fn log_type(&self) -> LogType {
189        match self {
190            Self::V1(message) => message.log_type(),
191            Self::V2(message) => message.log_type(),
192        }
193    }
194
195    pub(super) fn object_key_pattern(
196        &self,
197        session_id: &str,
198        timestamp_ms: UnixMillis,
199    ) -> ObjectKeyPattern {
200        match self {
201            Self::V1(message) => message.object_key_pattern(session_id, timestamp_ms),
202            Self::V2(message) => message.object_key_pattern(session_id, timestamp_ms),
203        }
204    }
205}
