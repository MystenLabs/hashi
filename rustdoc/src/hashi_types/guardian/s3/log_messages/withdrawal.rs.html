4use super::super::log_layout::ObjectKeyPattern;
5use super::super::log_layout::S3HourScopedDirectory;
6use crate::committee::CommitteeSignature;
7use crate::guardian::LimiterState;
8use crate::guardian::StandardWithdrawalRequestWire;
9use crate::guardian::StandardWithdrawalResponse;
10use crate::guardian::UnixMillis;
11use crate::guardian::WithdrawalID;
12use crate::guardian::unix_millis_to_seconds;
13use bitcoin::Txid;
14use serde::Deserialize;
15use serde::Serialize;
16
17#[derive(Debug, Serialize, Deserialize)]
18pub enum WithdrawalLogMessage {
19    Success {
20        txid: Txid,
21        request_data: StandardWithdrawalRequestWire,
22        request_sign: CommitteeSignature,
23        response: StandardWithdrawalResponse,
24        /// Limiter state after this withdrawal was consumed. The KP rotating in
25        /// the next enclave reads the max-seq Success log and uses its
26        /// `post_state` as the new enclave's initial limiter state.
27        post_state: LimiterState,
28    },
29    Failure {
30        request_data: StandardWithdrawalRequestWire,
31        request_sign: CommitteeSignature,
32        error: String,
33    },
34}
35
36impl WithdrawalLogMessage {
37    /// Prefix shared by all successful withdrawal object names within an hour.
38    pub const SUCCESS_OBJECT_KEY_PREFIX: &'static str = "success-";
39
40    /// Success keys lead with `success-{seq:020}` so that lexicographic listing
41    /// within an hour bucket is also seq-sorted — the last key is the max-seq
42    /// log, which the KP reads to recover limiter state. Failures don't have a
43    /// meaningful seq (the request's seq may be stale), so they use a random
44    /// suffix for dedup.
45    pub fn object_key_pattern(
46        &self,
47        session_id: &str,
48        timestamp_ms: UnixMillis,
49    ) -> ObjectKeyPattern {
50        let directory = S3HourScopedDirectory::withdraw(unix_millis_to_seconds(timestamp_ms));
51        match self {
52            Self::Success { request_data, .. } => ObjectKeyPattern::Fixed(format!(
53                "{directory}{}{:020}-{session_id}-wid{}.json",
54                Self::SUCCESS_OBJECT_KEY_PREFIX,
55                request_data.seq,
56                request_data.wid,
57            )),
58            Self::Failure { request_data, .. } => ObjectKeyPattern::RandomSuffix(format!(
59                "{directory}failure-{session_id}-wid{}-",
60                request_data.wid,
61            )),
62        }
63    }
64
65    pub fn wid(&self) -> WithdrawalID {
66        match self {
67            WithdrawalLogMessage::Success { request_data, .. } => request_data.wid,
68            WithdrawalLogMessage::Failure { request_data, .. } => request_data.wid,
69        }
70    }
71}
