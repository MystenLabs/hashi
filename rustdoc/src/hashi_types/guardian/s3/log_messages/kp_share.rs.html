4use super::super::log_layout::ObjectKeyPattern;
5use super::super::log_layout::S3_DIR_KP_SHARES;
6use crate::guardian::KpEncryptedShareRoster;
7use serde::Deserialize;
8use serde::Serialize;
9
10/// Current encrypted KP share state for a secret-sharing instance. The initial
11/// ceremony writes `cert_seq = 0`; later individual KP cert rotations can write
12/// higher `cert_seq` entries for the same `sharing_seq` without changing the
13/// `ceremony/` instance. Both S3 schema versions use this scalar-recipient
14/// payload so deployed V1 signing preimages remain unchanged.
15#[derive(Debug, Serialize, Deserialize, Clone, PartialEq)]
16pub struct KpShareStateLogMessage {
17    pub sharing_seq: u64,
18    pub cert_seq: u64,
19    pub encrypted_shares: KpEncryptedShareRoster,
20}
21
22impl KpShareStateLogMessage {
23    pub fn new(sharing_seq: u64, cert_seq: u64, encrypted_shares: KpEncryptedShareRoster) -> Self {
24        Self {
25            sharing_seq,
26            cert_seq,
27            encrypted_shares,
28        }
29    }
30
31    /// `kp-shares/{sharing_seq:020}/` — the slash-terminated S3 key prefix
32    /// containing every cert-state version for one `SecretSharingInstance`.
33    pub fn object_key_dir(sharing_seq: u64) -> String {
34        format!("{S3_DIR_KP_SHARES}/{sharing_seq:020}/")
35    }
36
37    /// `kp-shares/{sharing_seq:020}/{cert_seq:020}-{session_id}.json` — the
38    /// object key for one written KP share state.
39    pub fn object_key(session_id: &str, sharing_seq: u64, cert_seq: u64) -> String {
40        format!(
41            "{}{:020}-{session_id}.json",
42            Self::object_key_dir(sharing_seq),
43            cert_seq
44        )
45    }
46
47    pub fn object_key_pattern(&self, session_id: &str) -> ObjectKeyPattern {
48        ObjectKeyPattern::Fixed(Self::object_key(
49            session_id,
50            self.sharing_seq,
51            self.cert_seq,
52        ))
53    }
54}
