25/// Configuration recorded when operator initialization completes.
26#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
27pub struct OperatorInitInfo {
28    /// Installed deployment policy, including PCR pins.
29    pub deployment: DeploymentConfig,
30    /// KPs use this key to encrypt shares for the initialized session.
31    #[serde(with = "hex::serde")]
32    pub encryption_pubkey: EncPubKeyBytes,
33    pub mode: OperatorInitMode,
34}
35
36#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
37pub enum OperatorInitMode {
38    Ceremony,
39    Withdraw(Box<WithdrawOperatorInitInfo>),
40}

42/// Withdraw configuration installed during OI.
43/// The Hashi object ID and MPC master come from persisted or supplied genesis state.
44#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
45pub struct WithdrawOperatorInitInfo {
46    pub secret_sharing_instance: SecretSharingInstance,
47    #[serde(with = "hex::serde")]
48    pub config_hash: [u8; 32],
49    pub limiter_config: LimiterConfig,
50    /// Hashi shared object this guardian serves.
51    pub hashi_object_id: sui_sdk_types::Address,
52    /// MPC committee verifying key used for Bitcoin key derivation.
53    pub mpc_master_g: HashiMasterG,
54    /// Supplied genesis digest for KP authorization during PI.
55    /// Absent when using the persisted genesis record.
56    #[serde(with = "crate::guardian::serde::option_hex_32")]
57    pub genesis_state_hash: Option<[u8; 32]>,
58}

60impl OperatorInitInfo {
61    pub fn mode(&self) -> EnclaveMode {
62        match &self.mode {
63            OperatorInitMode::Ceremony => EnclaveMode::Ceremony,
64            OperatorInitMode::Withdraw(_) => EnclaveMode::Withdraw,
65        }
66    }
67
68    /// Check that live GuardianInfo matches this record and is at the post-OI stage.
69    pub fn match_post_oi_guardian_info(&self, live_info: &GuardianInfo) -> anyhow::Result<()> {
70        let expected_lifecycle = match &self.mode {
71            OperatorInitMode::Ceremony => CeremonyStage::OperatorInitialized.into(),
72            OperatorInitMode::Withdraw(_) => WithdrawStage::OperatorInitialized.into(),
73        };
74        anyhow::ensure!(
75            live_info.lifecycle == expected_lifecycle,
76            "S3 OI mode {:?} does not match live post-OperatorInit lifecycle {:?}",
77            self.mode(),
78            live_info.lifecycle
79        );
80        anyhow::ensure!(
81            live_info.deployment_info.as_ref() == Some(&self.deployment.summary()),
82            "S3 OI deployment differs from live post-OperatorInit GuardianInfo"
83        );
84        anyhow::ensure!(
85            live_info.encryption_pubkey == self.encryption_pubkey,
86            "S3 OI encryption pubkey differs from live post-OperatorInit GuardianInfo"
87        );
88        anyhow::ensure!(
89            live_info.enclave_btc_pubkey.is_none()
90                && live_info.limiter_state.is_none()
91                && live_info.current_committee_epoch.is_none(),
92            "live post-OperatorInit GuardianInfo contains later-stage state"
93        );
94        match &self.mode {
95            OperatorInitMode::Ceremony => anyhow::ensure!(
96                live_info.secret_sharing_instance.is_none()
97                    && live_info.config_hash.is_none()
98                    && live_info.limiter_config.is_none()
99                    && live_info.hashi_object_id.is_none()
100                    && live_info.mpc_master_g.is_none()
101                    && live_info.genesis_state_hash.is_none(),
102                "live ceremony GuardianInfo contains withdraw initialization state"
103            ),
104            OperatorInitMode::Withdraw(withdraw) => {
105                anyhow::ensure!(
106                    live_info.secret_sharing_instance.as_ref()
107                        == Some(&withdraw.secret_sharing_instance),
108                    "S3 OI secret-sharing instance differs from live post-OperatorInit GuardianInfo"
109                );
110                anyhow::ensure!(
111                    live_info.config_hash == Some(withdraw.config_hash),
112                    "S3 OI config_hash differs from live post-OperatorInit GuardianInfo"
113                );
114                anyhow::ensure!(
115                    live_info.limiter_config == Some(withdraw.limiter_config),
116                    "S3 OI limiter config differs from live post-OperatorInit GuardianInfo"
117                );
118                anyhow::ensure!(
119                    live_info.hashi_object_id == Some(withdraw.hashi_object_id),
120                    "S3 OI Hashi object ID differs from live post-OperatorInit GuardianInfo"
121                );
122                anyhow::ensure!(
123                    live_info.mpc_master_g == Some(withdraw.mpc_master_g),
124                    "S3 OI MPC master G differs from live post-OperatorInit GuardianInfo"
125                );
126                anyhow::ensure!(
127                    live_info.genesis_state_hash == withdraw.genesis_state_hash,
128                    "S3 OI genesis_state_hash differs from live post-OperatorInit GuardianInfo"
129                );
130            }
131        }
132        Ok(())
133    }
134}

136/// Initialization logs in order: OI attestation, OI completion, PI completion,
137/// then operator activation (OA). Ceremony mode stops after OI.
138#[derive(Debug, Serialize, Deserialize)]
139pub enum InitLogMessage {
140    /// OI attestation record, signed by the session key it attests.
141    OIAttestation {
142        attestation: NitroAttestation,
143        #[serde(with = "crate::guardian::serde::guardian_pubkey")]
144        signing_public_key: GuardianPubKey,
145    },
146    /// OI completion, signed by the key in the preceding attestation record.
147    OIGuardianInfo(Box<OperatorInitInfo>),
148    /// Provisioner initialization (PI) reconstructed the enclave BTC key.
149    PIEnclaveFullyInitialized {
150        sharing_seq: u64,
151        share_ids: Vec<ShareID>,
152        enclave_btc_pubkey: BitcoinPubkey,
153    },
154    /// Operator activation installed serving state.
155    OAActivated {
156        #[serde(with = "hex::serde")]
157        state_hash: [u8; 32],
158        #[serde(with = "hex::serde")]
159        config_hash: [u8; 32],
160        sharing_seq: u64,
161        committee_epoch: u64,
162        limiter_state: LimiterState,
163    },
164}
165
166impl InitLogMessage {
167    pub const OI_ATTESTATION: &'static str = "01-oi-attestation";
168    pub const OI_GUARDIAN_INFO: &'static str = "02-oi-guardian-info";
169    pub const PI_FULLY_INITIALIZED: &'static str = "03-pi-enclave-fully-initialized";
170    pub const OA_ACTIVATED: &'static str = "04-oa-activated";
171
172    pub fn object_key(&self, session_id: &str) -> String {
173        let suffix = match self {
174            InitLogMessage::OIAttestation { .. } => Self::OI_ATTESTATION,
175            InitLogMessage::OIGuardianInfo(_) => Self::OI_GUARDIAN_INFO,
176            InitLogMessage::PIEnclaveFullyInitialized { .. } => Self::PI_FULLY_INITIALIZED,
177            InitLogMessage::OAActivated { .. } => Self::OA_ACTIVATED,
178        };
179
180        Self::object_key_for_suffix(session_id, suffix)
181    }
182
183    pub fn attestation_object_key(session_id: &str) -> String {
184        Self::object_key_for_suffix(session_id, Self::OI_ATTESTATION)
185    }
186
187    pub fn guardian_info_object_key(session_id: &str) -> String {
188        Self::object_key_for_suffix(session_id, Self::OI_GUARDIAN_INFO)
189    }
190
191    pub fn pi_fully_initialized_object_key(session_id: &str) -> String {
192        Self::object_key_for_suffix(session_id, Self::PI_FULLY_INITIALIZED)
193    }
194
195    pub fn oa_activated_object_key(session_id: &str) -> String {
196        Self::object_key_for_suffix(session_id, Self::OA_ACTIVATED)
197    }

199    /// Verify facts repeated between 02 OIGuardianInfo and 03
200    /// PIEnclaveFullyInitialized.
201    pub fn verify_oi_pi_consistency(
202        oi_info: &OperatorInitInfo,
203        pi_message: &Self,
204    ) -> GuardianResult<()> {
205        let Self::PIEnclaveFullyInitialized {
206            sharing_seq: pi_sharing_seq,
207            share_ids,
208            ..
209        } = pi_message
210        else {
211            return Err(InvalidS3Log(
212                "expected PIEnclaveFullyInitialized init log".into(),
213            ));
214        };
215        let OperatorInitMode::Withdraw(withdraw) = &oi_info.mode else {
216            return Err(InvalidS3Log(
217                "PI requires withdraw-mode operator initialization".into(),
218            ));
219        };
220        let oi_instance = &withdraw.secret_sharing_instance;
221        let oi_sharing_seq = oi_instance.sharing_seq();
222
223        if *pi_sharing_seq != oi_sharing_seq {
224            return Err(InvalidS3Log(format!(
225                "PIEnclaveFullyInitialized sharing_seq {pi_sharing_seq} differs from OIGuardianInfo sharing_seq {oi_sharing_seq}"
226            )));
227        }
228
229        let unique_share_ids = share_ids.iter().copied().collect::<BTreeSet<_>>();
230        if unique_share_ids.len() != share_ids.len() {
231            return Err(InvalidS3Log(
232                "PIEnclaveFullyInitialized contains duplicate share_ids".into(),
233            ));
234        }
235        if share_ids.len() < oi_instance.threshold() || share_ids.len() > oi_instance.num_shares() {
236            return Err(InvalidS3Log(format!(
237                "PIEnclaveFullyInitialized has {} share_ids; expected between {} and {}",
238                share_ids.len(),
239                oi_instance.threshold(),
240                oi_instance.num_shares(),
241            )));
242        }
243
244        let commitment_ids = oi_instance
245            .commitments()
246            .iter()
247            .map(|commitment| commitment.id)
248            .collect::<BTreeSet<_>>();
249        if !unique_share_ids.is_subset(&commitment_ids) {
250            return Err(InvalidS3Log(
251                "PIEnclaveFullyInitialized contains share_ids absent from OIGuardianInfo commitments"
252                    .into(),
253            ));
254        }
255        Ok(())
256    }
257
258    /// Verify facts repeated between 02 OIGuardianInfo and 04 OAActivated.
259    pub fn verify_oi_oa_consistency(
260        oi_info: &OperatorInitInfo,
261        oa_message: &Self,
262    ) -> GuardianResult<()> {
263        let Self::OAActivated {
264            config_hash: oa_config_hash,
265            sharing_seq: oa_sharing_seq,
266            ..
267        } = oa_message
268        else {
269            return Err(InvalidS3Log("expected OAActivated init log".into()));
270        };
271        let OperatorInitMode::Withdraw(withdraw) = &oi_info.mode else {
272            return Err(InvalidS3Log(
273                "OA requires withdraw-mode operator initialization".into(),
274            ));
275        };
276        let oi_sharing_seq = withdraw.secret_sharing_instance.sharing_seq();
277        let oi_config_hash = withdraw.config_hash;
278
279        if *oa_sharing_seq != oi_sharing_seq {
280            return Err(InvalidS3Log(format!(
281                "OAActivated sharing_seq {oa_sharing_seq} differs from OIGuardianInfo sharing_seq {oi_sharing_seq}"
282            )));
283        }
284        if *oa_config_hash != oi_config_hash {
285            return Err(InvalidS3Log(
286                "OAActivated config_hash differs from OIGuardianInfo config_hash".into(),
287            ));
288        }
289        Ok(())
290    }
291
292    /// Verify facts repeated between 03 PIEnclaveFullyInitialized and 04
293    /// OAActivated.
294    pub fn verify_pi_oa_consistency(pi_message: &Self, oa_message: &Self) -> GuardianResult<()> {
295        let Self::PIEnclaveFullyInitialized {
296            sharing_seq: pi_sharing_seq,
297            ..
298        } = pi_message
299        else {
300            return Err(InvalidS3Log(
301                "expected PIEnclaveFullyInitialized init log".into(),
302            ));
303        };
304        let Self::OAActivated {
305            sharing_seq: oa_sharing_seq,
306            ..
307        } = oa_message
308        else {
309            return Err(InvalidS3Log("expected OAActivated init log".into()));
310        };
311
312        if oa_sharing_seq != pi_sharing_seq {
313            return Err(InvalidS3Log(format!(
314                "OAActivated sharing_seq {oa_sharing_seq} differs from PIEnclaveFullyInitialized sharing_seq {pi_sharing_seq}"
315            )));
316        }
317        Ok(())
318    }
319
320    fn object_key_for_suffix(session_id: &str, suffix: &str) -> String {
321        format!("{S3_DIR_INIT}/{session_id}/{suffix}.json")
322    }

365    #[test]
366    fn verifies_pairwise_init_log_consistency() {
367        let oi_info = OperatorInitInfo::mock_for_testing();
368        let pi = pi_message(0);
369        let oa = oa_message([2; 32], 0);
370
371        InitLogMessage::verify_oi_pi_consistency(&oi_info, &pi).unwrap();
372        InitLogMessage::verify_oi_oa_consistency(&oi_info, &oa).unwrap();
373        InitLogMessage::verify_pi_oa_consistency(&pi, &oa).unwrap();
374
375        assert!(InitLogMessage::verify_oi_pi_consistency(&oi_info, &pi_message(1)).is_err());
376        assert!(
377            InitLogMessage::verify_oi_pi_consistency(
378                &oi_info,
379                &pi_message_with_ids(0, share_ids(&[1, 1, 2])),
380            )
381            .is_err()
382        );
383        assert!(
384            InitLogMessage::verify_oi_pi_consistency(
385                &oi_info,
386                &pi_message_with_ids(0, share_ids(&[1, 2])),
387            )
388            .is_err()
389        );
390        assert!(
391            InitLogMessage::verify_oi_pi_consistency(
392                &oi_info,
393                &pi_message_with_ids(0, share_ids(&[1, 2, 6])),
394            )
395            .is_err()
396        );
397        assert!(
398            InitLogMessage::verify_oi_pi_consistency(
399                &oi_info,
400                &pi_message_with_ids(0, share_ids(&[1, 2, 3, 4, 5, 6])),
401            )
402            .is_err()
403        );
404        assert!(
405            InitLogMessage::verify_oi_oa_consistency(&oi_info, &oa_message([3; 32], 0)).is_err()
406        );
407        assert!(
408            InitLogMessage::verify_oi_oa_consistency(&oi_info, &oa_message([2; 32], 1)).is_err()
409        );
410        assert!(InitLogMessage::verify_pi_oa_consistency(&pi, &oa_message([2; 32], 1)).is_err());
