4//! Guardian S3 Object Lock retention policies.
5
6use serde::Deserialize;
7use serde::Serialize;
8use std::time::Duration;
9
10use super::LogType;
11
12const ONE_WEEK: Duration = Duration::from_secs(7 * 24 * 60 * 60);
13const THIRTY_DAYS: Duration = Duration::from_secs(30 * 24 * 60 * 60);
14const NINETY_DAYS: Duration = Duration::from_secs(90 * 24 * 60 * 60);
15const TEN_YEARS: Duration = Duration::from_secs(10 * 365 * 24 * 60 * 60);
16
17/// Object-lock retention policy for Guardian S3 logs.
18///
19/// Long-lived records include initialization, withdrawals, ceremonies,
20/// committee updates, and genesis. They remain usable only while both the
21/// record and its initialization chain are protected from replacement.
22///
23/// Object Lock expiry permits deletion; it does not trigger it. Heartbeats may
24/// be removed by a separate lifecycle policy. Encrypted KP shares use the same
25/// short-lived lock so stale shares can be explicitly deleted after their
26/// recovery window, for example when a KP encryption key is retired or lost.
27#[derive(Clone, Copy, Debug, Eq, PartialEq)]
28pub struct S3ObjectLockPolicy {
29    pub long_lived: Duration,
30    pub short_lived: Duration,
31}
32
33// TODO: Retire this enum once the repository has a canonical
34// Hashi network enum from which the retention policy can be derived.
35/// Hashi deployment class used to select an S3 object-lock policy.
36///
37/// This is intentionally independent of `bitcoin::Network`: a Hashi
38/// deployment comprises both a Sui chain and a Bitcoin chain, and neither one
39/// alone is the deployment identity.
40#[derive(Clone, Copy, Debug, Deserialize, Eq, PartialEq, Serialize)]
41#[serde(rename_all = "snake_case")]
42pub enum S3RetentionEnvironment {
43    Devnet,
44    Mainnet,
45    Testnet,
46}
47
48/// Devnet records balance useful debugging history with rapid iteration.
49pub const DEVNET_S3_OBJECT_LOCK_POLICY: S3ObjectLockPolicy = S3ObjectLockPolicy {
50    long_lived: NINETY_DAYS,
51    short_lived: ONE_WEEK,
52};
53
54/// Mainnet records are durable audit and recovery inputs.
55pub const MAINNET_S3_OBJECT_LOCK_POLICY: S3ObjectLockPolicy = S3ObjectLockPolicy {
56    long_lived: TEN_YEARS,
57    short_lived: THIRTY_DAYS,
58};
59
60/// Testnet is a durable deployment and uses the mainnet retention policy.
61pub const TESTNET_S3_OBJECT_LOCK_POLICY: S3ObjectLockPolicy = S3ObjectLockPolicy {
62    long_lived: TEN_YEARS,
63    short_lived: THIRTY_DAYS,
64};
65
66impl S3ObjectLockPolicy {
67    pub const fn for_environment(environment: S3RetentionEnvironment) -> Self {
68        match environment {
69            S3RetentionEnvironment::Devnet => DEVNET_S3_OBJECT_LOCK_POLICY,
70            S3RetentionEnvironment::Mainnet => MAINNET_S3_OBJECT_LOCK_POLICY,
71            S3RetentionEnvironment::Testnet => TESTNET_S3_OBJECT_LOCK_POLICY,
72        }
73    }
74
75    pub(super) const fn duration_for(self, log_type: LogType) -> Duration {
76        match log_type {
77            LogType::Heartbeat | LogType::KpShareState => self.short_lived,
78            LogType::Init
79            | LogType::Withdrawal
80            | LogType::Ceremony
81            | LogType::CommitteeUpdate
82            | LogType::Genesis => self.long_lived,
83        }
84    }
85}
