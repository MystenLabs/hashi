4//! Guardian S3 logs. `messages` holds the message families the enclave emits;
5//! `record` holds the `LogRecord` wrapper written to S3 and its keying/signing;
6//! `ceremony_state` combines the ceremony and KP-share messages for readers.
7//! The S3 layout constants live here, while object-lock durations live in
8//! `retention`. See `crates/hashi-guardian/README.md` for the canonical key
9//! layout.
10//!
11//! Writers call `LogRecord::new()`, which finalizes the pattern exactly once,
12//! stores the resulting key, and uses it for signing and upload.
13//! Readers either fetch a deterministic record using `object_key()` or list
14//! records in `object_key_dir()`. In both read paths, the S3 client rejects a
15//! record unless its signed key matches the actual key returned by S3.
16

18pub mod messages;
19pub mod record;
20pub mod retention;
21pub mod schema;

29pub enum ObjectKeyPattern {
30    Fixed(String),
31    /// Complete key prefix before the random suffix; finalize() appends the suffix.
32    RandomSuffix(String),
33}
34
35pub(super) enum LogType {
36    Heartbeat,
37    Init,
38    Withdrawal,
39    Ceremony,
40    KpShareState,
41    CommitteeUpdate,
42    Genesis,
43}
44
45impl ObjectKeyPattern {
46    /// Finalizes the pattern into the complete S3 object key.
47    pub fn finalize(self) -> String {
48        match self {
49            Self::Fixed(key) => key,
50            Self::RandomSuffix(prefix) => {
51                format!("{prefix}{:08x}.json", rand::random::<u32>())
52            }
53        }
54    }
55}
