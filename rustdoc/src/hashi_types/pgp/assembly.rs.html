22pub struct EncryptionRecipient<'a> {
23    pub cert: &'a openpgp::Cert,
24    pub subkey: Option<&'a Fingerprint>,
25}

27pub struct RecipientAttestation<'a> {
28    pub device_pem: &'a [u8],
29    pub dec_pem: &'a [u8],
30}

32pub fn assemble_public_cert(
33    primary: &openpgp::Cert,
34    recipients: &[EncryptionRecipient<'_>],
35    attestations: &BTreeMap<Fingerprint, RecipientAttestation<'_>>,
36    require_attestation: bool,
37    certifier: &mut dyn Signer,
38) -> Result<PgpPublicCert> {
39    ensure!(
40        !recipients.is_empty(),
41        "at least one encryption recipient is required"
42    );
43    ensure!(
44        !primary.is_tsk(),
45        "primary certificate must not contain secret key material"
46    );
47    let now = SystemTime::now();
48    let primary_key = primary
49        .keys()
50        .with_policy(&*POLICY, now)
51        .supported()
52        .alive()
53        .revoked(false)
54        .find(|key| key.primary())
55        .context("primary certificate has no usable primary key")?;
56    ensure!(
57        primary_key.for_certification() && primary_key.for_signing(),
58        "primary key must permit certification and signing"
59    );
60    ensure!(
61        !primary_key.for_transport_encryption() && !primary_key.for_storage_encryption(),
62        "encryption-capable primary keys cannot be used as the assembled identity"
63    );
64    ensure!(
65        certifier.public().fingerprint() == primary.fingerprint(),
66        "certifier does not match the primary certificate"
67    );
68    let primary_features = primary_key
69        .valid_cert()
70        .features()
71        .unwrap_or_else(Features::empty);
72
73    let mut selected = BTreeSet::new();
74    let mut bindings = Vec::with_capacity(recipients.len());
75    for recipient in recipients {
76        ensure!(
77            !recipient.cert.is_tsk(),
78            "recipient certificate must not contain secret key material"
79        );
80        let source_primary = recipient
81            .cert
82            .keys()
83            .with_policy(&*POLICY, now)
84            .supported()
85            .alive()
86            .revoked(false)
87            .find(|key| key.primary())
88            .context("recipient certificate has no usable primary key")?;
89        let mut candidates = recipient
90            .cert
91            .keys()
92            .subkeys()
93            .with_policy(&*POLICY, now)
94            .supported()
95            .alive()
96            .revoked(false)
97            .for_transport_encryption()
98            .filter(|key| {
99                recipient
100                    .subkey
101                    .is_none_or(|fp| key.key().fingerprint() == *fp)
102            });
103        let key = candidates
104            .next()
105            .context("recipient has no matching usable encryption subkey")?;
106        ensure!(
107            candidates.next().is_none(),
108            "recipient has ambiguous encryption subkeys; select a fingerprint"
109        );
110        let fingerprint = key.key().fingerprint();
111        let recipient_features = key.valid_cert().features().unwrap_or_else(Features::empty);
112        ensure!(
113            !primary_features.supports_seipdv2() || recipient_features.supports_seipdv2(),
114            "recipient encryption subkey {fingerprint} does not support SEIPDv2 \
115             required by certifier {}",
116            primary.fingerprint()
117        );
118        let attestation = attestations.get(&fingerprint);
119        ensure!(
120            selected.insert(fingerprint),
121            "duplicate encryption recipient"
122        );
123        match attestation {
124            Some(attestation) => super::yubikey_attestation::verify_yubikey_encryption_key(
125                key.key(),
126                attestation.device_pem,
127                attestation.dec_pem,
128            )
129            .context("recipient DEC attestation is invalid")?,
130            None => ensure!(
131                !require_attestation,
132                "recipient DEC attestation is required"
133            ),
134        }
135        let expiration = [
136            key.key_expiration_time(),
137            key.binding_signature().signature_expiration_time(),
138            key.direct_key_signature()
139                .ok()
140                .and_then(|sig| sig.signature_expiration_time()),
141            source_primary.key_expiration_time(),
142            source_primary
143                .binding_signature()
144                .signature_expiration_time(),
145            source_primary
146                .direct_key_signature()
147                .ok()
148                .and_then(|sig| sig.signature_expiration_time()),
149        ]
150        .into_iter()
151        .flatten()
152        .min();
153        ensure!(
154            expiration.is_none_or(|expiration| expiration > now),
155            "recipient has expired"
156        );
157        let binding = SignatureBuilder::new(SignatureType::SubkeyBinding)
158            .set_signature_creation_time(now)?
159            .set_key_flags(KeyFlags::empty().set_transport_encryption())?;
160        let binding = match expiration {
161            Some(expiration) => binding.set_key_expiration_time(key.key(), expiration)?,
162            // An absent subpacket inherits the primary's validity period,
163            // measured from this subkey's creation time. Encode no expiry.
164            None => binding.set_key_validity_period(std::time::Duration::ZERO)?,
165        };
166        bindings.push((key.key(), binding));
167    }
168    ensure!(
169        attestations
170            .keys()
171            .all(|fingerprint| selected.contains(fingerprint)),
172        "attestation supplied for an unselected encryption key"
173    );
174
175    let mut packets = Vec::with_capacity(bindings.len() * 2);
176    for (key, binding) in bindings {
177        let signature = binding
178            .sign_subkey_binding(certifier, primary.primary_key().key(), key)
179            .context("failed to certify encryption subkey")?;
180        packets.push(openpgp::Packet::PublicSubkey(key.clone()));
181        packets.push(openpgp::Packet::Signature(signature));
182    }
183
184    let (assembled, _) = primary
185        .clone()
186        .retain_subkeys(|_| false)
187        .insert_packets(packets)?;
188    let mut armored = Vec::new();
189    assembled.armored().serialize(&mut armored)?;
190    let result = PgpPublicCert::new(String::from_utf8(armored)?)?;
191    let actual = super::usable_keys(&result.cert)
192        .for_transport_encryption()
193        .map(|key| key.key().fingerprint())
194        .collect::<BTreeSet<_>>();
195    ensure!(
196        actual == selected,
197        "assembled certificate does not contain exactly the selected usable recipients"
198    );
199    Ok(result)
200}

216    fn signer(cert: &openpgp::Cert) -> openpgp::crypto::KeyPair {
217        cert.primary_key()
218            .key()
219            .clone()
220            .parts_into_secret()
221            .unwrap()
222            .into_keypair()

226    struct CountingSigner {
227        inner: openpgp::crypto::KeyPair,
228        calls: usize,
229    }
230
231    impl Signer for CountingSigner {
232        fn public(
233            &self,
234        ) -> &openpgp::packet::Key<
235            openpgp::packet::key::PublicParts,
236            openpgp::packet::key::UnspecifiedRole,
237        > {
238            self.inner.public()
239        }
240
241        fn sign(
242            &mut self,
243            hash_algo: openpgp::types::HashAlgorithm,
244            digest: &[u8],
245        ) -> openpgp::Result<openpgp::crypto::mpi::Signature> {
246            self.calls += 1;
247            self.inner.sign(hash_algo, digest)
248        }
249    }

252    fn assembly_preserves_identity_and_exact_public_subkeys() {
253        let primary = identity();
254        let mut signer = signer(&primary);
255        let primary = primary.strip_secret_key_material();
256        let recipients: Vec<_> = (0..3)
257            .map(|_| {
258                CertBuilder::new()
259                    .add_transport_encryption_subkey()
260                    .add_transport_encryption_subkey()
261                    .generate()
262                    .unwrap()
263                    .0
264                    .strip_secret_key_material()
265            })
266            .collect();
267        let keys: Vec<_> = recipients
268            .iter()
269            .enumerate()
270            .map(|(i, cert)| cert.keys().subkeys().nth(i % 2).unwrap().key())
271            .collect();
272        let fingerprints: Vec<_> = keys.iter().map(|key| key.fingerprint()).collect();
273        let selections: Vec<_> = recipients
274            .iter()
275            .zip(&fingerprints)

277                cert,
278                subkey: Some(fingerprint),

283                .unwrap();
284        assert_eq!(result.fingerprint(), primary.fingerprint());
285        assert_eq!(

287            primary,
288            "all original identity packets must be preserved"
289        );
290        assert!(!result.cert.is_tsk());

295        assert_eq!(

304                .iter()

312    fn assembly_rejects_expired_or_revoked_recipient_keys_before_signing() {
313        let primary = identity();
314        let mut signer = CountingSigner {
315            inner: signer(&primary),
316            calls: 0,
317        };
318        let primary = primary.strip_secret_key_material();
319        let creation = SystemTime::now() - Duration::from_secs(7200);
320        let valid = CertBuilder::new()
321            .set_creation_time(creation)
322            .set_validity_period(Duration::from_secs(86400))
323            .add_subkey(
324                KeyFlags::empty().set_transport_encryption(),
325                Duration::from_secs(86400),
326                None,
327            )
328            .generate()
329            .unwrap()
330            .0;
331        let valid_public = valid.clone().strip_secret_key_material();
332        for (case, primary_invalid, revoked) in [
333            ("expired subkey", false, false),
334            ("revoked subkey", false, true),
335            ("expired source primary", true, false),
336            ("revoked source primary", true, true),
337        ] {
338            let invalid = if revoked {
339                let mut source_signer = self::signer(&valid);
340                let revocation = if primary_invalid {
341                    CertRevocationBuilder::new()
342                        .build(&mut source_signer, &valid, None)
343                        .unwrap()
344                } else {
345                    SubkeyRevocationBuilder::new()
346                        .build(
347                            &mut source_signer,
348                            &valid,
349                            valid.keys().subkeys().next().unwrap().key(),
350                            None,
351                        )
352                        .unwrap()
353                };
354                valid.clone().insert_packets(revocation).unwrap().0
355            } else {
356                CertBuilder::new()
357                    .set_creation_time(creation)
358                    .set_validity_period(Duration::from_secs(if primary_invalid {
359                        3600
360                    } else {
361                        86400
362                    }))
363                    .add_subkey(
364                        KeyFlags::empty().set_transport_encryption(),
365                        Duration::from_secs(if primary_invalid { 86400 } else { 3600 }),
366                        None,
367                    )
368                    .generate()
369                    .unwrap()
370                    .0
371            }
372            .strip_secret_key_material();
373            let fingerprint = invalid.keys().subkeys().next().unwrap().key().fingerprint();
374            let error = assemble_public_cert(
375                &primary,
376                &[
377                    EncryptionRecipient {
378                        cert: &valid_public,
379                        subkey: None,
380                    },
381                    EncryptionRecipient {
382                        cert: &invalid,
383                        subkey: Some(&fingerprint),
384                    },
385                ],
386                &BTreeMap::new(),
387                false,
388                &mut signer,
389            )
390            .unwrap_err();
391            let reason = if primary_invalid {
392                "no usable primary key"
393            } else {
394                "no matching usable encryption subkey"
395            };
396            assert!(error.to_string().contains(reason), "{case}: {error:#}");
397            assert_eq!(signer.calls, 0, "{case} must be rejected before signing");
398        }
399        let assembled = assemble_public_cert(
400            &primary,
401            &[EncryptionRecipient {
402                cert: &valid_public,
403                subkey: None,
404            }],
405            &BTreeMap::new(),
406            false,
407            &mut signer,
408        )
409        .unwrap();
410        assert_eq!(signer.calls, 1);
411        assert_eq!(
412            super::super::usable_keys(&assembled.cert)
413                .for_transport_encryption()
414                .next()
415                .unwrap()
416                .key()
417                .fingerprint(),
418            valid_public
419                .keys()
420                .subkeys()
421                .next()
422                .unwrap()
423                .key()
424                .fingerprint()

703    fn feature_cert(features: Features, with_userid: bool, recipient: bool) -> openpgp::Cert {
704        let mut builder = CertBuilder::new()
705            .set_profile(openpgp::Profile::RFC4880)
706            .unwrap()
707            .set_features(if with_userid {
708                Features::empty()
709            } else {
710                features.clone()
711            })
712            .unwrap()
713            .set_primary_key_flags(KeyFlags::empty().set_certification().set_signing());
714        if recipient {
715            builder = builder.add_transport_encryption_subkey();
716        }
717        let mut cert = builder.generate().unwrap().0;
718        if with_userid {
719            // Deliberately disagree with the direct-key signature: the
720            // primary User ID's preferences must take precedence.
721            let userid = openpgp::packet::UserID::from("Backup identity");
722            let signature = SignatureBuilder::new(SignatureType::PositiveCertification)
723                .set_features(features.clone())
724                .unwrap()
725                .sign_userid_binding(&mut signer(&cert), cert.primary_key().key(), &userid)
726                .unwrap();
727            cert = cert
728                .insert_packets([
729                    openpgp::Packet::UserID(userid),
730                    openpgp::Packet::Signature(signature),
731                ])
732                .unwrap()
733                .0;
734        }
735        assert!(
736            cert.with_policy(&*POLICY, None)
737                .unwrap()
738                .features()
739                .unwrap()
740                .normalized_eq(&features)
741        );
742        cert
743    }

796    #[test]
797    fn assembly_preserves_compatible_feature_matrix_and_decryptability() {
798        use openpgp::parse::PacketParser;
799        use openpgp::parse::PacketParserResult;
800        use openpgp::parse::Parse;
801        use std::io::Cursor;
802        use std::io::Read;
803
804        let legacy = Features::empty().set_seipdv1();
805        let modern = Features::empty().set_seipdv1().set_seipdv2();
806        for (features, second_recipient_features, pkesk_version, seip_version) in [
807            (legacy.clone(), legacy.clone(), 3, 1),
808            (legacy, modern.clone(), 3, 1),
809            (modern.clone(), modern, 6, 2),
810        ] {
811            for with_userid in [false, true] {
812                let primary = feature_cert(features.clone(), with_userid, false);
813                let mut certifier = CountingSigner {
814                    inner: signer(&primary),
815                    calls: 0,
816                };
817                let primary = primary.strip_secret_key_material();
818                // Padding is not a capability difference.
819                let padded_features = Features::new([
820                    if second_recipient_features.supports_seipdv2() {
821                        0x09
822                    } else {
823                        0x01
824                    },
825                    0,
826                ]);
827                assert_ne!(second_recipient_features, padded_features);
828                assert!(second_recipient_features.normalized_eq(&padded_features));
829                let recipients = [
830                    feature_cert(features.clone(), false, true),
831                    feature_cert(padded_features, true, true),
832                ];
833                let public_recipients: Vec<_> = recipients
834                    .iter()
835                    .cloned()
836                    .map(openpgp::Cert::strip_secret_key_material)
837                    .collect();
838                let selections: Vec<_> = public_recipients
839                    .iter()
840                    .map(|cert| EncryptionRecipient { cert, subkey: None })
841                    .collect();
842                let assembled = assemble_public_cert(
843                    &primary,
844                    &selections,
845                    &BTreeMap::new(),
846                    false,
847                    &mut certifier,
848                )
849                .unwrap();
850                assert_eq!(certifier.calls, recipients.len());
851                assert_eq!(assembled.cert.fingerprint(), primary.fingerprint());
852                assert_eq!(
853                    assembled
854                        .cert
855                        .userids()
856                        .map(|uid| uid.userid())
857                        .collect::<Vec<_>>(),
858                    primary
859                        .userids()
860                        .map(|uid| uid.userid())
861                        .collect::<Vec<_>>()
862                );
863                assert_eq!(assembled.cert.clone().retain_subkeys(|_| false), primary);
864                let plaintext = b"backup decryptable by each original recipient";
865                let ciphertext = super::super::encrypt_armored(plaintext, &assembled).unwrap();
866                let mut parser = PacketParser::from_bytes(ciphertext.as_bytes()).unwrap();
867                let mut pkesks = 0;
868                let mut seips = 0;
869                while let PacketParserResult::Some(packet) = parser {
870                    let (packet, next) = packet.recurse().unwrap();
871                    match packet {
872                        openpgp::Packet::PKESK(pkesk) => {
873                            assert_eq!(pkesk.version(), pkesk_version);
874                            pkesks += 1;
875                        }
876                        openpgp::Packet::SEIP(seip) => {
877                            assert_eq!(seip.version(), seip_version);
878                            seips += 1;
879                        }
880                        _ => {}
881                    }
882                    parser = next;
883                }
884                assert_eq!(pkesks, recipients.len());
885                assert_eq!(seips, 1);
886                for recipient in &recipients {
887                    let mut secret = Vec::new();
888                    recipient.as_tsk().serialize(&mut secret).unwrap();
889                    let mut decrypted = Vec::new();
890                    super::super::decrypt_with_secret_key(
891                        Cursor::new(ciphertext.clone().into_bytes()),
892                        &secret,
893                    )
894                    .unwrap()
895                    .read_to_end(&mut decrypted)
896                    .unwrap();
897                    assert_eq!(decrypted, plaintext);
898                }
899            }
900        }
901    }
