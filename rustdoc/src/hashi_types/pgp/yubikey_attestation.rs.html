4//! Narrow OpenPGP attestation verification against four fixed Yubico issuer pins.
5//!
6//! This checks direct signatures, not RFC 5280 certification paths. The pins are
7//! the OpenPGP CA from <https://developers.yubico.com/PKI/yubico-opgp-ca-1.pem>
8//! and OPGP A 1, B 1, and B2 1 from
9//! <https://developers.yubico.com/PKI/yubico-intermediate.pem>, matched byte-for-byte
10//! against Yubico's published certificates on 2026-09-08. Intermediate pins are
11//! trusted directly; their ancestors are not additional trust anchors.
12//!
13//! X.509 validity dates, revocation, touch policy, and freshness are not enforced.
14//! OpenPGP keys must be usable under the existing OpenPGP policy. The sole
15//! signing key must be the certificate's primary key, binding its identity to
16//! SIG attestation; the sole transport-encryption key is attested separately.
17//! Attestation fingerprint metadata is deliberately ignored: the device
18//! administrator can overwrite it. See <https://developers.yubico.com/PGP/Attestation.html>.
19//!
20//! Statements support NIST P-256 uncompressed SEC1 points with named-curve SPKI
21//! parameters, or RFC 8410 Ed25519/X25519 raw 32-byte keys with absent parameters.
22//! The complete public key must match; legacy, lossy Curve25519 encodings are
23//! rejected.

41const SIG_COMMON_NAME: &str = "YubiKey OPGP Attestation SIG";
42const DEC_COMMON_NAME: &str = "YubiKey OPGP Attestation DEC";
43const YUBICO_KEY_SOURCE_OID: Oid<'static> = oid!(1.3.6.1.4.1.41482.5.2);
44const ED25519_OID: Oid<'static> = oid!(1.3.101.112);
45const X25519_OID: Oid<'static> = oid!(1.3.101.110);
46const EC_PUBLIC_KEY_OID: Oid<'static> = oid!(1.2.840.10045.2.1);
47const NIST_P256_OID: Oid<'static> = oid!(1.2.840.10045.3.1.7);
48static TRUSTED_ISSUERS: LazyLock<[Vec<u8>; 4]> = LazyLock::new(|| {
49    let mut input = include_bytes!("yubikey_attestation/yubico-openpgp-issuers.pem").as_slice();
50    let issuers = std::array::from_fn(|_| {
51        let (remainder, der) =
52            parse_certificate_pem(input).expect("embedded Yubico issuer PEM must be valid");
53        parse_certificate_der(&der, "embedded issuer")
54            .expect("embedded Yubico issuer DER must be valid");
55        input = remainder;
56        der
57    });
58    assert!(
59        input.trim_ascii().is_empty(),
60        "unexpected embedded issuer data"
61    );
62    issuers
63});

65/// Verify that `cert`'s sole usable signing key is its primary key and that it
66/// and the sole usable transport-encryption key were generated on a YubiKey,
67/// using its device certificate and SIG/DEC statements.
68///
69/// Each input must contain exactly one PEM `CERTIFICATE` block, with no other
70/// non-whitespace data and exactly one DER certificate inside. The device must
71/// be signed directly by a pinned Yubico OpenPGP issuer, and both statements by
72/// that device. Supported OpenPGP keys are NIST P-256 ECDSA/ECDH and legacy
73/// Ed25519/Cv25519, with standard statement SPKIs as described above.
74/// This does not enforce X.509 dates, revocation, touch, or freshness, and does
75/// not establish current possession of either private key.
76pub fn verify_yubikey_attestations(
77    cert: &PgpPublicCert,
78    device_pem: &[u8],
79    sig_pem: &[u8],
80    dec_pem: &[u8],
81) -> Result<()> {
82    verify_yubikey_attestations_and_keys(cert, device_pem, sig_pem, dec_pem).map(|_| ())
83}

85pub(crate) fn verify_yubikey_attestations_and_keys(
86    cert: &PgpPublicCert,
87    device_pem: &[u8],
88    sig_pem: &[u8],
89    dec_pem: &[u8],
90) -> Result<AttestedPgpKeys> {
91    let device_der = parse_single_certificate_pem(device_pem).context("device attestation")?;
92    let sig_der = parse_single_certificate_pem(sig_pem).context("SIG attestation")?;
93    let dec_der = parse_single_certificate_pem(dec_pem).context("DEC attestation")?;
94    let issuers = TRUSTED_ISSUERS.each_ref().map(Vec::as_slice);
95    // Dev KPs have no YubiKey: their software device signs its own certificate.
96    #[cfg(feature = "non-enclave-dev")]
97    let issuers: Vec<&[u8]> = issuers.into_iter().chain([device_der.as_slice()]).collect();
98    verify_yubikey_attestations_with_issuers(cert, &device_der, &sig_der, &dec_der, &issuers)
99}

101pub(crate) fn verify_yubikey_attestations_with_issuers(
102    cert: &PgpPublicCert,
103    device_der: &[u8],
104    sig_der: &[u8],
105    dec_der: &[u8],
106    trusted_issuers: &[&[u8]],
107) -> Result<AttestedPgpKeys> {
108    let device = parse_certificate_der(device_der, "device attestation")?;
109    let sig = parse_certificate_der(sig_der, "SIG attestation")?;
110    let dec = parse_certificate_der(dec_der, "DEC attestation")?;
111
112    verify_device_issuer(&device, trusted_issuers)?;
113    verify_statement(&sig, &device, SIG_COMMON_NAME)?;
114    verify_statement(&dec, &device, DEC_COMMON_NAME)?;
115    let (signing, signing_curve, signing_bytes) = signing_key(&cert.cert)?;
116    let (encryption, encryption_curve, encryption_bytes) = encryption_key(&cert.cert)?;
117    anyhow::ensure!(
118        statement_key(&sig, &signing_curve)? == signing_bytes,
119        "SIG attestation key does not match the OpenPGP signing key"
120    );
121    anyhow::ensure!(
122        statement_key(&dec, &encryption_curve)? == encryption_bytes,
123        "DEC attestation key does not match the OpenPGP encryption key"
124    );
125    Ok(AttestedPgpKeys {
126        signing,
127        encryption,
128    })
129}

131fn parse_single_certificate_pem(input: &[u8]) -> Result<Vec<u8>> {
132    let (remainder, der) = parse_certificate_pem(input)?;
133    anyhow::ensure!(
134        remainder.trim_ascii().is_empty(),
135        "certificate PEM contains trailing data"
136    );
137    Ok(der)
138}

140fn parse_certificate_pem(input: &[u8]) -> Result<(&[u8], Vec<u8>)> {
141    let input = input.trim_ascii_start();
142    // The library skips preambles and does not check that the footer label
143    // matches. Check those boundaries without reimplementing PEM decoding.
144    anyhow::ensure!(
145        input
146            .split(|byte| *byte == b'\n')
147            .next()
148            .unwrap_or_default()
149            .trim_ascii_end()
150            == b"-----BEGIN CERTIFICATE-----",
151        "expected a CERTIFICATE PEM header"
152    );
153    let (remainder, pem) = parse_x509_pem(input)
154        .map_err(|error| anyhow::anyhow!("invalid certificate PEM: {error}"))?;
155    let consumed = &input[..input.len() - remainder.len()];
156    anyhow::ensure!(
157        consumed
158            .trim_ascii_end()
159            .rsplit(|byte| *byte == b'\n')
160            .next()
161            .unwrap_or_default()
162            == b"-----END CERTIFICATE-----",
163        "expected a CERTIFICATE PEM footer"
164    );
165    Ok((remainder, pem.contents))
166}

168fn parse_certificate_der<'a>(der: &'a [u8], description: &str) -> Result<X509Certificate<'a>> {
169    let (remainder, certificate) = parse_x509_certificate(der)
170        .map_err(|error| anyhow::anyhow!("invalid {description} certificate DER: {error}"))?;
171    anyhow::ensure!(
172        remainder.is_empty(),
173        "{description} certificate DER contains trailing data"
174    );
175    Ok(certificate)
176}

178fn verify_device_issuer(device: &X509Certificate<'_>, trusted_issuers: &[&[u8]]) -> Result<()> {
179    for issuer_der in trusted_issuers {
180        let issuer = parse_certificate_der(issuer_der, "trusted issuer")?;
181        if device.issuer() == issuer.subject()
182            && device.verify_signature(Some(issuer.public_key())).is_ok()
183        {
184            return Ok(());
185        }
186    }
187    anyhow::bail!("device attestation certificate is not signed by a trusted issuer")
188}

190fn verify_statement(
191    statement: &X509Certificate<'_>,
192    device: &X509Certificate<'_>,
193    expected_common_name: &str,
194) -> Result<()> {
195    anyhow::ensure!(
196        statement.issuer() == device.subject(),
197        "{expected_common_name} was not issued by the device certificate"
198    );
199    statement
200        .verify_signature(Some(device.public_key()))
201        .with_context(|| format!("{expected_common_name} signature is invalid"))?;
202
203    let mut common_names = statement.subject().iter_common_name();
204    let common_name = common_names
205        .next()
206        .context("attestation has no subject common name")?
207        .as_str()
208        .context("attestation subject common name is not a string")?;
209    anyhow::ensure!(
210        common_names.next().is_none() && common_name == expected_common_name,
211        "attestation must have exactly the subject common name {expected_common_name:?}"
212    );
213    let key_source = statement
214        .get_extension_unique(&YUBICO_KEY_SOURCE_OID)
215        .context("attestation has duplicate key-source extensions")?
216        .context("attestation has no key-source extension")?;
217    anyhow::ensure!(
218        key_source.value == [0x02, 0x01, 0x01],
219        "attested key was not generated on the YubiKey"
220    );
221    Ok(())
222}

224fn statement_key<'a>(statement: &'a X509Certificate<'_>, curve: &Curve) -> Result<&'a [u8]> {
225    let spki = statement.public_key();
226    let (algorithm, valid_parameters, key_len) = match curve {
227        Curve::Ed25519 => (&ED25519_OID, spki.algorithm.parameters.is_none(), 32),
228        Curve::Cv25519 => (&X25519_OID, spki.algorithm.parameters.is_none(), 32),
229        Curve::NistP256 => (
230            &EC_PUBLIC_KEY_OID,
231            spki.algorithm
232                .parameters
233                .as_ref()
234                .is_some_and(|parameters| {
235                    parameters.as_oid().is_ok_and(|oid| oid == NIST_P256_OID)
236                }),
237            65,
238        ),
239        _ => anyhow::bail!("attestation public key uses an unsupported curve"),
240    };
241    anyhow::ensure!(
242        spki.algorithm.algorithm == *algorithm && valid_parameters,
243        "attestation public key has an unsupported algorithm or parameters"
244    );
245    let key = &spki.subject_public_key;
246    anyhow::ensure!(
247        key.unused_bits == 0
248            && key.data.len() == key_len
249            && (*curve != Curve::NistP256 || key.data.first() == Some(&0x04)),
250        "attestation public key has an unsupported encoding"
251    );
252    Ok(key.data.as_ref())
253}

255fn signing_key(cert: &openpgp::Cert) -> Result<(Fingerprint, Curve, &[u8])> {
256    let mut candidates = usable_keys(cert).for_signing();
257    let candidate = candidates
258        .next()
259        .context("OpenPGP certificate has no usable signing key")?;
260    anyhow::ensure!(
261        candidates.next().is_none(),
262        "OpenPGP certificate has multiple usable signing keys"
263    );
264    anyhow::ensure!(
265        candidate.key().fingerprint() == cert.fingerprint(),
266        "OpenPGP signing key must be the certificate's primary key"
267    );
268    match candidate.key().mpis() {
269        mpi::PublicKey::EdDSA {
270            curve: Curve::Ed25519,
271            q,
272        } => q
273            .decode_point(&Curve::Ed25519)
274            .map(|(key, _)| (candidate.key().fingerprint(), Curve::Ed25519, key))
275            .context("OpenPGP signing key has invalid Ed25519 encoding"),
276        mpi::PublicKey::ECDSA {
277            curve: Curve::NistP256,
278            q,
279        } => {
280            q.decode_point(&Curve::NistP256)
281                .context("OpenPGP signing key has invalid NIST P-256 encoding")?;
282            Ok((candidate.key().fingerprint(), Curve::NistP256, q.value()))
283        }
284        _ => anyhow::bail!("OpenPGP signing key does not use Ed25519 or NIST P-256 ECDSA"),
285    }
286}

288fn encryption_key(cert: &openpgp::Cert) -> Result<(Fingerprint, Curve, &[u8])> {
289    let mut candidates = usable_keys(cert).for_transport_encryption();
290    let candidate = candidates
291        .next()
292        .context("OpenPGP certificate has no usable encryption key")?;
293    anyhow::ensure!(
294        candidates.next().is_none(),
295        "OpenPGP certificate has multiple usable encryption keys"
296    );
297    match candidate.key().mpis() {
298        mpi::PublicKey::ECDH {
299            curve: Curve::Cv25519,
300            q,
301            ..
302        } => q
303            .decode_point(&Curve::Cv25519)
304            .map(|(key, _)| (candidate.key().fingerprint(), Curve::Cv25519, key))
305            .context("OpenPGP encryption key has invalid Cv25519 encoding"),
306        mpi::PublicKey::ECDH {
307            curve: Curve::NistP256,
308            q,
309            ..
310        } => {
311            q.decode_point(&Curve::NistP256)
312                .context("OpenPGP encryption key has invalid NIST P-256 encoding")?;
313            Ok((candidate.key().fingerprint(), Curve::NistP256, q.value()))
314        }
315        _ => anyhow::bail!("OpenPGP encryption key does not use Cv25519 or NIST P-256 ECDH"),
316    }
