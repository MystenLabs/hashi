19#[derive(Clone, Debug, Default)]
20pub struct RotationMessages {
21    pairs: Vec<(u32, Bcs)>,
22    held: Vec<u8>,
23    held_count: usize,
24}
25
26#[derive(Clone, PartialEq, Message)]
27struct Entry {
28    #[prost(uint32, tag = "1")]
29    key: u32,
30    #[prost(message, optional, tag = "2")]
31    value: Option<Bcs>,
32}
33
34impl RotationMessages {
35    pub fn len(&self) -> usize {
36        self.pairs.len() + self.held_count
37    }
38
39    pub fn is_empty(&self) -> bool {
40        self.len() == 0
41    }
42
43    pub fn iter(&self) -> impl Iterator<Item = Result<(u32, Bcs), DecodeError>> + '_ {
44        self.pairs
45            .iter()
46            .cloned()
47            .map(Ok)
48            .chain(self.held_entries().map(parse))
49    }
50
51    pub fn into_entries(self) -> impl Iterator<Item = Result<(u32, Bcs), DecodeError>> {
52        let held = Bytes::from(self.held);
53        let mut position = 0;
54        self.pairs
55            .into_iter()
56            .map(Ok)
57            .chain(std::iter::from_fn(move || {
58                next_entry(&held, &mut position).map(|range| parse(held.slice(range)))
59            }))
60    }
61
62    fn held_entries(&self) -> impl Iterator<Item = &[u8]> + '_ {
63        let mut position = 0;
64        std::iter::from_fn(move || {
65            next_entry(&self.held, &mut position).map(|range| &self.held[range])
66        })
67    }
68}
69
70fn parse(entry: impl Buf) -> Result<(u32, Bcs), DecodeError> {
71    Entry::decode(entry).map(|Entry { key, value }| (key, value.unwrap_or_default()))
72}
73
74fn next_entry(held: &[u8], position: &mut usize) -> Option<Range<usize>> {
75    let mut rest = held.get(*position..)?;
76    let len = usize::try_from(decode_varint(&mut rest).ok()?).ok()?;
77    let start = held.len() - rest.len();
78    let end = start.checked_add(len).filter(|end| *end <= held.len())?;
79    *position = end;
80    Some(start..end)
81}

83impl FromIterator<(u32, Bcs)> for RotationMessages {
84    fn from_iter<I: IntoIterator<Item = (u32, Bcs)>>(pairs: I) -> Self {
85        Self {
86            pairs: pairs.into_iter().collect(),
87            ..Self::default()
88        }
89    }
90}

92impl PartialEq for RotationMessages {
93    fn eq(&self, other: &Self) -> bool {
94        let sorted = |messages: &Self| {
95            let mut entries = messages.iter().collect::<Vec<_>>();
96            entries.sort_by_key(|entry| entry.as_ref().map(|(key, _)| *key).ok());
97            entries
98        };
99        sorted(self) == sorted(other)
100    }
101}
102
103impl Message for RotationMessages {
104    fn encode_raw(&self, buf: &mut impl BufMut) {
105        for (key, value) in &self.pairs {
106            encode_key(1, WireType::LengthDelimited, buf);
107            encode_varint(pair_len(*key, value) as u64, buf);
108            if *key != 0 {
109                prost::encoding::uint32::encode(1, key, buf);
110            }
111            if *value != Bcs::default() {
112                prost::encoding::message::encode(2, value, buf);
113            }
114        }
115        for entry in self.held_entries() {
116            encode_key(1, WireType::LengthDelimited, buf);
117            encode_varint(entry.len() as u64, buf);
118            buf.put_slice(entry);
119        }
120    }
121
122    fn merge_field(
123        &mut self,
124        tag: u32,
125        wire_type: WireType,
126        buf: &mut impl Buf,
127        ctx: DecodeContext,
128    ) -> Result<(), DecodeError> {
129        if tag != 1 {
130            return prost::encoding::skip_field(wire_type, tag, buf, ctx);
131        }
132        let mut entry = Bytes::new();
133        prost::encoding::bytes::merge(WireType::LengthDelimited, &mut entry, buf, ctx)?;
134        encode_varint(entry.len() as u64, &mut self.held);
135        self.held.extend_from_slice(&entry);
136        self.held_count += 1;
137        Ok(())
138    }
139
140    fn encoded_len(&self) -> usize {
141        self.pairs
142            .iter()
143            .map(|(key, value)| {
144                let len = pair_len(*key, value);
145                key_len(1) + encoded_len_varint(len as u64) + len
146            })
147            .sum::<usize>()
148            + self.held_count * key_len(1)
149            + self.held.len()
150    }
151
152    fn clear(&mut self) {
153        *self = Self::default();
154    }
155}
156
157fn pair_len(key: u32, value: &Bcs) -> usize {
158    let key_bytes = if key == 0 {
159        0
160    } else {
161        prost::encoding::uint32::encoded_len(1, &key)
162    };
163    let value_bytes = if *value == Bcs::default() {
164        0
165    } else {
166        prost::encoding::message::encoded_len(2, value)
167    };
168    key_bytes + value_bytes
169}

184    fn bcs(name: Option<String>, value: Option<Vec<u8>>) -> Bcs {
185        let mut bcs = Bcs::default();
186        bcs.name = name;
187        bcs.value = value.map(Bytes::from);
188        bcs
189    }

191    fn any_bcs() -> impl Strategy<Value = Bcs> {
192        (
193            proptest::option::of("[a-z]{0,3}"),
194            proptest::option::of(prop::collection::vec(any::<u8>(), 0..4)),
195        )
196            .prop_map(|(name, value)| bcs(name, value))
197    }
198
199    fn any_field() -> impl Strategy<Value = Vec<u8>> {
200        let entry = (0u32..4, proptest::option::of(any_bcs()))
201            .prop_map(|(key, value)| Entry { key, value }.encode_to_vec());
202        (
203            prop_oneof![3 => Just(1u64), 1 => 0u64..4],
204            prop_oneof![3 => Just(2u64), 1 => 0u64..6],
205            prop_oneof![entry, prop::collection::vec(any::<u8>(), 0..6)],
206        )
207            .prop_map(|(tag, wire_type, content)| {
208                let mut field = Vec::new();
209                encode_varint(tag << 3 | wire_type, &mut field);
210                encode_varint(content.len() as u64, &mut field);
211                field.extend(content);
212                field
213            })
214    }

216    #[test]
217    fn a_held_map_grows_with_its_entries() {
218        let mut wire = Vec::new();
219        let mut key = 0u32;
220        while wire.len() < 32 << 20 {
221            key += 1;
222            encode_key(1, WireType::LengthDelimited, &mut wire);
223            encode_varint(
224                (key_len(1) + encoded_len_varint(key.into())) as u64,
225                &mut wire,
226            );
227            prost::encoding::uint32::encode(1, &key, &mut wire);
228        }
229        let wire_len = wire.len();
230        let held = RotationMessages::decode(Bytes::from(wire)).unwrap();
231        assert_eq!(held.held_count, key as usize);
232        assert!(held.held.capacity() <= 2 * wire_len);
233
234        let mut wire = [(1, bcs(None, Some(vec![1; 16])))]
235            .into_iter()
236            .collect::<RotationMessages>()
237            .encode_to_vec();
238        let entry_len = wire.len();
239        prost::encoding::bytes::encode(2, &vec![0u8; 8 << 20], &mut wire);
240        let held = RotationMessages::decode(Bytes::from(wire)).unwrap();
241        assert_eq!(held.held_count, 1);
242        assert!(held.held.capacity() <= 2 * entry_len);
243        let buffer = held.held.as_ptr_range();
244        let (_, value) = held.into_entries().next().unwrap().unwrap();
245        assert!(buffer.contains(&value.value.unwrap().as_ptr()));
246    }

248    #[proptest]
249    fn a_held_map_reads_what_the_generated_map_decodes(
250        #[strategy(prop::collection::vec(any_field(), 0..6))] fields: Vec<Vec<u8>>,
251        #[strategy(prop::collection::vec(any::<u8>(), 0..3))] tail: Vec<u8>,
252    ) {
253        let wire = [fields.concat(), tail].concat();
254        let held = RotationMessages::decode(wire.as_slice());
255        if let Ok(held) = &held {
256            prop_assert_eq!(
257                held.iter().collect::<Vec<_>>(),
258                held.clone().into_entries().collect::<Vec<_>>()
259            );
260        }
261        if let Ok(generated) = GeneratedRotationMessages::decode(wire.as_slice()) {
262            let entries = held.unwrap().iter().collect::<Result<Vec<_>, _>>().unwrap();
263            prop_assert_eq!(
264                entries.into_iter().collect::<HashMap<_, _>>(),
265                generated.messages
266            );
267        }
268    }

270    #[proptest]
271    fn both_forms_encode_what_the_generated_map_decodes(
272        #[strategy(prop::collection::hash_map(prop_oneof![0u32..3, any::<u32>()], any_bcs(), 0..6))]
273        pairs: HashMap<u32, Bcs>,
274    ) {
275        let sent = pairs.clone().into_iter().collect::<RotationMessages>();
276        let wire = sent.encode_to_vec();
277        prop_assert_eq!(sent.encoded_len(), wire.len());
278        prop_assert_eq!(
279            &GeneratedRotationMessages::decode(wire.as_slice())
280                .unwrap()
281                .messages,
282            &pairs
283        );
284
285        let generated = GeneratedRotationMessages { messages: pairs }.encode_to_vec();
286        let held = RotationMessages::decode(generated.as_slice()).unwrap();
287        prop_assert_eq!(&held, &sent);
288        prop_assert_eq!(held.encoded_len(), generated.len());
289        prop_assert_eq!(held.encode_to_vec(), generated);
290    }
