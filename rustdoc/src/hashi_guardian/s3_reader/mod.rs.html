37/// Verified reader over the guardian's S3 logs.
38///
39/// Reads accept any allowlisted build unless the method explicitly requires
40/// the current build. Reuse one reader so repeated reads can share cached
41/// session attestations and signing keys. Every writing session must match the
42/// expected bucket, region, retention environment, and Bitcoin network.
43pub struct GuardianReader {
44    s3: GuardianS3Client,
45    expected_deployment: DeploymentConfig,
46    sessions: HashMap<SessionID, VerifiedSessionInfo>,
47}
48
49impl GuardianReader {
50    /// Create a reader after checking S3 connectivity and object-lock support.
51    pub async fn new(
52        expected_deployment: DeploymentConfig,
53        credentials: S3Credentials,
54    ) -> GuardianResult<Self> {
55        let s3 = GuardianS3Client::new(
56            &expected_deployment.bucket_info,
57            expected_deployment.retention_environment,
58            &credentials,
59        )
60        .await?;
61        Ok(Self::from_s3_client(s3, expected_deployment))
62    }
63
64    /// Reuse the enclave's S3 client, constructed from the same deployment
65    /// configuration, without another connectivity check.
66    pub(crate) fn from_s3_client(
67        s3: GuardianS3Client,
68        expected_deployment: DeploymentConfig,
69    ) -> Self {
70        Self {
71            s3,
72            expected_deployment,
73            sessions: HashMap::new(),
74        }
75    }

77    /// Load and verify a session's attestation and completed operator initialization on first use.
78    async fn ensure_session_info_loaded(&mut self, session_id: &str) -> GuardianResult<()> {
79        if !self.sessions.contains_key(session_id) {
80            let session_info =
81                VerifiedSessionInfo::read_from_s3(&self.s3, session_id, &self.expected_deployment)
82                    .await?;
83            self.sessions.insert(session_id.into(), session_info);
84        }
85        Ok(())
86    }
87
88    /// Verify a record and the initialization checkpoint required to emit it.
89    async fn verify_record(
90        &mut self,
91        record: VerifiableLogEntry,
92    ) -> GuardianResult<VerifiedLogEntry> {
93        self.ensure_session_info_loaded(record.session_id()).await?;
94        let session_info = self
95            .sessions
96            .get_mut(record.session_id())
97            .expect("session info was loaded above");
98        session_info.verify_record(&self.s3, record).await
99    }
100
101    /// Read an immutable S3 record and verify it against its writing session.
102    async fn read_verified_record(&mut self, key: &str) -> GuardianResult<VerifiedLogEntry> {
103        let record = self.s3.get_log_record(key).await?;
104        self.verify_record(record).await
105    }

107    /// Read and verify every immutable record in an hour-scoped directory.
108    ///
109    /// Each result retains its writing session's attested build PCRs because a
110    /// directory may contain records from more than one build.
111    pub async fn read_logs_in_dir(
112        &mut self,
113        dir: &S3HourDirectory,
114    ) -> GuardianResult<Vec<VerifiedLogEntry>> {
115        let all_logs = self.s3.list_all_log_records_in_dir(dir).await?;
116
117        let mut out = Vec::with_capacity(all_logs.len());
118        for record in all_logs {
119            let verified_record = self.verify_record(record).await?;
120            out.push(verified_record);
121        }
122        Ok(out)
123    }

125    /// Return verified session info after requiring the attested PCRs to match
126    /// the current build.
127    pub async fn get_current_session_info(
128        &mut self,
129        session_id: &str,
130    ) -> GuardianResult<VerifiedSessionInfo> {
131        self.ensure_session_info_loaded(session_id).await?;
132        let session_info = self
133            .sessions
134            .get(session_id)
135            .expect("session info was loaded above");
136        self.expected_deployment
137            .pcr_allowlist
138            .require_current_build(session_info.build_pcrs())?;
139        Ok(session_info.clone())
140    }

142    /// Read and verify the latest ceremony, with the session that wrote it,
143    /// or return `None` if none exists.
144    ///
145    /// Ceremony keys begin with a zero-padded `sharing_seq`, so the
146    /// lexicographically greatest key identifies the latest ceremony.
147    async fn read_latest_ceremony_log(
148        &mut self,
149        require_current: bool,
150    ) -> GuardianResult<Option<(CeremonyLogMessage, SessionID)>> {
151        let keys = self
152            .s3
153            .list_keys(&CeremonyLogMessage::object_key_dir())
154            .await?;
155        let Some(key) = keys.into_iter().max() else {
156            return Ok(None);
157        };
158        let verified_record = self.read_verified_record(&key).await?;
159        if require_current {
160            self.expected_deployment
161                .pcr_allowlist
162                .require_current_build(verified_record.build_pcrs())?;
163        }
164        let session_id = verified_record.entry().session_id().clone();
165        let msg = verified_record
166            .into_entry()
167            .into_message()
168            .into_ceremony()
169            .ok_or_else(|| InvalidS3Log(format!("expected a ceremony log at {key}")))?;
170        log_verified_read(&key, &session_id);
171        Ok(Some((*msg, session_id)))
172    }

174    /// Choose a sequence above every completed ceremony and occupied share directory.
175    /// Shares are published before the ceremony commit, so an interrupted attempt
176    /// can occupy a sequence even though no ceremony record exists for it.
177    ///
178    /// This allocates under the single ceremony-writer assumption; it does not
179    /// reserve the sequence. Conditional writes still reject competing records.
180    pub(crate) async fn next_sharing_seq(&mut self) -> GuardianResult<u64> {
181        let mut highest = self
182            .read_latest_ceremony_log(false)
183            .await?
184            .map(|(ceremony, _)| ceremony.sharing_seq());
185        // Count occupied directories even if their records are unreadable or
186        // delete-marked: an interrupted ceremony may have left shares here.
187        let shares_dir = KpShareStateLogMessage::root_dir();
188        let proposals_dir = CeremonyProposalLogMessage::object_key_dir();
189        for directory in self.s3.list_common_prefixes(&shares_dir).await? {
190            if directory == proposals_dir {
191                continue;
192            }
193            let seq = KpShareStateLogMessage::sharing_seq_from_dir(&directory)
194                .map_err(|err| InvalidS3Log(err.to_string()))?;
195            highest = Some(highest.map_or(seq, |previous| previous.max(seq)));
196        }
197        match highest {
198            None => Ok(0),
199            Some(seq) => seq
200                .checked_add(1)
201                .ok_or_else(|| InvalidS3Log("sharing_seq exhausted".into())),
202        }
203    }

205    /// Read and verify the latest encrypted KP-share state for `sharing_seq`.
206    ///
207    /// Keys begin with a zero-padded `cert_seq`, so the lexicographically
208    /// greatest key identifies the latest state. KP-share locks are expected to
209    /// expire, so this read authenticates the selected record without claiming
210    /// S3 immutability.
211    async fn read_latest_kp_share_state_log(
212        &mut self,
213        sharing_seq: u64,
214        require_current: bool,
215    ) -> GuardianResult<Option<KpShareStateLogMessage>> {
216        let prefix = KpShareStateLogMessage::object_key_dir(sharing_seq);
217        let keys = self.s3.list_keys_allowing_mutations(&prefix).await?;
218        let Some(key) = keys.into_iter().max() else {
219            return Ok(None);
220        };
221        let msg = self
222            .read_kp_share_state_log_at_key(&key, require_current)
223            .await?;
224        if msg.sharing_seq != sharing_seq {
225            return Err(InvalidS3Log(format!(
226                "sharing_seq mismatch: {} != {}",
227                msg.sharing_seq, sharing_seq
228            )));
229        }
230        Ok(Some(msg))
231    }

233    /// Read and verify an exact encrypted KP-share state written by the current
234    /// build.
235    ///
236    /// Read the requested sequence even if a later request has already advanced
237    /// the latest state.
238    pub async fn read_kp_share_state_log_from_current_build(
239        &mut self,
240        sharing_seq: u64,
241        cert_seq: u64,
242    ) -> GuardianResult<KpShareStateLogMessage> {
243        let key = KpShareStateLogMessage::object_key_for_sequences(sharing_seq, cert_seq);
244        self.read_kp_share_state_log_at_key(&key, true).await
245    }

247    /// Read and verify the proposal written by one live ceremony session.
248    pub async fn read_live_ceremony_proposal(
249        &mut self,
250        session_id: &SessionID,
251    ) -> GuardianResult<CeremonyState> {
252        let key = CeremonyProposalLogMessage::object_key(session_id);
253        // A live proposal has just been published, so its short-lived Compliance
254        // lock must still be active.
255        let verified_record = self.read_verified_record(&key).await?;
256        self.expected_deployment
257            .pcr_allowlist
258            .require_current_build(verified_record.build_pcrs())?;
259        let writing_session_id = verified_record.entry().session_id().clone();
260        let proposal = *verified_record
261            .into_entry()
262            .into_message()
263            .into_ceremony_proposal()
264            .ok_or_else(|| InvalidS3Log(format!("expected a ceremony proposal log at {key}")))?;
265        let state = CeremonyState::from_proposal(proposal).map_err(|error| {
266            InvalidS3Log(format!("invalid ceremony proposal at {key}: {error}"))
267        })?;
268        log_verified_read(&key, &writing_session_id);
269        Ok(state)
270    }

272    /// Read and verify one KP-share object under the requested build policy.
273    async fn read_kp_share_state_log_at_key(
274        &mut self,
275        key: &str,
276        require_current: bool,
277    ) -> GuardianResult<KpShareStateLogMessage> {
278        // KP-share locks are expected to expire, so authenticate the record
279        // without claiming that S3 still makes it immutable.
280        let record = self
281            .s3
282            .get_log_record_inner(key, ImmutabilityCheck::Skipped)
283            .await?;
284        let verified_record = self.verify_record(record).await?;
285        if require_current {
286            self.expected_deployment
287                .pcr_allowlist
288                .require_current_build(verified_record.build_pcrs())?;
289        }
290        let session_id = verified_record.entry().session_id().clone();
291        let msg = *verified_record
292            .into_entry()
293            .into_message()
294            .into_kp_share_state()
295            .ok_or_else(|| InvalidS3Log(format!("expected a kp-shares log at {key}")))?;
296        log_verified_read(key, &session_id);
297        Ok(msg)
298    }

327    /// Once a ceremony is present, its matching KP-share state must also exist
328    /// because writers publish `kp-shares/` before `ceremony/`.
329    async fn read_latest_ceremony_state_with_build_requirement(
330        &mut self,
331        require_current: bool,
332    ) -> GuardianResult<(CeremonyState, SessionID)> {
333        let (ceremony, dealer) = self
334            .read_latest_ceremony_log(require_current)
335            .await?
336            .ok_or_else(|| {
337                InvalidInputs("no ceremony log found; setup_new_key has not run".into())
338            })?;
339        let sharing_seq = ceremony.sharing_seq();
340        let kp_share_state = self
341            .read_latest_kp_share_state_log(sharing_seq, require_current)
342            .await?
343            .ok_or_else(|| {
344                InvalidS3Log(format!(
345                    "no kp-shares log found for latest ceremony sharing_seq {sharing_seq}"
346                ))
347            })?;
348        let state = CeremonyState::new(ceremony, kp_share_state)
349            .expect("ceremony and KP share state must have a consistent shape");
350        Ok((state, dealer))
351    }

353    /// Read the latest serving committee.
354    ///
355    /// Prefer the latest `committee-update/` record, then fall back
356    /// to the KP-authorized `genesis/record.json` bootstrap record. Return
357    /// `None` if neither source exists.
358    pub async fn read_latest_committee(&mut self) -> GuardianResult<Option<Committee>> {
359        if let Some(committee) = self.read_latest_committee_update().await? {
360            return Ok(Some(committee));
361        }
362        Ok(self.read_genesis().await?.map(|genesis| genesis.committee))
363    }

365    /// Read and verify the applied committee with the highest epoch, or return
366    /// `None` if no update exists.
367    ///
368    /// Keys begin with a zero-padded epoch, so the lexicographically
369    /// greatest key identifies the latest applied committee.
370    async fn read_latest_committee_update(&mut self) -> GuardianResult<Option<Committee>> {
371        let keys = self
372            .s3
373            .list_keys(&CommitteeUpdateLogMessage::object_key_dir())
374            .await?;
375        let Some(key) = keys.into_iter().max() else {
376            return Ok(None);
377        };
378        let verified_record = self.read_verified_record(&key).await?;
379        let session_id = verified_record.entry().session_id().clone();
380        let msg = verified_record
381            .into_entry()
382            .into_message()
383            .into_committee_update()
384            .ok_or_else(|| InvalidS3Log(format!("expected a committee-update log at {key}")))?;
385        log_verified_read(&key, &session_id);
386        Ok(Some(msg.new_committee))
387    }

389    /// Read and verify the fixed KP-authorized bootstrap record, or return
390    /// `None` if `genesis/record.json` has not been written.
391    pub async fn read_genesis(&mut self) -> GuardianResult<Option<Box<GenesisLogMessage>>> {
392        let key = GenesisLogMessage::object_key();
393        let keys = self
394            .s3
395            .list_keys(&GenesisLogMessage::object_key_dir())
396            .await?;
397        if keys.is_empty() {
398            return Ok(None);
399        }
400        if keys != [key.clone()] {
401            return Err(InvalidS3Log(format!(
402                "expected exactly one genesis record at {key}, found {keys:?}"
403            )));
404        }
405        let verified_record = self.read_verified_record(&key).await?;
406        let session_id = verified_record.entry().session_id().clone();
407        let genesis = verified_record
408            .into_entry()
409            .into_message()
410            .into_genesis()
411            .ok_or_else(|| InvalidS3Log(format!("expected a genesis log at {key}")))?;
412        log_verified_read(&key, &session_id);
413        Ok(Some(genesis))
414    }
415}

417fn log_verified_read(key: &str, session_id: &SessionID) {
418    info!("Successfully read {key} from session {session_id}.");
419}

421#[cfg(test)]
422pub(crate) fn reader_with_record_for_test(
423    record: Option<VerifiableLogEntry>,
424    signing_pubkey: hashi_types::guardian::GuardianPubKey,
425    extra_keys: Vec<String>,
426) -> GuardianReader {
427    use aws_sdk_s3::operation::get_object::GetObjectOutput;
428    use aws_sdk_s3::operation::list_object_versions::ListObjectVersionsOutput;
429    use aws_sdk_s3::primitives::ByteStream;
430    use aws_sdk_s3::primitives::DateTime;
431    use aws_sdk_s3::types::CommonPrefix;
432    use aws_sdk_s3::types::ObjectLockMode;
433    use aws_sdk_s3::types::ObjectVersion;
434    use aws_sdk_s3::Client;
435    use aws_smithy_mocks::mock;
436    use aws_smithy_mocks::mock_client;
437    use aws_smithy_mocks::RuleMode;
438    use hashi_types::guardian::InitConfig;
439    use hashi_types::guardian::S3ObjectLockPolicy;
440    use std::sync::Arc;
441    use std::sync::Mutex;
442
443    let mut keys = extra_keys;
444    if let Some(record) = &record {
445        keys.push(record.object_key().to_string());
446    }
447    let request = Arc::new(Mutex::new((String::new(), false)));
448    let captured_request = request.clone();
449    let list = mock!(Client::list_object_versions)
450        .match_requests(move |req| {
451            *captured_request.lock().unwrap() = (
452                req.prefix().unwrap_or_default().to_string(),
453                req.delimiter() == Some("/"),
454            );
455            true
456        })
457        .then_output(move || {
458            let (prefix, directories) = &*request.lock().unwrap();
459            if *directories {
460                let prefixes: std::collections::BTreeSet<_> = keys
461                    .iter()
462                    .filter_map(|key| {
463                        let rest = key.strip_prefix(prefix)?;
464                        let slash = rest.find('/')?;
465                        Some(format!("{prefix}{}", &rest[..=slash]))
466                    })
467                    .collect();
468                return ListObjectVersionsOutput::builder()
469                    .set_common_prefixes(Some(
470                        prefixes
471                            .into_iter()
472                            .map(|prefix| CommonPrefix::builder().prefix(prefix).build())
473                            .collect(),
474                    ))
475                    .build();
476            }
477            ListObjectVersionsOutput::builder()
478                .set_versions(Some(
479                    keys.iter()
480                        .filter(|key| key.starts_with(prefix.as_str()))
481                        .map(|key| ObjectVersion::builder().key(key).is_latest(true).build())
482                        .collect(),
483                ))
484                .build()
485        });
486    let config = InitConfig::mock_for_testing();
487    let policy = S3ObjectLockPolicy::for_environment(config.deployment().retention_environment);
488    let record_key = record
489        .as_ref()
490        .map(|record| record.object_key().to_string());
491    let get = mock!(Client::get_object)
492        .match_requests(move |req| req.key() == record_key.as_deref())
493        .then_output(move || {
494            let record = record.as_ref().unwrap();
495            GetObjectOutput::builder()
496                .object_lock_mode(ObjectLockMode::Compliance)
497                .object_lock_retain_until_date(DateTime::from(record.object_lock_expiry(policy)))
498                .body(ByteStream::from(serde_json::to_vec(record).unwrap()))
499                .build()
500        });
501    let client = mock_client!(aws_sdk_s3, RuleMode::MatchAny, &[&list, &get]);
502    let s3 = GuardianS3Client::from_client(
503        config.deployment().bucket_info.clone(),
504        config.deployment().retention_environment,
505        client,
506    );
507    let mut reader = GuardianReader::from_s3_client(s3, config.deployment().clone());
508    // Seed the attestation cache; the records still undergo normal signature,
509    // object-key, history, and lock verification.
510    reader.sessions.insert(
511        SessionID::from_signing_pubkey(&signing_pubkey),
512        VerifiedSessionInfo::new_for_test(
513            signing_pubkey,
514            config.deployment().pcr_allowlist.current_build().clone(),
515        ),
516    );
517    reader
518}
