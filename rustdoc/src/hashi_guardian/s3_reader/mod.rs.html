4//! Read and verify Guardian S3 logs.
5//!
6//! [`GuardianReader`] applies the S3 immutability policy for each log stream.
7//! It verifies each record with the signing key of the session that wrote it.
8//! It verifies the session attestation and the required initialization logs.
9//! It keeps verified session information in a cache for subsequent reads.

39/// A reader that verifies Guardian S3 logs.
40///
41/// Reads accept any build in the allowlist unless the method requires the current build.
42/// Reuse one reader to use the verified session information in its cache.
43/// Each session must report the expected bucket, region, retention environment, and Bitcoin network.
44pub struct GuardianReader {
45    s3: GuardianS3Client,
46    expected_deployment: DeploymentConfig,
47    sessions: HashMap<SessionID, VerifiedSessionInfo>,
48}
49

55    /// Create a reader from an existing S3 client, including the enclave's configured client.
56    /// The client must use the same deployment configuration.
57    /// This method does not repeat the S3 connection and Object Lock checks.
58    pub(crate) fn from_s3_client(
59        s3: GuardianS3Client,
60        expected_deployment: DeploymentConfig,
61    ) -> Self {
62        Self {
63            s3,
64            expected_deployment,
65            sessions: HashMap::new(),
66        }
67    }

69    /// Create a reader outside the enclave, using standard networking.
70    /// Check the S3 connection and Object Lock support before returning the reader.
71    /// Inside the enclave, use `from_s3_client()` with the configured S3 client.
72    pub async fn new(
73        expected_deployment: DeploymentConfig,
74        credentials: S3Credentials,
75    ) -> GuardianResult<Self> {
76        let s3 = GuardianS3Client::new(
77            &expected_deployment.bucket_info,
78            expected_deployment.retention_environment,
79            &credentials,
80        )
81        .await?;
82        Ok(Self::from_s3_client(s3, expected_deployment))
83    }

89    /// Read and verify the session attestation and operator initialization logs on first use.
90    async fn ensure_session_info_loaded(&mut self, session_id: &str) -> GuardianResult<()> {
91        if !self.sessions.contains_key(session_id) {
92            let session_info =
93                VerifiedSessionInfo::read_from_s3(&self.s3, session_id, &self.expected_deployment)
94                    .await?;
95            self.sessions.insert(session_id.into(), session_info);
96        }
97        Ok(())
98    }
99
100    /// Return verified session information and require the current build.
101    pub async fn get_current_session_info(
102        &mut self,
103        session_id: &str,
104    ) -> GuardianResult<VerifiedSessionInfo> {
105        self.ensure_session_info_loaded(session_id).await?;
106        let session_info = self
107            .sessions
108            .get(session_id)
109            .expect("session info was loaded above");
110        self.expected_deployment
111            .pcr_allowlist
112            .require_current_build(session_info.build_pcrs())?;
113        Ok(session_info.clone())
114    }

120    async fn verify_record(
121        &mut self,
122        log: SignedLogEntry,
123        require_current: bool,
124    ) -> GuardianResult<VerifiedLogEntry> {
125        self.ensure_session_info_loaded(log.session_id()).await?;
126        let session_info = self
127            .sessions
128            .get_mut(log.session_id())
129            .expect("session info was loaded above");
130        let verified_log = session_info.verify_record(&self.s3, log).await?;
131        if require_current {
132            self.expected_deployment
133                .pcr_allowlist
134                .require_current_build(verified_log.build_pcrs())?;
135        }
136        Ok(verified_log)
137    }
138
139    async fn read_log(
140        &mut self,
141        key: &str,
142        require_current: bool,
143    ) -> GuardianResult<VerifiedLogEntry> {
144        let log = self.s3.get_log_record(key).await?;
145        let verified_log = self.verify_record(log, require_current).await?;
146        info!(
147            key,
148            session_id = %verified_log.session_id(),
149            "Read and verified the S3 log record"
150        );
151        Ok(verified_log)
152    }

154    /// Read and verify each immutable record in a directory for one hour.
155    ///
156    /// A directory can contain records from more than one build.
157    /// Each result includes the build PCR values verified by the session attestation.
158    pub async fn read_logs_in_dir(
159        &mut self,
160        dir: &S3HourDirectory,
161    ) -> GuardianResult<Vec<VerifiedLogEntry>> {
162        let all_logs = self.s3.list_all_log_records_in_dir(dir).await?;
163
164        let mut out = Vec::with_capacity(all_logs.len());
165        for log in all_logs {
166            let verified_log = self.verify_record(log, false).await?;
167            out.push(verified_log);
168        }
169        info!(
170            directory = %dir,
171            record_count = out.len(),
172            "Read and verified the S3 log directory"
173        );
174        Ok(out)
175    }

181    /// Verify the KP-share record without requiring an active S3 lock, because these locks can expire.
182    async fn read_kp_share_state_log_at_key(
183        &mut self,
184        key: &str,
185        require_current: bool,
186    ) -> GuardianResult<KpShareStateLogMessage> {
187        // KP-share locks are expected to expire, so authenticate the record
188        // without claiming that S3 still makes it immutable.
189        let log = self
190            .s3
191            .get_log_record_inner(key, ImmutabilityCheck::Skipped)
192            .await?;
193        let verified_log = self.verify_record(log, require_current).await?;
194        info!(
195            key,
196            session_id = %verified_log.session_id(),
197            "Read and verified the S3 log record"
198        );
199        let msg = verified_log.extract("kp-shares", VersionedLogMessage::into_kp_share_state)?;
200        Ok(*msg)
201    }

203    /// Read and verify the latest encrypted KP-share state for `sharing_seq`.
204    ///
205    /// Each key starts with a `cert_seq` value padded with leading zeros.
206    /// The last key in lexicographic order identifies the latest state.
207    /// KP-share locks can expire.
208    /// This method verifies the selected record but does not require S3 immutability.
209    async fn read_latest_kp_share_state_log(
210        &mut self,
211        sharing_seq: u64,
212        require_current: bool,
213    ) -> GuardianResult<Option<KpShareStateLogMessage>> {
214        let keys = self
215            .s3
216            .list_keys_allowing_mutations(&KpShareStateLogMessage::object_key_dir(sharing_seq))
217            .await?;
218        if keys.is_empty() {
219            info!(
220                sharing_seq,
221                "No KP-share state log found for the sharing sequence"
222            );
223            return Ok(None);
224        }
225        let key = KpShareStateLogMessage::latest_key(sharing_seq, keys)?
226            .expect("the key list is nonempty");
227        let msg = self
228            .read_kp_share_state_log_at_key(&key, require_current)
229            .await?;
230        Ok(Some(msg))
231    }

234    pub async fn read_kp_share_state_log_from_current_build(

238    ) -> GuardianResult<KpShareStateLogMessage> {
239        let key = KpShareStateLogMessage::object_key_for_sequences(sharing_seq, cert_seq);

247    /// Read and verify the latest ceremony log and return it with its session ID.
248    /// Return `None` if no ceremony log exists.
249    ///
250    /// Each ceremony key starts with a `sharing_seq` value padded with leading zeros.
251    /// The last key in lexicographic order identifies the latest ceremony.
252    async fn read_latest_ceremony_log(
253        &mut self,
254        require_current: bool,
255    ) -> GuardianResult<Option<(CeremonyLogMessage, SessionID)>> {
256        let keys = self
257            .s3
258            .list_keys(&CeremonyLogMessage::object_key_dir())
259            .await?;
260        if keys.is_empty() {
261            info!("No completed ceremony log found");
262            return Ok(None);
263        }
264        let key = CeremonyLogMessage::latest_key(keys)?.expect("the key list is nonempty");
265        let verified_log = self.read_log(&key, require_current).await?;
266        let session_id = verified_log.session_id().clone();
267        let msg = verified_log.extract("ceremony", VersionedLogMessage::into_ceremony)?;
268        Ok(Some((*msg, session_id)))
269    }

271    /// Read the latest ceremony and its KP-share state with the specified build policy.
272    /// The matching KP-share state must exist if the ceremony record exists.
273    /// Writers publish `kp-shares/` records before `ceremony/` records.
274    async fn read_latest_ceremony_state_with_build_requirement(
275        &mut self,
276        require_current: bool,
277    ) -> GuardianResult<(CeremonyState, SessionID)> {
278        let (ceremony, dealer) = self
279            .read_latest_ceremony_log(require_current)
280            .await?
281            .ok_or_else(|| {
282                InvalidInputs("no ceremony log found; setup_new_key has not run".into())
283            })?;
284        let sharing_seq = ceremony.sharing_seq();
285        let kp_share_state = self
286            .read_latest_kp_share_state_log(sharing_seq, require_current)
287            .await?
288            .ok_or_else(|| {
289                InvalidS3Log(format!(
290                    "no kp-shares log found for latest ceremony sharing_seq {sharing_seq}"
291                ))
292            })?;
293        let state = CeremonyState::new(ceremony, kp_share_state)
294            .expect("ceremony and KP share state must have a consistent shape");
295        info!(
296            sharing_seq,
297            dealer_session_id = %dealer,
298            require_current_build = require_current,
299            "Loaded the latest ceremony and its KP-share state"
300        );
301        Ok((state, dealer))
302    }

304    /// Read the latest ceremony and the latest KP-share state for its `sharing_seq`.

322    /// Require the current build and an active S3 Compliance lock for the proposal.
323    pub async fn read_live_ceremony_proposal(
324        &mut self,
325        session_id: &SessionID,
326    ) -> GuardianResult<CeremonyState> {
327        let key = CeremonyProposalLogMessage::object_key(session_id);
328        // A live proposal has just been published, so its short-lived Compliance
329        // lock must still be active.
330        let verified_log = self.read_log(&key, true).await?;
331        let proposal = verified_log.extract(
332            "ceremony proposal",
333            VersionedLogMessage::into_ceremony_proposal,
334        )?;
335        CeremonyState::from_proposal(*proposal)
336            .map_err(|error| InvalidS3Log(format!("invalid ceremony proposal at {key}: {error}")))
337    }

339    /// Return the next unused sharing sequence number.
340    /// It must exceed each sequence number used by a completed ceremony or an existing share directory.
341    /// Return zero if neither source contains a sequence number.
342    /// Note that writers publish shares before they write the ceremony record.
343    /// So an interrupted write can leave a share directory without a ceremony record.
344    ///
345    /// This method assumes that only one writer creates ceremonies.
346    /// It does not reserve the sequence number.
347    /// Conditional writes reject records that compete for the same key.
348    pub(crate) async fn next_sharing_seq(&mut self) -> GuardianResult<u64> {
349        let max_ceremony_sharing_seq = self
350            .read_latest_ceremony_log(false)
351            .await?
352            .map(|(ceremony, _)| ceremony.sharing_seq());
353        // Count occupied directories even if their records are unreadable or
354        // delete-marked: an interrupted ceremony may have left shares here.
355        // Directory names are not authenticated. An operator with S3 write access
356        // can increase the next sequence or exhaust it by adding a directory.
357        let shares_dir = KpShareStateLogMessage::root_dir();
358        let proposals_dir = CeremonyProposalLogMessage::object_key_dir();
359        let mut max_kp_share_sharing_seq: Option<u64> = None;
360        for directory in self.s3.list_common_prefixes(&shares_dir).await? {
361            if directory == proposals_dir {
362                continue;
363            }
364            let seq = KpShareStateLogMessage::sharing_seq_from_dir(&directory)
365                .map_err(|err| InvalidS3Log(err.to_string()))?;
366            max_kp_share_sharing_seq = max_kp_share_sharing_seq.max(Some(seq));
367        }
368        let highest = max_ceremony_sharing_seq.max(max_kp_share_sharing_seq);
369        let next_seq = match highest {
370            None => Ok(0),
371            Some(seq) => seq
372                .checked_add(1)
373                .ok_or_else(|| InvalidS3Log("sharing_seq exhausted".into())),
374        }?;
375        info!(
376            max_ceremony_sharing_seq = ?max_ceremony_sharing_seq,
377            max_kp_share_sharing_seq = ?max_kp_share_sharing_seq,
378            next_sharing_seq = next_seq,
379            "Selected the next ceremony sharing sequence"
380        );
381        Ok(next_seq)
382    }

388    /// Read and verify the applied committee with the highest epoch.
389    /// Return `None` if no committee update exists.
390    ///
391    /// Each key starts with an epoch value padded with leading zeros.
392    /// The last key in lexicographic order identifies the latest applied committee.
393    async fn read_latest_committee_update(&mut self) -> GuardianResult<Option<Committee>> {
394        let keys = self
395            .s3
396            .list_keys(&CommitteeUpdateLogMessage::object_key_dir())
397            .await?;
398        if keys.is_empty() {
399            return Ok(None);
400        }
401        let key = CommitteeUpdateLogMessage::latest_key(keys)?.expect("the key list is nonempty");
402        let verified_log = self.read_log(&key, false).await?;
403        let msg = verified_log.extract(
404            "committee-update",
405            VersionedLogMessage::into_committee_update,
406        )?;
407        Ok(Some(msg.new_committee))
408    }

410    /// Read and verify the bootstrap record at the fixed key `genesis/record.json`.
411    /// The KPs authorize this record.
412    /// Return `None` if the record does not exist.
413    pub async fn read_genesis(&mut self) -> GuardianResult<Option<Box<GenesisLogMessage>>> {
414        let key = GenesisLogMessage::object_key();
415        let keys = self
416            .s3
417            .list_keys(&GenesisLogMessage::object_key_dir())
418            .await?;
419        if keys.is_empty() {
420            info!("No genesis record found");
421            return Ok(None);
422        }
423        if keys != [key.clone()] {
424            return Err(InvalidS3Log(format!(
425                "expected exactly one genesis record at {key}, found {keys:?}"
426            )));
427        }
428        let verified_log = self.read_log(&key, false).await?;
429        let genesis = verified_log.extract("genesis", VersionedLogMessage::into_genesis)?;
430        Ok(Some(genesis))
431    }
432
433    /// Read the latest committee in service.
434    ///
435    /// Use the latest `committee-update/` record if one exists.
436    /// Otherwise, use the bootstrap record at `genesis/record.json`, which the KPs authorize.
437    /// Return `None` if neither record exists.
438    pub async fn read_latest_committee(&mut self) -> GuardianResult<Option<Committee>> {
439        if let Some(committee) = self.read_latest_committee_update().await? {
440            return Ok(Some(committee));
441        }
442        info!("No committee update log found; checking the genesis record");
443        Ok(self.read_genesis().await?.map(|genesis| genesis.committee))
444    }

447/// Create a reader with mock S3 responses for one optional record and additional keys.
448/// Add session information to the cache without attestation verification.
449/// Reads still verify the record signature and apply the S3 checks required by the read method.
450#[cfg(test)]
451pub(crate) fn reader_with_record_for_test(
452    log: Option<SignedLogEntry>,
453    signing_pubkey: hashi_types::guardian::GuardianPubKey,
454    extra_keys: Vec<String>,
455) -> GuardianReader {
456    use aws_sdk_s3::operation::get_object::GetObjectOutput;
457    use aws_sdk_s3::operation::list_object_versions::ListObjectVersionsOutput;
458    use aws_sdk_s3::primitives::ByteStream;
459    use aws_sdk_s3::primitives::DateTime;
460    use aws_sdk_s3::types::CommonPrefix;
461    use aws_sdk_s3::types::ObjectLockMode;
462    use aws_sdk_s3::types::ObjectVersion;
463    use aws_sdk_s3::Client;
464    use aws_smithy_mocks::mock;
465    use aws_smithy_mocks::mock_client;
466    use aws_smithy_mocks::RuleMode;
467    use hashi_types::guardian::InitConfig;
468    use hashi_types::guardian::S3ObjectLockPolicy;
469    use std::sync::Arc;
470    use std::sync::Mutex;
471
472    let mut keys = extra_keys;
473    if let Some(log) = &log {
474        keys.push(log.object_key().to_string());
475    }
476    let request = Arc::new(Mutex::new((String::new(), false)));
477    let captured_request = request.clone();
478    let list = mock!(Client::list_object_versions)
479        .match_requests(move |req| {
480            *captured_request.lock().unwrap() = (
481                req.prefix().unwrap_or_default().to_string(),
482                req.delimiter() == Some("/"),
483            );
484            true
485        })
486        .then_output(move || {
487            let (prefix, directories) = &*request.lock().unwrap();
488            if *directories {
489                let prefixes: std::collections::BTreeSet<_> = keys
490                    .iter()
491                    .filter_map(|key| {
492                        let rest = key.strip_prefix(prefix)?;
493                        let slash = rest.find('/')?;
494                        Some(format!("{prefix}{}", &rest[..=slash]))
495                    })
496                    .collect();
497                return ListObjectVersionsOutput::builder()
498                    .set_common_prefixes(Some(
499                        prefixes
500                            .into_iter()
501                            .map(|prefix| CommonPrefix::builder().prefix(prefix).build())
502                            .collect(),
503                    ))
504                    .build();
505            }
506            ListObjectVersionsOutput::builder()
507                .set_versions(Some(
508                    keys.iter()
509                        .filter(|key| key.starts_with(prefix.as_str()))
510                        .map(|key| ObjectVersion::builder().key(key).is_latest(true).build())
511                        .collect(),
512                ))
513                .build()
514        });
515    let config = InitConfig::mock_for_testing();
516    let policy = S3ObjectLockPolicy::for_environment(config.deployment().retention_environment);
517    let log_key = log.as_ref().map(|log| log.object_key().to_string());
518    let get = mock!(Client::get_object)
519        .match_requests(move |req| req.key() == log_key.as_deref())
520        .then_output(move || {
521            let log = log.as_ref().unwrap();
522            GetObjectOutput::builder()
523                .object_lock_mode(ObjectLockMode::Compliance)
524                .object_lock_retain_until_date(DateTime::from(log.object_lock_expiry(policy)))
525                .body(ByteStream::from(serde_json::to_vec(log).unwrap()))
526                .build()
527        });
528    let client = mock_client!(aws_sdk_s3, RuleMode::MatchAny, &[&list, &get]);
529    let s3 = GuardianS3Client::from_client(
530        config.deployment().bucket_info.clone(),
531        config.deployment().retention_environment,
532        client,
533    );
534    let mut reader = GuardianReader::from_s3_client(s3, config.deployment().clone());
535    // Seed the attestation cache; the records still undergo normal signature,
536    // object-key, history, and lock verification.
537    reader.sessions.insert(
538        SessionID::from_signing_pubkey(&signing_pubkey),
539        VerifiedSessionInfo::new_for_test(
540            signing_pubkey,
541            config.deployment().pcr_allowlist.current_build().clone(),
542        ),
543    );
544    reader
545}
