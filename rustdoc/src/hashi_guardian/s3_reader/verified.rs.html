18/// Initialization checkpoint required by or verified for a session.
19///
20/// Variants are ordered by the durable log prefix each checkpoint proves.
21#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
22enum InitCheckpoint {
23    /// OI attestation and info (01-02), verified while initializing VerifiedSessionInfo.
24    OperatorInitialized,
25    /// The complete withdraw-mode initialization sequence (01-04).
26    OperatorActivated,
27}
28
29/// A session's attestation-anchored signing key, signed [`GuardianInfo`], build
30/// PCRs, and highest verified initialization checkpoint.
31#[derive(Debug, Clone)]
32pub struct VerifiedSessionInfo {
33    signing_pubkey: GuardianPubKey,
34    info: GuardianInfo,
35    build_pcrs: BuildPcrs,
36    verified_init_checkpoint: InitCheckpoint,
37}
38
39/// A log record whose message signature, writing session's attestation/PCRs,
40/// and required initialization checkpoint have been verified. The exact
41/// versioned entry is retained so callers can choose which schema versions they
42/// accept and how to interpret them.
43#[derive(Debug)]
44pub struct VerifiedLogRecord {
45    entry: LogEntry,
46    build_pcrs: BuildPcrs,
47}
48
49impl InitCheckpoint {
50    /// Return the initialization checkpoint required before serving a log.
51    /// Withdrawal and committee-update logs require 01-04; heartbeat, ceremony,
52    /// and genesis logs require only 01-02. KP-share state is mode-dependent.
53    /// Init logs use the dedicated init-log reader and are rejected here.
54    fn required_for(log_type: LogType, mode: EnclaveMode) -> GuardianResult<Self> {
55        let required = match log_type {
56            LogType::Init => {
57                return Err(InvalidS3Log(
58                    "unexpected init log in non-init-log reader".into(),
59                ));
60            }
61            LogType::Withdrawal | LogType::CommitteeUpdate => Self::OperatorActivated,
62            LogType::KpShareState => match mode {
63                EnclaveMode::Ceremony => Self::OperatorInitialized,
64                EnclaveMode::Withdraw => Self::OperatorActivated,
65            },
66            LogType::Heartbeat
67            | LogType::CeremonyCompleted
68            | LogType::CeremonyProposal
69            | LogType::Genesis => Self::OperatorInitialized,
70        };
71        Ok(required)
72    }
73}

86    pub(super) async fn read_from_s3(
87        s3: &GuardianS3Client,
88        session_id: &str,
89        expected_deployment: &DeploymentConfig,
90    ) -> GuardianResult<Self> {
91        // 1. Attestation (unsigned: authenticated by AWS, not the enclave key) →
92        //    the signing pubkey it commits to.
93        let att_key = InitLogMessage::attestation_object_key(session_id);
94        let attestation_message = Self::read_init_log(s3, &att_key, None).await?;
95        let InitLogMessage::OIAttestationUnsigned {
96            attestation,
97            signing_public_key: signing_pubkey,
98        } = *attestation_message
99        else {
100            return Err(InvalidS3Log(format!(
101                "expected OIAttestationUnsigned at key {att_key}"
102            )));
103        };
104
105        // 2. GuardianInfo, signature-verified under that pubkey → the reported build.
106        let info_key = InitLogMessage::guardian_info_object_key(session_id);
107        let info_message = Self::read_init_log(s3, &info_key, Some(&signing_pubkey)).await?;
108        let InitLogMessage::OIGuardianInfo(info) = *info_message else {
109            return Err(InvalidS3Log(format!(
110                "expected OIGuardianInfo at key {info_key}"
111            )));
112        };
113        let info = *info;
114
115        // 3. Anchor the pubkey and pin PCR0 to the allowlist entry for the
116        //    reported build. This replays a logged attestation whose short-lived
117        //    leaf cert has typically expired, so the chain is checked at the
118        //    document's own signed timestamp, not now.
119        let build_pcrs = verify_deployment_info(
120            session_id,
121            info.deployment_info.as_ref(),
122            expected_deployment,
123        )?;
124        attestation
125            .verify_replay(&signing_pubkey, &build_pcrs)
126            .map_err(|e| InvalidS3Log(format!("attestation at key {att_key}: {e}")))?;
127
128        Ok(Self {
129            signing_pubkey,
130            info,
131            build_pcrs,
132            verified_init_checkpoint: InitCheckpoint::OperatorInitialized,
133        })
134    }

136    /// Verify a record and the initialization checkpoint required to emit it.
137    pub(super) async fn verify_record(
138        &mut self,
139        s3: &GuardianS3Client,
140        record: LogRecord,
141    ) -> GuardianResult<VerifiedLogRecord> {
142        let entry = record.validate_into_entry(Some(&self.signing_pubkey))?;
143        let required = InitCheckpoint::required_for(entry.log_type(), self.info.lifecycle.mode())?;
144        self.ensure_init_checkpoint(s3, entry.session_id(), required)
145            .await?;
146        Ok(VerifiedLogRecord {
147            entry,
148            build_pcrs: self.build_pcrs.clone(),
149        })
150    }
151
152    async fn ensure_init_checkpoint(
153        &mut self,
154        s3: &GuardianS3Client,
155        session_id: &str,
156        required: InitCheckpoint,
157    ) -> GuardianResult<()> {
158        if self.verified_init_checkpoint >= required {
159            return Ok(());
160        }
161
162        match required {
163            InitCheckpoint::OperatorInitialized => {
164                unreachable!("session construction verifies operator initialization")
165            }
166            InitCheckpoint::OperatorActivated => {
167                let pi_key = InitLogMessage::pi_fully_initialized_object_key(session_id);
168                let pi_message =
169                    Self::read_init_log(s3, &pi_key, Some(&self.signing_pubkey)).await?;
170
171                let oa_key = InitLogMessage::oa_activated_object_key(session_id);
172                let oa_message =
173                    Self::read_init_log(s3, &oa_key, Some(&self.signing_pubkey)).await?;
174                InitLogMessage::verify_oi_pi_consistency(&self.info, &pi_message)?;
175                InitLogMessage::verify_oi_oa_consistency(&self.info, &oa_message)?;
176                InitLogMessage::verify_pi_oa_consistency(&pi_message, &oa_message)?;
177            }
178        }
179
180        self.verified_init_checkpoint = required;
181        Ok(())
182    }
183
184    /// Read an init log and validate it with the supplied signing key, if any.
185    async fn read_init_log(
186        s3: &GuardianS3Client,
187        key: &str,
188        signing_pubkey: Option<&GuardianPubKey>,
189    ) -> GuardianResult<Box<InitLogMessage>> {
190        let record = s3.get_log_record(key).await?;
191        let entry = record.validate_into_entry(signing_pubkey)?;
192        entry
193            .into_message()
194            .into_init()
195            .ok_or_else(|| InvalidS3Log(format!("expected an init log at key {key}")))
196    }

211/// Authenticate deployment identity separately from build selection: historical
212/// sessions may use older allowlisted builds, but must serve the same deployment.
213fn verify_deployment_info(
214    session_id: &str,
215    reported: Option<&DeploymentConfigSummary>,
216    expected: &DeploymentConfig,
217) -> GuardianResult<BuildPcrs> {
218    let reported = reported.ok_or_else(|| {
219        InvalidS3Log(format!(
220            "session {session_id} GuardianInfo is missing deployment_info"
221        ))
222    })?;
223    if reported.bucket_info != expected.bucket_info {
224        return Err(InvalidS3Log(format!(
225            "session {session_id} bucket/region {:?} does not match expected {:?}",
226            reported.bucket_info, expected.bucket_info
227        )));
228    }
229    if reported.retention_environment != expected.retention_environment {
230        return Err(InvalidS3Log(format!(
231            "session {session_id} retention environment {:?} does not match expected {:?}",
232            reported.retention_environment, expected.retention_environment
233        )));
234    }
235    if reported.bitcoin_network != expected.bitcoin_network {
236        return Err(InvalidS3Log(format!(
237            "session {session_id} Bitcoin network {:?} does not match expected {:?}",
238            reported.bitcoin_network, expected.bitcoin_network
239        )));
240    }
241    expected
242        .pcr_allowlist
243        .resolve(&reported.git_revision)
244        .cloned()
245}

295        let expected = DeploymentConfig::mock_for_testing();
296        let reported = expected.summary();

363            .object_lock_mode(ObjectLockMode::Compliance)

369    #[test]
370    fn required_init_checkpoint_matches_log_type_and_mode() {
371        use InitCheckpoint::OperatorActivated;
372        use InitCheckpoint::OperatorInitialized;
373
374        for mode in [EnclaveMode::Ceremony, EnclaveMode::Withdraw] {
375            assert_eq!(
376                InitCheckpoint::required_for(LogType::Withdrawal, mode).unwrap(),
377                OperatorActivated
378            );
379            assert_eq!(
380                InitCheckpoint::required_for(LogType::CommitteeUpdate, mode).unwrap(),
381                OperatorActivated
382            );
383            for log_type in [
384                LogType::Heartbeat,
385                LogType::CeremonyCompleted,
386                LogType::CeremonyProposal,
387                LogType::Genesis,
388            ] {
389                assert_eq!(
390                    InitCheckpoint::required_for(log_type, mode).unwrap(),
391                    OperatorInitialized
392                );
393            }
394        }
395
396        assert!(InitCheckpoint::required_for(LogType::Init, EnclaveMode::Ceremony).is_err());
397        assert_eq!(
398            InitCheckpoint::required_for(LogType::KpShareState, EnclaveMode::Ceremony).unwrap(),
399            OperatorInitialized
400        );
401        assert_eq!(
402            InitCheckpoint::required_for(LogType::KpShareState, EnclaveMode::Withdraw).unwrap(),
403            OperatorActivated
404        );
405    }

407    #[tokio::test]
408    async fn operator_activated_checkpoint_is_verified_once_per_session() {
409        let signing_key = GuardianSignKeyPair::from([8u8; 32]);
410        let signing_pubkey = signing_key.verification_key();
411        let session_id = SessionID::from_signing_pubkey(&signing_pubkey);
412        let pi_log = LogRecord::new(
413            session_id.clone(),
414            LogMessage::Init(Box::new(InitLogMessage::PIEnclaveFullyInitialized {
415                sharing_seq: 0,
416                share_ids: (1..=3).map(|id| ShareID::new(id).unwrap()).collect(),
417                enclave_btc_pubkey: hashi_types::bitcoin::create_btc_keypair_for_test(&[1; 32])
418                    .x_only_public_key()
419                    .0,
420            })),
421            &signing_key,
422        );
423        let oa_log = LogRecord::new(
424            session_id.clone(),
425            LogMessage::Init(Box::new(InitLogMessage::OAActivated {
426                state_hash: [1; 32],
427                config_hash: [2; 32],
428                sharing_seq: 0,
429                committee_epoch: 4,
430                limiter_state: LimiterState {
431                    num_tokens_available: 5,
432                    last_updated_at: 6,
433                    next_seq: 7,
434                },
435            })),
436            &signing_key,
437        );
438        let pi_key = pi_log.object_key().to_string();
439        let oa_key = oa_log.object_key().to_string();
440        let policy = S3ObjectLockPolicy::for_environment(S3RetentionEnvironment::Testnet);
441
442        let list_logs = mock!(Client::list_object_versions)
443            .sequence()
444            .output(move || listed_record(pi_key.clone()))
445            .output(move || listed_record(oa_key.clone()))
446            .build();
447        let get_logs = mock!(Client::get_object)
448            .sequence()
449            .output(move || locked_record(&pi_log, policy))
450            .output(move || locked_record(&oa_log, policy))
451            .build();
452        let client = mock_client!(aws_sdk_s3, RuleMode::MatchAny, &[&list_logs, &get_logs]);
453        let s3 = GuardianS3Client::from_client_for_tests(
454            S3BucketInfo::mock_for_testing(),

456            client,
457        );

460        session_info

471            InitCheckpoint::OperatorActivated
472        );
473        assert_eq!(list_logs.num_calls(), 2);
474        assert_eq!(get_logs.num_calls(), 2);
475    }
