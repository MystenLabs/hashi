23/// An initialization checkpoint required for a read or verified for a session.
24#[derive(Clone, Copy, Debug, Eq, PartialEq)]
25enum InitCheckpoint {
26    /// Operator initialization logs 01-02: the attestation and session information.
27    /// Construction of [`VerifiedSessionInfo`] verifies these logs.
28    OperatorInitialized,
29    /// Initialization logs 01-04 for withdrawal mode.
30    OperatorActivated,
31}
32
33/// Verified session information and the highest verified initialization checkpoint.
34/// This includes the signing key, signed [`OperatorInitInfo`], and build PCR values.
35/// The session attestation verifies the signing key and PCR values.
36#[derive(Debug, Clone)]
37pub struct VerifiedSessionInfo {
38    signing_pubkey: GuardianPubKey,
39    info: OperatorInitInfo,
40    build_pcrs: BuildPcrs,
41    verified_init_checkpoint: InitCheckpoint,
42}
43
44/// A log entry with a verified signature, session attestation, build PCR values, and required initialization checkpoint.
45/// This type keeps the original entry and its schema version.
46/// The caller selects which schema versions to accept and how to read them.
47#[derive(Debug)]
48pub struct VerifiedLogEntry {
49    entry: LogEntry,
50    build_pcrs: BuildPcrs,
51}
52
53impl InitCheckpoint {
54    /// Return the initialization checkpoint required to read a log.
55    /// Withdrawal and committee update logs require initialization logs 01-04.
56    /// Heartbeat, ceremony, ceremony proposal, and genesis logs require initialization logs 01-02.
57    /// KP-share state logs require logs 01-02 in ceremony mode and logs 01-04 in withdrawal mode.
58    /// Reject initialization logs here. Use the initialization log reader for those records.
59    fn required_for(log_type: LogType, mode: EnclaveMode) -> GuardianResult<Self> {
60        let required = match log_type {
61            LogType::Init => {
62                return Err(InvalidS3Log(
63                    "unexpected init log in non-init-log reader".into(),
64                ));
65            }
66            LogType::Withdrawal | LogType::CommitteeUpdate => Self::OperatorActivated,
67            LogType::KpShareState => match mode {
68                EnclaveMode::Ceremony => Self::OperatorInitialized,
69                EnclaveMode::Withdraw => Self::OperatorActivated,
70            },
71            LogType::Heartbeat
72            | LogType::CeremonyCompleted
73            | LogType::CeremonyProposal
74            | LogType::Genesis => Self::OperatorInitialized,
75        };
76        Ok(required)
77    }
78}

92    /// Read and verify the session attestation and operator initialization information from S3.
93    /// Select the build PCR values from the reader's deployment allowlist.
94    /// Verify the attestation before using its signing key to verify the two log signatures.
95    pub(super) async fn read_from_s3(
96        s3: &GuardianS3Client,
97        session_id: &str,
98        expected_deployment: &DeploymentConfig,
99    ) -> GuardianResult<Self> {
100        // Fetch the attestation and claimed signing key without trusting either.
101        let att_key = InitLogMessage::attestation_object_key(session_id);
102        let attestation_record = s3.get_log_record(&att_key).await?;
103        let InitLogMessage::OIAttestation {
104            attestation,
105            signing_public_key: signing_pubkey,
106        } = Self::unverified_init_message(&attestation_record)?
107        else {
108            return Err(InvalidS3Log(format!(
109                "expected OIAttestation at key {att_key}"
110            )));
111        };
112
113        // The unverified OI info selects a build from the reader's own allowlist.
114        let info_key = InitLogMessage::guardian_info_object_key(session_id);
115        let info_record = s3.get_log_record(&info_key).await?;
116        let InitLogMessage::OIGuardianInfo(info) = Self::unverified_init_message(&info_record)?
117        else {
118            return Err(InvalidS3Log(format!(
119                "expected OIGuardianInfo at key {info_key}"
120            )));
121        };
122        // Replay checks the certificate chain at the attestation's signed timestamp.
123        let build_pcrs =
124            verify_deployment_and_resolve_build(session_id, &info.deployment, expected_deployment)?;
125        attestation
126            .verify_replay(signing_pubkey, &build_pcrs)
127            .map_err(|e| InvalidS3Log(format!("attestation at key {att_key}: {e}")))?;
128
129        // An embedded key's signature is trusted only after its attestation verifies.
130        attestation_record.validate(signing_pubkey)?;
131        info_record.validate(signing_pubkey)?;
132
133        info!(
134            session_id,
135            git_revision = build_pcrs.git_revision(),
136            "Verified the session attestation and operator initialization logs"
137        );
138
139        Ok(Self {
140            signing_pubkey: *signing_pubkey,
141            info: info.as_ref().clone(),
142            build_pcrs,
143            verified_init_checkpoint: InitCheckpoint::OperatorInitialized,
144        })
145    }

147    /// Verify a record and the initialization checkpoint required to write it.
148    pub(super) async fn verify_record(
149        &mut self,
150        s3: &GuardianS3Client,
151        record: SignedLogEntry,
152    ) -> GuardianResult<VerifiedLogEntry> {
153        let entry = record.validate_into_entry(&self.signing_pubkey)?;
154        let required = InitCheckpoint::required_for(entry.log_type(), self.info.mode())?;
155        self.ensure_init_checkpoint(s3, entry.session_id(), required)
156            .await?;
157        Ok(VerifiedLogEntry {
158            entry,
159            build_pcrs: self.build_pcrs.clone(),
160        })
161    }

163    /// Verify the required initialization checkpoint if this session has not reached it.
164    /// Keep the current checkpoint if any verification fails.
165    /// The `session_id` must identify this session.
166    async fn ensure_init_checkpoint(
167        &mut self,
168        s3: &GuardianS3Client,
169        session_id: &str,
170        required: InitCheckpoint,
171    ) -> GuardianResult<()> {
172        match (self.verified_init_checkpoint, required) {
173            (_, InitCheckpoint::OperatorInitialized)
174            | (InitCheckpoint::OperatorActivated, InitCheckpoint::OperatorActivated) => Ok(()),
175            (InitCheckpoint::OperatorInitialized, InitCheckpoint::OperatorActivated) => {
176                let pi_key = InitLogMessage::pi_fully_initialized_object_key(session_id);
177                let pi_message =
178                    Self::read_verified_init_log(s3, &pi_key, &self.signing_pubkey).await?;
179
180                let oa_key = InitLogMessage::oa_activated_object_key(session_id);
181                let oa_message =
182                    Self::read_verified_init_log(s3, &oa_key, &self.signing_pubkey).await?;
183                InitLogMessage::verify_oi_pi_consistency(&self.info, &pi_message)?;
184                InitLogMessage::verify_oi_oa_consistency(&self.info, &oa_message)?;
185                InitLogMessage::verify_pi_oa_consistency(&pi_message, &oa_message)?;
186                self.verified_init_checkpoint = InitCheckpoint::OperatorActivated;
187                info!(session_id, "Verified the session activation checkpoint");
188                Ok(())
189            }
190        }
191    }

193    /// The caller must verify the attestation and log signatures before trusting the message.
194    fn unverified_init_message(record: &SignedLogEntry) -> GuardianResult<&InitLogMessage> {
195        record.message_unchecked().as_init().ok_or_else(|| {
196            InvalidS3Log(format!(
197                "expected an init log at key {}",
198                record.object_key()
199            ))
200        })
201    }

203    /// Read an initialization log and verify it with the authenticated signing key.
204    async fn read_verified_init_log(
205        s3: &GuardianS3Client,
206        key: &str,
207        signing_pubkey: &GuardianPubKey,
208    ) -> GuardianResult<Box<InitLogMessage>> {
209        let record = s3.get_log_record(key).await?;
210        let entry = record.validate_into_entry(signing_pubkey)?;
211        entry
212            .into_message()
213            .into_init()
214            .ok_or_else(|| InvalidS3Log(format!("expected an init log at key {key}")))
215    }

230/// Check the reported deployment and return its build from the reader's allowlist.
231/// The reported current build must have the same PCR values as the selected build.
232/// Log an error if a shared historical revision has different PCR values in the two allowlists.
233/// Historical mismatches do not prevent the read.
234/// The caller must then verify the Nitro attestation against the selected build's PCR values.
235fn verify_deployment_and_resolve_build(
236    session_id: &str,
237    reported: &DeploymentConfig,
238    expected: &DeploymentConfig,
239) -> GuardianResult<BuildPcrs> {
240    if reported.bucket_info != expected.bucket_info {
241        return Err(InvalidS3Log(format!(
242            "session {session_id} bucket/region {:?} does not match expected {:?}",
243            reported.bucket_info, expected.bucket_info
244        )));
245    }
246    if reported.retention_environment != expected.retention_environment {
247        return Err(InvalidS3Log(format!(
248            "session {session_id} retention environment {:?} does not match expected {:?}",
249            reported.retention_environment, expected.retention_environment
250        )));
251    }
252    if reported.bitcoin_network != expected.bitcoin_network {
253        return Err(InvalidS3Log(format!(
254            "session {session_id} Bitcoin network {:?} does not match expected {:?}",
255            reported.bitcoin_network, expected.bitcoin_network
256        )));
257    }
258    let reported_build = reported.pcr_allowlist.current_build();
259    let expected_build = expected
260        .pcr_allowlist
261        .resolve(reported_build.git_revision())?;
262    if reported_build != expected_build {
263        return Err(InvalidS3Log(format!(
264            "session {session_id} reported PCR values for build '{}' do not match the reader's allowlist",
265            reported_build.git_revision()
266        )));
267    }
268
269    // Historical revisions can be removed from either allowlist.
270    // Compare only revisions present in both allowlists.
271    // Log an error if their PCR values differ, but continue the read.
272    for reported_build in reported.pcr_allowlist.prev_builds() {
273        let Ok(expected_build) = expected
274            .pcr_allowlist
275            .resolve(reported_build.git_revision())
276        else {
277            continue;
278        };
279        if reported_build != expected_build {
280            error!(
281                session_id,
282                git_revision = reported_build.git_revision(),
283                "Reported historical PCR values do not match the reader's allowlist"
284            );
285        }
286    }
287
288    Ok(expected_build.clone())
289}

314    /// Extract the expected message.
315    /// If extraction fails, return an error with the expected message type and object key.
316    pub fn extract<T>(
317        self,
318        expected: &str,
319        extract: impl FnOnce(VersionedLogMessage) -> Option<T>,
320    ) -> GuardianResult<T> {
321        let key = self.entry.object_key().to_owned();
322        extract(self.entry.into_message())
323            .ok_or_else(|| InvalidS3Log(format!("expected a {expected} log at {key}")))
324    }
