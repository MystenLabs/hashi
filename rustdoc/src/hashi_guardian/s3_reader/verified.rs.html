19/// A session's verified guardian info: the attestation-anchored signing key,
20/// the signed [`GuardianInfo`], and the build PCRs proven by attestation.
21#[derive(Debug, Clone)]
22pub struct VerifiedSessionInfo {
23    signing_pubkey: GuardianPubKey,
24    info: GuardianInfo,
25    build_pcrs: BuildPcrs,
26}
27
28impl VerifiedSessionInfo {
29    pub(super) async fn read_from_s3(
30        s3: &GuardianS3Client,
31        session_id: &str,
32        allowlist: &PcrAllowlist,
33    ) -> GuardianResult<Self> {
34        // 1. Attestation (unsigned: authenticated by AWS, not the enclave key) →
35        //    the signing pubkey it commits to.
36        let att_key = InitLogMessage::attestation_object_key(session_id);
37        let attestation_record = s3.get_log_record(&att_key).await?;
38        let attestation_message = match attestation_record.validate_into_entry(None)?.into_message()
39        {
40            V1(LogMessageV1::Init(message)) | V2(LogMessageV2::Init(message)) => message,
41            V1(_) | V2(_) => {
42                return Err(InvalidS3Log(format!(
43                    "expected OIAttestationUnsigned at key {att_key}"
44                )));
45            }
46        };
47        let InitLogMessage::OIAttestationUnsigned {
48            attestation,
49            signing_public_key: signing_pubkey,
50        } = *attestation_message
51        else {
52            return Err(InvalidS3Log(format!(
53                "expected OIAttestationUnsigned at key {att_key}"
54            )));
55        };
56
57        // 2. GuardianInfo, signature-verified under that pubkey → the reported build.
58        let info_key = InitLogMessage::guardian_info_object_key(session_id);
59        let info_record = s3.get_log_record(&info_key).await?;
60        let info_message = match info_record
61            .validate_into_entry(Some(&signing_pubkey))?
62            .into_message()
63        {
64            V1(LogMessageV1::Init(message)) | V2(LogMessageV2::Init(message)) => message,
65            V1(_) | V2(_) => {
66                return Err(InvalidS3Log(format!(
67                    "expected OIGuardianInfo at key {info_key}"
68                )));
69            }
70        };
71        let InitLogMessage::OIGuardianInfo(info) = *info_message else {
72            return Err(InvalidS3Log(format!(
73                "expected OIGuardianInfo at key {info_key}"
74            )));
75        };
76        let info = *info;
77
78        // 3. Anchor the pubkey and pin PCR0 to the allowlist entry for the
79        //    reported build. This replays a logged attestation whose short-lived
80        //    leaf cert has typically expired, so the chain is checked at the
81        //    document's own signed timestamp, not now.
82        let build_pcrs = allowlist.resolve(&info.untrusted_git_revision)?.clone();
83        attestation
84            .verify_replay(&signing_pubkey, &build_pcrs)
85            .map_err(|e| InvalidS3Log(format!("attestation at key {att_key}: {e}")))?;
86
87        Ok(Self {
88            signing_pubkey,
89            info,
90            build_pcrs,
91        })
92    }
93
94    /// Verify a record with this session's attestation-anchored signing key.
95    pub(super) fn verify_record(&self, record: LogRecord) -> GuardianResult<VerifiedLogRecord> {
96        let entry = record.validate_into_entry(Some(&self.signing_pubkey))?;
97        Ok(VerifiedLogRecord {
98            entry,
99            build_pcrs: self.build_pcrs.clone(),
100        })
101    }
102
103    pub fn signing_pubkey(&self) -> &GuardianPubKey {
104        &self.signing_pubkey
105    }
106
107    pub fn info(&self) -> &GuardianInfo {
108        &self.info
109    }
110
111    pub fn build_pcrs(&self) -> &BuildPcrs {
112        &self.build_pcrs
113    }
114}

116/// A log record whose message signature and writing session's attestation/PCRs
117/// have both been verified. The exact versioned entry is retained so callers
118/// can choose which schema versions they accept and how to interpret them.
119#[derive(Debug)]
120pub struct VerifiedLogRecord {
121    entry: LogEntry,
122    build_pcrs: BuildPcrs,
123}
124
125impl VerifiedLogRecord {
126    #[cfg(test)]
127    pub(super) fn new_for_test(entry: LogEntry, build_pcrs: BuildPcrs) -> Self {
128        Self { entry, build_pcrs }
129    }
130
131    pub fn entry(&self) -> &LogEntry {
132        &self.entry
133    }
134
135    pub fn build_pcrs(&self) -> &BuildPcrs {
136        &self.build_pcrs
137    }
138
139    pub fn into_entry(self) -> LogEntry {
140        self.entry
141    }
142}
