25impl GuardianReader {
26    /// Enforces that `live_session` has heartbeated recently.
27    pub async fn ensure_session_live(&mut self, live_session: &str) -> GuardianResult<()> {
28        let (summary, now) = self.read_recent_heartbeat_summary().await?;
29        let live_session_info = validate_session_live(&summary, now, live_session)?;
30
31        info!(
32            session_id = %live_session,
33            first_heartbeat = live_session_info.first_heartbeat,
34            last_heartbeat = live_session_info.last_heartbeat,
35            age_secs = now.saturating_sub(live_session_info.last_heartbeat),
36            "session heartbeat check passed"
37        );
38        Ok(())
39    }
40
41    /// Enforces that `live_session` has heartbeated recently, while every other
42    /// guardian session has been quiet long enough to no longer be considered
43    /// active.
44    pub async fn ensure_session_live_and_others_quiet(
45        &mut self,
46        live_session: &str,
47    ) -> GuardianResult<()> {
48        let (summary, now) = self.read_recent_heartbeat_summary().await?;
49
50        let live_session_info = validate_session_live(&summary, now, live_session)?;
51        validate_other_sessions_quiet(
52            &summary,
53            now,
54            live_session,
55            OTHER_SESSION_QUIET_PERIOD.as_secs(),
56        )?;
57
58        info!(
59            session_id = %live_session,
60            first_heartbeat = live_session_info.first_heartbeat,
61            last_heartbeat = live_session_info.last_heartbeat,
62            age_secs = now.saturating_sub(live_session_info.last_heartbeat),
63            "activation heartbeat check passed"
64        );
65        Ok(())
66    }
67
68    async fn read_recent_heartbeat_summary(
69        &mut self,
70    ) -> GuardianResult<(Vec<GuardianSessionInfo>, UnixSeconds)> {
71        let recent_heartbeats = self.read_recent_heartbeat_logs().await?;
72        let summary = summarize_heartbeats_by_session(recent_heartbeats)?;
73        Ok((summary, now_timestamp_secs()))
74    }
75
76    async fn read_recent_heartbeat_logs(&mut self) -> GuardianResult<Vec<VerifiedLogRecord>> {
77        // Read from the previous, current, and next hour-scoped prefixes to
78        // cover clock-boundary cases and moderate clock skew.
79        let one_hour_ago = now_timestamp_secs().saturating_sub(60 * 60);
80        let mut cursor = S3HourScopedDirectory::heartbeat(one_hour_ago);
81        let mut logs = Vec::new();
82        for _ in 0..3 {
83            logs.extend(self.read_logs_in_dir(&cursor).await?);
84            cursor = cursor.next_dir();
85        }
86        Ok(logs)
87    }
88}

90#[derive(Debug, Clone)]
91struct GuardianSessionInfo {
92    session_id: SessionID,
93    first_heartbeat: UnixSeconds,
94    last_heartbeat: UnixSeconds,
95}
96
97fn summarize_heartbeats_by_session(
98    logs: Vec<VerifiedLogRecord>,
99) -> GuardianResult<Vec<GuardianSessionInfo>> {
100    let mut map: BTreeMap<SessionID, (UnixSeconds, UnixSeconds)> = BTreeMap::new();
101
102    for log in logs {
103        let entry = log.into_entry();
104        let session_id = entry.session_id().clone();
105        let ts = unix_millis_to_seconds(entry.timestamp_ms());
106        match entry.into_message() {
107            V1(LogMessageV1::Heartbeat(..)) | V2(LogMessageV2::Heartbeat(..)) => {}
108            V1(_) | V2(_) => {
109                return Err(InvalidS3Log(
110                    "non-heartbeat log found under the heartbeat prefix".into(),
111                ));
112            }
113        }
114
115        map.entry(session_id)
116            .and_modify(|(first, last)| {
117                *first = (*first).min(ts);
118                *last = (*last).max(ts);
119            })
120            .or_insert((ts, ts));
121    }
122
123    Ok(map
124        .into_iter()
125        .map(
126            |(session_id, (first_heartbeat, last_heartbeat))| GuardianSessionInfo {
127                session_id,
128                first_heartbeat,
129                last_heartbeat,
130            },
131        )
132        .collect())
133}

135fn validate_other_sessions_quiet(
136    summary: &[GuardianSessionInfo],
137    now: UnixSeconds,
138    live_session: &str,
139    other_session_quiet_secs: UnixSeconds,
140) -> GuardianResult<()> {
141    if let Some(most_recent_other_session) = summary
142        .iter()
143        .filter(|s| s.session_id.as_str() != live_session)
144        .max_by_key(|s| s.last_heartbeat)
145    {
146        let heartbeat_age_secs = now.saturating_sub(most_recent_other_session.last_heartbeat);
147        if heartbeat_age_secs < other_session_quiet_secs {
148            return Err(PriorSessionHeartbeatStillRecent {
149                session_id: most_recent_other_session.session_id.clone(),
150                heartbeat_age_secs,
151                required_quiet_secs: other_session_quiet_secs,
152            });
153        }
154    }
155    Ok(())
156}
157
158fn validate_session_live<'a>(
159    summary: &'a [GuardianSessionInfo],
160    now: UnixSeconds,
161    live_session: &str,
162) -> GuardianResult<&'a GuardianSessionInfo> {
163    let live_session_info = summary
164        .iter()
165        .find(|s| s.session_id.as_str() == live_session)
166        .ok_or_else(|| CurrentSessionHeartbeatNotLive {
167            session_id: live_session.into(),
168            heartbeat_age_secs: None,
169            retry_after_secs: HEARTBEAT_INTERVAL.as_secs(),
170        })?;
171    let live_session_age_secs = now.saturating_sub(live_session_info.last_heartbeat);
172    if live_session_age_secs > LIVE_SESSION_LATEST_HEARTBEAT_MAX_AGE.as_secs() {
173        return Err(CurrentSessionHeartbeatNotLive {
174            session_id: live_session.into(),
175            heartbeat_age_secs: Some(live_session_age_secs),
176            retry_after_secs: HEARTBEAT_INTERVAL.as_secs(),
177        });
178    }
179    Ok(live_session_info)
180}

195    fn heartbeat_log(session_id: &str, timestamp_secs: UnixSeconds) -> VerifiedLogRecord {
196        verified_log(
197            session_id,
198            timestamp_secs * 1_000,
199            LogMessage::Heartbeat(HeartbeatLogMessage::new(0)),
200        )
201    }

203    fn non_heartbeat_log() -> VerifiedLogRecord {
204        verified_log(
205            "test-session",
206            0,
207            LogMessage::Init(Box::new(InitLogMessage::PIEnclaveFullyInitialized {
208                sharing_seq: 0,
209                share_ids: vec![],
210                enclave_btc_pubkey: hashi_types::bitcoin::create_btc_keypair_for_test(&[1; 32])
211                    .x_only_public_key()
212                    .0,
213            })),
214        )
215    }

217    fn verified_log(session_id: &str, timestamp_ms: u64, message: LogMessage) -> VerifiedLogRecord {
218        let signing_key = GuardianSignKeyPair::from([7u8; 32]);
219        let entry = hashi_types::guardian::LogRecord::new_at_timestamp(
220            session_id.into(),
221            message,
222            &signing_key,
223            timestamp_ms,
224        )
225        .into_entry_unchecked();
226        VerifiedLogRecord::new_for_test(entry, build_pcrs())
227    }

229    #[test]
230    fn summarize_heartbeats_tracks_latest_per_session() {
231        let summary = summarize_heartbeats_by_session(vec![
232            heartbeat_log("b", 20),
233            heartbeat_log("a", 10),
234            heartbeat_log("a", 30),
235        ])
236        .unwrap();
237
238        assert_eq!(summary.len(), 2);
239        assert_eq!(summary[0].session_id.as_str(), "a");
240        assert_eq!(summary[0].first_heartbeat, 10);
241        assert_eq!(summary[0].last_heartbeat, 30);
242        assert_eq!(summary[1].session_id.as_str(), "b");
243        assert_eq!(summary[1].first_heartbeat, 20);
244        assert_eq!(summary[1].last_heartbeat, 20);
245    }
