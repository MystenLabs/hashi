4//! Recovers a standby enclave's activation limiter state from guardian S3
5//! withdrawal logs.
6//!
7//! Each withdrawal log carries the limiter `post_state` after that
8//! consume. The withdrawal seq is strictly monotonic across rotations, so the
9//! global max-seq withdrawal log holds the most recent limiter state.
10//!
11//! Finding that log is a 4-level S3 tree-walk over the hour-partitioned layout
12//! (`withdraw/YYYY/MM/DD/HH/`): at each level we list `CommonPrefixes`, pick the
13//! greatest numeric component, and descend. The first hour bucket containing
14//! any withdrawal key is the latest non-empty bucket. We read it and one bucket back
15//! (sub-hour clock-skew defense across hour boundaries), then take the max-seq
16//! withdrawal across both.
17//!
18//! We deliberately do not apply the auditor's `write_completion_time`
19//! (`DIR_WRITES_COMPLETION_DELAY`) gate when reading the found bucket. That gate
20//! exists for polling/auditor reads where the source might still be writing; if
21//! used here, an enclave that died late in an hour could have its final-hour
22//! bucket treated as not-yet-complete, and recovery would miss the most recent
23//! log. Activation instead calls this only after the heartbeat quiet check has
24//! confirmed every non-standby session has been silent long enough for S3
25//! read-after-write consistency to cover the old session's final writes.
26

39impl GuardianReader {
40    /// Derive the activation limiter state from withdrawal logs. Uses the
41    /// global max-seq withdrawal post-state when present, otherwise genesis, and
42    /// caps tokens to the supplied config in case capacity was lowered.
43    ///
44    /// Precondition: the caller must have already verified that every
45    /// non-standby session is quiet (`ensure_session_live_and_others_quiet`).
46    /// This read deliberately skips the `write_completion_time` gate, so it is
47    /// only sound once the prior session's final writes are guaranteed visible.
48    pub async fn recover_limiter_state(
49        &mut self,
50        limiter_config: &LimiterConfig,
51    ) -> GuardianResult<LimiterState> {
52        let Some(mut cursor) = find_latest_withdrawal_bucket(&self.s3).await? else {
53            // The search covers the complete S3 withdrawal history, so this
54            // branch is reachable only if no withdrawal has ever succeeded.
55            info!("no withdrawal logs found; using genesis limiter state");
56            return Ok(LimiterState::genesis(limiter_config));
57        };
58
59        // Read the found bucket + one bucket back, then take max-seq across
60        // both. The peek-back defends against sub-hour clock skew that may have
61        // placed a higher-seq log in the prior hour bucket.
62        let hit = bucket_max_post_state(self.read_logs_in_dir(&cursor).await?);
63        cursor = cursor.prev_dir();
64        let peek = bucket_max_post_state(self.read_logs_in_dir(&cursor).await?);
65        let recovered_state = [hit, peek]
66            .into_iter()
67            .flatten()
68            .max_by_key(|s| s.next_seq)
69            .ok_or_else(|| {
70                InvalidS3Log(
71                    "latest withdrawal bucket contained no verified withdrawal logs".into(),
72                )
73            })?;
74        let state = cap_limiter_state_to_config(recovered_state, limiter_config);
75        info!(
76            next_seq = state.next_seq,
77            last_updated_at = state.last_updated_at,
78            recovered_num_tokens_available = recovered_state.num_tokens_available,
79            capped_num_tokens_available = state.num_tokens_available,
80            "recovered limiter state from withdrawal logs"
81        );
82        Ok(state)
83    }
84}

86/// Finds the latest hour bucket under `withdraw/` containing at least one
87/// withdrawal key, by walking the YYYY/MM/DD/HH tree in descending numeric
88/// order at each level. Returns `None` if no withdrawal log exists anywhere.
89async fn find_latest_withdrawal_bucket(
90    s3_client: &GuardianS3Client,
91) -> GuardianResult<Option<S3HourDirectory>> {
92    let root = format!("{}/", S3_DIR_WITHDRAW);
93    for year in list_subdirs_desc(s3_client, &root).await? {
94        for month in list_subdirs_desc(s3_client, &year).await? {
95            for day in list_subdirs_desc(s3_client, &month).await? {
96                for hour in list_subdirs_desc(s3_client, &day).await? {
97                    if !s3_client.list_keys(&hour).await?.is_empty() {
98                        let dir = S3HourDirectory::from_path(&hour).map_err(|e| {
99                            InvalidS3Log(format!("invalid withdrawal-log directory {hour}: {e}"))
100                        })?;
101                        return Ok(Some(dir));
102                    }
103                }
104            }
105        }
106    }
107    Ok(None)
108}
109
110async fn list_subdirs_desc(
111    s3_client: &GuardianS3Client,
112    prefix: &str,
113) -> GuardianResult<Vec<String>> {
114    let mut dirs = s3_client
115        .list_common_prefixes(prefix)
116        .await?
117        .into_iter()
118        .map(|path| {
119            S3NumericDirectory::from_path(&path)
120                .map_err(|e| InvalidS3Log(format!("invalid withdrawal-log directory {path}: {e}")))
121        })
122        .collect::<GuardianResult<Vec<_>>>()?;
123    dirs.sort();
124    dirs.reverse();
125    Ok(dirs.into_iter().map(|dir| dir.to_string()).collect())
126}
127
128fn bucket_max_post_state(logs: Vec<VerifiedLogEntry>) -> Option<LimiterState> {
129    logs.into_iter()
130        .filter_map(|log| log.into_entry().into_message().into_withdrawal())
131        .map(|withdrawal| withdrawal.post_state)
132        .max_by_key(|s| s.next_seq)
133}
134
135fn cap_limiter_state_to_config(
136    mut state: LimiterState,
137    limiter_config: &LimiterConfig,
138) -> LimiterState {
139    state.num_tokens_available = state
140        .num_tokens_available
141        .min(limiter_config.max_bucket_capacity);
142    state
143}
