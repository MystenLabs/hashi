55impl OIWithdrawModeInstall {
56    pub fn from_parts(
57        init_config: InitConfig,
58        ceremony_state: CeremonyState,
59        genesis_state: Option<GenesisState>,
60        hashi_object_id: hashi_types::sui_sdk_types::Address,
61        mpc_master_g: HashiMasterG,
62    ) -> Self {

65            ceremony_state,
66            genesis_state,
67            hashi_object_id,
68            mpc_master_g,
69        }

72    /// Build the arming bundle from the stable config, verified ceremony state,
73    /// and either supplied bootstrap state or verified persisted genesis.
74    async fn from_ceremony_state(
75        reader: &mut GuardianReader,
76        config: InitConfig,
77        ceremony_state: CeremonyState,
78        genesis_state: Option<GenesisState>,
79    ) -> GuardianResult<Self> {
80        // First deployment pins these values for KP authorization during PI.
81        // Subsequent enclaves recover them from the verified immutable record.
82        let (hashi_object_id, mpc_master_g) = match &genesis_state {
83            Some(state) => {
84                let (_, hashi_object_id, mpc_master_g) = state.clone().into_parts();
85                (hashi_object_id, mpc_master_g)
86            }
87            None => {
88                let genesis = reader.read_genesis().await?.ok_or_else(|| {
89                    InvalidInputs("no genesis record found; genesis bootstrap is required".into())
90                })?;
91                (genesis.hashi_object_id, genesis.mpc_master_g)
92            }
93        };
94
95        Ok(Self::from_parts(
96            config,
97            ceremony_state,
98            genesis_state,
99            hashi_object_id,
100            mpc_master_g,
101        ))
102    }
103
104    /// Install the bundle onto a fresh enclave. Infallible by design (see the
105    /// `operator_init` invariant): every set runs once on a fresh enclave.
106    pub fn install_into(self, enclave: &Enclave) {
107        let config_hash = self.init_config.digest();
108        let limiter_config = *self.init_config.limiter_config();
109
110        info!(
111            "Setting secret-sharing instance: n={}, t={}, {} commitments.",
112            self.ceremony_state.secret_sharing_instance.num_shares(),
113            self.ceremony_state.secret_sharing_instance.threshold(),
114            self.ceremony_state
115                .secret_sharing_instance
116                .commitments()
117                .len()
118        );
119        if let Some(genesis_state) = &self.genesis_state {
120            info!(
121                genesis_state_hash = hex::encode(genesis_state.digest()),
122                "Storing genesis state."
123            );
124        }
125        enclave
126            .set_temporary_init_state(TemporaryInitState {
127                ceremony_state: self.ceremony_state,
128                genesis_state: self.genesis_state,
129                config_hash,
130            })
131            .expect("Unable to set temporary initialization state");
132
133        info!("Setting withdraw configuration.");
134        enclave
135            .install_config(self.mpc_master_g, limiter_config, self.hashi_object_id)
136            .expect("Unable to set enclave configuration");
137    }
138}

140/// Receives S3 API keys and mode-specific configuration. A ceremony enclave
141/// installs the shared deployment policy; a withdraw enclave additionally
142/// installs the stable `InitConfig`, arming state, and fixed `config_hash`.
143///
144/// Invariant: operator_init never returns an `Err` from a partially-initialized
145/// enclave. Every fallible preparation step (validation, attestation, S3 access)
146/// runs before any state is mutated, so an early `Err` leaves the enclave
147/// untouched and retryable. The mutation then happens entirely in
148/// `commit_operator_init`, which returns `()` — it cannot report an error, so a
149/// half-mutated enclave is never observed via an `Err`.
150/// Validate and commit operator initialization under the cancellation-safe
151/// control lock so concurrent callers cannot race the check-then-commit.
152pub async fn operator_init(
153    enclave: Arc<Enclave>,
154    request: OperatorInitRequest,
155) -> GuardianResult<()> {
156    info!("/operator_init - Received request.");
157
158    let uninitialized = match enclave.mode() {
159        EnclaveMode::Ceremony => CeremonyStage::Uninitialized.into(),
160        EnclaveMode::Withdraw => WithdrawStage::Uninitialized.into(),
161    };
162    enclave.require_lifecycle(uninitialized)?;
163    info!("Lifecycle stage validated.");
164
165    // ---- Validate & build: Nothing in this phase mutates enclave state, so any
166    // error here leaves the enclave untouched. ----
167
168    let (deployment, s3_credentials, withdraw_inputs) = match (enclave.mode(), request) {
169        (
170            EnclaveMode::Ceremony,
171            OperatorInitRequest::Ceremony(CeremonyOperatorInitRequest {
172                deployment,
173                s3_credentials,
174            }),
175        ) => (deployment, s3_credentials, None),
176        (EnclaveMode::Withdraw, OperatorInitRequest::Withdraw(request)) => {
177            let WithdrawOperatorInitRequest {
178                s3_credentials,
179                init_config,
180                genesis_state,
181            } = *request;
182            (
183                init_config.deployment().clone(),
184                s3_credentials,
185                Some((init_config, genesis_state)),
186            )
187        }
188        (EnclaveMode::Ceremony, OperatorInitRequest::Withdraw(_)) => {
189            return Err(InvalidInputs(
190                "ceremony-mode guardian received a withdraw operator-init request".into(),
191            ));
192        }
193        (EnclaveMode::Withdraw, OperatorInitRequest::Ceremony(_)) => {
194            return Err(InvalidInputs(
195                "withdraw-mode guardian received a ceremony operator-init request".into(),
196            ));
197        }
198    };
199    validate_deployment(&enclave, &deployment)?;
200    let attestation = get_attestation(&enclave.signing_pubkey())?;
201    attestation
202        .verify_live(
203            &enclave.signing_pubkey(),
204            deployment.pcr_allowlist.current_build(),
205        )
206        .map_err(|error| InvalidInputs(format!("deployment attestation check failed: {error}")))?;
207    let logger = GuardianS3Client::new(
208        &deployment.bucket_info,
209        deployment.retention_environment,
210        &s3_credentials,
211    )
212    .await?;
213    info!("S3 connectivity check complete.");
214
215    // Build the withdraw-mode install bundle up front; `None` for a ceremony enclave.
216    let withdraw_mode = match withdraw_inputs {
217        Some((config, genesis_state)) => {
218            let mut reader =
219                GuardianReader::from_s3_client(logger.clone(), config.deployment().clone());
220            let ceremony_state = reader.read_latest_ceremony_state().await?;
221            Some(
222                OIWithdrawModeInstall::from_ceremony_state(
223                    &mut reader,
224                    config,
225                    ceremony_state,
226                    genesis_state,
227                )
228                .await?,
229            )
230        }
231        None => None,
232    };
233    let install = OIInstall::new(deployment, attestation, logger, withdraw_mode);
234
235    // ---- All-or-nothing Commit: Nothing in this phase errors out. ----
236    info!("Committing S3 logger and mode-specific initialization state.");
237    commit_operator_init(&enclave, install).await;
238
239    info!("Operator initialization complete.");
240    Ok(())
241}

243/// This precursor retains the compiled revision. The runtime-config follow-up
244/// removes this comparison together with the corresponding build input.
245fn validate_deployment(enclave: &Enclave, deployment: &DeploymentConfig) -> GuardianResult<()> {
246    if deployment.pcr_allowlist.current_build().git_revision() != enclave.reported_git_revision() {
247        return Err(InvalidInputs(
248            "deployment revision does not match the compiled build".into(),
249        ));
250    }
251    Ok(())
252}

254/// Install the validated config on the enclave and write the operator_init logs.
255/// Infallible by design (returns `()`, see the `operator_init` invariant): every
256/// `set` here runs on a fresh enclave under the control lock, and S3 logging
257/// panics on failure rather than returning an error.
258async fn commit_operator_init(enclave: &Enclave, install: OIInstall) {
259    let OIInstall {
260        deployment,
261        attestation,
262        logger,
263        withdraw_mode,
264    } = install;
265
266    enclave
267        .config
268        .set_s3_logger(logger)
269        .expect("Unable to set logger");
270
271    enclave
272        .config
273        .set_deployment(deployment)
274        .expect("deployment is installed once");
275
276    // A ceremony enclave has no withdraw-mode arming state.
277    if let Some(withdraw_mode) = withdraw_mode {
278        withdraw_mode.install_into(enclave);
279    }
280
281    // Log to S3!
282    // 1) Attestation and pub key help authenticate all subsequent enclave-signed messages.
283    let signing_pk = enclave.signing_pubkey();
284    enclave
285        .log_init(OIAttestationUnsigned {
286            attestation,
287            signing_public_key: signing_pk,
288        })
289        .await
290        .expect("S3 logger must be initialized to log the OI attestation");
291
292    // 2) Share commitments help KPs confirm that the right private key will be constructed.
293    // This pre-transition snapshot reports `Uninitialized`; successfully
294    // writing it completes operator initialization.
295    // TODO(testnet-wipe): Replace the full GuardianInfo snapshot with a
296    // purpose-built OI payload containing only the data readers and KPs need;
297    // the evolving status response should not define the durable log schema.
298    enclave
299        .log_init(OIGuardianInfo(Box::new(enclave.info().await)))
300        .await
301        .expect("S3 logger must be initialized to log GuardianInfo");
302
303    let initialized = match enclave.mode() {
304        EnclaveMode::Ceremony => CeremonyStage::OperatorInitialized.into(),
305        EnclaveMode::Withdraw => WithdrawStage::OperatorInitialized.into(),
306    };
307    enclave
308        .advance_lifecycle_into(initialized)
309        .expect("operator_init should advance an uninitialized enclave");
310}

455    /// Run commit_operator_init on a fresh enclave for the given mode (withdraw =>
456    /// carries the InitConfig install bundle; ceremony => none).
457    async fn commit_for_mode(mode: EnclaveMode) -> (Arc<Enclave>, CapturedPuts) {
458        let enclave = Arc::new(Enclave::new(
459            GuardianSignKeyPair::new(rand::thread_rng()),
460            GuardianEncKeyPair::random(&mut rand::thread_rng()),
461            mode,
462        ));
463
464        let (logger, captures) = crate::test_utils::mock_logger_capturing();
465        let (deployment, withdraw_mode) = match mode {
466            EnclaveMode::Withdraw => {
467                let config = InitConfig::mock_for_testing();
468                let args = crate::test_utils::OperatorInitTestArgs::default();
469                (
470                    config.deployment().clone(),
471                    Some(OIWithdrawModeInstall::from_parts(
472                        config,
473                        args.ceremony_state,
474                        None,
475                        args.hashi_object_id,
476                        args.mpc_master_g,
477                    )),
478                )
479            }
480            EnclaveMode::Ceremony => (DeploymentConfig::mock_for_testing(), None),
481        };
482
483        let attestation = get_attestation(&enclave.signing_pubkey()).unwrap();
484        let install = OIInstall::new(deployment, attestation, logger, withdraw_mode);
485        commit_operator_init(&enclave, install).await;
486        (enclave, captures)
487    }

489    fn assert_operator_init_logs(
490        enclave: &Enclave,
491        captures: &CapturedPuts,
492        expected_lifecycle: EnclaveLifecycle,
493    ) {
494        let captured = captures.lock().unwrap();
495        assert_eq!(captured.len(), 2, "operator init should write two records");
496        let session_id = enclave.s3_session_id();
497        assert_eq!(
498            captured[0].0,
499            InitLogMessage::attestation_object_key(&session_id)
500        );
501        assert_eq!(
502            captured[1].0,
503            InitLogMessage::guardian_info_object_key(&session_id)
504        );
505
506        let attestation: LogRecord = serde_json::from_slice(&captured[0].1).unwrap();
507        assert!(matches!(
508            attestation.message(),
509            VersionedLogMessage::V1(LogMessageV1::Init(message))
510                if matches!(message.as_ref(), OIAttestationUnsigned { .. })
511        ));
512
513        let guardian_info: LogRecord = serde_json::from_slice(&captured[1].1).unwrap();
514        let VersionedLogMessage::V1(LogMessageV1::Init(message)) = guardian_info.message() else {
515            panic!("expected V1 init record");
516        };
517        let OIGuardianInfo(info) = message.as_ref() else {
518            panic!("expected operator-init GuardianInfo record");
519        };
520        assert_eq!(info.lifecycle, expected_lifecycle);
521    }
