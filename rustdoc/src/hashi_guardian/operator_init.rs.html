56    pub fn from_parts(
57        init_config: InitConfig,
58        ceremony_state: CeremonyState,
59        genesis_state: Option<GenesisState>,
60        hashi_object_id: hashi_types::sui_sdk_types::Address,
61        mpc_master_g: HashiMasterG,
62    ) -> Self {

64            init_config,
65            ceremony_state,
66            genesis_state,
67            hashi_object_id,
68            mpc_master_g,
69        }
70    }

72    /// Build the arming bundle from the stable config, verified ceremony state,
73    /// and either supplied bootstrap state or verified persisted genesis.
74    async fn from_ceremony_state(
75        reader: &mut GuardianReader,
76        config: InitConfig,
77        ceremony_state: CeremonyState,
78        genesis_state: Option<GenesisState>,
79    ) -> GuardianResult<Self> {
80        // First deployment pins these values for KP authorization during PI.
81        // Subsequent enclaves recover them from the verified immutable record.
82        let (hashi_object_id, mpc_master_g) = match &genesis_state {
83            Some(state) => {
84                if reader.read_latest_committee().await?.is_some() {
85                    return Err(InvalidInputs(
86                        "genesis bootstrap is rejected after a committee-update or genesis record exists"
87                            .into(),
88                    ));
89                }
90                let (_, hashi_object_id, mpc_master_g) = state.clone().into_parts();
91                (hashi_object_id, mpc_master_g)
92            }
93            None => {
94                let genesis = reader.read_genesis().await?.ok_or_else(|| {
95                    InvalidInputs("no genesis record found; genesis bootstrap is required".into())
96                })?;
97                (genesis.hashi_object_id, genesis.mpc_master_g)
98            }
99        };
100
101        Ok(Self::from_parts(
102            config,
103            ceremony_state,
104            genesis_state,
105            hashi_object_id,
106            mpc_master_g,
107        ))
108    }

110    /// Install the bundle onto a fresh enclave. Infallible by design (see the
111    /// `operator_init` invariant): every set runs once on a fresh enclave.
112    pub fn install_into(self, enclave: &mut Enclave) {
113        let config_hash = self.init_config.digest();
114        let limiter_config = *self.init_config.limiter_config();
115
116        info!(
117            "Setting secret-sharing instance: n={}, t={}, {} commitments.",
118            self.ceremony_state.secret_sharing_instance.num_shares(),
119            self.ceremony_state.secret_sharing_instance.threshold(),
120            self.ceremony_state
121                .secret_sharing_instance
122                .commitments()
123                .len()
124        );
125        if let Some(genesis_state) = &self.genesis_state {
126            info!(
127                genesis_state_hash = hex::encode(genesis_state.digest()),
128                "Storing genesis state."
129            );
130        }
131        enclave
132            .state
133            .set_temporary_init_state(TemporaryInitState {
134                ceremony_state: self.ceremony_state,
135                genesis_state: self.genesis_state,
136                config_hash,
137            })
138            .expect("Unable to set temporary initialization state");
139
140        info!("Setting withdraw configuration.");
141        enclave
142            .config
143            .install_config(self.mpc_master_g, limiter_config, self.hashi_object_id)
144            .expect("Unable to set enclave configuration");
145    }
146}

148/// Receives S3 API keys and mode-specific configuration. A ceremony enclave
149/// installs the shared deployment policy; a withdraw enclave additionally
150/// installs the stable `InitConfig`, arming state, and fixed `config_hash`.
151///
152/// Invariant: operator_init never returns an `Err` from a partially-initialized
153/// enclave. Every fallible preparation step (validation, attestation, S3 access)
154/// runs before any state is mutated, so an early `Err` leaves the enclave
155/// untouched and retryable. The mutation then happens entirely in
156/// `commit_operator_init`, which returns `()` — it cannot report an error, so a
157/// half-mutated enclave is never observed via an `Err`.
158/// Validate and commit operator initialization under the cancellation-safe
159/// control lock so concurrent callers cannot race the check-then-commit.
160pub async fn operator_init(
161    enclave: &mut Enclave,
162    request: OperatorInitRequest,
163) -> GuardianResult<()> {
164    info!("/operator_init - Received request.");
165
166    enclave.require_lifecycle(None)?;
167
168    // ---- Validate & build: Nothing in this phase mutates enclave state, so any
169    // error here leaves the enclave untouched. ----
170
171    let (deployment, s3_credentials, withdraw_inputs) = match request {
172        OperatorInitRequest::Ceremony(CeremonyOperatorInitRequest {
173            deployment,
174            s3_credentials,
175        }) => (deployment, s3_credentials, None),
176        OperatorInitRequest::Withdraw(request) => {
177            let WithdrawOperatorInitRequest {
178                s3_credentials,
179                init_config,
180                genesis_state,
181            } = *request;
182            (
183                init_config.deployment().clone(),
184                s3_credentials,
185                Some((init_config, genesis_state)),
186            )
187        }
188    };
189    let attestation = get_attestation(&enclave.config.signing_pubkey())?;
190    attestation
191        .verify_live(
192            &enclave.config.signing_pubkey(),
193            deployment.pcr_allowlist.current_build(),
194        )
195        .map_err(|error| InvalidInputs(format!("deployment attestation check failed: {error}")))?;
196    let logger = GuardianS3Client::new_in_enclave(
197        &deployment.bucket_info,
198        deployment.retention_environment,
199        &s3_credentials,
200    )
201    .await?;
202    info!("S3 connectivity check complete.");
203
204    // Build the withdraw-mode install bundle up front; `None` for a ceremony enclave.
205    let withdraw_mode = match withdraw_inputs {
206        Some((config, genesis_state)) => {
207            let mut reader =
208                GuardianReader::from_s3_client(logger.clone(), config.deployment().clone());
209            let ceremony_state = reader.read_latest_ceremony_state().await?;
210            Some(
211                OIWithdrawModeInstall::from_ceremony_state(
212                    &mut reader,
213                    config,
214                    ceremony_state,
215                    genesis_state,
216                )
217                .await?,
218            )
219        }
220        None => None,
221    };
222    let install = OIInstall::new(deployment, attestation, logger, withdraw_mode);
223
224    // ---- All-or-nothing Commit: Nothing in this phase errors out. ----
225    info!("Committing S3 logger and mode-specific initialization state.");
226    commit_operator_init(enclave, install).await;
227
228    info!("Operator initialization complete.");
229    Ok(())
230}

232/// Install the validated config on the enclave and write the operator_init logs.
233/// Infallible by design (returns `()`, see the `operator_init` invariant): every
234/// `set` here runs on a fresh enclave under the control lock, and S3 logging
235/// panics on failure rather than returning an error.
236async fn commit_operator_init(enclave: &mut Enclave, install: OIInstall) {
237    let OIInstall {
238        deployment,
239        attestation,
240        logger,
241        withdraw_mode,
242    } = install;
243
244    let oi_info = OperatorInitInfo {
245        deployment: deployment.clone(),
246        encryption_pubkey: enclave.config.encryption_public_key().to_bytes().to_vec(),
247        mode: match &withdraw_mode {
248            None => OperatorInitMode::Ceremony,
249            Some(withdraw) => OperatorInitMode::Withdraw(Box::new(WithdrawOperatorInitInfo {
250                secret_sharing_instance: withdraw.ceremony_state.secret_sharing_instance.clone(),
251                config_hash: withdraw.init_config.digest(),
252                limiter_config: *withdraw.init_config.limiter_config(),
253                hashi_object_id: withdraw.hashi_object_id,
254                mpc_master_g: withdraw.mpc_master_g,
255                genesis_state_hash: withdraw.genesis_state.as_ref().map(GenesisState::digest),
256            })),
257        },
258    };
259
260    enclave
261        .config
262        .set_s3_logger(logger)
263        .expect("Unable to set logger");
264
265    enclave
266        .config
267        .set_deployment(deployment)
268        .expect("deployment is installed once");
269
270    let initialized = if withdraw_mode.is_some() {
271        WithdrawStage::OperatorInitialized.into()
272    } else {
273        CeremonyStage::OperatorInitialized.into()
274    };
275
276    // A ceremony enclave has no withdraw-mode arming state.
277    if let Some(withdraw_mode) = withdraw_mode {
278        withdraw_mode.install_into(enclave);
279    }
280
281    // Log to S3!
282    // 1) Attest the session key that signs this record and all subsequent logs.
283    let signing_pk = enclave.config.signing_pubkey();
284    enclave
285        .log_init(OIAttestation {
286            attestation,
287            signing_public_key: signing_pk,
288        })
289        .await
290        .expect("S3 logger must be initialized to log the OI attestation");
291
292    // 2) Share commitments help KPs confirm that the right private key will be constructed.
293    // Successfully writing this record completes operator initialization before
294    // the live lifecycle advances. Its schema contains only durable OI facts.
295    enclave
296        .log_init(OIGuardianInfo(Box::new(oi_info)))
297        .await
298        .expect("S3 logger must be initialized to log operator initialization");
299
300    enclave
301        .advance_lifecycle_into(initialized)
302        .expect("operator_init should advance an uninitialized enclave");
303}

310    fn genesis_record(state: GenesisState, key: &GuardianSignKeyPair) -> SignedLogEntry {
311        let (committee, hashi_object_id, mpc_master_g) = state.into_parts();
312        SignedLogEntry::new(
313            SessionID::from_signing_pubkey(&key.verification_key()),
314            LogMessage::Genesis(Box::new(GenesisLogMessage {
315                committee,
316                hashi_object_id,
317                mpc_master_g,
318            })),
319            key,
320        )
321    }

437    /// Run commit_operator_init on a fresh enclave for the given mode (withdraw =>
438    /// carries the InitConfig install bundle; ceremony => none).
439    async fn commit_for_mode(mode: EnclaveMode) -> (Enclave, CapturedPuts) {
440        let mut enclave = Enclave::new(
441            GuardianSignKeyPair::new(rand::thread_rng()),
442            GuardianEncKeyPair::random(&mut rand::thread_rng()),
443        );
444
445        let (logger, captures) = crate::test_utils::mock_logger_capturing();
446        let (deployment, withdraw_mode) = match mode {
447            EnclaveMode::Withdraw => {
448                let config = InitConfig::mock_for_testing();
449                let args = crate::test_utils::OperatorInitTestArgs::default();
450                (
451                    config.deployment().clone(),
452                    Some(OIWithdrawModeInstall::from_parts(
453                        config,
454                        args.ceremony_state,
455                        None,
456                        args.hashi_object_id,
457                        args.mpc_master_g,
458                    )),
459                )
460            }
461            EnclaveMode::Ceremony => (DeploymentConfig::mock_for_testing(), None),
462        };
463
464        let attestation = get_attestation(&enclave.config.signing_pubkey()).unwrap();
465        let install = OIInstall::new(deployment, attestation, logger, withdraw_mode);
466        commit_operator_init(&mut enclave, install).await;
467        (enclave, captures)
468    }

470    async fn assert_operator_init_logs(
471        enclave: &Enclave,
472        captures: &CapturedPuts,
473        expected_mode: EnclaveMode,
474    ) {
475        let live_info = enclave.info();
476        let state = &enclave.state;
477        let temporary_init_state = state.temporary_init_state().ok();
478        let captured = captures.lock().unwrap();
479        assert_eq!(captured.len(), 2, "operator init should write two records");
480        let session_id = enclave.config.s3_session_id();
481        assert_eq!(
482            captured[0].0,
483            InitLogMessage::attestation_object_key(&session_id)
484        );
485        assert_eq!(
486            captured[1].0,
487            InitLogMessage::guardian_info_object_key(&session_id)
488        );
489
490        let attestation: SignedLogEntry = serde_json::from_slice(&captured[0].1).unwrap();
491        assert!(matches!(
492            attestation.message_unchecked(),
493            VersionedLogMessage::V1(LogMessageV1::Init(message))
494                if matches!(message.as_ref(), OIAttestation { .. })
495        ));
496
497        let guardian_info: SignedLogEntry = serde_json::from_slice(&captured[1].1).unwrap();
498        let VersionedLogMessage::V1(LogMessageV1::Init(message)) =
499            guardian_info.message_unchecked()
500        else {
501            panic!("expected V1 init record");
502        };
503        let OIGuardianInfo(info) = message.as_ref() else {
504            panic!("expected operator-init completion record");
505        };
506        assert_eq!(info.mode(), expected_mode);
507        assert_eq!(&info.deployment, enclave.config.deployment().unwrap());
508        assert_eq!(
509            info.encryption_pubkey,
510            enclave.config.encryption_public_key().to_bytes().to_vec()
511        );
512        if let OperatorInitMode::Withdraw(withdraw) = &info.mode {
513            let state = temporary_init_state.unwrap();
514            assert_eq!(
515                withdraw.secret_sharing_instance,
516                state.ceremony_state.secret_sharing_instance
517            );
518            assert_eq!(withdraw.config_hash, state.config_hash);
519            assert_eq!(
520                withdraw.genesis_state_hash,
521                state.genesis_state.as_ref().map(GenesisState::digest)
522            );
523            assert_eq!(
524                withdraw.limiter_config,
525                enclave.config.limiter_config().unwrap()
526            );
527            assert_eq!(Some(withdraw.hashi_object_id), live_info.hashi_object_id);
528            assert_eq!(Some(withdraw.mpc_master_g), live_info.mpc_master_g);
529        }
530        guardian_info
531            .validate(&enclave.config.signing_pubkey())
532            .unwrap();
533    }
