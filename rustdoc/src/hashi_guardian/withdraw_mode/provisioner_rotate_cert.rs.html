4//! Replace the authenticated KP signer's sole certificate. The live guardian
5//! verifies that the caller submitted its currently committed share, then
6//! appends a complete `kp-shares/` snapshot with only that share re-encrypted.
7

24pub async fn provisioner_rotate_cert(
25    enclave: Arc<Enclave>,
26    signed_request: KpSigned<ProvisionerRotateCertRequest>,
27) -> GuardianResult<GuardianSignedResponse<ProvisionerRotateCertResponse>> {
28    info!("/provisioner_rotate_cert - Received request.");
29
30    enclave.require_fully_initialized()?;
31
32    let signer_fingerprint = signed_request.signer_fingerprint().to_hex();
33    let request = signed_request
34        .verify_into_data()
35        .map_err(|error| Unauthenticated(error.to_string()))?;
36
37    let live_session_id = enclave.s3_session_id();
38    request.validate_session(&live_session_id)?;
39
40    let mut reader = enclave.new_guardian_reader()?;
41    let latest_state = reader.read_latest_ceremony_state().await?;
42    apply_cert_rotation(&enclave, signer_fingerprint, request, latest_state).await
43}
44
45async fn apply_cert_rotation(
46    enclave: &Enclave,
47    signer_fingerprint: String,
48    request: ProvisionerRotateCertRequest,
49    latest_state: CeremonyState,
50) -> GuardianResult<GuardianSignedResponse<ProvisionerRotateCertResponse>> {
51    let (_, expected_cert_seq, new_kp_pgp_cert, encrypted_share) = request.into_parts();
52    let share_id = encrypted_share.id;
53    let new_recipient_fingerprint = new_kp_pgp_cert.fingerprint().to_hex();
54
55    let enclave_btc_pubkey = enclave.config.enclave_btc_pubkey()?;
56    if latest_state.btc_master_pubkey != enclave_btc_pubkey {
57        return Err(InvalidInputs(format!(
58            "latest ceremony BTC pubkey differs from initialized enclave BTC pubkey: latest \
59             {:?}, initialized {enclave_btc_pubkey:?}",
60            latest_state.btc_master_pubkey
61        )));
62    }
63    let sharing_seq = latest_state.secret_sharing_instance.sharing_seq();
64
65    if latest_state.cert_seq != expected_cert_seq {
66        return Err(InvalidInputs(format!(
67            "provisioner_rotate_cert request expected cert_seq {expected_cert_seq}, latest cert_seq \
68             is {}",
69            latest_state.cert_seq
70        )));
71    }
72
73    let latest_instance = latest_state.secret_sharing_instance;
74    let encrypted_shares = latest_state.encrypted_shares;
75    encrypted_shares.validate_share_assignment(&signer_fingerprint, share_id)?;
76    let next_cert_seq = latest_state
77        .cert_seq
78        .checked_add(1)
79        .ok_or_else(|| InvalidInputs("cert_seq overflow".into()))?;
80
81    // The KP signature binds the ciphertext and the rest of this request to the
82    // current session and cert sequence, so no additional HPKE AAD is needed.
83    let share = decrypt_share(&encrypted_share, enclave.encryption_secret_key(), None)?;
84    latest_instance.commitments().verify_share(&share)?;
85
86    let replacement_ciphertext = encrypt_share_for_provisioner(&share, &new_kp_pgp_cert);
87    let (encrypted_shares, changed_entry) = encrypted_shares.replace_recipient(
88        &signer_fingerprint,
89        new_recipient_fingerprint.clone(),
90        replacement_ciphertext,
91    )?;
92
93    enclave
94        .log_kp_share_state(sharing_seq, next_cert_seq, encrypted_shares)
95        .await?;
96
97    info!(
98        sharing_seq,
99        cert_seq = next_cert_seq,
100        share_id = share_id.get(),
101        signer_fingerprint = %signer_fingerprint,
102        new_fingerprint = %new_recipient_fingerprint,
103        "KP certificate rotation complete",
104    );
105
106    Ok(enclave.sign(ProvisionerRotateCertResponse {
107        cert_seq: next_cert_seq,
108        encrypted_share: changed_entry,
109    }))
110}

139    const TEST_N: usize = 5;

143    fn signed_rotate_request(
144        enclave: &Arc<Enclave>,
145        new_cert: PgpPublicCert,
146        share: &Share,
147        signer_cert: &PgpPublicCert,
148        signer_secret: &str,
149    ) -> KpSigned<ProvisionerRotateCertRequest> {
150        let request = ProvisionerRotateCertRequest::new(
151            enclave.s3_session_id(),
152            INITIAL_CERT_SEQ,
153            new_cert,
154            share,
155            enclave.encryption_public_key(),
156            &mut rand::thread_rng(),
157        );
158        let signature = sign_detached_in_process(signer_secret, &KpSigned::signed_bytes(&request));
159        KpSigned::from_parts(request, signer_cert.clone(), signature)
160    }

207    #[tokio::test]
208    async fn rotates_only_authenticated_signers_share_and_rejects_another_share() {
209        let (enclave, shares, ceremony_state, mut secret_keys, captures, cert_roster) =
210            initialized_rotation_context().await;
211        let old_encrypted_shares = ceremony_state.encrypted_shares.clone();
212        let original_commitments = ceremony_state.secret_sharing_instance.commitments().clone();
213        let btc_master_pubkey = ceremony_state.btc_master_pubkey;
214        let signer_cert = cert_roster.cert_for_share(shares[0].id).unwrap();
215        let signer_fingerprint = signer_cert.fingerprint().to_hex();
216        let signer_secret = secret_keys.get(&signer_fingerprint).unwrap();
217        let (new_public, new_secret) = mock_pgp_keypair();
218        let new_cert = PgpPublicCert::new(new_public).unwrap();
219        let new_fingerprint = new_cert.fingerprint().to_hex();
220
221        let signed_wrong_share = signed_rotate_request(
222            &enclave,
223            new_cert.clone(),
224            &shares[1],
225            signer_cert,
226            signer_secret,
227        );
228        let wrong_signer_fingerprint = signed_wrong_share.signer_fingerprint().to_hex();
229        let wrong_share_request = signed_wrong_share
230            .verify_into_data()
231            .expect("old certificate should authenticate the request");
232        let error = apply_cert_rotation(
233            &enclave,
234            wrong_signer_fingerprint,
235            wrong_share_request,
236            ceremony_state.clone(),
237        )
238        .await
239        .expect_err("a signer must not rotate another provisioner's share");
240        assert!(matches!(error, InvalidInputs(_)));
241        assert!(captures.lock().unwrap().is_empty());
242
243        let signed_request =
244            signed_rotate_request(&enclave, new_cert, &shares[0], signer_cert, signer_secret);
245        let authenticated_fingerprint = signed_request.signer_fingerprint().to_hex();
246        let request = signed_request
247            .verify_into_data()
248            .expect("old certificate should authenticate the request");
249        let signed_response =
250            apply_cert_rotation(&enclave, authenticated_fingerprint, request, ceremony_state)
251                .await
252                .expect("the committed recipient should rotate its sole certificate");
253        let response = signed_response
254            .verify_into_data(&enclave.signing_pubkey())
255            .expect("rotation response should be signed by the enclave")
256            .response;
257
258        assert_eq!(response.cert_seq, INITIAL_CERT_SEQ + 1);
259        assert_eq!(response.encrypted_share.id, shares[0].id);
260        assert_eq!(
261            response.encrypted_share.recipient_fingerprint,
262            new_fingerprint
263        );
264        assert!(response
265            .encrypted_share
266            .armored_ciphertext
267            .starts_with("-----BEGIN PGP MESSAGE-----"));
268
269        let captured = captures.lock().unwrap();
270        assert_eq!(captured.len(), 1);
271        assert!(captured[0]
272            .0
273            .starts_with("kp-shares/00000000000000000000/00000000000000000008-"));
274        let record: LogRecord = serde_json::from_slice(&captured[0].1).unwrap();
275        let VersionedLogMessage::V2(LogMessageV2::KpShareState(persisted)) = record.message()
276        else {
277            panic!("certificate rotation should persist a V2 KP-share snapshot");
278        };
279        assert_eq!(persisted.sharing_seq, 0);
280        assert_eq!(persisted.cert_seq, INITIAL_CERT_SEQ + 1);
281        let (expected_roster, expected_changed_entry) = old_encrypted_shares
282            .replace_recipient(
283                &signer_fingerprint,
284                new_fingerprint.clone(),
285                response.encrypted_share.armored_ciphertext.clone(),
286            )
287            .unwrap();
288        assert_eq!(response.encrypted_share, expected_changed_entry);
289        assert_eq!(persisted.encrypted_shares, expected_roster);
290
291        secret_keys.insert(new_fingerprint, new_secret);
292        let decrypted = decrypt_kp_shares(&persisted.encrypted_shares, &secret_keys);
293        for (actual, expected) in decrypted.iter().zip(&shares) {
294            assert_eq!(actual.id, expected.id);
295            assert_eq!(actual.value, expected.value);
296        }
297        assert_eq!(
298            original_commitments,
299            ShareCommitments::from_shares(&decrypted).unwrap()
300        );
301        assert_eq!(
302            enclave.config.enclave_btc_pubkey().unwrap(),
303            btc_master_pubkey
304        );
305    }

308    async fn rejects_wrong_lifecycle_before_verifying_signature() {
309        let enclave = Enclave::create_with_random_keys();
310        let request = ProvisionerRotateCertRequest::from_encrypted_share_for_testing(
311            "mock-session".into(),
312            0,
313            mock_pgp_cert(),
314            GuardianEncryptedShare {
315                id: NonZeroU16::new(1).unwrap(),
316                ciphertext: Ciphertext {
317                    encapsulated_key: vec![],
318                    aes_ciphertext: vec![],
319                },
320            },
321        );
322        let signed_request =
323            KpSigned::from_parts(request, mock_pgp_cert(), "invalid signature".into());

325        let err = provisioner_rotate_cert(enclave, signed_request)
326            .await
327            .expect_err("wrong lifecycle should be rejected first");

329        assert!(matches!(err, LifecycleMismatch { .. }));
