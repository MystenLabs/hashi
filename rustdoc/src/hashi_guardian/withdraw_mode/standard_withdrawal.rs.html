19const MAX_CLOCK_SKEW_SECS: u64 = 5 * 60;
20// Requests are minted per attempt and should not remain usable indefinitely.
21const MAX_REQUEST_AGE_SECS: u64 = 30 * 60;
22
23pub async fn standard_withdrawal(
24    enclave: &mut Enclave,
25    signed_request: SignedStandardWithdrawalRequestWire,
26) -> GuardianResult<GuardianSignedResponse<StandardWithdrawalResponse>> {
27    info!("/standard_withdrawal - Received request.");
28
29    let network = enclave.config.bitcoin_network()?;
30    let signed_request =
31        HashiSigned::<StandardWithdrawalRequest>::validate_addr(signed_request, network)?;
32    let wid = *signed_request.message().wid();
33    // 0) Validation
34    enclave.require_fully_initialized()?;
35
36    // 1) Verify certificate
37    let committee = enclave.state.get_committee()?;
38
39    info!("Verifying request certificate.");
40    verify_hashi_cert(
41        enclave.config.hashi_object_id()?,
42        committee,
43        &signed_request,
44    )?;
45    info!("Request certificate verified.");
46
47    let (request_sign, request) = signed_request.into_parts();
48
49    // 2) Rate limits: consume tokens. The control lock keeps this consumption
50    //    exclusive until it is durably logged or the enclave aborts.
51    validate_request_timestamp(request.timestamp_secs(), now_timestamp_secs())?;
52
53    info!("Checking rate limits.");
54    // Gross outflow (= inputs - change = external_out + miner_fee).
55    // Miner fee leaves the pool too, so it must consume the limit;
56    // change flows back, so it must not.
57    let consumed_amount_sats = request.utxos().gross_outflow_amount().to_sat();
58    let post_state = enclave.state.consume_from_limiter(
59        request.seq(),
60        request.timestamp_secs(),
61        consumed_amount_sats,
62    )?;
63    info!("Rate limit check passed.");
64
65    // 3) Sign tx (while holding the control lock)
66    info!("Generating BTC signatures.");
67    let (txid, signatures) = enclave
68        .config
69        .btc_sign(request.utxos())
70        .expect("All BTC keys should be set");
71    let response = StandardWithdrawalResponse {
72        enclave_signatures: signatures,
73    };
74    info!("BTC signatures generated.");
75
76    // 4) Log while holding the control lock, before returning signatures.
77    info!("Withdrawal {} processed successfully. Logging to S3.", wid);
78    let msg = WithdrawalLogMessage {
79        txid,
80        request_data: StandardWithdrawalRequestWire::from(request),
81        request_sign,
82        response: response.clone(),
83        post_state,
84    };
85    enclave
86        .log_withdraw(msg)
87        .await
88        .expect("S3 logger must be initialized to log a withdrawal");
89    info!("Withdrawal {} logged.", wid);
90    Ok(enclave.sign(response))
91}
92
93fn validate_request_timestamp(
94    request_timestamp_secs: u64,
95    guardian_now: u64,
96) -> GuardianResult<()> {
97    // Ensure that a compromised hashi committee cannot set an arbitrary time
98    // in the future, which in turn allows bypassing the limiter completely.
99    if request_timestamp_secs > guardian_now + MAX_CLOCK_SKEW_SECS {
100        return Err(InvalidInputs(format!(
101            "request timestamp {} is too far in the future (guardian clock: {})",
102            request_timestamp_secs, guardian_now
103        )));
104    }
105
106    // This check is useful for situations where there is a big time gap between
107    // consecutive withdrawals. A compromised hashi committee can exploit that
108    // gap by carefully choosing timestamps until gap * refill_rate is extracted.
109    if guardian_now.saturating_sub(request_timestamp_secs) > MAX_REQUEST_AGE_SECS {
110        return Err(InvalidInputs(format!(
111            "request timestamp {} is too old (guardian clock: {}, maximum age: {} seconds)",
112            request_timestamp_secs, guardian_now, MAX_REQUEST_AGE_SECS
113        )));
114    }
115
116    Ok(())
117}

141    fn wire(
142        request: HashiSigned<StandardWithdrawalRequest>,
143    ) -> SignedStandardWithdrawalRequestWire {
144        let (signature, request) = request.into_parts();
145        SignedStandardWithdrawalRequestWire {
146            data: request.into(),
147            signature: hashi_types::move_types::CommitteeSignature {
148                epoch: signature.epoch(),
149                signature: signature.signature_bytes().to_vec(),
150                signers_bitmap: signature.signers_bitmap_bytes().to_vec(),
151            },
152        }
153    }

179            OperatorInitTestArgs::default()
180                .with_s3_logger(logger)
181                .with_config(config)
182                .with_genesis_bindings(
183                    hashi_types::guardian::test_utils::TEST_HASHI_OBJECT_ID,
184                    hashi_btc_master_pubkey,
185                ),
186        );
187
188        // The reconstructed BTC keypair (set by provisioner_init in production).
189        enclave
190            .config
191            .set_btc_keypair(
192                BitcoinKeypair::from_seckey_slice(&BTC_LIB, &[8u8; 32])
193                    .expect("valid test secret key"),
194            )
195            .unwrap();
196
197        enclave
198            .advance_lifecycle_into(WithdrawStage::ProvisionerInitialized.into())
199            .expect("test setup should advance provisioner init lifecycle");

201            .expect("activate_enclave_for_testing should succeed on a fresh enclave");

203        assert!(enclave.require_fully_initialized().is_ok());
204        (enclave, captures)
205    }

276    #[tokio::test]
277    async fn test_standard_withdrawal_rate_limit_exceeded() {
278        let timestamp_secs = now_timestamp_secs();
279        let (req1, committee) = StandardWithdrawalRequest::mock_signed_and_committee_with_seq(
280            Network::Regtest,
281            WithdrawalID::new([0x01; 32]),
282            timestamp_secs,
283            0,
284        );
285        let amount_sats = req1.message().utxos().gross_outflow_amount().to_sat();
286        // Bucket capacity == one withdrawal, so second will be rejected.
287        let (mut enclave, captures) =
288            setup_fully_initialized_enclave(Network::Regtest, committee, amount_sats).await;
289
290        let first = standard_withdrawal(&mut enclave, wire(req1)).await;
291        assert!(first.is_ok());
292
293        // Second withdrawal with seq=1 and later timestamp — bucket is empty, no refill (rate=0).
294        let (req2, _) = StandardWithdrawalRequest::mock_signed_and_committee_with_seq(
295            Network::Regtest,
296            WithdrawalID::new([0x02; 32]),
297            timestamp_secs + 1,
298            1,
299        );
300        let second = standard_withdrawal(&mut enclave, wire(req2)).await;
301        assert!(matches!(
302            second.unwrap_err(),
303            GuardianError::RateLimitExceeded
304        ));
305
306        let captured = captures.lock().unwrap();
307        assert_eq!(
308            captured.len(),
309            1,
310            "only successful withdrawals should be logged"
311        );
312        let success: SignedLogEntry = serde_json::from_slice(&captured[0].1).unwrap();
313        assert_eq!(captured[0].0, success.object_key());
314        let VersionedLogMessage::V1(LogMessageV1::Withdrawal(message)) =
315            success.message_unchecked()
316        else {
317            panic!("expected V1 withdrawal record");
318        };
319        let WithdrawalLogMessage {
320            request_data,
321            post_state,
322            ..
323        } = message.as_ref();
324        assert_eq!(request_data.seq, 0);
325        assert_eq!(post_state.next_seq, 1);
326        assert_eq!(post_state.num_tokens_available, 0);
327    }

329    #[test]
330    fn test_request_timestamp_bounds() {
331        const GUARDIAN_NOW: u64 = 1_000_000;
332
333        assert!(
334            validate_request_timestamp(GUARDIAN_NOW - MAX_REQUEST_AGE_SECS, GUARDIAN_NOW).is_ok()
335        );
336        assert!(
337            validate_request_timestamp(GUARDIAN_NOW + MAX_CLOCK_SKEW_SECS, GUARDIAN_NOW).is_ok()
338        );
339
340        let too_old =
341            validate_request_timestamp(GUARDIAN_NOW - MAX_REQUEST_AGE_SECS - 1, GUARDIAN_NOW);
342        assert!(matches!(too_old, Err(InvalidInputs(message)) if message.contains("too old")));
343
344        let too_far_in_the_future =
345            validate_request_timestamp(GUARDIAN_NOW + MAX_CLOCK_SKEW_SECS + 1, GUARDIAN_NOW);
346        assert!(matches!(
347            too_far_in_the_future,
348            Err(InvalidInputs(message)) if message.contains("too far in the future")
349        ));
350    }
