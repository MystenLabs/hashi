32fn to_status(e: GuardianError) -> Status {
33    match e {
34        InvalidInputs(msg) => Status::invalid_argument(msg),
35        Unauthenticated(msg) => Status::unauthenticated(msg),
36        BuildNotAllowlisted(msg) | BuildNotCurrent(msg) => Status::failed_precondition(msg),
37        LifecycleMismatch { expected, actual } => Status::failed_precondition(format!(
38            "expected enclave lifecycle {expected:?}, but enclave is {actual:?}"
39        )),
40        // Guardian reserves `Aborted` and `ResourceExhausted` exclusively for
41        // limiter sequence mismatches and rate limiting, respectively, so
42        // Hashi can classify them without parsing error messages.
43        LimiterSequenceMismatch { expected, actual } => Status::aborted(format!(
44            "limiter sequence mismatch: expected {expected}, got {actual}"
45        )),
46        CurrentSessionHeartbeatNotLive {
47            session_id,
48            heartbeat_age_secs,
49            retry_after_secs,
50        } => {
51            let detail = match heartbeat_age_secs {
52                Some(heartbeat_age_secs) => {
53                    format!("last heartbeated {heartbeat_age_secs}s ago")
54                }
55                None => "has no heartbeat in the recent scan".into(),
56            };
57            Status::unavailable(format!(
58                "operator_activate blocked: current guardian session {session_id} {detail}; \
59                 expected a heartbeat within {}s; retry in {retry_after_secs}s",
60                crate::LIVE_SESSION_LATEST_HEARTBEAT_MAX_AGE.as_secs()
61            ))
62        }
63        PriorSessionHeartbeatStillRecent {
64            session_id,
65            heartbeat_age_secs,
66            required_quiet_secs,
67        } => Status::unavailable(format!(
68            "operator_activate blocked: prior guardian session {session_id} heartbeated \
69             {heartbeat_age_secs}s ago; required quiet period is {required_quiet_secs}s; retry in \
70             {}s",
71            required_quiet_secs.saturating_sub(heartbeat_age_secs)
72        )),
73        RateLimitExceeded => Status::resource_exhausted("Rate limit exceeded"),
74        S3Error(msg) => Status::internal(msg),
75        InvalidS3Log(msg) => Status::internal(msg),
76        InternalError(msg) => Status::internal(msg),
77        Unavailable(msg) => Status::unavailable(msg),
78    }
79}

81#[tonic::async_trait]
82impl proto::guardian_service_server::GuardianService for GuardianGrpc {
83    async fn get_guardian_info(
84        &self,
85        _request: Request<proto::GetGuardianInfoRequest>,
86    ) -> anyhow::Result<Response<proto::GetGuardianInfoResponse>, Status> {
87        let resp = info::get_guardian_info(self.enclave.clone())
88            .await
89            .map_err(to_status)?;
90
91        let resp_pb = proto_conversions::get_guardian_info_response_to_pb(resp);
92
93        Ok(Response::new(resp_pb))
94    }
95
96    async fn setup_new_key(
97        &self,
98        request: Request<proto::SetupNewKeyRequest>,
99    ) -> anyhow::Result<Response<proto::SignedSetupNewKeyResponse>, Status> {
100        let domain_req: SetupNewKeyRequest = request.into_inner().try_into().map_err(to_status)?;
101
102        let signed = task_spawner::setup_new_key(self.enclave.clone(), domain_req)
103            .await
104            .map_err(to_status)?;
105
106        let resp = proto_conversions::setup_new_key_response_signed_to_pb(signed);
107
108        Ok(Response::new(resp))
109    }

182    async fn standard_withdrawal(
183        &self,
184        request: Request<proto::SignedStandardWithdrawalRequest>,
185    ) -> Result<Response<proto::SignedStandardWithdrawalResponse>, Status> {
186        // proto to domain
187        let domain_req = SignedStandardWithdrawalRequestWire::try_from(request.into_inner())
188            .map_err(to_status)?;
189
190        // validate address with network
191        let network = self.enclave.config.bitcoin_network().map_err(to_status)?;
192        let validated_req =
193            HashiSigned::<StandardWithdrawalRequest>::validate_addr(domain_req, network)
194                .map_err(to_status)?;
195
196        // core withdraw call
197        let response = task_spawner::standard_withdrawal(self.enclave.clone(), validated_req)
198            .await
199            .map_err(to_status)?;
200
201        // domain to proto
202        let resp_pb = proto_conversions::standard_withdrawal_response_signed_to_pb(response);
203        Ok(Response::new(resp_pb))
204    }
205
206    async fn update_committee(
207        &self,
208        request: Request<proto::SignedCommitteeTransition>,
209    ) -> Result<Response<proto::UpdateCommitteeResponse>, Status> {
210        let signed = HashiSigned::<CommitteeTransitionRequest>::try_from(request.into_inner())
211            .map_err(to_status)?;
212        let current_committee_epoch = task_spawner::update_committee(self.enclave.clone(), signed)
213            .await
214            .map_err(to_status)?;
215
216        Ok(Response::new(proto::UpdateCommitteeResponse {
217            current_committee_epoch: Some(current_committee_epoch),
218        }))
219    }
220
221    async fn update_committee_chain(
222        &self,
223        request: Request<proto::UpdateCommitteeChainRequest>,
224    ) -> Result<Response<proto::UpdateCommitteeResponse>, Status> {
225        let transitions = request
226            .into_inner()
227            .transitions
228            .into_iter()
229            .map(HashiSigned::<CommitteeTransitionRequest>::try_from)
230            .collect::<Result<Vec<_>, _>>()
231            .map_err(to_status)?;
232        let current_committee_epoch =
233            task_spawner::update_committee_chain(self.enclave.clone(), transitions)
234                .await
235                .map_err(to_status)?;
236
237        Ok(Response::new(proto::UpdateCommitteeResponse {
238            current_committee_epoch: Some(current_committee_epoch),
239        }))
240    }
241}

243#[cfg(test)]
244mod tests {
245    use super::*;
246
247    #[test]
248    fn heartbeat_readiness_errors_have_actionable_status_codes() {
249        let missing_error = CurrentSessionHeartbeatNotLive {
250            session_id: "current".into(),
251            heartbeat_age_secs: None,
252            retry_after_secs: 60,
253        };
254        assert_eq!(missing_error.retry_after_secs(), Some(60));
255        let missing = to_status(missing_error);
256        assert_eq!(missing.code(), tonic::Code::Unavailable);
257
258        let stale_error = CurrentSessionHeartbeatNotLive {
259            session_id: "current".into(),
260            heartbeat_age_secs: Some(200),
261            retry_after_secs: 60,
262        };
263        assert_eq!(stale_error.retry_after_secs(), Some(60));
264        let stale = to_status(stale_error);
265        assert_eq!(stale.code(), tonic::Code::Unavailable);
266
267        let prior_error = PriorSessionHeartbeatStillRecent {
268            session_id: "prior".into(),
269            heartbeat_age_secs: 30,
270            required_quiet_secs: 600,
271        };
272        assert_eq!(prior_error.retry_after_secs(), Some(570));
273        let prior = to_status(prior_error);
274        assert_eq!(prior.code(), tonic::Code::Unavailable);
275    }
