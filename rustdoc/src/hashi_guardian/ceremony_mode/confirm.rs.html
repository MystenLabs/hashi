15pub async fn confirm_ceremony(
16    enclave: Arc<Enclave>,
17    signed: KpSigned<CeremonyConfirmationRequest>,
18) -> GuardianResult<CeremonyConfirmationResponse> {
19    let lifecycle = enclave.lifecycle();
20    if lifecycle != CeremonyStage::AwaitingKeyProvisionerConfirmations.into()
21        && lifecycle != CeremonyStage::Completed.into()
22    {
23        enclave.require_lifecycle(CeremonyStage::AwaitingKeyProvisionerConfirmations.into())?;
24    }
25
26    let pending = enclave.pending_ceremony()?;
27    let signer_fingerprint = signed.signer_fingerprint().to_hex();
28    let request = signed
29        .verify_signature()
30        .map_err(|error| GuardianError::Unauthenticated(error.to_string()))?;
31    request.validate_session(&enclave.s3_session_id())?;
32    let (share_id, already_confirmed) =
33        pending.validate_confirmation(&signer_fingerprint, request.ceremony_digest())?;
34    if already_confirmed {
35        return pending.status();
36    }
37
38    let status = pending.record_confirmation(share_id)?;
39    info!(
40        share_id = share_id.get(),
41        signer_fingerprint,
42        have = status.have,
43        need = status.need,
44        "Accepted key provisioner ceremony confirmation."
45    );
46
47    if status.completed && lifecycle == CeremonyStage::AwaitingKeyProvisionerConfirmations.into() {
48        enclave
49            .advance_lifecycle_into(CeremonyStage::Completed.into())
50            .expect("all KP confirmations should complete the ceremony lifecycle");
51        info!("Every key provisioner confirmed the ceremony; ceremony complete.");
52    }
53
54    Ok(status)
55}

72    const TEST_N: usize = 3;
73    const TEST_T: usize = 2;
74
75    struct TestContext {
76        enclave: Arc<Enclave>,
77        ceremony_digest: [u8; 32],
78        roster: KpCertRoster,
79        secret_keys: MockKpSecretKeys,
80    }

82    async fn setup_context() -> TestContext {
83        let (roster, secret_keys) = mock_kp_certs_roster_with_secrets(TEST_N);
84        let (logger, _) = mock_logger_capturing();
85        let enclave = Enclave::create_operator_initialized_ceremony(logger);
86        let response = setup_new_key(
87            enclave.clone(),
88            SetupNewKeyRequest::new(roster.clone(), TEST_N, TEST_T).unwrap(),
89        )
90        .await
91        .unwrap()
92        .verify_into_data(&enclave.signing_pubkey())
93        .unwrap()
94        .response;
95        TestContext {
96            enclave,
97            ceremony_digest: CeremonyState::from(response).digest(),
98            roster,
99            secret_keys,
100        }
101    }
102
103    impl TestContext {
104        fn signed_confirmation(&self, index: usize) -> KpSigned<CeremonyConfirmationRequest> {
105            self.signed_confirmation_with(index, self.enclave.s3_session_id(), self.ceremony_digest)
106        }
107
108        fn signed_confirmation_with_digest(
109            &self,
110            index: usize,
111            ceremony_digest: [u8; 32],
112        ) -> KpSigned<CeremonyConfirmationRequest> {
113            self.signed_confirmation_with(index, self.enclave.s3_session_id(), ceremony_digest)
114        }
115
116        fn signed_confirmation_with(
117            &self,
118            index: usize,
119            session_id: SessionID,
120            ceremony_digest: [u8; 32],
121        ) -> KpSigned<CeremonyConfirmationRequest> {
122            let cert = self.roster.iter().nth(index).unwrap().clone();
123            let request = CeremonyConfirmationRequest::new(session_id, ceremony_digest);
124            let signature = sign_detached_in_process(
125                self.secret_keys.get(&cert.fingerprint().to_hex()).unwrap(),
126                &KpSigned::signed_bytes(&request),
127            );
128            KpSigned::from_parts(request, cert, signature)
129        }
130    }

132    #[tokio::test]
133    async fn requires_every_kp_confirmation() {
134        let context = setup_context().await;
135        assert_eq!(
136            context.enclave.lifecycle(),
137            CeremonyStage::AwaitingKeyProvisionerConfirmations.into()
138        );
139
140        let first = context.signed_confirmation(0);
141        let status = confirm_ceremony(context.enclave.clone(), first.clone())
142            .await
143            .unwrap();
144        assert_eq!(status.have, 1);
145        assert!(!status.completed);
146        let repeated = confirm_ceremony(context.enclave.clone(), first)
147            .await
148            .unwrap();
149        assert_eq!(repeated.have, 1);
150
151        for index in 1..TEST_N {
152            let status =
153                confirm_ceremony(context.enclave.clone(), context.signed_confirmation(index))
154                    .await
155                    .unwrap();
156            assert_eq!(status.have as usize, index + 1);
157            assert_eq!(status.need as usize, TEST_N);
158            assert_eq!(status.completed, index + 1 == TEST_N);
159        }
160        assert_eq!(context.enclave.lifecycle(), CeremonyStage::Completed.into());
161        let repeated = confirm_ceremony(
162            context.enclave.clone(),
163            context.signed_confirmation(TEST_N - 1),
164        )
165        .await
166        .unwrap();
167        assert!(repeated.completed);
168    }
