39/// Enclave's config & state
40pub struct Enclave {
41    /// Immutable config (set once during init)
42    pub config: EnclaveConfig,
43    /// Mutable state
44    pub state: EnclaveState,
45    /// Temporary state retained until operator activation commits.
46    temporary_init_state: RwLock<Option<TemporaryInitState>>,
47    /// Serializes lifecycle and control-plane transitions so concurrent
48    /// operations cannot race a check-then-set.
49    control_lock: tokio::sync::Mutex<()>,
50    /// Ceremony state retained until every KP confirms successful share recovery.
51    pending_ceremony: OnceLock<PendingCeremony>,
52    /// Serializes and fences every S3 log write from this enclave session.
53    log_writer: LogWriter,
54}

56/// Configuration set during initialization (immutable after set)
57pub struct EnclaveConfig {
58    /// Ephemeral signing keypair generated at boot.
59    signing_keys: GuardianSignKeyPair,
60    /// Ephemeral encryption keypair generated at boot.
61    encryption_keys: GuardianEncKeyPair,
62    /// S3 client & config (set in operator_init)
63    s3_logger: OnceLock<GuardianS3Client>,
64    /// Enclave BTC private key (set in provisioner_init)
65    enclave_btc_keypair: OnceLock<Keypair>,
66    /// BTC network: mainnet, testnet, regtest (set in operator_init)
67    btc_network: OnceLock<Network>,
68    /// Raw MPC verifying key as a curve point. Stored with y-parity so the
69    /// 2-of-2 child-key derivation matches the MPC's signing protocol.
70    /// Set in operator_init.
71    hashi_btc_master_pubkey: OnceLock<HashiMasterG>,
72    /// Guardian build PCR pins used to verify attested guardian sessions.
73    pcr_allowlist: OnceLock<PcrAllowlist>,
74    /// Operator-supplied limiter configuration.
75    /// Note: This struct is duplicated in two places: `RateLimiter` stores a copy after activation.
76    limiter_config: OnceLock<LimiterConfig>,
77    /// The Hashi shared-object id this guardian serves; bound into every
78    /// committee-certificate preimage verified here (set in operator_init).
79    hashi_object_id: OnceLock<hashi_types::sui_sdk_types::Address>,
80}

82/// Mutable state that changes during operation.
83/// Committee + rate limiter are installed during operator_activate.
84pub struct EnclaveState {
85    /// Authoritative mode-specific lifecycle, initialized at boot.
86    lifecycle: RwLock<EnclaveLifecycle>,
87    /// Current Hashi committee.
88    committee: RwLock<Option<Arc<HashiCommittee>>>,
89    /// Rate limiter. Set once during operator_activate.
90    /// Uses `Arc<tokio::Mutex>` so the guard can be held across `.await`.
91    rate_limiter: OnceLock<Arc<tokio::sync::Mutex<RateLimiter>>>,
92    /// Mirrors the limiter's state so status reads never wait on the limiter
93    /// lock, which a withdrawal holds across its durable log write.
94    limiter_snapshot: RwLock<Option<LimiterState>>,
95}

106pub(crate) struct PendingCeremony {
107    digest: [u8; 32],
108    encrypted_shares: KpEncryptedShareRoster,
109    confirmed_share_ids: RwLock<BTreeSet<ShareID>>,
110}
111
112impl PendingCeremony {
113    fn new(state: CeremonyState) -> Self {
114        Self {
115            digest: state.digest(),
116            encrypted_shares: state.encrypted_shares,
117            confirmed_share_ids: RwLock::new(BTreeSet::new()),
118        }
119    }
120
121    pub(crate) fn validate_confirmation(
122        &self,
123        signer_fingerprint: &str,
124        ceremony_digest: &[u8; 32],
125    ) -> GuardianResult<(ShareID, bool)> {
126        if ceremony_digest != &self.digest {
127            return Err(InvalidInputs(
128                "ceremony confirmation digest differs from pending ceremony".into(),
129            ));
130        }
131        let share = self
132            .encrypted_shares
133            .find_by_fingerprint(signer_fingerprint)
134            .ok_or_else(|| {
135                Unauthenticated(format!(
136                    "KP fingerprint {signer_fingerprint} is not present in the pending ceremony roster"
137                ))
138            })?;
139        let confirmed = self
140            .confirmed_share_ids
141            .read()
142            .expect("pending ceremony lock poisoned")
143            .contains(&share.id);
144        Ok((share.id, confirmed))
145    }
146
147    pub(crate) fn record_confirmation(
148        &self,
149        share_id: ShareID,
150    ) -> GuardianResult<CeremonyConfirmationResponse> {
151        let mut confirmed = self
152            .confirmed_share_ids
153            .write()
154            .expect("pending ceremony lock poisoned");
155        confirmed.insert(share_id);
156        CeremonyConfirmationResponse::new(confirmed.len(), self.encrypted_shares.share_count())
157    }
158
159    pub(crate) fn status(&self) -> GuardianResult<CeremonyConfirmationResponse> {
160        CeremonyConfirmationResponse::new(
161            self.confirmed_share_ids
162                .read()
163                .expect("pending ceremony lock poisoned")
164                .len(),
165            self.encrypted_shares.share_count(),
166        )
167    }
168
169    fn is_complete(&self) -> bool {
170        self.status().is_ok_and(|status| status.completed)
171    }
172}

261    /// Install the activation-derived committee + rate limiter. Called from operator_activate.
262    pub fn init(&self, committee: HashiCommittee, rate_limiter: RateLimiter) -> GuardianResult<()> {
263        self.set_committee(committee)?;

272    /// Get the current committee.
273    pub fn get_committee(&self) -> GuardianResult<Arc<HashiCommittee>> {
274        let guard = self
275            .committee
276            .read()
277            .expect("rwlock should never throw an error");
278        guard
279            .as_ref()
280            .cloned()
281            .ok_or_else(|| InvalidInputs("committee not initialized".into()))
282    }
283
284    /// Whether the committee is installed, without cloning the `Arc`.
285    fn has_committee(&self) -> bool {
286        self.committee
287            .read()
288            .expect("rwlock should never throw an error")
289            .is_some()
290    }
291
292    /// Set committee. Called only from `init` during operator activation.
293    fn set_committee(&self, committee: HashiCommittee) -> GuardianResult<()> {
294        info!("Setting committee for epoch {}.", committee.epoch());
295
296        let mut guard = self
297            .committee
298            .write()
299            .expect("rwlock should never throw an error");
300        if guard.is_some() {
301            return Err(InvalidInputs("committee already initialized".into()));
302        }
303        *guard = Some(Arc::new(committee));
304        Ok(())
305    }
306
307    /// Replace an already-initialized committee. Rejects the swap unless
308    /// the in-memory epoch matches `expected_current_epoch`.
309    pub fn replace_committee(
310        &self,
311        committee: HashiCommittee,
312        expected_current_epoch: u64,
313    ) -> GuardianResult<()> {
314        info!("Replacing committee for epoch {}.", committee.epoch());
315
316        let mut guard = self
317            .committee
318            .write()
319            .expect("rwlock should never throw an error");
320        let current_epoch = guard
321            .as_ref()
322            .ok_or_else(|| InvalidInputs("committee not initialized".into()))?
323            .epoch();
324        if current_epoch != expected_current_epoch {
325            return Err(InvalidInputs(format!(
326                "committee epoch mismatch: expected {expected_current_epoch}, actual {current_epoch}"
327            )));
328        }
329        *guard = Some(Arc::new(committee));
330        Ok(())
331    }
332
333    // ========================================================================
334    // Rate Limiter Management
335    // ========================================================================
336
337    fn set_rate_limiter(&self, limiter: RateLimiter) -> GuardianResult<()> {
338        info!("Setting rate limiter.");
339
340        let state = *limiter.state();
341        self.rate_limiter
342            .set(Arc::new(tokio::sync::Mutex::new(limiter)))
343            .map_err(|_| InvalidInputs("rate_limiter already initialized".into()))?;
344        *self.limiter_snapshot.write().unwrap() = Some(state);
345        Ok(())
346    }
347
348    /// Timeout for acquiring the limiter lock. If a withdrawal is in progress and
349    /// takes longer than this, callers bail rather than wait indefinitely.
350    const LIMITER_LOCK_TIMEOUT: Duration = Duration::from_secs(10);
351
352    /// Acquire exclusive access to the limiter, bounded by
353    /// `LIMITER_LOCK_TIMEOUT`.
354    async fn lock_limiter(&self) -> GuardianResult<OwnedMutexGuard<RateLimiter>> {
355        let rate_limiter = self
356            .rate_limiter
357            .get()
358            .ok_or_else(|| InternalError("rate limiter not initialized".into()))?;
359        tokio::time::timeout(
360            Self::LIMITER_LOCK_TIMEOUT,
361            rate_limiter.clone().lock_owned(),
362        )
363        .await
364        .map_err(|_| Unavailable("timed out waiting for rate limiter lock".into()))
365    }
366
367    /// Acquire exclusive access to the limiter, consume tokens, and return a guard.
368    /// The guard is held through signing and durable logging so no other withdrawal
369    /// can start until this one is durably logged or the enclave aborts.
370    pub async fn consume_from_limiter(
371        &self,
372        seq: u64,
373        timestamp: u64,
374        amount_sats: u64,
375    ) -> GuardianResult<OwnedMutexGuard<RateLimiter>> {
376        let mut guard = self.lock_limiter().await?;
377        guard.consume(seq, timestamp, amount_sats)?;
378        Ok(guard)
379    }
380
381    /// Record the consumption a withdrawal has just made durable, then release
382    /// the limiter. Consuming the guard orders the record before the release.
383    pub fn set_limiter_snapshot(&self, guard: OwnedMutexGuard<RateLimiter>) {
384        *self.limiter_snapshot.write().unwrap() = Some(*guard.state());
385    }
386
387    /// The limiter state as of the last durably logged withdrawal. `None` means
388    /// no limiter exists; it never reports absence for a limiter that is busy.
389    pub fn limiter_snapshot(&self) -> Option<LimiterState> {
390        *self.limiter_snapshot.read().unwrap()
391    }
392}

426    /// Spawn a root-owned Tokio task so caller cancellation only detaches the
427    /// waiter and does not cancel accepted work. Use this when the task needs
428    /// no serialization or owns a narrower lock for the state it mutates.
429    /// E.g., used in `standard_withdrawal` that owns a separate serialization lock inside.
430    pub async fn spawn_task<Input, Output, Task, Fut>(
431        self: Arc<Self>,
432        input: Input,
433        task: Task,
434    ) -> GuardianResult<Output>
435    where
436        Input: Send + 'static,
437        Output: Send + 'static,
438        Task: FnOnce(Arc<Self>, Input) -> Fut + Send + 'static,
439        Fut: Future<Output = GuardianResult<Output>> + Send + 'static,
440    {
441        tokio::spawn(async move { task(self, input).await })
442            .await
443            .expect("guardian task failed")
444    }
445
446    /// Spawn a root-owned Tokio task that holds the control lock while running.
447    /// Use this for lifecycle and control-plane transitions whose shared-state
448    /// checks and mutations must not interleave with another control task.
449    pub async fn spawn_control_task<Input, Output, Task, Fut>(
450        self: Arc<Self>,
451        input: Input,
452        task: Task,
453    ) -> GuardianResult<Output>
454    where
455        Input: Send + 'static,
456        Output: Send + 'static,
457        Task: FnOnce(Arc<Self>, Input) -> Fut + Send + 'static,
458        Fut: Future<Output = GuardianResult<Output>> + Send + 'static,
459    {
460        self.spawn_task(input, move |enclave, input| async move {
461            let task_enclave = enclave.clone();
462            let _guard = enclave.control_lock.lock().await;
463            task(task_enclave, input).await
464        })
465        .await
466    }

472    /// Which flows this enclave serves (fixed at boot).
473    pub fn mode(&self) -> EnclaveMode {

477    pub fn lifecycle(&self) -> EnclaveLifecycle {
478        *self
479            .state
480            .lifecycle
481            .read()
482            .expect("lifecycle lock poisoned")
483    }
484
485    /// Require an exact mode and lifecycle stage.
486    pub fn require_lifecycle(&self, expected: EnclaveLifecycle) -> GuardianResult<()> {
487        let actual = self.lifecycle();
488        if actual != expected {
489            return Err(LifecycleMismatch { expected, actual });
490        }
491        Ok(())
492    }
493
494    /// Transition after the operation's durable log succeeds. The lifecycle is
495    /// the single source of completion state.
496    pub fn advance_lifecycle_into(&self, next: EnclaveLifecycle) -> GuardianResult<()> {
497        let expected = next
498            .predecessor()
499            .ok_or_else(|| InvalidInputs(format!("cannot advance lifecycle into {next:?}")))?;
500        let mut lifecycle = self
501            .state
502            .lifecycle
503            .write()
504            .expect("lifecycle lock poisoned");
505        if *lifecycle != expected {
506            return Err(LifecycleMismatch {
507                expected,
508                actual: *lifecycle,
509            });
510        }
511        self.assert_state_installed_for(next);
512        *lifecycle = next;
513        Ok(())
514    }
515
516    fn assert_state_installed_for(&self, next: EnclaveLifecycle) {
517        let installed = match next {
518            EnclaveLifecycle::Ceremony(CeremonyStage::Uninitialized)
519            | EnclaveLifecycle::Withdraw(WithdrawStage::Uninitialized) => {
520                unreachable!("cannot advance lifecycle into {next:?}")
521            }
522            EnclaveLifecycle::Ceremony(CeremonyStage::OperatorInitialized) => {
523                self.operator_init_state_installed(EnclaveMode::Ceremony)
524            }
525            EnclaveLifecycle::Ceremony(CeremonyStage::AwaitingKeyProvisionerConfirmations) => {
526                self.pending_ceremony.get().is_some()
527            }
528            EnclaveLifecycle::Ceremony(CeremonyStage::Completed) => self
529                .pending_ceremony
530                .get()
531                .is_some_and(PendingCeremony::is_complete),
532            EnclaveLifecycle::Withdraw(WithdrawStage::OperatorInitialized) => {
533                self.operator_init_state_installed(EnclaveMode::Withdraw)
534            }
535            EnclaveLifecycle::Withdraw(WithdrawStage::ProvisionerInitialized) => {
536                self.config.is_enclave_btc_keypair_set() && self.temporary_init_state_is_available()
537            }
538            EnclaveLifecycle::Withdraw(WithdrawStage::Activated) => {
539                self.state.has_committee()
540                    && self.state.rate_limiter.get().is_some()
541                    && !self.temporary_init_state_is_available()
542            }
543        };
544        assert!(
545            installed,
546            "cannot advance lifecycle to {next:?}: state is incomplete"
547        );
548    }
549
550    /// Whether every field operator_init installs is present (mode-aware).
551    fn operator_init_state_installed(&self, mode: EnclaveMode) -> bool {
552        // Both modes install the S3 logger; a ceremony enclave installs nothing else.
553        if self.config.s3_logger.get().is_none() {
554            return false;
555        }
556        match mode {
557            EnclaveMode::Ceremony => true,
558            // Withdraw enclaves additionally install their stable configuration.
559            EnclaveMode::Withdraw => {
560                self.config.btc_network.get().is_some()
561                    && self.config.pcr_allowlist.get().is_some()
562                    && self.config.limiter_config.get().is_some()
563                    && self.temporary_init_state_is_available()
564                    && self.config.hashi_btc_master_pubkey.get().is_some()
565                    && self.config.hashi_object_id.get().is_some()
566            }
567        }
568    }

607    /// Build identity for `GuardianInfo.untrusted_git_revision` / the
608    /// `PcrAllowlist` key. A real ceremony enclave is a distinct measured build
609    /// (its own PCR0) from the same-commit withdraw enclave, so it reports a
610    /// distinct identity — the allowlist forbids two entries per revision, so
611    /// otherwise the withdraw enclave and KPs couldn't pin both PCR0s.
612    /// `test`/`non-enclave-dev` skip attestation and share one entry, so the
613    /// suffix is compiled out (existing mock flow unchanged).
614    fn reported_git_revision(&self) -> String {
615        // Injected at build time (docker/CI); defaults outside a real build.
616        let base = option_env!("GIT_REVISION").unwrap_or("unknown");
617        if cfg!(not(any(test, feature = "non-enclave-dev"))) && self.mode() == EnclaveMode::Ceremony
618        {
619            format!("{base}-ceremony")
620        } else {
621            base.to_string()
622        }
623    }

625    pub async fn info(&self) -> GuardianInfo {
626        let temporary_init_state = self.temporary_init_state().ok();
627        GuardianInfo {
628            lifecycle: self.lifecycle(),
629            secret_sharing_instance: temporary_init_state
630                .as_ref()
631                .map(|state| state.ceremony_state.secret_sharing_instance.clone()),
632            bucket_info: self
633                .config
634                .s3_logger()
635                .ok()
636                .map(|l| l.bucket_info().clone()),
637            encryption_pubkey: self.encryption_public_key().to_bytes().to_vec(),
638            config_hash: temporary_init_state.as_ref().map(|state| state.config_hash),
639            genesis_state_hash: temporary_init_state
640                .as_ref()

643            enclave_btc_pubkey: self.config.enclave_btc_pubkey().ok(),
644            limiter_state: self.state.limiter_snapshot(),
645            limiter_config: self.limiter_config().ok(),
646            current_committee_epoch: self.state.get_committee().ok().map(|c| c.epoch()),
647            mpc_master_g: self.config.hashi_btc_master_pubkey.get().copied(),
648            hashi_object_id: self.config.hashi_object_id.get().copied(),
649        }

731    // Temporary Initialization State
732    // ========================================================================
733
734    /// Return an owned initialization snapshot. Activation takes the stored
735    /// value, so this capability is unavailable once the enclave is active.
736    /// Keep this as the only accessor for `TemporaryInitState`. Its readers are
737    /// restricted to provisioner initialization, operator activation, and
738    /// status reporting; active request handlers must not depend on it.
739    pub fn temporary_init_state(&self) -> GuardianResult<TemporaryInitState> {
740        self.temporary_init_state
741            .read()
742            .expect("temporary initialization lock poisoned")
743            .clone()
744            .ok_or_else(|| InvalidInputs("Temporary initialization state not set".into()))
745    }
746
747    /// Operator initialization is the sole installer of temporary state.
748    pub fn set_temporary_init_state(&self, state: TemporaryInitState) -> GuardianResult<()> {
749        let mut slot = self
750            .temporary_init_state
751            .write()
752            .expect("temporary initialization lock poisoned");
753        if slot.is_some() {
754            return Err(InvalidInputs(
755                "Temporary initialization state already set".into(),
756            ));
757        }
758        *slot = Some(state);
759        Ok(())
760    }
761
762    fn temporary_init_state_is_available(&self) -> bool {
763        self.temporary_init_state
764            .read()
765            .expect("temporary initialization lock poisoned")
766            .is_some()
767    }
768
769    pub fn clear_temporary_init_state(&self) {
770        self.temporary_init_state
771            .write()
772            .expect("temporary initialization lock poisoned")
773            .take()
774            .expect("temporary initialization state must exist before activation");
775    }

781    /// Construct a verified reader with the enclave's fixed S3 client and PCR
782    /// allowlist. Each operation gets a fresh, operation-scoped session cache.
783    pub fn new_guardian_reader(&self) -> GuardianResult<GuardianReader> {
784        let pcr_allowlist = self
785            .config
786            .pcr_allowlist
787            .get()
788            .cloned()
789            .ok_or_else(|| InvalidInputs("PCR allowlist is uninitialized".into()))?;
790        self.new_guardian_reader_with_allowlist(pcr_allowlist)
791    }

793    /// Construct a verified reader with an operation-authorized PCR allowlist.
794    /// Ceremony rotation uses the allowlist agreed in the current KPs' signed
795    /// requests because ceremony-mode operator init installs only S3 config.
796    pub(crate) fn new_guardian_reader_with_allowlist(
797        &self,
798        pcr_allowlist: PcrAllowlist,
799    ) -> GuardianResult<GuardianReader> {
800        let s3 = self.config.s3_logger()?.clone();

812    pub fn install_config(
813        &self,
814        network: Network,
815        hashi_btc_master_pubkey: HashiMasterG,
816        pcr_allowlist: PcrAllowlist,
817        limiter_config: LimiterConfig,
818        hashi_object_id: hashi_types::sui_sdk_types::Address,
819    ) -> GuardianResult<()> {
820        self.config
821            .btc_network
822            .set(network)
823            .map_err(|_| InvalidInputs("Network is already initialized".into()))?;
824        self.config
825            .hashi_btc_master_pubkey
826            .set(hashi_btc_master_pubkey)
827            .map_err(|_| InvalidInputs("Hashi BTC key is already initialized".into()))?;
828        self.config
829            .pcr_allowlist
830            .set(pcr_allowlist)
831            .map_err(|_| InvalidInputs("PCR allowlist is already initialized".into()))?;
832        self.config
833            .limiter_config
834            .set(limiter_config)
835            .map_err(|_| InvalidInputs("Limiter config is already initialized".into()))?;
836        self.config
837            .hashi_object_id
838            .set(hashi_object_id)
839            .map_err(|_| InvalidInputs("Hashi object id is already initialized".into()))
840    }
