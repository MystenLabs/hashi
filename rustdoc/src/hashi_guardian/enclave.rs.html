40/// Configuration set during initialization (immutable after set)
41pub struct EnclaveConfig {
42    /// Ephemeral signing keypair generated at boot.
43    signing_keys: GuardianSignKeyPair,
44    /// Ephemeral encryption keypair generated at boot.
45    encryption_keys: GuardianEncKeyPair,

50    /// Complete deployment policy, installed once in either mode during OI.
51    deployment: InitOnce<DeploymentConfig>,
52    /// Raw MPC verifying key as a curve point. Stored with y-parity so the
53    /// 2-of-2 child-key derivation matches the MPC's signing protocol.
54    /// Set in operator_init.

56    /// Operator-supplied limiter configuration.
57    /// Note: This struct is duplicated in two places: `RateLimiter` stores a copy after activation.
58    limiter_config: InitOnce<LimiterConfig>,
59    /// The Hashi shared-object id this guardian serves; bound into every
60    /// committee-certificate preimage verified here (set in operator_init).
61    hashi_object_id: InitOnce<hashi_types::sui_sdk_types::Address>,

64/// Mutable state that changes during operation.
65/// Committee and rate limiter are installed during operator activation.
66/// The service exclusively borrows the whole enclave; fields need no inner locks.
67#[derive(Default)]
68pub struct EnclaveState {
69    /// Advanced only after a durable heartbeat write.
70    next_heartbeat_seq: u64,
71    /// Authoritative mode-specific lifecycle, absent until operator init commits.
72    lifecycle: Option<EnclaveLifecycle>,
73    /// Current Hashi committee.
74    committee: Option<RuntimeCommittee>,
75    /// Rate limiter. Set once during operator_activate.
76    rate_limiter: Option<RateLimiter>,
77    /// Retained until operator activation commits.
78    temporary_init_state: Option<TemporaryInitState>,
79    /// Ceremony proposal and KP confirmations, retained even after completion.
80    pending_ceremony: Option<PendingCeremony>,
81}

92pub struct PendingCeremony {

98impl PendingCeremony {
99    fn new(
100        proposal: CeremonyProposalLogMessage,
101        deployment: &DeploymentConfig,
102    ) -> GuardianResult<Self> {
103        let artifacts = CeremonyArtifacts {
104            deployment: deployment.clone(),
105            ceremony_state: CeremonyState::from_proposal(proposal.clone())?,
106        };
107        Ok(Self {
108            proposal,
109            ceremony_artifacts_digest: artifacts.digest(),
110            confirmed_share_ids: BTreeSet::new(),
111        })
112    }
113
114    pub fn validate_confirmation(
115        &self,
116        signer_fingerprint: &str,
117        ceremony_artifacts_digest: &[u8; 32],
118    ) -> GuardianResult<(ShareID, bool)> {
119        if ceremony_artifacts_digest != &self.ceremony_artifacts_digest {
120            return Err(InvalidInputs(
121                "ceremony confirmation digest differs from pending ceremony".into(),
122            ));
123        }
124        let share = self
125            .proposal
126            .encrypted_shares
127            .find_by_fingerprint(signer_fingerprint)
128            .ok_or_else(|| {
129                Unauthenticated(format!(
130                    "KP fingerprint {signer_fingerprint} is not present in the pending ceremony roster"
131                ))
132            })?;
133        let confirmed = self.confirmed_share_ids.contains(&share.id);
134        Ok((share.id, confirmed))
135    }
136
137    pub fn record_confirmation(
138        &mut self,
139        share_id: ShareID,
140    ) -> GuardianResult<CeremonyConfirmationResponse> {
141        self.confirmed_share_ids.insert(share_id);
142        self.status()
143    }
144
145    pub fn status(&self) -> GuardianResult<CeremonyConfirmationResponse> {
146        CeremonyConfirmationResponse::new(
147            self.confirmed_share_ids.len(),
148            self.proposal.encrypted_shares.share_count(),
149        )
150    }
151
152    fn is_complete(&self) -> bool {
153        self.status().is_ok_and(|status| status.completed)
154    }
155}

247    /// Sign a BTC tx. Returns an Err if enclave btc keypair or hashi btc pk is not set.
248    pub fn btc_sign(&self, tx_utxos: &TxUTXOs) -> GuardianResult<(Txid, Vec<BitcoinSignature>)> {
249        let enclave_keypair = self
250            .enclave_btc_keypair
251            .get()
252            .ok_or(InvalidInputs("Bitcoin key is not initialized".into()))?;
253        let hashi_btc_pk = self

259        let (messages, txid) = tx_utxos.signing_messages_and_txid(&enclave_btc_pk, hashi_btc_pk);
260        Ok((txid, sign_btc_tx(&messages, enclave_keypair)))

498    /// Require an exact mode and lifecycle stage, or no lifecycle for `None`.
499    pub fn require_lifecycle(&self, expected: Option<EnclaveLifecycle>) -> GuardianResult<()> {
500        let actual = self.state.lifecycle();
501        if actual != expected {
502            return Err(LifecycleMismatch { expected, actual });
503        }
504        Ok(())
505    }
506
507    /// Check the predecessor and installed state, then advance the lifecycle.
508    /// Callers must persist the operation's completion record before calling this.
509    pub fn advance_lifecycle_into(&mut self, next: EnclaveLifecycle) -> GuardianResult<()> {
510        let expected = next.predecessor();
511        if self.state.lifecycle != expected {
512            return Err(LifecycleMismatch {
513                expected,
514                actual: self.state.lifecycle,
515            });
516        }
517        self.assert_state_installed_for(next);
518        self.state.lifecycle = Some(next);
519        Ok(())
520    }
521
522    fn assert_state_installed_for(&self, next: EnclaveLifecycle) {
523        let installed = match next {
524            EnclaveLifecycle::Ceremony(CeremonyStage::OperatorInitialized) => {
525                self.operator_init_state_installed(EnclaveMode::Ceremony)
526            }
527            EnclaveLifecycle::Ceremony(CeremonyStage::AwaitingKeyProvisionerConfirmations) => {
528                self.state.pending_ceremony.is_some()
529            }
530            EnclaveLifecycle::Ceremony(CeremonyStage::Completed) => self
531                .state
532                .pending_ceremony
533                .as_ref()
534                .is_some_and(PendingCeremony::is_complete),
535            EnclaveLifecycle::Withdraw(WithdrawStage::OperatorInitialized) => {
536                self.operator_init_state_installed(EnclaveMode::Withdraw)
537            }
538            EnclaveLifecycle::Withdraw(WithdrawStage::ProvisionerInitialized) => {
539                self.config.is_enclave_btc_keypair_set()
540                    && self.state.temporary_init_state.is_some()
541            }
542            EnclaveLifecycle::Withdraw(WithdrawStage::Activated) => {
543                self.state.committee.is_some()
544                    && self.state.rate_limiter.is_some()
545                    && self.state.temporary_init_state.is_none()
546            }
547        };
548        assert!(
549            installed,
550            "cannot advance lifecycle to {next:?}: state is incomplete"
551        );
552    }
553
554    /// Whether every field operator_init installs is present (mode-aware).
555    fn operator_init_state_installed(&self, mode: EnclaveMode) -> bool {
556        // Both modes install S3 and the shared deployment policy.
557        if self.config.s3_logger.get().is_none() || self.config.deployment.get().is_none() {
558            return false;
559        }
560        match mode {
561            EnclaveMode::Ceremony => true,
562            // Withdraw enclaves additionally install their stable configuration.
563            EnclaveMode::Withdraw => {
564                self.config.limiter_config.get().is_some()
565                    && self.state.temporary_init_state.is_some()
566                    && self.config.hashi_btc_master_pubkey.get().is_some()
567                    && self.config.hashi_object_id.get().is_some()
568            }
569        }
570    }
571
572    /// Require the activated withdraw lifecycle.
573    pub fn require_fully_initialized(&self) -> GuardianResult<()> {
574        self.require_lifecycle(WithdrawStage::Activated.into())
575    }

594    /// Collect status from the enclave already borrowed by the caller.

606            encryption_pubkey: self.config.encryption_public_key().to_bytes().to_vec(),
607            config_hash: temporary_init_state.as_ref().map(|state| state.config_hash),
608            genesis_state_hash: temporary_init_state
609                .as_ref()
610                .and_then(|state| state.genesis_state.as_ref().map(GenesisState::digest)),
611            enclave_btc_pubkey: self.config.enclave_btc_pubkey().ok(),
612            limiter_state: self.state.limiter_state(),
613            limiter_config: self.config.limiter_config().ok(),
614            current_committee_epoch: self.state.get_committee().ok().map(|c| c.epoch()),
615            mpc_master_g: self.config.hashi_btc_master_pubkey.get().copied(),
616            hashi_object_id: self.config.hashi_object_id.get().copied(),

653        self.write_log(LogMessage::Genesis(Box::new(msg))).await
654    }

656    pub async fn log_ceremony_proposal(
657        &mut self,
658        proposal: CeremonyProposalLogMessage,
659    ) -> GuardianResult<()> {
660        self.write_log(LogMessage::CeremonyProposal(Box::new(proposal)))
661            .await
662    }

664    /// Persist the current encrypted KP share state to `kp-shares/` for recovery.
665    /// `sharing_seq` pairs it with the matching `ceremony/` instance, while
666    /// `cert_seq` versions recipient-cert rotations within that instance.
667    pub async fn log_kp_share_state(
668        &mut self,
669        sharing_seq: u64,
670        cert_seq: u64,
671        encrypted_shares: KpEncryptedShareRoster,
672    ) -> GuardianResult<()> {
673        self.write_log(LogMessage::KpShareState(Box::new(
674            KpShareStateLogMessage::new(sharing_seq, cert_seq, encrypted_shares),
675        )))
676        .await
677    }

679    // ========================================================================
680    // Ceremony Publication
681    // ========================================================================
682
683    /// Publish the pending ceremony to the established authoritative locations.
684    /// The ceremony record is written last and therefore acts as the commit.
685    pub async fn publish_pending_ceremony(&mut self) -> GuardianResult<()> {
686        let CeremonyProposalLogMessage {
687            ceremony,
688            encrypted_shares,
689        } = self.state.pending_ceremony()?.proposal.clone();
690        self.log_kp_share_state(ceremony.sharing_seq(), 0, encrypted_shares)
691            .await?;
692        self.write_log(LogMessage::Ceremony(Box::new(ceremony)))
693            .await
694    }
695
696    // ========================================================================
697    // Heartbeats
698    // ========================================================================
699
700    /// Persist a heartbeat and advance its sequence after the durable write.
701    /// Called under the service control lock; a no-op outside withdraw mode.
702    pub async fn heartbeat(&mut self) -> GuardianResult<()> {
703        if self.state.mode() != Some(EnclaveMode::Withdraw) {
704            return Ok(());
705        }
706
707        self.write_log(LogMessage::Heartbeat(HeartbeatLogMessage::new(
708            self.state.next_heartbeat_seq,
709        )))
710        .await?;
711        self.state.next_heartbeat_seq += 1;
712        Ok(())
713    }
