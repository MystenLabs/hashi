34/// What the files say, from the stack and the environment file.
35pub struct Settings {
36    pub proxy_url: String,
37    pub tunnel_endpoint: String,
38    pub bucket: String,
39    pub region: String,
40    pub retention_environment: String,
41    pub bitcoin_network: String,
42    pub commit: String,
43    pub pcr0: String,
44    pub prev_builds: Vec<Build>,
45    pub sui_rpc: String,
46    pub package_id: String,
47    pub object_id: String,
48    pub refill_rate: u64,
49    pub max_bucket_capacity: u64,
50    pub access_key: String,
51    pub secret_key: String,
52}
53
54impl Settings {
55    pub fn read(ctx: &Ctx, stack: &Stack, view: &SlotView) -> Result<Self> {
56        let bucket = stack.bucket()?;
57        let pcr0 = stack.config(&view.pcr0_key).unwrap_or("");
58        if !is_hex(pcr0, 96) {
59            bail!(
60                "{} is not a PCR0 measurement: {}",
61                view.pcr0_key,
62                if pcr0.is_empty() { "unset" } else { pcr0 }
63            );
64        }
65        let sats = |key: &str| -> Result<u64> {
66            let value = stack
67                .config(key)
68                .ok_or_else(|| anyhow!("stack {} has no {key}", stack.name))?;
69            value.parse().map_err(|_| {
70                anyhow!(
71                    "stack {}'s {key} is not a whole number: {value}",
72                    stack.name
73                )
74            })
75        };
76        let refill_rate = sats("refill-rate-sats-per-sec")?;
77        let max_bucket_capacity = sats("max-bucket-capacity-sats")?;
78        let prev_builds = prev_builds(ctx, stack, view, pcr0)?;
79        // The enclave keeps the key it is initialized with, so this must be the stack's long-lived one.
80        let access_key = stack
81            .output("s3_access_key_id")
82            .ok_or_else(|| anyhow!("stack {} has no s3_access_key_id output", stack.name))?
83            .to_owned();
84        let secret_key = stack
85            .output_value(ctx, "s3_secret_access_key", true)
86            .context("could not read the guardian's S3 secret key")?;
87        let env = &ctx.env;
88        // The files quote every value as JSON does, which YAML reads the same only for printable ASCII.
89        let plain = |key: &str| -> Result<String> {
90            let value = env.require(key)?;
91            if !value.chars().all(|c| c == ' ' || c.is_ascii_graphic()) {
92                bail!("{key} holds characters other than printable ASCII");
93            }
94            Ok(value.to_owned())
95        };
96        Ok(Self {
97            proxy_url: env.proxy_url()?.url,
98            tunnel_endpoint: view.endpoint.clone(),
99            bucket: bucket.name,
100            region: bucket.region,
101            retention_environment: plain("RETENTION_ENVIRONMENT")?,
102            bitcoin_network: plain("BITCOIN_NETWORK")?,
103            commit: view.commit.clone(),
104            pcr0: pcr0.to_owned(),
105            prev_builds,
106            sui_rpc: plain("SUI_RPC")?,
107            package_id: plain("HASHI_PACKAGE_ID")?,
108            object_id: plain("HASHI_OBJECT_ID")?,
109            refill_rate,
110            max_bucket_capacity,
111            access_key,
112            secret_key,
113        })
114    }
115}

117/// A build the configurations allowlist besides the slot's own.

119pub struct Build {
120    pub commit: String,
121    pub pcr0: String,
122}

124/// An entry of the stack's `prev-builds`, named as the configurations name a build.
125#[derive(Deserialize)]
126#[serde(deny_unknown_fields)]
127struct ListedBuild {
128    git_revision: String,
129    pcr0: String,
130}
131
132/// The builds the slot's configurations allowlist besides its own: the other slot's, then the stack's `prev-builds`.
133/// Every reader must allowlist each build that wrote the log, so `prev-builds` keeps one once no slot names it.
134pub fn prev_builds(ctx: &Ctx, stack: &Stack, view: &SlotView, pcr0: &str) -> Result<Vec<Build>> {
135    let other = view.slot.other();
136    let suffix = other.suffix();
137    let mut builds: Vec<Build> = Vec::new();
138    if let Some(commit) = stack.config(&format!("hashi-commit{suffix}")) {
139        let other_pcr0 = stack.config(&format!("eif-pcr0{suffix}")).unwrap_or("");
140        let shown = if other_pcr0.is_empty() {
141            "unset"
142        } else {
143            other_pcr0
144        };
145        if commit == view.commit {
146            if other_pcr0 != pcr0 {
147                bail!(
148                    "slots a and b of stack {} both build {commit}, but eif-pcr0{suffix} is {shown} and {} is {pcr0}; one commit has one measurement: give both the same",
149                    stack.name,
150                    view.pcr0_key
151                );
152            }
153        } else {
154            if !is_hex(commit, 40) {
155                bail!(
156                    "stack {}'s hashi-commit{suffix} is not a full commit: {commit}",
157                    stack.name
158                );
159            }
160            if !is_hex(other_pcr0, 96) {
161                bail!(
162                    "stack {}'s eif-pcr0{suffix} is not a PCR0 measurement: {shown}; slot {}'s configurations list slot {other}'s build {commit}, which needs its measurement",
163                    stack.name,
164                    view.slot
165                );
166            }
167            require_same_formats(ctx, &view.commit, commit, &format!("slot {other}'s build"))?;
168            builds.push(Build {
169                commit: commit.to_owned(),
170                pcr0: other_pcr0.to_owned(),
171            });
172        }
173    }
174    let listed: Vec<ListedBuild> = match stack.config("prev-builds") {
175        Some(text) => serde_json::from_str(text).map_err(|_| {
176            anyhow!(
177                "stack {}'s prev-builds is not a list of builds, each a git_revision and a pcr0: {text}",
178                stack.name
179            )
180        })?,
181        None => Vec::new(),
182    };
183    for ListedBuild {
184        git_revision: commit,
185        pcr0: listed_pcr0,
186    } in listed
187    {
188        if !is_hex(&commit, 40) {
189            bail!(
190                "stack {}'s prev-builds lists {commit}, which is not a full commit",
191                stack.name
192            );
193        }
194        if !is_hex(&listed_pcr0, 96) {
195            bail!(
196                "stack {}'s prev-builds lists {commit} with {listed_pcr0}, which is not a PCR0 measurement",
197                stack.name
198            );
199        }
200        let known = std::iter::once((view.commit.as_str(), pcr0))
201            .chain(builds.iter().map(|b| (b.commit.as_str(), b.pcr0.as_str())))
202            .find(|(known, _)| *known == commit);
203        match known {
204            Some((_, known_pcr0)) if known_pcr0 == listed_pcr0 => continue,
205            Some((_, known_pcr0)) => bail!(
206                "stack {} gives {commit} two measurements, {known_pcr0} and {listed_pcr0} in prev-builds; one commit has one measurement: give it the same in both",
207                stack.name
208            ),
209            None => {}
210        }
211        require_same_formats(ctx, &view.commit, &commit, "a build in prev-builds")?;
212        builds.push(Build {
213            commit,
214            pcr0: listed_pcr0,
215        });
216    }
217    Ok(builds)
218}

220/// A verified roster, as the download-kp-pubkeys step writes it: one directory per key provisioner.
221pub struct Roster {
222    pub dir: String,
223    pub threshold: u32,
224    certs: Vec<Cert>,
225}
226
227struct Cert {
228    name: String,
229    /// The certificate and its attestation files, by file name.
230    files: Vec<(String, Vec<u8>)>,
231}
232
233impl Roster {
234    pub fn load(dir: &str, threshold: u32) -> Result<Self> {
235        let root = Path::new(dir);
236        if !root.join("roster.txt").is_file() {
237            bail!("no verified roster in {dir}; run the download-kp-pubkeys step first");
238        }
239        let mut found = Vec::new();
240        for id in entries(root)? {
241            if root.join(&id).is_dir() {
242                for name in entries(&root.join(&id))? {
243                    if name.ends_with("-kp-pubkey.asc") {
244                        // The configurations list certificates unquoted, as guardian.sh did.
245                        let plain = |part: &str| {
246                            part.bytes()
247                                .all(|b| b.is_ascii_alphanumeric() || b"._-".contains(&b))
248                        };
249                        if !plain(&id) || !plain(&name) {
250                            bail!(
251                                "{dir}/{id}/{name} has a name the configurations can't list: use letters, digits, dots, hyphens and underscores only"
252                            );
253                        }
254                        found.push((id.clone(), name));
255                    }
256                }
257            }
258        }
259        // In the order of bash's `*/*-kp-pubkey.asc`, which sorts whole paths: `a-b/` comes before `a/`.
260        found.sort_by_key(|(id, name)| format!("{id}/{name}"));
261        let mut certs = Vec::new();
262        for (id, name) in found {
263            let dir = root.join(&id);
264            let stem = name.trim_end_matches(".asc");
265            let mut files = vec![(name.clone(), read(&dir.join(&name))?)];
266            for suffix in ATTESTATION_SUFFIXES {
267                let sibling = format!("{stem}.{suffix}");
268                files.push((sibling.clone(), read(&dir.join(&sibling))?));
269            }
270            certs.push(Cert { name, files });
271        }
272        if certs.is_empty() {
273            bail!("no certificates in {dir}");
274        }
275        Ok(Self {
276            dir: dir.to_owned(),
277            threshold,
278            certs,
279        })
280    }
281
282    pub fn kp_count(&self) -> usize {
283        self.certs.len()
284    }
285
286    fn names(&self) -> Vec<&str> {
287        self.certs.iter().map(|cert| cert.name.as_str()).collect()
288    }
289}

291/// Non-hidden entries of a directory, which bash's globs match.
292fn entries(dir: &Path) -> Result<Vec<String>> {
293    let mut names = Vec::new();
294    for entry in
295        std::fs::read_dir(dir).with_context(|| format!("could not read {}", dir.display()))?
296    {
297        let name = entry?.file_name();
298        let name = name
299            .to_str()
300            .ok_or_else(|| anyhow!("{} holds a name that is not UTF-8", dir.display()))?;
301        if !name.starts_with('.') {
302            names.push(name.to_owned());
303        }
304    }
305    Ok(names)
306}
307
308fn read(path: &Path) -> Result<Vec<u8>> {
309    std::fs::read(path).with_context(|| format!("could not read {}", path.display()))
310}

312/// Rendered files by path relative to the slot's directory. `operator.yaml` holds the stack's S3 key.
313pub struct Rendered {
314    files: BTreeMap<String, Vec<u8>>,
315}
316
317impl Rendered {
318    /// The slot's files. A KP-set rotation adds `rotate-kp-set/`, whose files list both rosters for the re-deal, while
319    /// the slot's own files list the new roster, which provisions guardians once the re-deal completes.
320    pub fn new(settings: &Settings, dealt: &Roster, new: Option<&Roster>) -> Result<Self> {
321        let mut rendered = Self {
322            files: BTreeMap::new(),
323        };
324        let Some(new) = new else {
325            rendered.add_configs("", settings, dealt, None);
326            rendered.add_certs("", dealt, None)?;
327            return Ok(rendered);
328        };
329        rendered.add_configs("", settings, new, None);
330        rendered.add_certs("", new, None)?;
331        rendered.add_configs("rotate-kp-set/", settings, dealt, Some(new));
332        rendered.add_certs("rotate-kp-set/", dealt, None)?;
333        rendered.add_certs("rotate-kp-set/", new, Some(dealt))?;
334        Ok(rendered)
335    }
336
337    fn add_configs(
338        &mut self,
339        prefix: &str,
340        settings: &Settings,
341        roster: &Roster,
342        new: Option<&Roster>,
343    ) {
344        self.files.insert(
345            format!("{prefix}guardian-init.yaml"),
346            config(settings, &settings.proxy_url, roster, new).into_bytes(),
347        );
348        let operator = format!(
349            "{}\ns3_credentials:\n  access_key: {}\n  secret_key: {}\n",
350            config(settings, &settings.tunnel_endpoint, roster, new),
351            quote(&settings.access_key),
352            quote(&settings.secret_key)
353        );
354        self.files
355            .insert(format!("{prefix}operator.yaml"), operator.into_bytes());
356    }
357
358    /// Adds a roster's files under `certs/`. A configuration lists two rosters by file name, so one name must be one
359    /// certificate in both.
360    fn add_certs(&mut self, prefix: &str, roster: &Roster, other: Option<&Roster>) -> Result<()> {
361        for cert in &roster.certs {
362            for (name, bytes) in &cert.files {
363                let path = format!("{prefix}certs/{name}");
364                if let Some(existing) = self.files.get(&path)
365                    && existing != bytes
366                {
367                    bail!(
368                        "{name} differs between {} and {}; a key provisioner with a new certificate needs a new user ID",
369                        other.map_or(roster.dir.as_str(), |other| other.dir.as_str()),
370                        roster.dir
371                    );
372                }
373                self.files.insert(path, bytes.clone());
374            }
375        }
376        Ok(())
377    }
378
379    pub fn file(&self, path: &str) -> Option<&[u8]> {
380        self.files.get(path).map(Vec::as_slice)
381    }
382
383    /// Every file, by its path in the slot's directory.
384    pub fn files(&self) -> impl Iterator<Item = (&str, &[u8])> {
385        self.files
386            .iter()
387            .map(|(path, bytes)| (path.as_str(), bytes.as_slice()))
388    }
389
390    /// The configuration files, without the certificates they list.
391    pub fn configs(&self) -> impl Iterator<Item = &str> {
392        self.files
393            .keys()
394            .map(String::as_str)
395            .filter(|path| path.ends_with(".yaml"))
396    }
397
398    /// Where [`Rendered::write_unchecked`] puts a file, relative to the slot's directory: each configuration waits
399    /// under another name until the guardian's tools have checked it, so no step finds one they never passed.
400    pub fn written(path: &str) -> String {
401        if path.ends_with(".yaml") {
402            format!("{path}.unchecked")
403        } else {
404            path.to_owned()
405        }
406    }
407
408    /// Removes the configurations, so no step runs with one the guardian's tools refused.
409    pub fn remove_configs(&self, dir: &Path) -> Result<()> {
410        for config in self.configs() {
411            for path in [dir.join(config), dir.join(Self::written(config))] {
412                if path.exists() {
413                    std::fs::remove_file(&path)
414                        .with_context(|| format!("could not remove {}", path.display()))?;
415                }
416            }
417        }
418        Ok(())
419    }
420
421    /// Writes the files into `dir`, replacing the certificates and configurations an earlier render left. Other files
422    /// there, such as a ceremony's records, stay.
423    pub fn write(&self, dir: &Path) -> Result<()> {
424        self.write_unchecked(dir)?;
425        self.place_configs(dir)
426    }
427
428    /// Writes the files as [`Rendered::write`] does, with the configurations where [`Rendered::written`] says until
429    /// [`Rendered::place_configs`].
430    pub fn write_unchecked(&self, dir: &Path) -> Result<()> {
431        self.remove_configs(dir)?;
432        for stale in ["certs", "rotate-kp-set"] {
433            let path = dir.join(stale);
434            if path.exists() {
435                std::fs::remove_dir_all(&path)
436                    .with_context(|| format!("could not remove {}", path.display()))?;
437            }
438        }
439        for (relative, bytes) in &self.files {
440            let path = dir.join(Self::written(relative));
441            std::fs::create_dir_all(path.parent().unwrap_or(dir))?;
442            if relative.ends_with("operator.yaml") {
443                OpenOptions::new()
444                    .write(true)
445                    .create_new(true)
446                    .mode(0o600)
447                    .open(&path)
448                    .and_then(|mut file| file.write_all(bytes))
449            } else {
450                std::fs::write(&path, bytes)
451            }
452            .with_context(|| format!("could not write {}", path.display()))?;
453        }
454        Ok(())
455    }
456
457    /// Moves the configurations into place, once checked.
458    pub fn place_configs(&self, dir: &Path) -> Result<()> {
459        for config in self.configs() {
460            let (from, to) = (dir.join(Self::written(config)), dir.join(config));
461            std::fs::rename(&from, &to).with_context(|| {
462                format!("could not move {} to {}", from.display(), to.display())
463            })?;
464        }
465        Ok(())
466    }
467
468    /// The files in `dir` that differ from this render. Only the files a render writes count: Finder, for one, adds
469    /// its own to a directory it has shown.
470    pub fn stale(&self, dir: &Path) -> Vec<&str> {
471        self.files
472            .iter()
473            .filter(|(relative, bytes)| {
474                std::fs::read(dir.join(relative)).ok().as_ref() != Some(*bytes)
475            })
476            .map(|(relative, _)| relative.as_str())
477            .collect()
478    }
479}

481/// A slot's files as a render would write them now, with what they were rendered from.
482pub struct SlotRender {
483    pub settings: Settings,
484    pub dealt: Roster,
485    /// The roster a KP-set rotation re-deals to.
486    pub new: Option<Roster>,
487    pub rendered: Rendered,
488}
489
490impl SlotRender {
491    pub fn new(ctx: &Ctx, stack: &Stack, view: &SlotView) -> Result<Self> {
492        let settings = Settings::read(ctx, stack, view)?;
493        let dealt = Roster::load(
494            ctx.env.require("KP_ROSTER_DIR")?,
495            threshold(ctx, "KP_THRESHOLD")?,
496        )?;
497        let new = match ctx.env.get("NEW_KP_ROSTER_DIR") {
498            Some(dir) => Some(Roster::load(dir, threshold(ctx, "NEW_KP_THRESHOLD")?)?),
499            None => None,
500        };
501        let rendered = Rendered::new(&settings, &dealt, new.as_ref())?;
502        Ok(Self {
503            settings,
504            dealt,
505            new,
506            rendered,
507        })
508    }
509}

511pub(crate) fn threshold(ctx: &Ctx, key: &str) -> Result<u32> {
512    let value = ctx.env.require(key)?;
513    value
514        .parse()
515        .map_err(|_| anyhow!("{key} is not a number: {value}"))
516}

518/// Stops unless the slot's rendered files are what a render would write now: the stack, the environment file and
519/// the rosters can each have changed since the render step ran.
520pub fn require_fresh(ctx: &Ctx, stack: &Stack, view: &SlotView) -> Result<()> {
521    let operator = view.dir.join("operator.yaml");
522    if !operator.is_file() {
523        bail!(
524            "no {}; run the render step for slot {} first",
525            operator.display(),
526            view.slot
527        );
528    }
529    let render = SlotRender::new(ctx, stack, view)?;
530    let stale = render.rendered.stale(&view.dir);
531    if !stale.is_empty() {
532        bail!(
533            "out of date in {}: {}; the stack, {} or the roster changed after the render step; run it again, then publish again and have every key provisioner download again, unless only operator.yaml is out of date",
534            view.dir.display(),
535            stale.join(", "),
536            ctx.env.path().display()
537        );
538    }
539    Ok(())
540}

542/// A rendered configuration. `endpoint` is how its reader reaches the guardian: the proxy for key provisioners, the
543/// slot's tunnel for the operator.
544fn config(settings: &Settings, endpoint: &str, roster: &Roster, new: Option<&Roster>) -> String {
545    let s = settings;
546    let mut lines = vec![
547        format!("guardian_endpoint: {}", quote(endpoint)),
548        format!("relay_endpoint: {}", quote(&s.proxy_url)),
549        String::new(),
550        "deployment:".to_owned(),
551        "  bucket_info:".to_owned(),
552        format!("    name: {}", quote(&s.bucket)),
553        format!("    region: {}", quote(&s.region)),
554        format!(
555            "  retention_environment: {}",
556            quote(&s.retention_environment)
557        ),
558        format!("  bitcoin_network: {}", quote(&s.bitcoin_network)),
559        "  pcr_allowlist:".to_owned(),
560        "    current_build:".to_owned(),
561        format!("      git_revision: {}", quote(&s.commit)),
562        format!("      pcr0: {}", quote(&s.pcr0)),
563    ];
564    if s.prev_builds.is_empty() {
565        lines.push("    prev_builds: []".to_owned());
566    } else {
567        lines.push("    prev_builds:".to_owned());
568        for build in &s.prev_builds {
569            lines.push(format!("      - git_revision: {}", quote(&build.commit)));
570            lines.push(format!("        pcr0: {}", quote(&build.pcr0)));
571        }
572    }
573    lines.extend([
574        String::new(),
575        "hashi:".to_owned(),
576        format!("  sui_rpc: {}", quote(&s.sui_rpc)),
577        format!("  package_id: {}", quote(&s.package_id)),
578        format!("  hashi_object_id: {}", quote(&s.object_id)),
579        String::new(),
580    ]);
581    lines.extend(roster_lines("kp_roster", roster));
582    if let Some(new) = new {
583        lines.push(String::new());
584        lines.extend(roster_lines("new_kp_roster", new));
585    }
586    lines.extend([
587        String::new(),
588        "limiter_config:".to_owned(),
589        format!("  refill_rate: {}", s.refill_rate),
590        format!("  max_bucket_capacity: {}", s.max_bucket_capacity),
591    ]);
592    lines.join("\n") + "\n"
593}
594
595fn roster_lines(name: &str, roster: &Roster) -> Vec<String> {
596    let mut lines = vec![
597        format!("{name}:"),
598        format!("  num_shares: {}", roster.kp_count()),
599        format!("  threshold: {}", roster.threshold),
600        "  kp_pgp_cert_paths:".to_owned(),
601    ];
602    lines.extend(
603        roster
604            .names()
605            .into_iter()
606            .map(|cert| format!("    - certs/{cert}")),
607    );
608    lines
609}

611/// A YAML double-quoted string. JSON's escaping is valid YAML, and matches bash's `"%s"` for any plain value.
612fn quote(value: &str) -> String {
613    serde_json::Value::from(value).to_string()
614}

646    /// A roster directory as the download-kp-pubkeys step writes it, each file holding `tag` and its own name.
647    pub fn roster_dir(root: &Path, ids: &[&str], tag: &str) -> String {
648        std::fs::create_dir_all(root).unwrap();
649        std::fs::write(root.join("roster.txt"), "1  x\n").unwrap();
650        for id in ids {
651            let dir = root.join(id);
652            std::fs::create_dir_all(&dir).unwrap();
653            let stem = format!("{id}-kp-pubkey");
654            for name in std::iter::once(format!("{stem}.asc"))
655                .chain(ATTESTATION_SUFFIXES.iter().map(|s| format!("{stem}.{s}")))
656            {
657                std::fs::write(dir.join(&name), format!("{tag} {name}")).unwrap();
658            }
659            std::fs::write(dir.join(format!("{id}-kp-fingerprint.txt")), "F").unwrap();
660        }
661        root.display().to_string()
662    }
