22/// Loads the stack and the slot a step names, or the stack's only one.
23fn slot(ctx: &Ctx, arg: Option<Slot>) -> Result<(Stack, SlotView)> {
24    let stack = Stack::load(ctx)?;
25    let slot = stack.select(arg)?;
26    let view = stack.slot(ctx, slot)?;
27    Ok((stack, view))
28}
