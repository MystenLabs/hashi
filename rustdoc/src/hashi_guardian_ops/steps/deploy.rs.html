36/// Has hashi's guardian-enclave workflow measure the slot's commit on two runners, or checks run `run_id`, and stops
37/// unless both agree on the PCR0 the stack configures.
38pub fn measure(ctx: &Ctx, arg: Option<Slot>, run_id: Option<String>) -> Result<()> {
39    const REPO: &str = "MystenLabs/hashi";
40    let (stack, view) = slot(ctx, arg)?;
41    print_info(&format!(
42        "Measuring slot {}'s guardian build at {}",
43        view.slot, view.commit
44    ));
45    let suffix = view.slot.suffix();
46    let featureless = |stack: &Stack| -> Result<()> {
47        if stack.config(&format!("eif-features{suffix}")).is_some() {
48            bail!(
49                "stack {} sets eif-features{suffix}, so slot {} builds an image this workflow's real build doesn't measure",
50                stack.name,
51                view.slot
52            );
53        }
54        Ok(())
55    };
56    featureless(&stack)?;
57    let run_id = match run_id {
58        Some(run_id) => run_id,
59        None => {
60            github::require_run_urls(ctx)?;
61            github::dispatch(
62                ctx,
63                REPO,
64                "guardian-enclave.yml",
65                "main",
66                &[("git_revision", &view.commit)],
67            )?
68        }
69    };
70    if run_id.is_empty() || !run_id.bytes().all(|b| b.is_ascii_digit()) {
71        bail!("could not tell which run measures the build: {run_id}");
72    }
73    print_info(&format!(
74        "Measurement run: https://github.com/{REPO}/actions/runs/{run_id}"
75    ));
76    // Only a dispatch builds the inputs it is given; other runs check reproducibility with canonical ones.
77    let run = github::run(ctx, REPO, &run_id)?;
78    if run.event != "workflow_dispatch" || run.workflow_name != "Guardian Enclave Build" {
79        bail!(
80            "run {run_id} is a {} run of {}, not a dispatched Guardian Enclave Build, so it measures no deployment",
81            run.event,
82            run.workflow_name
83        );
84    }
85    github::watch(ctx, REPO, &run_id)?;
86    // The run builds whatever commit and features it was given, so check it built this one on both runners.
87    let log = read(
88        ctx.exec,
89        &Cmd::new("gh").args(["run", "view", &run_id, "--repo", REPO, "--log"]),
90    )
91    .with_context(|| format!("could not read the log of run {run_id}"))?;
92    github::require_canonical_build(&log, &run_id, &view.commit)?;
93    let downloads = tempfile::tempdir()?;
94    read(
95        ctx.exec,
96        &Cmd::new("gh")
97            .args(["run", "download", &run_id, "--repo", REPO])
98            .args(["--pattern", "nitro-pcrs-*", "--dir"])
99            .arg(downloads.path()),
100    )
101    .with_context(|| format!("could not download the measurements of run {run_id}"))?;
102    let mut pcrs = Vec::new();
103    for entry in std::fs::read_dir(downloads.path())? {
104        let path = entry?.path().join("nitro.pcrs");
105        if path.is_file() {
106            pcrs.push(std::fs::read_to_string(path)?);
107        }
108    }
109    let measured = github::agreed_pcr0(&pcrs)?;
110    println!("PCR0: {measured}");
111    // The watch takes minutes, during which the operator may set the key it compares.
112    let stack = Stack::load(ctx)?;
113    featureless(&stack)?;
114    let now = stack.slot(ctx, view.slot)?;
115    if now.commit != view.commit {
116        bail!(
117            "stack {}'s hashi-commit{suffix} changed to {} while run {run_id} measured {}; measure that one",
118            stack.name,
119            now.commit,
120            view.commit
121        );
122    }
123    let configured = stack.config(&view.pcr0_key);
124    if configured != Some(measured.as_str()) {
125        bail!(
126            "stack {} has {} {}; set it to the PCR0 above in its Pulumi.<stack>.yaml",
127            stack.name,
128            view.pcr0_key,
129            configured.unwrap_or("unset")
130        );
131    }
132    print_success(&format!(
133        "It is the {} stack {} configures.",
134        view.pcr0_key, stack.name
135    ));
136    if view.instance.is_none() {
137        next(&if stack.slots()?.len() == 1 {
138            "deploy".to_owned()
139        } else {
140            format!("arm --slot {} next to the serving guardian", view.slot)
141        });
142    }
143    Ok(())
144}

146/// Builds the proxy at the active slot's commit and applies the proxy stack, through the workflow
147/// `GUARDIAN_PROXY_WORKFLOW` names.
148pub fn proxy(ctx: &Ctx) -> Result<()> {
149    let stack = Stack::load(ctx)?;
150    let view = deployed_active(ctx, &stack)?;
151    print_info(&format!("Rolling the guardian proxy to {}", view.commit));
152    let url = ctx.env.proxy_url()?.url;
153    let (workflow, git_ref) = github::proxy_workflow(ctx)?;
154    github::require_run_urls(ctx)?;
155    // The workflow builds the image and applies the proxy stack from the ref it runs at.
156    print_info(&format!("Proxy workflow: {workflow}@{git_ref}"));
157    let tree = Tree::at(
158        ctx,
159        &view.commit,
160        &format!("the commit slot {}'s guardian runs", view.slot),
161    )?;
162    require_no_kp_round(ctx, &tree)?;
163    print_warning(
164        "Rolling a proxy that is serving takes the guardian's address down while it restarts, and drops the shares key provisioners have sent in a provisioning round that has not finished.",
165    );
166    confirm(
167        ctx.ask,
168        &format!("Roll the proxy of {}?", stack.short_name()),
169        &Unchanged {
170            no_input: "no input received; nothing was rolled",
171            declined: "not confirmed; nothing was rolled",
172        },
173    )?;
174    // The question can wait any time, while a deploy or a switchover moves the active slot's guardian.
175    let now = deployed_active(ctx, &Stack::load(ctx)?)?;
176    if now.slot != view.slot || now.commit != view.commit {
177        bail!(
178            "stack {}'s active slot or its commit changed while this step waited for you; nothing was rolled: run this step again",
179            stack.name
180        );
181    }
182    require_no_kp_round(ctx, &tree)?;
183    let run_id = github::dispatch(
184        ctx,
185        &workflow.repo,
186        &workflow.file,
187        &git_ref,
188        &[("env", stack.short_name()), ("hashi_commit", &view.commit)],
189    )?;
190    print_info(&format!(
191        "Proxy deploy run: https://github.com/{}/actions/runs/{run_id}",
192        workflow.repo
193    ));
194    github::watch(ctx, &workflow.repo, &run_id).context(
195        "check the proxy deploy before going on; running this step again restarts the proxy again",
196    )?;
197    proxy_health(ctx)
198        .with_context(|| format!("the proxy deployed but {url}/health does not answer"))?;
199    print_success(&format!("The proxy answers at {url}."));
200    next(&format!(
201        "tunnel --slot {}, left running in its own terminal; then info --slot {}",
202        view.slot, view.slot
203    ));
204    Ok(())
205}

207/// The active slot, once the stack has deployed the commit it configures: the proxy decodes that guardian, so it runs
208/// the build that guardian was deployed from.
209fn deployed_active(ctx: &Ctx, stack: &Stack) -> Result<SlotView> {
210    let view = stack.slot(ctx, stack.active()?)?;
211    view.require_deployed(ctx, stack)?;
212    Ok(view)
213}

215/// Stops while the relay's guardian is in a key provisioner round, whose shares a roll drops. A relay that doesn't
216/// answer is left to the warning: rolling the proxy may be what brings it back.
217fn require_no_kp_round(ctx: &Ctx, tree: &Tree) -> Result<()> {
218    match relay_info(ctx, tree) {
219        Ok(relay) if relay.lifecycle.in_kp_round() => bail!(
220            "the guardian the relay serves is in a key provisioner round ({}); roll the proxy once that round has finished",
221            relay.lifecycle
222        ),
223        Ok(_) => Ok(()),
224        Err(error) => {
225            print_warning(&format!(
226                "Could not read the guardian the relay serves, so this step can't tell whether a key provisioner round is open: {error:#}"
227            ));
228            Ok(())
229        }
230    }
231}

233/// Applies the whole enclave stack. An untargeted up re-renders every slot it plans, so it refuses a stack with two
234/// slots, and has the operator name every guardian instance it could destroy.
235pub fn deploy(ctx: &Ctx) -> Result<()> {
236    let stack = Stack::load(ctx)?;
237    print_info(&format!(
238        "Deploying the guardian enclave stack {}",
239        stack.name
240    ));
241    if stack.slots()?.len() > 1 {
242        bail!(
243            "stack {} has two guardian slots; the deploy step runs an untargeted pulumi up, which can replace either guardian: use the rotation steps instead",
244            stack.name
245        );
246    }
247    require_pulumi_version(ctx)?;
248    let plan = preview(ctx, &stack.dir, &stack.name, None)?;
249    let changed = changes(&plan);
250    for (op, urn) in &changed {
251        let parts: Vec<&str> = urn.split("::").collect();
252        print_detail(&format!(
253            "  {op}  {}",
254            parts[parts.len().saturating_sub(2)..].join("  ")
255        ));
256    }
257    print_detail(
258        &plan
259            .change_summary
260            .iter()
261            .map(|(op, count)| format!("{count} {op}"))
262            .collect::<Vec<_>>()
263            .join(", "),
264    );
265    if changed.is_empty() {
266        print_info("Nothing to change.");
267        return Ok(());
268    }
269
270    // The plan leaves out an instance it deletes because the log bucket it depends on is replaced, so any change is
271    // taken to risk every instance the stack has. An enclave holds the only live copy of its key, and naming each
272    // instance is the consent to lose it.
273    let existing = instance_ids(ctx, &stack)
274        .with_context(|| format!("could not read the instances of stack {}", stack.name))?;
275    if existing.is_empty() {
276        confirm(
277            ctx.ask,
278            &format!("Apply this plan to {}?", stack.name),
279            &NOTHING_CHANGED,
280        )?;
281    } else {
282        print_warning(
283            "This plan can destroy the guardian instances below. A destroyed guardian's key is gone from memory for good: it needs every step from provisioning again, or from the ceremony if the log bucket changes too.",
284        );
285        for id in &existing {
286            confirm_typed(ctx.ask, id, "to destroy that guardian", &NOTHING_CHANGED)?;
287        }
288    }
289    // The prompts can wait any time, and the up plans again from scratch, so the consent must still describe it.
290    let reloaded = Stack::load(ctx)?;
291    if reloaded.slots()?.len() > 1
292        || changes(&preview(ctx, &reloaded.dir, &reloaded.name, None)?) != changed
293    {
294        bail!(
295            "stack {}'s configuration or plan changed while this step waited for you; nothing was changed: run this step again",
296            stack.name
297        );
298    }
299    let now = instance_ids(ctx, &stack)
300        .with_context(|| format!("could not read the instances of stack {}", stack.name))?;
301    if now != existing {
302        bail!(
303            "stack {}'s instances changed while this step waited for you: {} then, {} now; nothing was changed: run this step again",
304            stack.name,
305            existing.join(" "),
306            now.join(" ")
307        );
308    }
309    let applied = up(ctx, &stack.dir, &stack.name, None)?;
310    let outcome = if applied {
311        "deployed, but"
312    } else {
313        "the deploy did not finish, and"
314    };
315    // Rotation state describes earlier instances, so it is stale once none of them is left. A deploy that fails
316    // partway can already have destroyed them, so this runs either way.
317    let remaining = instance_ids(ctx, &stack).with_context(|| {
318        format!(
319            "{outcome} could not read the instances of stack {}",
320            stack.name
321        )
322    })?;
323    if !existing.iter().any(|id| remaining.contains(id)) {
324        let tags = stack.rotation_tags(ctx).with_context(|| {
325            format!("{outcome} could not list the stack's rotation tags; clear them before the next rotation")
326        })?;
327        for tag in tags {
328            read(
329                ctx.exec,
330                &pulumi(&stack.dir).args(["stack", "tag", "rm", &tag, "--stack", &stack.name]),
331            )
332            .with_context(|| {
333                format!("{outcome} could not clear {tag}; clear it before the next rotation")
334            })?;
335        }
336    }
337    if !applied {
338        bail!("the deploy did not finish; preview the stack before running this step again");
339    }
340    print_success(
341        "Deployed. The host builds the enclave image at boot; follow it with the host step.",
342    );
343    next("host");
344    Ok(())
345}

347/// A plan's changes, by operation and URN, sorted: Pulumi can list one plan's steps in another order.
348fn changes(plan: &Plan) -> Vec<(String, String)> {
349    let mut changes: Vec<(String, String)> = plan
350        .steps
351        .iter()
352        .filter(|step| step.op != "same")
353        .map(|step| (step.op.clone(), step.urn.clone()))
354        .collect();
355    changes.sort();
356    changes
357}

359fn instance_ids(ctx: &Ctx, stack: &Stack) -> Result<Vec<String>> {
360    Ok(stack
361        .instances(ctx)?
362        .into_iter()
363        .filter_map(|instance| instance.id)
364        .collect())
365}

406            json!({ "hashi-commit": COMMIT, "hashi-commit-b": COMMIT, "slot-b-enabled": "true" })
407        } else {
408            json!({ "hashi-commit": COMMIT })
409        };

425        let plan = json!({ "steps": steps, "changeSummary": { "same": 15 } }).to_string();

430        let exec = Fake::new(move |words| {

480                    .ok()
481                    .map(|output| (output.success, output.stdout)),

677            }

735    /// A stack whose slot a's host was launched to build `launched[n]` as of its `n`th read, and the last one after
736    /// those, a relay whose guardian is `relayed`, and a proxy workflow that succeeds.
737    fn proxy_world(launched: &'static [&'static str], relayed: &'static str) -> Fake {
738        let stack = stack_world(json!({ "hashi-commit": COMMIT }), json!({}));
739        let reads = std::cell::Cell::new(0);
740        Fake::new(move |words| {
741            let w: Vec<&str> = words.iter().map(String::as_str).collect();
742            let ok = |stdout: &str| Some((true, stdout.to_owned()));
743            match w.as_slice() {
744                ["pulumi", "-C", _, "stack", "output", "--json", ..] => ok(&json!({
745                    "active_slot": "a",
746                    "enclave_instance_id": "i-0aaa",
747                    "s3_bucket_name": "guardian-bucket",
748                    "s3_bucket_region": "us-west-2",
749                })
750                .to_string()),
751                ["aws", "ec2", "describe-instance-attribute", ..] => {
752                    let read = reads.get();
753                    reads.set(read + 1);
754                    ok(&user_data(launched[read.min(launched.len() - 1)]))
755                }
756                [
757                    "grpcurl",
758                    ..,
759                    "sui.hashi.v1alpha.GuardianRelayService/GetProvisioningTargetInfo",
760                ] => ok(&json!({
761                    "signedInfo": {
762                        "data": crate::guardian::tests::info_json(&"aa".repeat(32), None, relayed)
763                    }
764                })
765                .to_string()),
766                ["gh", "--version"] => ok("gh version 2.87.0 (2026-01-01)\n"),
767                ["gh", "workflow", "run", ..] => {
768                    ok("https://github.com/org/infra/actions/runs/456\n")
769                }
770                ["gh", "run", "watch", "456", ..] => ok(""),
771                ["curl", .., url] if url.ends_with("/health") => ok(""),
772                ["pulumi", ..] => stack
773                    .output(&Cmd::new("pulumi").args(&words[1..]))
774                    .ok()
775                    .map(|output| (output.success, output.stdout)),
776                _ => None,
777            }
778        })
779    }
