32const ROSTER: &str = "roster.txt";

34const KP_FILE_SUFFIXES: [&str; 5] = [
35    "kp-pubkey.asc",
36    "kp-fingerprint.txt",
37    "kp-pubkey.attestation-device.pem",
38    "kp-pubkey.attestation-sig.pem",
39    "kp-pubkey.attestation-dec.pem",
40];

42#[derive(Deserialize)]
43struct Listing {
44    #[serde(rename = "Versions", default)]
45    versions: Vec<Version>,
46}
47
48#[derive(Deserialize)]
49struct Version {
50    #[serde(rename = "Key")]
51    key: String,
52    #[serde(rename = "VersionId")]
53    version_id: String,
54    #[serde(rename = "IsLatest")]
55    is_latest: bool,
56}
57
58/// What the bucket's listing holds, sorted out.
59#[derive(Default)]
60struct Uploads {
61    /// The latest version of each key provisioner file: key and version, by user ID.
62    latest: BTreeMap<String, Vec<(String, String)>>,
63    /// Key provisioner files with more than one version.
64    reuploaded: BTreeSet<String>,
65    /// The latest version of anything else.
66    unexpected: Vec<String>,
67}
68
69impl Uploads {
70    fn sort(versions: Vec<Version>) -> Self {
71        let mut uploads = Self::default();
72        let mut seen = BTreeSet::new();
73        for version in versions {
74            if [CONFIG_PREFIX, ROTATION_PREFIX]
75                .iter()
76                .any(|prefix| version.key.starts_with(&format!("{prefix}/")))
77            {
78                continue;
79            }
80            match kp_file_owner(&version.key) {
81                Some(id) => {
82                    if !seen.insert(version.key.clone()) {
83                        uploads.reuploaded.insert(version.key.clone());
84                    }
85                    if version.is_latest {
86                        uploads
87                            .latest
88                            .entry(id)
89                            .or_default()
90                            .push((version.key, version.version_id));
91                    }
92                }
93                None if version.is_latest => uploads.unexpected.push(version.key),
94                None => {}
95            }
96        }
97        uploads
98    }
99}

101/// The user ID a key provisioner file belongs to, if `key` is one: `<id>/<id>-<suffix>`. The download writes the roster
102/// beside the user IDs' directories, so none may take its name, in any case.
103fn kp_file_owner(key: &str) -> Option<String> {
104    let (id, file) = key.split_once('/')?;
105    let valid_id = !id.eq_ignore_ascii_case(ROSTER)
106        && id.starts_with(|c: char| c.is_ascii_alphanumeric())
107        && id
108            .bytes()
109            .all(|b| b.is_ascii_alphanumeric() || b"._-".contains(&b));
110    let valid_file = KP_FILE_SUFFIXES
111        .iter()
112        .any(|suffix| file == format!("{id}-{suffix}"));
113    (valid_id && valid_file).then(|| id.to_owned())
114}

116enum Status {
117    Verified(String),
118    Incomplete(String),
119    Invalid(String, Option<String>),
120}

122/// Downloads and verifies every key provisioner upload in `KP_BUCKET`, then writes the roster in the fingerprint order a
123/// ceremony deals by.
124pub fn download_kp_pubkeys(ctx: &Ctx, out: Option<PathBuf>) -> Result<()> {
125    let bucket = kp_bucket(ctx)?;
126    let out = match out {
127        Some(dir) if dir.is_absolute() => dir,
128        Some(dir) => std::env::current_dir()?.join(dir),
129        None => ctx.repo.join(".hashi/kp-pubkeys").join(format!(
130            "{bucket}-{}",
131            jiff::Timestamp::now().strftime("%Y%m%dT%H%M%SZ")
132        )),
133    };
134    print_info("Downloading the key provisioners' uploads");
135    print_info(&format!(
136        "This step downloads every key provisioner upload in s3://{},",
137        bucket
138    ));
139    print_detail(
140        "verifies each certificate and its YubiKey attestations as certificate-loading commands do,",
141    );
142    print_detail("and prints the roster once every upload verifies.");
143
144    print_info("Checking the AWS account");
145    let (account, arn) = identity(ctx)?;
146    print_detail(&format!("AWS account:  {account}\nAWS identity: {arn}"));
147
148    // Verify as the key provisioners' tools will: with the guardian's own build.
149    print_info("Building the certificate verifier");
150    let stack = Stack::load(ctx)?;
151    let view = stack.slot(ctx, stack.active()?)?;
152    let tree = Tree::at(
153        ctx,
154        &view.commit,
155        &format!("the commit slot {}'s guardian runs", view.slot),
156    )?;
157    let guardian_init = tree.guardian_init(ctx)?;
158    if !guardian_init.has_tool(ctx, "verify-kp-cert")? {
159        bail!(
160            "hashi-guardian-init at {}, slot {}'s commit, has no tools verify-kp-cert, so it cannot verify uploads",
161            view.commit,
162            view.slot
163        );
164    }
165    print_info(&format!(
166        "Verifying with hashi-guardian-init at {}, slot {}'s commit.",
167        view.commit, view.slot
168    ));
169
170    print_info("Listing the uploads");
171    let listing: Listing = read_json(
172        ctx.exec,
173        &Cmd::new("aws")
174            .args(["s3api", "list-object-versions", "--bucket", &bucket])
175            .args(["--expected-bucket-owner", &account, "--output", "json"]),
176    )
177    .with_context(|| format!("could not list s3://{}", bucket))?;
178    let uploads = Uploads::sort(listing.versions);
179    if uploads.latest.is_empty() && uploads.unexpected.is_empty() {
180        bail!("no uploads in s3://{} yet", bucket);
181    }
182    print_info(&format!("User IDs: {}", uploads.latest.len()));
183
184    let width = uploads
185        .latest
186        .keys()
187        .map(String::len)
188        .max()
189        .unwrap_or(0)
190        .max(2);
191    if !uploads.latest.is_empty() {
192        print_info("Downloading the uploads");
193        if let Some(parent) = out.parent() {
194            std::fs::create_dir_all(parent)?;
195        }
196        std::fs::create_dir(&out).with_context(|| {
197            format!(
198                "could not create a new output directory at {}",
199                out.display()
200            )
201        })?;
202        let mut count = 0;
203        for (id, files) in &uploads.latest {
204            let dir = out.join(id);
205            std::fs::create_dir_all(&dir)
206                .with_context(|| format!("could not create {}", dir.display()))?;
207            for (key, version_id) in files {
208                read(
209                    ctx.exec,
210                    &Cmd::new("aws")
211                        .args(["s3api", "get-object", "--bucket", &bucket])
212                        .args(["--expected-bucket-owner", &account, "--key", key])
213                        .args(["--version-id", version_id])
214                        .arg(out.join(key)),
215                )
216                .with_context(|| format!("could not download s3://{}/{key}", bucket))?;
217                count += 1;
218            }
219        }
220        print_success(&format!("Downloaded {count} files into {}", out.display()));
221    }
222
223    print_info("Verifying each key provisioner");
224    // Downloads land on a case-insensitive filesystem, and each ID must name one key provisioner.
225    let mut by_case: BTreeMap<String, usize> = BTreeMap::new();
226    for id in uploads.latest.keys() {
227        *by_case.entry(id.to_lowercase()).or_default() += 1;
228    }
229    let mut statuses: Vec<(String, Status)> = Vec::new();
230    for id in uploads.latest.keys() {
231        let status = verify_upload(
232            ctx,
233            &guardian_init,
234            &out,
235            id,
236            by_case[&id.to_lowercase()] > 1,
237        )?;
238        statuses.push((id.clone(), status));
239    }
240    let mut fingerprints: BTreeMap<String, usize> = BTreeMap::new();
241    for (_, status) in &statuses {
242        if let Status::Verified(fingerprint) = status {
243            *fingerprints.entry(fingerprint.clone()).or_default() += 1;
244        }
245    }
246    let mut verified = Vec::new();
247    for (id, status) in statuses {
248        let status = match status {
249            Status::Verified(fingerprint) if fingerprints[&fingerprint] > 1 => Status::Invalid(
250                format!("same certificate as another user ID ({fingerprint})"),
251                None,
252            ),
253            status => status,
254        };
255        match status {
256            Status::Verified(fingerprint) => {
257                println!(
258                    "{id:<width$}  {:<10}  {}",
259                    "VERIFIED",
260                    format_fingerprint(&fingerprint)
261                );
262                verified.push((fingerprint, id));
263            }
264            Status::Incomplete(detail) => println!("{id:<width$}  {:<10}  {detail}", "INCOMPLETE"),
265            Status::Invalid(detail, errors) => {
266                println!("{id:<width$}  {:<10}  {detail}", "INVALID");
267                for line in errors.iter().flat_map(|errors| errors.lines()) {
268                    println!("    {}", shown(line));
269                }
270            }
271        }
272    }
273
274    let reuploaded: Vec<&String> = uploads
275        .latest
276        .iter()
277        .filter(|(_, files)| {
278            files
279                .iter()
280                .any(|(key, _)| uploads.reuploaded.contains(key))
281        })
282        .map(|(id, _)| id)
283        .collect();
284    if !reuploaded.is_empty() {
285        print_info("Notes:");
286        for id in reuploaded {
287            print_detail(&format!(
288                "  {id} uploaded some files more than once; the latest upload of each file is used."
289            ));
290        }
291    }
292    if !uploads.unexpected.is_empty() {
293        print_warning(
294            "Unexpected objects (inspect them, remove them with aws s3 rm, then download again):",
295        );
296        for key in &uploads.unexpected {
297            print_detail(&format!("  s3://{bucket}/{}", shown(key)));
298        }
299    }
300    if !uploads.unexpected.is_empty()
301        || verified.is_empty()
302        || verified.len() != uploads.latest.len()
303    {
304        if out.is_dir() {
305            println!("\nDownloaded files: {}", out.display());
306        }
307        bail!(
308            "not every upload verified, so no roster was written; fix the problems above and download again"
309        );
310    }
311
312    // Share IDs follow the ascending fingerprint order in which a ceremony deals these certificates.
313    print_info("Roster");
314    verified.sort();
315    let roster: String = verified
316        .iter()
317        .enumerate()
318        .map(|(index, (fingerprint, id))| {
319            format!(
320                "{}  {id:<width$}  {}\n",
321                index + 1,
322                format_fingerprint(fingerprint)
323            )
324        })
325        .collect();
326    let roster_path = out.join(ROSTER);
327    std::fs::write(&roster_path, &roster)
328        .with_context(|| format!("could not write {}", roster_path.display()))?;
329    print!("{roster}");
330    println!(
331        "\nKey provisioners: {}\nRoster and files: {}",
332        verified.len(),
333        out.display()
334    );
335    print_success(
336        "Roster verified successfully! Post it in a code block and have each key provisioner confirm their row.",
337    );
338    Ok(())
339}

341/// Checks one key provisioner's downloaded files, as certificate-loading commands do.
342fn verify_upload(
343    ctx: &Ctx,
344    guardian_init: &GuardianInit,
345    out: &std::path::Path,
346    id: &str,
347    case_collision: bool,
348) -> Result<Status> {
349    let missing: Vec<String> = KP_FILE_SUFFIXES
350        .iter()
351        .map(|suffix| format!("{id}-{suffix}"))
352        .filter(|file| !out.join(id).join(file).is_file())
353        .collect();
354    if !missing.is_empty() {
355        return Ok(Status::Incomplete(format!("missing {}", missing.join(" "))));
356    }
357    if case_collision {
358        return Ok(Status::Invalid(
359            "another user ID differs only by letter case".to_owned(),
360            None,
361        ));
362    }
363    let output = ctx
364        .exec
365        .output(
366            &guardian_init
367                .cmd()
368                .args(["tools", "verify-kp-cert", "--kp-pgp-cert-path"])
369                .arg(out.join(id).join(format!("{id}-kp-pubkey.asc")))
370                .capture_stderr(),
371        )
372        .context("could not run the guardian's tools")?;
373    if !output.success {
374        return Ok(Status::Invalid(
375            "certificate verification failed:".to_owned(),
376            Some(output.stderr),
377        ));
378    }
379    let fingerprint = output.stdout.trim();
380    if fingerprint.len() != 40
381        || !fingerprint
382            .bytes()
383            .all(|b| b.is_ascii_digit() || (b'A'..=b'F').contains(&b))
384    {
385        return Ok(Status::Invalid(
386            format!("unexpected fingerprint: {}", shown(fingerprint)),
387            None,
388        ));
389    }
390    let recorded_path = out.join(id).join(format!("{id}-kp-fingerprint.txt"));
391    // Anything the upload holds is checked against the certificate, so bytes that aren't UTF-8 just don't match.
392    let recorded_bytes = std::fs::read(&recorded_path)
393        .with_context(|| format!("could not read {}", recorded_path.display()))?;
394    let recorded_text = String::from_utf8_lossy(&recorded_bytes).into_owned();
395    let recorded = recorded_text.lines().next().unwrap_or("");
396    if recorded != fingerprint {
397        return Ok(Status::Invalid(
398            format!(
399                "{id}-kp-fingerprint.txt ({}) does not match the certificate ({fingerprint})",
400                shown(recorded)
401            ),
402            None,
403        ));
404    }
405    Ok(Status::Verified(fingerprint.to_owned()))
406}

408/// `text` with its control characters escaped: anyone with the upload key chooses what an upload holds.
409pub(crate) fn shown(text: &str) -> String {
410    text.chars()
411        .map(|c| {
412            if c.is_control() {
413                c.escape_default().to_string()
414            } else {
415                c.to_string()
416            }
417        })
418        .collect()
419}

518                    let name = std::path::Path::new(path).file_name()?.to_str()?;
519                    let id = name.strip_suffix("-kp-pubkey.asc")?;
520                    Some(match fingerprints.get(id) {
521                        Some(fingerprint) => (true, format!("{fingerprint}\n")),
522                        None => (false, String::new()),
523                    })
524                }
525                _ => None,
526            }
527        })
528    }

596    fn writes_no_roster_past_any_one_refusal() {
597        let fingerprint = |id: &'static str, digit: &str| (id, digit.repeat(40));
598        let with =
599            |ids: &[&str], extra: &[&str]| {
600                let mut versions: Vec<serde_json::Value> =
601                    ids.iter().flat_map(|id| bundle(id)).collect();
602                versions.extend(extra.iter().map(
603                    |key| serde_json::json!({ "Key": key, "VersionId": "1", "IsLatest": true }),
604                ));
605                versions
606            };
607        let cases = [
608            (
609                "an upload missing a file",
610                bundle("alice")[1..].to_vec(),
611                vec![fingerprint("alice", "B")],
612                &[][..],
613            ),
614            (
615                "an object that is no upload",
616                with(&["alice"], &["stray.txt"]),
617                vec![fingerprint("alice", "B")],
618                &[],
619            ),
620            (
621                "one certificate under two user IDs",
622                with(&["alice", "bob"], &[]),
623                vec![fingerprint("alice", "B"), fingerprint("bob", "B")],
624                &[],
625            ),
626            (
627                "user IDs that differ only by case",
628                with(&["alice", "Alice"], &[]),
629                vec![fingerprint("alice", "B"), fingerprint("Alice", "C")],
630                &[],
631            ),
632            (
633                "a user ID the roster file would take",
634                with(&["alice", "Roster.txt"], &[]),
635                vec![fingerprint("alice", "B"), fingerprint("Roster.txt", "C")],
