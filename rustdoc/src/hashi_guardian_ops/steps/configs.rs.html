41/// Writes the slot's configuration, then has the guardian's own tools load it as the ceremony will.
42pub fn render(ctx: &Ctx, arg: Option<Slot>) -> Result<()> {
43    let (stack, view) = slot(ctx, arg)?;
44    require_re_deal_standby(ctx, &view)?;
45    print_info(&format!(
46        "Rendering slot {}'s guardian configuration",
47        view.slot
48    ));
49    let render = SlotRender::new(ctx, &stack, &view)?;
50    if view.slot != stack.active()? {
51        require_keeps_serving(ctx, &stack, &view, &render.settings)?;
52    } else {
53        require_lists_log_builds(ctx, &stack, &view, &render.settings, Vec::new())?;
54    }
55    let tree = Tree::at(
56        ctx,
57        &view.commit,
58        &format!("the commit slot {}'s guardian runs", view.slot),
59    )?;
60    let guardian_init = tree.guardian_init(ctx)?;
61    let check = guardian_init.has_tool(ctx, "check-config")?;
62    std::fs::create_dir_all(&view.dir)?;
63    render.rendered.write_unchecked(&view.dir)?;
64    if check {
65        for config in render.rendered.configs() {
66            let path = view.dir.join(Rendered::written(config));
67            // The configuration lists certificates relative to its own directory.
68            let checked = ctx.exec.output(
69                &guardian_init
70                    .cmd()
71                    .args(["tools", "check-config", "--config"])
72                    .arg(path.file_name().unwrap_or_default())
73                    .cwd(path.parent().unwrap_or(&view.dir)),
74            );
75            if !checked.as_ref().is_ok_and(|output| output.success) {
76                render.rendered.remove_configs(&view.dir)?;
77                checked.with_context(|| {
78                    format!("could not run the guardian's tools on {}", path.display())
79                })?;
80                bail!(
81                    "the guardian's tools refuse {}, so no step will run with it; fix what they name, then render again",
82                    path.display()
83                );
84            }
85        }
86    } else {
87        print_warning(
88            "This guardian's build has no tools check-config, so its tools have not checked the rendered files.",
89        );
90    }
91    render.rendered.place_configs(&view.dir)?;
92
93    let settings = &render.settings;
94    println!("Guardian commit:  {}", settings.commit);
95    println!("PCR0:             {}", settings.pcr0);
96    for build in &settings.prev_builds {
97        println!("Earlier build:    {}, PCR0 {}", build.commit, build.pcr0);
98    }
99    println!("Log bucket:       s3://{}", settings.bucket);
100    println!("Bitcoin network:  {}", settings.bitcoin_network);
101    println!("Retention class:  {}", settings.retention_environment);
102    let dir = view.dir.display();
103    match &render.new {
104        Some(new) => {
105            println!(
106                "KP-set rotation:  {} key provisioners, any {} sign, re-deal to {}, any {} provision",
107                render.dealt.kp_count(),
108                render.dealt.threshold,
109                new.kp_count(),
110                new.threshold
111            );
112            println!("For the signers:      {dir}/rotate-kp-set/guardian-init.yaml");
113            println!("For the re-deal:      {dir}/rotate-kp-set/operator.yaml");
114        }
115        None => println!(
116            "Key provisioners: {}, any {} provision",
117            render.dealt.kp_count(),
118            render.dealt.threshold
119        ),
120    }
121    println!("For key provisioners: {dir}/guardian-init.yaml");
122    println!("For the operator:     {dir}/operator.yaml");
123    next(&match render.new {
124        Some(_) => format!("publish --slot {} --rotate-kp-set", view.slot),
125        None => format!("publish --slot {}", view.slot),
126    });
127    Ok(())
128}

130/// Publishes the slot's configuration for key provisioners to the KP bucket's `_config/`, or a KP-set rotation's
131/// with `rotate_kp_set`, and prints the digest they compare with the operator's post.
132pub fn publish(ctx: &Ctx, arg: Option<Slot>, rotate_kp_set: bool) -> Result<()> {
133    if !ctx.ask.interactive() {
134        bail!("this publication is interactive and must run in a terminal");
135    }
136    let (stack, view) = slot(ctx, arg)?;
137    require_re_deal_standby(ctx, &view)?;
138    let prefix = if rotate_kp_set {
139        if ctx.env.get("NEW_KP_ROSTER_DIR").is_none() {
140            bail!(
141                "set NEW_KP_ROSTER_DIR and NEW_KP_THRESHOLD in {} and render again to publish a KP-set rotation",
142                ctx.env.path().display()
143            );
144        }
145        "rotate-kp-set/"
146    } else {
147        ""
148    };
149    let config_dir = view.dir.join(prefix);
150    let config_file = config_dir.join("guardian-init.yaml");
151    let config = config_file.display();
152    if !config_file.is_file() {
153        bail!(
154            "no {config}; run the render step for slot {} first",
155            view.slot
156        );
157    }
158    // What goes out is this render, which the slot's files are shown to equal, so nothing written to them in between
159    // can mix in. The operator's own files are never published, so they don't count.
160    let render = SlotRender::new(ctx, &stack, &view)?;
161    let published = published(&render.rendered, rotate_kp_set);
162    let stale: Vec<&str> = published
163        .iter()
164        .filter(|(path, bytes)| {
165            std::fs::read(config_dir.join(path)).ok().as_deref() != Some(*bytes)
166        })
167        .map(|(path, _)| *path)
168        .collect();
169    if !stale.is_empty() {
170        bail!(
171            "out of date in {}: {}; the stack, {} or the roster changed after the render step; run it again, then publish",
172            config_dir.display(),
173            stale.join(", "),
174            ctx.env.path().display()
175        );
176    }
177    let guardian_bucket = stack.bucket()?.name;
178    let bucket = kp_bucket(ctx)?;
179    // Publishing replaces everything under its prefix.
180    if bucket == guardian_bucket {
181        bail!(
182            "KP_BUCKET names the guardian's log bucket, s3://{bucket}; set it to the key provisioners' bucket"
183        );
184    }
185    let tree = Tree::at(
186        ctx,
187        &view.commit,
188        &format!("the commit slot {}'s guardian runs", view.slot),
189    )?;
190    // A re-deal's signers sign before `submit`, and the new set confirms once its proposal is out.
191    let re_deal_stage = || -> Result<()> {
192        if ctx.env.get("NEW_KP_ROSTER_DIR").is_none() {
193            return Ok(());
194        }
195        let slot = view.slot;
196        let lifecycle = require_reached(ctx, &stack, &view, &tree)?.info.lifecycle;
197        let next = re_deal_step(&lifecycle, slot);
198        if rotate_kp_set && !lifecycle.is("none") && !lifecycle.is("ceremony/OPERATOR_INITIALIZED")
199        {
200            bail!(
201                "slot {slot}'s guardian is {lifecycle}; the signers' configuration goes out before rotate-kp-set submit --slot {slot}; next: {next}"
202            );
203        }
204        if !rotate_kp_set && !lifecycle.is("ceremony/AWAITING_KEY_PROVISIONER_CONFIRMATIONS") {
205            bail!(
206                "slot {slot}'s guardian is {lifecycle}; during a re-deal, the new set's configuration goes out once rotate-kp-set submit --slot {slot} logs `ceremony proposal published`; next: {next}"
207            );
208        }
209        Ok(())
210    };
211    re_deal_stage()?;
212    // Key provisioners run download-config.sh at the guardian's commit, so this digest must be the one it computes.
213    let scheme = DigestScheme::of(&tree)?;
214    let guardian_init = tree.guardian_init(ctx)?;
215    let commit = &render.settings.commit;
216
217    print_info("Publishing the key provisioners' configuration");
218    print_info(&format!(
219        "This step publishes {config} and its certificates to s3://{bucket}/{CONFIG_PREFIX}/."
220    ));
221    print_detail("Key provisioners fetch them with key-provisioner/scripts/download-config.sh.");
222
223    print_info("Verifying the key provisioner certificates");
224    let scratch = tempfile::tempdir()?;
225    let publish_dir = scratch.path().join(CONFIG_PREFIX);
226    for (path, bytes) in &published {
227        let to = publish_dir.join(path);
228        std::fs::create_dir_all(to.parent().unwrap_or(&publish_dir))?;
229        std::fs::write(&to, bytes).with_context(|| format!("could not write {}", to.display()))?;
230    }
231    let mut count = 0;
232    for (path, _) in published
233        .iter()
234        .filter(|(path, _)| path.starts_with("certs/") && path.ends_with(".asc"))
235    {
236        let cert = publish_dir.join(path);
237        let fingerprint = read(
238            ctx.exec,
239            &guardian_init
240                .cmd()
241                .args(["tools", "verify-kp-cert", "--kp-pgp-cert-path"])
242                .arg(&cert),
243        )
244        .with_context(|| format!("certificate {path} did not verify, so nothing was published"))?;
245        println!(
246            "{}  {}",
247            format_fingerprint(fingerprint.trim()),
248            path.trim_start_matches("certs/")
249                .trim_end_matches("-kp-pubkey.asc")
250        );
251        count += 1;
252    }
253    let leaked = files_holding_access_keys(&publish_dir)?;
254    if !leaked.is_empty() {
255        bail!(
256            "found an AWS access key ID, or could not check for one, so nothing was published: {}",
257            leaked.join(" ")
258        );
259    }
260    let digest = scheme.digest(&publish_dir)?;
261    let digest_lines = scheme.lines(&digest);
262    print_detail(&format!(
263        "Guardian commit:      {commit}\nKey provisioners:     {count}\nConfiguration digest: {}",
264        digest_lines[0]
265    ));
266    for line in &digest_lines[1..] {
267        print_detail(&format!("                      {line}"));
268    }
269
270    print_info("Checking the AWS account");
271    let (account, arn) = identity(ctx)?;
272    print_detail(&format!("AWS account:  {account}\nAWS identity: {arn}"));
273    let head_bucket = |bucket: &str| {
274        read(
275            ctx.exec,
276            &Cmd::new("aws")
277                .args(["s3api", "head-bucket", "--bucket", bucket])
278                .args(["--expected-bucket-owner", &account]),
279        )
280    };
281    head_bucket(&bucket).with_context(|| {
282        format!("no bucket s3://{bucket} in account {account}; check KP_BUCKET")
283    })?;
284    head_bucket(&guardian_bucket).with_context(|| {
285        format!("no guardian log bucket s3://{guardian_bucket} in account {account}")
286    })?;
287    confirm(
288        ctx.ask,
289        &format!("Publish this configuration to s3://{bucket}/{CONFIG_PREFIX}/?"),
290        &Unchanged {
291            no_input: "no input received; nothing was published",
292            declined: "publication not confirmed; nothing was published",
293        },
294    )?;
295    // The question can wait any time, while another terminal moves a re-deal on.
296    re_deal_stage()?;
297
298    print_info("Uploading the configuration");
299    let prefix = format!("s3://{bucket}/{CONFIG_PREFIX}/");
300    read(
301        ctx.exec,
302        &Cmd::new("aws")
303            .args([
304                "s3",
305                "rm",
306                "--recursive",
307                "--only-show-errors",
308                "--region",
309                REGION,
310            ])
311            .arg(&prefix),
312    )
313    .with_context(|| format!("could not clear {prefix}; publish again"))?;
314    read(
315        ctx.exec,
316        &Cmd::new("aws")
317            .args([
318                "s3",
319                "cp",
320                "--recursive",
321                "--only-show-errors",
322                "--region",
323                REGION,
324            ])
325            .arg(format!("{}/", publish_dir.display()))
326            .arg(&prefix),
327    )
328    .with_context(|| format!("could not upload the configuration to {prefix}; publish again"))?;
329    print_success(&format!(
330        "Published guardian-init.yaml and {count} certificates."
331    ));
332
333    print_info("Publication complete");
334    print_info("Post the commit and the digest, so each key provisioner can compare them:");
335    println!("  Guardian commit:      {commit}");
336    println!("  Configuration digest: {}", digest_lines[0]);
337    for line in &digest_lines[1..] {
338        println!("                        {line}");
339    }
340    if scheme == DigestScheme::Legacy {
341        print_warning(
342            "This digest covers guardian-init.yaml alone, so post the fingerprints above too, for each key provisioner to compare with what the download prints.",
343        );
344    }
345    print_info("Every key provisioner runs, from a hashi checkout at that commit:");
346    print_detail("  ./key-provisioner/scripts/download-config.sh");
347    print_success(&format!(
348        "Configuration published successfully! It lists {count} key provisioners."
349    ));
350    next(&if rotate_kp_set {
351        format!(
352            "rotate-kp-set init --slot {}, once the key provisioners who sign have downloaded it",
353            view.slot
354        )
355    } else {
356        "the step this configuration is for, once every key provisioner has downloaded it"
357            .to_owned()
358    });
359    Ok(())
360}

362/// Stops while `NEW_KP_ROSTER_DIR` is set unless the slot is the standby the KP-set rotation re-deals on: any other slot
363/// would render the new set, which holds no shares until the re-deal completes.
364fn require_re_deal_standby(ctx: &Ctx, view: &SlotView) -> Result<()> {
365    if ctx.env.get("NEW_KP_ROSTER_DIR").is_some() && view.role != Role::Standby {
366        bail!(
367            "{} sets NEW_KP_ROSTER_DIR for a KP-set rotation, which re-deals on a standby, and slot {} is the {}; render and publish only the standby until the re-deal completes",
368            ctx.env.path().display(),
369            view.slot,
370            view.role
371        );
372    }
373    Ok(())
374}

376/// What `publish` sends, by its path under the KP bucket's prefix: the slot's own files, or with `rotate_kp_set` the
377/// re-deal's. The operator's files never go out.
378fn published(rendered: &Rendered, rotate_kp_set: bool) -> Vec<(&str, &[u8])> {
379    let prefix = if rotate_kp_set { "rotate-kp-set/" } else { "" };
380    rendered
381        .files()
382        .filter_map(|(path, bytes)| {
383            let path = path.strip_prefix(prefix)?;
384            let own = rotate_kp_set || !path.starts_with("rotate-kp-set/");
385            (own && !path.ends_with("operator.yaml")).then_some((path, bytes))
386        })
387        .collect()
388}

390fn files_holding_access_keys(dir: &Path) -> Result<Vec<String>> {
391    let mut leaked = Vec::new();
392    let mut pending = vec![dir.to_owned()];
393    while let Some(next) = pending.pop() {
394        for entry in std::fs::read_dir(&next)? {
395            let path = entry?.path();
396            if path.is_dir() {
397                pending.push(path);
398            } else if holds_access_key(&String::from_utf8_lossy(&std::fs::read(&path)?)) {
399                leaked.push(path.display().to_string());
400            }
401        }
402    }
403    Ok(leaked)
404}
