28/// Prints the stack's slots, the proxy, the relay and the epoch. A read that fails is shown rather than fatal, so it
29/// still reports a partly broken stack.
30pub fn status(ctx: &Ctx) -> Result<()> {
31    let stack = Stack::load(ctx)?;
32    println!("Stack:         {}", stack.name);
33    let pulumi = match (pulumi_pin(ctx), pulumi_version(ctx)) {
34        (Ok(pinned), Ok(current)) if pinned == current => {
35            format!("{current}, as PULUMI_VERSION pins")
36        }
37        (Ok(pinned), Ok(current)) => format!(
38            "{current}, but PULUMI_VERSION pins {pinned}: put it first on your PATH before a step that changes a stack"
39        ),
40        (Err(error), _) | (_, Err(error)) => format!("unknown: {error:#}"),
41    };
42    println!("Pulumi:        {pulumi}");
43    let written = match stack.written_by(ctx) {
44        Ok(written) if written.is_empty() => "unknown".to_owned(),
45        Ok(written) => written,
46        Err(error) => format!("unknown: {error:#}"),
47    };
48    println!("Written by:    Pulumi {written}");
49    let tags = match stack.rotation_tags(ctx) {
50        Ok(tags) if tags.is_empty() => "none".to_owned(),
51        Ok(tags) => tags.join(", "),
52        Err(error) => format!("unknown: {error:#}"),
53    };
54    println!("Rotation tags: {tags}");
55    for slot in Slot::ALL {
56        println!("Slot {slot}:        {}", describe_slot(&stack, slot));
57    }
58
59    let proxy = match proxy_info(ctx) {
60        Ok(info) => describe_proxy(&info),
61        Err(error) => format!("unknown: {error:#}"),
62    };
63    println!("Proxy:         {proxy}");
64    // The relay leads to a standby once the proxy follows it, so its answer is read with the standby's build.
65    let relay = stack
66        .active()
67        .and_then(|active| {
68            let read_as = match stack.role(active.other())? {
69                Role::Standby => active.other(),
70                _ => active,
71            };
72            stack.slot(ctx, read_as)
73        })
74        .and_then(|view| {
75            view.require_deployed(ctx, &stack)?;
76            Tree::at(
77                ctx,
78                &view.commit,
79                &format!("the commit slot {}'s guardian runs", view.slot),
80            )
81        })
82        .and_then(|tree| relay_info(ctx, &tree));
83    let relay = match relay {
84        Ok(info) => format!(
85            "sends key provisioners to {} ({}){}",
86            info.signing_key.as_deref().unwrap_or("no guardian"),
87            info.lifecycle,
88            if info.lifecycle.in_kp_round() {
89                "; a key provisioner round is open"
90            } else {
91                ""
92            }
93        ),
94        Err(error) => format!("unknown: {error:#}"),
95    };
96    println!("Relay:         {relay}");
97    if ctx.env.get("SUI_GRAPHQL_URL").is_some() {
98        let epoch = match sui::window(ctx, sui::now()) {
99            Ok(window) => format!(
100                "{}, began {} ago, ends in about {}{}",
101                window.epoch,
102                sui::duration(window.since),
103                sui::duration(window.remaining),
104                if window.refusal().is_none() {
105                    "; the time allows a switchover"
106                } else {
107                    "; too close to the turn for a switchover"
108                }
109            ),
110            Err(error) => format!("unknown: {error:#}"),
111        };
112        println!("Sui epoch:     {epoch}");
113    }
114    Ok(())
115}

117fn describe_slot(stack: &Stack, slot: Slot) -> String {
118    let enabled = match stack.slot_enabled(slot) {
119        Ok(true) => "enabled".to_owned(),
120        Ok(false) => "disabled".to_owned(),
121        Err(error) => format!("enabled or not unknown ({error:#})"),
122    };
123    let role = stack.role(slot);
124    let deployed = match (&role, stack.instance(slot)) {
125        (Ok(Role::Absent), _) => "not deployed".to_owned(),
126        (Ok(role), Ok(Some(instance))) => format!("{role} on {instance}"),
127        (Ok(role), _) => role.to_string(),
128        (Err(error), _) => format!("{error:#}"),
129    };
130    let suffix = slot.suffix();
131    let commit = stack.config(&format!("hashi-commit{suffix}"));
132    let build = match (commit, stack.config(&format!("eif-pcr0{suffix}"))) {
133        (None, None) => String::new(),
134        (commit, pcr0) => format!(
135            ", commit {}, PCR0 {}",
136            commit.unwrap_or("unset"),
137            pcr0.unwrap_or("unset")
138        ),
139    };
140    // The stack's last update recorded the active slot's commit; one that differs has not been applied.
141    let applied = match stack.output("guardian_hashi_commit") {
142        Some(applied) if matches!(role, Ok(Role::Active)) && commit != Some(applied) => {
143            format!(", but the stack last applied commit {applied}")
144        }
145        _ => String::new(),
146    };
147    format!("{enabled}, {deployed}{build}{applied}")
148}

150fn describe_proxy(info: &ProxyInfo) -> String {
151    let age = match info.age() {
152        Some(age) if age > CURRENT_FOR => format!(
153            ", as of {}s ago, too old to show that the proxy still reaches that guardian",
154            age.as_secs()
155        ),
156        Some(age) => format!(", as of {}s ago", age.as_secs()),
157        None => ", as of an unknown time".to_owned(),
158    };
159    format!(
160        "fronts {} (BTC key {}, build {}), {}{age}",
161        info.signing_pub_key.as_deref().unwrap_or("no guardian"),
162        info.btc_pubkey.as_deref().unwrap_or("none"),
163        info.git_revision.as_deref().unwrap_or("unknown"),
164        match &info.committee_epoch {
165            Some(epoch) => format!("serving at committee epoch {epoch}"),
166            None => "not activated".to_owned(),
167        }
168    )
169}

171/// Prints a slot's guardian through its tunnel, once the proxy is shown to lead to it.
172pub fn info(ctx: &Ctx, arg: Option<Slot>) -> Result<()> {
173    let (stack, view) = slot(ctx, arg)?;
174    let tree = Tree::at(
175        ctx,
176        &view.commit,
177        &format!("the commit slot {}'s guardian runs", view.slot),
178    )?;
179    let reached = require_reached(ctx, &stack, &view, &tree)?;
180    let url = ctx.env.proxy_url()?.url;
181    println!(
182        "Slot:        {}, {}, on {}",
183        view.slot,
184        view.role,
185        view.instance.as_deref().unwrap_or("")
186    );
187    println!("Signing key: {}", reached.signing_key);
188    println!(
189        "BTC key:     {}",
190        reached
191            .info
192            .btc_key
193            .as_deref()
194            .unwrap_or("none; the guardian is not provisioned")
195    );
196    println!("Lifecycle:   {}", reached.info.lifecycle);
197    match &reached.proxy {
198        Some(proxy) => {
199            println!("Proxy:       fronts this guardian at {url}");
200            match &proxy.committee_epoch {
201                Some(epoch) => println!("Serving:     committee epoch {epoch}"),
202                None => println!("Serving:     no; the guardian is not activated"),
203            }
204        }
205        None => println!("Relay:       sends key provisioners to this guardian at {url}"),
206    }
207    Ok(())
208}

210/// Prints a slot's host: the end of its boot log, its enclave and its units.
211pub fn host(ctx: &Ctx, arg: Option<Slot>) -> Result<()> {
212    let (stack, view) = slot(ctx, arg)?;
213    let bucket = stack.bucket()?;
214    let instance = view.require_instance(&stack)?;
215    println!("Guardian host {instance}, slot {}", view.slot);
216    let commands = host::report_commands();
217    let commands: Vec<&str> = commands.iter().map(String::as_str).collect();
218    let report = host::run(ctx, &bucket.region, instance, &commands)?;
219    println!("{}", report.text);
220    if !report.success {
221        bail!("could not read the host; a new instance takes a minute to accept commands");
222    }
223    // Only a cross-check: key provisioners' tools verify PCR0 from a fresh attestation.
224    if let Some(reported) = host::reported_pcr0(&report.text) {
225        if Some(reported) == stack.config(&view.pcr0_key) {
226            println!(
227                "\nThe enclave reports the PCR0 that {} configures.",
228                view.pcr0_key
229            );
230        } else {
231            print_warning(&format!(
232                "The enclave reports PCR0 {reported}, not the {} stack {} configures. Key provisioners' tools will refuse it.",
233                view.pcr0_key, stack.name
234            ));
235        }
236    }
237    Ok(())
238}

240/// Forwards the slot's local port to its guardian. It runs until stopped, so it gets a terminal of its own.
241pub fn tunnel(ctx: &Ctx, arg: Option<Slot>) -> Result<()> {
242    let (stack, view) = slot(ctx, arg)?;
243    let bucket = stack.bucket()?;
244    let instance = view.require_instance(&stack)?;
245    print_info(&format!(
246        "Forwarding {} to slot {}'s guardian on {instance}",
247        view.endpoint, view.slot
248    ));
249    print_info("Leave this running in its own terminal while operator steps run.");
250    let error = ctx.exec.exec(
251        &Cmd::new("aws")
252            .args(["ssm", "start-session", "--region", &bucket.region])
253            .args(["--target", instance])
254            .args(["--document-name", "AWS-StartPortForwardingSession"])
255            .arg("--parameters")
256            .arg(format!("portNumber=3000,localPortNumber={}", view.port)),
257    );
258    Err(anyhow::Error::new(error).context("could not start the tunnel"))
259}

286    /// [`world`], with the proxy's view `age` old, or undated.
287    fn world_as_of(fronted: &'static str, relayed: &'static str, age: Option<Duration>) -> Fake {
288        let stack = crate::stack::tests::two_slots();
289        let answered = age.map(|age| {
290            (SystemTime::now() - age)
291                .duration_since(UNIX_EPOCH)
292                .unwrap()
293                .as_millis()
294                .to_string()
295        });
296        Fake::new(move |words| {
297            if words[0] == "pulumi" {
298                return crate::exec::Exec::output(&stack, &Cmd::new("pulumi").args(&words[1..]))
299                    .ok()
300                    .map(|output| (output.success, output.stdout));
301            }
302            let last = words.last().unwrap().as_str();
303            match (words[0].as_str(), last) {
304                ("grpcurl", "sui.hashi.v1alpha.GuardianService/GetGuardianInfo") => {
305                    let key = if words.contains(&"127.0.0.1:3000".to_owned()) {
306                        KEY_A
307                    } else {
308                        KEY_B
309                    };
310                    Some((
311                        true,
312                        json!({ "info": info_json(key, None, "none") }).to_string(),
313                    ))
314                }
315                ("grpcurl", "sui.hashi.v1alpha.GuardianRelayService/GetProvisioningTargetInfo") => {
316                    Some((
317                        true,
318                        json!({ "signedInfo": { "data": info_json(relayed, None, "none") } })
319                            .to_string(),
320                    ))
321                }
322                ("curl", url) if url.ends_with("/info") => {
323                    let mut info = json!({ "signingPubKey": fronted, "committeeEpoch": "1253" });
324                    if let Some(answered) = &answered {
325                        info["timestampMs"] = answered.clone().into();
326                    }
327                    Some((true, info.to_string()))
328                }
329                ("aws", _) if words[1] == "ec2" => Some((true, user_data(COMMIT))),
330                _ => None,
331            }
332        })
333    }

335    /// A world where `stack`'s configuration and outputs load, its hosts build `launched` (or can't be read), and every
336    /// other read fails. Any other command is a change, which fails the test.
337    fn only_the_stack(stack: Fake, launched: Option<&'static str>) -> Fake {
338        Fake::new(move |words| {
339            let w: Vec<&str> = words.iter().map(String::as_str).collect();
340            match w.as_slice() {
341                ["pulumi", "-C", _, "config", ..] | ["pulumi", "-C", _, "stack", "output", ..] => {
342                    crate::exec::Exec::output(&stack, &Cmd::new("pulumi").args(&words[1..]))
343                        .ok()
344                        .map(|output| (output.success, output.stdout))
345                }
346                ["aws", "ec2", "describe-instance-attribute", ..] => Some(
347                    launched.map_or((false, String::new()), |commit| (true, user_data(commit))),
348                ),
349                ["pulumi", "version"]
350                | ["pulumi", "-C", _, "stack", "export", ..]
351                | ["pulumi", "-C", _, "stack", "tag", "ls", ..]
352                | ["curl", ..] => Some((false, String::new())),
353                _ => None,
354            }
355        })
356    }

424    fn reads_no_guardian_whose_host_builds_another_commit() {
425        // Until the stack deploys a slot's new commit, its host runs the old one, which the new commit's protos may
426        // misread. Both hosts here build COMMIT.
427        let pending = "c4".repeat(20);
428        let outputs = json!({
429            "active_slot": "a",
430            "enclave_instance_id": "i-0aaa",
431            "standby_instance_id": "i-0bbb",
432            "s3_bucket_name": "guardian-bucket",
433            "s3_bucket_region": "us-west-2",
434        });
435        let mut no_standby = outputs.clone();
436        no_standby["standby_instance_id"] = "".into();
437        for (slot, host, config, outputs) in [
438            (
439                Slot::A,
440                "i-0aaa",
441                json!({ "hashi-commit": pending }),
442                no_standby,
443            ),
444            (
445                Slot::B,
446                "i-0bbb",
447                json!({ "hashi-commit": COMMIT, "hashi-commit-b": pending, "slot-b-enabled": "true" }),
448                outputs,
449            ),
450        ] {
451            let repo = tempfile::tempdir().unwrap();
452            let exec = only_the_stack(
453                crate::stack::tests::stack_world(config, outputs),
454                Some(COMMIT),
455            );
456            let ctx = test_ctx(repo.path(), &exec);
457            std::fs::create_dir_all(repo.path().join(".hashi/guardian-init").join(&pending))
458                .unwrap();
459            assert_eq!(
460                info(&ctx, Some(slot)).unwrap_err().to_string(),
461                format!(
462                    "stack org/testnet configures commit {pending} for slot {slot}, but its host {host} was launched to build another commit; set it back, or deploy it, first"
463                )
464            );
465            // status reads the relay as this slot, and grpcurl would fail the test.
466            status(&ctx).unwrap();
467        }
468    }
