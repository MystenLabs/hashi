53/// Runs `operator rotate-kp-set <action>` on the slot's standby with the re-deal's files, which list both rosters.
54pub fn rotate_kp_set(ctx: &Ctx, arg: Option<Slot>, action: Action) -> Result<()> {
55    let (stack, view) = slot(ctx, arg)?;
56    let slot = view.slot;
57    if ctx.env.get("NEW_KP_ROSTER_DIR").is_none() {
58        bail!(
59            "set NEW_KP_ROSTER_DIR and NEW_KP_THRESHOLD in {}, then render, for a KP-set rotation",
60            ctx.env.path().display()
61        );
62    }
63    if view.role != Role::Standby {
64        bail!(
65            "a KP-set rotation re-deals on a standby guardian, and slot {slot}'s is {}",
66            match view.role {
67                Role::Absent => "not deployed".to_owned(),
68                role => role.to_string(),
69            }
70        );
71    }
72    let tree = Tree::at(
73        ctx,
74        &view.commit,
75        &format!("the commit slot {slot}'s guardian runs"),
76    )?;
77    let guardian_init = tree.guardian_init(ctx)?;
78    require_fresh(ctx, &stack, &view)?;
79    let reached = require_reached(ctx, &stack, &view, &tree)?;
80    require_monitor_allows(ctx, &stack, &[&view])?;
81    let lifecycle = &reached.info.lifecycle;
82    if lifecycle.is("ceremony/COMPLETED") {
83        bail!(
84            "slot {slot}'s guardian has completed the re-deal; switch your environment file to the new set, then new-session --slot {slot}"
85        );
86    }
87    let awaiting = lifecycle.is("ceremony/AWAITING_KEY_PROVISIONER_CONFIRMATIONS");
88    match &action {
89        Action::Init | Action::Submit(_) if awaiting => bail!(
90            "slot {slot}'s guardian has the signers' files and waits for the new set's confirmations: rotate-kp-set wait --slot {slot}"
91        ),
92        Action::Submit(_) if !lifecycle.is("ceremony/OPERATOR_INITIALIZED") => bail!(
93            "slot {slot}'s guardian is {lifecycle}, not initialized for a re-deal; next: {}",
94            re_deal_step(lifecycle, slot)
95        ),
96        Action::Wait if !awaiting => bail!(
97            "slot {slot}'s guardian is {lifecycle}, waiting for no confirmations; next: {}",
98            re_deal_step(lifecycle, slot)
99        ),
100        _ => {}
101    }
102    let mut submissions = Vec::new();
103    match &action {
104        Action::Submit(files) if files.is_empty() => {
105            if !uploads_submissions(&tree) {
106                bail!(
107                    "slot {slot}'s build doesn't upload the signers' submissions to the KP bucket; name the files they sent: rotate-kp-set submit --slot {slot} --submission <file>, once per file"
108                );
109            }
110            submissions = uploaded_submissions(ctx, &view, &reached.signing_key[..16])?;
111        }
112        Action::Submit(files) => {
113            for file in files {
114                if !file.is_file() {
115                    bail!("no submission file at {}", file.display());
116                }
117                // The CLI runs from the re-deal's directory, so it gets every path absolute.
118                submissions.push(
119                    std::fs::canonicalize(file)
120                        .with_context(|| format!("could not resolve {}", file.display()))?,
121                );
122            }
123        }
124        Action::Init => {
125            if !lifecycle.is("none") && !lifecycle.is("ceremony/OPERATOR_INITIALIZED") {
126                bail!(
127                    "slot {slot}'s guardian is {lifecycle}; a KP-set rotation needs a fresh guardian: start one with new-session --slot {slot}"
128                );
129            }
130            print_warning(&format!(
131                "This makes slot {slot}'s guardian a ceremony guardian for good: it never serves withdrawals. Once the new key provisioners have all confirmed, start a fresh session on it with new-session --slot {slot}."
132            ));
133            confirm(
134                ctx.ask,
135                &format!("Re-deal the key on slot {slot}'s guardian?"),
136                &NOTHING_CHANGED,
137            )?;
138            // The question can wait any time, while the files, the tunnel, the guardian or the monitor change.
139            require_fresh(ctx, &stack, &view)?;
140            let now = require_reached(ctx, &stack, &view, &tree)?;
141            if now.signing_key != reached.signing_key || now.info.lifecycle != *lifecycle {
142                bail!(
143                    "slot {slot}'s guardian changed while this step waited for you; nothing was changed: run this step again"
144                );
145            }
146            require_monitor_allows(ctx, &stack, &[&view])?;
147        }
148        Action::Wait => {}
149    }
150    next(&match action {
151        Action::Init if uploads_submissions(&tree) => format!(
152            "once the signers have run key-provisioner rotate-kp-set, rotate-kp-set submit --slot {slot}, which collects what they uploaded to the KP bucket"
153        ),
154        Action::Init => {
155            format!(
156                "once the signers have sent their files, rotate-kp-set submit --slot {slot} --submission <file>, once per file"
157            )
158        }
159        // The re-deal leaves the slot's own files as they are, but a render with NEW_KP_* set would point at the
160        // signers' publication, whose old roster the new set can't provision with.
161        Action::Submit(_) | Action::Wait => format!(
162            "once this logs `ceremony proposal published`, publish --slot {slot} in another terminal, and have every key provisioner of the new set confirm into a new --encrypted-shares-path, so an earlier ceremony's file stays. Once this reports the rotation complete, set KP_ROSTER_DIR and KP_THRESHOLD to the new set's and remove NEW_KP_ROSTER_DIR and NEW_KP_THRESHOLD, without publishing again; then new-session --slot {slot}"
163        ),
164    });
165    let mut cmd = guardian_init
166        .cmd()
167        .args([
168            "operator",
169            "rotate-kp-set",
170            action.name(),
171            "--config",
172            "operator.yaml",
173        ])
174        // The configuration lists certificates relative to its own directory.
175        .cwd(view.dir.join("rotate-kp-set"));
176    for submission in &submissions {
177        cmd = cmd.arg("--submission").arg(submission);
178    }
179    Err(anyhow::Error::new(ctx.exec.exec(&cmd)).context("could not run hashi-guardian-init"))
180}

182/// The step a re-deal's standby is ready for, by its lifecycle.
183pub(crate) fn re_deal_step(lifecycle: &Lifecycle, slot: Slot) -> String {
184    match lifecycle.to_string().as_str() {
185        "none" => format!("rotate-kp-set init --slot {slot}"),
186        "ceremony/OPERATOR_INITIALIZED" => format!("rotate-kp-set submit --slot {slot}"),
187        "ceremony/AWAITING_KEY_PROVISIONER_CONFIRMATIONS" => {
188            format!("rotate-kp-set wait --slot {slot}")
189        }
190        "ceremony/COMPLETED" => {
191            format!("switch your environment file to the new set, then new-session --slot {slot}")
192        }
193        _ => format!("new-session --slot {slot}, for a fresh guardian to re-deal on"),
194    }
195}

197/// Whether the build's key provisioner command uploads its submission to the KP bucket, which it does from hashi #1464
198/// on.
199fn uploads_submissions(tree: &Tree) -> bool {
200    std::fs::read_to_string(
201        tree.dir
202            .join("crates/hashi-guardian-init/src/submission.rs"),
203    )
204    .is_ok_and(|source| source.contains(&format!("\"{ROTATION_PREFIX}\"")))
205}

207/// Downloads the submissions the signers uploaded for `session`, in place of an earlier download, and stops unless
208/// they reach the dealt set's threshold.
209fn uploaded_submissions(ctx: &Ctx, view: &SlotView, session: &str) -> Result<Vec<PathBuf>> {
210    let source = format!("s3://{}/{ROTATION_PREFIX}/{session}/", kp_bucket(ctx)?);
211    let dir = view
212        .dir
213        .join("rotate-kp-set")
214        .join("submissions")
215        .join(session);
216    if dir.exists() {
217        std::fs::remove_dir_all(&dir)?;
218    }
219    std::fs::create_dir_all(&dir)?;
220    print_info("Collecting the signers' submissions");
221    read(
222        ctx.exec,
223        &Cmd::new("aws")
224            .args(["s3", "cp", "--recursive", "--only-show-errors"])
225            .args(["--region", REGION])
226            .arg(&source)
227            .arg(format!("{}/", dir.display())),
228    )
229    .with_context(|| format!("could not download the submissions in {source}"))?;
230    let mut files = Vec::new();
231    for entry in std::fs::read_dir(&dir)? {
232        let entry = entry?;
233        let path = entry.path();
234        if entry.file_type()?.is_file()
235            && path
236                .extension()
237                .is_some_and(|extension| extension == "rotation")
238        {
239            files.push(std::fs::canonicalize(&path)?);
240        }
241    }
242    files.sort();
243    let signers: Vec<String> = files
244        .iter()
245        .filter_map(|file| file.file_stem())
246        .map(|stem| shown(&stem.to_string_lossy()))
247        .collect();
248    let threshold = threshold(ctx, "KP_THRESHOLD")?;
249    if signers.len() < threshold as usize {
250        bail!(
251            "found {} of the {threshold} submissions the current set's threshold needs in {source}{}; run this again once more key provisioners have run key-provisioner rotate-kp-set",
252            signers.len(),
253            if signers.is_empty() {
254                String::new()
255            } else {
256                format!(", from {}", signers.join(", "))
257            }
258        );
259    }
260    print_info(&format!("Submissions in {source}, by signer:"));
261    for signer in &signers {
262        print_detail(&format!("  {signer}"));
263    }
264    Ok(files)
265}

615            .ends_with(

619        // The active slot keeps the roster that holds the key.

637            run.step(&[], |ctx| crate::steps::configs::publish(
638                ctx,
639                Some(Slot::B),
640                false
641            ))
642            .0
643            .unwrap_err()
