104/// Polls `check` until it returns something, up to two minutes.
105fn poll<T>(mut check: impl FnMut() -> Option<T>) -> Option<T> {
106    for attempt in 0..POLLS {
107        if let Some(found) = check() {
108            return Some(found);
109        }
110        if attempt + 1 < POLLS {
111            std::thread::sleep(POLL_INTERVAL);
112        }
113    }
114    None
115}

117/// What a host says of its guardian unit: `GUARDIAN_RUNNING` while it is enabled or active, either of which can start
118/// a session, and `GUARDIAN_STOPPED` otherwise.
119fn guardian_unit_state(ctx: &Ctx, region: &str, instance: &str) -> Result<String> {
120    let state = host::run(ctx, region, instance, &[GUARDIAN_UNIT_STATE])?;
121    if !state.success {
122        bail!("{}", state.text.trim());
123    }
124    Ok(state.text)
125}

127/// Creates a standby guardian host in `slot`, next to the serving one in the other slot.
128pub fn arm(ctx: &Ctx, slot: Slot) -> Result<()> {
129    let (stack, arming) = arm_checks(ctx, slot, true)?;
130    let changes = preview_slot(ctx, &stack, &arming.urn)?;
131    if changes != [Change::new("create", &arming.urn)] {
132        bail!(
133            "arming slot {slot} must create its instance and change nothing else, but this plan does not; nothing was changed"
134        );
135    }
136    confirm(
137        ctx.ask,
138        &format!("Create slot {slot}'s guardian host?"),
139        &NOTHING_CHANGED,
140    )?;
141    // The question can wait any time, while the stack's configuration, its plan or the serving guardian changes.
142    let (stack, now) = arm_checks(ctx, slot, false)?;
143    if now != arming
144        || preview(ctx, &stack.dir, &stack.name, Some(&arming.urn))?.changes() != changes
145    {
146        bail!(
147            "slot {slot}'s plan changed while this step waited for you; nothing was changed: run this step again"
148        );
149    }
150    require_no_guardian_runs(ctx)?;
151    let Arming {
152        active,
153        serving,
154        urn,
155    } = arming;
156    up_slot(ctx, &stack, &urn)?;
157    let after = Stack::load(ctx)?;
158    if after.output("enclave_instance_id") != Some(serving.as_str()) || after.active()? != active {
159        bail!(
160            "stack {} no longer shows slot {active}'s guardian {serving} as active; check its outputs before going on",
161            stack.name
162        );
163    }
164    let Some(standby) = after.output("standby_instance_id") else {
165        bail!(
166            "stack {} lists no standby after arming slot {slot}; check its outputs",
167            stack.name
168        );
169    };
170    // Only the program knew which instance it would create, so check it is the standby the outputs name.
171    if after.instance_urn(ctx, standby)?.as_deref() != Some(urn.as_str()) {
172        bail!(
173            "stack {}'s standby {standby} is not the instance arming slot {slot} created, {urn}; check the stack before going on",
174            stack.name
175        );
176    }
177    print_success(&format!(
178        "Armed slot {slot} on {standby}. The host builds the enclave image at boot; follow it with host --slot {slot}, then run proxy-update so the relay reaches it."
179    ));
180    next(&format!(
181        "host --slot {slot}, until its boot completes; then proxy-update"
182    ));
183    Ok(())
184}

194/// Checks the stack, the serving guardian and the workflows for arming `slot`, and finds the instance it would create.
195fn arm_checks(ctx: &Ctx, slot: Slot, announce: bool) -> Result<(Stack, Arming)> {
196    let stack = Stack::load(ctx)?;
197    let view = stack.slot(ctx, slot)?;
198    stack.bucket()?;
199    let active = stack.active()?;
200    let serving = stack.output("enclave_instance_id");
201    if announce {
202        print_info(&format!(
203            "Arming slot {slot} next to slot {active}'s guardian"
204        ));
205    }
206    if view.role != Role::Absent {
207        bail!("slot {slot} already runs a guardian, as the {}", view.role);
208    }
209    require_pulumi_version(ctx)?;
210    require_no_guardian_runs(ctx)?;
211    let Some(serving) = serving.filter(|_| active != slot) else {
212        bail!(
213            "slot {slot} is the stack's active slot; arm adds a standby next to a serving guardian"
214        );
215    };
216    if !stack.slot_enabled(slot)? {
217        bail!(
218            "enable slot {slot} first: set slot-{slot}-enabled to \"true\" in stack {}'s Pulumi.<stack>.yaml",
219            stack.name
220        );
221    }
222    let configured_active = stack.config("active-slot").unwrap_or("a");
223    if configured_active != active.to_string() {
224        bail!(
225            "stack {} configures active-slot {configured_active}, but slot {active} is active; set it back before arming",
226            stack.name
227        );
228    }
229    let pcr0 = stack.config(&view.pcr0_key).unwrap_or("");
230    if !is_hex(pcr0, 96) {
231        bail!(
232            "stack {}'s {} is not a PCR0 measurement: {}; measure slot {slot}'s build with measure --slot {slot}, and set it to the PCR0 that prints",
233            stack.name,
234            view.pcr0_key,
235            if pcr0.is_empty() { "unset" } else { pcr0 }
236        );
237    }
238    let settings = Settings::read(ctx, &stack, &view)?;
239    require_proxy_decodes(ctx, &stack, &view)?;
240    let (suffix, active_suffix) = (slot.suffix(), active.suffix());
241    for key in ["eif-features", "eif-ceremony-mode"] {
242        if stack.config(&format!("{key}{suffix}")).is_some()
243            || stack.config(&format!("{key}{active_suffix}")).is_some()
244        {
245            bail!(
246                "stack {} sets {key} for a slot; a real guardian builds with neither: unset it",
247                stack.name
248            );
249        }
250    }
251    let tags = stack.rotation_tags(ctx)?;
252    if !tags.is_empty() {
253        bail!(
254            "stack {} has rotation tags, which hold rotation state apart from its configuration: {}; this flow keeps it in Pulumi.<stack>.yaml, so clear them once you know the yaml says the same",
255            stack.name,
256            tags.join(", ")
257        );
258    }
259    serving_info(ctx).context("arm a standby only next to a serving guardian")?;
260    require_keeps_serving(ctx, &stack, &view, &settings)?;
261    let urn = instance_to_create(ctx, &stack, slot)?;
262    let serving = serving.to_owned();
263    Ok((
264        stack,
265        Arming {
266            active,
267            serving,
268            urn,
269        },
270    ))
271}

273/// The URN of the instance the stack would create for `slot`: the one instance an untargeted plan creates, since only
274/// the program knows its name. Anything else that plan creates is refused, as the targeted plan arming applies leaves
275/// it out, though the new host may need it.
276fn instance_to_create(ctx: &Ctx, stack: &Stack, slot: Slot) -> Result<String> {
277    let plan = preview(ctx, &stack.dir, &stack.name, None)?;
278    let (instances, others): (Vec<&str>, Vec<&str>) = plan
279        .creates()
280        .into_iter()
281        .partition(|urn| urn_type(urn) == Some(INSTANCE));
282    let [urn] = instances[..] else {
283        bail!(
284            "stack {}'s plan creates {} guardian instances, not slot {slot}'s one: {}",
285            stack.name,
286            instances.len(),
287            instances.join(" ")
288        );
289    };
290    if !others.is_empty() {
291        bail!(
292            "stack {} also plans to create resources that arming slot {slot} would leave out; create them first, from {}: pulumi up --stack {} {}",
293            stack.name,
294            stack.dir,
295            stack.name,
296            others
297                .iter()
298                .map(|other| format!("--target {other}"))
299                .collect::<Vec<_>>()
300                .join(" ")
301        );
302    }
303    Ok(urn.to_owned())
304}

306/// Stops unless the proxy's build decodes what the slot's guardian sends, as it must to relay or front it. A guardian
307/// on the active slot's build needs no check: the proxy already serves that one.
308fn require_proxy_decodes(ctx: &Ctx, stack: &Stack, view: &SlotView) -> Result<()> {
309    let active = stack.active()?;
310    if stack.config(&format!("hashi-commit{}", active.suffix())) != Some(view.commit.as_str()) {
311        require_same_formats(
312            ctx,
313            &view.commit,
314            &image_commit(ctx, stack)?,
315            "the proxy's build",
316        )?;
317    }
318    Ok(())
319}

321/// Applies the proxy stack so the proxy follows the enclave stack's slots.
322pub fn proxy_update(ctx: &Ctx) -> Result<()> {
323    let stack = Stack::load(ctx)?;
324    print_info(&format!(
325        "Updating the proxy of stack {} to follow its guardian slots",
326        stack.name
327    ));
328    require_no_guardian_runs(ctx)?;
329    let other = stack.active()?.other();
330    if stack.role(other)? == Role::Standby {
331        require_proxy_decodes(ctx, &stack, &stack.slot(ctx, other)?)?;
332    }
333    let view = stack.slot(ctx, stack.active()?)?;
334    update_proxy(
335        ctx,
336        &stack,
337        &tree(ctx, &view)?,
338        &NOTHING_CHANGED,
339        &|| Ok(()),
340    )?;
341    print_success(&format!(
342        "The proxy answers at {}.",
343        ctx.env.proxy_url()?.url
344    ));
345    Ok(())
346}

348/// Makes the slot's provisioned standby the serving guardian: flips the active slot, points the proxy at it, stops the
349/// old guardian for good, and activates the new one. Each phase is decided from live state, so a rerun resumes where
350/// an earlier run stopped.
351pub fn switchover(ctx: &Ctx, slot_arg: Slot) -> Result<()> {
352    let mut stack = Stack::load(ctx)?;
353    let mut view = stack.slot(ctx, slot_arg)?;
354    let bucket = stack.bucket()?;
355    let slot = view.slot;
356    let other = slot.other();
357    let tree = tree(ctx, &view)?;
358    let guardian_init = tree.guardian_init(ctx)?;
359    // The stack reports only an enabled slot's instance, and every run must find the old guardian to stop it.
360    if !stack.slot_enabled(other)? {
361        bail!(
362            "keep slot-{other}-enabled \"true\" in stack {}'s Pulumi.<stack>.yaml until slot {other}'s teardown: switchover finds the guardian it stops through it",
363            stack.name
364        );
365    }
366    // One reading of the epoch sets this run's deadlines, which each phase checks right before it acts.
367    let window = sui::window(ctx, sui::now())?;
368
369    if stack.active()? != slot {
370        print_info(&format!(
371            "Checking that slot {slot}'s guardian is ready to serve"
372        ));
373        if view.role != Role::Standby {
374            bail!("slot {slot} runs no standby guardian to switch over to");
375        }
376        require_fresh(ctx, &stack, &view)?;
377        let reached = require_reached(ctx, &stack, &view, &tree)?;
378        let lifecycle = &reached.info.lifecycle;
379        if !lifecycle.is("withdraw/PROVISIONER_INITIALIZED") {
380            bail!(
381                "slot {slot}'s guardian is {lifecycle}, not provisioned and waiting for activation"
382            );
383        }
384        let serving = serving_info(ctx)?;
385        let standby_btc = reached.info.btc_key.as_deref();
386        if standby_btc.is_none() || standby_btc != serving.btc_pubkey.as_deref() {
387            bail!(
388                "slot {slot}'s guardian holds BTC key {}, but the serving guardian holds {}",
389                standby_btc.unwrap_or("none"),
390                serving.btc_pubkey.as_deref().unwrap_or("none")
391            );
392        }
393        let serving_key = serving
394            .signing_pub_key
395            .as_deref()
396            .filter(|key| crate::stack::is_hex(key, 64))
397            .ok_or_else(|| anyhow!("the proxy's /info names no serving guardian's signing key"))?;
398        let serving_init = OperatorInit::read(ctx, &bucket, serving_key)?;
399        let standby_config = reached.info.config_hash.as_deref();
400        if standby_config != Some(serving_init.config_hash.as_str()) {
401            // The hash covers the PCR allowlist too, which a rotation onto another build extends.
402            let standby_init = OperatorInit::read(ctx, &bucket, &reached.signing_key)?;
403            if standby_config != Some(standby_init.config_hash.as_str())
404                || !standby_init.keeps(&serving_init)
405            {
406                bail!(
407                    "slot {slot}'s guardian has config hash {}, but the serving guardian has {}; their limiter or deployment block differ, and switching over would change them",
408                    standby_config.unwrap_or("none"),
409                    serving_init.config_hash
410                );
411            }
412        }
413        require_initialized_as_rendered(ctx, &stack, &view, &bucket, &reached.signing_key)?;
414        // The monitor reads both builds' records: the serving one's through the switchover and its re-audits after.
415        require_monitor_allows(ctx, &stack, &[&view, &stack.slot(ctx, other)?])?;
416        require_proxy_decodes(ctx, &stack, &view)?;
417        require_pulumi_version(ctx)?;
418        require_no_guardian_runs(ctx)?;
419        require_epoch_window(&window, serving.committee_epoch.as_deref())?;
420        if stack.config("active-slot") != Some(&slot.to_string()) {
421            bail!(
422                "set active-slot to {slot} in stack {}'s Pulumi.<stack>.yaml first, and commit it",
423                stack.name
424            );
425        }
426
427        print_info(&format!("Making slot {slot} the stack's active slot"));
428        let old_instance = stack.output("enclave_instance_id").map(str::to_owned);
429        let instance = view.require_instance(&stack)?.to_owned();
430        let Some(urn) = stack.instance_urn(ctx, &instance)? else {
431            bail!(
432                "could not find slot {slot}'s instance {instance} in stack {}'s state",
433                stack.name
434            );
435        };
436        if !preview_slot(ctx, &stack, &urn)?.is_empty() {
437            bail!(
438                "flipping the active slot must change no resource, but this plan does; nothing was changed"
439            );
440        }
441        print_warning(
442            "From here until activation, nodes' new withdrawals fail and are retried: through the proxy's update, then the ten minutes activation waits once the old guardian stops, about 15 to 20 minutes in all. Until the old guardian stops, you can still roll back to it.",
443        );
444        confirm(
445            ctx.ask,
446            &format!("Make slot {slot} the active slot?"),
447            &NOTHING_CHANGED,
448        )?;
449        // The question can wait any time, while a yaml edit or a workflow changes what the flip would apply.
450        if !preview(ctx, &stack.dir, &stack.name, Some(&urn))?
451            .changes()
452            .is_empty()
453        {
454            bail!(
455                "slot {slot}'s plan changed while this step waited for you; nothing was changed: run this step again"
456            );
457        }
458        require_no_guardian_runs(ctx)?;
459        up_slot(ctx, &stack, &urn)?;
460        stack = Stack::load(ctx)?;
461        // A target that names nothing applies cleanly too, so check the flip happened before relying on it.
462        if stack.active()? != slot
463            || stack.output("enclave_instance_id") != Some(instance.as_str())
464            || stack.output("standby_instance_id").map(str::to_owned) != old_instance
465        {
466            bail!(
467                "stack {}'s outputs did not flip to slot {slot}; nothing else was changed: check its outputs",
468                stack.name
469            );
470        }
471        view = stack.slot(ctx, slot)?;
472    }
473
474    let info = slot_info(ctx, &tree, &view)?;
475    let (Some(signing_key), Some(slot_btc)) = (info.signing_key.clone(), info.btc_key.clone())
476    else {
477        bail!(
478            "slot {slot}'s guardian did not report its signing and BTC keys at {}",
479            view.endpoint
480        );
481    };
482    let activated = info.lifecycle.is("withdraw/ACTIVATED");
483    if !activated {
484        // A rerun can come long after the checks before the flip.
485        require_no_guardian_runs(ctx)?;
486        require_initialized_as_rendered(ctx, &stack, &view, &bucket, &signing_key)?;
487    }
488    let fronts = |info: &ProxyInfo| info.signing_pub_key.as_deref() == Some(signing_key.as_str());
489    if !proxy_info(ctx).is_ok_and(|info| fronts(&info)) {
490        print_info(&format!("Pointing the proxy at slot {slot}'s guardian"));
491        if !activated && let Some(refusal) = window.refusal() {
492            bail!("{refusal}");
493        }
494        let front_by = window.end - sui::REMAINING_SECS;
495        update_proxy(ctx, &stack, &tree, &FLIPPED, &|| {
496            if !activated && sui::now() > front_by {
497                bail!(
498                    "Sui epoch {} ends in under {} now, too soon for slot {slot}'s guardian to activate in it; the stack has flipped, but the proxy still fronts slot {other}'s guardian: run this step again once the next epoch has settled, or roll back",
499                    window.epoch,
500                    sui::duration(sui::REMAINING_SECS)
501                );
502            }
503            Ok(())
504        })?;
505        if poll(|| proxy_info(ctx).ok().filter(fronts)).is_none() {
506            bail!(
507                "the proxy does not front slot {slot}'s guardian two minutes after its update; the old guardian still runs: check the proxy before going on"
508            );
509        }
510    }
511
512    if let Some(old_instance) = stack.output("standby_instance_id") {
513        let state = guardian_unit_state(ctx, &bucket.region, old_instance)
514            .with_context(|| format!("could not read slot {other}'s host {old_instance}"))?;
515        if state.contains("GUARDIAN_RUNNING") {
516            print_info(&format!("Stopping slot {other}'s guardian"));
517            let stop_by = window.end - sui::STOP_REMAINING_SECS;
518            // What the stop and the activation after it rely on, checked again once the question below is answered.
519            let ready = || -> Result<()> {
520                // A run that replaces a host meanwhile could leave neither guardian holding the key.
521                require_no_guardian_runs(ctx)?;
522                // Nothing activation needs may be found wrong only after the stop.
523                if !activated {
524                    require_fresh(ctx, &stack, &view)?;
525                }
526                require_monitor_allows(ctx, &stack, &[&view, &stack.slot(ctx, other)?])?;
527                let now = slot_info(ctx, &tree, &view)?;
528                if now.signing_key.as_deref() != Some(signing_key.as_str())
529                    || now.lifecycle != info.lifecycle
530                {
531                    bail!(
532                        "slot {slot}'s guardian is {} now, not the session {} in {} that this run checked; nothing was stopped: run this step again",
533                        now.lifecycle,
534                        &signing_key[..16],
535                        info.lifecycle
536                    );
537                }
538                // The relay still reaches the old guardian: the last check that it is the one that served, with one key.
539                let old = relay_info(ctx, &tree).with_context(|| {
540                    format!(
541                        "could not confirm through the relay that slot {other}'s guardian is the activated one; nothing was stopped"
542                    )
543                })?;
544                if old.btc_key.as_deref() != Some(slot_btc.as_str())
545                    || old.signing_key.as_deref() == Some(signing_key.as_str())
546                    || !old.lifecycle.is("withdraw/ACTIVATED")
547                {
548                    bail!(
549                        "could not confirm through the relay that slot {other}'s guardian is the activated one, holding slot {slot}'s BTC key {slot_btc}; nothing was stopped"
550                    );
551                }
552                let left = window.end - sui::now();
553                if !activated && sui::now() > stop_by {
554                    bail!(
555                        "Sui epoch {} ends in about {}, too soon for slot {slot}'s guardian to activate before it once slot {other}'s stops; nothing was stopped; so that slot {other}'s guardian serves through the epoch's turn, roll back to it: set active-slot back to {other}, apply that with a pulumi up targeted at the URN the flip's plan printed, and run proxy-update; then switch over again once the next epoch has settled",
556                        window.epoch,
557                        sui::duration(left)
558                    );
559                }
560                Ok(())
561            };
562            ready()?;
563            print_warning(&format!(
564                "Stopping slot {other}'s guardian on {old_instance} cannot be undone: its copy of the key is gone, and only slot {slot}'s guardian holds it in memory after this."
565            ));
566            confirm_typed(
567                ctx.ask,
568                old_instance,
569                &format!("to stop slot {other}'s guardian for good"),
570                &STOP_DECLINED,
571            )?;
572            ready()?;
573            let stopped = host::run(
574                ctx,
575                &bucket.region,
576                old_instance,
577                &[
578                    "systemctl disable --now hashi-guardian-enclave.service hashi-guardian-bridge.service",
579                    STOPPED_CHECKS[0],
580                    STOPPED_CHECKS[1],
581                ],
582            )
583            .with_context(|| {
584                format!(
585                    "the stop of slot {other}'s guardian on {old_instance} may have run; run this step again: it checks the host first"
586                )
587            })?;
588            if !stopped.success {
589                print_detail(&stopped.text);
590                bail!(
591                    "stopping slot {other}'s guardian did not finish; run this step again: it checks the host first"
592                );
593            }
594            print_success(&format!("Stopped slot {other} guardian on {old_instance}."));
595        } else if !activated {
596            // Activation waits for the old guardian to fall quiet, which an enclave left running never does.
597            let stopped = host::run(ctx, &bucket.region, old_instance, &STOPPED_CHECKS)?;
598            if !stopped.success {
599                print_detail(&stopped.text);
600                bail!(
601                    "slot {other}'s guardian unit on {old_instance} is stopped, but the host still runs an enclave or could not say; terminate it there with nitro-cli terminate-enclave --all, then run this step again"
602                );
603            }
604        }
605    }
606
607    if !activated {
608        print_info(&format!("Activating slot {slot}'s guardian"));
609        require_fresh(ctx, &stack, &view)?;
610        activate(ctx, &tree, &guardian_init, &view)?;
611    }
612
613    print_info(&format!(
614        "Checking that the proxy serves slot {slot}'s guardian"
615    ));
616    let serves = |info: &ProxyInfo| {
617        fronts(info)
618            && info.btc_pubkey.as_deref() == Some(slot_btc.as_str())
619            && info.git_revision.as_deref() == Some(view.commit.as_str())
620            && info.committee_epoch.is_some()
621            && info.age().is_some_and(|age| age <= CURRENT_FOR)
622    };
623    let Some(serving) = poll(|| proxy_info(ctx).ok().filter(serves)) else {
624        bail!(
625            "the proxy's /info does not show slot {slot}'s activated guardian: {}",
626            proxy_info(ctx).map_or_else(|error| format!("{error:#}"), |info| format!("{info:?}"))
627        );
628    };
629    print_success(&format!(
630        "Slot {slot}'s guardian serves at committee epoch {}, with BTC key {slot_btc}.",
631        serving.committee_epoch.as_deref().unwrap_or("")
632    ));
633    // A committee handoff while the proxy fronted the guardian before its activation reached it as nothing. The
634    // guardian serves by now, so a failed read only goes unchecked.
635    match sui::window(ctx, sui::now()) {
636        Ok(now)
637            if now.since >= sui::SETTLE_SECS
638                && serving.committee_epoch.as_deref() != Some(now.epoch.to_string().as_str()) =>
639        {
640            print_warning(&format!(
641                "Slot {slot}'s guardian holds committee epoch {}, but Sui is at epoch {}, and it refuses what the newer committee signs. Restart hashi's leader, which hands the guardian the committees it missed.",
642                serving.committee_epoch.as_deref().unwrap_or("none"),
643                now.epoch
644            ));
645        }
646        Ok(_) => {}
647        Err(error) => print_warning(&format!(
648            "Could not read the Sui epoch, so this step did not check that slot {slot}'s guardian holds the current committee: {error:#}"
649        )),
650    }
651    print_info(&format!(
652        "Once it has served for a while, tear down slot {other} with teardown --slot {other}."
653    ));
654    next(&format!(
655        "after a soak, set slot-{other}-enabled to \"false\" and run teardown --slot {other}"
656    ));
657    Ok(())
658}

680/// A withdraw session's configuration, from its operator-init record. Activation drops the configuration hash from a
681/// guardian's own report, so a serving session's is read here.
682struct OperatorInit {
683    config_hash: String,
684    limiter: Value,
685    deployment: Value,
686}
687
688impl OperatorInit {
689    /// The record of the session named by the start of its signing key. It must have one version, as a guardian
690    /// requires of each record it reads.
691    fn read(ctx: &Ctx, bucket: &Bucket, signing_key: &str) -> Result<Self> {
692        let session = signing_key
693            .get(..16)
694            .ok_or_else(|| anyhow!("{signing_key} is not a guardian's signing key"))?;
695        let key = format!("init/{session}/02-oi-guardian-info.json");
696        let url = format!("s3://{}/{key}", bucket.name);
697        let versions = read(
698            ctx.exec,
699            &Cmd::new("aws")
700                .args(["s3api", "list-object-versions", "--region", &bucket.region])
701                .args(["--bucket", &bucket.name, "--prefix", &key])
702                .arg("--query")
703                .arg(format!(
704                    "[length(Versions[?Key=='{key}'] || `[]`), length(DeleteMarkers[?Key=='{key}'] || `[]`)]"
705                ))
706                .args(["--output", "text"]),
707        )
708        .with_context(|| format!("could not list the versions of {url}"))?;
709        match versions.split_whitespace().collect::<Vec<_>>()[..] {
710            ["1", "0"] => {}
711            ["0", "0"] => bail!("there is no {url}"),
712            _ => bail!(
713                "{url} has versions or delete markers beside the one its guardian wrote ({}); find out who wrote them before going on",
714                versions.trim()
715            ),
716        }
717        let record: Value = read_json(
718            ctx.exec,
719            &Cmd::new("aws")
720                .args(["s3", "cp", "--only-show-errors", "--region", &bucket.region])
721                .arg(&url)
722                .arg("-"),
723        )
724        .with_context(|| format!("could not read {url}"))?;
725        let info = record.pointer("/message/Init/OIGuardianInfo");
726        let withdraw = info.and_then(|info| info.pointer("/mode/Withdraw"));
727        let (Some(config_hash), Some(limiter), Some(deployment)) = (
728            withdraw
729                .and_then(|withdraw| withdraw.get("config_hash"))
730                .and_then(Value::as_str),
731            withdraw.and_then(|withdraw| withdraw.get("limiter_config")),
732            info.and_then(|info| info.get("deployment")),
733        ) else {
734            bail!("{url} is not a withdraw session's operator-init record");
735        };
736        Ok(Self {
737            config_hash: config_hash.to_owned(),
738            limiter: limiter.clone(),
739            deployment: deployment.clone(),
740        })
741    }
742
743    /// Stops unless `settings` give this session's limiter and deployment exactly: activation recomputes its
744    /// configuration hash from them, the allowlist's order included.
745    fn require_rendered(&self, settings: &Settings, slot: Slot) -> Result<()> {
746        let deployment = json!({
747            "bucket_info": { "name": settings.bucket, "region": settings.region },
748            "retention_environment": settings.retention_environment,
749            "bitcoin_network": logged_network(settings),
750            "pcr_allowlist": {
751                "current_build": {
752                    "git_revision": settings.commit,
753                    "pcr0": settings.pcr0.to_ascii_lowercase(),
754                },
755                "prev_builds": settings
756                    .prev_builds
757                    .iter()
758                    .map(|build| json!({
759                        "git_revision": build.commit,
760                        "pcr0": build.pcr0.to_ascii_lowercase(),
761                    }))
762                    .collect::<Vec<_>>(),
763            },
764        });
765        let limiter = json!({
766            "refill_rate": settings.refill_rate,
767            "max_bucket_capacity": settings.max_bucket_capacity,
768        });
769        if self.deployment != deployment || self.limiter != limiter {
770            bail!(
771                "slot {slot}'s configurations give limiter {limiter} and deployment {deployment}, but its guardian was initialized with limiter {} and deployment {}; activating it would fail once the old guardian has stopped: render the settings it was initialized with",
772                self.limiter,
773                self.deployment
774            );
775        }
776        Ok(())
777    }
778
779    /// Whether this session keeps `serving`'s limiter and deployment, its PCR allowlist only adding builds to
780    /// `serving`'s.
781    fn keeps(&self, serving: &Self) -> bool {
782        let split = |deployment: &Value| -> Option<(Value, Vec<Value>)> {
783            let mut rest = deployment.as_object()?.clone();
784            let allowlist = rest.remove("pcr_allowlist")?;
785            let mut builds = vec![allowlist.get("current_build")?.clone()];
786            builds.extend(allowlist.get("prev_builds")?.as_array()?.iter().cloned());
787            Some((Value::Object(rest), builds))
788        };
789        let (Some((rest, builds)), Some((serving_rest, serving_builds))) =
790            (split(&self.deployment), split(&serving.deployment))
791        else {
792            return false;
793        };
794        self.limiter == serving.limiter
795            && rest == serving_rest
796            && serving_builds.iter().all(|build| builds.contains(build))
797    }
798
799    fn prev_builds(&self) -> Option<Vec<Build>> {
800        self.deployment
801            .pointer("/pcr_allowlist/prev_builds")?
802            .as_array()?
803            .iter()
804            .map(build_of)
805            .collect()
806    }
807}

809fn build_of(value: &Value) -> Option<Build> {
810    Some(Build {
811        commit: value.get("git_revision")?.as_str()?.to_owned(),
812        pcr0: value.get("pcr0")?.as_str()?.to_owned(),
813    })
814}

816/// The build each session of the log ran, from its operator-init record.
817fn session_builds(ctx: &Ctx, bucket: &Bucket) -> Result<Vec<(String, Build)>> {
818    let keys: Vec<String> = read_json(
819        ctx.exec,
820        &Cmd::new("aws")
821            .args(["s3api", "list-objects-v2", "--region", &bucket.region])
822            .args(["--bucket", &bucket.name, "--prefix", "init/"])
823            .args(["--query", "Contents[].Key || `[]`", "--output", "json"]),
824    )
825    .with_context(|| format!("could not list s3://{}/init/", bucket.name))?;
826    let mut builds = Vec::new();
827    for key in keys {
828        let Some(session) = key
829            .strip_prefix("init/")
830            .and_then(|rest| rest.strip_suffix("/02-oi-guardian-info.json"))

851/// Stops unless a standby's settings keep the serving guardian's limiter and deployment, and allowlist every build that
852/// ran a session in the log and every one the serving guardian allowlists. Guardians and key provisioners verify the
853/// records of every recent session, and switchover refuses less, but only once the standby is provisioned.
854pub(crate) fn require_keeps_serving(
855    ctx: &Ctx,
856    stack: &Stack,
857    view: &SlotView,
858    settings: &Settings,
859) -> Result<()> {
860    let bucket = stack.bucket()?;
861    let serving_key = proxy_info(ctx)?
862        .signing_pub_key
863        .filter(|key| is_hex(key, 64))
864        .ok_or_else(|| anyhow!("the proxy's /info names no serving guardian's signing key"))?;
865    let serving = OperatorInit::read(ctx, &bucket, &serving_key)?;
866    for (name, theirs, ours) in [
867        (
868            "refill-rate-sats-per-sec",
869            serving.limiter.get("refill_rate"),
870            Value::from(settings.refill_rate),
871        ),
872        (
873            "max-bucket-capacity-sats",
874            serving.limiter.get("max_bucket_capacity"),
875            Value::from(settings.max_bucket_capacity),
876        ),
877        (
878            "the log bucket",
879            serving.deployment.pointer("/bucket_info/name"),
880            Value::from(settings.bucket.as_str()),
881        ),
882        (
883            "the log bucket's region",
884            serving.deployment.pointer("/bucket_info/region"),
885            Value::from(settings.region.as_str()),
886        ),
887        (
888            "RETENTION_ENVIRONMENT",
889            serving.deployment.get("retention_environment"),
890            Value::from(settings.retention_environment.as_str()),
891        ),
892        (
893            "BITCOIN_NETWORK",
894            serving.deployment.get("bitcoin_network"),
895            Value::from(logged_network(settings)),
896        ),
897    ] {
898        if theirs != Some(&ours) {
899            bail!(
900                "slot {}'s configurations give {name} as {ours}, but the serving guardian's give {}; a switchover keeps the serving guardian's limiter and deployment block: set it back",
901                view.slot,
902                theirs.map_or("none".to_owned(), Value::to_string)
903            );
904        }
905    }
906    let serving_prev = serving.prev_builds().ok_or_else(|| {
907        anyhow!(
908            "could not read the builds the serving guardian {} allowlists",
909            &serving_key[..16]
910        )
911    })?;
912    require_lists_log_builds(
913        ctx,
914        stack,
915        view,
916        settings,
917        serving_prev
918            .into_iter()
919            .map(|build| (build, "the serving guardian allowlists it".to_owned()))
920            .collect(),
921    )
922}

924/// Stops unless `settings` allowlist each build that wrote the stack's log, and each of `also`: a reader resolves a
925/// record's writer by that session's own build, so a guardian that leaves one out can't read what it wrote.
926pub(crate) fn require_lists_log_builds(
927    ctx: &Ctx,
928    stack: &Stack,
929    view: &SlotView,
930    settings: &Settings,
931    also: Vec<(Build, String)>,
932) -> Result<()> {
933    let required = session_builds(ctx, &stack.bucket()?)?
934        .into_iter()
935        .map(|(session, build)| (build, format!("session {session} ran it")))
936        .chain(also);
937    let own = Build {
938        commit: settings.commit.clone(),
939        pcr0: settings.pcr0.clone(),
940    };
941    let mut missing: Vec<(Build, String)> = Vec::new();
942    for (build, why) in required {
943        match std::iter::once(&own)
944            .chain(&settings.prev_builds)
945            .find(|listed| listed.commit == build.commit)
946        {
947            Some(listed) if listed.pcr0 == build.pcr0 => {}
948            Some(listed) => bail!(
949                "slot {}'s configurations measure {} as {}, but {why} as {}",
950                view.slot,
951                build.commit,
952                listed.pcr0,
953                build.pcr0
954            ),
955            None if missing.iter().any(|(seen, _)| seen.commit == build.commit) => {}
956            None => missing.push((build, why)),
957        }
958    }
959    if !missing.is_empty() {
960        bail!(
961            "slot {}'s configurations leave out {}; every guardian of the log must allowlist those builds: add each to prev-builds in stack {}'s Pulumi.<stack>.yaml, as its git_revision and the pcr0 its own measurement gave, never one read from the log bucket",
962            view.slot,
963            missing
964                .iter()
965                .map(|(build, why)| format!("{} ({why})", build.commit))
966                .collect::<Vec<_>>()
967                .join("; "),
968            stack.name
969        );
970    }
971    Ok(())
972}

974/// Stops unless hashi-monitor, where `HASHI_MONITOR_PULUMI_DIR` names its program, allowlists each of these slots'
975/// builds: it rejects every record from a build it doesn't, and reads the latest ceremony record when it starts, which
976/// a re-deal writes.
977pub(crate) fn require_monitor_allows(ctx: &Ctx, stack: &Stack, views: &[&SlotView]) -> Result<()> {
978    let Some(dir) = ctx.env.get("HASHI_MONITOR_PULUMI_DIR") else {
979        print_warning(
980            "HASHI_MONITOR_PULUMI_DIR is unset, so this step did not check that hashi-monitor allowlists the guardians' builds.",
981        );
982        return Ok(());
983    };
984    project_name(dir)?;
985    let entry: Value = read_json(
986        ctx.exec,
987        &pulumi(dir).args(["config", "get", "audit", "--json", "--stack", &stack.name]),
988    )
989    .with_context(|| {
990        format!(
991            "could not read the audit configuration of hashi-monitor's stack {} in {dir}",
992            stack.name
993        )
994    })?;
995    let audit = entry
996        .get("objectValue")
997        .cloned()
998        .or_else(|| serde_json::from_str(entry.get("value")?.as_str()?).ok());
999    let allowlist = audit
1000        .as_ref()
1001        .and_then(|audit| audit.pointer("/deployment/pcr_allowlist"));
1002    let builds: Vec<&Value> = allowlist
1003        .and_then(|allowlist| allowlist.get("current_build"))
1004        .into_iter()
1005        .chain(
1006            allowlist
1007                .and_then(|allowlist| allowlist.get("prev_builds"))
1008                .and_then(Value::as_array)
1009                .into_iter()
1010                .flatten(),
1011        )
1012        .collect();
1013    for view in views {
1014        let pcr0 = stack.config(&view.pcr0_key).unwrap_or("");
1015        let listed = builds
1016            .iter()
1017            .filter_map(|build| build_of(build))
1018            .any(|build| {
1019                build.commit == view.commit
1020                    && !pcr0.is_empty()
1021                    && build.pcr0.eq_ignore_ascii_case(pcr0)
1022            });
1023        if !listed {
1024            bail!(
1025                "hashi-monitor does not allowlist {} with PCR0 {}; add it to deployment.pcr_allowlist in the audit configuration of stack {} in {dir}, keeping every build already there, deploy that stack, then run this step again",
1026                view.commit,
1027                if pcr0.is_empty() { "unset" } else { pcr0 },
1028                stack.name
1029            );
1030        }
1031    }
1032    Ok(())
1033}

1035/// Destroys a standby slot's host: the retired guardian after a switchover, or a standby that never served.
1036pub fn teardown(ctx: &Ctx, slot_arg: Slot) -> Result<()> {
1037    let (stack, view) = slot(ctx, Some(slot_arg))?;
1038    stack.bucket()?;
1039    let instance = view.require_instance(&stack)?;
1040    let slot = view.slot;
1041    print_info(&format!(
1042        "Tearing down slot {slot}'s guardian host {instance}"
1043    ));
1044    if view.role != Role::Standby {
1045        bail!("slot {slot} runs the active guardian; only a standby can be torn down");
1046    }
1047    require_pulumi_version(ctx)?;
1048    require_no_guardian_runs(ctx)?;
1049    if stack.slot_enabled(slot)? {
1050        bail!(
1051            "disable slot {slot} first: set slot-{slot}-enabled to \"false\" in stack {}'s Pulumi.<stack>.yaml, and commit it",
1052            stack.name
1053        );
1054    }
1055    let active = stack.active()?;
1056    let configured_active = stack.config("active-slot").unwrap_or("a");
1057    if configured_active != active.to_string() {
1058        bail!(
1059            "stack {} configures active-slot {configured_active}, but slot {active} is active",
1060            stack.name
1061        );
1062    }
1063    serving_info(ctx).context("tear nothing down until a guardian serves")?;
1064    let Some(host) = stack
1065        .instances(ctx)?
1066        .into_iter()
1067        .find(|resource| resource.id.as_deref() == Some(instance))
1068    else {
1069        bail!(
1070            "could not find slot {slot}'s instance {instance} in stack {}'s state",
1071            stack.name
1072        );
1073    };
1074    // EC2 and SSM are asked in the host's own region and account, which its ARN names.
1075    let arn: Vec<&str> = host
1076        .outputs
1077        .as_ref()
1078        .and_then(|outputs| outputs.get("arn")?.as_str())
1079        .unwrap_or_default()
1080        .split(':')
1081        .collect();
1082    let (region, account) = (
1083        arn.get(3).copied().unwrap_or_default(),
1084        arn.get(4).copied().unwrap_or_default(),
1085    );
1086    // What the guardian on the host is, if anything: `None` once it can't serve again.
1087    let guardian = || -> Result<Option<Lifecycle>> {
1088        // EC2 is asked first: SSM can't reach a host it has stopped or ended, which a teardown cut off partway leaves.
1089        match host::instance_state(ctx, region, instance, account)?.as_deref() {
1090            None | Some("stopping" | "stopped" | "shutting-down" | "terminated") => {
1091                return Ok(None);
1092            }
1093            _ => {}
1094        }
1095        let state = guardian_unit_state(ctx, region, instance)
1096            .with_context(|| format!("could not read slot {slot}'s host"))?;
1097        if state.contains("GUARDIAN_STOPPED") {
1098            return Ok(None);
1099        }
1100        // A retiring guardian still signs until the switchover step stops it; a standby never has.
1101        let lifecycle = slot_info(ctx, &tree(ctx, &view)?, &view)?.lifecycle;
1102        if lifecycle.is("withdraw/ACTIVATED") {
1103            bail!(
1104                "slot {slot}'s guardian on {instance} is activated and still runs; stop it with the switchover step first"
1105            );
1106        }
1107        Ok(Some(lifecycle))
1108    };
1109    let found = guardian()?;
1110    match &found {
1111        None => print_info(&format!(
1112            "No guardian runs on slot {slot}'s host {instance}."
1113        )),
1114        Some(lifecycle) => print_warning(&format!(
1115            "Slot {slot}'s guardian still runs ({lifecycle}). Tearing it down abandons it: what was provisioned or dealt to it is lost."
1116        )),
1117    }
1118    let changes = preview_slot(ctx, &stack, &host.urn)?;
1119    if changes != [Change::new("delete", &host.urn)] {
1120        bail!(
1121            "tearing down slot {slot} must delete its instance and change nothing else, but this plan does not; nothing was changed"
1122        );
1123    }
1124    confirm_typed(
1125        ctx.ask,
1126        instance,
1127        &format!("to destroy slot {slot}'s host"),
1128        &NOTHING_CHANGED,
1129    )?;
1130    // The question can wait any time, while the stack's configuration, either guardian or the plan changes.
1131    serving_info(ctx).context("tear nothing down until a guardian serves")?;
1132    let now = Stack::load(ctx)?;
1133    if now.slot_enabled(slot)?
1134        || now.active()? != active
1135        || now.config("active-slot").unwrap_or("a") != configured_active
1136        || now.instance(slot)? != Some(instance)
1137        || guardian()? != found

1145    up_slot(ctx, &now, &host.urn)?;

1147    if after.output("standby_instance_id").is_some() || after.active()? != active {

1151        );
1152    }
1153    print_success(&format!(
1154        "Tore down slot {slot}. Run proxy-update, so the relay stops pointing at it."

1159            view.commit,
1160            slot.suffix(),
1161            stack.name
1162        ));
1163    }
1164    next("proxy-update");
1165    Ok(())
