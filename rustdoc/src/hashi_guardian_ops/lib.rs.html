32use env::Env;

35pub struct Ctx<'a> {
36    pub env: Env,
37    pub exec: &'a dyn Exec,
38    pub ask: &'a dyn Ask,
39    /// The hashi checkout this tool runs from, which holds its rendered files under `.hashi/`.
40    pub repo: PathBuf,
41}
42
43impl<'a> Ctx<'a> {
44    pub fn new(env_file: &Path, exec: &'a dyn Exec, ask: &'a dyn Ask) -> Result<Self> {
45        let repo = Path::new(env!("CARGO_MANIFEST_DIR")).join("../..");
46        Ok(Self {
47            env: Env::load(env_file)?,
48            exec,
49            ask,
50            repo: std::fs::canonicalize(&repo)
51                .with_context(|| format!("could not find this checkout at {}", repo.display()))?,
52        })
53    }
54
55    #[cfg(test)]
56    pub fn for_test(repo: &Path, exec: &'a dyn Exec, values: &[(&str, &str)]) -> Self {
57        let values = values
58            .iter()
59            .map(|(key, value)| ((*key).to_owned(), (*value).to_owned()))
60            .collect();
61        Self {
62            env: Env::new(PathBuf::from("testnet.env"), values).unwrap(),
63            exec,
64            ask: &ask::scripted::NO_ANSWERS,
65            repo: repo.to_owned(),
66        }
67    }
68
69    #[cfg(test)]
70    pub fn with_ask(self, ask: &'a dyn Ask) -> Self {
71        Self { ask, ..self }
72    }
73}

75/// Prints a note for the operator. Notes go to stderr: stdout carries only a step's results (a report, a roster, a
76/// digest to post), so piping a step's stdout never picks up its progress.
77pub fn print_detail(msg: &str) {
78    eprintln!("{msg}");
79}
80
81pub fn print_info(msg: &str) {
82    print_detail(&format!("{} {msg}", "ℹ".blue().bold()));
83}
84
85pub fn print_success(msg: &str) {
86    print_detail(&format!("{} {msg}", "✓".green().bold()));
87}
88
89pub fn print_warning(msg: &str) {
90    print_detail(&format!("{} {msg}", "⚠".yellow().bold()));
91}
92
93/// Ends a step by naming the one after it, so the steps and the runbook can't drift apart.
94pub fn next(step: &str) {
95    print_info(&format!("Next: {step}"));
96}

99mod tests {
100    use std::path::Path;
101
102    /// hashi is public, so fixtures keep the shapes of real records with none of their account data.
103    #[test]
104    fn fixtures_hold_no_credentials_or_account_data() {
105        let mut stack = vec![Path::new(env!("CARGO_MANIFEST_DIR")).join("testdata")];
106        while let Some(dir) = stack.pop() {
107            for entry in std::fs::read_dir(&dir).unwrap() {
108                let path = entry.unwrap().path();
109                if path.is_dir() {
110                    stack.push(path);
111                    continue;
112                }
113                let text = std::fs::read_to_string(&path).unwrap();
114                let words: Vec<&str> = text
115                    .split(|c: char| !(c.is_ascii_alphanumeric() || c == '.' || c == ':'))
116                    .collect();
117                for word in words {
118                    let found = (word.len() >= 20
119                        && word.starts_with("AKIA")
120                        && word[4..20]
121                            .bytes()
122                            .all(|b| b.is_ascii_uppercase() || b.is_ascii_digit()))
123                        || (word.len() == 12 && word.bytes().all(|b| b.is_ascii_digit()))
124                        || word.starts_with("arn:aws:")
125                        || is_private_ip(word);
126                    assert!(!found, "{} holds {word}", path.display());
127                }
128            }
129        }
130    }
131
132    fn is_private_ip(word: &str) -> bool {
133        let octets: Vec<u8> = word
134            .split('.')
135            .filter_map(|part| part.parse().ok())
136            .collect();
137        octets.len() == 4
138            && word.split('.').count() == 4
139            && match octets[..2] {
140                [10, _] => true,
141                [172, second] => (16..=31).contains(&second),
142                [192, 168] => true,
143                _ => false,
144            }
145    }
