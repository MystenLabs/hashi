4//! The Sui epoch, from Sui GraphQL. hashi's committee hands off to the guardian once per epoch, so a switchover must
5//! sit well inside one.
6
7use anyhow::Context;
8use anyhow::Result;
9use anyhow::anyhow;
10use serde::Deserialize;
11
12use crate::Ctx;
13use crate::exec::Cmd;
14use crate::exec::read_json;

16/// Minimum time since the epoch began, for the committee handoff to settle.
17pub const SETTLE_SECS: i64 = 30 * 60;
18/// Minimum time left in the epoch to point the proxy at the standby, for the switchover to finish in it.
19pub const REMAINING_SECS: i64 = 60 * 60;
20/// Minimum time left in the epoch to stop the old guardian: activation then waits ten minutes for it to fall quiet.
21pub const STOP_REMAINING_SECS: i64 = 20 * 60;
22
23/// Where now falls in the current epoch, taking it to last as long as the previous one did.
24pub struct Window {
25    pub epoch: u64,
26    pub since: i64,
27    pub remaining: i64,
28    /// When the epoch ends, in seconds since the Unix epoch.
29    pub end: i64,
30}
31
32impl Window {
33    pub fn new(epoch: u64, start: i64, previous_start: i64, now: i64) -> Self {
34        let end = 2 * start - previous_start;
35        Self {
36            epoch,
37            since: now - start,
38            remaining: end - now,
39            end,
40        }
41    }
42
43    /// Why a switchover can't start now, or `None` when it can.
44    pub fn refusal(&self) -> Option<String> {
45        let epoch = self.epoch;
46        if self.since < SETTLE_SECS {
47            Some(format!(
48                "Sui epoch {epoch} began {} minutes ago; switch over once hashi's committee handoff has settled, 30 minutes after the epoch turns",
49                self.since / 60
50            ))
51        } else if self.remaining < REMAINING_SECS {
52            Some(format!(
53                "Sui epoch {epoch} ends in about {} minutes; switch over after the next epoch has settled, so its committee handoff reaches the new guardian",
54                self.remaining / 60
55            ))
56        } else {
57            None
58        }
59    }
60}

62pub fn window(ctx: &Ctx, now: i64) -> Result<Window> {
63    #[derive(Deserialize)]
64    #[serde(rename_all = "camelCase")]
65    struct Epoch {
66        epoch_id: Option<u64>,
67        start_timestamp: String,
68    }
69    #[derive(Deserialize)]
70    struct Data {
71        epoch: Epoch,
72    }
73    #[derive(Deserialize)]
74    struct Response {
75        data: Data,
76    }
77    let url = ctx.env.require("SUI_GRAPHQL_URL")?;
78    let query = |query: &str| -> Result<Epoch> {
79        let response: Response = read_json(
80            ctx.exec,
81            &Cmd::new("curl")
82                .args(["-fsS", "--max-time", "15"])
83                .args(["-H", "Content-Type: application/json"])
84                .arg("-d")
85                .arg(serde_json::json!({ "query": query }).to_string())
86                .arg(url),
87        )?;
88        Ok(response.data.epoch)
89    };
90    let current = query("{ epoch { epochId startTimestamp } }")
91        .with_context(|| format!("could not read the current Sui epoch from {url}"))?;
92    let epoch = current
93        .epoch_id
94        .ok_or_else(|| anyhow!("could not read the current Sui epoch from {url}"))?;
95    let start = seconds(&current.start_timestamp)?;
96    let previous = query(&format!(
97        "{{ epoch(epochId: {}) {{ startTimestamp }} }}",
98        epoch.saturating_sub(1)
99    ))
100    .with_context(|| format!("could not read the previous Sui epoch from {url}"))?;
101    Ok(Window::new(
102        epoch,
103        start,
104        seconds(&previous.start_timestamp)?,
105        now,
106    ))
107}

109/// Stops unless the Sui epoch, and hashi's committee with it, stays put while guardians switch, and the serving guardian
110/// holds this epoch's committee. hashi's leader hands the guardian each new committee once per epoch, and a standby the
111/// proxy fronts before its activation takes none, which leaves it a committee behind until the next epoch.
112pub fn require_epoch_window(window: &Window, committee_epoch: Option<&str>) -> Result<()> {
113    if let Some(refusal) = window.refusal() {
114        anyhow::bail!("{refusal}");
115    }
116    let epoch = window.epoch;
117    if committee_epoch != Some(epoch.to_string().as_str()) {
118        anyhow::bail!(
119            "the serving guardian is at committee epoch {}, but Sui is at epoch {epoch}; wait for hashi's committee handoff to reach it",
120            committee_epoch.unwrap_or("none")
121        );
122    }
123    Ok(())
124}

126/// A GraphQL timestamp in whole seconds since the Unix epoch.
127fn seconds(timestamp: &str) -> Result<i64> {
128    Ok(timestamp
129        .parse::<jiff::Timestamp>()
130        .with_context(|| format!("{timestamp} is not a timestamp"))?
131        .as_second())
132}
133
134pub fn now() -> i64 {
135    jiff::Timestamp::now().as_second()
136}
137
138/// A duration as hours and minutes, for messages.
139pub fn duration(secs: i64) -> String {
140    let minutes = secs / 60;
141    if minutes.abs() < 60 {
142        format!("{minutes} min")
143    } else {
144        format!("{} h {} min", minutes / 60, (minutes % 60).abs())
145    }
146}

148#[cfg(test)]
149mod tests {
150    use super::*;
151
152    #[test]
153    fn measures_the_window() {
154        let start = seconds("2026-10-14T21:24:13.402Z").unwrap();
155        let previous = seconds("2026-10-13T21:24:13Z").unwrap();
156        let window = Window::new(1254, start, previous, start + 3 * 3600);
157        assert_eq!(window.since, 3 * 3600);
158        assert_eq!(window.remaining, 21 * 3600);
159        assert_eq!(window.refusal(), None);
160        assert!(
161            Window::new(1254, start, previous, start + 20 * 60)
162                .refusal()
163                .unwrap()
164                .starts_with("Sui epoch 1254 began 20 minutes ago;")
165        );
166        assert!(
167            Window::new(1254, start, previous, start + 23 * 3600 + 1)
168                .refusal()
169                .unwrap()
170                .starts_with("Sui epoch 1254 ends in about 59 minutes;")
171        );
172        assert_eq!(duration(3 * 3600 + 25 * 60), "3 h 25 min");
173        assert_eq!(duration(20 * 60), "20 min");
174    }
