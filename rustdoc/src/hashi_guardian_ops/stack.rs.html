25/// The type of a guardian's host in the stack's state and plans.

28#[derive(Clone, Copy, Debug, PartialEq, Eq, clap::ValueEnum)]
29pub enum Slot {
30    #[value(name = "a")]
31    A,
32    #[value(name = "b")]
33    B,
34}
35
36impl Slot {
37    pub const ALL: [Self; 2] = [Self::A, Self::B];
38
39    /// The suffix of this slot's own configuration keys, such as `hashi-commit-b`.
40    pub fn suffix(self) -> &'static str {
41        match self {
42            Self::A => "",
43            Self::B => "-b",
44        }
45    }
46
47    pub fn other(self) -> Self {
48        match self {
49            Self::A => Self::B,
50            Self::B => Self::A,
51        }
52    }
53}
54
55impl fmt::Display for Slot {
56    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
57        f.write_str(match self {
58            Self::A => "a",
59            Self::B => "b",
60        })
61    }
62}

64#[derive(Clone, Copy, Debug, PartialEq, Eq)]
65pub enum Role {
66    Active,
67    Standby,
68    Absent,
69}
70
71impl fmt::Display for Role {
72    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
73        f.write_str(match self {
74            Self::Active => "active",
75            Self::Standby => "standby",
76            Self::Absent => "absent",
77        })
78    }
79}

86/// The stack's configuration and outputs as a step found them when it started, so it acts on one view of both.
87pub struct Stack {
88    pub name: String,
89    pub dir: String,
90    /// The program's project, which prefixes its own configuration keys.
91    project: String,
92    config: BTreeMap<String, String>,
93    outputs: BTreeMap<String, String>,
94}
95
96#[derive(Deserialize)]
97struct ConfigEntry {
98    value: Option<Value>,
99}
100
101impl Stack {
102    pub fn load(ctx: &Ctx) -> Result<Self> {
103        let dir = ctx.env.require("GUARDIAN_PULUMI_DIR")?;
104        let name = ctx.env.require("GUARDIAN_STACK")?;
105        let project = project_name(dir)?;
106        let config: BTreeMap<String, ConfigEntry> = read_json(
107            ctx.exec,
108            &pulumi(dir).args(["config", "--stack", name, "--json"]),
109        )
110        .with_context(|| format!("could not read the configuration of stack {name} in {dir}"))?;
111        let outputs: BTreeMap<String, Value> = read_json(
112            ctx.exec,
113            &pulumi(dir).args(["stack", "output", "--json", "--stack", name]),
114        )
115        .with_context(|| format!("could not read the outputs of stack {name} in {dir}"))?;
116        Ok(Self {
117            name: name.to_owned(),
118            dir: dir.to_owned(),
119            project,
120            config: config
121                .into_iter()
122                .filter_map(|(key, entry)| Some((key, scalar(entry.value?))))
123                .collect(),
124            outputs: outputs
125                .into_iter()
126                .map(|(key, value)| (key, scalar(value)))
127                .collect(),
128        })
129    }
130
131    /// The stack's name without its organization, which names its directory of rendered files.
132    pub fn short_name(&self) -> &str {
133        self.name.rsplit('/').next().unwrap_or(&self.name)
134    }
135
136    /// A configuration value, or `None` when the key is unset.
137    pub fn config(&self, key: &str) -> Option<&str> {
138        self.config
139            .get(&format!("{}:{key}", self.project))
140            .map(String::as_str)
141            .filter(|value| !value.is_empty())
142    }
143
144    pub fn output(&self, key: &str) -> Option<&str> {
145        self.outputs
146            .get(key)
147            .map(String::as_str)
148            .filter(|value| !value.is_empty())
149    }
150
151    /// As the enclave program reads the key: slot a is on unless disabled, slot b is off unless enabled.
152    pub fn slot_enabled(&self, slot: Slot) -> Result<bool> {
153        match self.config(&format!("slot-{slot}-enabled")) {
154            Some("1" | "t" | "T" | "TRUE" | "true" | "True") => Ok(true),
155            Some("0" | "f" | "F" | "FALSE" | "false" | "False") => Ok(false),
156            None => Ok(slot == Slot::A),
157            Some(other) => bail!(
158                "stack {} sets slot-{slot}-enabled to {other}, not true or false",
159                self.name
160            ),
161        }
162    }
163
164    /// The slot the outputs follow.
165    pub fn active(&self) -> Result<Slot> {
166        match self.output("active_slot") {
167            None | Some("a") => Ok(Slot::A),
168            Some("b") => Ok(Slot::B),
169            Some(other) => bail!(
170                "stack {}'s active_slot output is {other}, not a or b",
171                self.name
172            ),
173        }
174    }
175
176    pub fn role(&self, slot: Slot) -> Result<Role> {
177        Ok(if self.output("enclave_instance_id").is_none() {
178            Role::Absent
179        } else if slot == self.active()? {
180            Role::Active
181        } else if self.output("standby_instance_id").is_some() {
182            Role::Standby
183        } else {
184            Role::Absent
185        })
186    }
187
188    /// The slots the stack configures or still runs.
189    pub fn slots(&self) -> Result<Vec<Slot>> {
190        let mut slots = Vec::new();
191        for slot in Slot::ALL {
192            if self.slot_enabled(slot)? || self.role(slot)? != Role::Absent {
193                slots.push(slot);
194            }
195        }
196        Ok(slots)
197    }
198
199    pub fn instance(&self, slot: Slot) -> Result<Option<&str>> {
200        Ok(match self.role(slot)? {
201            Role::Active => self.output("enclave_instance_id"),
202            Role::Standby => self.output("standby_instance_id"),
203            Role::Absent => None,
204        })
205    }
206
207    /// The guardian's log bucket, as the stack deployed it.
208    pub fn bucket(&self) -> Result<Bucket> {
209        match (
210            self.output("s3_bucket_name"),
211            self.output("s3_bucket_region"),
212        ) {
213            (Some(name), Some(region)) => Ok(Bucket {
214                name: name.to_owned(),
215                region: region.to_owned(),
216            }),
217            _ => bail!(
218                "stack {} in {} has no deployed guardian; run the deploy step first",
219                self.name,
220                self.dir
221            ),
222        }
223    }
224
225    /// The slot named on the command line, or else the stack's only one.
226    pub fn select(&self, arg: Option<Slot>) -> Result<Slot> {
227        if let Some(slot) = arg {
228            return Ok(slot);
229        }
230        match self.slots()?.as_slice() {
231            [slot] => Ok(*slot),
232            [] => bail!("stack {} enables no guardian slot", self.name),
233            _ => bail!(
234                "stack {} has guardian slots a and b; name the one this step acts on with --slot a or --slot b",
235                self.name
236            ),
237        }
238    }

240    /// What a step needs about a slot: its build, its deployed guardian, its tunnel and its rendered files.
241    pub fn slot(&self, ctx: &Ctx, slot: Slot) -> Result<SlotView> {
242        let suffix = slot.suffix();
243        let commit = self.config(&format!("hashi-commit{suffix}")).unwrap_or("");
244        if !is_hex(commit, 40) {
245            bail!(
246                "stack {}'s hashi-commit{suffix} is not a full commit: {}",
247                self.name,
248                if commit.is_empty() { "unset" } else { commit }
249            );
250        }
251        let base = ctx.env.tunnel_port()?;
252        let port = match slot {
253            Slot::A => base,
254            Slot::B => base + 1,
255        };
256        Ok(SlotView {
257            slot,
258            commit: commit.to_owned(),
259            pcr0_key: format!("eif-pcr0{suffix}"),
260            role: self.role(slot)?,
261            instance: self.instance(slot)?.map(str::to_owned),
262            port,
263            endpoint: format!("http://127.0.0.1:{port}"),
264            dir: ctx
265                .repo
266                .join(".hashi/guardian")
267                .join(self.short_name())
268                .join(slot.to_string()),
269        })
270    }

272    /// One output read on its own, so a secret one can be shown.
273    pub fn output_value(&self, ctx: &Ctx, key: &str, secret: bool) -> Result<String> {
274        let mut cmd = pulumi(&self.dir).args(["stack", "output", key]);
275        if secret {
276            cmd = cmd.arg("--show-secrets");
277        }
278        let value = read(ctx.exec, &cmd.args(["--stack", &self.name]))?;
279        Ok(value.trim_end_matches('\n').to_owned())
280    }

282    /// The stack's `rotation:*` tags, which hold rotation state apart from its configuration.
283    pub fn rotation_tags(&self, ctx: &Ctx) -> Result<Vec<String>> {
284        let tags: BTreeMap<String, Value> = read_json(

303    /// The guardian instances in the stack's state, by URN and ID.

313    /// The state URN of the guardian instance with this ID.
314    pub fn instance_urn(&self, ctx: &Ctx, id: &str) -> Result<Option<String>> {
315        Ok(self
316            .instances(ctx)?
317            .into_iter()
318            .find(|resource| resource.id.as_deref() == Some(id))
319            .map(|resource| resource.urn))
320    }

323/// The project a Pulumi program's own directory declares, whose name prefixes the program's configuration keys. `pulumi
324/// -C` would use a parent directory's project instead, so a directory without one is refused.
325pub fn project_name(dir: &str) -> Result<String> {
326    #[derive(Deserialize)]
327    struct Project {
328        name: String,
329    }
330    let files: Vec<PathBuf> = ["Pulumi.yaml", "Pulumi.yml"]
331        .iter()
332        .map(|file| Path::new(dir).join(file))
333        .filter(|path| path.is_file())
334        .collect();
335    let [file] = &files[..] else {
336        bail!(
337            "{dir} holds {} Pulumi project files, not one: name the Pulumi program's own directory",
338            files.len()
339        );
340    };
341    let text = std::fs::read_to_string(file)
342        .with_context(|| format!("could not read {}", file.display()))?;
343    let project: Project = serde_yaml::from_str(&text)
344        .with_context(|| format!("{} names no Pulumi project", file.display()))?;
345    if project.name.is_empty() || project.name.contains(':') {
346        bail!(
347            "{} names the Pulumi project {:?}, which no configuration key can carry",
348            file.display(),
349            project.name
350        );
351    }
352    Ok(project.name)
353}

394/// What a step needs about one slot.
395pub struct SlotView {
396    pub slot: Slot,
397    /// The commit the slot configures, which its guardian runs once the stack has deployed it.
398    pub commit: String,
399    /// The configuration key holding the slot's PCR0.
400    pub pcr0_key: String,
401    pub role: Role,
402    pub instance: Option<String>,
403    /// The local port its tunnel forwards.
404    pub port: u16,
405    pub endpoint: String,
406    /// Its rendered files.
407    pub dir: PathBuf,
408}
409
410impl SlotView {
411    pub fn tunnel_down(&self) -> String {
412        format!(
413            "Could not reach slot {slot}'s guardian at {}. Run the tunnel step for slot {slot} again: it closes once idle.",
414            self.endpoint,
415            slot = self.slot
416        )
417    }
418
419    /// The slot's instance, for steps that act on a deployed guardian.
420    pub fn require_instance(&self, stack: &Stack) -> Result<&str> {
421        self.instance.as_deref().ok_or_else(|| {
422            anyhow!(
423                "slot {} of stack {} has no deployed guardian",
424                self.slot,
425                stack.name
426            )
427        })
428    }
429
430    /// Stops unless the slot's host was launched to build the commit the slot configures, whose protos decode its
431    /// guardian's answers. The stack records no standby's commit, but every host's user-data names its own.
432    pub fn require_deployed(&self, ctx: &Ctx, stack: &Stack) -> Result<()> {
433        let instance = self.require_instance(stack)?;
434        if !host::user_data(ctx, &stack.bucket()?.region, instance)?.contains(&self.commit) {
435            bail!(
436                "stack {} configures commit {} for slot {}, but its host {instance} was launched to build another commit; set it back, or deploy it, first",
437                stack.name,
438                self.commit,
439                self.slot
440            );
441        }
442        Ok(())
443    }
444}

463pub fn is_hex(value: &str, length: usize) -> bool {
464    value.len() == length
465        && value
466            .bytes()
467            .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
468}
469
470fn scalar(value: Value) -> String {
471    match value {
472        Value::String(text) => text,
473        Value::Null => String::new(),
474        other => other.to_string(),
475    }
