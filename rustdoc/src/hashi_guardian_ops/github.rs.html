20/// A workflow file of a GitHub repository, or with a trailing `*`, every workflow file its name prefixes.
21#[derive(Debug, PartialEq, Eq)]
22pub struct Workflow {
23    pub repo: String,
24    pub file: String,
25}
26
27impl Workflow {
28    /// Reads `<owner>/<repo>/.github/workflows/<file>`, as a `uses:` line names a workflow, and the `@<ref>` after it.
29    fn parse(text: &str) -> Option<(Self, Option<&str>)> {
30        let (path, git_ref) = match text.split_once('@') {
31            Some((path, git_ref)) => (path, Some(git_ref)),
32            None => (text, None),
33        };
34        let (owner, rest) = path.split_once('/')?;
35        let (repo, rest) = rest.split_once('/')?;
36        let file = rest.strip_prefix(".github/workflows/")?;
37        let named = |part: &str| {
38            !part.is_empty()
39                && part
40                    .bytes()
41                    .all(|b| b.is_ascii_alphanumeric() || b"._-".contains(&b))
42        };
43        let workflow = Self {
44            repo: format!("{owner}/{repo}"),
45            file: file.to_owned(),
46        };
47        (named(owner) && named(repo) && !file.is_empty() && !file.contains('/'))
48            .then_some((workflow, git_ref))
49    }
50
51    fn matches(&self, file: &str) -> bool {
52        match self.file.strip_suffix('*') {
53            Some(prefix) => file.starts_with(prefix),
54            None => file == self.file,
55        }
56    }
57}
58
59impl fmt::Display for Workflow {
60    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
61        write!(f, "{}/.github/workflows/{}", self.repo, self.file)
62    }
63}

65/// `GUARDIAN_PROXY_WORKFLOW`, the workflow the proxy step dispatches, and the branch or tag it runs at.
66pub fn proxy_workflow(ctx: &Ctx) -> Result<(Workflow, String)> {
67    let text = ctx.env.require("GUARDIAN_PROXY_WORKFLOW")?;
68    match Workflow::parse(text) {
69        Some((workflow, Some(git_ref)))
70            if !git_ref.is_empty()
71                && !workflow.file.contains('*')
72                && (workflow.file.ends_with(".yml") || workflow.file.ends_with(".yaml")) =>
73        {
74            Ok((workflow, git_ref.to_owned()))
75        }
76        _ => bail!(
77            "GUARDIAN_PROXY_WORKFLOW is not <owner>/<repo>/.github/workflows/<file>@<branch or tag>: {text}"
78        ),
79    }
80}

82/// `GUARDIAN_WORKFLOWS`, the workflows that apply the guardian stacks.
83fn guardian_workflows(ctx: &Ctx) -> Result<Vec<Workflow>> {
84    ctx.env
85        .require("GUARDIAN_WORKFLOWS")?
86        .split_whitespace()
87        .map(|entry| match Workflow::parse(entry) {
88            Some((workflow, None))
89                if workflow
90                    .file
91                    .find('*')
92                    .is_none_or(|at| at + 1 == workflow.file.len()) =>
93            {
94                Ok(workflow)
95            }
96            _ => bail!(
97                "GUARDIAN_WORKFLOWS lists {entry}, not <owner>/<repo>/.github/workflows/<file>, where a trailing * matches by prefix"
98            ),
99        })
100        .collect()
101}

103/// Stops unless `gh` reports the run a dispatch starts, which it does from 2.87 on.
104pub fn require_run_urls(ctx: &Ctx) -> Result<()> {
105    let version = read(ctx.exec, &Cmd::new("gh").arg("--version"))
106        .context("could not read the GitHub CLI's version")?;
107    let first = version.lines().next().unwrap_or("");
108    let numbers: Vec<u32> = first
109        .strip_prefix("gh version ")
110        .unwrap_or("")
111        .split(|c: char| !c.is_ascii_digit())
112        .take(2)
113        .filter_map(|part| part.parse().ok())
114        .collect();
115    match numbers[..] {
116        [major, minor] if major > 2 || (major == 2 && minor >= 87) => Ok(()),
117        _ => bail!(
118            "this step needs GitHub CLI 2.87 or later, which reports the run it dispatches; found: {first}"
119        ),
120    }
121}

123/// Stops while a workflow that applies the guardian stacks has a run that hasn't completed, since it would race this
124/// step: one `GUARDIAN_WORKFLOWS` lists, or the proxy's. A reusable workflow's runs are listed under its caller's.
125pub fn require_no_guardian_runs(ctx: &Ctx) -> Result<()> {
126    let mut workflows = guardian_workflows(ctx)?;
127    if ctx.env.get("GUARDIAN_PROXY_WORKFLOW").is_some() {
128        workflows.push(proxy_workflow(ctx)?.0);
129    }
130    let repos: BTreeSet<&str> = workflows
131        .iter()
132        .map(|workflow| workflow.repo.as_str())
133        .collect();
134    for repo in repos {
135        // A run that waits for an approval or a concurrency slot starts once it has one.
136        for status in ["requested", "waiting", "pending", "queued", "in_progress"] {
137            let paths = read(
138                ctx.exec,
139                &Cmd::new("gh")
140                    .args(["api", "--paginate"])
141                    .arg(format!(
142                        "repos/{repo}/actions/runs?status={status}&per_page=100"
143                    ))
144                    .args(["--jq", ".workflow_runs[].path"]),
145            )
146            .with_context(|| format!("could not list {repo}'s {status} workflow runs"))?;
147            // GitHub can name the ref a run's workflow came from after an @.
148            if let Some(path) = paths
149                .lines()
150                .map(|path| path.split('@').next().unwrap_or(path))
151                .find(|path| {
152                    path.strip_prefix(".github/workflows/").is_some_and(|file| {
153                        workflows
154                            .iter()
155                            .any(|workflow| workflow.repo == repo && workflow.matches(file))
156                    })
157                })
158            {
159                bail!(
160                    "{repo}'s {path} has a run {status}, and it applies the guardian stacks too; wait for it to finish"
161                );
162            }
163        }
164    }
165    Ok(())
166}

168/// Dispatches a workflow and returns the ID of the run it started.
169pub fn dispatch(
170    ctx: &Ctx,
171    repo: &str,
172    workflow: &str,
173    git_ref: &str,
174    inputs: &[(&str, &str)],
175) -> Result<String> {
176    let mut cmd = Cmd::new("gh").args([
177        "workflow", "run", workflow, "--repo", repo, "--ref", git_ref,
178    ]);
179    for (key, value) in inputs {
180        cmd = cmd.arg("-f").arg(format!("{key}={value}"));
181    }
182    // A failure may come after GitHub has started the run, so neither message reads as though dispatching again were
183    // safe.
184    let printed = read(ctx.exec, &cmd).with_context(|| {
185        format!(
186            "could not tell whether {workflow} was dispatched; check {repo}'s runs before going on"
187        )
188    })?;
189    let url = printed.trim().lines().last().unwrap_or("");
190    // GitHub spells the owner and repository its own way, whatever case the environment file used.
191    let runs = format!("https://github.com/{repo}/actions/runs/");
192    let id = url
193        .get(..runs.len())
194        .filter(|start| start.eq_ignore_ascii_case(&runs))
195        .and_then(|_| url.get(runs.len()..))
196        .unwrap_or("");
197    if id.is_empty() || !id.bytes().all(|b| b.is_ascii_digit()) {
198        bail!("dispatched {workflow}, but could not tell which run it started: {url}");
199    }
200    Ok(id.to_owned())
201}

203/// What GitHub says of a run.
204#[derive(Deserialize)]
205#[serde(rename_all = "camelCase")]
206pub struct Run {
207    pub event: String,
208    pub workflow_name: String,
209    pub conclusion: String,
210}
211
212pub fn run(ctx: &Ctx, repo: &str, run_id: &str) -> Result<Run> {
213    read_json(
214        ctx.exec,
215        &Cmd::new("gh")
216            .args(["run", "view", run_id, "--repo", repo])
217            .args(["--json", "event,workflowName,conclusion"]),
218    )
219    .with_context(|| format!("could not read run {run_id} of {repo}"))
220}

222/// Waits for a run to finish, and stops unless it succeeded. A watch that fails tells nothing of the run itself, which
223/// may still be going or may have succeeded, so its conclusion is asked for then.
224pub fn watch(ctx: &Ctx, repo: &str, run_id: &str) -> Result<()> {
225    let watched = read(
226        ctx.exec,
227        &Cmd::new("gh").args(["run", "watch", run_id, "--repo", repo, "--exit-status"]),
228    );
229    if watched.is_ok() {
230        return Ok(());
231    }
232    let url = format!("https://github.com/{repo}/actions/runs/{run_id}");
233    match run(ctx, repo, run_id) {
234        Ok(run) if run.conclusion == "success" => Ok(()),
235        Ok(run) if !run.conclusion.is_empty() => bail!("run {url} ended in {}", run.conclusion),
236        _ => bail!("could not tell how run {url} ends; check it there before going on"),
237    }
238}

240/// The PCR0 both runners of a measurement agree on, from their `nitro.pcrs` files.
241pub fn agreed_pcr0(pcrs: &[String]) -> Result<String> {
242    let measured: Vec<&str> = pcrs
243        .iter()
244        .flat_map(|file| file.lines())
245        .filter(|line| line.contains("PCR0"))
246        .filter_map(|line| line.split_whitespace().next())
247        .collect();
248    match measured[..] {
249        [first, second] if first == second && crate::stack::is_hex(first, 96) => {
250            Ok(first.to_owned())
251        }
252        _ => bail!(
253            "the two runners did not agree on one PCR0: {}",
254            measured.join(" ")
255        ),
256    }
257}

259/// Stops unless a measurement run's log shows it built `commit` on both runners, with no feature that changes the
260/// image, such as `non-enclave-dev`.
261pub fn require_canonical_build(log: &str, run_id: &str, commit: &str) -> Result<()> {
262    let checkout = format!("checkout --progress --force {commit}");
263    let checkouts = log.lines().filter(|line| line.contains(&checkout)).count();
264    if checkouts != 2 {
265        bail!("run {run_id} did not build {commit} on both runners");
266    }
267    // Each runner's steps print their environment, FEATURES among it; the build recipe's own `${FEATURES:+...}` has no
268    // space after the colon.
269    let features: Vec<&str> = log
270        .lines()
271        .filter_map(|line| line.split_once("FEATURES:").map(|(_, rest)| rest))
272        .filter(|rest| rest.starts_with(char::is_whitespace))
273        .collect();
274    if features.len() < 2 {
275        bail!(
276            "run {run_id}'s log does not show the features of both builds, so it cannot tell they had none"
277        );
278    }
279    if features.iter().any(|rest| !rest.trim().is_empty()) {
280        bail!(
281            "run {run_id} built with extra features, so it does not measure the guardian's image"
282        );
283    }
284    Ok(())
285}

298    #[test]
299    fn reads_workflows_as_uses_names_them() {
300        let repo = tempfile::tempdir().unwrap();
301        let exec = crate::exec::fake::Fake::new(|_| None);
302        let proxy = |text: &str| {
303            let ctx = Ctx::for_test(repo.path(), &exec, &[("GUARDIAN_PROXY_WORKFLOW", text)]);
304            proxy_workflow(&ctx).map_err(|error| error.to_string())
305        };
306        assert_eq!(
307            proxy("org/infra/.github/workflows/proxy.yaml@main").unwrap(),
308            (workflow("proxy.yaml"), "main".to_owned())
309        );
310        assert_eq!(
311            proxy("org/infra/.github/workflows/proxy.yml@release/1").unwrap(),
312            (workflow("proxy.yml"), "release/1".to_owned())
313        );
314        for bad in [
315            "org/infra/.github/workflows/proxy.yaml",
316            "org/infra/.github/workflows/proxy.yaml@",
317            "org/infra/.github/workflows/proxy-*@main",
318            "org/infra/.github/workflows/nested/proxy.yaml@main",
319            "org/infra/workflows/proxy.yaml@main",
320            "org/infra/.github/workflows/proxy.json@main",
321            "org/in fra/.github/workflows/proxy.yaml@main",
322        ] {
323            assert_eq!(
324                proxy(bad).unwrap_err(),
325                format!(
326                    "GUARDIAN_PROXY_WORKFLOW is not <owner>/<repo>/.github/workflows/<file>@<branch or tag>: {bad}"
327                )
328            );
329        }
330
331        let listed = |text: &str| {
332            let ctx = Ctx::for_test(repo.path(), &exec, &[("GUARDIAN_WORKFLOWS", text)]);
333            guardian_workflows(&ctx).map_err(|error| error.to_string())
334        };
335        let found = listed(
336            "org/infra/.github/workflows/guardian-* org/infra/.github/workflows/deploy.yaml",
337        )
338        .unwrap();
339        assert_eq!(found, [workflow("guardian-*"), workflow("deploy.yaml")]);
340        assert!(found[0].matches("guardian-proxy.yaml"));
341        assert!(!found[0].matches("lint.yml"));
342        assert!(found[1].matches("deploy.yaml"));
343        assert!(!found[1].matches("deploy.yaml.old"));
344        for bad in [
345            "org/infra/.github/workflows/guardian-*@main",
346            "org/infra/.github/workflows/*-guardian.yaml",

348            assert!(
349                listed(bad)
350                    .unwrap_err()
351                    .starts_with(&format!("GUARDIAN_WORKFLOWS lists {bad}"))
352            );
353        }
354    }

384    #[test]
385    fn needs_both_runners_to_agree() {
386        let pcr0 = "0c".repeat(48);
387        let file = |value: &str| format!("{value} PCR0\n{} PCR1\n", "1".repeat(96));
388        assert_eq!(agreed_pcr0(&[file(&pcr0), file(&pcr0)]).unwrap(), pcr0);
389        assert!(agreed_pcr0(&[file(&pcr0), file(&"0d".repeat(48))]).is_err());
390        assert!(agreed_pcr0(&[file(&pcr0)]).is_err());
391        assert!(agreed_pcr0(&[file(&pcr0), file(&pcr0), file(&pcr0)]).is_err());
392    }

395    fn reads_what_the_runners_built() {
396        let commit = "0873917ea3daf0ac99e123215808ad602eb5fafb";
397        let log = format!(
398            "a\tgit checkout --progress --force {commit}\nb\tgit checkout --progress --force {commit}\na\tFEATURES: \nb\tFEATURES: \na\tRUN cargo build ${{FEATURES:+--features \"$FEATURES\"}}\n"

406        let one = format!("a\tgit checkout --progress --force {commit}\n");
