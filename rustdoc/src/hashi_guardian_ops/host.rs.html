17/// The host's units: the enclave, the bridge that serves its gRPC port, and its vsock proxies to S3.

20pub struct HostOutput {
21    /// What the commands printed, stdout then stderr.
22    pub text: String,
23    pub success: bool,
24}
25
26/// Runs shell commands on a guardian host as one script and returns what they printed, and whether the script ended in
27/// success. SSM reports only the last command's status, so a check that must fail the run exits non-zero itself.
28pub fn run(ctx: &Ctx, region: &str, instance: &str, commands: &[&str]) -> Result<HostOutput> {
29    let command_id = read(
30        ctx.exec,
31        &aws_ssm(region, "send-command")
32            .args(["--instance-ids", instance])
33            .args(["--document-name", "AWS-RunShellScript"])
34            .args(["--query", "Command.CommandId", "--output", "text"])
35            .arg("--parameters")
36            .arg(format!("commands={}", serde_json::to_string(commands)?)),
37    )
38    .with_context(|| {
39        format!(
40            "could not send commands to {instance}; a new instance takes a minute to accept them"
41        )
42    })?
43    .trim()
44    .to_owned();
45    let success = ctx
46        .exec
47        .output(
48            &Cmd::new("aws")
49                .args(["ssm", "wait", "command-executed", "--region", region])
50                .args(["--command-id", &command_id, "--instance-id", instance]),
51        )
52        .context("could not wait for the host's commands")?
53        .success;
54    let text = read(
55        ctx.exec,
56        &aws_ssm(region, "get-command-invocation")
57            .args(["--command-id", &command_id, "--instance-id", instance])
58            .args(["--query", "[StandardOutputContent, StandardErrorContent]"])
59            .args(["--output", "text"]),
60    )
61    .with_context(|| format!("could not read what the commands printed on {instance}"))?;
62    Ok(HostOutput { text, success })
63}

65/// The EC2 state of a host, such as `running` or `terminated`, or `None` once EC2 no longer knows it. EC2 says the
66/// same of an instance in another account, so `None` needs credentials for `account`, the host's.
67pub fn instance_state(
68    ctx: &Ctx,
69    region: &str,
70    instance: &str,
71    account: &str,
72) -> Result<Option<String>> {
73    let output = ctx.exec.output(
74        &Cmd::new("aws")
75            .args(["ec2", "describe-instances", "--region", region])
76            .args(["--instance-ids", instance])
77            .args(["--query", "Reservations[].Instances[].State.Name"])
78            .args(["--output", "text"])
79            .capture_stderr(),
80    )?;
81    if output.success {
82        return Ok(Some(output.stdout.trim().to_owned()));
83    }
84    if !output.stderr.contains("InvalidInstanceID.NotFound") {
85        bail!(
86            "could not read the state of {instance}: {}",
87            output.stderr.trim()
88        );
89    }
90    let caller = read(
91        ctx.exec,
92        &Cmd::new("aws")
93            .args(["sts", "get-caller-identity", "--region", region])
94            .args(["--query", "Account", "--output", "text"]),
95    )
96    .context("could not read which AWS account your credentials are for")?;
97    if caller.trim() != account {
98        bail!(
99            "EC2 does not know {instance}, but your AWS credentials are for account {}, not {account}, which runs it",
100            caller.trim()
101        );
102    }
103    Ok(None)
104}

106fn aws_ssm(region: &str, action: &str) -> Cmd {
107    Cmd::new("aws").args(["ssm", action, "--region", region])
108}

110/// The user-data the host was launched with, which names the commit it builds its guardian from.
111pub fn user_data(ctx: &Ctx, region: &str, instance: &str) -> Result<String> {
112    let encoded = read(
113        ctx.exec,
114        &Cmd::new("aws")
115            .args(["ec2", "describe-instance-attribute", "--region", region])
116            .args(["--instance-id", instance, "--attribute", "userData"])
117            .args(["--query", "UserData.Value", "--output", "text"]),
118    )
119    .with_context(|| format!("could not read the user-data of {instance}"))?;
120    let decoded = Base64::decode_vec(encoded.trim())
121        .map_err(|_| anyhow!("the user-data of {instance} is not base64"))?;
122    Ok(String::from_utf8_lossy(&decoded).into_owned())
123}

125/// The commands the host step runs: the end of the boot log, whether the boot finished, the enclave and the units.
126pub fn report_commands() -> Vec<String> {
127    vec![
128        "tail -n 12 /var/log/hashi-guardian-bootstrap.log".to_owned(),
129        // The boot log traces its own commands, so match only the lines the boot itself prints.
130        r"grep -E '^\[[^]]*\] hashi-guardian user-data (complete|FAILED)' /var/log/hashi-guardian-bootstrap.log || echo 'The boot has not finished.'".to_owned(),
131        "nitro-cli describe-enclaves".to_owned(),
132        format!("systemctl is-active {UNITS} || true"),
133    ]
134}

136/// The PCR0 a host's `nitro-cli describe-enclaves` reports.
137pub fn reported_pcr0(report: &str) -> Option<&str> {
138    const KEY: &str = "\"PCR0\": \"";
139    let start = report.find(KEY)? + KEY.len();
140    let value = report.get(start..start + 96)?;
141    (report[start + 96..].starts_with('"') && crate::stack::is_hex(value, 96)).then_some(value)
142}

149    #[test]
150    fn runs_commands_as_bash_did() {
151        let exec = Fake::new(|words| match (words[2].as_str(), words[3].as_str()) {
152            ("send-command", _) => Some((true, "cmd-1\n".to_owned())),
153            ("wait", "command-executed") => Some((false, String::new())),
154            ("get-command-invocation", _) => Some((true, "out\terr\n".to_owned())),
155            _ => None,
156        });
157        let repo = tempfile::tempdir().unwrap();
158        let ctx = Ctx::for_test(repo.path(), &exec, &[]);
159        let commands = report_commands();
160        let commands: Vec<&str> = commands.iter().map(String::as_str).collect();
161        let output = run(&ctx, "us-west-2", "i-0aaa", &commands).unwrap();
162        assert!(!output.success);
163        assert_eq!(output.text, "out\terr\n");
164        let log = exec.log.borrow();
165        assert_eq!(
166            log[0][..12],
167            [
168                "aws",
169                "ssm",
170                "send-command",
171                "--region",
172                "us-west-2",
173                "--instance-ids",
174                "i-0aaa",
175                "--document-name",
176                "AWS-RunShellScript",
177                "--query",
178                "Command.CommandId",
179                "--output",
180            ]
181        );
182        // SSM reads the JSON array; the grep's backslashes reach the host's shell as one each.
183        let parameters: Vec<String> =
184            serde_json::from_str(log[0][14].strip_prefix("commands=").unwrap()).unwrap();
185        assert_eq!(
186            parameters[1],
187            r"grep -E '^\[[^]]*\] hashi-guardian user-data (complete|FAILED)' /var/log/hashi-guardian-bootstrap.log || echo 'The boot has not finished.'"
188        );
189        assert_eq!(
190            log[1][6..],
191            ["--command-id", "cmd-1", "--instance-id", "i-0aaa"]
192        );
193    }
