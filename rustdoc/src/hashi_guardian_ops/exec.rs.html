22/// The external programs the steps run.
23pub const REQUIRED: [(&str, &str); 8] = [
24    ("aws", "AWS CLI v2"),
25    ("cargo", "Rust toolchain (rustup)"),
26    ("curl", "standard system utilities"),
27    ("gh", "GitHub CLI"),
28    ("git", "git"),
29    ("grpcurl", "grpcurl"),
30    ("pulumi", "Pulumi CLI"),
31    ("tar", "standard system utilities"),
32];

34#[derive(Clone, Debug)]
35pub struct Cmd {
36    program: OsString,
37    args: Vec<OsString>,
38    cwd: Option<PathBuf>,
39    env_remove: Vec<OsString>,
40    capture_stderr: bool,
41}
42
43impl Cmd {
44    pub fn new(program: impl AsRef<OsStr>) -> Self {
45        Self {
46            program: program.as_ref().to_owned(),
47            args: Vec::new(),
48            cwd: None,
49            env_remove: Vec::new(),
50            capture_stderr: false,
51        }
52    }
53
54    /// Captures stderr too, apart from stdout, for a step that reports it itself.
55    pub fn capture_stderr(mut self) -> Self {
56        self.capture_stderr = true;
57        self
58    }
59
60    /// Leaves `name` out of the command's environment.
61    pub fn env_remove(mut self, name: impl AsRef<OsStr>) -> Self {
62        self.env_remove.push(name.as_ref().to_owned());
63        self
64    }
65
66    pub fn removed_env(&self) -> &[OsString] {
67        &self.env_remove
68    }
69
70    pub fn arg(mut self, arg: impl AsRef<OsStr>) -> Self {
71        self.args.push(arg.as_ref().to_owned());
72        self
73    }
74
75    pub fn args<I: IntoIterator<Item = S>, S: AsRef<OsStr>>(mut self, args: I) -> Self {
76        self.args
77            .extend(args.into_iter().map(|arg| arg.as_ref().to_owned()));
78        self
79    }
80
81    pub fn cwd(mut self, dir: impl AsRef<Path>) -> Self {
82        self.cwd = Some(dir.as_ref().to_owned());
83        self
84    }
85
86    /// The program and its arguments as text, for tests and messages.
87    pub fn words(&self) -> Vec<String> {
88        std::iter::once(&self.program)
89            .chain(&self.args)
90            .map(|word| word.to_string_lossy().into_owned())
91            .collect()
92    }
93
94    pub fn working_dir(&self) -> Option<&Path> {
95        self.cwd.as_deref()
96    }
97
98    fn command(&self) -> Command {
99        let mut command = Command::new(&self.program);
100        command.args(&self.args);
101        // Keeps parsed output stable regardless of the operator's locale, and the AWS CLI on its default endpoints
102        // and out of a pager.
103        command
104            .env("LC_ALL", "C")
105            .env("AWS_PAGER", "")
106            .env("AWS_IGNORE_CONFIGURED_ENDPOINT_URLS", "true");
107        if let Some(dir) = &self.cwd {
108            command.current_dir(dir);
109        }
110        for name in &self.env_remove {
111            command.env_remove(name);
112        }
113        command
114    }
115
116    fn describe(&self) -> String {
117        const LIMIT: usize = 160;
118        let text = self.words().join(" ");
119        match text.char_indices().nth(LIMIT) {
120            Some((end, _)) => format!("`{}…`", &text[..end]),
121            None => format!("`{text}`"),
122        }
123    }
124}
125
126pub struct Output {
127    pub success: bool,
128    pub stdout: String,
129    /// Empty unless the command captures stderr.
130    pub stderr: String,
131}
132
133pub trait Exec {
134    /// Runs `cmd` to completion with its stdout captured and its stderr on the terminal, as `$(...)` does, unless the
135    /// command captures stderr too.
136    fn output(&self, cmd: &Cmd) -> io::Result<Output>;
137
138    /// Runs `cmd` on the terminal, as a plain command line in a script does, and returns whether it succeeded.
139    fn run(&self, cmd: &Cmd) -> io::Result<bool>;
140
141    /// Replaces this process with `cmd`, as bash's `exec` does. Returns only if that fails.
142    fn exec(&self, cmd: &Cmd) -> io::Error;
143
144    /// Whether `program` is on PATH.
145    fn has(&self, program: &str) -> bool;
146}
147
148/// The real world.
149pub struct System;
150
151impl Exec for System {
152    fn output(&self, cmd: &Cmd) -> io::Result<Output> {
153        let stderr = if cmd.capture_stderr {
154            Stdio::piped()
155        } else {
156            Stdio::inherit()
157        };
158        // Without the terminal, a command can neither take an answer typed ahead for a step's prompt nor hang on a
159        // prompt of its own that nobody sees.
160        let output = cmd.command().stdin(Stdio::null()).stderr(stderr).output()?;
161        let text = |bytes: Vec<u8>| {
162            String::from_utf8(bytes)
163                .map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error))
164        };
165        Ok(Output {
166            success: output.status.success(),
167            stdout: text(output.stdout)?,
168            stderr: text(output.stderr)?,
169        })
170    }
171
172    fn run(&self, cmd: &Cmd) -> io::Result<bool> {
173        let mut child = cmd.command().spawn()?;
174        // Ctrl-C still reaches the child, which shares the terminal, but this waits for it to stop, as bash does: an
175        // interrupted `pulumi up` must finish its own cleanup before the operator gets the prompt back.
176        // SAFETY: `signal` only swaps this process's SIGINT disposition, and nothing else here touches it.
177        let previous = unsafe { libc::signal(libc::SIGINT, libc::SIG_IGN) };
178        let status = child.wait();
179        // SAFETY: as above, restoring the disposition `signal` returned.
180        unsafe { libc::signal(libc::SIGINT, previous) };
181        Ok(status?.success())
182    }
183
184    fn exec(&self, cmd: &Cmd) -> io::Error {
185        cmd.command().exec()
186    }
187
188    fn has(&self, program: &str) -> bool {
189        std::env::var_os("PATH").is_some_and(|paths| {
190            std::env::split_paths(&paths).any(|dir| {
191                dir.join(program)
192                    .metadata()
193                    .is_ok_and(|meta| meta.is_file() && meta.permissions().mode() & 0o111 != 0)
194            })
195        })
196    }
197}

199/// Runs `cmd` and returns its stdout, failing unless it exits successfully.
200pub fn read(exec: &dyn Exec, cmd: &Cmd) -> Result<String> {
201    let output = exec
202        .output(cmd)
203        .with_context(|| format!("could not run {}", cmd.describe()))?;
204    if !output.success {
205        return Err(anyhow!("{} failed", cmd.describe()));
206    }
207    Ok(output.stdout)
208}
209
210/// Runs `cmd` and decodes its stdout as JSON.
211pub fn read_json<T: DeserializeOwned>(exec: &dyn Exec, cmd: &Cmd) -> Result<T> {
212    let stdout = read(exec, cmd)?;
213    serde_json::from_str(&stdout).with_context(|| {
214        format!(
215            "{} printed something other than the JSON expected",
216            cmd.describe()
217        )
218    })
219}
220
221/// Stops unless every program the steps run is on PATH.
222pub fn require_programs(exec: &dyn Exec) -> Result<()> {
223    let missing: Vec<String> = REQUIRED
224        .iter()
225        .filter(|(program, _)| !exec.has(program))
226        .map(|(program, package)| format!("  - {program} ({package})"))
227        .collect();
228    if missing.is_empty() {
229        return Ok(());
230    }
231    Err(anyhow!(
232        "the following required CLI tools are not installed or not on PATH:\n{}\n\nInstall the listed tools, then run this step again",
233        missing.join("\n")
234    ))
235}

237#[cfg(test)]
238mod tests {
239    use super::*;
240
241    #[test]
242    fn runs_commands_in_the_scripts_environment() {
243        let command = Cmd::new("aws").env_remove("RUSTUP_TOOLCHAIN").command();
244        let envs: Vec<(String, Option<String>)> = command
245            .get_envs()
246            .map(|(key, value)| {
247                (
248                    key.to_string_lossy().into_owned(),
249                    value.map(|value| value.to_string_lossy().into_owned()),
250                )
251            })
252            .collect();
253        for (key, value) in [
254            ("LC_ALL", Some("C")),
255            ("AWS_PAGER", Some("")),
256            ("AWS_IGNORE_CONFIGURED_ENDPOINT_URLS", Some("true")),
257            ("RUSTUP_TOOLCHAIN", None),
258        ] {
259            assert!(
260                envs.contains(&(key.to_owned(), value.map(str::to_owned))),
261                "{key}"
262            );
263        }
264    }

267#[cfg(test)]
268pub mod fake {
269    //! A stand-in world: each command is answered by a handler, and every command is recorded.
270
271    use std::cell::RefCell;
272
273    use super::*;
274
275    pub type Handler = Box<dyn Fn(&[String]) -> Option<Output>>;
276
277    pub struct Fake {
278        handler: Handler,
279        pub log: RefCell<Vec<Vec<String>>>,
280        /// The variables each logged command leaves out of its environment.
281        pub removed_env: RefCell<Vec<Vec<String>>>,
282        /// The working directory of each command exec'd in place of this tool.
283        pub exec_dirs: RefCell<Vec<Option<PathBuf>>>,
284    }
285
286    impl Fake {
287        /// A world answering each command with `handler`'s (success, stdout); `None` means the command is unknown,
288        /// which fails the test.
289        pub fn new(handler: impl Fn(&[String]) -> Option<(bool, String)> + 'static) -> Self {
290            Self::with_stderr(move |words| {
291                handler(words).map(|(success, stdout)| Output {
292                    success,
293                    stdout,
294                    stderr: String::new(),
295                })
296            })
297        }
298
299        /// A world whose answers include what each command prints on stderr.
300        pub fn with_stderr(handler: impl Fn(&[String]) -> Option<Output> + 'static) -> Self {
301            Self {
302                handler: Box::new(handler),
303                log: RefCell::default(),
304                removed_env: RefCell::default(),
305                exec_dirs: RefCell::default(),
306            }
307        }
308
309        pub fn ran(&self, prefix: &[&str]) -> bool {
310            self.log.borrow().iter().any(|words| {
311                words.len() >= prefix.len() && words.iter().zip(prefix).all(|(w, p)| w == p)
312            })
313        }
314    }
315
316    impl Exec for Fake {
317        fn output(&self, cmd: &Cmd) -> io::Result<Output> {
318            let words = cmd.words();
319            self.log.borrow_mut().push(words.clone());
320            self.removed_env.borrow_mut().push(
321                cmd.removed_env()
322                    .iter()
323                    .map(|name| name.to_string_lossy().into_owned())
324                    .collect(),
325            );
326            Ok((self.handler)(&words).unwrap_or_else(|| panic!("unexpected command: {words:?}")))
327        }
328
329        fn run(&self, cmd: &Cmd) -> io::Result<bool> {
330            self.output(cmd).map(|output| output.success)
331        }
332
333        fn exec(&self, cmd: &Cmd) -> io::Error {
334            self.log.borrow_mut().push(cmd.words());
335            self.exec_dirs
336                .borrow_mut()
337                .push(cmd.working_dir().map(Path::to_owned));
338            io::Error::other("exec in a test")
339        }
340
341        fn has(&self, _program: &str) -> bool {
342            true
343        }
344    }
