4//! Pulumi previews and updates. A target that names nothing is no error to Pulumi, so every step compares a plan's
5//! changes with exactly what it expects before applying it.

21#[derive(Deserialize)]
22pub struct Plan {
23    #[serde(default)]
24    pub steps: Vec<Step>,
25    #[serde(rename = "changeSummary", default)]
26    pub change_summary: BTreeMap<String, u64>,
27}
28
29#[derive(Clone, Deserialize)]
30pub struct Step {
31    pub op: String,
32    pub urn: String,
33    #[serde(rename = "oldState")]
34    pub old_state: Option<Value>,
35    #[serde(rename = "newState")]
36    pub new_state: Option<Value>,
37}
38
39/// One resource change: its operation and URN.
40#[derive(Clone, Debug, PartialEq, Eq)]
41pub struct Change {
42    pub op: String,
43    pub urn: String,
44}
45
46impl Change {
47    pub fn new(op: &str, urn: &str) -> Self {
48        Self {
49            op: op.to_owned(),
50            urn: urn.to_owned(),
51        }
52    }
53}
54
55impl Plan {
56    /// The plan's resource changes, leaving out the stack's own outputs and reads, such as the proxy's reference to
57    /// the enclave stack.
58    pub fn changes(&self) -> Vec<Change> {
59        self.steps
60            .iter()
61            .filter(|step| !matches!(step.op.as_str(), "same" | "read" | "refresh"))
62            .filter(|step| !step.urn.contains("::pulumi:pulumi:Stack::"))
63            .map(|step| Change::new(&step.op, &step.urn))
64            .collect()
65    }
66
67    /// The URNs of the resources the plan creates. A replacement isn't one: it plans `create-replacement`.
68    pub fn creates(&self) -> Vec<&str> {
69        self.steps
70            .iter()
71            .filter(|step| step.op == "create")
72            .map(|step| step.urn.as_str())
73            .collect()
74    }
75
76    /// Prints the changes, as a step shows what it is about to apply.
77    pub fn print_changes(&self, target: &str) {
78        let changes = self.changes();
79        if changes.is_empty() {
80            print_info(&format!("Plan for {target}:\nno resource changes"));
81        } else {
82            print_info(&format!("Plan for {target}:"));
83            for change in changes {
84                print_detail(&format!("{} {}", change.op, change.urn));
85            }
86        }
87    }
88}

90/// Previews an up of the stack in `dir`, or of `target` and what depends on it.
91pub fn preview(ctx: &Ctx, dir: &str, stack: &str, target: Option<&str>) -> Result<Plan> {
92    let mut cmd = pulumi(dir).args(["preview", "--stack", stack]);
93    if let Some(target) = target {
94        cmd = cmd.args(["--target", target, "--target-dependents"]);
95    }
96    let output = ctx.exec.output(&cmd.arg("--json"))?;
97    if !output.success {
98        // With --json, Pulumi reports what failed inside the plan it prints.
99        #[derive(Deserialize)]
100        struct Failed {
101            #[serde(default)]
102            diagnostics: Vec<Diagnostic>,
103        }
104        #[derive(Deserialize)]
105        struct Diagnostic {
106            message: Option<String>,
107        }
108        match serde_json::from_str::<Failed>(&output.stdout) {
109            Ok(failed) => {
110                for message in failed.diagnostics.into_iter().filter_map(|d| d.message) {
111                    print_detail(&message);
112                }
113            }
114            Err(_) => print_detail(&output.stdout),
115        }
116        bail!("could not preview stack {stack}");
117    }
118    serde_json::from_str(&output.stdout)
119        .map_err(|error| anyhow!("Pulumi printed a plan this step cannot read: {error}"))
120}

122/// Applies the stack in `dir`, or `target` and what depends on it, with Pulumi's progress on the terminal.
123pub fn up(ctx: &Ctx, dir: &str, stack: &str, target: Option<&str>) -> Result<bool> {
124    let mut cmd = pulumi(dir).args(["up", "--stack", stack]);
125    if let Some(target) = target {
126        cmd = cmd.args(["--target", target, "--target-dependents"]);
127    }
128    Ok(ctx.exec.run(&cmd.arg("--yes"))?)
129}

131/// Previews a targeted up of one guardian instance and what depends on it, and prints it.
132pub fn preview_slot(ctx: &Ctx, stack: &Stack, urn: &str) -> Result<Vec<Change>> {
133    let plan = preview(ctx, &stack.dir, &stack.name, Some(urn))?;
134    plan.print_changes(urn);
135    Ok(plan.changes())
136}
137
138/// Applies a targeted up of one guardian instance.
139pub fn up_slot(ctx: &Ctx, stack: &Stack, urn: &str) -> Result<()> {
140    if !up(ctx, &stack.dir, &stack.name, Some(urn))? {
141        bail!(
142            "the update of stack {} did not finish; run this step again: it previews the stack first",
143            stack.name
144        );
145    }
146    Ok(())
147}

149/// Stops unless Pulumi is the version `PULUMI_VERSION` pins: the stacks' own automation applies them too, and a newer
150/// CLI can write state an older one cannot read.
151pub fn require_pulumi_version(ctx: &Ctx) -> Result<()> {
152    let pinned = crate::stack::pulumi_pin(ctx)?;
153    let current = crate::stack::pulumi_version(ctx)?;
154    if current != pinned {
155        bail!(
156            "this is Pulumi {current}, but PULUMI_VERSION is {pinned}; put Pulumi {pinned} first on your PATH"
157        );
158    }
159    Ok(())
160}

162/// A resource's type, from its URN `urn:pulumi:<stack>::<project>::<type>::<name>`, where a child's type follows its
163/// parents' after a `$`.
164pub fn urn_type(urn: &str) -> Option<&str> {
165    urn.split("::").nth(2)?.rsplit('$').next()
166}

168/// A preview's JSON with these steps, as Pulumi prints it, for tests.
169#[cfg(test)]
170pub fn plan_json(steps: &[(&str, &str)]) -> String {
171    let mut steps: Vec<Value> = steps
172        .iter()
173        .map(|(op, urn)| serde_json::json!({ "op": op, "urn": urn }))
174        .collect();
175    // As real previews list them: the stack itself, and the proxy's read of the enclave stack.
176    steps.push(
177        serde_json::json!({ "op": "same", "urn": "urn:pulumi:testnet::enclave::pulumi:pulumi:Stack::enclave-testnet" }),
178    );
179    steps.push(serde_json::json!({ "op": "read", "urn": "urn:pulumi:testnet::proxy::pulumi:pulumi:StackReference::org/enclave/testnet" }));
180    serde_json::json!({ "steps": steps, "changeSummary": { "same": 15 } }).to_string()
181}

183#[cfg(test)]
184mod tests {
185    use super::*;
186
187    const INSTANCE_B: &str = "urn:pulumi:testnet::enclave::aws:ec2/instance:Instance::guardian-b";
188
189    #[test]
190    fn leaves_out_the_stack_and_its_reads() {
191        let plan: Plan = serde_json::from_str(&plan_json(&[
192            ("same", "urn:a"),
193            ("create", INSTANCE_B),
194            ("refresh", "urn:c"),
195        ]))
196        .unwrap();
197        assert_eq!(plan.changes(), [Change::new("create", INSTANCE_B)]);
198        assert_eq!(plan.creates(), [INSTANCE_B]);
199        let replaced: Plan =
200            serde_json::from_str(&plan_json(&[("create-replacement", INSTANCE_B)])).unwrap();
201        assert!(replaced.creates().is_empty());
202    }
203
204    #[test]
205    fn reads_a_resources_type_from_its_urn() {
206        assert_eq!(urn_type(INSTANCE_B), Some(crate::stack::INSTANCE));
207        assert_eq!(
208            urn_type("urn:pulumi:testnet::enclave::pkg:index:Host$aws:ec2/instance:Instance::a::b"),
209            Some(crate::stack::INSTANCE)
210        );
211        assert_eq!(urn_type("urn:a"), None);
212    }
