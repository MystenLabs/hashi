25    /// The tree at `commit`, extracted once into `.hashi/guardian-init/<commit>/`. `whose` names what runs the
26    /// commit, for the message when this checkout doesn't have it.
27    pub fn at(ctx: &Ctx, commit: &str, whose: &str) -> Result<Self> {
28        let root = ctx.repo.join(".hashi/guardian-init");
29        let dir = root.join(commit);
30        if !dir.is_dir() {
31            if resolve(ctx, commit)?.is_none() {
32                bail!("this checkout does not have {commit}, {whose}; fetch it first");
33            }
34            extract(ctx, commit, &root, &dir)
35                .with_context(|| format!("could not extract {commit} into {}", dir.display()))?;
36        }
37        Ok(Self {
38            commit: commit.to_owned(),
39            dir,
40        })
41    }
42
43    /// The protos the guardian at this commit speaks, for grpcurl's `-import-path`.
44    pub fn protos(&self) -> PathBuf {
45        self.dir.join("crates/hashi-types/proto")
46    }

48    /// Builds `hashi-guardian-init` from this tree with its own toolchain and lock file. The first build of a commit
49    /// takes minutes; later ones only confirm it is current.
50    pub fn guardian_init(&self, ctx: &Ctx) -> Result<GuardianInit> {
51        let cargo = |args: &[&str]| {
52            build_overrides()
53                .into_iter()
54                .fold(Cmd::new("cargo").args(args).cwd(&self.dir), Cmd::env_remove)
55        };
56        if let Some(channel) = toolchain_channel(&self.dir)? {
57            let version = read(ctx.exec, &cargo(&["--version"]))?;
58            if !uses_channel(&version, &channel) {
59                bail!(
60                    "cargo in {} is {}, not toolchain {channel}, which {} pins",
61                    self.dir.display(),
62                    version.trim(),
63                    self.commit
64                );
65            }
66        }
67        let path = |relative: &str| {
68            self.dir
69                .join(relative)
70                .into_os_string()
71                .into_string()
72                .map_err(|_| anyhow!("the tree's path is not UTF-8"))
73        };
74        let messages = read(
75            ctx.exec,
76            &cargo(&[
77                "build",
78                "--manifest-path",
79                &path("Cargo.toml")?,
80                "--release",
81                "--locked",
82                "-p",
83                "hashi-guardian-init",
84                "--message-format=json-render-diagnostics",
85                "--target-dir",
86                &path("target")?,
87            ]),
88        )
89        .with_context(|| format!("could not build hashi-guardian-init at {}", self.commit))?;
90        Ok(GuardianInit {
91            path: executable(&messages)?,
92        })
93    }
94}

106/// The full commit `rev` names in this checkout, if it names one.
107pub fn resolve(ctx: &Ctx, rev: &str) -> Result<Option<String>> {
108    let output = ctx
109        .exec
110        .output(&git(&ctx.repo).args([
111            "rev-parse",
112            "--verify",
113            "--quiet",
114            &format!("{rev}^{{commit}}"),
115        ]))
116        .with_context(|| format!("could not run git in {}", ctx.repo.display()))?;
117    Ok(output.success.then(|| output.stdout.trim().to_owned()))
118}
119
120/// Stops unless `commit` has the guardian protos and log types of `other`, `whose` build: each guardian must read the
121/// records the other wrote.
122pub fn require_same_formats(ctx: &Ctx, commit: &str, other: &str, whose: &str) -> Result<()> {
123    for rev in [commit, other] {
124        if resolve(ctx, rev)?.is_none() {
125            bail!(
126                "this checkout does not have {rev}, so it cannot compare {commit} with {whose}; fetch it first"
127            );
128        }
129    }
130    let changed = read(
131        ctx.exec,
132        &git(&ctx.repo)
133            .args(["diff", "--name-only", other, commit, "--"])
134            .args(FORMATS),
135    )
136    .with_context(|| format!("could not compare {commit} with {other}"))?;
137    if !changed.trim().is_empty() {
138        bail!(
139            "{commit} changes the guardian's protos or log records from {whose}, {other}: {}; a rotation cannot move the key across that without a compatibility review",
140            changed.split_whitespace().collect::<Vec<_>>().join(", ")
141        );
142    }
143    Ok(())
144}

156    /// Whether this build has `tools <name>`; older builds lack some.
157    pub fn has_tool(&self, ctx: &Ctx, name: &str) -> Result<bool> {
158        let help = read(ctx.exec, &self.cmd().args(["tools", "--help"]))
159            .context("could not run the guardian's tools")?;
160        Ok(help.contains(name))
161    }

164/// Environment variables that would build with another toolchain or configuration: all of cargo's but where it finds
165/// its home, its network and its registries. A parent `cargo run` sets `RUSTUP_TOOLCHAIN`, which outranks the tree's
166/// `rust-toolchain.toml`.
167fn is_build_override(name: &str) -> bool {
168    let keeps = name == "CARGO_HOME"
169        || [
170            "CARGO_NET_",
171            "CARGO_HTTP_",
172            "CARGO_REGISTRIES_",
173            "CARGO_REGISTRY_",
174        ]
175        .iter()
176        .any(|prefix| name.starts_with(prefix));
177    matches!(name, "RUSTUP_TOOLCHAIN" | "RUSTFLAGS" | "RUSTDOCFLAGS")
178        || name.starts_with("RUSTC")
179        || (name.starts_with("CARGO") && !keeps)
180}
181
182fn build_overrides() -> Vec<std::ffi::OsString> {
183    std::env::vars_os()
184        .map(|(name, _)| name)
185        .filter(|name| name.to_str().is_some_and(is_build_override))
186        .collect()
187}

189/// The channel the tree's `rust-toolchain.toml` pins, if it has one.
190fn toolchain_channel(dir: &Path) -> Result<Option<String>> {
191    let path = dir.join("rust-toolchain.toml");
192    if !path.exists() {
193        return Ok(None);
194    }
195    let text = std::fs::read_to_string(&path)?;
196    let channel = text.lines().find_map(|line| {
197        let (key, value) = line.split_once('=')?;
198        (key.trim() == "channel").then(|| value.trim().trim_matches('"').to_owned())
199    });
200    channel
201        .map(Some)
202        .with_context(|| format!("{} names no channel", path.display()))
203}
204
205/// Whether `cargo --version` output is the pinned channel. Only a version number can be checked this way.
206fn uses_channel(version: &str, channel: &str) -> bool {
207    if !channel.bytes().all(|b| b.is_ascii_digit() || b == b'.') {
208        return true;
209    }
210    version
211        .strip_prefix("cargo ")
212        .and_then(|rest| rest.strip_prefix(channel))
213        .is_some_and(|rest| rest.starts_with(['.', ' ', '-']))
214}

216/// The path of the `hashi-guardian-init` binary in cargo's JSON messages.
217fn executable(messages: &str) -> Result<PathBuf> {
218    #[derive(serde::Deserialize)]
219    struct Message {
220        reason: String,
221        target: Option<Target>,
222        executable: Option<PathBuf>,
223    }
224    #[derive(serde::Deserialize)]
225    struct Target {
226        name: String,
227    }
228    messages
229        .lines()
230        .filter_map(|line| serde_json::from_str::<Message>(line).ok())
231        .filter(|message| message.reason == "compiler-artifact")
232        .filter(|message| {
233            message
234                .target
235                .as_ref()
236                .is_some_and(|target| target.name == "hashi-guardian-init")
237        })
238        .find_map(|message| message.executable)
239        .context("cargo built no hashi-guardian-init binary")
240}

242/// Extracts beside `dir` and renames into place, so an interrupted extraction never looks complete. Each process
243/// extracts into its own directory, since two steps can reach a new commit at once.
244fn extract(ctx: &Ctx, commit: &str, root: &Path, dir: &Path) -> Result<()> {
245    let partial = root.join(format!(".{commit}.partial-{}", std::process::id()));
246    if partial.exists() {
247        std::fs::remove_dir_all(&partial)?;
248    }
249    let renamed = unpack(ctx, commit, &partial).map(|tree| std::fs::rename(tree, dir));
250    // The partial directory goes whether or not the extraction finished.
251    let cleaned = std::fs::remove_dir_all(&partial);
252    match renamed? {
253        // Another step's extraction finished first.
254        Err(_) if dir.is_dir() => {}
255        renamed => renamed?,
256    }
257    Ok(cleaned?)
258}
259
260/// Unpacks `commit` into `partial`'s `tree/`.
261fn unpack(ctx: &Ctx, commit: &str, partial: &Path) -> Result<PathBuf> {
262    let tree = partial.join("tree");
263    std::fs::create_dir_all(&tree)?;
264    let archive = partial.join("tree.tar");
265    read(
266        ctx.exec,
267        &git(&ctx.repo)
268            .args(["archive", "--format=tar", "-o"])
269            .arg(&archive)
270            .arg(commit),
271    )?;
272    read(
273        ctx.exec,
274        &Cmd::new("tar")
275            .arg("-x")
276            .arg("-f")
277            .arg(&archive)
278            .arg("-C")
279            .arg(&tree),
280    )?;
281    Ok(tree)
282}

285    Cmd::new("git").arg("-C").arg(repo)
