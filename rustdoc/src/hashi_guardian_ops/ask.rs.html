4//! Questions to the operator before a step changes anything.
5
6use std::io;
7use std::io::BufRead;
8use std::io::IsTerminal;
9
10use anyhow::Result;
11use anyhow::bail;
12
13pub trait Ask {
14    /// Whether stdin and stderr are a terminal, as a step that must be watched requires.
15    fn interactive(&self) -> bool;
16
17    /// The line typed after `prompt`, or `None` when stdin has closed.
18    fn line(&self, prompt: &str) -> io::Result<Option<String>>;
19}
20
21/// The operator at the terminal.
22pub struct Terminal;
23
24impl Ask for Terminal {
25    fn interactive(&self) -> bool {
26        io::stdin().is_terminal() && io::stderr().is_terminal()
27    }
28
29    fn line(&self, prompt: &str) -> io::Result<Option<String>> {
30        // Discard what was typed before the question was shown, so it can't answer it.
31        // SAFETY: tcflush takes no pointers; it only drops the terminal's unread input.
32        if io::stdin().is_terminal()
33            && unsafe { libc::tcflush(libc::STDIN_FILENO, libc::TCIFLUSH) } != 0
34        {
35            return Err(io::Error::last_os_error());
36        }
37        eprint!("{prompt}");
38        let mut line = String::new();
39        if io::stdin().lock().read_line(&mut line)? == 0 {
40            return Ok(None);
41        }
42        Ok(Some(line.trim_end_matches(['\n', '\r']).to_owned()))
43    }
44}
45
46/// What a step says when the operator doesn't go ahead: on no input, and on any answer but the one asked for.
47pub struct Unchanged {
48    pub no_input: &'static str,
49    pub declined: &'static str,
50}
51
52/// The usual wording, for steps that would have changed anything.
53pub const NOTHING_CHANGED: Unchanged = Unchanged {
54    no_input: "no input received; nothing was changed",
55    declined: "not confirmed; nothing was changed",
56};
57
58/// Asks for y or yes, and stops on anything else.
59pub fn confirm(ask: &dyn Ask, question: &str, unchanged: &Unchanged) -> Result<()> {
60    let answer = ask
61        .line(&format!("{question} [y/N] "))?
62        .map(|answer| answer.trim().to_ascii_lowercase());
63    match answer.as_deref() {
64        None => bail!("{}", unchanged.no_input),
65        Some("y" | "yes") => Ok(()),
66        Some(_) => bail!("{}", unchanged.declined),
67    }
68}
69
70/// Asks the operator to type `expected`, such as the instance ID a step would destroy, and stops on anything else.
71pub fn confirm_typed(
72    ask: &dyn Ask,
73    expected: &str,
74    purpose: &str,
75    unchanged: &Unchanged,
76) -> Result<()> {
77    match ask.line(&format!("Type {expected} {purpose}: "))? {
78        None => bail!("{}", unchanged.no_input),
79        Some(typed) if typed == expected => Ok(()),
80        Some(_) => bail!("{}", unchanged.declined),
81    }
82}

84#[cfg(test)]
85pub mod scripted {
86    //! Answers given in advance, recording each prompt.
87
88    use std::cell::RefCell;
89    use std::collections::VecDeque;
90
91    use super::*;
92
93    /// A terminal nobody answers at.
94    pub struct NoAnswers;
95
96    pub static NO_ANSWERS: NoAnswers = NoAnswers;
97
98    impl Ask for NoAnswers {
99        fn interactive(&self) -> bool {
100            true
101        }
102
103        fn line(&self, _prompt: &str) -> io::Result<Option<String>> {
104            Ok(None)
105        }
106    }
107
108    /// No terminal at all, as when a step runs from a script.
109    pub struct NoTerminal;
110
111    impl Ask for NoTerminal {
112        fn interactive(&self) -> bool {
113            false
114        }
115
116        fn line(&self, _prompt: &str) -> io::Result<Option<String>> {
117            Ok(None)
118        }
119    }
120
121    #[derive(Default)]
122    pub struct Scripted {
123        answers: RefCell<VecDeque<String>>,
124        pub prompts: RefCell<Vec<String>>,
125    }
126
127    impl Scripted {
128        pub fn new(answers: &[&str]) -> Self {
129            Self {
130                answers: RefCell::new(answers.iter().map(|answer| (*answer).to_owned()).collect()),
131                prompts: RefCell::default(),
132            }
133        }
134    }
135
136    impl Ask for Scripted {
137        fn interactive(&self) -> bool {
138            true
139        }
140
141        fn line(&self, prompt: &str) -> io::Result<Option<String>> {
142            self.prompts.borrow_mut().push(prompt.to_owned());
143            Ok(self.answers.borrow_mut().pop_front())
144        }
145    }
146
147    #[test]
148    fn confirms_only_y_or_yes() {
149        assert!(confirm(&Scripted::new(&["yes"]), "Go?", &NOTHING_CHANGED).is_ok());
150        assert!(confirm(&Scripted::new(&["y"]), "Go?", &NOTHING_CHANGED).is_ok());
151        assert!(confirm(&Scripted::new(&[" Y "]), "Go?", &NOTHING_CHANGED).is_ok());
152        for answer in ["yes please", "", "no", "n"] {
153            assert_eq!(
154                confirm(&Scripted::new(&[answer]), "Go?", &NOTHING_CHANGED)
155                    .unwrap_err()
156                    .to_string(),
157                "not confirmed; nothing was changed"
158            );
159        }
160        let ask = Scripted::new(&[]);
161        assert_eq!(
162            confirm(&ask, "Go?", &NOTHING_CHANGED)
163                .unwrap_err()
164                .to_string(),
165            "no input received; nothing was changed"
166        );
167        assert_eq!(ask.prompts.borrow()[0], "Go? [y/N] ");
168        assert!(
169            confirm_typed(
170                &Scripted::new(&["i-0aaa"]),
171                "i-0aaa",
172                "to stop it",
173                &NOTHING_CHANGED
174            )
175            .is_ok()
176        );
177        assert!(
178            confirm_typed(
179                &Scripted::new(&["i-0aab"]),
180                "i-0aaa",
181                "to stop it",
182                &NOTHING_CHANGED
183            )
184            .is_err()
185        );
186    }
