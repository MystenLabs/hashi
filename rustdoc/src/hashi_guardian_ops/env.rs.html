15/// The keys the file may set. It used to be sourced by bash, which let it set anything, such as `PATH` or
16/// `AWS_PROFILE`; those belong in the shell now, so any other key is refused.
17const KEYS: &[&str] = &[
18    "GUARDIAN_PULUMI_DIR",
19    "GUARDIAN_STACK",
20    "GUARDIAN_PROXY_PULUMI_DIR",
21    "HASHI_MONITOR_PULUMI_DIR",
22    "GUARDIAN_PROXY_URL",
23    "PULUMI_VERSION",
24    "GUARDIAN_PROXY_WORKFLOW",
25    "GUARDIAN_WORKFLOWS",
26    "KP_BUCKET",
27    "KP_ROSTER_DIR",
28    "KP_THRESHOLD",
29    "NEW_KP_ROSTER_DIR",
30    "NEW_KP_THRESHOLD",
31    "SUI_RPC",
32    "SUI_GRAPHQL_URL",
33    "HASHI_PACKAGE_ID",
34    "HASHI_OBJECT_ID",
35    "BITCOIN_NETWORK",
36    "RETENTION_ENVIRONMENT",
37    "TUNNEL_PORT",
38];
39
40const DEFAULT_TUNNEL_PORT: u16 = 3000;

42pub struct Env {
43    path: PathBuf,
44    values: BTreeMap<String, String>,
45}

53impl Env {
54    /// Reads `path`. Unlike when bash sourced it, a key the file leaves out stays unset: a shell value meant for
55    /// another stack would otherwise mix in.
56    pub fn load(path: &Path) -> Result<Self> {
57        let text = match std::fs::read_to_string(path) {
58            Ok(text) => text,
59            Err(error) if error.kind() == std::io::ErrorKind::NotFound => bail!(
60                "no environment file at {}; copy crates/hashi-guardian-ops/guardian.env.sample",
61                path.display()
62            ),
63            Err(error) => {
64                return Err(error).with_context(|| {
65                    format!("could not read the environment file {}", path.display())
66                });
67            }
68        };
69        let home = std::env::var("HOME").ok();
70        let values = parse(&text, home.as_deref())
71            .with_context(|| format!("could not read the environment file {}", path.display()))?;
72        Self::new(path.to_owned(), values)
73    }
74
75    pub fn new(path: PathBuf, values: BTreeMap<String, String>) -> Result<Self> {
76        debug_assert!(
77            values.keys().all(|key| KEYS.contains(&key.as_str())),
78            "values hold a key the file may not set: {:?}",
79            values.keys().collect::<Vec<_>>()
80        );
81        let env = Self { path, values };
82        if env.get("NEW_KP_ROSTER_DIR").is_some() != env.get("NEW_KP_THRESHOLD").is_some() {
83            bail!(
84                "{} sets only one of NEW_KP_ROSTER_DIR and NEW_KP_THRESHOLD; set both for a KP-set rotation, or neither",
85                env.path.display()
86            );
87        }
88        Ok(env)
89    }
90
91    /// A key's value. An empty value counts as unset, as `${KEY:-}` reads it.
92    pub fn get(&self, key: &str) -> Option<&str> {
93        debug_assert!(KEYS.contains(&key), "{key} is not a key the file may set");
94        self.values
95            .get(key)
96            .map(String::as_str)
97            .filter(|value| !value.is_empty())
98    }
99
100    pub fn require(&self, key: &str) -> Result<&str> {
101        self.get(key)
102            .ok_or_else(|| anyhow!("{} does not set {key}", self.path.display()))
103    }
104
105    pub fn path(&self) -> &Path {
106        &self.path
107    }
108
109    /// The local port slot a's tunnel uses; slot b's is the next one.
110    pub fn tunnel_port(&self) -> Result<u16> {
111        let Some(value) = self.get("TUNNEL_PORT") else {
112            return Ok(DEFAULT_TUNNEL_PORT);
113        };
114        value
115            .parse::<u16>()
116            .ok()
117            .filter(|port| port.checked_add(1).is_some())
118            .ok_or_else(|| anyhow!("TUNNEL_PORT is not a port: {value}"))
119    }
120
121    pub fn proxy_url(&self) -> Result<ProxyUrl> {
122        let url = self.require("GUARDIAN_PROXY_URL")?;
123        let host = url.strip_prefix("https://").filter(|host| {
124            !host.is_empty()
125                && host
126                    .chars()
127                    .all(|c| c.is_ascii_alphanumeric() || c == '.' || c == '-')
128        });
129        let Some(host) = host else {
130            bail!("GUARDIAN_PROXY_URL must be https:// and a host name, with no path: {url}");
131        };
132        Ok(ProxyUrl {
133            url: url.to_owned(),
134            authority: format!("{host}:443"),
135        })
136    }
137}

139fn parse(text: &str, home: Option<&str>) -> Result<BTreeMap<String, String>> {
140    let mut values = BTreeMap::new();
141    for (index, line) in text.lines().enumerate() {
142        let number = index + 1;
143        let line = line.trim();
144        if line.is_empty() || line.starts_with('#') {
145            continue;
146        }
147        let Some((key, raw)) = line.split_once('=') else {
148            bail!("line {number} is not KEY=VALUE");
149        };
150        if key.starts_with("export ") {
151            bail!("line {number} starts with export: write KEY=VALUE");
152        }
153        if !KEYS.contains(&key) {
154            bail!("line {number} sets {key}, which no step reads; set it in your shell instead");
155        }
156        let value = parse_value(raw, home).with_context(|| format!("line {number} sets {key}"))?;
157        if values.insert(key.to_owned(), value).is_some() {
158            bail!("line {number} sets {key} a second time");
159        }
160    }
161    Ok(values)
162}

164/// Reads a value as bash would: double-quoted, single-quoted or bare, then an optional comment. Only `$HOME` is
165/// expanded; anything else bash would act on is refused rather than read differently.
166fn parse_value(raw: &str, home: Option<&str>) -> Result<String> {
167    let (value, rest) = match raw.chars().next() {
168        Some(quote @ ('"' | '\'')) => {
169            let end = raw[1..]
170                .find(quote)
171                .ok_or_else(|| anyhow!("a value with no closing {quote}"))?;
172            let inner = &raw[1..1 + end];
173            let value = if quote == '"' {
174                expand(inner, home)?
175            } else {
176                inner.to_owned()
177            };
178            (value, &raw[2 + end..])
179        }
180        _ => {
181            let end = raw.find(char::is_whitespace).unwrap_or(raw.len());
182            let word = &raw[..end];
183            if word.contains(['"', '\'', ';', '&', '|', '<', '>', '(', ')']) {
184                bail!("a value with quotes or shell operators inside it");
185            }
186            if word.starts_with('~') || word.contains(":~") {
187                bail!("a ~, which bash would expand: write $HOME instead");
188            }
189            (expand(word, home)?, &raw[end..])
190        }
191    };
192    let comment = rest.trim_start();
193    if !comment.is_empty() && (comment.len() == rest.len() || !comment.starts_with('#')) {
194        bail!("text after its value");
195    }
196    Ok(value)
197}
198
199fn expand(text: &str, home: Option<&str>) -> Result<String> {
200    if text.contains(['\\', '`']) {
201        bail!("a backslash or backtick");
202    }
203    let mut out = String::new();
204    let mut rest = text;
205    while let Some(start) = rest.find('$') {
206        out.push_str(&rest[..start]);
207        let after = &rest[start + 1..];
208        let length = if after.starts_with("{HOME}") {
209            "{HOME}".len()
210        } else if after.starts_with("HOME")
211            && !after[4..].starts_with(|c: char| c.is_ascii_alphanumeric() || c == '_')
212        {
213            "HOME".len()
214        } else {
215            bail!("a $ other than $HOME, which is the only variable this file can use");
216        };
217        out.push_str(home.ok_or_else(|| anyhow!("$HOME, but HOME is not set"))?);
218        rest = &after[length..];
219    }
220    out.push_str(rest);
221    Ok(out)
222}

243    /// The sample documents every key, set or commented out, so an operator learns of each from it.
244    #[test]
245    fn the_sample_names_every_key() {
246        let named: Vec<&str> = SAMPLE
247            .lines()
248            .map(|line| line.trim_start_matches("# "))
249            .filter_map(|line| line.split_once('=').map(|(key, _)| key))
250            .filter(|key| {
251                !key.is_empty() && key.bytes().all(|b| b.is_ascii_uppercase() || b == b'_')
252            })
253            .collect();
254        let mut sorted = named.clone();
255        sorted.sort_unstable();
256        sorted.dedup();
257        assert_eq!(sorted.len(), named.len(), "the sample names a key twice");
258        let mut keys = KEYS.to_vec();
259        keys.sort_unstable();
260        assert_eq!(sorted, keys);
261    }
