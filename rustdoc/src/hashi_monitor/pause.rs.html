16/// Explicitly pause Hashi using the configured governance identities.
17///
18/// Nothing calls this on monitor startup. Success means the on-chain paused flag
19/// was observed, not that nodes acknowledged it or Bitcoin broadcasts stopped.
20/// Errors after submission can mean a transaction landed: inspect the chain
21/// before retrying. This function neither resumes proposals nor unpauses Hashi.
22pub async fn initiate_pause(config: &Config) -> anyhow::Result<()> {
23    let pause = config
24        .pause
25        .as_ref()
26        .context("pause configuration is required to initiate a pause")?;
27    // Check on-chain state before loading any signing credentials.
28    let read_config = CliConfig {
29        sui_rpc_url: pause.transaction_rpc_url.clone(),
30        package_id: Some(config.sui.package_id()),
31        hashi_object_id: Some(pause.object_id()),
32        ..CliConfig::default()
33    };
34    let reader = HashiClient::new(&read_config).await?;
35    if reader.onchain_state().state().hashi().config.paused() {
36        return Ok(());
37    }
38    // The first configured operator creates the proposal and automatically votes.
39    let (first_path, remaining_paths) = pause
40        .operator_key_paths
41        .split_first()
42        .context("at least one pause operator key must be configured")?;
43    let first_config = CliConfig {
44        keypair_path: Some(first_path.clone()),
45        ..read_config.clone()
46    };
47    let first = HashiClient::new(&first_config).await?;
48
49    let options = TxOptions {
50        gas_budget: None,
51        skip_confirm: true,
52        dry_run: false,
53        serialize_unsigned: false,
54        sender: None,
55        gas_object: None,
56        gas_price: None,
57    };
58    let create = first.build_create_proposal_transaction(CreateProposalParams::EmergencyPause {
59        pause: true,
60        metadata: vec![],
61    })?;
62    let TxOutcome::Executed(response) = first
63        .finalize_tx(create, &options)
64        .await
65        .context("creating pause proposal; submission may have landed")?
66    else {
67        anyhow::bail!("pause proposal was not executed");
68    };
69    let proposal_id = extract_proposal_id_from_response(&response)
70        .context("pause proposal creation succeeded but its ID could not be read")?;
71    let proposal_context = || {
72        format!(
73            "pause proposal {proposal_id} did not complete confirmation; submitted transactions may have landed"
74        )
75    };
76
77    // Collect approvals from the remaining configured operators.
78    for path in remaining_paths {
79        let voter = HashiClient::new(&CliConfig {
80            keypair_path: Some(path.clone()),
81            ..read_config.clone()
82        })
83        .await
84        .with_context(proposal_context)?;
85        let vote = voter
86            .build_vote_transaction(
87                proposal_id,
88                voter
89                    .proposal_type_arg(&ProposalType::EmergencyPause)
90                    .with_context(proposal_context)?,
91            )
92            .with_context(proposal_context)?;
93        voter
94            .finalize_tx(vote, &options)
95            .await
96            .with_context(proposal_context)?;
97    }
98
99    // Execute the approved proposal using the proposer's credentials.
100    let executor = HashiClient::new(&first_config)
101        .await
102        .with_context(proposal_context)?;
103    let execute = executor
104        .build_execute_proposal_transaction(proposal_id, &ProposalType::EmergencyPause)
105        .with_context(proposal_context)?;
106    executor
107        .finalize_tx(execute, &options)
108        .await
109        .with_context(proposal_context)?;
110
111    // Read fresh state so success means the paused flag was actually observed.
112    let confirmed = HashiClient::new(&read_config)
113        .await
114        .with_context(proposal_context)?;
115    ensure!(
116        confirmed.onchain_state().state().hashi().config.paused(),
117        "on-chain paused flag was not observed after executing pause proposal {proposal_id}"
118    );
119    Ok(())
120}

122#[cfg(test)]
123mod tests {
124    use super::*;
125    use e2e_tests::TestNetworksBuilder;
126    use hashi_types::move_types::ConfigValue;
127
128    use crate::config::PauseConfig;
129
130    #[tokio::test]
131    async fn pause_requires_configuration() {
132        let error = initiate_pause(&Config::for_tests()).await.unwrap_err();
133        assert_eq!(
134            error.to_string(),
135            "pause configuration is required to initiate a pause"
136        );
137    }
138
139    #[tokio::test]
140    async fn pause_with_three_operators() -> anyhow::Result<()> {
141        assert_pause_with_operators(3, 4000).await
142    }
143
144    #[tokio::test]
145    async fn pause_with_one_operator() -> anyhow::Result<()> {
146        assert_pause_with_operators(1, 2000).await
147    }
148
149    async fn assert_pause_with_operators(count: usize, threshold: u64) -> anyhow::Result<()> {
150        let networks = TestNetworksBuilder::new()
151            .with_nodes(4)
152            .with_onchain_config(
153                "governance_emergency_pause_threshold_bps",
154                ConfigValue::U64(threshold),
155            )
156            .build()
157            .await?;
158        let ids = networks.hashi_network().ids();
159        let keys = tempfile::tempdir()?;
160        let paths: Vec<_> = (0..count)
161            .map(|index| keys.path().join(format!("operator-{index}.pem")))
162            .collect();
163        for (node, path) in networks.hashi_network().nodes().iter().zip(&paths) {
164            std::fs::write(path, node.config().operator_private_key.as_ref().unwrap())?;
165        }
166        let mut config = Config::for_tests();
167        // Keep the default unreachable audit RPC to prove pause uses only the fullnode.
168        config.sui.package_id = Some(ids.package_id);
169        config.pause = Some(PauseConfig {
170            transaction_rpc_url: networks.sui_network().rpc_url.clone(),
171            operator_key_paths: paths,
172            object_id: Some(ids.hashi_object_id),
173        });
174        let read_config = CliConfig {
175            sui_rpc_url: networks.sui_network().rpc_url.clone(),
176            package_id: Some(ids.package_id),
177            hashi_object_id: Some(ids.hashi_object_id),
178            ..CliConfig::default()
179        };
180        let before = HashiClient::new(&read_config).await?;
181        assert!(!before.onchain_state().state().hashi().config.paused());
182        initiate_pause(&config).await?;
183        let paused = HashiClient::new(&read_config).await?;
184        assert!(paused.onchain_state().state().hashi().config.paused());
185        let executed = paused.fetch_executed_proposals();
186        let proposal = executed
187            .iter()
188            .find(|proposal| proposal.proposal_type == ProposalType::EmergencyPause)
189            .context("executed pause proposal not found")?;
190        let details = paused.fetch_proposal_details(proposal.id).await?;
191        assert_eq!(details.votes.len(), count);
192
193        // An already-paused bridge needs no signing credentials.
194        keys.close()?;
195        config.pause.as_mut().unwrap().operator_key_paths.clear();
196        initiate_pause(&config).await?;
197        Ok(())
198    }
