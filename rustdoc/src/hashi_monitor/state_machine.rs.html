23/// A record of all the events tracking a single withdrawal.
24///
25/// `add_event` validates and stores an event. A structurally valid late event
26/// is retained even though the method returns its timing finding.
27///
28/// `violations(cursors)` checks if there are any violations given current cursors
29///
30/// Invariant: `expected_events` should not contain an event type that exists in `seen_events`.
31///
32/// Four current scope/progress combinations are possible:
33/// - Window: Out, In
34/// - Progress: Expecting events, Complete
35///
36/// `is_expecting_events()` distinguishes the two progress states. Findings are
37/// emitted to the caller and are not retained in the WSM, so `Complete` does not
38/// mean that no earlier finding was reported.
39pub struct WithdrawalStateMachine {
40    /// the set of non-zero events we have seen until now related to this withdrawal.
41    seen_events: Vec<MonitorWithdrawalEvent>,
42    /// Each `(event, deadline, relation)` entry means `event` is expected by
43    /// `deadline`; `relation` records whether it precedes or follows an event
44    /// already observed for this withdrawal.
45    expected_events: Vec<(WithdrawalEventType, UnixSeconds, EventRelation)>,
46    /// last time at which we checked for a btc withdrawal tx
47    btc_checked_at: Option<UnixSeconds>,
48    /// immutable wid
49    wid: WithdrawalID,
50    /// immutable txid
51    btc_txid: Txid,
52}
53
54pub enum BtcFetchOutcome {
55    NotExpected,
56    Unconfirmed,
57    Confirmed(Vec<MonitorFinding>),
58}
59
60impl WithdrawalStateMachine {
61    /// Note: Initialization ensures that the state machine has at least one event.
62    pub fn new(event: MonitorWithdrawalEvent, cfg: &Config) -> Self {
63        let mut sm = Self {
64            seen_events: Vec::new(),
65            expected_events: Vec::new(),
66            btc_checked_at: None,
67            wid: event.wid,
68            btc_txid: event.btc_txid,
69        };
70        assert!(
71            sm.add_event(event, cfg).is_empty(),
72            "first event cannot produce a finding"
73        );
74        sm
75    }
76
77    pub fn get(&self, event_type: WithdrawalEventType) -> Option<&MonitorWithdrawalEvent> {
78        self.seen_events
79            .iter()
80            .find(|event| event.event_type == event_type)
81    }
82
83    pub fn btc_txid(&self) -> Txid {
84        self.btc_txid
85    }
86
87    pub fn wid(&self) -> WithdrawalID {
88        self.wid
89    }
90
91    pub fn expects(&self, event_type: WithdrawalEventType) -> bool {
92        self.expected_events
93            .iter()
94            .any(|(event, _, _)| *event == event_type)
95    }
96
97    /// Are any expected neighboring events still outstanding?
98    ///
99    /// This does not mean that no timing finding was emitted during ingestion.
100    /// Callers must ensure `is_in_audit_window()` is true before using this for
101    /// garbage collection. Out-of-window withdrawals with pending expectations
102    /// may remain in memory, but they arise only from the bounded lookback or
103    /// lookahead ranges.
104    pub fn is_expecting_events(&self) -> bool {
105        !self.expected_events.is_empty()
106    }
107
108    // TODO: If we fully move to strict guardian-led audits, this can be relaxed to only include
109    // withdrawals with guardian E2 in the user window.
110    pub fn is_in_audit_window(&self, window: &impl AuditWindow) -> bool {
111        self.seen_events
112            .iter()
113            .any(|e| window.in_window(e.timestamp_secs))
114    }
115
116    /// Add an event and update expectations for its immediate neighbors.
117    ///
118    /// A missing predecessor is expected by the event timestamp plus
119    /// `clock_skew`; a missing successor is expected by the configured
120    /// next-event deadline. Events may be ingested in any order.
121    ///
122    /// Event retention is based on structural validity, not finding category.
123    /// `MonitorFinding::InvalidEventAdded` denotes a contradictory, definite
124    /// safety issue, so it is returned immediately without storing the event.
125    /// A structurally valid event is stored and updates subsequent expectations
126    /// even when its timing produces a safety or liveness finding.
127    ///
128    /// The return value contains every resulting finding. An empty vector means
129    /// the event was accepted without one.
130    ///
131    /// TODO: Represent a missing neighbor with its full allowed timestamp
132    /// interval, and apply both bounds regardless of event ingestion order. For
133    /// consecutive events `E_i` and `E_{i+1}`, observing `E_i` should constrain
134    /// `E_{i+1}` to `[E_i - clock_skew, E_i + next_event_delay]`; observing
135    /// `E_{i+1}` first should impose the equivalent inverse interval on `E_i`.
136    /// The current single-deadline representation checks only one side of that
137    /// interval, so timing findings can depend on which event is ingested first.
138    /// Before applying this uniformly to E2/E3, account for a Bitcoin block
139    /// header's timestamp being an approximate miner-provided time rather than
140    /// the wall-clock time at which the transaction was confirmed.
141    pub fn add_event(
142        &mut self,
143        new_event: MonitorWithdrawalEvent,
144        cfg: &Config,
145    ) -> Vec<MonitorFinding> {
146        if let Some(existing_event) = self.get(new_event.event_type) {
147            return if *existing_event == new_event {
148                Vec::new()
149            } else {
150                vec![MonitorFinding::InvalidEventAdded(
151                    "duplicate event for same wid with different contents".to_string(),
152                )]
153            };
154        }
155
156        if self.wid != new_event.wid {
157            return vec![MonitorFinding::InvalidEventAdded("invalid wid".to_string())];
158        }
159
160        if self.btc_txid != new_event.btc_txid {
161            return vec![MonitorFinding::InvalidEventAdded(
162                "invalid btc_txid".to_string(),
163            )];
164        }
165
166        // If a neighbor is already expected, record a timing finding but still
167        // ingest the event. The monitor must retain a late event so it can
168        // distinguish liveness from safety and continue validating the flow.
169        let mut timing_findings = Vec::new();
170        for (src, deadline, relation) in self.expected_events.iter() {
171            if *src == new_event.event_type && *deadline < new_event.timestamp_secs {
172                timing_findings.push(MonitorFinding::EventOccurredAfterDeadline {
173                    event: MonitorEvent::Withdrawal(new_event.clone()),
174                    relation: *relation,
175                    deadline: *deadline,
176                    occurred_at: new_event.timestamp_secs,
177                });
178            }
179        }
180
181        // if neighbor is not there, then we add an expectation indicating when we expect to see it.
182        if let Some(predecessor_event_type) = new_event.event_type.predecessor()
183            && self.get(predecessor_event_type).is_none()
184        {
185            let predecessor_deadline = new_event.timestamp_secs + cfg.clock_skew;
186            self.expected_events.push((
187                predecessor_event_type,
188                predecessor_deadline,
189                EventRelation::Predecessor,
190            ));
191        }
192        if let Some(successor_event_type) = new_event.event_type.successor()
193            && self.get(successor_event_type).is_none()
194        {
195            let successor_deadline = new_event.timestamp_secs
196                + cfg
197                    .next_event_delay(new_event.event_type)
198                    .expect("has a successor");
199            self.expected_events.push((
200                successor_event_type,
201                successor_deadline,
202                EventRelation::Successor,
203            ));
204        }
205
206        // remove any previously stored expected events
207        self.expected_events
208            .retain(|(src, _, _)| *src != new_event.event_type);
209        // add to seen events
210        self.seen_events.push(new_event);
211        timing_findings
212    }
213
214    /// If expecting BTC confirmation, query BTC RPC and add the event if confirmed.
215    ///     - Returns `Ok(BtcFetchOutcome::NotExpected)` if a BTC event is not expected.
216    ///     - Returns `Ok(BtcFetchOutcome::Unconfirmed)` if checked but block not yet mined.
217    ///     - Returns `Ok(BtcFetchOutcome::Confirmed(findings))` if confirmed; `findings` may be empty.
218    ///     - Returns `Err` for BTC RPC/infrastructure failures.
219    pub fn try_fetch_btc_tx(
220        &mut self,
221        cfg: &Config,
222        btc_rpc_client: &BtcRpcClient,
223    ) -> anyhow::Result<BtcFetchOutcome> {
224        if !self.expects(WithdrawalEventType::E3BtcConfirmed) {
225            return Ok(BtcFetchOutcome::NotExpected);
226        }
227        let btc_txid = self.btc_txid;
228        let wid = self.wid;
229        let cur_time = now_unix_seconds();
230
231        match btc_rpc_client.lookup_confirmation(btc_txid) {
232            Ok(Some(block_time)) => {
233                self.btc_checked_at = Some(cur_time);
234                let e_btc = MonitorWithdrawalEvent {
235                    event_type: WithdrawalEventType::E3BtcConfirmed,
236                    wid,
237                    btc_txid,
238                    timestamp_secs: block_time,
239                };
240                Ok(BtcFetchOutcome::Confirmed(self.add_event(e_btc, cfg)))
241            }
242            Ok(None) => {
243                self.btc_checked_at = Some(cur_time);
244                Ok(BtcFetchOutcome::Unconfirmed)
245            }
246            Err(e) => Err(e),
247        }
248    }
249
250    /// Check for violations given per-source cursors.
251    /// Only reports a missing event if its deadline has passed relative to the relevant cursor.
252    /// Callers must ensure is_in_audit_window() is true before calling this function.
253    pub fn violations(&self, cursors: &Cursors) -> Vec<MonitorFinding> {
254        let mut out = Vec::new();
255        for (event_type, deadline, relation) in &self.expected_events {
256            let cursor = match event_type {
257                WithdrawalEventType::E3BtcConfirmed => match self.btc_checked_at {
258                    Some(checked_at) => checked_at,
259                    None => {
260                        tracing::warn!(
261                            "callers should avoid reaching this branch by calling try_fetch_btc_tx before"
262                        );
263                        continue;
264                    }
265                },
266                _ => cursors.for_event_type(*event_type),
267            };
268            if *deadline <= cursor {
269                out.push(MonitorFinding::ExpectedEventMissing {
270                    event_type: MonitorEventType::Withdrawal(*event_type),
271                    relation: *relation,
272                    deadline: *deadline,
273                    cursor,
274                });
275            }
276        }
277        out
278    }
279}

281/// Deposit State Machine. Unlike withdrawal state machine, here we only listen for a sui event,
282/// which in turn triggers a lookup for a specific btc event. So we simplify the struct & its impl's.
283pub struct DepositStateMachine {
284    /// The hashi deposit event
285    hashi_deposit_event: MonitorDepositEvent,
286    /// None initially and Some post BTC event find
287    btc_event: Option<MonitorDepositEvent>,
288    btc_event_expected_at: UnixSeconds,
289    btc_checked_at: Option<UnixSeconds>,
290}
291
292impl DepositStateMachine {
293    pub fn new(event: MonitorDepositEvent, cfg: &Config) -> Self {
294        if event.event_type != DepositEventType::E2HashiDeposited {
295            panic!("unexpected event type");
296        }
297        // btc confirmation is a predecessor event => we set the deadline to now (+skew).
298        let t_btc_expected = event.timestamp_secs + cfg.clock_skew;
299        Self {
300            hashi_deposit_event: event,
301            btc_event: None,
302            btc_event_expected_at: t_btc_expected,
303            btc_checked_at: None,
304        }
305    }
306
307    pub fn btc_txid(&self) -> Txid {
308        self.hashi_deposit_event.btc_txid
309    }
310
311    pub fn hashi_deposit_event(&self) -> &MonitorDepositEvent {
312        &self.hashi_deposit_event
313    }
314
315    pub fn is_expecting_events(&self) -> bool {
316        self.btc_event.is_none()
317    }
318
319    pub fn try_fetch_btc_tx(
320        &mut self,
321        btc_rpc_client: &BtcRpcClient,
322    ) -> anyhow::Result<BtcFetchOutcome> {
323        if !self.is_expecting_events() {
324            return Ok(BtcFetchOutcome::NotExpected);
325        }
326
327        let deadline = self.btc_event_expected_at;
328        let btc_txid = self.hashi_deposit_event.btc_txid;
329        let cur_time = now_unix_seconds();
330
331        match btc_rpc_client.lookup_confirmation(btc_txid) {
332            Ok(Some(block_time)) => {
333                self.btc_checked_at = Some(cur_time);
334                let e_btc = MonitorDepositEvent {
335                    event_type: DepositEventType::E1BtcConfirmed,
336                    btc_txid,
337                    btc_vout: self.hashi_deposit_event.btc_vout,
338                    timestamp_secs: block_time,
339                };
340
341                let mut findings = Vec::new();
342                if deadline < block_time {
343                    findings.push(MonitorFinding::EventOccurredAfterDeadline {
344                        event: MonitorEvent::Deposit(e_btc.clone()),
345                        relation: EventRelation::Predecessor,
346                        deadline,
347                        occurred_at: block_time,
348                    });
349                }
350                self.btc_event = Some(e_btc);
351                Ok(BtcFetchOutcome::Confirmed(findings))
352            }
353            Ok(None) => {
354                self.btc_checked_at = Some(cur_time);
355                Ok(BtcFetchOutcome::Unconfirmed)
356            }
357            Err(e) => Err(e),
358        }
359    }
360
361    pub fn violations(&self) -> Vec<MonitorFinding> {
362        if self.btc_event.is_some() {
363            // btc event found => no violations!
364            return Vec::new();
365        };
366
367        // btc event not yet found
368        let Some(cursor) = self.btc_checked_at else {
369            tracing::warn!(
370                txid = %self.hashi_deposit_event.btc_txid,
371                "callers should avoid this branch by calling try_fetch_btc_tx before violations"
372            );
373            return Vec::new();
374        };
375
376        let deadline = self.btc_event_expected_at;
377        if deadline > cursor {
378            return Vec::new();
379        }
380
381        vec![MonitorFinding::ExpectedEventMissing {
382            event_type: MonitorEventType::Deposit(DepositEventType::E1BtcConfirmed),
383            relation: EventRelation::Predecessor,
384            deadline,
385            cursor,
386        }]
387    }
388}
