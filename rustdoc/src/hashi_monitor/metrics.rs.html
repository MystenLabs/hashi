4//! Prometheus metrics for continuous audits, served at `/metrics`.
5
6use std::sync::Arc;
7
8use hashi_types::guardian::time::UnixSeconds;
9use prometheus::Encoder;
10use prometheus::IntGauge;
11use prometheus::IntGaugeVec;
12use prometheus::Opts;
13use prometheus::Registry;
14use prometheus::TextEncoder;
15use tracing::info;
16
17use crate::findings::MonitorFinding;
18
19pub const SOURCE_SUI: &str = "sui";
20pub const SOURCE_GUARDIAN: &str = "guardian";
21pub const SOURCE_BTC: &str = "btc";
22/// The daily check of every UTXO in the UTXO pool against Bitcoin.
23pub const SOURCE_UTXO: &str = "utxo";
24/// The snapshot. Its time is the last successful save.
25pub const SOURCE_SNAPSHOT: &str = "snapshot";
26
27const FLOW_WITHDRAWAL: &str = "withdrawal";
28const FLOW_DEPOSIT: &str = "deposit";
29
30pub struct MonitorMetrics {
31    registry: Registry,
32    pool_utxo_count: IntGauge,
33    oldest_utxo_timestamp_seconds: IntGauge,
34    /// The sum of the UTXO amounts in Move's pool, in satoshis. See `UtxoPool`.
35    pool_value_sats: IntGauge,
36    /// Sui's hBTC supply, in satoshis. 0 until the first read.
37    hbtc_supply_sats: IntGauge,
38    /// The open flows, by kind: `withdrawal` or `deposit`. Set at each
39    /// removal of the closed machines.
40    open_flows: IntGaugeVec,
41    /// Unix time of the latest finding of each kind. A kind has no series
42    /// until its first finding.
43    last_finding_timestamp_seconds: IntGaugeVec,
44    /// Unix time each source has been checked through: the Sui and guardian
45    /// cursors, and for Bitcoin, which is looked up rather than scanned, the
46    /// start of the last pass over every pending confirmation, or the startup
47    /// time before the first pass. For the snapshot, the time of the last
48    /// successful save.
49    /// 0 until the auditor starts, so a monitor that never starts reads stale.
50    checked_through_timestamp_seconds: IntGaugeVec,
51}
52
53impl Default for MonitorMetrics {
54    fn default() -> Self {
55        Self::new()
56    }
57}
58
59impl MonitorMetrics {
60    pub fn new() -> Self {
61        let registry = Registry::new();
62        let last_finding_timestamp_seconds = IntGaugeVec::new(
63            Opts::new(
64                "hashi_monitor_last_finding_timestamp_seconds",
65                "Unix time of the latest monitor finding of each kind",
66            ),
67            &["kind"],
68        )
69        .expect("valid metric");
70        let checked_through_timestamp_seconds = IntGaugeVec::new(
71            Opts::new(
72                "hashi_monitor_checked_through_timestamp_seconds",
73                "Unix time through which each source has been checked; for btc, the start of the last pass that looked up every pending confirmation",
74            ),
75            &["source"],
76        )
77        .expect("valid metric");
78        let open_flows = IntGaugeVec::new(
79            Opts::new(
80                "hashi_monitor_open_flows",
81                "Number of withdrawal and deposit flows the monitor follows and has not closed",
82            ),
83            &["kind"],
84        )
85        .expect("valid metric");
86
87        let pool_utxo_count = IntGauge::new(
88            "hashi_monitor_pool_utxo_count",
89            "Number of UTXOs in Move's pool, by the monitor's account",
90        )
91        .expect("valid metric");
92        let oldest_utxo_timestamp_seconds = IntGauge::new(
93            "hashi_monitor_oldest_utxo_timestamp_seconds",
94            "Earliest block timestamp of a UTXO in Move's pool, in Unix seconds; 0 with no such UTXO",
95        )
96        .expect("valid metric");
97        let pool_value_sats = IntGauge::new(
98            "hashi_monitor_pool_value_sats",
99            "Sum of the UTXO amounts in Move's pool, by the monitor's account, in satoshis",
100        )
101        .expect("valid metric");
102        let hbtc_supply_sats = IntGauge::new(
103            "hashi_monitor_hbtc_supply_sats",
104            "Total supply of hBTC on Sui, in satoshis",
105        )
106        .expect("valid metric");
107        registry
108            .register(Box::new(pool_value_sats.clone()))
109            .expect("register");
110        registry
111            .register(Box::new(hbtc_supply_sats.clone()))
112            .expect("register");
113        registry
114            .register(Box::new(pool_utxo_count.clone()))
115            .expect("register");
116        registry
117            .register(Box::new(oldest_utxo_timestamp_seconds.clone()))
118            .expect("register");
119
120        registry
121            .register(Box::new(last_finding_timestamp_seconds.clone()))
122            .expect("register");
123        registry
124            .register(Box::new(checked_through_timestamp_seconds.clone()))
125            .expect("register");
126        registry
127            .register(Box::new(open_flows.clone()))
128            .expect("register");
129
130        for source in [
131            SOURCE_SUI,
132            SOURCE_GUARDIAN,
133            SOURCE_BTC,
134            SOURCE_UTXO,
135            SOURCE_SNAPSHOT,
136        ] {
137            checked_through_timestamp_seconds
138                .with_label_values(&[source])
139                .set(0);
140        }
141        for kind in [FLOW_WITHDRAWAL, FLOW_DEPOSIT] {
142            open_flows.with_label_values(&[kind]).set(0);
143        }
144
145        Self {
146            registry,
147            pool_utxo_count,
148            oldest_utxo_timestamp_seconds,
149            pool_value_sats,
150            hbtc_supply_sats,
151            open_flows,
152            last_finding_timestamp_seconds,
153            checked_through_timestamp_seconds,
154        }
155    }
156
157    pub fn record_findings<'a>(
158        &self,
159        findings: impl IntoIterator<Item = &'a MonitorFinding>,
160        now: UnixSeconds,
161    ) {
162        for finding in findings {
163            self.last_finding_timestamp_seconds
164                .with_label_values(&[finding.kind()])
165                .set(gauge_value(now));
166        }
167    }
168
169    pub fn set_hbtc_backing(&self, hbtc_supply: u64, pool_value: u64) {
170        self.hbtc_supply_sats
171            .set(i64::try_from(hbtc_supply).unwrap_or(i64::MAX));
172        self.pool_value_sats
173            .set(i64::try_from(pool_value).unwrap_or(i64::MAX));
174    }
175
176    pub fn set_checked_through(&self, source: &str, timestamp: UnixSeconds) {
177        self.checked_through_timestamp_seconds
178            .with_label_values(&[source])
179            .set(gauge_value(timestamp));
180    }
181
182    pub fn set_utxo_pool(&self, count: usize, oldest_timestamp: Option<UnixSeconds>) {
183        self.pool_utxo_count
184            .set(i64::try_from(count).unwrap_or(i64::MAX));
185        self.oldest_utxo_timestamp_seconds
186            .set(gauge_value(oldest_timestamp.unwrap_or_default()));
187    }
188
189    pub fn set_open_flows(&self, withdrawals: usize, deposits: usize) {
190        for (kind, count) in [(FLOW_WITHDRAWAL, withdrawals), (FLOW_DEPOSIT, deposits)] {
191            self.open_flows
192                .with_label_values(&[kind])
193                .set(i64::try_from(count).unwrap_or(i64::MAX));
194        }
195    }
196
197    pub(crate) fn render(&self) -> String {
198        let mut buf = Vec::new();
199        TextEncoder::new()
200            .encode(&self.registry.gather(), &mut buf)
201            .expect("encode metrics");
202        String::from_utf8(buf).expect("metrics are utf-8")
203    }
204
205    /// Serve `GET /metrics` on an already bound listener, forever.
206    pub async fn serve(self: Arc<Self>, listener: tokio::net::TcpListener) -> anyhow::Result<()> {
207        let app = axum::Router::new().route(
208            "/metrics",
209            axum::routing::get(move || {
210                let metrics = self.clone();
211                async move { metrics.render() }
212            }),
213        );
214        info!("Metrics listening on {}.", listener.local_addr()?);
215        axum::serve(listener, app).await?;
216        Ok(())
217    }
218}
219
220fn gauge_value(timestamp: UnixSeconds) -> i64 {
221    i64::try_from(timestamp).unwrap_or(i64::MAX)
222}

228    #[tokio::test]
229    async fn the_metrics_route_serves_the_registry() {
230        use tokio::io::AsyncReadExt as _;
231        use tokio::io::AsyncWriteExt as _;
232
233        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
234        let addr = listener.local_addr().unwrap();
235        let metrics = Arc::new(MonitorMetrics::new());
236        metrics.set_checked_through(SOURCE_SUI, 1_700_000_000);
237        tokio::spawn(metrics.serve(listener));
238
239        let mut stream = tokio::net::TcpStream::connect(addr).await.unwrap();
240        stream
241            .write_all(b"GET /metrics HTTP/1.1\r\nHost: localhost\r\nConnection: close\r\n\r\n")
242            .await
243            .unwrap();
244        let mut response = String::new();
245        stream.read_to_string(&mut response).await.unwrap();
246
247        assert!(
248            response.contains(
249                r#"hashi_monitor_checked_through_timestamp_seconds{source="sui"} 1700000000"#
250            ),
251            "{response}"
252        );
253    }
254
255    #[test]
256    fn every_series_is_exported_before_the_auditor_runs() {
257        let rendered = MonitorMetrics::new().render();
258
259        for line in [
260            r#"hashi_monitor_checked_through_timestamp_seconds{source="sui"} 0"#,
261            r#"hashi_monitor_checked_through_timestamp_seconds{source="guardian"} 0"#,
262            r#"hashi_monitor_checked_through_timestamp_seconds{source="btc"} 0"#,
263            r#"hashi_monitor_checked_through_timestamp_seconds{source="snapshot"} 0"#,
264            r#"hashi_monitor_open_flows{kind="withdrawal"} 0"#,
265            r#"hashi_monitor_open_flows{kind="deposit"} 0"#,
266        ] {
267            assert!(rendered.contains(line), "missing {line} in:\n{rendered}");
268        }
269        // A kind has no series until its first finding.
270        assert!(!rendered.contains("hashi_monitor_last_finding_timestamp_seconds{"));
271    }
272
273    #[test]
274    fn a_finding_sets_the_latest_timestamp_of_its_kind() {
275        let metrics = MonitorMetrics::new();
276        let finding = MonitorFinding::SourceStale {
277            source: SOURCE_GUARDIAN,
278            checked_through: 0,
279            limit_secs: 3_600,
280        };
281
282        metrics.record_findings(std::slice::from_ref(&finding), 100);
283        metrics.record_findings(&[], 150);
284        metrics.record_findings(&[finding], 200);
285        let stale = MonitorFinding::SnapshotStale {
286            last_saved: 0,
287            limit_secs: 3_600,
288        };
289        metrics.record_findings(&[stale], 250);
290        metrics.set_checked_through(SOURCE_GUARDIAN, 300);
291        metrics.set_open_flows(3, 5);
292
293        let rendered = metrics.render();
294        for line in [
295            r#"hashi_monitor_last_finding_timestamp_seconds{kind="SourceStale"} 200"#,
296            r#"hashi_monitor_last_finding_timestamp_seconds{kind="SnapshotStale"} 250"#,
297            r#"hashi_monitor_checked_through_timestamp_seconds{source="guardian"} 300"#,
298            r#"hashi_monitor_open_flows{kind="withdrawal"} 3"#,
299            r#"hashi_monitor_open_flows{kind="deposit"} 5"#,
300        ] {
301            assert!(rendered.contains(line), "missing {line} in:\n{rendered}");
302        }
303    }
