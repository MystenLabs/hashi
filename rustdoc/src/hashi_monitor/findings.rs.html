4//! The findings the monitor reports.
5//!
6//! Why it exists: every check in the pollers and the state machines reports
7//! its result as one value of `MonitorFinding`. The auditor logs each
8//! finding, and the metrics record the time of the last one.
9//!
10//! Assumptions:
11//! - Every finding needs an operator action. A condition that needs no
12//!   action is not a finding.
13//! - Every operator must act on an essential finding. It shows a loss of
14//!   Bitcoin backing, a broken protocol rule, or a check that cannot run.
15//!   An event that comes before the event it depends on breaks a rule.
16//! - Only the Hashi team can act on the other findings. They show a delay or
17//!   a fault that moves no Bitcoin. `AlertMode` decides if they alert.
18//! - `BtcRpcInconsistency` means a Bitcoin RPC answer contradicts one the
19//!   monitor already holds. The bridge cannot cause it. The inner enum names
20//!   the contradiction.
21//! - `GuardianInconsistency` means a guardian log that an honest guardian
22//!   cannot write. The guardian is the cause. The inner enum names the rule
23//!   it breaks.
24

48/// The findings that alert: an ERROR log line, and the finding metric. A
49/// finding that does not alert is a WARN log line.
50#[derive(Clone, Copy, Debug, Default, Deserialize, Eq, PartialEq)]
51#[serde(rename_all = "snake_case")]
52pub enum AlertMode {
53    /// Every finding alerts.
54    #[default]
55    Strict,
56    /// Only the essential findings alert. See `MonitorFinding::is_essential`.
57    Essential,
58}

75/// Why the monitor cannot trust a withdrawal's transaction or events.
76// Findings are rare, so the large variant does not cost much.
77#[allow(clippy::large_enum_variant)]
78#[derive(Clone, Debug, Eq, PartialEq)]
79pub enum WithdrawalInvalidReason {
80    /// A single event's txid differs from the txid of the transaction that
81    /// its own inputs and outputs build.
82    TxidMismatch {
83        event: MonitorWithdrawalEvent,
84        computed: Txid,
85    },
86    /// The inputs and outputs of a single event do not build a transaction.
87    Unbuildable {
88        event: MonitorWithdrawalEvent,
89        error: String,
90    },
91    /// A new event disagrees with an event the monitor holds.
92    EventsDiffer {
93        kind: EventDiff,
94        held: MonitorWithdrawalEvent,
95        new: MonitorWithdrawalEvent,
96    },
97    /// The withdrawal lists `outpoint` as an input, and the pending
98    /// withdrawal `other` lists it too. Move locks an input to one
99    /// withdrawal, so one of the two has no Hashi approval.
100    InputClaimed {
101        outpoint: OutPoint,
102        other: WithdrawalID,
103    },
104}

106/// What a new withdrawal event disagrees on. The monitor checks in this order.
107#[derive(Clone, Copy, Debug, Eq, PartialEq)]
108pub enum EventDiff {
109    /// The new event names another txid.
110    Txid,
111    /// The full event differs from a held event of the same type.
112    Full,
113    /// The events name the same txid but disagree on what it does not
114    /// cover, such as which outputs are change.
115    Tx,
116}

118/// Why two events of one deposit request disagree.
119#[derive(Clone, Debug, Eq, PartialEq)]
120pub enum DepositMismatchReason {
121    /// E2 and E3: the Bitcoin payment and the approval differ in amount or
122    /// script.
123    PaymentDiffers {
124        sui: SuiDepositApprovedEvent,
125        btc: BtcConfirmedDeposit,
126    },
127    /// E1 and E3: the approval differs from its request. Move copies the
128    /// request, so Move has a fault or the Sui answer is wrong.
129    ApprovalDiffers {
130        request: SuiDepositRequestedEvent,
131        approval: SuiDepositApprovedEvent,
132    },
133    /// E1 and E1: a request came again with other contents.
134    RequestDiffers {
135        held: SuiDepositRequestedEvent,
136        new: SuiDepositRequestedEvent,
137    },
138}

140/// Findings emitted by the monitor.
141#[derive(Clone, Debug, Eq, PartialEq)]
142pub enum MonitorFinding {
143    // ========================================================================
144    // Withdrawals
145    // E1: `WithdrawalPickedForProcessing`, the Hashi approval on Sui.
146    // E2: `WithdrawalLogMessage`, the guardian's log of its signature.
147    // E3: `BtcConfirmation`, the Bitcoin confirmation.
148    // ========================================================================
149    /// The monitor saw the successor of `event_type`, but `event_type` is
150    /// missing or more than its skew late. The monitor looks up E3 from E1,
151    /// so a Bitcoin spend with no guardian log gives a missing E2.
152    WithdrawalUnapproved {
153        wid: WithdrawalID,
154        event_type: WithdrawalEventType,
155        deadline: UnixSeconds,
156        timing: EventTiming,
157    },
158    /// The monitor saw the predecessor of `event_type`, but `event_type` is
159    /// missing or more than its delay late.
160    WithdrawalStuck {
161        wid: WithdrawalID,
162        event_type: WithdrawalEventType,
163        deadline: UnixSeconds,
164        timing: EventTiming,
165    },
166    /// The withdrawal's transaction or events contradict each other. See
167    /// `WithdrawalInvalidReason`.
168    WithdrawalInvalid {
169        wid: WithdrawalID,
170        reason: WithdrawalInvalidReason,
171    },
172
173    // ========================================================================
174    // Deposits
175    // E1: `SuiDepositRequestedEvent`, the request on Sui.
176    // E2: `BtcConfirmedDeposit`, the Bitcoin payment.
177    // E3: `SuiDepositApprovedEvent`, the Hashi approval on Sui.
178    // ========================================================================
179    /// E3 without E2: Hashi approved a deposit, but its outpoint holds no
180    /// unspent Bitcoin output.
181    DepositUnbacked { sui: SuiDepositApprovedEvent },
182    /// Two events of one deposit request disagree. See
183    /// `DepositMismatchReason`.
184    DepositMismatch {
185        request_id: Address,
186        reason: DepositMismatchReason,
187    },
188    /// E3 and E2 found, but the E2 block is later than E3 plus the skew.
189    ///
190    /// Hashi approved a deposit whose block is later than `deadline`, the
191    /// approval plus the skew. The skew covers a block that a miner dates
192    /// ahead, so Hashi approved before Bitcoin confirmed the payment.
193    DepositLate {
194        sui: SuiDepositApprovedEvent,
195        deadline: UnixSeconds,
196        block_time: UnixSeconds,
197    },
198    /// At least `MIN_STALLED_DEPOSITS` deposits that look valid are unapproved
199    /// `deposit_approval_overdue` after their latest request and E2, and no
200    /// deposit was approved since. A sanction also leaves one such deposit
201    /// unapproved, so a single one is not a finding.
202    DepositApprovalStalled {
203        outpoints: Vec<OutPoint>,
204        last_approval: Option<UnixSeconds>,
205    },
206
207    // ========================================================================
208    // UTXO pool and hBTC supply
209    // ========================================================================
210    /// A UTXO in the monitor's UTXO pool was spent on Bitcoin, and no Hashi
211    /// withdrawal confirmed on Bitcoin spent it.
212    UtxoUnexpectedSpend { outpoint: OutPoint, amount: u64 },
213    /// A confirmed Hashi withdrawal spent a UTXO that is not in Move's pool,
214    /// by the monitor's account. `held` is `None` if the pool never held it.
215    /// The monitor missed an event, or Move has a fault.
216    UtxoOutsideMovePoolSpent {
217        outpoint: OutPoint,
218        txid: Txid,
219        held: Option<UtxoOrigin>,
220    },
221    /// A deposit and a withdrawal's change claim one outpoint. Move refuses a
222    /// request for held change, so an insider or a Move fault is the cause.
223    UtxoOriginConflict {
224        outpoint: OutPoint,
225        held: UtxoOrigin,
226        new: UtxoOrigin,
227    },
228    /// A UTXO in Move's pool has sat unspent on Bitcoin past the warn age.
229    /// The committee can spend a UTXO alone 60 days after it confirms.
230    UtxoTooOld {
231        outpoint: OutPoint,
232        age_secs: u64,
233        threshold_secs: u64,
234    },
235    /// Sui's hBTC supply has exceeded the value of the UTXO pool since
236    /// `since`, for longer than the grace. Bitcoin does not back every hBTC.
237    HbtcSupplyExceedsPool {
238        hbtc_supply: u64,
239        pool_value: u64,
240        since: UnixSeconds,
241    },
242
243    // ========================================================================
244    // Sources and the monitor
245    // Bitcoin answers, guardian logs, and the monitor's own progress.
246    // ========================================================================
247    /// A Bitcoin RPC answer contradicts one the monitor already holds. See
248    /// `BtcRpcInconsistency`.
249    BtcRpcInconsistency(BtcRpcInconsistency),
250    /// The guardian's logs break a rule that an honest guardian keeps. See
251    /// `GuardianInconsistency`.
252    GuardianInconsistency(GuardianInconsistency),
253    /// The time the guardian signed into a log differs from the time S3
254    /// recorded the object by more than `MAX_EXPECTED_LOG_CLOCK_SKEW`.
255    GuardianClockSkew {
256        session_id: SessionID,
257        object_key: String,
258        /// The guardian's signed time.
259        logged_at_ms: UnixMillis,
260        /// S3's `LastModified` time.
261        stored_at_ms: UnixMillis,
262    },
263    /// A source was last checked longer ago than its limit. Its events go
264    /// unchecked until it catches up.
265    SourceStale {
266        source: &'static str,
267        checked_through: UnixSeconds,
268        limit_secs: u64,
269    },
270    /// No snapshot save has succeeded for longer than the limit. A restart
271    /// repeats the work done since the last save.
272    SnapshotStale {
273        last_saved: UnixSeconds,
274        limit_secs: u64,
275    },
276}

278impl MonitorFinding {
279    /// The finding for a withdrawal event that broke its deadline. A late
280    /// or missing `Predecessor` is `WithdrawalUnapproved`, else
281    /// `WithdrawalStuck`.
282    pub fn withdrawal_timing(
283        wid: WithdrawalID,
284        event_type: WithdrawalEventType,
285        relation: EventRelation,
286        deadline: UnixSeconds,
287        timing: EventTiming,
288    ) -> Self {
289        match relation {
290            EventRelation::Predecessor => Self::WithdrawalUnapproved {
291                wid,
292                event_type,
293                deadline,
294                timing,
295            },
296            EventRelation::Successor => Self::WithdrawalStuck {
297                wid,
298                event_type,
299                deadline,
300                timing,
301            },
302        }
303    }
304
305    /// Whether every operator must act on the finding. See the module doc.
306    pub fn is_essential(&self) -> bool {
307        match self {
308            // No hBTC was minted. Users wait for it.
309            Self::DepositApprovalStalled { .. } => false,
310            // A bad guardian clock or a slow S3 write. The checks still run.
311            Self::GuardianClockSkew { .. } => false,
312            // A user waits for Bitcoin. A spend with no guardian log is a
313            // `WithdrawalUnapproved`, and a theft of an input is a `UtxoUnexpectedSpend`.
314            Self::WithdrawalStuck { .. } => false,
315            Self::WithdrawalUnapproved { .. }
316            | Self::WithdrawalInvalid { .. }
317            | Self::DepositUnbacked { .. }
318            | Self::DepositMismatch { .. }
319            | Self::DepositLate { .. }
320            | Self::BtcRpcInconsistency(_)
321            | Self::UtxoTooOld { .. }
322            | Self::UtxoUnexpectedSpend { .. }
323            | Self::UtxoOutsideMovePoolSpent { .. }
324            | Self::UtxoOriginConflict { .. }
325            | Self::HbtcSupplyExceedsPool { .. }
326            | Self::SourceStale { .. }
327            | Self::SnapshotStale { .. }
328            | Self::GuardianInconsistency(_) => true,
329        }
330    }
331
332    /// Whether the finding alerts in `mode`.
333    pub fn alerts_in(&self, mode: AlertMode) -> bool {
334        mode == AlertMode::Strict || self.is_essential()
335    }
336
337    /// The variant name. Alerting labels each finding with it.
338    pub fn kind(&self) -> &'static str {
339        match self {
340            Self::WithdrawalUnapproved { .. } => "WithdrawalUnapproved",
341            Self::WithdrawalStuck { .. } => "WithdrawalStuck",
342            Self::WithdrawalInvalid { .. } => "WithdrawalInvalid",
343            Self::DepositUnbacked { .. } => "DepositUnbacked",
344            Self::DepositLate { .. } => "DepositLate",
345            Self::DepositMismatch { .. } => "DepositMismatch",
346            Self::DepositApprovalStalled { .. } => "DepositApprovalStalled",
347            Self::BtcRpcInconsistency(_) => "BtcRpcInconsistency",
348            Self::UtxoTooOld { .. } => "UtxoTooOld",
349            Self::UtxoUnexpectedSpend { .. } => "UtxoUnexpectedSpend",
350            Self::UtxoOutsideMovePoolSpent { .. } => "UtxoOutsideMovePoolSpent",
351            Self::UtxoOriginConflict { .. } => "UtxoOriginConflict",
352            Self::HbtcSupplyExceedsPool { .. } => "HbtcSupplyExceedsPool",
353            Self::SourceStale { .. } => "SourceStale",
354            Self::SnapshotStale { .. } => "SnapshotStale",
355            Self::GuardianClockSkew { .. } => "GuardianClockSkew",
356            Self::GuardianInconsistency(_) => "GuardianInconsistency",
357        }
358    }
359}

361/// A Bitcoin RPC answer that contradicts one the monitor already holds. A
362/// chain cannot produce it, so it is a lying node, a reorg, or a monitor
363/// bug. The bridge is not the cause, and an engineer has to look.
364#[derive(Clone, Debug, Eq, PartialEq)]
365pub enum BtcRpcInconsistency {
366    /// Bitcoin confirmed an outpoint the monitor's UTXO pool knows with a
367    /// second amount or script. An output has one of each.
368    UtxoMismatch {
369        outpoint: OutPoint,
370        held_amount: u64,
371        seen_amount: u64,
372        held_script: ScriptBuf,
373        seen_script: ScriptBuf,
374    },
375    /// A UTXO the monitor's UTXO pool holds does not exist on Bitcoin at the
376    /// daily pool check. Bitcoin confirmed it once.
377    UtxoVanished {
378        outpoint: OutPoint,
379        first_seen: UnixSeconds,
380    },
381}

383/// A guardian log that an honest guardian cannot write. The guardian is
384/// the cause, and an engineer has to look.
385#[derive(Clone, Debug, Eq, PartialEq)]
386pub enum GuardianInconsistency {
387    /// Two withdrawal logs have the same seq. The guardian's limiter takes
388    /// each seq once, so two guardian sessions signed from one limiter state.
389    SeqDuplicate { seq: u64, object_key: String },
390    /// No withdrawal log has `seq` by `expect_by`, but a later seq has one.
391    /// The guardian signed a withdrawal without a log, or a log was lost.
392    SeqGap { seq: u64, expect_by: UnixSeconds },
393    /// A withdrawal log skips more than a few seqs: its seq is far above
394    /// `next_seq`, the seq that the monitor expects next.
395    SeqJump {
396        next_seq: u64,
397        seq: u64,
398        object_key: String,
399    },
400    /// The monitor cannot read an output address in the withdrawal log at
401    /// `object_key`.
402    WithdrawalLogUnreadable {
403        object_key: String,
404        wid: WithdrawalID,
405        error: String,
406    },
407}

409impl fmt::Display for GuardianInconsistency {
410    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
411        match self {
412            Self::SeqDuplicate { seq, object_key } => {
413                write!(f, "SeqDuplicate(seq={seq}, key={object_key})")
414            }
415            Self::SeqGap { seq, expect_by } => write!(
416                f,
417                "SeqGap(seq={seq}, expect_by={})",
418                utc_timestamp(*expect_by)
419            ),
420            Self::SeqJump {
421                next_seq,
422                seq,
423                object_key,
424            } => write!(
425                f,
426                "SeqJump(next_seq={next_seq}, seq={seq}, key={object_key})"
427            ),
428            Self::WithdrawalLogUnreadable {
429                object_key,
430                wid,
431                error,
432            } => write!(
433                f,
434                "WithdrawalLogUnreadable(key={object_key}, wid={wid}, error={error})"
435            ),
436        }
437    }
438}

465impl EventTiming {
466    /// The timing for a log line, against the `deadline` it broke.
467    fn describe(&self, deadline: UnixSeconds) -> String {
468        match self {
469            Self::Late { event } => format!(
470                "late_by={}, event={}",
471                human_duration(event.timestamp_secs.saturating_sub(deadline)),
472                MonitorEvent::Withdrawal(event.clone()),
473            ),
474            Self::Missing { cursor } => {
475                format!("missing, checked_through={}", utc_timestamp(*cursor))
476            }
477        }
478    }
479}

481impl fmt::Display for WithdrawalInvalidReason {
482    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
483        match self {
484            Self::TxidMismatch { event, computed } => write!(
485                f,
486                "TxidMismatch(event={}, event_tx={{{}}}, computed={computed})",
487                MonitorEvent::Withdrawal(event.clone()),
488                event.tx,
489            ),
490            Self::Unbuildable { event, error } => write!(
491                f,
492                "Unbuildable(event={}, event_tx={{{}}}, error={error})",
493                MonitorEvent::Withdrawal(event.clone()),
494                event.tx,
495            ),
496            Self::EventsDiffer { kind, held, new } => write!(
497                f,
498                "EventsDiffer(kind={kind:?}, held={}, held_tx={{{}}}, new={}, new_tx={{{}}})",
499                MonitorEvent::Withdrawal(held.clone()),
500                held.tx,
501                MonitorEvent::Withdrawal(new.clone()),
502                new.tx,
503            ),
504            Self::InputClaimed { outpoint, other } => {
505                write!(f, "InputClaimed(outpoint={outpoint}, other_wid={other})")
506            }
507        }
508    }

535impl fmt::Display for MonitorFinding {
536    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
537        match self {
538            Self::WithdrawalUnapproved {
539                wid,
540                event_type,
541                deadline,
542                timing,
543            } => write!(
544                f,
545                "WithdrawalUnapproved(wid={wid}, event_type={event_type:?}, deadline={}, {})",
546                utc_timestamp(*deadline),
547                timing.describe(*deadline),
548            ),
549            Self::WithdrawalStuck {
550                wid,
551                event_type,
552                deadline,
553                timing,
554            } => write!(
555                f,
556                "WithdrawalStuck(wid={wid}, event_type={event_type:?}, deadline={}, {})",
557                utc_timestamp(*deadline),
558                timing.describe(*deadline),
559            ),
560            Self::WithdrawalInvalid { wid, reason } => {
561                write!(f, "WithdrawalInvalid(wid={wid}, {reason})")
562            }
563            Self::DepositUnbacked { sui } => write!(
564                f,
565                "DepositUnbacked({})",
566                MonitorEvent::DepositApproved(sui.clone()),
567            ),
568            Self::DepositLate {
569                sui,
570                deadline,
571                block_time,
572            } => write!(
573                f,
574                "DepositLate({}, deadline={}, block_time={})",
575                MonitorEvent::DepositApproved(sui.clone()),
576                utc_timestamp(*deadline),
577                utc_timestamp(*block_time),
578            ),
579            Self::DepositMismatch { request_id, reason } => {
580                write!(f, "DepositMismatch(request_id={request_id}, {reason})")
581            }
582            Self::DepositApprovalStalled {
583                outpoints,
584                last_approval,
585            } => {
586                let outpoints = outpoints.iter().map(ToString::to_string);
587                write!(
588                    f,
589                    "DepositApprovalStalled(outpoints=[{}], last_approval=",
590                    outpoints.collect::<Vec<_>>().join(", "),
591                )?;
592                match last_approval {
593                    Some(time) => write!(f, "{})", utc_timestamp(*time)),
594                    None => write!(f, "none)"),
595                }
596            }
597            Self::BtcRpcInconsistency(inconsistency) => {
598                write!(f, "BtcRpcInconsistency({inconsistency})")
599            }
600            Self::UtxoTooOld {
601                outpoint,
602                age_secs,
603                threshold_secs,
604            } => write!(
605                f,
606                "UtxoTooOld(outpoint={outpoint}, age={}, threshold={})",
607                human_duration(*age_secs),
608                human_duration(*threshold_secs),
609            ),
610            Self::UtxoUnexpectedSpend { outpoint, amount } => write!(
611                f,
612                "UtxoUnexpectedSpend(outpoint={outpoint}, amount={amount})"
613            ),
614            Self::UtxoOutsideMovePoolSpent {
615                outpoint,
616                txid,
617                held,
618            } => write!(
619                f,
620                "UtxoOutsideMovePoolSpent(outpoint={outpoint}, txid={txid}, held={held:?})"
621            ),
622            Self::UtxoOriginConflict {
623                outpoint,
624                held,
625                new,
626            } => write!(
627                f,
628                "UtxoOriginConflict(outpoint={outpoint}, held={held:?}, new={new:?})"
629            ),
630            Self::HbtcSupplyExceedsPool {
631                hbtc_supply,
632                pool_value,
633                since,
634            } => write!(
635                f,
636                "HbtcSupplyExceedsPool(hbtc_supply={hbtc_supply}, pool_value={pool_value}, excess={}, since={})",
637                hbtc_supply.saturating_sub(*pool_value),
638                utc_timestamp(*since),
639            ),
640            Self::SourceStale {
641                source,
642                checked_through,
643                limit_secs,
644            } => write!(
645                f,
646                "SourceStale(source={source}, checked_through={}, limit={})",
647                utc_timestamp(*checked_through),
648                human_duration(*limit_secs),
649            ),
650            Self::SnapshotStale {
651                last_saved,
652                limit_secs,
653            } => write!(
654                f,
655                "SnapshotStale(last_saved={}, limit={})",
656                utc_timestamp(*last_saved),
657                human_duration(*limit_secs),
658            ),
659            Self::GuardianClockSkew {
660                session_id,
661                object_key,
662                logged_at_ms,
663                stored_at_ms,
664            } => {
665                let logged_at = unix_millis_to_seconds(*logged_at_ms);
666                let stored_at = unix_millis_to_seconds(*stored_at_ms);
667                let skew_ms = i128::from(*stored_at_ms) - i128::from(*logged_at_ms);
668                write!(
669                    f,
670                    "GuardianClockSkew(session_id={session_id}, key={object_key}, guardian_time={}, s3_time={}, s3_minus_guardian={} ({skew_ms}ms))",
671                    utc_timestamp(logged_at),
672                    utc_timestamp(stored_at),
673                    human_timestamp_delta(stored_at, logged_at),
674                )
675            }
676            Self::GuardianInconsistency(inconsistency) => {
677                write!(f, "GuardianInconsistency({inconsistency})")
678            }
679        }
680    }
681}
