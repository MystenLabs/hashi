4//! The findings the monitor reports.
5//!
6//! Why it exists: every check in the pollers and the state machines reports
7//! its result as one value of `MonitorFinding`. The auditor logs each
8//! finding, and the metrics record the time of the last one.
9//!
10//! Assumptions:
11//! - Every finding needs an operator action, so all findings have one level.
12//!   A condition that needs no action is not a finding.
13//! - `BtcRpcInconsistency` means a Bitcoin RPC answer contradicts one the
14//!   monitor already holds. The bridge cannot cause it. The inner enum names
15//!   the contradiction.
16//! - `GuardianInconsistency` means a guardian log that an honest guardian
17//!   cannot write. The guardian is the cause. The inner enum names the rule
18//!   it breaks.
19

42pub enum EventRelation {
43    Predecessor,
44    Successor,
45}

47/// Findings emitted by the monitor.
48#[derive(Clone, Debug, Eq, PartialEq)]
49pub enum MonitorFinding {
50    InvalidEventAdded(String),
51    /// A withdrawal event came after its deadline in the timing table.
52    WithdrawalEventLate {
53        event: MonitorWithdrawalEvent,
54        relation: EventRelation,
55        deadline: UnixSeconds,
56        occurred_at: UnixSeconds, // same as event.timestamp
57    },
58    /// A withdrawal event is still missing after its deadline in the timing
59    /// table, at the cursor of its source.
60    WithdrawalEventMissing {
61        wid: WithdrawalID,
62        event_type: WithdrawalEventType,
63        relation: EventRelation,
64        deadline: UnixSeconds,
65        cursor: UnixSeconds,
66    },
67    /// The Hashi approval's txid is not the txid of the transaction that its
68    /// inputs and outputs build.
69    WithdrawalTxidMismatch {
70        wid: WithdrawalID,
71        claimed: Txid,
72        computed: Txid,
73    },
74    /// The Hashi approval's inputs and outputs do not build a Bitcoin transaction.
75    WithdrawalTxUnbuildable {
76        wid: WithdrawalID,
77        claimed: Txid,
78        reason: String,
79    },
80    /// The Hashi approval and the guardian's log state different inputs,
81    /// user payouts or change for the withdrawal.
82    WithdrawalMismatch {
83        wid: WithdrawalID,
84        sui: WithdrawalTx,
85        guardian: WithdrawalTx,
86    },
87    /// Hashi approved a deposit for minting that Bitcoin shows no matching
88    /// deposit for.
89    DepositUnbacked {
90        sui: SuiDepositApprovedEvent,
91    },
92    /// Hashi approved a deposit whose block is later than `deadline`, the
93    /// approval plus the skew. Either Hashi approved before Bitcoin
94    /// confirmed, or a miner dated the block past the skew.
95    DepositLate {
96        sui: SuiDepositApprovedEvent,
97        deadline: UnixSeconds,
98        block_time: UnixSeconds,
99    },
100    /// The deposit Bitcoin confirmed differs in amount or script from the
101    /// one Hashi approved.
102    DepositMismatch {
103        sui: SuiDepositApprovedEvent,
104        btc: ConfirmedDeposit,
105    },
106    /// A Bitcoin RPC answer contradicts one the monitor already holds. See
107    /// `BtcRpcInconsistency`.
108    BtcRpcInconsistency(BtcRpcInconsistency),
109    /// A UTXO in the monitor's UTXO pool has sat unspent on Bitcoin past the
110    /// warn age. The committee can spend a UTXO alone 60 days after it
111    /// confirms.
112    UtxoTooOld {
113        outpoint: OutPoint,
114        age_secs: u64,
115        threshold_secs: u64,
116    },
117    /// A UTXO in the monitor's UTXO pool was spent on Bitcoin, and no Hashi
118    /// withdrawal confirmed on Bitcoin spent it.
119    UtxoUnexpectedSpend {
120        outpoint: OutPoint,
121        amount: u64,
122    },
123    /// A confirmed Hashi withdrawal spent a UTXO the monitor's UTXO pool
124    /// never held. The bridge spent a UTXO that no verified deposit or
125    /// change brought in, or the pool missed the UTXO's entry.
126    UtxoUnknownInputSpent {
127        outpoint: OutPoint,
128        txid: Txid,
129    },
130    /// Sui's hBTC supply has exceeded the value of the UTXO pool since
131    /// `since`, for longer than the grace. Bitcoin does not back every hBTC.
132    HbtcSupplyExceedsPool {
133        hbtc_supply: u64,
134        pool_value: u64,
135        since: UnixSeconds,
136    },
137    /// A source was last checked longer ago than its limit. Its events go
138    /// unchecked until it catches up.
139    SourceStale {
140        source: &'static str,
141        checked_through: UnixSeconds,
142        limit_secs: u64,
143    },
144    /// No snapshot save has succeeded for longer than the limit. A restart
145    /// repeats the work done since the last save.
146    SnapshotStale {
147        last_saved: UnixSeconds,
148        limit_secs: u64,
149    },
150    /// The time the guardian signed into a log differs from the time S3
151    /// recorded the object by more than `MAX_EXPECTED_LOG_CLOCK_SKEW`.
152    GuardianClockSkew {
153        session_id: SessionID,
154        object_key: String,
155        /// The guardian's signed time.
156        logged_at_ms: UnixMillis,
157        /// S3's `LastModified` time.
158        stored_at_ms: UnixMillis,
159    },
160    /// The guardian's logs break a rule that an honest guardian keeps. See
161    /// `GuardianInconsistency`.
162    GuardianInconsistency(GuardianInconsistency),
163}
164
165impl MonitorFinding {
166    /// The variant name. Alerting labels each finding with it.
167    pub fn kind(&self) -> &'static str {
168        match self {
169            Self::InvalidEventAdded(_) => "InvalidEventAdded",
170            Self::WithdrawalEventLate { .. } => "WithdrawalEventLate",
171            Self::WithdrawalEventMissing { .. } => "WithdrawalEventMissing",
172            Self::WithdrawalTxidMismatch { .. } => "WithdrawalTxidMismatch",
173            Self::WithdrawalTxUnbuildable { .. } => "WithdrawalTxUnbuildable",
174            Self::WithdrawalMismatch { .. } => "WithdrawalMismatch",
175            Self::DepositUnbacked { .. } => "DepositUnbacked",
176            Self::DepositLate { .. } => "DepositLate",
177            Self::DepositMismatch { .. } => "DepositMismatch",
178            Self::BtcRpcInconsistency(_) => "BtcRpcInconsistency",
179            Self::UtxoTooOld { .. } => "UtxoTooOld",
180            Self::UtxoUnexpectedSpend { .. } => "UtxoUnexpectedSpend",
181            Self::UtxoUnknownInputSpent { .. } => "UtxoUnknownInputSpent",
182            Self::HbtcSupplyExceedsPool { .. } => "HbtcSupplyExceedsPool",
183            Self::SourceStale { .. } => "SourceStale",
184            Self::SnapshotStale { .. } => "SnapshotStale",
185            Self::GuardianClockSkew { .. } => "GuardianClockSkew",
186            Self::GuardianInconsistency(_) => "GuardianInconsistency",
187        }
188    }
189}

191/// A Bitcoin RPC answer that contradicts one the monitor already holds. A
192/// chain cannot produce it, so it is a lying node, a reorg, or a monitor
193/// bug. The bridge is not the cause, and an engineer has to look.
194#[derive(Clone, Debug, Eq, PartialEq)]
195pub enum BtcRpcInconsistency {
196    /// Bitcoin confirmed an outpoint the monitor's UTXO pool knows with a
197    /// second amount or script. An output has one of each.
198    UtxoMismatch {
199        outpoint: OutPoint,
200        held_amount: u64,
201        seen_amount: u64,
202        held_script: ScriptBuf,
203        seen_script: ScriptBuf,
204    },
205    /// A UTXO the monitor's UTXO pool holds does not exist on Bitcoin, and
206    /// its grace period has passed. Bitcoin confirmed it once.
207    UtxoVanished {
208        outpoint: OutPoint,
209        first_seen: UnixSeconds,
210    },
211}

213/// A guardian log that an honest guardian cannot write. The guardian is
214/// the cause, and an engineer has to look.
215#[derive(Clone, Debug, Eq, PartialEq)]
216pub enum GuardianInconsistency {
217    /// Two withdrawal logs have the same seq. The guardian's limiter takes
218    /// each seq once, so two guardian sessions signed from one limiter state.
219    SeqDuplicate { seq: u64, object_key: String },
220    /// No withdrawal log has `seq` by `expect_by`, but a later seq has one.
221    /// The guardian signed a withdrawal without a log, or a log was lost.
222    SeqGap { seq: u64, expect_by: UnixSeconds },
223    /// A withdrawal log skips more than a few seqs: its seq is far above
224    /// `next_seq`, the seq that the monitor expects next.
225    SeqJump {
226        next_seq: u64,
227        seq: u64,
228        object_key: String,
229    },
230}
231
232impl fmt::Display for GuardianInconsistency {
233    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
234        match self {
235            Self::SeqDuplicate { seq, object_key } => {
236                write!(f, "SeqDuplicate(seq={seq}, key={object_key})")
237            }
238            Self::SeqGap { seq, expect_by } => write!(
239                f,
240                "SeqGap(seq={seq}, expect_by={})",
241                utc_timestamp(*expect_by)
242            ),
243            Self::SeqJump {
244                next_seq,
245                seq,
246                object_key,
247            } => write!(
248                f,
249                "SeqJump(next_seq={next_seq}, seq={seq}, key={object_key})"
250            ),
251        }
252    }
253}
254
255impl fmt::Display for BtcRpcInconsistency {
256    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
257        match self {
258            Self::UtxoMismatch {
259                outpoint,
260                held_amount,
261                seen_amount,
262                held_script,
263                seen_script,
264            } => write!(
265                f,
266                "UtxoMismatch(outpoint={outpoint}, held_amount={held_amount}, seen_amount={seen_amount}, held_script={held_script:x}, seen_script={seen_script:x})"
267            ),
268            Self::UtxoVanished {
269                outpoint,
270                first_seen,
271            } => write!(
272                f,
273                "UtxoVanished(outpoint={outpoint}, first_seen={})",
274                utc_timestamp(*first_seen),
275            ),
276        }
277    }
278}

280impl fmt::Display for MonitorFinding {
281    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
282        match self {
283            Self::InvalidEventAdded(message) => {
284                write!(f, "InvalidEventAdded(message={message})")
285            }
286            Self::WithdrawalEventLate {
287                event,
288                relation,
289                deadline,
290                occurred_at,
291            } => write!(
292                f,
293                "WithdrawalEventLate(event={}, relation={relation:?}, deadline={}, occurred_at={}, late_by={})",
294                MonitorEvent::Withdrawal(event.clone()),
295                utc_timestamp(*deadline),
296                utc_timestamp(*occurred_at),
297                human_duration(occurred_at.saturating_sub(*deadline)),
298            ),
299            Self::WithdrawalEventMissing {
300                wid,
301                event_type,
302                relation,
303                deadline,
304                cursor,
305            } => write!(
306                f,
307                "WithdrawalEventMissing(wid={wid}, event_type={event_type:?}, relation={relation:?}, deadline={}, cursor={})",
308                utc_timestamp(*deadline),
309                utc_timestamp(*cursor),
310            ),
311            Self::WithdrawalTxidMismatch {
312                wid,
313                claimed,
314                computed,
315            } => write!(
316                f,
317                "WithdrawalTxidMismatch(wid={wid}, claimed={claimed}, computed={computed})"
318            ),
319            Self::WithdrawalTxUnbuildable {
320                wid,
321                claimed,
322                reason,
323            } => write!(
324                f,
325                "WithdrawalTxUnbuildable(wid={wid}, claimed={claimed}, reason={reason})"
326            ),
327            Self::WithdrawalMismatch { wid, sui, guardian } => write!(
328                f,
329                "WithdrawalMismatch(wid={wid}, sui={{{sui}}}, guardian={{{guardian}}})",
330            ),
331            Self::DepositUnbacked { sui } => write!(
332                f,
333                "DepositUnbacked({})",
334                MonitorEvent::DepositApproved(sui.clone()),
335            ),
336            Self::DepositLate {
337                sui,
338                deadline,
339                block_time,
340            } => write!(
341                f,
342                "DepositLate({}, deadline={}, block_time={})",
343                MonitorEvent::DepositApproved(sui.clone()),
344                utc_timestamp(*deadline),
345                utc_timestamp(*block_time),
346            ),
347            Self::DepositMismatch { sui, btc } => write!(
348                f,
349                "DepositMismatch(request_id={}, deposit_id={}, hashi_amount={}, btc_amount={}, hashi_script={:x}, btc_script={:x})",
350                sui.request_id,
351                sui.deposit_id,
352                sui.amount,
353                btc.amount,
354                sui.script_pubkey,
355                btc.script_pubkey,
356            ),
357            Self::BtcRpcInconsistency(inconsistency) => {
358                write!(f, "BtcRpcInconsistency({inconsistency})")
359            }
360            Self::UtxoTooOld {
361                outpoint,
362                age_secs,
363                threshold_secs,
364            } => write!(
365                f,
366                "UtxoTooOld(outpoint={outpoint}, age={}, threshold={})",
367                human_duration(*age_secs),
368                human_duration(*threshold_secs),
369            ),
370            Self::UtxoUnexpectedSpend { outpoint, amount } => write!(
371                f,
372                "UtxoUnexpectedSpend(outpoint={outpoint}, amount={amount})"
373            ),
374            Self::UtxoUnknownInputSpent { outpoint, txid } => {
375                write!(f, "UtxoUnknownInputSpent(outpoint={outpoint}, txid={txid})")
376            }
377            Self::HbtcSupplyExceedsPool {
378                hbtc_supply,
379                pool_value,
380                since,
381            } => write!(
382                f,
383                "HbtcSupplyExceedsPool(hbtc_supply={hbtc_supply}, pool_value={pool_value}, excess={}, since={})",
384                hbtc_supply.saturating_sub(*pool_value),
385                utc_timestamp(*since),
386            ),
387            Self::SourceStale {
388                source,
389                checked_through,
390                limit_secs,
391            } => write!(
392                f,
393                "SourceStale(source={source}, checked_through={}, limit={})",
394                utc_timestamp(*checked_through),
395                human_duration(*limit_secs),
396            ),
397            Self::SnapshotStale {
398                last_saved,
399                limit_secs,
400            } => write!(
401                f,
402                "SnapshotStale(last_saved={}, limit={})",
403                utc_timestamp(*last_saved),
404                human_duration(*limit_secs),
405            ),
406            Self::GuardianClockSkew {
407                session_id,
408                object_key,
409                logged_at_ms,
410                stored_at_ms,
411            } => {
412                let logged_at = unix_millis_to_seconds(*logged_at_ms);
413                let stored_at = unix_millis_to_seconds(*stored_at_ms);
414                let skew_ms = i128::from(*stored_at_ms) - i128::from(*logged_at_ms);
415                write!(
416                    f,
417                    "GuardianClockSkew(session_id={session_id}, key={object_key}, guardian_time={}, s3_time={}, s3_minus_guardian={} ({skew_ms}ms))",
418                    utc_timestamp(logged_at),
419                    utc_timestamp(stored_at),
420                    human_timestamp_delta(stored_at, logged_at),
421                )
422            }
423            Self::GuardianInconsistency(inconsistency) => {
424                write!(f, "GuardianInconsistency({inconsistency})")
425            }
426        }
427    }
