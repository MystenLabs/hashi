4//! The Bitcoin client: confirmation lookups over HTTP JSON-RPC.
5//!
6//! Why it exists: the withdrawal E3 and the deposit E1 are Bitcoin
7//! confirmations. Bitcoin has no event stream to scan, so the auditor looks
8//! each pending transaction up by txid.
9//!
10//! Why one height:
11//! - A parent transaction creates an output that a dependent transaction spends.
12//! - At one height, the dependent cannot have more confirmations than its parent.
13//! - All lookups in a tick use one captured height, including lookups in different batches.
14//! - New blocks between RPC responses must not complete a dependent machine before its parent is ready to close.
15//! - Pool maintenance needs each complete flow's parent machine already closed or ready to close.
16//!
17//! How it works:
18//! 1. At startup, the auditor reads Hashi's on-chain
19//!    `bitcoin_confirmation_threshold` and gives it to the client. A
20//!    transaction counts as confirmed at that depth, as it does for the
21//!    committee. On mainnet, a threshold below `MAINNET_MIN_CONFIRMATIONS`
22//!    stops the monitor.
23//! 2. Before a pass, `synced_height` checks the node and returns its height.
24//!    All batches use this height to calculate confirmation depth.
25//!    Blocks that arrive during the pass count on the next pass.
26//! 3. `prefetch_confirmations` looks all pending txids up in batches of
27//!    `HTTP_JSON_RPC_BATCH_SIZE`, with `HTTP_JSON_RPC_BATCHES_IN_FLIGHT`
28//!    batches sent at once, and `lookup_confirmation` serves each one from
29//!    that cache. A confirmed answer is kept across passes: a confirmation
30//!    at the threshold depth does not change. `forget_unconfirmed` drops
31//!    the rest at the start of each pass, so an unconfirmed transaction is
32//!    looked up again. An answer for a txid no longer asked for is dropped.
33//!    When a lookup fails, the batches that finished are kept, so the next
34//!    pass asks only about the rest.
35//! 4. A transaction at the threshold depth is returned with the time of
36//!    its own block and the time of the block that gave it that depth.
37//! 5. Requests are not paced: the node is the bridge's own. A request that
38//!    fails to send, gets an HTTP 502, 503 or 504, or is rate limited is
39//!    retried with exponential backoff, `MAX_REQUEST_RETRIES` times. Core
40//!    answers a JSON-RPC error with HTTP 500, so 500 is not retried.
41//!
42//! Assumptions:
43//! - The threshold is read once. A governance change takes effect in the
44//!   monitor at its next restart.
45//! - An unconfirmed answer is never reused across passes. So the auditor
46//!   can stamp every answer with the pass's start time: the real lookup
47//!   is no earlier. A cached confirmed answer does not need a time.
48//! - The endpoint serves `getrawtransaction` with verbose output, which
49//!   needs `txindex` or a hosted provider.
50//! - A request that still fails after the retries fails the pass, which
51//!   is retried on the next tick.

82/// Calls per JSON-RPC batch request.
83const HTTP_JSON_RPC_BATCH_SIZE: usize = 100;
84/// Confirmation batches sent at the same time. A probe of the testnet node
85/// (2026-10-19) served eight batches of 100 at once fastest, with no errors.
86const HTTP_JSON_RPC_BATCHES_IN_FLIGHT: usize = 8;
87/// Txids between two progress log lines of a confirmation lookup.
88const LOOKUP_PROGRESS_EVERY: usize = 10_000;
89/// Retries of one request. The delays double from one second, so the
90/// last retry comes about a minute after the first failure.
91const MAX_REQUEST_RETRIES: usize = 6;
92/// The lowest confirmation threshold the monitor accepts on Bitcoin
93/// mainnet. Hashi's launch tool refuses a lower one there too.
94pub const MAINNET_MIN_CONFIRMATIONS: u64 = 6;
95
96/// The on-chain confirmation threshold, refused on mainnet below
97/// `MAINNET_MIN_CONFIRMATIONS`.
98pub fn checked_confirmation_threshold(threshold: u64, network: Network) -> anyhow::Result<u64> {
99    anyhow::ensure!(
100        network != Network::Bitcoin || threshold >= MAINNET_MIN_CONFIRMATIONS,
101        "Hashi's bitcoin_confirmation_threshold is {threshold} on mainnet; \
102         the monitor needs at least {MAINNET_MIN_CONFIRMATIONS}"
103    );
104    Ok(threshold)
105}

107/// What Bitcoin holds at an outpoint. `Missing` and `Spent` differ by the
108/// transaction lookup, `Spent` and `Unspent` by the UTXO set.
109#[derive(Clone, Debug, PartialEq, Eq)]
110pub enum OutpointState {
111    /// The transaction is unknown or not confirmed deep enough, or it has no
112    /// output at that vout.
113    Missing,
114    /// The output exists, and a confirmed transaction has spent it.
115    Spent,
116    /// The output exists and is in the UTXO set. A mempool spend does not
117    /// change this.
118    Unspent {
119        /// The time of the block that holds the output.
120        block_time: UnixSeconds,
121    },
122}

171impl BlockchainInfo {
172    fn ensure_synced(&self) -> anyhow::Result<()> {
173        anyhow::ensure!(
174            !self.initialblockdownload && self.blocks >= self.headers,
175            "bitcoind is not synced: initialblockdownload={}, blocks={}, headers={}",

177            self.blocks,

180        Ok(())
181    }

184/// The answers of a confirmation lookup, and the first error if a batch
185/// failed. The answers of the batches that finished are kept either way.
186struct LookupOutcome {
187    confirmations: HashMap<Txid, Option<BtcConfirmation>>,
188    error: Option<anyhow::Error>,
189}
190
191/// A mined transaction before its block header is read.
192struct MinedTx {
193    block_hash: BlockHash,
194    inputs: Vec<OutPoint>,
195    outputs: Vec<TxOut>,
196}
197
198impl RawTransactionInfo {
199    /// The block hash, inputs, and outputs of a mined transaction.
200    fn mined(self, txid: Txid) -> anyhow::Result<Option<MinedTx>> {
201        let Some(block_hash) = self.blockhash else {
202            debug!(%txid, "bitcoin tx found but not mined yet");
203            return Ok(None);
204        };
205        if self.confirmations == 0 {
206            return Ok(None);
207        }
208
209        let block_hash = BlockHash::from_str(&block_hash)
210            .with_context(|| format!("invalid bitcoin block hash for {txid}"))?;
211        let tx: Transaction = deserialize_hex(&self.hex)
212            .with_context(|| format!("invalid bitcoin transaction hex for {txid}"))?;
213        anyhow::ensure!(
214            tx.compute_txid() == txid,
215            "bitcoin returned transaction {} for {txid}",
216            tx.compute_txid()
217        );
218        Ok(Some(MinedTx {
219            block_hash,
220            inputs: tx.input.iter().map(|input| input.previous_output).collect(),
221            outputs: tx.output,
222        }))
223    }
224}

226impl BtcRpcClient {
227    /// A client that counts a transaction as confirmed at `required_confirmations`.
228    pub fn new(cfg: &Config, required_confirmations: u64) -> anyhow::Result<Self> {
229        Ok(Self {
230            transport: HttpJsonRpcTransport {
231                rpc_url: cfg.btc.rpc_url.clone(),
232                headers: cfg.btc.http_headers.clone(),
233                required_confirmations,
234            },
235            confirmation_cache: RefCell::new(HashMap::new()),
236        })
237    }

239    /// Return the node height after the synchronization check succeeds.
240    pub fn synced_height(&self) -> anyhow::Result<u64> {
241        let response = self
242            .transport
243            .call("getblockchaininfo", serde_json::json!([]))?;
244        if let Some(error) = response.error {
245            anyhow::bail!(
246                "bitcoin getblockchaininfo failed: {} ({})",
247                error.message,
248                error.code
249            );
250        }
251        let info: BlockchainInfo = serde_json::from_value(
252            response
253                .result
254                .context("bitcoin getblockchaininfo response is missing its result")?,
255        )
256        .context("failed to parse bitcoin blockchain info")?;
257        info.ensure_synced()?;
258        Ok(info.blocks)
259    }
260
261    /// Drop the unconfirmed answers, so the next lookup asks again. The
262    /// confirmed ones stay: a confirmation at the threshold depth does not
263    /// change.
264    pub fn forget_unconfirmed(&self) {
265        self.confirmation_cache
266            .borrow_mut()
267            .retain(|_, confirmation| confirmation.is_some());
268    }
269
270    /// Fetch missing confirmations at `tip_height` and remove answers absent from `txids`.
271    /// Keep completed batches if another batch fails.
272    pub fn prefetch_confirmations(&self, txids: &[Txid], tip_height: u64) -> anyhow::Result<()> {
273        let asked = txids.iter().copied().collect::<HashSet<_>>();
274        let missing = {
275            let cache = self.confirmation_cache.borrow();
276            txids
277                .iter()
278                .filter(|txid| !matches!(cache.get(*txid), Some(Some(_))))
279                .copied()
280                .collect::<Vec<_>>()
281        };
282        info!(
283            cached = txids.len() - missing.len(),
284            looked_up = missing.len(),
285            "bitcoin confirmations served from the cache"
286        );
287        let outcome = self.transport.lookup_confirmations(&missing, tip_height);
288        let mut cache = self.confirmation_cache.borrow_mut();
289        cache.extend(outcome.confirmations);
290        cache.retain(|txid, _| asked.contains(txid));
291        outcome.error.map_or(Ok(()), Err)
292    }
293
294    /// Query BTC RPC to check if a transaction is confirmed.
295    /// Returns
296    /// - `Ok(Some(confirmation))` if txid is confirmed,
297    /// - `Ok(None)` if txid is not seen or txid is seen but not confirmed,
298    /// - `Err(...)` for all other errors
299    pub fn lookup_confirmation(&self, txid: Txid) -> anyhow::Result<Option<BtcConfirmation>> {
300        if let Some(result) = self.confirmation_cache.borrow().get(&txid) {
301            return Ok(result.clone());
302        }
303        let tip_height = self.synced_height()?;
304        let result = self.transport.lookup_confirmation(txid, tip_height)?;
305        self.confirmation_cache
306            .borrow_mut()
307            .insert(txid, result.clone());
308        Ok(result)
309    }

311    /// The state of each outpoint on Bitcoin. The transactions are fetched
312    /// fresh, and the confirmation cache is not touched.
313    pub fn lookup_outpoint_states(
314        &self,
315        outpoints: &[OutPoint],
316        tip_height: u64,
317    ) -> anyhow::Result<HashMap<OutPoint, OutpointState>> {
318        let txids = outpoints
319            .iter()
320            .map(|outpoint| outpoint.txid)
321            .collect::<HashSet<_>>()
322            .into_iter()
323            .collect::<Vec<_>>();
324        let outcome = self.transport.lookup_confirmations(&txids, tip_height);
325        if let Some(error) = outcome.error {
326            return Err(error);
327        }
328
329        // Only an output that exists can be in the UTXO set.
330        let mut states = HashMap::with_capacity(outpoints.len());
331        let mut existing = Vec::new();
332        for outpoint in outpoints {
333            let confirmation = outcome
334                .confirmations
335                .get(&outpoint.txid)
336                .with_context(|| format!("missing bitcoin lookup result for {}", outpoint.txid))?;
337            match confirmation {
338                Some(confirmation) if (outpoint.vout as usize) < confirmation.outputs.len() => {
339                    existing.push((*outpoint, confirmation.block_time));
340                }
341                _ => {
342                    states.insert(*outpoint, OutpointState::Missing);
343                }
344            }
345        }
346
347        let existing_outpoints = existing
348            .iter()
349            .map(|(outpoint, _)| *outpoint)
350            .collect::<Vec<_>>();
351        let unspent = self.transport.lookup_unspent(&existing_outpoints)?;
352        for (outpoint, block_time) in existing {
353            let state = match unspent.get(&outpoint) {
354                Some(true) => OutpointState::Unspent { block_time },
355                Some(false) => OutpointState::Spent,
356                None => anyhow::bail!("missing bitcoin gettxout result for {outpoint}"),
357            };
358            states.insert(outpoint, state);
359        }
360        Ok(states)
361    }
362}

371impl HttpJsonRpcTransport {
372    fn call(&self, method: &str, params: Value) -> anyhow::Result<JsonRpcEnvelope> {
373        let body = serde_json::json!({
374            "jsonrpc": "1.0",
375            "id": "hashi-monitor",
376            "method": method,
377            "params": params,
378        });
379        serde_json::from_value(self.post_json(&body)?)
380            .context("failed to decode bitcoin JSON-RPC response")
381    }
382
383    fn call_batch(&self, body: &[Value]) -> anyhow::Result<Vec<JsonRpcEnvelope>> {
384        let body = Value::Array(body.to_vec());
385        serde_json::from_value(self.post_json(&body)?)
386            .context("failed to decode bitcoin JSON-RPC batch response")
387    }
388
389    /// Send one request. A send that fails, a timeout among them, an HTTP
390    /// 502, 503 or 504 and a rate limit are retried with bounded exponential
391    /// backoff. Core sends 503 when its work queue is full. Any other status
392    /// goes to the JSON-RPC decoding: Core answers an RPC error with 500.
393    fn post_json(&self, body: &Value) -> anyhow::Result<Value> {
394        for attempt in 0..=MAX_REQUEST_RETRIES {
395            let request = minreq::post(&self.rpc_url)
396                .with_timeout(60)
397                .with_headers(self.headers.clone())
398                .with_json(body)
399                .context("failed to encode bitcoin JSON-RPC request")?;
400            let response = match request.send() {
401                Ok(response) => response,
402                Err(error) => {
403                    Self::wait_before_retry(attempt, &format!("request failed: {error}"))?;
404                    continue;
405                }
406            };
407
408            if matches!(response.status_code, 429 | 502 | 503 | 504) {
409                Self::wait_before_retry(attempt, &format!("HTTP {}", response.status_code))?;
410                continue;
411            }
412
413            let response_body: Value = response.json().with_context(|| {
414                format!(
415                    "failed to decode bitcoin JSON-RPC response (HTTP {})",
416                    response.status_code
417                )
418            })?;
419            if json_rpc_rate_limited(&response_body) {
420                Self::wait_before_retry(attempt, "rate limited")?;
421                continue;
422            }
423
424            return Ok(response_body);
425        }
426
427        unreachable!("the retry loop either returns or errors")
428    }
429
430    fn wait_before_retry(attempt: usize, reason: &str) -> anyhow::Result<()> {
431        if attempt == MAX_REQUEST_RETRIES {
432            anyhow::bail!("bitcoin JSON-RPC {reason} after {MAX_REQUEST_RETRIES} retries");
433        }
434
435        let delay = Duration::from_secs(1 << attempt.min(5));
436        warn!(
437            retry = attempt + 1,
438            max_retries = MAX_REQUEST_RETRIES,
439            delay_secs = delay.as_secs(),
440            reason,
441            "bitcoin JSON-RPC request will be retried"
442        );
443        thread::sleep(delay);
444        Ok(())
445    }
446
447    fn lookup_confirmation(
448        &self,
449        txid: Txid,
450        tip_height: u64,
451    ) -> anyhow::Result<Option<BtcConfirmation>> {
452        self.lookup_confirmation_batch(&[txid], tip_height)?
453            .remove(&txid)
454            .with_context(|| format!("missing bitcoin lookup result for {txid}"))
455    }
456
457    /// Look every txid up, `HTTP_JSON_RPC_BATCHES_IN_FLIGHT` batches at a
458    /// time. Each worker takes the next batch until none is left. The first
459    /// failure stops the others at their next batch. The answers of the
460    /// batches that finished are returned with that error.
461    fn lookup_confirmations(&self, txids: &[Txid], tip_height: u64) -> LookupOutcome {
462        let batches = txids.chunks(HTTP_JSON_RPC_BATCH_SIZE).collect::<Vec<_>>();
463        let next_batch = AtomicUsize::new(0);
464        let processed = AtomicUsize::new(0);
465        let results = Mutex::new(Vec::with_capacity(batches.len()));
466        thread::scope(|scope| {
467            for _ in 0..HTTP_JSON_RPC_BATCHES_IN_FLIGHT.min(batches.len()) {
468                scope.spawn(|| {
469                    while let Some(batch) = batches.get(next_batch.fetch_add(1, Ordering::Relaxed))
470                    {
471                        let result = self.lookup_confirmation_batch(batch, tip_height);
472                        if result.is_err() {
473                            next_batch.store(batches.len(), Ordering::Relaxed);
474                        }
475                        let processed_txids =
476                            processed.fetch_add(batch.len(), Ordering::Relaxed) + batch.len();
477                        if processed_txids.is_multiple_of(LOOKUP_PROGRESS_EVERY)
478                            || processed_txids == txids.len()
479                        {
480                            info!(
481                                processed_txids,
482                                total_txids = txids.len(),
483                                "bitcoin confirmation lookup progress"
484                            );
485                        }
486                        results.lock().expect("a worker never panics").push(result);
487                    }
488                });
489            }
490        });
491        let mut outcome = LookupOutcome {
492            confirmations: HashMap::with_capacity(txids.len()),
493            error: None,
494        };
495        for result in results.into_inner().expect("a worker never panics") {
496            match result {
497                Ok(confirmations) => outcome.confirmations.extend(confirmations),
498                Err(error) => {
499                    outcome.error.get_or_insert(error);
500                }
501            }
502        }
503        outcome
504    }
505
506    fn lookup_confirmation_batch(
507        &self,
508        txids: &[Txid],
509        tip_height: u64,
510    ) -> anyhow::Result<HashMap<Txid, Option<BtcConfirmation>>> {
511        let requests = txids
512            .iter()
513            .enumerate()
514            .map(|(id, txid)| {
515                serde_json::json!({
516                    "jsonrpc": "2.0",
517                    "id": id,
518                    "method": "getrawtransaction",
519                    "params": [txid.to_string(), true],
520                })
521            })
522            .collect::<Vec<_>>();
523        let responses = self.call_batch(&requests)?;
524        let mut seen = HashSet::with_capacity(txids.len());
525        let mut mined = HashMap::with_capacity(txids.len());
526
527        for response in responses {
528            let index = response_index(&response, txids.len())?;
529            anyhow::ensure!(
530                seen.insert(index),
531                "duplicate bitcoin getrawtransaction batch response ID {index}"
532            );
533            let txid = txids[index];
534            if let Some(error) = response.error {
535                if error.code == -5 {
536                    mined.insert(txid, None);
537                    continue;
538                }
539                anyhow::bail!(
540                    "bitcoin getrawtransaction failed for {txid}: {} ({})",
541                    error.message,
542                    error.code
543                );
544            }
545            let info: RawTransactionInfo = serde_json::from_value(
546                response
547                    .result
548                    .context("bitcoin getrawtransaction batch response is missing its result")?,
549            )
550            .with_context(|| format!("failed to parse transaction info for {txid}"))?;
551            mined.insert(txid, info.mined(txid)?);
552        }
553        anyhow::ensure!(
554            seen.len() == txids.len(),
555            "bitcoin getrawtransaction batch returned {}/{} responses",
556            seen.len(),
557            txids.len()
558        );
559
560        let mined_block_hashes = mined
561            .values()
562            .flatten()
563            .map(|tx| tx.block_hash)
564            .collect::<HashSet<_>>()
565            .into_iter()
566            .collect::<Vec<_>>();
567        let confirmed_block_times = self.confirmed_block_times(&mined_block_hashes, tip_height)?;
568
569        Ok(mined
570            .into_iter()
571            .map(|(txid, transaction)| {
572                let confirmation = transaction.and_then(|tx| {
573                    confirmed_block_times.get(&tx.block_hash).map(
574                        |&(block_time, confirmed_time)| BtcConfirmation {
575                            block_time,
576                            confirmed_time,
577                            inputs: tx.inputs,
578                            outputs: tx.outputs,
579                        },
580                    )
581                });
582                (txid, confirmation)
583            })
584            .collect())
585    }

587    /// Return block and confirmation times for blocks deep enough at `tip_height`.
588    /// The holding block counts as the first confirmation.
589    fn confirmed_block_times(
590        &self,
591        block_hashes: &[BlockHash],
592        tip_height: u64,
593    ) -> anyhow::Result<HashMap<BlockHash, (UnixSeconds, UnixSeconds)>> {
594        let mut headers = self.block_headers(block_hashes)?;
595        let confirming_height =
596            |header: &RawBlockHeader| header.height + self.required_confirmations - 1;
597        headers.retain(|_, header| confirming_height(header) <= tip_height);
598        let confirming_heights = headers
599            .values()
600            .map(confirming_height)
601            .collect::<HashSet<_>>()
602            .into_iter()
603            .collect::<Vec<_>>();
604        let confirming_hashes = self.block_hashes(&confirming_heights)?;
605        let confirming_headers =
606            self.block_headers(&confirming_hashes.values().copied().collect::<Vec<_>>())?;
607
608        headers
609            .iter()
610            .map(|(hash, header)| {
611                let height = confirming_height(header);
612                let confirming_hash = confirming_hashes
613                    .get(&height)
614                    .with_context(|| format!("missing block hash at height {height}"))?;
615                let confirming = confirming_headers
616                    .get(confirming_hash)
617                    .with_context(|| format!("missing block header for {confirming_hash}"))?;
618                Ok((*hash, (header.time, confirming.time)))
619            })
620            .collect()
621    }

623    /// `getblockheader` for each hash, in batches.
624    fn block_headers(
625        &self,
626        hashes: &[BlockHash],
627    ) -> anyhow::Result<HashMap<BlockHash, RawBlockHeader>> {
628        let mut headers = HashMap::with_capacity(hashes.len());
629        for chunk in hashes.chunks(HTTP_JSON_RPC_BATCH_SIZE) {
630            let requests = chunk
631                .iter()
632                .enumerate()
633                .map(|(id, hash)| {
634                    serde_json::json!({
635                        "jsonrpc": "2.0",
636                        "id": id,
637                        "method": "getblockheader",
638                        "params": [hash.to_string(), true],
639                    })
640                })
641                .collect::<Vec<_>>();
642            let responses = self.call_batch(&requests)?;
643            let mut seen = HashSet::with_capacity(chunk.len());
644            for response in responses {
645                let index = response_index(&response, chunk.len())?;
646                anyhow::ensure!(
647                    seen.insert(index),
648                    "duplicate bitcoin getblockheader batch response ID {index}"
649                );
650                let block_hash = chunk[index];
651                if let Some(error) = response.error {
652                    anyhow::bail!(
653                        "bitcoin getblockheader failed for {block_hash}: {} ({})",
654                        error.message,
655                        error.code
656                    );
657                }
658                let header: RawBlockHeader = serde_json::from_value(
659                    response
660                        .result
661                        .context("bitcoin getblockheader batch response is missing its result")?,
662                )
663                .with_context(|| format!("failed to parse block header for {block_hash}"))?;
664                headers.insert(block_hash, header);
665            }
666            anyhow::ensure!(
667                seen.len() == chunk.len(),
668                "bitcoin getblockheader batch returned {}/{} responses",
669                seen.len(),
670                chunk.len()
671            );
672        }
673        Ok(headers)
674    }
675
676    /// `getblockhash` for each height, in batches.
677    fn block_hashes(&self, heights: &[u64]) -> anyhow::Result<HashMap<u64, BlockHash>> {
678        let mut hashes = HashMap::with_capacity(heights.len());
679        for chunk in heights.chunks(HTTP_JSON_RPC_BATCH_SIZE) {
680            let requests = chunk
681                .iter()
682                .enumerate()
683                .map(|(id, height)| {
684                    serde_json::json!({
685                        "jsonrpc": "2.0",
686                        "id": id,
687                        "method": "getblockhash",
688                        "params": [height],
689                    })
690                })
691                .collect::<Vec<_>>();
692            let responses = self.call_batch(&requests)?;
693            let mut seen = HashSet::with_capacity(chunk.len());
694            for response in responses {
695                let index = response_index(&response, chunk.len())?;
696                anyhow::ensure!(
697                    seen.insert(index),
698                    "duplicate bitcoin getblockhash batch response ID {index}"
699                );
700                let height = chunk[index];
701                if let Some(error) = response.error {
702                    anyhow::bail!(
703                        "bitcoin getblockhash failed for height {height}: {} ({})",
704                        error.message,
705                        error.code
706                    );
707                }
708                let hash: String = serde_json::from_value(
709                    response
710                        .result
711                        .context("bitcoin getblockhash batch response is missing its result")?,
712                )
713                .with_context(|| format!("failed to parse block hash at height {height}"))?;
714                let hash = BlockHash::from_str(&hash)
715                    .with_context(|| format!("invalid bitcoin block hash at height {height}"))?;
716                hashes.insert(height, hash);
717            }
718            anyhow::ensure!(
719                seen.len() == chunk.len(),
720                "bitcoin getblockhash batch returned {}/{} responses",
721                seen.len(),
722                chunk.len()
723            );
724        }
725        Ok(hashes)
726    }
727}

729impl HttpJsonRpcTransport {
730    /// Whether each outpoint is in the UTXO set, with the mempool excluded.
731    /// Callers pass outpoints that exist: `gettxout` answers null for a spent
732    /// output and for one that never existed alike.
733    fn lookup_unspent(&self, outpoints: &[OutPoint]) -> anyhow::Result<HashMap<OutPoint, bool>> {
734        let mut unspent = HashMap::with_capacity(outpoints.len());
735        for chunk in outpoints.chunks(HTTP_JSON_RPC_BATCH_SIZE) {
736            let requests = chunk
737                .iter()
738                .enumerate()
739                .map(|(id, outpoint)| {
740                    serde_json::json!({
741                        "jsonrpc": "2.0",
742                        "id": id,
743                        "method": "gettxout",
744                        "params": [outpoint.txid.to_string(), outpoint.vout, false],
745                    })
746                })
747                .collect::<Vec<_>>();
748            let responses = self.call_batch(&requests)?;
749            let mut seen = HashSet::with_capacity(chunk.len());
750            for response in responses {
751                let index = response_index(&response, chunk.len())?;
752                anyhow::ensure!(
753                    seen.insert(index),
754                    "duplicate bitcoin gettxout batch response ID {index}"
755                );
756                let outpoint = chunk[index];
757                if let Some(error) = response.error {
758                    anyhow::bail!(
759                        "bitcoin gettxout failed for {outpoint}: {} ({})",
760                        error.message,
761                        error.code
762                    );
763                }
764                let is_unspent = !matches!(response.result, None | Some(Value::Null));
765                unspent.insert(outpoint, is_unspent);
766            }
767            anyhow::ensure!(
768                seen.len() == chunk.len(),
769                "bitcoin gettxout batch returned {}/{} responses",
770                seen.len(),
771                chunk.len()
772            );
773        }
774        Ok(unspent)
775    }
776}

778fn json_rpc_rate_limited(response: &Value) -> bool {
779    let is_rate_limited = |envelope: &Value| {
780        envelope
781            .get("error")
782            .and_then(|error| error.get("code"))
783            .and_then(Value::as_i64)
784            == Some(429)
785    };
786
787    match response {
788        Value::Array(envelopes) => envelopes.iter().any(is_rate_limited),
789        envelope => is_rate_limited(envelope),
790    }
791}
792
793fn response_index(response: &JsonRpcEnvelope, batch_len: usize) -> anyhow::Result<usize> {
794    let index = response
795        .id
796        .as_u64()
797        .context("bitcoin JSON-RPC batch response ID is not an integer")?;
798    let index =
799        usize::try_from(index).context("bitcoin JSON-RPC batch response ID is too large")?;
800    anyhow::ensure!(
801        index < batch_len,
802        "bitcoin JSON-RPC batch response ID {index} is outside batch of {batch_len}"
803    );
804    Ok(index)
805}

1026    /// Accept one connection, read its request, and answer `response`.
1027    fn respond(listener: &TcpListener, response: &str) {
1028        let (stream, _) = listener.accept().unwrap();
1029        let mut reader = BufReader::new(stream);
1030        let mut content_length = 0;
1031        loop {
1032            let mut line = String::new();
1033            reader.read_line(&mut line).unwrap();
1034            if line == "\r\n" {
1035                break;
1036            }
1037            let line = line.to_ascii_lowercase();
1038            if let Some(value) = line.strip_prefix("content-length:") {
1039                content_length = value.trim().parse().unwrap();
1040            }
1041        }
1042        reader.read_exact(&mut vec![0; content_length]).unwrap();
1043        reader.into_inner().write_all(response.as_bytes()).unwrap();
1044    }
