24const HTTP_JSON_RPC_BATCH_SIZE: usize = 20;
25const HTTP_JSON_RPC_BATCH_DELAY: Duration = Duration::from_millis(200);
26const MAX_RATE_LIMIT_RETRIES: usize = 6;
27const MIN_CONFIRMATIONS: u64 = 6;

39#[derive(Deserialize)]
40struct JsonRpcEnvelope {
41    id: Value,
42    result: Option<Value>,
43    error: Option<JsonRpcError>,

52#[derive(Deserialize)]
53struct RawTransactionInfo {
54    blockhash: Option<String>,
55    #[serde(default)]
56    confirmations: u64,
57}

61    time: u64,

64impl RawTransactionInfo {
65    fn sufficiently_confirmed_block_hash(self, txid: Txid) -> anyhow::Result<Option<BlockHash>> {
66        let Some(block_hash) = self.blockhash else {
67            debug!(%txid, "bitcoin tx found but not mined yet");
68            return Ok(None);
69        };
70        if self.confirmations < MIN_CONFIRMATIONS {
71            debug!(
72                %txid,
73                confirmations = self.confirmations,
74                required_confirmations = MIN_CONFIRMATIONS,
75                "bitcoin tx has insufficient confirmation depth"
76            );
77            return Ok(None);
78        }
79
80        BlockHash::from_str(&block_hash)
81            .with_context(|| format!("invalid bitcoin block hash for {txid}"))
82            .map(Some)
83    }
84}

86impl BtcRpcClient {
87    pub fn new(cfg: &Config) -> anyhow::Result<Self> {
88        Ok(Self {
89            transport: HttpJsonRpcTransport {
90                rpc_url: cfg.btc.resolve_rpc_url()?,
91                headers: cfg.btc.http_headers.clone(),
92            },
93            confirmation_cache: RefCell::new(HashMap::new()),
94        })
95    }
96
97    /// Start a fresh lookup cycle. Confirmed and unconfirmed results are shared
98    /// by all outpoints from the same transaction within one audit tick, while
99    /// later ticks still retry transactions that were previously unconfirmed.
100    pub fn clear_confirmation_cache(&self) {
101        self.confirmation_cache.borrow_mut().clear();
102    }
103
104    pub fn prefetch_confirmations(&self, txids: &[Txid]) -> anyhow::Result<()> {
105        let confirmations = self.transport.lookup_confirmations(txids)?;
106        self.confirmation_cache.borrow_mut().extend(confirmations);
107        Ok(())
108    }
109
110    /// Query BTC RPC to check if a transaction is confirmed.
111    /// Returns
112    /// - `Ok(Some(block_time))` if txid is confirmed,
113    /// - `Ok(None)` if txid is not seen or txid is seen but not confirmed,
114    /// - `Err(...)` for all other errors
115    pub fn lookup_confirmation(&self, txid: Txid) -> anyhow::Result<Option<UnixSeconds>> {
116        if let Some(result) = self.confirmation_cache.borrow().get(&txid) {
117            return Ok(*result);
118        }
119        let result = self.transport.lookup_confirmation(txid)?;
120        self.confirmation_cache.borrow_mut().insert(txid, result);
121        Ok(result)
122    }
123}

125impl HttpJsonRpcTransport {
126    fn call(&self, method: &str, params: Value) -> anyhow::Result<JsonRpcEnvelope> {
127        let body = serde_json::json!({
128            "jsonrpc": "1.0",
129            "id": "hashi-monitor",
130            "method": method,
131            "params": params,
132        });
133        serde_json::from_value(self.post_json(&body)?)
134            .context("failed to decode bitcoin JSON-RPC response")
135    }
136
137    fn call_batch(&self, body: &[Value]) -> anyhow::Result<Vec<JsonRpcEnvelope>> {
138        let body = Value::Array(body.to_vec());
139        serde_json::from_value(self.post_json(&body)?)
140            .context("failed to decode bitcoin JSON-RPC batch response")
141    }
142
143    /// Pace requests so a historical audit does not exhaust a provider's
144    /// rolling throughput bucket, and retry the standard HTTP/JSON-RPC 429
145    /// responses with bounded exponential backoff.
146    // TODO: Classify and retry transient non-rate-limit failures, such as
147    // timeouts, connection errors, and HTTP 5xx responses. They currently
148    // abort a batch audit or defer a continuous retry until the next BTC tick.
149    fn post_json(&self, body: &Value) -> anyhow::Result<Value> {
150        for attempt in 0..=MAX_RATE_LIMIT_RETRIES {
151            let response = minreq::post(&self.rpc_url)
152                .with_timeout(60)
153                .with_headers(self.headers.clone())
154                .with_json(body)
155                .context("failed to encode bitcoin JSON-RPC request")?
156                .send()
157                .context("bitcoin JSON-RPC request failed")?;
158
159            if response.status_code == 429 {
160                Self::wait_before_rate_limit_retry(attempt)?;
161                continue;
162            }
163
164            let response_body: Value = response
165                .json()
166                .context("failed to decode bitcoin JSON-RPC response")?;
167            if json_rpc_rate_limited(&response_body) {
168                Self::wait_before_rate_limit_retry(attempt)?;
169                continue;
170            }
171
172            thread::sleep(HTTP_JSON_RPC_BATCH_DELAY);
173            return Ok(response_body);
174        }
175
176        unreachable!("rate-limit retry loop either returns or errors")
177    }
178
179    fn wait_before_rate_limit_retry(attempt: usize) -> anyhow::Result<()> {
180        if attempt == MAX_RATE_LIMIT_RETRIES {
181            anyhow::bail!(
182                "bitcoin JSON-RPC remained rate limited after {MAX_RATE_LIMIT_RETRIES} retries"
183            );
184        }
185
186        let delay = Duration::from_secs(1 << attempt.min(5));
187        warn!(
188            retry = attempt + 1,
189            max_retries = MAX_RATE_LIMIT_RETRIES,
190            delay_secs = delay.as_secs(),
191            "bitcoin JSON-RPC rate limited; retrying"
192        );
193        thread::sleep(delay);
194        Ok(())
195    }
196
197    fn lookup_confirmation(&self, txid: Txid) -> anyhow::Result<Option<UnixSeconds>> {
198        let response = self.call(
199            "getrawtransaction",
200            serde_json::json!([txid.to_string(), true]),
201        )?;
202        if let Some(error) = response.error {
203            if error.code == -5 {
204                debug!(%txid, "bitcoin tx not found in mempool or chain yet");
205                return Ok(None);
206            }
207            anyhow::bail!(
208                "bitcoin getrawtransaction failed for {txid}: {} ({})",
209                error.message,
210                error.code
211            );
212        }
213        let tx_info: RawTransactionInfo = serde_json::from_value(
214            response
215                .result
216                .context("bitcoin getrawtransaction response is missing its result")?,
217        )
218        .with_context(|| format!("failed to parse transaction info for {txid}"))?;
219        let Some(block_hash) = tx_info.sufficiently_confirmed_block_hash(txid)? else {
220            return Ok(None);
221        };
222
223        let response = self.call(
224            "getblockheader",
225            serde_json::json!([block_hash.to_string(), true]),
226        )?;
227        if let Some(error) = response.error {
228            anyhow::bail!(
229                "bitcoin getblockheader failed for {block_hash}: {} ({})",
230                error.message,
231                error.code
232            );
233        }
234        let header: RawBlockHeader = serde_json::from_value(
235            response
236                .result
237                .context("bitcoin getblockheader response is missing its result")?,
238        )
239        .with_context(|| format!("failed to parse block header for {block_hash}"))?;
240        Ok(Some(header.time))
241    }
242
243    fn lookup_confirmations(
244        &self,
245        txids: &[Txid],
246    ) -> anyhow::Result<HashMap<Txid, Option<UnixSeconds>>> {
247        let mut confirmations = HashMap::with_capacity(txids.len());
248        for (batch_index, chunk) in txids.chunks(HTTP_JSON_RPC_BATCH_SIZE).enumerate() {
249            confirmations.extend(self.lookup_confirmation_batch(chunk)?);
250            let processed_txids = ((batch_index + 1) * HTTP_JSON_RPC_BATCH_SIZE).min(txids.len());
251            if processed_txids.is_multiple_of(200) || processed_txids == txids.len() {
252                info!(
253                    processed_txids,
254                    total_txids = txids.len(),
255                    "bitcoin confirmation lookup progress"
256                );
257            }
258        }
259        Ok(confirmations)
260    }
261
262    fn lookup_confirmation_batch(
263        &self,
264        txids: &[Txid],
265    ) -> anyhow::Result<HashMap<Txid, Option<UnixSeconds>>> {
266        let requests = txids
267            .iter()
268            .enumerate()
269            .map(|(id, txid)| {
270                serde_json::json!({
271                    "jsonrpc": "2.0",
272                    "id": id,
273                    "method": "getrawtransaction",
274                    "params": [txid.to_string(), true],
275                })
276            })
277            .collect::<Vec<_>>();
278        let responses = self.call_batch(&requests)?;
279        let mut seen = HashSet::with_capacity(txids.len());
280        let mut block_hashes = HashMap::with_capacity(txids.len());
281
282        for response in responses {
283            let index = response_index(&response, txids.len())?;
284            anyhow::ensure!(
285                seen.insert(index),
286                "duplicate bitcoin getrawtransaction batch response ID {index}"
287            );
288            let txid = txids[index];
289            if let Some(error) = response.error {
290                if error.code == -5 {
291                    block_hashes.insert(txid, None);
292                    continue;
293                }
294                anyhow::bail!(
295                    "bitcoin getrawtransaction failed for {txid}: {} ({})",
296                    error.message,
297                    error.code
298                );
299            }
300            let info: RawTransactionInfo = serde_json::from_value(
301                response
302                    .result
303                    .context("bitcoin getrawtransaction batch response is missing its result")?,
304            )
305            .with_context(|| format!("failed to parse transaction info for {txid}"))?;
306            let block_hash = info.sufficiently_confirmed_block_hash(txid)?;
307            block_hashes.insert(txid, block_hash);
308        }
309        anyhow::ensure!(
310            seen.len() == txids.len(),
311            "bitcoin getrawtransaction batch returned {}/{} responses",
312            seen.len(),
313            txids.len()
314        );
315
316        let unique_block_hashes = block_hashes
317            .values()
318            .flatten()
319            .copied()
320            .collect::<HashSet<_>>()
321            .into_iter()
322            .collect::<Vec<_>>();
323        let mut block_times = HashMap::with_capacity(unique_block_hashes.len());
324        for hashes in unique_block_hashes.chunks(HTTP_JSON_RPC_BATCH_SIZE) {
325            let requests = hashes
326                .iter()
327                .enumerate()
328                .map(|(id, hash)| {
329                    serde_json::json!({
330                        "jsonrpc": "2.0",
331                        "id": id,
332                        "method": "getblockheader",
333                        "params": [hash.to_string(), true],
334                    })
335                })
336                .collect::<Vec<_>>();
337            let responses = self.call_batch(&requests)?;
338            let mut seen = HashSet::with_capacity(hashes.len());
339            for response in responses {
340                let index = response_index(&response, hashes.len())?;
341                anyhow::ensure!(
342                    seen.insert(index),
343                    "duplicate bitcoin getblockheader batch response ID {index}"
344                );
345                let block_hash = hashes[index];
346                if let Some(error) = response.error {
347                    anyhow::bail!(
348                        "bitcoin getblockheader failed for {block_hash}: {} ({})",
349                        error.message,
350                        error.code
351                    );
352                }
353                let header: RawBlockHeader = serde_json::from_value(
354                    response
355                        .result
356                        .context("bitcoin getblockheader batch response is missing its result")?,
357                )
358                .with_context(|| format!("failed to parse block header for {block_hash}"))?;
359                block_times.insert(block_hash, header.time);
360            }
361            anyhow::ensure!(
362                seen.len() == hashes.len(),
363                "bitcoin getblockheader batch returned {}/{} responses",
364                seen.len(),
365                hashes.len()
366            );
367        }
368
369        block_hashes
370            .into_iter()
371            .map(|(txid, block_hash)| {
372                let confirmation = block_hash
373                    .map(|hash| {
374                        block_times
375                            .get(&hash)
376                            .copied()
377                            .with_context(|| format!("missing block time for {hash}"))
378                    })
379                    .transpose()?;
380                Ok((txid, confirmation))
381            })
382            .collect()
383    }
384}

386fn json_rpc_rate_limited(response: &Value) -> bool {
387    let is_rate_limited = |envelope: &Value| {
388        envelope
389            .get("error")
390            .and_then(|error| error.get("code"))
391            .and_then(Value::as_i64)
392            == Some(429)
393    };
394
395    match response {
396        Value::Array(envelopes) => envelopes.iter().any(is_rate_limited),
397        envelope => is_rate_limited(envelope),
398    }
399}
400
401fn response_index(response: &JsonRpcEnvelope, batch_len: usize) -> anyhow::Result<usize> {
402    let index = response
403        .id
404        .as_u64()
405        .context("bitcoin JSON-RPC batch response ID is not an integer")?;
406    let index =
407        usize::try_from(index).context("bitcoin JSON-RPC batch response ID is too large")?;
408    anyhow::ensure!(
409        index < batch_len,
410        "bitcoin JSON-RPC batch response ID {index} is outside batch of {batch_len}"
411    );
412    Ok(index)
413}

488    // Note that this test requires local bitcoind running.
489    #[tokio::test]
490    async fn lookup_btc_confirmation_with_local_regtest() -> Result<()> {
491        init_test_tracing();
492
493        let temp_dir = TempDir::new()?;
494        let node = BitcoinNodeBuilder::new()
495            .dir(temp_dir.path())
496            .build()
497            .await?;
498        let cfg = test_config(node.rpc_url().to_string());
499        let btc_rpc_client = BtcRpcClient::new(&cfg)?;
500
501        let unknown_txid = Txid::from_slice(&[7u8; 32])?;
502        let unknown = btc_rpc_client.lookup_confirmation(unknown_txid)?;
503        assert!(
504            unknown.is_none(),
505            "expected unknown tx lookup to return none"
506        );
507
508        let destination = node.get_new_address()?;
509        let txid = node.send_to_address(&destination, Amount::from_sat(50_000))?;
510
511        let unconfirmed = btc_rpc_client.lookup_confirmation(txid)?;
512        assert!(unconfirmed.is_none(), "expected unconfirmed transaction");
513
514        node.generate_blocks(1)?;
515        btc_rpc_client.clear_confirmation_cache();
516
517        let insufficiently_confirmed = btc_rpc_client.lookup_confirmation(txid)?;
518        assert!(
519            insufficiently_confirmed.is_none(),
520            "expected one-confirmation transaction to remain pending"
521        );
522
523        node.generate_blocks(MIN_CONFIRMATIONS - 1)?;
524        btc_rpc_client.clear_confirmation_cache();
525
526        let confirmed = btc_rpc_client.lookup_confirmation(txid)?;
527        assert!(confirmed.is_some(), "expected confirmed transaction");
528
529        Ok(())
530    }
