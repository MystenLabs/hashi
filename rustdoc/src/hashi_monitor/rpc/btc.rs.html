4//! The Bitcoin client: confirmation lookups over HTTP JSON-RPC.
5//!
6//! Why it exists: the withdrawal E3 and the deposit E1 are Bitcoin
7//! confirmations. Bitcoin has no event stream to scan, so the auditor looks
8//! each pending transaction up by txid.
9//!
10//! How it works:
11//! 1. Before a pass, `ensure_synced` fails while bitcoind lags its peers,
12//!    since a lagging node reports recent transactions as unconfirmed.
13//! 2. `prefetch_confirmations` looks all pending txids up in batches of
14//!    `HTTP_JSON_RPC_BATCH_SIZE`, and `lookup_confirmation` serves each one
15//!    from that cache. The cache is cleared at the start of each pass, so an
16//!    unconfirmed transaction is looked up again next pass.
17//! 3. A transaction with at least `MIN_CONFIRMATIONS` confirmations is
18//!    returned with the time of its own block and the time of the block that
19//!    gave it `MIN_CONFIRMATIONS`.
20//! 4. Requests are paced by `HTTP_JSON_RPC_BATCH_DELAY`, and a rate-limited
21//!    request is retried with exponential backoff.
22//!
23//! Assumptions:
24//! - The endpoint serves `getrawtransaction` with verbose output, which
25//!   needs `txindex` or a hosted provider.
26//! - Only rate limits are retried. Other transient failures fail the pass
27//!   and are retried on the next tick.
28

103impl BlockchainInfo {
104    fn ensure_synced(&self) -> anyhow::Result<()> {
105        anyhow::ensure!(
106            !self.initialblockdownload && self.blocks >= self.headers,
107            "bitcoind is not synced: initialblockdownload={}, blocks={}, headers={}",
108            self.initialblockdownload,
109            self.blocks,
110            self.headers
111        );
112        Ok(())
113    }
114}

116impl RawTransactionInfo {
117    /// The block hash and outputs of a sufficiently confirmed transaction.
118    fn confirmed(self, txid: Txid) -> anyhow::Result<Option<(BlockHash, Vec<TxOut>)>> {
119        let Some(block_hash) = self.blockhash else {
120            debug!(%txid, "bitcoin tx found but not mined yet");
121            return Ok(None);
122        };
123        if self.confirmations < MIN_CONFIRMATIONS {
124            debug!(
125                %txid,
126                confirmations = self.confirmations,
127                required_confirmations = MIN_CONFIRMATIONS,
128                "bitcoin tx has insufficient confirmation depth"
129            );
130            return Ok(None);
131        }
132
133        let block_hash = BlockHash::from_str(&block_hash)
134            .with_context(|| format!("invalid bitcoin block hash for {txid}"))?;
135        let tx: Transaction = deserialize_hex(&self.hex)
136            .with_context(|| format!("invalid bitcoin transaction hex for {txid}"))?;
137        anyhow::ensure!(
138            tx.compute_txid() == txid,
139            "bitcoin returned transaction {} for {txid}",
140            tx.compute_txid()
141        );
142        Ok(Some((block_hash, tx.output)))
143    }
144}

146impl BtcRpcClient {
147    pub fn new(cfg: &Config) -> anyhow::Result<Self> {
148        Ok(Self {
149            transport: HttpJsonRpcTransport {
150                rpc_url: cfg.btc.rpc_url.clone(),
151                headers: cfg.btc.http_headers.clone(),
152            },
153            confirmation_cache: RefCell::new(HashMap::new()),
154        })
155    }
156
157    /// Fail while bitcoind lags its peers: a lagging node reports recent
158    /// transactions as unconfirmed, which reads as a missing Bitcoin event.
159    pub fn ensure_synced(&self) -> anyhow::Result<()> {
160        let response = self
161            .transport
162            .call("getblockchaininfo", serde_json::json!([]))?;
163        if let Some(error) = response.error {
164            anyhow::bail!(
165                "bitcoin getblockchaininfo failed: {} ({})",
166                error.message,
167                error.code
168            );
169        }
170        let info: BlockchainInfo = serde_json::from_value(
171            response
172                .result
173                .context("bitcoin getblockchaininfo response is missing its result")?,
174        )
175        .context("failed to parse bitcoin blockchain info")?;
176        info.ensure_synced()
177    }
178
179    /// Start a fresh lookup cycle. Confirmed and unconfirmed results are shared
180    /// by all outpoints from the same transaction within one audit tick, while
181    /// later ticks still retry transactions that were previously unconfirmed.
182    pub fn clear_confirmation_cache(&self) {
183        self.confirmation_cache.borrow_mut().clear();
184    }
185
186    pub fn prefetch_confirmations(&self, txids: &[Txid]) -> anyhow::Result<()> {
187        let confirmations = self.transport.lookup_confirmations(txids)?;
188        self.confirmation_cache.borrow_mut().extend(confirmations);
189        Ok(())
190    }
191
192    /// Query BTC RPC to check if a transaction is confirmed.
193    /// Returns
194    /// - `Ok(Some(confirmation))` if txid is confirmed,
195    /// - `Ok(None)` if txid is not seen or txid is seen but not confirmed,
196    /// - `Err(...)` for all other errors
197    pub fn lookup_confirmation(&self, txid: Txid) -> anyhow::Result<Option<BtcConfirmation>> {
198        if let Some(result) = self.confirmation_cache.borrow().get(&txid) {
199            return Ok(result.clone());
200        }
201        let result = self.transport.lookup_confirmation(txid)?;
202        self.confirmation_cache
203            .borrow_mut()
204            .insert(txid, result.clone());
205        Ok(result)
206    }
207}

209impl HttpJsonRpcTransport {
210    fn call(&self, method: &str, params: Value) -> anyhow::Result<JsonRpcEnvelope> {
211        let body = serde_json::json!({
212            "jsonrpc": "1.0",
213            "id": "hashi-monitor",
214            "method": method,
215            "params": params,
216        });
217        serde_json::from_value(self.post_json(&body)?)
218            .context("failed to decode bitcoin JSON-RPC response")
219    }
220
221    fn call_batch(&self, body: &[Value]) -> anyhow::Result<Vec<JsonRpcEnvelope>> {
222        let body = Value::Array(body.to_vec());
223        serde_json::from_value(self.post_json(&body)?)
224            .context("failed to decode bitcoin JSON-RPC batch response")
225    }
226
227    /// Pace requests so a historical audit does not exhaust a provider's
228    /// rolling throughput bucket, and retry the standard HTTP/JSON-RPC 429
229    /// responses with bounded exponential backoff.
230    // TODO: Classify and retry transient non-rate-limit failures, such as
231    // timeouts, connection errors, and HTTP 5xx responses. They currently
232    // abort a batch audit or defer a continuous retry until the next BTC tick.
233    fn post_json(&self, body: &Value) -> anyhow::Result<Value> {
234        for attempt in 0..=MAX_RATE_LIMIT_RETRIES {
235            let response = minreq::post(&self.rpc_url)
236                .with_timeout(60)
237                .with_headers(self.headers.clone())
238                .with_json(body)
239                .context("failed to encode bitcoin JSON-RPC request")?
240                .send()
241                .context("bitcoin JSON-RPC request failed")?;
242
243            if response.status_code == 429 {
244                Self::wait_before_rate_limit_retry(attempt)?;
245                continue;
246            }
247
248            let response_body: Value = response.json().with_context(|| {
249                format!(
250                    "failed to decode bitcoin JSON-RPC response (HTTP {})",
251                    response.status_code
252                )
253            })?;
254            if json_rpc_rate_limited(&response_body) {
255                Self::wait_before_rate_limit_retry(attempt)?;
256                continue;
257            }
258
259            thread::sleep(HTTP_JSON_RPC_BATCH_DELAY);
260            return Ok(response_body);
261        }
262
263        unreachable!("rate-limit retry loop either returns or errors")
264    }
265
266    fn wait_before_rate_limit_retry(attempt: usize) -> anyhow::Result<()> {
267        if attempt == MAX_RATE_LIMIT_RETRIES {
268            anyhow::bail!(
269                "bitcoin JSON-RPC remained rate limited after {MAX_RATE_LIMIT_RETRIES} retries"
270            );
271        }
272
273        let delay = Duration::from_secs(1 << attempt.min(5));
274        warn!(
275            retry = attempt + 1,
276            max_retries = MAX_RATE_LIMIT_RETRIES,
277            delay_secs = delay.as_secs(),
278            "bitcoin JSON-RPC rate limited; retrying"
279        );
280        thread::sleep(delay);
281        Ok(())
282    }

284    fn lookup_confirmation(&self, txid: Txid) -> anyhow::Result<Option<BtcConfirmation>> {
285        let response = self.call(
286            "getrawtransaction",
287            serde_json::json!([txid.to_string(), true]),
288        )?;
289        if let Some(error) = response.error {
290            if error.code == -5 {
291                debug!(%txid, "bitcoin tx not found in mempool or chain yet");
292                return Ok(None);
293            }
294            anyhow::bail!(
295                "bitcoin getrawtransaction failed for {txid}: {} ({})",
296                error.message,
297                error.code
298            );
299        }
300        let tx_info: RawTransactionInfo = serde_json::from_value(
301            response
302                .result
303                .context("bitcoin getrawtransaction response is missing its result")?,
304        )
305        .with_context(|| format!("failed to parse transaction info for {txid}"))?;
306        let Some((block_hash, outputs)) = tx_info.confirmed(txid)? else {
307            return Ok(None);
308        };
309
310        let times = self.confirmation_times(&[block_hash])?;
311        let (block_time, confirmed_time) = times
312            .get(&block_hash)
313            .copied()
314            .with_context(|| format!("missing block times for {block_hash}"))?;
315        Ok(Some(BtcConfirmation {
316            block_time,
317            confirmed_time,
318            outputs,
319        }))
320    }
321
322    fn lookup_confirmations(
323        &self,
324        txids: &[Txid],
325    ) -> anyhow::Result<HashMap<Txid, Option<BtcConfirmation>>> {
326        let mut confirmations = HashMap::with_capacity(txids.len());
327        for (batch_index, chunk) in txids.chunks(HTTP_JSON_RPC_BATCH_SIZE).enumerate() {
328            confirmations.extend(self.lookup_confirmation_batch(chunk)?);
329            let processed_txids = ((batch_index + 1) * HTTP_JSON_RPC_BATCH_SIZE).min(txids.len());
330            if processed_txids.is_multiple_of(200) || processed_txids == txids.len() {
331                info!(
332                    processed_txids,
333                    total_txids = txids.len(),
334                    "bitcoin confirmation lookup progress"
335                );
336            }
337        }
338        Ok(confirmations)
339    }
340
341    fn lookup_confirmation_batch(
342        &self,
343        txids: &[Txid],
344    ) -> anyhow::Result<HashMap<Txid, Option<BtcConfirmation>>> {
345        let requests = txids
346            .iter()
347            .enumerate()
348            .map(|(id, txid)| {
349                serde_json::json!({
350                    "jsonrpc": "2.0",
351                    "id": id,
352                    "method": "getrawtransaction",
353                    "params": [txid.to_string(), true],
354                })
355            })
356            .collect::<Vec<_>>();
357        let responses = self.call_batch(&requests)?;
358        let mut seen = HashSet::with_capacity(txids.len());
359        let mut confirmed = HashMap::with_capacity(txids.len());
360
361        for response in responses {
362            let index = response_index(&response, txids.len())?;
363            anyhow::ensure!(
364                seen.insert(index),
365                "duplicate bitcoin getrawtransaction batch response ID {index}"
366            );
367            let txid = txids[index];
368            if let Some(error) = response.error {
369                if error.code == -5 {
370                    confirmed.insert(txid, None);
371                    continue;
372                }
373                anyhow::bail!(
374                    "bitcoin getrawtransaction failed for {txid}: {} ({})",
375                    error.message,
376                    error.code
377                );
378            }
379            let info: RawTransactionInfo = serde_json::from_value(
380                response
381                    .result
382                    .context("bitcoin getrawtransaction batch response is missing its result")?,
383            )
384            .with_context(|| format!("failed to parse transaction info for {txid}"))?;
385            confirmed.insert(txid, info.confirmed(txid)?);
386        }
387        anyhow::ensure!(
388            seen.len() == txids.len(),
389            "bitcoin getrawtransaction batch returned {}/{} responses",
390            seen.len(),
391            txids.len()
392        );
393
394        let unique_block_hashes = confirmed
395            .values()
396            .flatten()
397            .map(|(block_hash, _)| *block_hash)
398            .collect::<HashSet<_>>()
399            .into_iter()
400            .collect::<Vec<_>>();
401        let times = self.confirmation_times(&unique_block_hashes)?;
402
403        confirmed
404            .into_iter()
405            .map(|(txid, confirmed)| {
406                let confirmation = confirmed
407                    .map(|(hash, outputs)| {
408                        let (block_time, confirmed_time) = times
409                            .get(&hash)
410                            .copied()
411                            .with_context(|| format!("missing block times for {hash}"))?;
412                        Ok::<_, anyhow::Error>(BtcConfirmation {
413                            block_time,
414                            confirmed_time,
415                            outputs,
416                        })
417                    })
418                    .transpose()?;
419                Ok((txid, confirmation))
420            })
421            .collect()
422    }

424    /// For each block that holds a confirmed transaction: its own time, and
425    /// the time of the block `MIN_CONFIRMATIONS - 1` above it. Core counts the
426    /// holding block as the first confirmation, so that block is the one that
427    /// gave the transaction `MIN_CONFIRMATIONS`.
428    fn confirmation_times(
429        &self,
430        block_hashes: &[BlockHash],
431    ) -> anyhow::Result<HashMap<BlockHash, (UnixSeconds, UnixSeconds)>> {
432        let headers = self.block_headers(block_hashes)?;
433        let confirming_height = |header: &RawBlockHeader| header.height + MIN_CONFIRMATIONS - 1;
434        let confirming_heights = headers
435            .values()
436            .map(confirming_height)
437            .collect::<HashSet<_>>()
438            .into_iter()
439            .collect::<Vec<_>>();
440        let confirming_hashes = self.block_hashes(&confirming_heights)?;
441        let confirming_headers =
442            self.block_headers(&confirming_hashes.values().copied().collect::<Vec<_>>())?;
443
444        block_hashes
445            .iter()
446            .map(|hash| {
447                let header = headers
448                    .get(hash)
449                    .with_context(|| format!("missing block header for {hash}"))?;
450                let height = confirming_height(header);
451                let confirming_hash = confirming_hashes
452                    .get(&height)
453                    .with_context(|| format!("missing block hash at height {height}"))?;
454                let confirming = confirming_headers
455                    .get(confirming_hash)
456                    .with_context(|| format!("missing block header for {confirming_hash}"))?;
457                Ok((*hash, (header.time, confirming.time)))
458            })
459            .collect()
460    }

462    /// `getblockheader` for each hash, in batches.
463    fn block_headers(
464        &self,
465        hashes: &[BlockHash],
466    ) -> anyhow::Result<HashMap<BlockHash, RawBlockHeader>> {
467        let mut headers = HashMap::with_capacity(hashes.len());
468        for chunk in hashes.chunks(HTTP_JSON_RPC_BATCH_SIZE) {
469            let requests = chunk
470                .iter()
471                .enumerate()
472                .map(|(id, hash)| {
473                    serde_json::json!({
474                        "jsonrpc": "2.0",
475                        "id": id,
476                        "method": "getblockheader",
477                        "params": [hash.to_string(), true],
478                    })
479                })
480                .collect::<Vec<_>>();
481            let responses = self.call_batch(&requests)?;
482            let mut seen = HashSet::with_capacity(chunk.len());
483            for response in responses {
484                let index = response_index(&response, chunk.len())?;
485                anyhow::ensure!(
486                    seen.insert(index),
487                    "duplicate bitcoin getblockheader batch response ID {index}"
488                );
489                let block_hash = chunk[index];
490                if let Some(error) = response.error {
491                    anyhow::bail!(
492                        "bitcoin getblockheader failed for {block_hash}: {} ({})",
493                        error.message,
494                        error.code
495                    );
496                }
497                let header: RawBlockHeader = serde_json::from_value(
498                    response
499                        .result
500                        .context("bitcoin getblockheader batch response is missing its result")?,
501                )
502                .with_context(|| format!("failed to parse block header for {block_hash}"))?;
503                headers.insert(block_hash, header);
504            }
505            anyhow::ensure!(
506                seen.len() == chunk.len(),
507                "bitcoin getblockheader batch returned {}/{} responses",
508                seen.len(),
509                chunk.len()
510            );
511        }
512        Ok(headers)
513    }

515    /// `getblockhash` for each height, in batches.
516    fn block_hashes(&self, heights: &[u64]) -> anyhow::Result<HashMap<u64, BlockHash>> {
517        let mut hashes = HashMap::with_capacity(heights.len());
518        for chunk in heights.chunks(HTTP_JSON_RPC_BATCH_SIZE) {
519            let requests = chunk
520                .iter()
521                .enumerate()
522                .map(|(id, height)| {
523                    serde_json::json!({
524                        "jsonrpc": "2.0",
525                        "id": id,
526                        "method": "getblockhash",
527                        "params": [height],
528                    })
529                })
530                .collect::<Vec<_>>();
531            let responses = self.call_batch(&requests)?;
532            let mut seen = HashSet::with_capacity(chunk.len());
533            for response in responses {
534                let index = response_index(&response, chunk.len())?;
535                anyhow::ensure!(
536                    seen.insert(index),
537                    "duplicate bitcoin getblockhash batch response ID {index}"
538                );
539                let height = chunk[index];
540                if let Some(error) = response.error {
541                    anyhow::bail!(
542                        "bitcoin getblockhash failed for height {height}: {} ({})",
543                        error.message,
544                        error.code
545                    );
546                }
547                let hash: String = serde_json::from_value(
548                    response
549                        .result
550                        .context("bitcoin getblockhash batch response is missing its result")?,
551                )
552                .with_context(|| format!("failed to parse block hash at height {height}"))?;
553                let hash = BlockHash::from_str(&hash)
554                    .with_context(|| format!("invalid bitcoin block hash at height {height}"))?;
555                hashes.insert(height, hash);
556            }
557            anyhow::ensure!(
558                seen.len() == chunk.len(),
559                "bitcoin getblockhash batch returned {}/{} responses",
560                seen.len(),
561                chunk.len()
562            );
563        }
564        Ok(hashes)
565    }

568fn json_rpc_rate_limited(response: &Value) -> bool {
569    let is_rate_limited = |envelope: &Value| {
570        envelope
571            .get("error")
572            .and_then(|error| error.get("code"))
573            .and_then(Value::as_i64)
574            == Some(429)
575    };
576
577    match response {
578        Value::Array(envelopes) => envelopes.iter().any(is_rate_limited),
579        envelope => is_rate_limited(envelope),
580    }
581}
582
583fn response_index(response: &JsonRpcEnvelope, batch_len: usize) -> anyhow::Result<usize> {
584    let index = response
585        .id
586        .as_u64()
587        .context("bitcoin JSON-RPC batch response ID is not an integer")?;
588    let index =
589        usize::try_from(index).context("bitcoin JSON-RPC batch response ID is too large")?;
590    anyhow::ensure!(
591        index < batch_len,
592        "bitcoin JSON-RPC batch response ID {index} is outside batch of {batch_len}"
593    );
594    Ok(index)
595}

685        let insufficiently_confirmed = btc_rpc_client.lookup_confirmation(txid)?;
686        assert!(
687            insufficiently_confirmed.is_none(),
688            "expected one-confirmation transaction to remain pending"
689        );
690
691        node.generate_blocks(MIN_CONFIRMATIONS - 1)?;
692        btc_rpc_client.clear_confirmation_cache();
693
694        let confirmed = btc_rpc_client
695            .lookup_confirmation(txid)?
696            .expect("expected confirmed transaction");
697        assert!(
698            confirmed.outputs.iter().any(|output| {
699                output.value == Amount::from_sat(50_000)
700                    && output.script_pubkey == destination.script_pubkey()
701            }),
702            "expected the paid output among {:?}",
703            confirmed.outputs
704        );
705
706        // The tip is exactly the block that gave the transaction its
707        // confirmations, so its header time is the confirmed time.
708        let tip_height = node.get_block_count()?;
709        let tip_hash: String = node
710            .rpc_client()
711            .call("getblockhash", &[serde_json::json!(tip_height)])?;
712        let tip_header: serde_json::Value = node.rpc_client().call(
713            "getblockheader",
714            &[serde_json::json!(tip_hash), serde_json::json!(true)],
715        )?;
716        assert_eq!(
717            confirmed.confirmed_time,
718            tip_header["time"].as_u64().unwrap()
719        );
720        assert!(confirmed.block_time <= confirmed.confirmed_time);
721
722        Ok(())
723    }

747    #[test]
748    fn undecodable_response_error_names_the_http_status() {
749        let listener = TcpListener::bind("127.0.0.1:0").unwrap();
750        let rpc_url = format!("http://{}", listener.local_addr().unwrap());
751        let server = std::thread::spawn(move || {
752            let (stream, _) = listener.accept().unwrap();
753            let mut reader = BufReader::new(stream);
754            let mut content_length = 0;
755            loop {
756                let mut line = String::new();
757                reader.read_line(&mut line).unwrap();
758                if line == "\r\n" {
759                    break;
760                }
761                let line = line.to_ascii_lowercase();
762                if let Some(value) = line.strip_prefix("content-length:") {
763                    content_length = value.trim().parse().unwrap();
764                }
765            }
766            reader.read_exact(&mut vec![0; content_length]).unwrap();
767            reader
768                .into_inner()
769                .write_all(b"HTTP/1.1 401 Unauthorized\r\nContent-Length: 0\r\n\r\n")
770                .unwrap();
771        });
772        let transport = HttpJsonRpcTransport {
773            rpc_url,
774            headers: BTreeMap::new(),
775        };
776
777        let error = transport
778            .call("getblockchaininfo", serde_json::json!([]))
779            .err()
780            .expect("an empty 401 body is not JSON");
781
782        server.join().unwrap();
783        assert!(format!("{error:#}").contains("HTTP 401"), "{error:#}");
784    }
