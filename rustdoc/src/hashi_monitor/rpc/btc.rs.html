4//! The Bitcoin client: confirmation lookups over HTTP JSON-RPC.
5//!
6//! Why it exists: the withdrawal E3 and the deposit E1 are Bitcoin
7//! confirmations. Bitcoin has no event stream to scan, so the auditor looks
8//! each pending transaction up by txid.
9//!
10//! Why one height:
11//! - A parent transaction creates an output that a dependent transaction spends.
12//! - At one height, the dependent cannot have more confirmations than its parent.
13//! - All lookups in a tick use one captured height, including lookups in different batches.
14//! - New blocks between RPC responses must not complete a dependent machine before its parent is ready to close.
15//! - Pool maintenance needs each complete flow's parent machine already closed or ready to close.
16//!
17//! How it works:
18//! 1. At startup, the auditor reads Hashi's on-chain
19//!    `bitcoin_confirmation_threshold` and gives it to the client. A
20//!    transaction counts as confirmed at that depth, as it does for the
21//!    committee. On mainnet, a threshold below `MAINNET_MIN_CONFIRMATIONS`
22//!    stops the monitor.
23//! 2. Before a pass, `synced_height` checks the node and returns its height.
24//!    The node must follow `deployment.bitcoin_network`, be out of initial
25//!    block download, and have a tip no older than `max_tip_age`, which the
26//!    auditor sets to its `SOURCE_STALE_LIMIT`. At the first check, a node
27//!    that reports its indexes must have a synced `txindex`. A failed check
28//!    fails the pass, so a wrong node, or one with an older tip, becomes a
29//!    `SourceStale` finding. Until its tip passes that age, a frozen node
30//!    still gives lookup findings.
31//!    All batches use this height to calculate confirmation depth.
32//!    Blocks that arrive during the pass count on the next pass.
33//! 3. `prefetch_confirmations` looks all pending txids up in batches of
34//!    `HTTP_JSON_RPC_BATCH_SIZE`, with `HTTP_JSON_RPC_BATCHES_IN_FLIGHT`
35//!    batches sent at once, and `lookup_confirmation` serves each one from
36//!    that cache. A confirmed answer is kept across passes: a confirmation
37//!    at the threshold depth does not change. `forget_unconfirmed` drops
38//!    the rest at the start of each pass, so an unconfirmed transaction is
39//!    looked up again. An answer for a txid no longer asked for is dropped.
40//!    When a lookup fails, the batches that finished are kept, so the next
41//!    pass asks only about the rest.
42//! 4. A transaction at the threshold depth is returned with the time of
43//!    its own block and the time of the block that gave it that depth.
44//! 5. Requests are not paced: the node is the bridge's own. A request that
45//!    fails to send, gets an HTTP 502, 503 or 504, or is rate limited is
46//!    retried with exponential backoff, `MAX_REQUEST_RETRIES` times. Core
47//!    answers a JSON-RPC error with HTTP 500, so 500 is not retried.
48//!
49//! Assumptions:
50//! - The threshold is read once. A governance change takes effect in the
51//!   monitor at its next restart.
52//! - An unconfirmed answer is never reused across passes. So the auditor
53//!   can stamp every answer with the pass's start time: the real lookup
54//!   is no earlier. A cached confirmed answer does not need a time.
55//! - The endpoint serves `getrawtransaction` with verbose output, which
56//!   needs `txindex` or a hosted provider. A hosted provider that does not
57//!   serve `getindexinfo` is not checked for an index.
58//! - A request that still fails after the retries fails the pass, which
59//!   is retried on the next tick.

93/// Calls per JSON-RPC batch request.
94const HTTP_JSON_RPC_BATCH_SIZE: usize = 100;
95/// Confirmation batches sent at the same time. A probe of the testnet node
96/// (2026-10-19) served eight batches of 100 at once fastest, with no errors.
97const HTTP_JSON_RPC_BATCHES_IN_FLIGHT: usize = 8;
98/// Txids between two progress log lines of a confirmation lookup.
99const LOOKUP_PROGRESS_EVERY: usize = 10_000;
100/// Retries of one request. The delays double from one second, so the
101/// last retry comes about a minute after the first failure.
102const MAX_REQUEST_RETRIES: usize = 6;
103/// The lowest confirmation threshold the monitor accepts on Bitcoin
104/// mainnet. Hashi's launch tool refuses a lower one there too.
105pub const MAINNET_MIN_CONFIRMATIONS: u64 = 6;
106
107/// The on-chain confirmation threshold, refused on mainnet below
108/// `MAINNET_MIN_CONFIRMATIONS`.
109pub fn checked_confirmation_threshold(threshold: u64, network: Network) -> anyhow::Result<u64> {
110    anyhow::ensure!(
111        network != Network::Bitcoin || threshold >= MAINNET_MIN_CONFIRMATIONS,
112        "Hashi's bitcoin_confirmation_threshold is {threshold} on mainnet; \
113         the monitor needs at least {MAINNET_MIN_CONFIRMATIONS}"
114    );
115    Ok(threshold)
116}

118/// What Bitcoin holds at an outpoint. `Missing` and `Spent` differ by the
119/// transaction lookup, `Spent` and `Unspent` by the UTXO set.
120#[derive(Clone, Debug, PartialEq, Eq)]
121pub enum OutpointState {
122    /// The transaction is unknown or not confirmed deep enough, or it has no
123    /// output at that vout.
124    Missing,
125    /// The output exists, and a confirmed transaction has spent it. With
126    /// `include_mempool`, a mempool transaction also counts.
127    Spent,
128    /// The output exists and is in the UTXO set. Without `include_mempool`,
129    /// a mempool spend does not change this.
130    Unspent {
131        /// The time of the block that holds the output.
132        block_time: UnixSeconds,
133    },
134}

182#[derive(Deserialize)]
183struct BlockchainInfo {
184    /// The chain name, as in `bitcoind -chain`.
185    chain: String,
186    blocks: u64,
187    headers: u64,
188    /// The time of the tip block.
189    time: UnixSeconds,
190    initialblockdownload: bool,
191}
192
193impl BlockchainInfo {
194    /// Fail unless the node follows `network` and is synced at `now`, with
195    /// a tip at most `max_tip_age` old. Core clears `initialblockdownload`
196    /// once and never sets it again, so the tip time also catches a node
197    /// that stopped getting blocks.
198    fn ensure_synced(
199        &self,
200        network: Network,
201        max_tip_age: u64,
202        now: UnixSeconds,
203    ) -> anyhow::Result<()> {
204        anyhow::ensure!(
205            self.chain == network.to_core_arg(),
206            "bitcoind follows chain {}, not {}",
207            self.chain,
208            network.to_core_arg()
209        );
210        anyhow::ensure!(
211            !self.initialblockdownload && self.blocks >= self.headers,
212            "bitcoind is not synced: initialblockdownload={}, blocks={}, headers={}",
213            self.initialblockdownload,
214            self.blocks,
215            self.headers
216        );
217        anyhow::ensure!(
218            now.saturating_sub(self.time) <= max_tip_age,
219            "bitcoind tip at height {} is from {}, more than {max_tip_age} s ago",
220            self.blocks,
221            utc_timestamp(self.time)
222        );
223        Ok(())
224    }
225}

227/// Fail unless a `getindexinfo` result shows a synced `txindex`.
228fn ensure_txindex(index_info: &Value) -> anyhow::Result<()> {
229    let synced = index_info
230        .get("txindex")
231        .context("bitcoind has no txindex; getrawtransaction needs one")?
232        .get("synced")
233        .and_then(Value::as_bool)
234        .context("bitcoind getindexinfo has no txindex.synced")?;
235    anyhow::ensure!(synced, "bitcoind txindex is still building");
236    Ok(())
237}

239/// The answers of a confirmation lookup, and the first error if a batch
240/// failed. The answers of the batches that finished are kept either way.
241struct LookupOutcome {
242    confirmations: HashMap<Txid, Option<BtcConfirmation>>,
243    error: Option<anyhow::Error>,
244}

246/// A mined transaction before its block header is read.
247struct MinedTx {
248    block_hash: BlockHash,
249    inputs: Vec<OutPoint>,
250    outputs: Vec<TxOut>,
251}
252
253impl RawTransactionInfo {
254    /// The block hash, inputs, and outputs of a mined transaction.
255    fn mined(self, txid: Txid) -> anyhow::Result<Option<MinedTx>> {
256        let Some(block_hash) = self.blockhash else {
257            debug!(%txid, "bitcoin tx found but not mined yet");
258            return Ok(None);
259        };
260        if self.confirmations == 0 {
261            return Ok(None);
262        }
263
264        let block_hash = BlockHash::from_str(&block_hash)
265            .with_context(|| format!("invalid bitcoin block hash for {txid}"))?;
266        let tx: Transaction = deserialize_hex(&self.hex)
267            .with_context(|| format!("invalid bitcoin transaction hex for {txid}"))?;
268        anyhow::ensure!(
269            tx.compute_txid() == txid,
270            "bitcoin returned transaction {} for {txid}",
271            tx.compute_txid()
272        );
273        Ok(Some(MinedTx {
274            block_hash,
275            inputs: tx.input.iter().map(|input| input.previous_output).collect(),
276            outputs: tx.output,
277        }))
278    }
279}

281impl BtcRpcClient {
282    /// A client that counts a transaction as confirmed at
283    /// `required_confirmations`, and a node as synced with a tip at most
284    /// `max_tip_age` seconds old.
285    pub fn new(
286        cfg: &Config,
287        required_confirmations: u64,
288        max_tip_age: u64,
289    ) -> anyhow::Result<Self> {
290        Ok(Self {
291            transport: HttpJsonRpcTransport {
292                rpc_url: cfg.btc.rpc_url.clone(),
293                headers: cfg.btc.http_headers.clone(),
294                required_confirmations,
295            },
296            confirmation_cache: RefCell::new(HashMap::new()),
297            network: cfg.deployment.bitcoin_network,
298            max_tip_age,
299            txindex_checked: Cell::new(false),
300        })
301    }
302
303    /// Return the node height after the synchronization check succeeds.
304    /// See the module doc for the checks.
305    pub fn synced_height(&self) -> anyhow::Result<u64> {
306        let response = self
307            .transport
308            .call("getblockchaininfo", serde_json::json!([]))?;
309        if let Some(error) = response.error {
310            anyhow::bail!(
311                "bitcoin getblockchaininfo failed: {} ({})",
312                error.message,
313                error.code
314            );
315        }
316        let info: BlockchainInfo = serde_json::from_value(
317            response
318                .result
319                .context("bitcoin getblockchaininfo response is missing its result")?,
320        )
321        .context("failed to parse bitcoin blockchain info")?;
322        info.ensure_synced(self.network, self.max_tip_age, now_timestamp_secs())?;
323        if !self.txindex_checked.get() {
324            self.check_txindex()?;
325            self.txindex_checked.set(true);
326        }
327        Ok(info.blocks)
328    }
329
330    /// Fail unless the node has a synced `txindex`. A node that does not
331    /// serve `getindexinfo`, such as a hosted provider, passes.
332    fn check_txindex(&self) -> anyhow::Result<()> {
333        let response = self.transport.call("getindexinfo", serde_json::json!([]))?;
334        if let Some(error) = response.error {
335            // Method not found.
336            if error.code == -32601 {
337                info!("bitcoin endpoint does not serve getindexinfo; txindex not checked");
338                return Ok(());
339            }
340            anyhow::bail!(
341                "bitcoin getindexinfo failed: {} ({})",
342                error.message,
343                error.code
344            );
345        }
346        ensure_txindex(
347            &response
348                .result
349                .context("bitcoin getindexinfo response is missing its result")?,
350        )
351    }

353    /// Drop the unconfirmed answers, so the next lookup asks again. The
354    /// confirmed ones stay: a confirmation at the threshold depth does not
355    /// change.
356    pub fn forget_unconfirmed(&self) {
357        self.confirmation_cache
358            .borrow_mut()
359            .retain(|_, confirmation| confirmation.is_some());
360    }
361
362    /// Fetch missing confirmations at `tip_height` and remove answers absent from `txids`.
363    /// Keep completed batches if another batch fails.
364    pub fn prefetch_confirmations(&self, txids: &[Txid], tip_height: u64) -> anyhow::Result<()> {
365        let asked = txids.iter().copied().collect::<HashSet<_>>();
366        let missing = {
367            let cache = self.confirmation_cache.borrow();
368            txids
369                .iter()
370                .filter(|txid| !matches!(cache.get(*txid), Some(Some(_))))
371                .copied()
372                .collect::<Vec<_>>()
373        };
374        info!(
375            cached = txids.len() - missing.len(),
376            looked_up = missing.len(),
377            "bitcoin confirmations served from the cache"
378        );
379        let outcome = self.transport.lookup_confirmations(&missing, tip_height);
380        let mut cache = self.confirmation_cache.borrow_mut();
381        cache.extend(outcome.confirmations);
382        cache.retain(|txid, _| asked.contains(txid));
383        outcome.error.map_or(Ok(()), Err)
384    }
385
386    /// Query BTC RPC to check if a transaction is confirmed.
387    /// Returns
388    /// - `Ok(Some(confirmation))` if txid is confirmed,
389    /// - `Ok(None)` if txid is not seen or txid is seen but not confirmed,
390    /// - `Err(...)` for all other errors
391    pub fn lookup_confirmation(&self, txid: Txid) -> anyhow::Result<Option<BtcConfirmation>> {
392        if let Some(result) = self.confirmation_cache.borrow().get(&txid) {
393            return Ok(result.clone());
394        }
395        let tip_height = self.synced_height()?;
396        let result = self.transport.lookup_confirmation(txid, tip_height)?;
397        self.confirmation_cache
398            .borrow_mut()
399            .insert(txid, result.clone());
400        Ok(result)
401    }

403    /// The state of each outpoint on Bitcoin. The transactions are fetched
404    /// fresh, and the confirmation cache is not touched. `include_mempool`
405    /// also counts a spend in the mempool.
406    pub fn lookup_outpoint_states(
407        &self,
408        outpoints: &[OutPoint],
409        tip_height: u64,
410        include_mempool: bool,
411    ) -> anyhow::Result<HashMap<OutPoint, OutpointState>> {
412        let txids = outpoints
413            .iter()
414            .map(|outpoint| outpoint.txid)
415            .collect::<HashSet<_>>()
416            .into_iter()
417            .collect::<Vec<_>>();
418        let outcome = self.transport.lookup_confirmations(&txids, tip_height);
419        if let Some(error) = outcome.error {
420            return Err(error);
421        }
422
423        // Only an output that exists can be in the UTXO set.
424        let mut states = HashMap::with_capacity(outpoints.len());
425        let mut existing = Vec::new();
426        for outpoint in outpoints {
427            let confirmation = outcome
428                .confirmations
429                .get(&outpoint.txid)
430                .with_context(|| format!("missing bitcoin lookup result for {}", outpoint.txid))?;
431            match confirmation {
432                Some(confirmation) if (outpoint.vout as usize) < confirmation.outputs.len() => {
433                    existing.push((*outpoint, confirmation.block_time));
434                }
435                _ => {
436                    states.insert(*outpoint, OutpointState::Missing);
437                }
438            }
439        }
440
441        let existing_outpoints = existing
442            .iter()
443            .map(|(outpoint, _)| *outpoint)
444            .collect::<Vec<_>>();
445        let unspent = self
446            .transport
447            .lookup_unspent(&existing_outpoints, include_mempool)?;
448        for (outpoint, block_time) in existing {
449            let state = match unspent.get(&outpoint) {
450                Some(true) => OutpointState::Unspent { block_time },
451                Some(false) => OutpointState::Spent,
452                None => anyhow::bail!("missing bitcoin gettxout result for {outpoint}"),
453            };
454            states.insert(outpoint, state);
455        }
456        Ok(states)
457    }
458}

467impl HttpJsonRpcTransport {
468    fn call(&self, method: &str, params: Value) -> anyhow::Result<JsonRpcEnvelope> {
469        let body = serde_json::json!({
470            "jsonrpc": "1.0",
471            "id": "hashi-monitor",
472            "method": method,
473            "params": params,
474        });
475        serde_json::from_value(self.post_json(&body)?)
476            .context("failed to decode bitcoin JSON-RPC response")
477    }
478
479    fn call_batch(&self, body: &[Value]) -> anyhow::Result<Vec<JsonRpcEnvelope>> {
480        let body = Value::Array(body.to_vec());
481        serde_json::from_value(self.post_json(&body)?)
482            .context("failed to decode bitcoin JSON-RPC batch response")
483    }
484
485    /// Send one request. A send that fails, a timeout among them, an HTTP
486    /// 502, 503 or 504 and a rate limit are retried with bounded exponential
487    /// backoff. Core sends 503 when its work queue is full. Any other status
488    /// goes to the JSON-RPC decoding: Core answers an RPC error with 500.
489    fn post_json(&self, body: &Value) -> anyhow::Result<Value> {
490        for attempt in 0..=MAX_REQUEST_RETRIES {
491            let request = minreq::post(&self.rpc_url)
492                .with_timeout(60)
493                .with_headers(self.headers.clone())
494                .with_json(body)
495                .context("failed to encode bitcoin JSON-RPC request")?;
496            let response = match request.send() {
497                Ok(response) => response,
498                Err(error) => {
499                    Self::wait_before_retry(attempt, &format!("request failed: {error}"))?;
500                    continue;
501                }
502            };
503
504            if matches!(response.status_code, 429 | 502 | 503 | 504) {
505                Self::wait_before_retry(attempt, &format!("HTTP {}", response.status_code))?;
506                continue;
507            }
508
509            let response_body: Value = response.json().with_context(|| {
510                format!(
511                    "failed to decode bitcoin JSON-RPC response (HTTP {})",
512                    response.status_code
513                )
514            })?;
515            if json_rpc_rate_limited(&response_body) {
516                Self::wait_before_retry(attempt, "rate limited")?;
517                continue;
518            }
519
520            return Ok(response_body);
521        }
522
523        unreachable!("the retry loop either returns or errors")
524    }
525
526    fn wait_before_retry(attempt: usize, reason: &str) -> anyhow::Result<()> {
527        if attempt == MAX_REQUEST_RETRIES {
528            anyhow::bail!("bitcoin JSON-RPC {reason} after {MAX_REQUEST_RETRIES} retries");
529        }
530
531        let delay = Duration::from_secs(1 << attempt.min(5));
532        warn!(
533            retry = attempt + 1,
534            max_retries = MAX_REQUEST_RETRIES,
535            delay_secs = delay.as_secs(),
536            reason,
537            "bitcoin JSON-RPC request will be retried"
538        );
539        thread::sleep(delay);
540        Ok(())
541    }
542
543    fn lookup_confirmation(
544        &self,
545        txid: Txid,
546        tip_height: u64,
547    ) -> anyhow::Result<Option<BtcConfirmation>> {
548        self.lookup_confirmation_batch(&[txid], tip_height)?
549            .remove(&txid)
550            .with_context(|| format!("missing bitcoin lookup result for {txid}"))
551    }
552
553    /// Look every txid up, `HTTP_JSON_RPC_BATCHES_IN_FLIGHT` batches at a
554    /// time. Each worker takes the next batch until none is left. The first
555    /// failure stops the others at their next batch. The answers of the
556    /// batches that finished are returned with that error.
557    fn lookup_confirmations(&self, txids: &[Txid], tip_height: u64) -> LookupOutcome {
558        let batches = txids.chunks(HTTP_JSON_RPC_BATCH_SIZE).collect::<Vec<_>>();
559        let next_batch = AtomicUsize::new(0);
560        let processed = AtomicUsize::new(0);
561        let results = Mutex::new(Vec::with_capacity(batches.len()));
562        thread::scope(|scope| {
563            for _ in 0..HTTP_JSON_RPC_BATCHES_IN_FLIGHT.min(batches.len()) {
564                scope.spawn(|| {
565                    while let Some(batch) = batches.get(next_batch.fetch_add(1, Ordering::Relaxed))
566                    {
567                        let result = self.lookup_confirmation_batch(batch, tip_height);
568                        if result.is_err() {
569                            next_batch.store(batches.len(), Ordering::Relaxed);
570                        }
571                        let processed_txids =
572                            processed.fetch_add(batch.len(), Ordering::Relaxed) + batch.len();
573                        if processed_txids.is_multiple_of(LOOKUP_PROGRESS_EVERY)
574                            || processed_txids == txids.len()
575                        {
576                            info!(
577                                processed_txids,
578                                total_txids = txids.len(),
579                                "bitcoin confirmation lookup progress"
580                            );
581                        }
582                        results.lock().expect("a worker never panics").push(result);
583                    }
584                });
585            }
586        });
587        let mut outcome = LookupOutcome {
588            confirmations: HashMap::with_capacity(txids.len()),
589            error: None,
590        };
591        for result in results.into_inner().expect("a worker never panics") {
592            match result {
593                Ok(confirmations) => outcome.confirmations.extend(confirmations),
594                Err(error) => {
595                    outcome.error.get_or_insert(error);
596                }
597            }
598        }
599        outcome
600    }
601
602    fn lookup_confirmation_batch(
603        &self,
604        txids: &[Txid],
605        tip_height: u64,
606    ) -> anyhow::Result<HashMap<Txid, Option<BtcConfirmation>>> {
607        let requests = txids
608            .iter()
609            .enumerate()
610            .map(|(id, txid)| {
611                serde_json::json!({
612                    "jsonrpc": "2.0",
613                    "id": id,
614                    "method": "getrawtransaction",
615                    "params": [txid.to_string(), true],
616                })
617            })
618            .collect::<Vec<_>>();
619        let responses = self.call_batch(&requests)?;
620        let mut seen = HashSet::with_capacity(txids.len());
621        let mut mined = HashMap::with_capacity(txids.len());
622
623        for response in responses {
624            let index = response_index(&response, txids.len())?;
625            anyhow::ensure!(
626                seen.insert(index),
627                "duplicate bitcoin getrawtransaction batch response ID {index}"
628            );
629            let txid = txids[index];
630            if let Some(error) = response.error {
631                if error.code == -5 {
632                    mined.insert(txid, None);
633                    continue;
634                }
635                anyhow::bail!(
636                    "bitcoin getrawtransaction failed for {txid}: {} ({})",
637                    error.message,
638                    error.code
639                );
640            }
641            let info: RawTransactionInfo = serde_json::from_value(
642                response
643                    .result
644                    .context("bitcoin getrawtransaction batch response is missing its result")?,
645            )
646            .with_context(|| format!("failed to parse transaction info for {txid}"))?;
647            mined.insert(txid, info.mined(txid)?);
648        }
649        anyhow::ensure!(
650            seen.len() == txids.len(),
651            "bitcoin getrawtransaction batch returned {}/{} responses",
652            seen.len(),
653            txids.len()
654        );
655
656        let mined_block_hashes = mined
657            .values()
658            .flatten()
659            .map(|tx| tx.block_hash)
660            .collect::<HashSet<_>>()
661            .into_iter()
662            .collect::<Vec<_>>();
663        let confirmed_block_times = self.confirmed_block_times(&mined_block_hashes, tip_height)?;
664
665        Ok(mined
666            .into_iter()
667            .map(|(txid, transaction)| {
668                let confirmation = transaction.and_then(|tx| {
669                    confirmed_block_times.get(&tx.block_hash).map(
670                        |&(block_time, confirmed_time)| BtcConfirmation {
671                            block_time,
672                            confirmed_time,
673                            inputs: tx.inputs,
674                            outputs: tx.outputs,
675                        },
676                    )
677                });
678                (txid, confirmation)
679            })
680            .collect())
681    }

683    /// Return block and confirmation times for blocks deep enough at `tip_height`.
684    /// The holding block counts as the first confirmation.
685    fn confirmed_block_times(
686        &self,
687        block_hashes: &[BlockHash],
688        tip_height: u64,
689    ) -> anyhow::Result<HashMap<BlockHash, (UnixSeconds, UnixSeconds)>> {
690        let mut headers = self.block_headers(block_hashes)?;
691        let confirming_height =
692            |header: &RawBlockHeader| header.height + self.required_confirmations - 1;
693        headers.retain(|_, header| confirming_height(header) <= tip_height);
694        let confirming_heights = headers
695            .values()
696            .map(confirming_height)
697            .collect::<HashSet<_>>()
698            .into_iter()
699            .collect::<Vec<_>>();
700        let confirming_hashes = self.block_hashes(&confirming_heights)?;
701        let confirming_headers =
702            self.block_headers(&confirming_hashes.values().copied().collect::<Vec<_>>())?;
703
704        headers
705            .iter()
706            .map(|(hash, header)| {
707                let height = confirming_height(header);
708                let confirming_hash = confirming_hashes
709                    .get(&height)
710                    .with_context(|| format!("missing block hash at height {height}"))?;
711                let confirming = confirming_headers
712                    .get(confirming_hash)
713                    .with_context(|| format!("missing block header for {confirming_hash}"))?;
714                Ok((*hash, (header.time, confirming.time)))
715            })
716            .collect()
717    }

719    /// `getblockheader` for each hash, in batches.
720    fn block_headers(
721        &self,
722        hashes: &[BlockHash],
723    ) -> anyhow::Result<HashMap<BlockHash, RawBlockHeader>> {
724        let mut headers = HashMap::with_capacity(hashes.len());
725        for chunk in hashes.chunks(HTTP_JSON_RPC_BATCH_SIZE) {
726            let requests = chunk
727                .iter()
728                .enumerate()
729                .map(|(id, hash)| {
730                    serde_json::json!({
731                        "jsonrpc": "2.0",
732                        "id": id,
733                        "method": "getblockheader",
734                        "params": [hash.to_string(), true],
735                    })
736                })
737                .collect::<Vec<_>>();
738            let responses = self.call_batch(&requests)?;
739            let mut seen = HashSet::with_capacity(chunk.len());
740            for response in responses {
741                let index = response_index(&response, chunk.len())?;
742                anyhow::ensure!(
743                    seen.insert(index),
744                    "duplicate bitcoin getblockheader batch response ID {index}"
745                );
746                let block_hash = chunk[index];
747                if let Some(error) = response.error {
748                    anyhow::bail!(
749                        "bitcoin getblockheader failed for {block_hash}: {} ({})",
750                        error.message,
751                        error.code
752                    );
753                }
754                let header: RawBlockHeader = serde_json::from_value(
755                    response
756                        .result
757                        .context("bitcoin getblockheader batch response is missing its result")?,
758                )
759                .with_context(|| format!("failed to parse block header for {block_hash}"))?;
760                headers.insert(block_hash, header);
761            }
762            anyhow::ensure!(
763                seen.len() == chunk.len(),
764                "bitcoin getblockheader batch returned {}/{} responses",
765                seen.len(),
766                chunk.len()
767            );
768        }
769        Ok(headers)
770    }

772    /// `getblockhash` for each height, in batches.
773    fn block_hashes(&self, heights: &[u64]) -> anyhow::Result<HashMap<u64, BlockHash>> {
774        let mut hashes = HashMap::with_capacity(heights.len());
775        for chunk in heights.chunks(HTTP_JSON_RPC_BATCH_SIZE) {
776            let requests = chunk
777                .iter()
778                .enumerate()
779                .map(|(id, height)| {
780                    serde_json::json!({
781                        "jsonrpc": "2.0",
782                        "id": id,
783                        "method": "getblockhash",
784                        "params": [height],
785                    })
786                })
787                .collect::<Vec<_>>();
788            let responses = self.call_batch(&requests)?;
789            let mut seen = HashSet::with_capacity(chunk.len());
790            for response in responses {
791                let index = response_index(&response, chunk.len())?;
792                anyhow::ensure!(
793                    seen.insert(index),
794                    "duplicate bitcoin getblockhash batch response ID {index}"
795                );
796                let height = chunk[index];
797                if let Some(error) = response.error {
798                    anyhow::bail!(
799                        "bitcoin getblockhash failed for height {height}: {} ({})",
800                        error.message,
801                        error.code
802                    );
803                }
804                let hash: String = serde_json::from_value(
805                    response
806                        .result
807                        .context("bitcoin getblockhash batch response is missing its result")?,
808                )
809                .with_context(|| format!("failed to parse block hash at height {height}"))?;
810                let hash = BlockHash::from_str(&hash)
811                    .with_context(|| format!("invalid bitcoin block hash at height {height}"))?;
812                hashes.insert(height, hash);
813            }
814            anyhow::ensure!(
815                seen.len() == chunk.len(),
816                "bitcoin getblockhash batch returned {}/{} responses",
817                seen.len(),
818                chunk.len()
819            );
820        }
821        Ok(hashes)
822    }

825impl HttpJsonRpcTransport {
826    /// Whether each outpoint is in the UTXO set. `include_mempool` also
827    /// counts a spend in the mempool. Callers pass outpoints that exist:
828    /// `gettxout` answers null for a spent output and for one that never
829    /// existed alike.
830    fn lookup_unspent(
831        &self,
832        outpoints: &[OutPoint],
833        include_mempool: bool,
834    ) -> anyhow::Result<HashMap<OutPoint, bool>> {
835        let mut unspent = HashMap::with_capacity(outpoints.len());
836        for chunk in outpoints.chunks(HTTP_JSON_RPC_BATCH_SIZE) {
837            let requests = chunk
838                .iter()
839                .enumerate()
840                .map(|(id, outpoint)| {
841                    serde_json::json!({
842                        "jsonrpc": "2.0",
843                        "id": id,
844                        "method": "gettxout",
845                        "params": [outpoint.txid.to_string(), outpoint.vout, include_mempool],
846                    })
847                })
848                .collect::<Vec<_>>();
849            let responses = self.call_batch(&requests)?;
850            let mut seen = HashSet::with_capacity(chunk.len());
851            for response in responses {
852                let index = response_index(&response, chunk.len())?;
853                anyhow::ensure!(
854                    seen.insert(index),
855                    "duplicate bitcoin gettxout batch response ID {index}"
856                );
857                let outpoint = chunk[index];
858                if let Some(error) = response.error {
859                    anyhow::bail!(
860                        "bitcoin gettxout failed for {outpoint}: {} ({})",
861                        error.message,
862                        error.code
863                    );
864                }
865                let is_unspent = !matches!(response.result, None | Some(Value::Null));
866                unspent.insert(outpoint, is_unspent);
867            }
868            anyhow::ensure!(
869                seen.len() == chunk.len(),
870                "bitcoin gettxout batch returned {}/{} responses",
871                seen.len(),
872                chunk.len()
873            );
874        }
875        Ok(unspent)
876    }
877}

879fn json_rpc_rate_limited(response: &Value) -> bool {
880    let is_rate_limited = |envelope: &Value| {
881        envelope
882            .get("error")
883            .and_then(|error| error.get("code"))
884            .and_then(Value::as_i64)
885            == Some(429)
886    };
887
888    match response {
889        Value::Array(envelopes) => envelopes.iter().any(is_rate_limited),
890        envelope => is_rate_limited(envelope),
891    }
892}
893
894fn response_index(response: &JsonRpcEnvelope, batch_len: usize) -> anyhow::Result<usize> {
895    let index = response
896        .id
897        .as_u64()
898        .context("bitcoin JSON-RPC batch response ID is not an integer")?;
899    let index =
900        usize::try_from(index).context("bitcoin JSON-RPC batch response ID is too large")?;
901    anyhow::ensure!(
902        index < batch_len,
903        "bitcoin JSON-RPC batch response ID {index} is outside batch of {batch_len}"
904    );
905    Ok(index)
906}

1141    /// Accept one connection, read its request, and answer `response`.
1142    fn respond(listener: &TcpListener, response: &str) {
1143        let (stream, _) = listener.accept().unwrap();
1144        let mut reader = BufReader::new(stream);
1145        let mut content_length = 0;
1146        loop {
1147            let mut line = String::new();
1148            reader.read_line(&mut line).unwrap();
1149            if line == "\r\n" {
1150                break;
1151            }
1152            let line = line.to_ascii_lowercase();
1153            if let Some(value) = line.strip_prefix("content-length:") {
1154                content_length = value.trim().parse().unwrap();
1155            }
1156        }
1157        reader.read_exact(&mut vec![0; content_length]).unwrap();
1158        reader.into_inner().write_all(response.as_bytes()).unwrap();
1159    }

1323    /// A client of a one-request server that answers `response`.
1324    fn client_answered_by(response: &'static str) -> (BtcRpcClient, std::thread::JoinHandle<()>) {
1325        let listener = TcpListener::bind("127.0.0.1:0").unwrap();
1326        let mut cfg = Config::for_tests();
1327        cfg.btc.rpc_url = format!("http://{}", listener.local_addr().unwrap());
1328        let server = std::thread::spawn(move || respond(&listener, response));
1329        let client = BtcRpcClient::new(&cfg, MAINNET_MIN_CONFIRMATIONS, MAX_TIP_AGE).unwrap();
1330        (client, server)
1331    }
