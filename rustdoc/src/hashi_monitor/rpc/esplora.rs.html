4//! A reference Bitcoin source over the Esplora HTTP API, such as mempool.space.
5//!
6//! Why it exists: whoever controls the peers of the monitor's node can withhold
7//! blocks from it. The node then cannot see an honest deposit, and a fake one
8//! looks the same. The auditor counts a deposit approval as paid when either
9//! source shows its payment, so one stalled source cannot cause a pause or
10//! hide a fake deposit.
11
12use std::collections::HashMap;
13
14use anyhow::Context;
15use bitcoin::OutPoint;
16use bitcoin::ScriptBuf;
17use serde::Deserialize;
18
19const REQUEST_TIMEOUT_SECS: u64 = 10;
20
21/// What the reference source holds at an outpoint. See `OutpointState`.
22#[derive(Clone, Debug, PartialEq, Eq)]
23pub enum ReferenceOutput {
24    /// The transaction is unknown or not confirmed deep enough, or it has no
25    /// output at that vout.
26    Missing,
27    /// The output exists, and a transaction, also in the mempool, spent it.
28    Spent {
29        amount: u64,
30        script_pubkey: ScriptBuf,
31    },
32    /// The output exists and is unspent.
33    Unspent {
34        amount: u64,
35        script_pubkey: ScriptBuf,
36    },
37}
38
39pub struct EsploraClient {
40    url: String,
41    /// Confirmations a transaction needs to count as confirmed.
42    confirmations: u64,
43}
44
45#[derive(Deserialize)]
46struct Transaction {
47    status: TransactionStatus,
48    vout: Vec<TransactionOutput>,
49}
50
51#[derive(Deserialize)]
52struct TransactionStatus {
53    confirmed: bool,
54    block_height: Option<u64>,
55}
56
57#[derive(Deserialize)]
58struct TransactionOutput {
59    value: u64,
60    scriptpubkey: String,
61}
62
63#[derive(Deserialize)]
64struct Outspend {
65    spent: bool,
66}
67
68impl EsploraClient {
69    pub fn new(url: &str, confirmations: u64) -> Self {
70        Self {
71            url: url.trim_end_matches('/').to_owned(),
72            confirmations,
73        }
74    }
75
76    /// The state of each outpoint at the source's current tip. An outpoint
77    /// whose lookup fails is logged and left out, so it does not cost the
78    /// others their answers.
79    pub fn lookup_outputs(
80        &self,
81        outpoints: &[OutPoint],
82    ) -> anyhow::Result<HashMap<OutPoint, ReferenceOutput>> {
83        let tip_height = self
84            .get("/blocks/tip/height")?
85            .context("esplora has no tip height")?
86            .as_str()?
87            .trim()
88            .parse::<u64>()
89            .context("failed to parse the esplora tip height")?;
90        Ok(outpoints
91            .iter()
92            .filter_map(|outpoint| match self.lookup(outpoint, tip_height) {
93                Ok(output) => Some((*outpoint, output)),
94                Err(error) => {
95                    tracing::warn!(?error, %outpoint, "esplora lookup failed");
96                    None
97                }
98            })
99            .collect())
100    }
101
102    fn lookup(&self, outpoint: &OutPoint, tip_height: u64) -> anyhow::Result<ReferenceOutput> {
103        let Some(response) = self.get(&format!("/tx/{}", outpoint.txid))? else {
104            return Ok(ReferenceOutput::Missing);
105        };
106        let transaction: Transaction = response
107            .json()
108            .context("failed to parse an esplora transaction")?;
109        let (true, Some(block_height)) = (
110            transaction.status.confirmed,
111            transaction.status.block_height,
112        ) else {
113            return Ok(ReferenceOutput::Missing);
114        };
115        if block_height.saturating_add(self.confirmations) > tip_height.saturating_add(1) {
116            return Ok(ReferenceOutput::Missing);
117        }
118        let Some(output) = transaction.vout.get(outpoint.vout as usize) else {
119            return Ok(ReferenceOutput::Missing);
120        };
121        let outspend: Outspend = self
122            .get(&format!("/tx/{}/outspend/{}", outpoint.txid, outpoint.vout))?
123            .with_context(|| format!("esplora has no outspend for {outpoint}"))?
124            .json()
125            .context("failed to parse an esplora outspend")?;
126        let amount = output.value;
127        let script_pubkey = ScriptBuf::from_hex(&output.scriptpubkey)
128            .context("failed to parse an esplora script")?;
129        Ok(if outspend.spent {
130            ReferenceOutput::Spent {
131                amount,
132                script_pubkey,
133            }
134        } else {
135            ReferenceOutput::Unspent {
136                amount,
137                script_pubkey,
138            }
139        })
140    }
141
142    /// The response to `GET path`, or `None` on HTTP 404.
143    fn get(&self, path: &str) -> anyhow::Result<Option<minreq::Response>> {
144        let response = minreq::get(format!("{}{path}", self.url))
145            .with_timeout(REQUEST_TIMEOUT_SECS)
146            .send()
147            .with_context(|| format!("esplora request {path} failed"))?;
148        match response.status_code {
149            200 => Ok(Some(response)),
150            404 => Ok(None),
151            code => anyhow::bail!("esplora request {path} answered HTTP {code}"),
152        }
153    }
154}

156#[cfg(test)]
157mod tests {
158    use super::*;
159    use crate::test_utils::respond;
160    use crate::test_utils::txid;
161    use std::net::TcpListener;
162
163    /// A URL whose next requests get `responses`, in order.
164    fn esplora_answering(responses: Vec<String>) -> (String, std::thread::JoinHandle<()>) {
165        let listener = TcpListener::bind("127.0.0.1:0").unwrap();
166        let url = format!("http://{}", listener.local_addr().unwrap());
167        let server = std::thread::spawn(move || {
168            for response in responses {
169                respond(&listener, &response);
170            }
171        });
172        (url, server)
173    }
174
175    fn ok(body: &str) -> String {
176        format!(
177            "HTTP/1.1 200 OK\r\nContent-Length: {}\r\n\r\n{body}",
178            body.len()
179        )
180    }
181
182    fn transaction(block_height: u64) -> String {
183        ok(&format!(
184            r#"{{"status":{{"confirmed":true,"block_height":{block_height}}},"vout":[{{"value":5000,"scriptpubkey":"51"}}]}}"#
185        ))
186    }
187
188    #[test]
189    fn outputs_are_read_at_the_confirmation_depth() {
190        let not_found = "Transaction not found";
191        let (url, server) = esplora_answering(vec![
192            ok("100"),
193            // The first outpoint's lookup fails.
194            "HTTP/1.1 500 Internal Server Error\r\nContent-Length: 0\r\n\r\n".to_owned(),
195            transaction(95),
196            ok(r#"{"spent":false}"#),
197            transaction(95),
198            ok(r#"{"spent":true}"#),
199            transaction(96),
200            format!(
201                "HTTP/1.1 404 Not Found\r\nContent-Length: {}\r\n\r\n{not_found}",
202                not_found.len()
203            ),
204        ]);
205        let outpoint = |fill| OutPoint {
206            txid: txid(fill),
207            vout: 0,
208        };
209        let failed = outpoint(5);
210        let outpoints = [failed, outpoint(1), outpoint(2), outpoint(3), outpoint(4)];
211
212        let outputs = EsploraClient::new(&url, 6)
213            .lookup_outputs(&outpoints)
214            .unwrap();
215        server.join().unwrap();
216
217        assert!(!outputs.contains_key(&failed));
218        assert_eq!(
219            outputs[&outpoints[1]],
220            ReferenceOutput::Unspent {
221                amount: 5000,
222                script_pubkey: ScriptBuf::from_bytes(vec![0x51]),
223            }
224        );
225        assert_eq!(
226            outputs[&outpoints[2]],
227            ReferenceOutput::Spent {
228                amount: 5000,
229                script_pubkey: ScriptBuf::from_bytes(vec![0x51]),
230            }
231        );
232        // Five confirmations at tip 100.
233        assert_eq!(outputs[&outpoints[3]], ReferenceOutput::Missing);
234        assert_eq!(outputs[&outpoints[4]], ReferenceOutput::Missing);
235    }
