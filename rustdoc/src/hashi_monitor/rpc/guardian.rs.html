4//! The guardian poller: the withdrawal E2 events read from the guardian's
5//! S3 logs, one hourly directory at a time, while the directory fills.
6//!
7//! Why it exists: the guardian's signature on a withdrawal is only recorded
8//! in its log. The poller reads the log through `GuardianReader`, which
9//! verifies each session's attestation against the allowlist and each
10//! record's signature.
11//!
12//! How it works:
13//! 1. The `GuardianCursor` names an hourly directory. Each poll lists it and
14//!    reads only the keys it has not read yet, so an E2 arrives within a poll.
15//! 2. Every verified withdrawal log becomes an E2 event. A log whose signed
16//!    time differs from its S3 `LastModified` time by more than
17//!    `MAX_EXPECTED_LOG_CLOCK_SKEW` is also a finding.
18//! 3. The guardian's limiter takes each `seq` once, in order, so the logs
19//!    have the seqs 0, 1, 2, ... A second log for a seq is a
20//!    `GuardianInconsistency::SeqDuplicate` finding.
21//! 4. A seq that a higher seq skips is a gap. It may still appear in the
22//!    next directory. A skip of more than `MAX_SEQ_SKIP` seqs is a `SeqJump`
23//!    finding at once. A gap still open at the end of that directory is a
24//!    `GuardianInconsistency::SeqGap` finding.
25//! 5. Once the directory's write completion time has passed, a last poll
26//!    reads it, and the cursor moves to the next directory.
27//!
28//! Assumptions:
29//! - The bridge keys are read once, before polling, from the genesis and
30//!   ceremony records. They never change.
31//! - The guardian's clock does not step back by more than one directory
32//!   between two logs. A later seq is then never two directories early.
33//! - A first run starts at the bridge's genesis, before the first
34//!   withdrawal log. Its seq is 0.
35

58/// The most seqs one log may skip before a `SeqJump` finding. A skip is
59/// rare, so a larger one is a finding, and the skipped seqs are not tracked.
60const MAX_SEQ_SKIP: u64 = 10;
61
62/// How far the guardian log is read, and the seqs it has.
63#[derive(Clone, Debug, Deserialize, PartialEq, Serialize)]
64pub struct GuardianCursor {
65    /// The directory read now. Every earlier directory is read in full.
66    dir: S3HourDirectory,
67    /// The keys read in `dir`.
68    dir_keys: BTreeSet<String>,
69    /// The seq after the highest seq read. The first log in a bucket has
70    /// seq 0, because the guardian's limiter starts there.
71    next_seq: u64,
72    /// The seqs skipped in `dir`. Their logs may come until the end of the
73    /// next directory.
74    gaps: BTreeSet<u64>,
75    /// The seqs skipped in the directory before `dir`. Their logs must come
76    /// by the end of `dir`.
77    due_gaps: BTreeSet<u64>,
78}
79
80impl GuardianCursor {
81    pub fn new(dir: S3HourDirectory) -> Self {
82        Self {
83            dir,
84            dir_keys: BTreeSet::new(),
85            next_seq: 0,
86            gaps: BTreeSet::new(),
87            due_gaps: BTreeSet::new(),
88        }
89    }
90
91    /// The start of `dir`. Every log before it is read.
92    pub fn read_until(&self) -> UnixSeconds {
93        self.dir.to_unix_seconds()
94    }
95
96    fn is_new(&self, key: &str) -> bool {
97        !self.dir_keys.contains(key)
98    }
99
100    /// Record the read log `key` with `seq`. Return the inconsistency it
101    /// shows, if any.
102    fn record(&mut self, key: String, seq: u64) -> Option<GuardianInconsistency> {
103        // `poll` reads only new keys, and a list has each key once.
104        let new = self.dir_keys.insert(key.clone());
105        assert!(new, "the guardian log {key} was read twice");
106        match seq.cmp(&self.next_seq) {
107            // A seq below `next_seq` is a late log for a gap, or a duplicate.
108            Ordering::Less => {
109                if !self.gaps.remove(&seq) && !self.due_gaps.remove(&seq) {
110                    return Some(GuardianInconsistency::SeqDuplicate {
111                        seq,
112                        object_key: key,
113                    });
114                }
115                tracing::info!(seq, key, "a late guardian log filled a seq gap");
116                None
117            }
118            Ordering::Equal => {
119                self.next_seq = seq.saturating_add(1);
120                None
121            }
122            // A seq above `next_seq` skips the seqs between them.
123            Ordering::Greater => {
124                let next_seq = self.next_seq;
125                self.next_seq = seq.saturating_add(1);
126                if seq - next_seq > MAX_SEQ_SKIP {
127                    return Some(GuardianInconsistency::SeqJump {
128                        next_seq,
129                        seq,
130                        object_key: key,
131                    });
132                }
133                tracing::info!(
134                    first_missing = next_seq,
135                    last_missing = seq - 1,
136                    key,
137                    "a guardian log skipped seqs; they are gaps"
138                );
139                self.gaps.extend(next_seq..seq);
140                None
141            }
142        }
143    }
144
145    /// Move to the next directory. Return a gap for each due seq that no
146    /// log filled.
147    fn finish_dir(&mut self) -> anyhow::Result<Vec<GuardianInconsistency>> {
148        self.dir = self.dir.next_dir()?;
149        self.dir_keys.clear();
150        let expect_by = self.dir.to_unix_seconds();
151        let due = std::mem::take(&mut self.due_gaps);
152        self.due_gaps = std::mem::take(&mut self.gaps);
153        Ok(due
154            .into_iter()
155            .map(|seq| GuardianInconsistency::SeqGap { seq, expect_by })
156            .collect())
157    }
158}

160/// What one guardian poll read.
161pub struct GuardianPoll {
162    pub events: Vec<MonitorEvent>,
163    pub findings: Vec<MonitorFinding>,
164    /// Whether the poll finished the directory and moved the cursor.
165    pub dir_done: bool,
166}

168pub struct GuardianWithdrawalsPoller {
169    /// Owns the S3 client + the trusted-key cache, so a session's attestation is
170    /// verified once for the poller's lifetime.
171    reader: GuardianReader,
172    cursor: GuardianCursor,
173    /// Set by `read_hashi_btc_keys`. The guardian's internal outputs state a
174    /// derivation path, and the keys turn it into the address they pay.
175    hashi_btc_keys: Option<HashiBTCKeys>,
176}
177
178impl PollCursor for GuardianWithdrawalsPoller {
179    type Position = GuardianCursor;
180
181    fn read_until(&self) -> UnixSeconds {
182        self.cursor.read_until()
183    }
184
185    fn next_item_to_poll(&self) -> Self::Position {
186        self.cursor.clone()
187    }
188}
189
190impl GuardianWithdrawalsPoller {
191    /// Fails if S3 cannot be reached.
192    pub async fn new(config: &Config, start: GuardianCursor) -> anyhow::Result<Self> {
193        let s3_credentials =
194            hashi_guardian::resolve_s3_credentials(config.s3_credentials.as_ref()).await?;
195        Ok(Self {
196            reader: GuardianReader::new(config.deployment.clone(), s3_credentials).await?,
197            cursor: start,
198            hashi_btc_keys: None,
199        })
200    }
201
202    /// The two keys every bridge address derives from, and the Hashi object
203    /// id. They never change. The MPC master key and the object id come from
204    /// the KP-authorized genesis record and the guardian's Bitcoin key from
205    /// the latest ceremony record, both verified against the allowlist.
206    pub async fn read_hashi_identity(
207        &mut self,
208        network: Network,
209    ) -> anyhow::Result<(HashiBTCKeys, Address)> {
210        let genesis = self
211            .reader
212            .read_genesis()
213            .await
214            .context("failed to read the guardian genesis record")?
215            .context("the guardian genesis record does not exist")?;
216        let ceremony = self
217            .reader
218            .read_latest_ceremony_state()
219            .await
220            .context("failed to read the guardian ceremony record")?;
221        let keys = HashiBTCKeys::new(ceremony.btc_master_pubkey, genesis.mpc_master_g, network);
222        tracing::info!(
223            change_script = %keys.script_pubkey(None),
224            hashi_object_id = %genesis.hashi_object_id,
225            "read the Hashi identity from the guardian records"
226        );
227        self.hashi_btc_keys = Some(keys.clone());
228        Ok((keys, genesis.hashi_object_id))
229    }
230
231    /// Read the logs of the current directory that are not read yet. Once
232    /// the directory's write completion time has passed, finish it.
233    pub async fn poll(&mut self) -> anyhow::Result<GuardianPoll> {
234        // Decide before the list, so the last list sees every write.
235        let dir_done = now_timestamp_secs() >= self.cursor.dir.write_completion_time();
236        let dir = self.cursor.dir.clone();
237        let cursor = &self.cursor;
238        let mut verified_logs = self
239            .reader
240            .read_logs_in_dir_where(&dir, |key| cursor.is_new(key))
241            .await?;
242        // The seq rules need seq order. Sort by the signed seq, so the order
243        // does not depend on the key format.
244        verified_logs.sort_by_key(|log| {
245            let message = log.entry().message().as_withdrawal();
246            message.map(|withdrawal| withdrawal.request_data.seq)
247        });
248        // A failed poll leaves the cursor as it was, so the retry reads the
249        // same logs again.
250        let mut next = self.cursor.clone();
251        // Withdrawal polling may replay historical buckets during an upgrade, so
252        // this caller accepts any record whose session build verifies against the
253        // configured allowlist. Add a cursor/cutoff policy here if tailing must
254        // require the current build after the upgrade window.
255        let keys = self
256            .hashi_btc_keys
257            .as_ref()
258            .context("the bridge keys must be read before polling guardian logs")?;
259        let mut findings = Vec::new();
260        let mut withdrawal_events = Vec::with_capacity(verified_logs.len());
261        for log in verified_logs {
262            // Each record is one sample of the guardian's clock against AWS
263            // time, plus the write latency.
264            if log.exceeds_clock_skew(MAX_EXPECTED_LOG_CLOCK_SKEW) {
265                findings.push(MonitorFinding::GuardianClockSkew {
266                    session_id: log.session_id().clone(),
267                    object_key: log.entry().object_key().to_owned(),
268                    logged_at_ms: log.entry().timestamp_ms(),
269                    stored_at_ms: log.stored_at_ms(),
270                });
271            }
272            let entry = log.into_entry();
273            let object_key = entry.object_key().to_owned();
274            let timestamp_ms = entry.timestamp_ms();
275            let withdrawal = entry
276                .into_message()
277                .into_withdrawal()
278                .context("non-withdrawal log in the guardian's withdrawal logs")?;
279            let inconsistency = next.record(object_key, withdrawal.request_data.seq);
280            findings.extend(inconsistency.map(MonitorFinding::GuardianInconsistency));
281            match MonitorWithdrawalEvent::from_guardian(&withdrawal, timestamp_ms, keys) {
282                Ok(event) => {
283                    debug!(wid = %event.wid, txid = %event.btc_txid, "guardian withdrawal log");
284                    withdrawal_events.push(MonitorEvent::Withdrawal(event));
285                }
286                Err(found) => findings.extend(found),
287            }
288        }
289        if dir_done {
290            let gaps = next.finish_dir()?;
291            findings.extend(gaps.into_iter().map(MonitorFinding::GuardianInconsistency));
292        }
293        self.cursor = next;
294        tracing::info!(
295            dir = %utc_timestamp(dir.to_unix_seconds()),
296            events = withdrawal_events.len(),
297            dir_done,
298            "read guardian logs"
299        );
300        Ok(GuardianPoll {
301            events: withdrawal_events,
302            findings,
303            dir_done,
304        })
305    }
306}

322#[cfg(test)]
323mod tests {
324    use super::*;
325
326    const HOUR: UnixSeconds = 1_700_000_000 / 3_600 * 3_600;
327
328    fn cursor() -> GuardianCursor {
329        GuardianCursor::new(S3HourDirectory::withdraw(HOUR).unwrap())
330    }
331
332    fn record(cursor: &mut GuardianCursor, seq: u64) -> Option<GuardianInconsistency> {
333        cursor.record(format!("key-{seq}"), seq)
334    }
335
336    #[test]
337    fn seqs_in_order_give_no_finding() {
338        let mut cursor = cursor();
339        for seq in 0..4 {
340            assert_eq!(record(&mut cursor, seq), None);
341        }
342        assert!(!cursor.is_new("key-2"));
343        assert_eq!(cursor.finish_dir().unwrap(), vec![]);
344        assert!(cursor.is_new("key-2"));
345        assert_eq!(cursor.read_until(), HOUR + 3_600);
346    }
347
348    #[test]
349    fn a_second_log_for_a_seq_is_a_duplicate() {
350        let mut cursor = cursor();
351        assert_eq!(record(&mut cursor, 0), None);
352        assert_eq!(
353            cursor.record("other-0".to_string(), 0),
354            Some(GuardianInconsistency::SeqDuplicate {
355                seq: 0,
356                object_key: "other-0".to_string(),
357            })
358        );
359    }
360
361    #[test]
362    fn a_gap_may_close_in_the_next_directory() {
363        let mut cursor = cursor();
364        assert_eq!(record(&mut cursor, 0), None);
365        assert_eq!(record(&mut cursor, 2), None);
366        assert_eq!(cursor.finish_dir().unwrap(), vec![]);
367
368        assert_eq!(record(&mut cursor, 1), None);
369        assert_eq!(cursor.finish_dir().unwrap(), vec![]);
370        // A seq whose gap closed is a duplicate after that.
371        assert!(matches!(
372            cursor.record("again-1".to_string(), 1),
373            Some(GuardianInconsistency::SeqDuplicate { seq: 1, .. })
374        ));
375    }
376
377    #[test]
378    fn a_gap_open_at_the_end_of_the_next_directory_is_a_finding() {
379        let mut cursor = cursor();
380        // The first log in a bucket has seq 0, so seq 0 is a gap too.
381        assert_eq!(record(&mut cursor, 2), None);
382        assert_eq!(cursor.finish_dir().unwrap(), vec![]);
383
384        assert_eq!(record(&mut cursor, 1), None);
385        assert_eq!(
386            cursor.finish_dir().unwrap(),
387            vec![GuardianInconsistency::SeqGap {
388                seq: 0,
389                expect_by: HOUR + 2 * 3_600,
390            }]
391        );
392    }
393
394    #[test]
395    fn a_skip_of_more_than_the_limit_is_a_jump() {
396        let mut cursor = cursor();
397        assert_eq!(record(&mut cursor, 0), None);
398        assert_eq!(record(&mut cursor, 1 + MAX_SEQ_SKIP), None);
399        assert_eq!(
400            record(&mut cursor, 3 + 2 * MAX_SEQ_SKIP),
401            Some(GuardianInconsistency::SeqJump {
402                next_seq: 2 + MAX_SEQ_SKIP,
403                seq: 3 + 2 * MAX_SEQ_SKIP,
404                object_key: format!("key-{}", 3 + 2 * MAX_SEQ_SKIP),
405            })
406        );
407        // A jump adds no gaps; the next seq continues after it.
408        assert_eq!(record(&mut cursor, 4 + 2 * MAX_SEQ_SKIP), None);
409    }
410
411    #[test]
412    fn a_cursor_with_gaps_round_trips_through_json() {
413        let mut cursor = cursor();
414        assert_eq!(record(&mut cursor, 1), None);
415        cursor.finish_dir().unwrap();
416        assert_eq!(record(&mut cursor, 3), None);
417
418        let json = serde_json::to_string(&cursor).unwrap();
419        assert_eq!(
420            serde_json::from_str::<GuardianCursor>(&json).unwrap(),
421            cursor
422        );
423    }
