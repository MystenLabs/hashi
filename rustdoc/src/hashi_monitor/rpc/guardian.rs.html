4//! The guardian poller: the withdrawal E2 events read from the guardian's
5//! S3 logs, one hourly directory at a time.
6//!
7//! Why it exists: the guardian's signature on a withdrawal is only recorded
8//! in its log. The poller reads the log through `GuardianReader`, which
9//! verifies each session's attestation against the allowlist and each
10//! record's signature.
11//!
12//! How it works:
13//! 1. The cursor is an hourly directory. A directory is read once its write
14//!    completion time has passed.
15//! 2. Every verified withdrawal log in the directory becomes an E2 event.
16//!    A log whose signed time differs from its S3 `LastModified` time by more
17//!    than `MAX_EXPECTED_LOG_CLOCK_SKEW` is also a finding.
18//! 3. The cursor moves to the next directory.
19//!
20//! Assumptions:
21//! - The bridge keys are read once, before polling, from the genesis and
22//!   ceremony records. They never change.
23//! - Concurrent guardian sessions are not detected. One way to detect them
24//!   is to keep each session's first and last observed timestamp and require
25//!   that the ranges do not overlap.
26
27use crate::config::Config;
28use crate::domain::MonitorEvent;
29use crate::domain::MonitorWithdrawalEvent;
30use crate::domain::PollOutcome;
31use crate::findings::MonitorFinding;
32use crate::keys::HashiBTCKeys;
33use crate::time::utc_timestamp;
34use anyhow::Context;
35use bitcoin::Network;
36use hashi_guardian::MAX_EXPECTED_LOG_CLOCK_SKEW;
37use hashi_guardian::s3_reader::GuardianReader;
38use hashi_guardian::s3_reader::VerifiedLogEntry;
39use hashi_types::guardian::s3::S3HourDirectory;
40use hashi_types::guardian::time::UnixSeconds;
41use hashi_types::guardian::time::now_timestamp_secs;
42use tracing::debug;
43
44pub struct GuardianWithdrawalsPoller {
45    /// Owns the S3 client + the trusted-key cache, so a session's attestation is
46    /// verified once for the poller's lifetime.
47    reader: GuardianReader,
48    cursor: S3HourDirectory,
49    /// Set by `read_hashi_btc_keys`. The guardian's internal outputs state a
50    /// derivation path, and the keys turn it into the address they pay.
51    hashi_btc_keys: Option<HashiBTCKeys>,
52}
53
54impl GuardianWithdrawalsPoller {
55    /// Fails if S3 cannot be reached.
56    pub async fn new(config: &Config, start: UnixSeconds) -> anyhow::Result<Self> {
57        let s3_credentials =
58            hashi_guardian::resolve_s3_credentials(config.s3_credentials.as_ref()).await?;
59        Ok(Self {
60            reader: GuardianReader::new(config.deployment.clone(), s3_credentials).await?,
61            cursor: S3HourDirectory::withdraw(start)?,
62            hashi_btc_keys: None,
63        })
64    }
65
66    /// The E2 event of a verified guardian withdrawal log.
67    fn withdrawal_event(&self, log: VerifiedLogEntry) -> anyhow::Result<MonitorWithdrawalEvent> {
68        let keys = self
69            .hashi_btc_keys
70            .as_ref()
71            .context("the bridge keys must be read before polling guardian logs")?;
72        let event = MonitorWithdrawalEvent::from_guardian(log, keys)?;
73        debug!(wid = %event.wid, txid = %event.btc_txid, "guardian withdrawal log");
74        Ok(event)
75    }
76
77    pub fn cursor_seconds(&self) -> UnixSeconds {
78        self.cursor.to_unix_seconds()
79    }
80
81    /// The two keys every bridge address derives from. They never change. The
82    /// MPC master key comes from the KP-authorized genesis record and the
83    /// guardian's Bitcoin key from the latest ceremony record, both verified
84    /// against the allowlist.
85    pub async fn read_hashi_btc_keys(&mut self, network: Network) -> anyhow::Result<HashiBTCKeys> {
86        let genesis = self
87            .reader
88            .read_genesis()
89            .await
90            .context("failed to read the guardian genesis record")?
91            .context("the guardian genesis record does not exist")?;
92        let ceremony = self
93            .reader
94            .read_latest_ceremony_state()
95            .await
96            .context("failed to read the guardian ceremony record")?;
97        let keys = HashiBTCKeys::new(ceremony.btc_master_pubkey, genesis.mpc_master_g, network);
98        tracing::info!(
99            change_script = %keys.script_pubkey(None),
100            "read the bridge keys from the guardian records"
101        );
102        self.hashi_btc_keys = Some(keys.clone());
103        Ok(keys)
104    }
105
106    /// Time after which the next unread hourly partition is considered complete.
107    pub fn next_partition_ready_at(&self) -> UnixSeconds {
108        self.cursor.write_completion_time()
109    }
110
111    /// Poll one hourly Guardian S3 directory and advance to the next directory.
112    pub async fn poll_one_hour(&mut self) -> anyhow::Result<PollOutcome> {
113        if now_timestamp_secs() < self.cursor.write_completion_time() {
114            return Ok(PollOutcome::CursorUnmoved);
115        }
116
117        let start = self.cursor.to_unix_seconds();
118        let next_cursor = self.cursor.next_dir()?;
119        let end = next_cursor.to_unix_seconds();
120        let verified_logs = self.reader.read_logs_in_dir(&self.cursor).await?;
121        // Withdrawal polling may replay historical buckets during an upgrade, so
122        // this caller accepts any record whose session build verifies against the
123        // configured allowlist. Add a cursor/cutoff policy here if tailing must
124        // require the current build after the upgrade window.
125        let mut findings = Vec::new();
126        let mut withdrawal_events = Vec::with_capacity(verified_logs.len());
127        for log in verified_logs {
128            // Each record is one sample of the guardian's clock against AWS
129            // time, plus the write latency.
130            if log.exceeds_clock_skew(MAX_EXPECTED_LOG_CLOCK_SKEW) {
131                findings.push(MonitorFinding::GuardianClockSkew {
132                    session_id: log.session_id().clone(),
133                    object_key: log.entry().object_key().to_owned(),
134                    logged_at_ms: log.entry().timestamp_ms(),
135                    stored_at_ms: log.stored_at_ms(),
136                });
137            }
138            withdrawal_events.push(MonitorEvent::Withdrawal(self.withdrawal_event(log)?));
139        }
140
141        self.cursor = next_cursor;
142        tracing::info!(
143            start = %utc_timestamp(start),
144            end = %utc_timestamp(end),
145            cursor = %utc_timestamp(self.cursor.to_unix_seconds()),
146            events = withdrawal_events.len(),
147            "completed Guardian event range"
148        );
149        Ok(PollOutcome::CursorAdvanced {
150            events: withdrawal_events,
151            findings,
152        })
153    }
154}

156#[cfg(test)]
157impl GuardianWithdrawalsPoller {
158    /// A poller over a mock S3 client, for tests that never poll it.
159    pub(crate) fn for_tests(config: &Config, start: UnixSeconds) -> Self {
160        Self {
161            reader: hashi_guardian::test_utils::mock_reader(config.deployment.clone()),
162            cursor: S3HourDirectory::withdraw(start).expect("valid test timestamp"),
163            hashi_btc_keys: Some(crate::test_utils::test_hashi_btc_keys()),
164        }
165    }
