4//! Withdrawal-side checks on the Hashi approvals that the monitor reads from Sui.
5//!
6//! Why it exists: Move validates a withdrawal's user outputs against the user
7//! requests, but not its txid or its change outputs. A compromised committee
8//! controls both, so the monitor checks them here before an approval enters
9//! the state machine.
10//!
11//! Goals of the approval checks, together with Move and the state machine:
12//! 1. The transaction the guardian signs is the one posted on Sui: the same
13//!    input UTXOs and the same output UTXOs.
14//! 2. Every output is either a user-requested payout, with the requested
15//!    amount less the fee share, or change to a Hashi address.
16//!
17//! Chain of trust, from the user's request to the guardian's signature:
18//! 1. Move builds the `WithdrawalTransaction` object in `new_withdrawal_txn`
19//!    (`withdrawal_queue.move`). It asserts that each user output pays the
20//!    request's address and the request's amount less the per-user miner fee,
21//!    and it caps that fee. The inputs are copied from the UTXO pool by id in
22//!    `commit_withdrawal_tx` (`withdraw.move`). Change outputs are not checked.
23//! 2. `commit_withdrawal_tx` then calls `emit_withdrawal_picked_for_processing`,
24//!    which emits `WithdrawalPickedForProcessing` from that object's own
25//!    fields. So the event and the object carry the same txid, inputs and
26//!    outputs, and both reach this file.
27//! 3. This file rebuilds the transaction from those inputs and outputs and
28//!    rejects the approval if the claimed txid differs.
29//! 4. The guardian computes its own txid from the outputs it signs, and the
30//!    state machine requires the guardian's txid to equal the approval's.
31//!    Together: the guardian signed exactly the outputs that Move validated.
32//! 5. A txid does not commit to which outputs Sui calls change, so the event
33//!    carries the inputs and outputs in Move's shape, and the state machine
34//!    requires the guardian's log, converted into that shape, to be equal.
35//!    The guardian derives every change address from a path it logs, so
36//!    equality means each change output pays the bridge, which completes
37//!    both goals.
38//!
39//! Assumptions:
40//! - The two keys every bridge address derives from never change. The caller
41//!   reads them once.
42//! - An approval comes from a `WithdrawalPickedForProcessing` event or from the
43//!   `WithdrawalTransaction` object created with it. Both carry the same txid,
44//!   inputs, outputs and timestamp.

63/// A Hashi approval read from Sui.
64#[derive(Debug, PartialEq, Eq)]
65pub enum HashiApproval {
66    /// The approval passed every check.
67    Valid(MonitorWithdrawalEvent),
68    /// The approval failed a check. It is reported and not ingested.
69    Rejected(Vec<MonitorFinding>),
70}
71
72impl From<Result<MonitorWithdrawalEvent, Vec<MonitorFinding>>> for HashiApproval {
73    fn from(checked: Result<MonitorWithdrawalEvent, Vec<MonitorFinding>>) -> Self {
74        match checked {
75            Ok(event) => Self::Valid(event),
76            Err(findings) => Self::Rejected(findings),
77        }
78    }
79}

81/// The Hashi approval recorded by the object at `wid`, or `None` if that object
82/// is not a Hashi `WithdrawalTransaction`.
83pub fn parse_withdrawal_object(
84    package_versions: &PackageVersions,
85    wid: WithdrawalID,
86    object: &Object,
87) -> anyhow::Result<Option<HashiApproval>> {
88    let is_withdrawal_transaction = object
89        .object_type_opt()
90        .context("Sui object is missing its type")?
91        .parse::<StructTag>()
92        .is_ok_and(|tag| WithdrawalTransaction::matches(package_versions, &tag));
93    if !is_withdrawal_transaction {
94        return Ok(None);
95    }
96    let txn: WithdrawalTransaction = object
97        .contents()
98        .deserialize()
99        .with_context(|| format!("failed to decode withdrawal transaction {wid}"))?;
100    anyhow::ensure!(
101        txn.id == wid,
102        "Sui returned withdrawal transaction {} for {wid}",
103        txn.id
104    );
105    Ok(Some(
106        MonitorWithdrawalEvent::from_sui_object(wid, txn).into(),
107    ))
108}

110/// Parse the monitored event a Sui event carries into `events`, or its
111/// findings into `findings` if it fails a check.
112pub fn parse_event(
113    package_versions: &PackageVersions,
114    hashi_btc_keys: &HashiBTCKeys,
115    event: Event,
116    transaction_timestamp_secs: UnixSeconds,
117    events: &mut Vec<MonitorEvent>,
118    findings: &mut Vec<MonitorFinding>,
119) -> anyhow::Result<()> {
120    let contents = event
121        .contents
122        .context("Sui event is missing BCS contents")?;
123    let event = HashiEvent::try_parse(package_versions, &contents)
124        .context("failed to parse Hashi Sui event")?;
125
126    match event {
127        Some(HashiEvent::WithdrawalPickedForProcessing(event)) => {
128            match MonitorWithdrawalEvent::from_sui_event(event) {
129                Ok(event) => events.push(MonitorEvent::Withdrawal(event)),
130                Err(rejection) => findings.extend(rejection),
131            }
132        }
133        Some(HashiEvent::DepositConfirmed(event)) => {
134            // ListTransactions supplies the containing checkpoint's timestamp
135            // alongside the nested events.
136            events.push(MonitorEvent::Deposit(MonitorDepositEvent::from_sui(
137                event,
138                transaction_timestamp_secs,
139                hashi_btc_keys,
140            )));
141        }
142        Some(_) | None => {}
143    }
144    Ok(())
145}

184    /// The inputs and outputs of `txn` as its Hashi approval carries them.
185    pub fn approval_tx(txn: &WithdrawalTransaction) -> WithdrawalTx {
186        WithdrawalTx::new(
187            txn.inputs.clone(),
188            txn.withdrawal_outputs.clone(),
189            txn.change_outputs.clone(),
190        )
191    }

193    /// `txn` with the txid that its inputs and outputs build.
194    pub fn with_rebuilt_txid(mut txn: WithdrawalTransaction) -> WithdrawalTransaction {
195        let txid = approval_tx(&txn).unsigned_tx().unwrap().compute_txid();
196        txn.txid = txid.into();
197        txn
198    }

200    /// The `withdrawal_tx` of one input into one user output and one change
201    /// output to the bridge, as a `WithdrawalTransaction` object whose txid is
202    /// the one its inputs and outputs build.
203    pub fn withdrawal_transaction(id: Address) -> WithdrawalTransaction {
204        let tx = withdrawal_tx(witness_program_from_address(&test_change_address()).unwrap());
205        let txn = WithdrawalTransaction {
206            id,
207            txid: BitcoinTxid::ZERO,
208            request_ids: vec![Address::new([0x55; 32])],
209            inputs: tx.inputs,
210            withdrawal_outputs: tx.withdrawal_outputs,
211            change_outputs: tx.change_outputs,
212            created_timestamp_ms: 1_789_805_327_448,
213            signed_timestamp_ms: None,
214            confirmed_timestamp_ms: None,
215            randomness: vec![],
216            signing: SigningBatch {
217                signatures: vec![],
218                epoch: 0,
219            },
220            guardian_signatures: None,
221        };
222        with_rebuilt_txid(txn)
223    }

242    /// The `WithdrawalPickedForProcessing` event emitted alongside `txn`.
243    pub fn picked_for_processing_event(txn: &WithdrawalTransaction) -> Event {
244        // `WithdrawalPickedForProcessing` fields, in declaration order.
245        let mut contents = Bcs::serialize(&(
246            txn.id,
247            txn.txid,
248            txn.request_ids.clone(),
249            txn.inputs.clone(),
250            txn.withdrawal_outputs.clone(),
251            txn.change_outputs.clone(),
252            txn.created_timestamp_ms,
253            txn.randomness.clone(),
254        ))
255        .unwrap();
256        contents.name = Some(format!(
257            "{PACKAGE_ID}::withdrawal_queue::WithdrawalPickedForProcessing"
258        ));
259        let mut event = Event::default();
260        event.contents = Some(contents);
261        event
262    }

271    /// The findings of a rejected approval.
272    fn rejection(approval: HashiApproval) -> Vec<MonitorFinding> {
273        match approval {
274            HashiApproval::Rejected(findings) => findings,
275            HashiApproval::Valid(event) => panic!("approval was not rejected: {event:?}"),
276        }
277    }

315    #[test]
316    fn a_change_output_to_another_address_is_carried_for_the_guardian_check() {
317        let mut txn = withdrawal_transaction(WID);
318        txn.change_outputs[0].bitcoin_address = vec![0x54; 32];
319        // An honest txid, as a committee paying itself as change would commit.
320        let txn = with_rebuilt_txid(txn);
321
322        // The address is not checked here: the state machine compares Sui's
323        // transaction with the guardian's once E2 arrives.
324        let HashiApproval::Valid(event) = approval(&txn) else {
325            panic!("the approval was rejected");
326        };
327        assert_eq!(event.tx.change_outputs[0].bitcoin_address, vec![0x54; 32]);
328    }

404    #[test]
405    fn lookup_and_event_scan_build_the_same_approval() {
406        let hashi_btc_keys = test_hashi_btc_keys();
407        for txn in [
408            withdrawal_transaction(WID),
409            with_wrong_txid(withdrawal_transaction(WID)),
410        ] {
411            let looked_up =
412                parse_withdrawal_object(&package_versions(), WID, &object_at_wid(PACKAGE_ID, &txn))
413                    .unwrap()
414                    .unwrap();
415            let expected = match looked_up {
416                HashiApproval::Valid(event) => (vec![MonitorEvent::Withdrawal(event)], vec![]),
417                HashiApproval::Rejected(findings) => (vec![], findings),
418            };
419
420            let (mut events, mut findings) = (Vec::new(), Vec::new());
421            parse_event(
422                &package_versions(),
423                &hashi_btc_keys,
424                picked_for_processing_event(&txn),
425                0,
426                &mut events,
427                &mut findings,
428            )
429            .unwrap();
430            assert_eq!((events, findings), expected);
431        }
