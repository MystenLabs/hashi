4//! The hBTC supply: the `total_supply` of Hashi's `TreasuryCap<BTC>`.
5//!
6//! Why it exists: the auditor compares the supply with the value of its UTXO
7//! pool. More hBTC than the pool holds means that Bitcoin does not back every
8//! hBTC.
9//!
10//! How it works:
11//! 1. At startup, `find_hbtc_treasury_cap` reads the Hashi object for the id
12//!    of its treasury `ObjectBag`.
13//! 2. Derive the id of the bag's `Key<TreasuryCap<BTC>>` field the same way
14//!    Move does. Read the field for the id of the cap.
15//! 3. Each check reads the cap and decodes its supply.
16//!
17//! Assumptions:
18//! - The cap stays in the treasury bag under the original package's `Key`,
19//!   so its id does not change.
20//! - Only `GetObject` is used. An archive node does not serve `GetCoinInfo`
21//!   or `ListDynamicFields`.
22

41/// The id of the dynamic object field that holds `TreasuryCap<BTC>` in the
42/// treasury bag `bag_id`.
43pub fn treasury_cap_field_id(bag_id: Address, package_id: Address) -> Address {
44    let btc = struct_tag(package_id, "btc", "BTC", vec![]);
45    let cap = struct_tag(Address::TWO, "coin", "TreasuryCap", vec![btc]);
46    let key = struct_tag(package_id, "treasury", "Key", vec![cap]);
47    let wrapper = struct_tag(Address::TWO, "dynamic_object_field", "Wrapper", vec![key]);
48    // `Key` has no fields, so Move gives it one `dummy_field: bool`.
49    bag_id.derive_dynamic_child_id(&wrapper, &[0])
50}

52impl SuiPoller {
53    /// Find Hashi's `TreasuryCap<BTC>` through the Hashi object
54    /// `hashi_object_id`. Call once, before `hbtc_supply`.
55    pub async fn find_hbtc_treasury_cap(&mut self, hashi_object_id: Address) -> anyhow::Result<()> {
56        let package_id = Address::from_str(&self.package_id)
57            .with_context(|| format!("invalid Hashi package ID {}", self.package_id))?;
58        let hashi: Hashi =
59            read_object(&mut self.client, hashi_object_id, "the Hashi object").await?;
60        let field_id = treasury_cap_field_id(hashi.treasury.objects.id, package_id);
61        let field: Field<bool, Address> =
62            read_object(&mut self.client, field_id, "the hBTC TreasuryCap field").await?;
63        self.hbtc_treasury_cap_id = field.value;
64        tracing::info!(hbtc_treasury_cap_id = %field.value, "found the hBTC TreasuryCap");
65        Ok(())
66    }
67
68    /// The `total_supply` of Hashi's `TreasuryCap<BTC>`, in satoshis.
69    pub async fn hbtc_supply(&mut self) -> anyhow::Result<u64> {
70        let id = self.hbtc_treasury_cap_id;
71        let cap: TreasuryCap = read_object(&mut self.client, id, "the hBTC TreasuryCap").await?;
72        Ok(cap.supply)
73    }
74}

76fn struct_tag(address: Address, module: &str, name: &str, type_params: Vec<TypeTag>) -> TypeTag {
77    TypeTag::Struct(Box::new(StructTag::new(
78        address,
79        Identifier::new(module).expect("valid module name"),
80        Identifier::new(name).expect("valid type name"),
81        type_params,
82    )))
83}

85/// The BCS contents of object `id`, decoded as `T`.
86async fn read_object<T: DeserializeOwned>(
87    client: &mut sui_rpc::Client,
88    id: Address,
89    what: &str,
90) -> anyhow::Result<T> {
91    let request =
92        GetObjectRequest::new(&id).with_read_mask(FieldMask::from_paths([Object::path_builder()
93            .contents()
94            .finish()]));
95    client
96        .ledger_client()
97        .get_object(request)
98        .await
99        .with_context(|| format!("failed to fetch {what} {id}"))?
100        .into_inner()
101        .object
102        .with_context(|| format!("Sui GetObject response is missing {what} {id}"))?
103        .contents()
104        .deserialize()
105        .with_context(|| format!("failed to decode {what} {id}"))
106}

112    #[test]
113    fn the_field_id_matches_testnet() {
114        // Read from testnet on 2026-10-22: the cap's owner is this field, and
115        // the field's owner is the treasury bag.
116        let package_id =
117            Address::from_str("0x5e0b99c31f691742acea5de4ca1bf53c398d5639fed6503a51ffe8035dc68485")
118                .unwrap();
119        let bag_id =
120            Address::from_str("0xe4029de51c3972ce6f0b2be70b6bff52f1f54ece7ad27e26b383bc328fc9e38d")
121                .unwrap();
122        assert_eq!(
123            treasury_cap_field_id(bag_id, package_id).to_string(),
124            "0x8ef8a4b5f9826796f896896624c0817ac232eccda18e3872420b7504524130f7"
125        );
