4//! The Sui poller: the Hashi events read from Sui, in checkpoint order, and
5//! the hBTC supply.
6//!
7//! Why it exists: the withdrawal E1 and the deposit E2 are Sui events. The
8//! poller scans the checkpoints for them and gives each one its time. It
9//! also reads the hBTC supply; see `hbtc_supply`.
10//!
11//! How it works:
12//! 1. The poller starts at a checkpoint: on a first run, the one that holds
13//!    the Hashi package's publish transaction; on a resume, the one the
14//!    snapshot saved. Before the first poll, it checks that the node still
15//!    serves the start. It sets the progress timestamp from the checkpoint
16//!    before the start.
17//! 2. Each poll lists the transactions that emit a monitored event, from the
18//!    next checkpoint through the checkpoint the caller names, but at most
19//!    `MAX_CHECKPOINTS_PER_POLL` checkpoints. The caller names the chain head
20//!    it read at the tick's start. A failed range is retried,
21//!    then split.
22//! 3. The cursor advances through the last checkpoint the node declared
23//!    complete. The transactions of later checkpoints wait for the next poll,
24//!    so a checkpoint is ingested whole.
25//!
26//! Assumptions:
27//! - Every monitored event type keeps the original package's address across
28//!   upgrades.
29

70const REQUEST_TIMEOUT: Duration = Duration::from_secs(60);
71const PAGE_SIZE: u32 = 1_000;
72/// Checkpoints one poll scans at most. A backlog, such as a first run from
73/// genesis, is walked in ranges of this size: each range fits one request
74/// timeout, and its events are released before the next range is read.
75const MAX_CHECKPOINTS_PER_POLL: u64 = 10_000;
76const MIN_CHECKPOINTS_PER_RETRY: u64 = 25;
77const MAX_RANGE_ATTEMPTS: u32 = 3;
78const INITIAL_RETRY_DELAY: Duration = Duration::from_millis(250);
79/// Distance kept above a pruning node's reported oldest checkpoint: about an hour
80/// of checkpoints, since the backends behind one endpoint prune unevenly.
81const PRUNING_MARGIN_CHECKPOINTS: u64 = 6 * 60 * 60;
82
83struct TransactionScan {
84    transactions: Vec<ExecutedTransaction>,
85    completed_checkpoint: Option<u64>,
86    end_reason: QueryEndReason,
87}
88
89fn completed_checkpoint_for_scan(
90    start_checkpoint: u64,
91    end_checkpoint: u64,
92    end_reason: QueryEndReason,
93    watermark_checkpoint: Option<u64>,
94) -> anyhow::Result<Option<u64>> {
95    anyhow::ensure!(
96        start_checkpoint < end_checkpoint,
97        "empty Sui transaction checkpoint range"
98    );
99    let completed_checkpoint = match end_reason {
100        QueryEndReason::CheckpointBound => end_checkpoint - 1,
101        QueryEndReason::LedgerTip if watermark_checkpoint.is_none() => return Ok(None),
102        _ => watermark_checkpoint.context(format!(
103            "Sui transaction scan ended with {end_reason:?} without a completion watermark"
104        ))?,
105    };
106    anyhow::ensure!(
107        completed_checkpoint < end_checkpoint,
108        "Sui transaction scan watermark checkpoint {completed_checkpoint} is outside requested range ending at {end_checkpoint}"
109    );
110    Ok((completed_checkpoint >= start_checkpoint).then_some(completed_checkpoint))
111}

113/// A Sui v2 gRPC client for `config`, with `REQUEST_TIMEOUT` on every request.
114fn connect(config: &SuiConfig) -> anyhow::Result<sui_rpc::Client> {
115    let client = sui_rpc::Client::new(&config.rpc_url)
116        .with_context(|| format!("invalid Sui RPC URL {}", config.rpc_url))?
117        .request_layer(tower::timeout::TimeoutLayer::new(REQUEST_TIMEOUT));
118    Ok(client)
119}

121/// The bridge's genesis on Sui: the transaction that published the original
122/// Hashi package. No Hashi event precedes it.

124pub struct Genesis {
125    /// The checkpoint that holds the publish transaction.
126    pub checkpoint: u64,
127    /// The time of that checkpoint.

131/// The genesis of the Hashi package in `config`.
132pub async fn package_genesis(config: &SuiConfig) -> anyhow::Result<Genesis> {
133    let mut client = connect(config)?;
134    publish_genesis(&mut client, &config.package_id).await
135}

137/// The checkpoint and time of the transaction that published `package_id`.
138/// The package object is immutable, so its previous transaction is its publish.
139async fn publish_genesis(
140    client: &mut sui_rpc::Client,
141    package_id: &str,
142) -> anyhow::Result<Genesis> {
143    let address = Address::from_str(package_id)
144        .with_context(|| format!("invalid Hashi package ID {package_id}"))?;
145    let request = GetObjectRequest::new(&address)
146        .with_read_mask(FieldMask::from_paths(["previous_transaction"]));
147    let publish_digest = client
148        .ledger_client()
149        .get_object(request)
150        .await
151        .with_context(|| format!("failed to fetch Hashi package {package_id}"))?
152        .into_inner()
153        .object
154        .with_context(|| format!("Sui GetObject response is missing package {package_id}"))?
155        .previous_transaction
156        .with_context(|| format!("Hashi package {package_id} is missing previous_transaction"))?;
157    let digest = Digest::from_str(&publish_digest).with_context(|| {
158        format!("invalid publish transaction digest {publish_digest} of Hashi package {package_id}")
159    })?;
160    let request = GetTransactionRequest::new(&digest)
161        .with_read_mask(FieldMask::from_paths(["checkpoint", "timestamp"]));
162    let transaction = client
163        .ledger_client()
164        .get_transaction(request)
165        .await
166        .with_context(|| format!("failed to fetch the publish transaction of Hashi package {package_id}"))?
167        .into_inner()
168        .transaction
169        .with_context(|| format!("Sui GetTransaction response is missing the publish transaction of Hashi package {package_id}"))?;
170    let checkpoint = transaction.checkpoint.with_context(|| {
171        format!("publish transaction of Hashi package {package_id} is missing checkpoint")
172    })?;
173    let timestamp = transaction.timestamp.with_context(|| {
174        format!("publish transaction of Hashi package {package_id} is missing timestamp")
175    })?;
176    let timestamp_ms = proto_to_timestamp_ms(timestamp).with_context(|| {
177        format!("invalid publish transaction timestamp of Hashi package {package_id}")
178    })?;
179    Ok(Genesis {
180        checkpoint,
181        timestamp: unix_millis_to_seconds(timestamp_ms),
182    })
183}

185/// The Hashi settings the monitor reads once at startup.
186#[derive(Clone, Copy, Debug, Eq, PartialEq)]
187pub struct HashiSettings {
188    /// `bitcoin_confirmation_threshold`: the confirmations the committee
189    /// waits for. It is 6 when unset, as in the Hashi node.
190    pub confirmation_threshold: u64,
191    /// `bitcoin_deposit_time_delay_ms` in seconds: the minimum time from a
192    /// deposit approval to its mint. Move's `init` sets 10 minutes.
193    pub deposit_delay_secs: u64,
194}

196/// Read `HashiSettings` from the Hashi object. Fail unless the object's
197/// type is from `config.package_id`: the event filter uses that id, so a
198/// wrong one reads no events.
199pub async fn read_hashi_settings(
200    config: &SuiConfig,
201    hashi_object_id: Address,
202) -> anyhow::Result<HashiSettings> {
203    let mut client = connect(config)?;
204    let request = GetObjectRequest::new(&hashi_object_id).with_read_mask(FieldMask::from_paths([
205        Object::path_builder().object_type(),
206        Object::path_builder().contents().finish(),
207    ]));
208    let object = client
209        .ledger_client()
210        .get_object(request)
211        .await
212        .with_context(|| format!("failed to fetch the Hashi object {hashi_object_id}"))?
213        .into_inner()
214        .object
215        .with_context(|| {
216            format!("Sui GetObject response is missing the Hashi object {hashi_object_id}")
217        })?;
218    let package_id = Address::from_str(&config.package_id)
219        .with_context(|| format!("invalid Hashi package ID {}", config.package_id))?;
220    let expected_type = StructTag::new(
221        package_id,
222        Identifier::from_static("hashi"),
223        Identifier::from_static("Hashi"),
224        vec![],
225    )
226    .to_string();
227    anyhow::ensure!(
228        object.object_type() == expected_type,
229        "the Hashi object {hashi_object_id} has type {}, not {expected_type}; \
230         sui.package_id must be the package that defines it",
231        object.object_type(),
232    );
233    let hashi: Hashi = object
234        .contents()
235        .deserialize()
236        .with_context(|| format!("failed to decode the Hashi object {hashi_object_id}"))?;
237    Ok(HashiSettings {
238        confirmation_threshold: hashi.config.get_u64("bitcoin_confirmation_threshold", 6),
239        deposit_delay_secs: unix_millis_to_seconds(
240            hashi
241                .config
242                .get_u64("bitcoin_deposit_time_delay_ms", 10 * 60 * 1_000),
243        ),
244    })
245}

247/// What one Sui poll read: the events of the new range, and the findings of
248/// events that failed a check.
249pub struct SuiPoll {
250    pub events: Vec<MonitorEvent>,
251    pub findings: Vec<MonitorFinding>,
252}

254pub struct SuiPoller {
255    /// Sui v2 gRPC client used for checkpoint, transaction and object requests.
256    client: sui_rpc::Client,
257    /// Deployed package versions used to identify Hashi event and object types.
258    package_versions: PackageVersions,
259    /// The keys every bridge address derives from.
260    hashi_btc_keys: HashiBTCKeys,
261    /// Original Hashi package used to construct server-side transaction filters.
262    package_id: String,
263    /// The time of the last checkpoint scanned. `ensure_start_is_served`
264    /// sets it for the start.
265    last_read_checkpoint_timestamp: UnixSeconds,
266    /// The first checkpoint not yet scanned.
267    next_checkpoint: u64,
268    /// Hashi's `TreasuryCap<BTC>`. `find_hbtc_treasury_cap` sets it.
269    hbtc_treasury_cap_id: Address,
270}

284impl SuiPoller {
285    /// A poller whose scan starts at checkpoint `start`.
286    pub fn new(
287        config: &SuiConfig,
288        hashi_btc_keys: HashiBTCKeys,
289        start: u64,
290    ) -> anyhow::Result<Self> {
291        let package_id = Address::from_str(&config.package_id)
292            .with_context(|| format!("invalid Hashi package ID {}", config.package_id))?;
293
294        let package_versions = PackageVersions::new(BTreeMap::from([(1, package_id)]));
295
296        Ok(Self {
297            client: connect(config)?,
298            package_versions,
299            hashi_btc_keys,
300            package_id: config.package_id.clone(),
301            last_read_checkpoint_timestamp: 0,
302            next_checkpoint: start,
303            hbtc_treasury_cap_id: Address::ZERO,
304        })
305    }
306
307    /// The newest checkpoint the node serves.
308    pub async fn latest_checkpoint(&mut self) -> anyhow::Result<u64> {
309        Ok(self.checkpoint_range().await?.1)
310    }
311
312    /// Scan up to checkpoint `up_to`, inclusive, but at most
313    /// `MAX_CHECKPOINTS_PER_POLL` checkpoints.
314    ///
315    /// The progress timestamp advances only through `watermark.checkpoint`, which
316    /// is the inclusive boundary that `ListTransactions` reports as fully
317    /// covered. Each transaction's nested events provide the Hashi payloads.
318    /// The Sui SDK handles item and scan limits, resumable scans,
319    /// and retryable partial streams. A terminal failure discards the partial
320    /// range; that range is retried and may be split. Return `None` when the
321    /// cursor does not move.
322    pub async fn poll(&mut self, up_to: u64) -> anyhow::Result<Option<SuiPoll>> {
323        let start_checkpoint = self.next_checkpoint;
324        if start_checkpoint > up_to {
325            return Ok(None);
326        }
327        // A backlog is scanned in bounded ranges. The caller polls again while
328        // the cursor advances, so the rest follows in the next polls.
329        let mut end_checkpoint = up_to
330            .saturating_add(1)
331            .min(start_checkpoint.saturating_add(MAX_CHECKPOINTS_PER_POLL));
332
333        let mut range_attempt = 0u32;
334        let scan = loop {
335            let scan = tokio::time::timeout(
336                REQUEST_TIMEOUT,
337                self.list_transactions_in_range(start_checkpoint, end_checkpoint),
338            )
339            .await
340            .context("Sui transaction scan timed out")
341            .and_then(|result| result);
342            match scan {
343                Ok(transactions) => break transactions,
344                Err(error) if range_attempt + 1 < MAX_RANGE_ATTEMPTS => {
345                    let delay = INITIAL_RETRY_DELAY.saturating_mul(1 << range_attempt);
346                    range_attempt += 1;
347                    tracing::warn!(
348                        start_checkpoint,
349                        end_checkpoint,
350                        range_attempt,
351                        ?delay,
352                        ?error,
353                        "Sui transaction scan failed; retrying range"
354                    );
355                    tokio::time::sleep(delay).await;
356                }
357                Err(error) => {
358                    let checkpoint_count = end_checkpoint.saturating_sub(start_checkpoint);
359                    if checkpoint_count <= MIN_CHECKPOINTS_PER_RETRY {
360                        return Err(error).context("Sui transaction scan failed after retries");
361                    }
362                    end_checkpoint = start_checkpoint + checkpoint_count / 2;
363                    tracing::warn!(
364                        start_checkpoint,
365                        end_checkpoint,
366                        ?error,
367                        "Sui transaction scan failed after retries; retrying a smaller range"
368                    );
369                    range_attempt = 0;
370                }
371            }
372        };
373
374        let Some(completed_checkpoint) = completed_checkpoint_for_scan(
375            start_checkpoint,
376            end_checkpoint,
377            scan.end_reason,
378            scan.completed_checkpoint,
379        )?
380        else {
381            return Ok(None);
382        };
383
384        let mut events = Vec::with_capacity(scan.transactions.len());
385        let mut findings = Vec::new();
386        for transaction in scan.transactions {
387            let checkpoint = transaction
388                .checkpoint
389                .context("filtered Sui transaction is missing checkpoint")?;
390            // A LedgerTip response can include transactions from a checkpoint
391            // that its watermark does not yet declare complete. Leave those
392            // transactions for the next poll so the checkpoint is ingested
393            // atomically and cannot be skipped as indexing catches up.
394            if checkpoint > completed_checkpoint {
395                continue;
396            }
397            self.parse_transaction(transaction, &mut events, &mut findings)?;
398        }
399
400        let scanned_through_timestamp = self.checkpoint_timestamp(completed_checkpoint).await?;
401        self.next_checkpoint = completed_checkpoint.saturating_add(1);
402        self.last_read_checkpoint_timestamp = self
403            .last_read_checkpoint_timestamp
404            .max(scanned_through_timestamp);
405        tracing::info!(
406            start_checkpoint,
407            end_checkpoint,
408            completed_checkpoint,
409            end_reason = ?scan.end_reason,
410            last_read_checkpoint_timestamp = %utc_timestamp(self.last_read_checkpoint_timestamp),
411            events = events.len(),
412            findings = findings.len(),
413            "completed Sui event range"
414        );
415        Ok(Some(SuiPoll { events, findings }))
416    }
417
418    /// Verify that the node retains the required history.
419    /// Set `last_read_checkpoint_timestamp` from the checkpoint before `next_checkpoint`.
420    pub async fn ensure_start_is_served(&mut self) -> anyhow::Result<()> {
421        let (lowest, _) = self.checkpoint_range().await?;
422        let previous = self.next_checkpoint.saturating_sub(1);
423        anyhow::ensure!(
424            lowest == 0 || lowest.saturating_add(PRUNING_MARGIN_CHECKPOINTS) <= previous,
425            "the Sui node has pruned checkpoint {previous}, which the scan from checkpoint {} needs: it serves from {lowest}. Use a Sui node that keeps the history from there",
426            self.next_checkpoint,
427        );
428        self.last_read_checkpoint_timestamp = self.checkpoint_timestamp(previous).await?;
429        Ok(())
430    }

432    /// The oldest and the newest checkpoint the node serves. The oldest is 0
433    /// unless the node prunes.
434    async fn checkpoint_range(&mut self) -> anyhow::Result<(u64, u64)> {
435        let service_info = self
436            .client
437            .ledger_client()
438            .get_service_info(GetServiceInfoRequest::default())
439            .await
440            .context("failed to fetch Sui service info")?
441            .into_inner();
442        let latest = service_info
443            .checkpoint_height
444            .context("Sui service info is missing checkpoint_height")?;
445        Ok((
446            service_info.lowest_available_checkpoint.unwrap_or(0),
447            latest,
448        ))
449    }

451    /// The timestamp of checkpoint `sequence_number`, read from the node.
452    async fn checkpoint_timestamp(&mut self, sequence_number: u64) -> anyhow::Result<UnixSeconds> {
453        let request = GetCheckpointRequest::by_sequence_number(sequence_number).with_read_mask(
454            FieldMask::from_paths(["sequence_number", "summary.timestamp"]),
455        );
456        let checkpoint = self
457            .client
458            .ledger_client()
459            .get_checkpoint(request)
460            .await
461            .context("failed to fetch Sui checkpoint")?
462            .into_inner()
463            .checkpoint
464            .context("missing Sui checkpoint")?;
465        let actual_sequence = checkpoint
466            .sequence_number
467            .context("Sui checkpoint is missing sequence_number")?;
468        let timestamp = checkpoint
469            .summary
470            .and_then(|summary| summary.timestamp)
471            .context("Sui checkpoint is missing summary.timestamp")?;
472        let timestamp_ms =
473            proto_to_timestamp_ms(timestamp).context("invalid Sui checkpoint timestamp")?;
474        let timestamp_secs = unix_millis_to_seconds(timestamp_ms);
475        anyhow::ensure!(
476            actual_sequence == sequence_number,
477            "requested Sui checkpoint {sequence_number}, received {actual_sequence}"
478        );
479        Ok(timestamp_secs)
480    }
481
482    async fn list_transactions_in_range(
483        &mut self,
484        start_checkpoint: u64,
485        end_checkpoint: u64,
486    ) -> anyhow::Result<TransactionScan> {
487        if start_checkpoint >= end_checkpoint {
488            anyhow::bail!("empty Sui transaction checkpoint range");
489        }
490
491        let request = ListTransactionsRequest::default()
492            .with_read_mask(FieldMask::from_paths([
493                "checkpoint",
494                "events.events.event_type",
495                "events.events.contents",
496            ]))
497            .with_start_checkpoint(start_checkpoint)
498            .with_end_checkpoint(end_checkpoint)
499            .with_filter(self.transaction_filter())
500            .with_options(
501                QueryOptions::default()
502                    .with_limit(PAGE_SIZE)
503                    .with_ordering(Ordering::Ascending),
504            );
505        let stream = self.client.list_transactions(request);
506        futures::pin_mut!(stream);
507
508        let mut all_transactions = Vec::new();
509        let mut completed_checkpoint = None;
510        let mut end_reason = None;
511        tracing::info!(
512            start_checkpoint,
513            end_checkpoint,
514            "starting Sui transaction scan"
515        );
516
517        while let Some(frame) = stream.next().await {
518            // The SDK validates watermarks and transparently resumes page
519            // limits and retryable partial streams before yielding frames.
520            let frame = frame.context("Sui ListTransactions stream failed")?;
521            if let Some(checkpoint) = frame
522                .watermark
523                .as_ref()
524                .and_then(|watermark| watermark.checkpoint)
525            {
526                completed_checkpoint = Some(checkpoint);
527            }
528            if let Some(transaction) = frame.transaction {
529                all_transactions.push(transaction);
530            }
531            if let Some(end) = frame.end {
532                end_reason = end
533                    .reason
534                    .and_then(|reason| QueryEndReason::try_from(reason).ok());
535            }
536        }
537
538        Ok(TransactionScan {
539            transactions: all_transactions,
540            completed_checkpoint,
541            end_reason: end_reason.context("Sui ListTransactions ended without QueryEnd")?,
542        })
543    }
544
545    fn transaction_filter(&self) -> TransactionFilter {
546        let event_types = [
547            format!(
548                "{}::withdrawal_queue::WithdrawalPickedForProcessing",
549                self.package_id
550            ),
551            format!("{}::deposit::DepositApproved", self.package_id),
552            format!("{}::deposit::DepositRequested", self.package_id),
553        ];
554        TransactionFilter::any(event_types.into_iter().map(tx_filter::event_type))
555    }
556
557    /// Parse the monitored events of one transaction into `events`. An event
558    /// that fails a check adds its findings to `findings` instead.
559    fn parse_transaction(
560        &self,
561        transaction: ExecutedTransaction,
562        events: &mut Vec<MonitorEvent>,
563        findings: &mut Vec<MonitorFinding>,
564    ) -> anyhow::Result<()> {
565        let sui_events = transaction
566            .events
567            .context("filtered Sui transaction is missing events")?
568            .events;
569
570        for sui_event in sui_events {
571            approval::parse_event(
572                &self.package_versions,
573                &self.hashi_btc_keys,
574                sui_event,
575                events,
576                findings,
577            )?;
578        }
579        Ok(())
580    }

611    /// Checkpoints `lowest..=head`, one a second from `GENESIS_SECS`. Like a pruning
612    /// fullnode, it reports `lowest` and answers older checkpoints with `NotFound`.
613    #[derive(Clone)]
614    struct PrunedLedger {
615        lowest: u64,
616        head: u64,
617    }
618
619    const GENESIS_SECS: UnixSeconds = 1_700_000_000;
620
621    #[tonic::async_trait]
622    impl LedgerService for PrunedLedger {
623        async fn get_service_info(
624            &self,
625            _: tonic::Request<GetServiceInfoRequest>,
626        ) -> Result<tonic::Response<GetServiceInfoResponse>, tonic::Status> {
627            let mut info = GetServiceInfoResponse::default();
628            info.checkpoint_height = Some(self.head);
629            info.timestamp = Some(timestamp_ms_to_proto((GENESIS_SECS + self.head) * 1_000));
630            info.lowest_available_checkpoint = Some(self.lowest);
631            Ok(tonic::Response::new(info))
632        }
633
634        async fn get_checkpoint(
635            &self,
636            request: tonic::Request<GetCheckpointRequest>,
637        ) -> Result<tonic::Response<GetCheckpointResponse>, tonic::Status> {
638            let Some(CheckpointId::SequenceNumber(sequence)) = request.into_inner().checkpoint_id
639            else {
640                return Err(tonic::Status::invalid_argument(
641                    "expected a sequence number",
642                ));
643            };
644            if !(self.lowest..=self.head).contains(&sequence) {
645                return Err(tonic::Status::not_found("pruned"));
646            }
647            let mut summary = CheckpointSummary::default();
648            summary.timestamp = Some(timestamp_ms_to_proto((GENESIS_SECS + sequence) * 1_000));
649            let mut checkpoint = Checkpoint::default();
650            checkpoint.sequence_number = Some(sequence);
651            checkpoint.summary = Some(summary);
652            let mut response = GetCheckpointResponse::default();
653            response.checkpoint = Some(checkpoint);
654            Ok(tonic::Response::new(response))
655        }
656    }

658    /// A ledger holding the package `PACKAGE_ID`, published by the transaction
659    /// `PUBLISH_DIGEST` at `PUBLISH_SECS`. Each `None` leaves that field out.
660    #[derive(Clone)]
661    struct PublishedPackageLedger {
662        previous_transaction: Option<String>,
663        timestamp: Option<UnixSeconds>,
664    }
665
666    const PUBLISH_DIGEST: &str = "11111111111111111111111111111111";
667    const PUBLISH_CHECKPOINT: u64 = 4_200;
668    const PUBLISH_SECS: UnixSeconds = 1_690_000_000;
669
670    #[tonic::async_trait]
671    impl LedgerService for PublishedPackageLedger {
672        async fn get_object(
673            &self,
674            request: tonic::Request<GetObjectRequest>,
675        ) -> Result<tonic::Response<GetObjectResponse>, tonic::Status> {
676            if request.into_inner().object_id != Some(PACKAGE_ID.to_string()) {
677                return Err(tonic::Status::not_found("no such object"));
678            }
679            let mut object = Object::default();
680            object.previous_transaction = self.previous_transaction.clone();
681            Ok(tonic::Response::new(GetObjectResponse::new(object)))
682        }
683
684        async fn get_transaction(
685            &self,
686            request: tonic::Request<GetTransactionRequest>,
687        ) -> Result<tonic::Response<GetTransactionResponse>, tonic::Status> {
688            if request.into_inner().digest.as_deref() != Some(PUBLISH_DIGEST) {
689                return Err(tonic::Status::not_found("no such transaction"));
690            }
691            let mut transaction = ExecutedTransaction::default();
692            transaction.checkpoint = Some(PUBLISH_CHECKPOINT);
693            transaction.timestamp = self
694                .timestamp
695                .map(|seconds| timestamp_ms_to_proto(seconds * 1_000));
696            Ok(tonic::Response::new(GetTransactionResponse::new(
697                transaction,
698            )))
699        }
700    }

719    /// Serve `ledger` on a local port and return the config that reaches it.
720    async fn serve(ledger: impl LedgerService) -> SuiConfig {
721        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
722        let addr = listener.local_addr().unwrap();
723        let incoming = futures::stream::unfold(listener, |listener| async move {
724            let result = listener.accept().await.map(|(stream, _)| stream);
725            Some((result, listener))
726        });
727        tokio::spawn(
728            tonic::transport::Server::builder()
729                .add_service(LedgerServiceServer::new(ledger))
730                .serve_with_incoming(incoming),
731        );
732        SuiConfig {
733            rpc_url: format!("http://{addr}"),
734            package_id: PACKAGE_ID.to_string(),
735        }
736    }

884    /// `PrunedLedger`'s checkpoints, plus a `ListTransactions` that records each
885    /// requested range and answers it with no transactions, up to its bound.
886    #[derive(Clone)]
887    struct ScanningLedger {
888        checkpoints: PrunedLedger,
889        ranges: Arc<Mutex<Vec<(u64, u64)>>>,
890    }
891
892    #[tonic::async_trait]
893    impl LedgerService for ScanningLedger {
894        async fn get_service_info(
895            &self,
896            request: tonic::Request<GetServiceInfoRequest>,
897        ) -> Result<tonic::Response<GetServiceInfoResponse>, tonic::Status> {
898            self.checkpoints.get_service_info(request).await
899        }
900
901        async fn get_checkpoint(
902            &self,
903            request: tonic::Request<GetCheckpointRequest>,
904        ) -> Result<tonic::Response<GetCheckpointResponse>, tonic::Status> {
905            self.checkpoints.get_checkpoint(request).await
906        }
907
908        async fn list_transactions(
909            &self,
910            request: tonic::Request<ListTransactionsRequest>,
911        ) -> Result<
912            tonic::Response<tonic::codegen::BoxStream<ListTransactionsResponse>>,
913            tonic::Status,
914        > {
915            let request = request.into_inner();
916            let (start, end) = (
917                request.start_checkpoint.unwrap_or(0),
918                request.end_checkpoint.unwrap_or(0),
919            );
920            self.ranges.lock().unwrap().push((start, end));
921            let mut watermark = Watermark::default();
922            watermark.cursor = Some(b"bound".to_vec().into());
923            watermark.checkpoint = Some(end - 1);
924            let mut query_end = QueryEnd::default();
925            query_end.reason = Some(QueryEndReason::CheckpointBound as i32);
926            let mut frame = ListTransactionsResponse::default();
927            frame.watermark = Some(watermark);
928            frame.end = Some(query_end);
929            Ok(tonic::Response::new(Box::pin(futures::stream::once(
930                async { Ok(frame) },
931            ))))
932        }
933    }

935    #[tokio::test]
936    async fn a_poll_scans_at_most_the_range_cap_and_the_next_poll_continues() {
937        let ledger = ScanningLedger {
938            checkpoints: PrunedLedger {
939                lowest: 0,
940                head: 3 * MAX_CHECKPOINTS_PER_POLL,
941            },
942            ranges: Arc::default(),
943        };
944        let mut poller = poller_for(ledger.clone()).await;
945
946        for expected_cursor_checkpoint in [MAX_CHECKPOINTS_PER_POLL, 2 * MAX_CHECKPOINTS_PER_POLL] {
947            assert!(
948                poller
949                    .poll(3 * MAX_CHECKPOINTS_PER_POLL)
950                    .await
951                    .unwrap()
952                    .is_some()
953            );
954            assert_eq!(poller.next_item_to_poll(), expected_cursor_checkpoint);
955            assert_eq!(
956                poller.read_until(),
957                GENESIS_SECS + expected_cursor_checkpoint - 1
958            );
959        }
960        assert_eq!(
961            *ledger.ranges.lock().unwrap(),
962            vec![
963                (0, MAX_CHECKPOINTS_PER_POLL),
964                (MAX_CHECKPOINTS_PER_POLL, 2 * MAX_CHECKPOINTS_PER_POLL),
965            ]
966        );
