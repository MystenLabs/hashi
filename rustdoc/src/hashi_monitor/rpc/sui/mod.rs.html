4//! The Sui poller: the Hashi events read from Sui, in checkpoint order.
5//!
6//! Why it exists: the withdrawal E1 and the deposit E2 are Sui events. The
7//! poller scans the checkpoints for them and gives each one its time.
8//!
9//! How it works:
10//! 1. The poller keeps a cursor in seconds. On the first poll it finds the
11//!    checkpoint before the cursor by probing checkpoint timestamps, or
12//!    starts at the oldest checkpoint a pruning node serves.
13//! 2. Each poll lists the transactions that emit a monitored event, from the
14//!    next checkpoint through the one that covers the poll end. A failed
15//!    range is retried, then split.
16//! 3. The cursor advances through the last checkpoint the node declared
17//!    complete. The transactions of later checkpoints wait for the next poll,
18//!    so a checkpoint is ingested whole.
19//! 4. A Hashi approval missing from the scan is read from its
20//!    `WithdrawalTransaction` object by withdrawal id.
21//!
22//! Assumptions:
23//! - Every monitored event type keeps the original package's address across
24//!   upgrades.
25//! - Checkpoint timestamps are cached. A checkpoint's time never changes.
26

64const REQUEST_TIMEOUT: Duration = Duration::from_secs(60);
65const PAGE_SIZE: u32 = 1_000;
66const MIN_CHECKPOINTS_PER_RETRY: u64 = 25;
67const MAX_RANGE_ATTEMPTS: u32 = 3;
68const MAX_LOOKUP_ATTEMPTS: u32 = 3;
69const INITIAL_RETRY_DELAY: Duration = Duration::from_millis(250);
70/// Distance kept above a pruning node's reported oldest checkpoint: an hour at the probe's
71/// six checkpoints per second, since the backends behind one endpoint prune unevenly.
72const PRUNING_MARGIN_CHECKPOINTS: u64 = 6 * 60 * 60;

74struct TransactionScan {
75    transactions: Vec<ExecutedTransaction>,
76    completed_checkpoint: Option<u64>,
77    end_reason: QueryEndReason,
78}

80fn completed_checkpoint_for_scan(
81    start_checkpoint: u64,
82    end_checkpoint: u64,
83    end_reason: QueryEndReason,
84    watermark_checkpoint: Option<u64>,
85) -> anyhow::Result<Option<u64>> {
86    anyhow::ensure!(
87        start_checkpoint < end_checkpoint,
88        "empty Sui transaction checkpoint range"
89    );
90    let completed_checkpoint = match end_reason {
91        QueryEndReason::CheckpointBound => end_checkpoint - 1,
92        QueryEndReason::LedgerTip if watermark_checkpoint.is_none() => return Ok(None),
93        _ => watermark_checkpoint.context(format!(
94            "Sui transaction scan ended with {end_reason:?} without a completion watermark"
95        ))?,
96    };
97    anyhow::ensure!(
98        completed_checkpoint < end_checkpoint,
99        "Sui transaction scan watermark checkpoint {completed_checkpoint} is outside requested range ending at {end_checkpoint}"
100    );
101    Ok((completed_checkpoint >= start_checkpoint).then_some(completed_checkpoint))
102}

104/// Connection-level failures that a fresh connection can clear, as the node's peer
105/// retries classify them: a `GOAWAY` surfaces as `Internal`, a broken pipe as `Unknown`.
106fn is_transient(status: &tonic::Status) -> bool {
107    matches!(
108        status.code(),
109        tonic::Code::Unavailable
110            | tonic::Code::Unknown
111            | tonic::Code::Internal
112            | tonic::Code::Cancelled
113            | tonic::Code::DeadlineExceeded
114    )
115}

176    /// Whether the scan covered `timestamp_secs`. The cursor's own second is
177    /// excluded, since later unscanned checkpoints can share it.
178    pub fn has_scanned(&self, timestamp_secs: UnixSeconds) -> bool {
179        (self.start_seconds..self.cursor_seconds).contains(&timestamp_secs)
180    }
181
182    /// Scan through the checkpoint covering `up_to`, or the observed chain head.
183    ///
184    /// The timestamp cursor advances only through `watermark.checkpoint`, which
185    /// is the inclusive boundary that `ListTransactions` reports as fully
186    /// covered. Each transaction provides the checkpoint timestamp that
187    /// `DepositConfirmed` needs, and its nested events provide the Hashi
188    /// payloads. The Sui SDK handles item and scan limits, resumable cursors,
189    /// and retryable partial streams. A terminal failure discards the partial
190    /// range; that range is retried and may be split.
191    pub async fn poll(&mut self, up_to: UnixSeconds) -> anyhow::Result<PollOutcome> {
192        if up_to <= self.cursor_seconds {
193            return Ok(PollOutcome::CursorUnmoved);
194        }
195
196        let start_checkpoint = match self.next_checkpoint_to_scan {
197            Some(checkpoint) => checkpoint,
198            None => self.first_checkpoint_to_scan().await?,
199        };
200        let (mut latest_sequence, mut latest_timestamp) = self
201            .observed_chain_head
202            .context("latest Sui checkpoint was not resolved")?;
203        if start_checkpoint > latest_sequence {
204            (latest_sequence, latest_timestamp) = self.refresh_chain_head().await?;
205            if start_checkpoint > latest_sequence {
206                return Ok(PollOutcome::CursorUnmoved);
207            }
208        }
209
210        let mut end_checkpoint = if latest_timestamp < up_to {
211            latest_sequence.saturating_add(1)
212        } else {
213            let (_, boundary) = self
214                .checkpoint_bracket_from_head(up_to, (latest_sequence, latest_timestamp))
215                .await?
216                .context("Sui does not yet have a checkpoint at the poll end time")?;
217            boundary.saturating_add(1)
218        };
219        if end_checkpoint <= start_checkpoint {
220            return Ok(PollOutcome::CursorUnmoved);
221        }
222
223        let mut range_attempt = 0u32;
224        let scan = loop {
225            let scan = tokio::time::timeout(
226                REQUEST_TIMEOUT,
227                self.list_transactions_in_range(start_checkpoint, end_checkpoint),
228            )
229            .await
230            .context("Sui transaction scan timed out")
231            .and_then(|result| result);
232            match scan {
233                Ok(transactions) => break transactions,
234                Err(error) if range_attempt + 1 < MAX_RANGE_ATTEMPTS => {
235                    let delay = INITIAL_RETRY_DELAY.saturating_mul(1 << range_attempt);
236                    range_attempt += 1;
237                    tracing::warn!(
238                        start_checkpoint,
239                        end_checkpoint,
240                        range_attempt,
241                        ?delay,
242                        ?error,
243                        "Sui transaction scan failed; retrying range"
244                    );
245                    tokio::time::sleep(delay).await;
246                }
247                Err(error) => {
248                    let checkpoint_count = end_checkpoint.saturating_sub(start_checkpoint);
249                    if checkpoint_count <= MIN_CHECKPOINTS_PER_RETRY {
250                        return Err(error).context("Sui transaction scan failed after retries");
251                    }
252                    end_checkpoint = start_checkpoint + checkpoint_count / 2;
253                    tracing::warn!(
254                        start_checkpoint,
255                        end_checkpoint,
256                        ?error,
257                        "Sui transaction scan failed after retries; retrying a smaller range"
258                    );
259                    range_attempt = 0;
260                }
261            }
262        };
263
264        let Some(completed_checkpoint) = completed_checkpoint_for_scan(
265            start_checkpoint,
266            end_checkpoint,
267            scan.end_reason,
268            scan.completed_checkpoint,
269        )?
270        else {
271            return Ok(PollOutcome::CursorUnmoved);
272        };
273
274        let mut events = Vec::with_capacity(scan.transactions.len());
275        let mut findings = Vec::new();
276        for transaction in scan.transactions {
277            let checkpoint = transaction
278                .checkpoint
279                .context("filtered Sui transaction is missing checkpoint")?;
280            // A LedgerTip response can include transactions from a checkpoint
281            // that its watermark does not yet declare complete. Leave those
282            // transactions for the next poll so the checkpoint is ingested
283            // atomically and cannot be skipped as indexing catches up.
284            if checkpoint > completed_checkpoint {
285                continue;
286            }
287            self.parse_transaction(transaction, &mut events, &mut findings)?;
288        }
289
290        let scanned_through_timestamp = self.checkpoint_timestamp(completed_checkpoint).await?;
291        self.next_checkpoint_to_scan = Some(completed_checkpoint.saturating_add(1));
292        self.cursor_seconds = self.cursor_seconds.max(scanned_through_timestamp);
293        tracing::info!(
294            start_checkpoint,
295            end_checkpoint,
296            completed_checkpoint,
297            end_reason = ?scan.end_reason,
298            cursor = %utc_timestamp(self.cursor_seconds),
299            events = events.len(),
300            findings = findings.len(),
301            "completed Sui event range"
302        );
303        Ok(PollOutcome::CursorAdvanced { events, findings })
304    }

306    /// Read the Hashi approval of `wid` from its `WithdrawalTransaction` object, which is
307    /// created alongside `WithdrawalPickedForProcessing` with the same txid and timestamp.
308    pub async fn fetch_withdrawal_approval(
309        &mut self,
310        wid: WithdrawalID,
311    ) -> anyhow::Result<Option<HashiApproval>> {
312        let request = GetObjectRequest::new(&wid).with_read_mask(FieldMask::from_paths([
313            Object::path_builder().object_type(),
314            Object::path_builder().contents().finish(),
315        ]));
316        let lookup = async {
317            let mut attempt = 0u32;
318            loop {
319                match self
320                    .client
321                    .ledger_client()
322                    .get_object(request.clone())
323                    .await
324                {
325                    Ok(response) => {
326                        break response
327                            .into_inner()
328                            .object
329                            .context("Sui GetObject response is missing the object")
330                            .map(Some);
331                    }
332                    Err(status) if status.code() == tonic::Code::NotFound => break Ok(None),
333                    Err(status) if attempt + 1 < MAX_LOOKUP_ATTEMPTS && is_transient(&status) => {
334                        let delay = INITIAL_RETRY_DELAY.saturating_mul(1 << attempt);
335                        attempt += 1;
336                        tracing::warn!(%wid, attempt, ?delay, ?status, "Sui object lookup failed; retrying");
337                        tokio::time::sleep(delay).await;
338                    }
339                    Err(status) => {
340                        break Err(status).with_context(|| {
341                            format!("failed to fetch withdrawal transaction {wid}")
342                        });
343                    }
344                }
345            }
346        };
347        // Retries share one request's timeout, so a lookup never stalls the audit longer than one request.
348        let object = tokio::time::timeout(REQUEST_TIMEOUT, lookup)
349            .await
350            .with_context(|| format!("Sui lookup of withdrawal transaction {wid} timed out"))??;
351        match object {
352            Some(object) => approval::parse_withdrawal_object(&self.package_versions, wid, &object),
353            None => Ok(None),
354        }
355    }

357    /// The checkpoint before the cursor or, once the node has pruned that far, the oldest
358    /// one it serves, moving the scan's start past that checkpoint's second.
359    async fn first_checkpoint_to_scan(&mut self) -> anyhow::Result<u64> {
360        let (before, _) = self
361            .checkpoint_bracket(self.cursor_seconds)
362            .await?
363            .context("Sui does not yet have a checkpoint at the poll start time")?;
364        if before > 0 {
365            let before_timestamp = self.checkpoint_timestamp(before).await?;
366            if before_timestamp >= self.cursor_seconds {
367                // Transactions in unscanned earlier checkpoints are at most this second.
368                self.start_seconds = before_timestamp + 1;
369                tracing::warn!(
370                    requested_start = %utc_timestamp(self.cursor_seconds),
371                    scan_start = %utc_timestamp(self.start_seconds),
372                    checkpoint = before,
373                    "Sui node has pruned the requested start; scanning from its oldest checkpoint"
374                );
375            }
376        }
377        Ok(before)
378    }

380    /// Find a safe checkpoint bracket `(before, at_or_after)` for a timestamp.
381    ///
382    /// Exponential probing finds a lower bound, then binary search resolves the
383    /// exact adjacent checkpoint boundary. When a predecessor checkpoint exists,
384    /// the lower bound is before the timestamp and the upper bound is at or after
385    /// it. At or before the oldest checkpoint the node serves, genesis unless it
386    /// prunes, both bounds are that checkpoint.
387    async fn checkpoint_bracket(
388        &mut self,
389        timestamp_secs: UnixSeconds,
390    ) -> anyhow::Result<Option<(u64, u64)>> {
391        let head = self.refresh_chain_head().await?;
392        self.checkpoint_bracket_from_head(timestamp_secs, head)
393            .await
394    }
395
396    async fn checkpoint_bracket_from_head(
397        &mut self,
398        timestamp_secs: UnixSeconds,
399        (latest_sequence, latest_timestamp): (u64, UnixSeconds),
400    ) -> anyhow::Result<Option<(u64, u64)>> {
401        tracing::info!(
402            target = %utc_timestamp(timestamp_secs),
403            latest_checkpoint = latest_sequence,
404            latest_timestamp = %utc_timestamp(latest_timestamp),
405            "resolving Sui checkpoint boundary"
406        );
407        if latest_timestamp < timestamp_secs {
408            return Ok(None);
409        }
410
411        let oldest = match self.lowest_available_checkpoint {
412            0 => 0,
413            lowest => lowest
414                .saturating_add(PRUNING_MARGIN_CHECKPOINTS)
415                .min(latest_sequence),
416        };
417        let elapsed = latest_timestamp.saturating_sub(timestamp_secs);
418        let mut distance = elapsed.saturating_mul(6).max(1);
419        let mut low_sequence = loop {
420            let probe = latest_sequence.saturating_sub(distance).max(oldest);
421            let timestamp = self.checkpoint_timestamp(probe).await?;
422            if timestamp < timestamp_secs {
423                break probe;
424            }
425            if probe == oldest {
426                return Ok(Some((oldest, oldest)));
427            }
428            distance = distance.saturating_mul(2);
429        };
430        let mut high_sequence = latest_sequence;
431
432        // Resolve the exact adjacent checkpoint boundary. A previously capped
433        // interpolation could leave a very wide but technically safe bracket
434        // for historical timestamps, forcing ListTransactions to scan many
435        // unrelated checkpoint ranges. Binary search keeps the lookup
436        // logarithmic even when checkpoint production has varied over time.
437        while high_sequence > low_sequence.saturating_add(1) {
438            let midpoint = low_sequence + (high_sequence - low_sequence) / 2;
439            let midpoint_timestamp = self.checkpoint_timestamp(midpoint).await?;
440            if midpoint_timestamp < timestamp_secs {
441                low_sequence = midpoint;
442            } else {
443                high_sequence = midpoint;
444            }
445        }
446
447        tracing::info!(
448            target = %utc_timestamp(timestamp_secs),
449            before_checkpoint = low_sequence,
450            at_or_after_checkpoint = high_sequence,
451            "resolved Sui checkpoint boundary"
452        );
453        Ok(Some((low_sequence, high_sequence)))
454    }

456    async fn refresh_chain_head(&mut self) -> anyhow::Result<(u64, UnixSeconds)> {
457        let service_info = self
458            .client
459            .ledger_client()
460            .get_service_info(GetServiceInfoRequest::default())
461            .await
462            .context("failed to fetch Sui service info")?
463            .into_inner();
464        let sequence_number = service_info
465            .checkpoint_height
466            .context("Sui service info is missing checkpoint_height")?;
467        let timestamp = service_info
468            .timestamp
469            .context("Sui service info is missing timestamp")?;
470        let timestamp_ms =
471            proto_to_timestamp_ms(timestamp).context("invalid Sui service info timestamp")?;
472        let latest = (sequence_number, unix_millis_to_seconds(timestamp_ms));
473        self.checkpoint_timestamps.insert(latest.0, latest.1);
474        self.observed_chain_head = Some(latest);
475        self.lowest_available_checkpoint = service_info.lowest_available_checkpoint.unwrap_or(0);
476        Ok(latest)
477    }

479    /// The timestamp of checkpoint `sequence_number`, from the cache or the node.
480    async fn checkpoint_timestamp(&mut self, sequence_number: u64) -> anyhow::Result<UnixSeconds> {
481        if let Some(timestamp) = self.checkpoint_timestamps.get(&sequence_number) {
482            return Ok(*timestamp);
483        }
484        let request = GetCheckpointRequest::by_sequence_number(sequence_number).with_read_mask(
485            FieldMask::from_paths(["sequence_number", "summary.timestamp"]),
486        );
487        let checkpoint = self
488            .client
489            .ledger_client()
490            .get_checkpoint(request)
491            .await
492            .context("failed to fetch Sui checkpoint")?
493            .into_inner()
494            .checkpoint
495            .context("missing Sui checkpoint")?;
496        let actual_sequence = checkpoint
497            .sequence_number
498            .context("Sui checkpoint is missing sequence_number")?;
499        let timestamp = checkpoint
500            .summary
501            .and_then(|summary| summary.timestamp)
502            .context("Sui checkpoint is missing summary.timestamp")?;
503        let timestamp_ms =
504            proto_to_timestamp_ms(timestamp).context("invalid Sui checkpoint timestamp")?;
505        let timestamp_secs = unix_millis_to_seconds(timestamp_ms);
506        self.checkpoint_timestamps
507            .insert(actual_sequence, timestamp_secs);
508        anyhow::ensure!(
509            actual_sequence == sequence_number,
510            "requested Sui checkpoint {sequence_number}, received {actual_sequence}"
511        );
512        Ok(timestamp_secs)
513    }

515    async fn list_transactions_in_range(
516        &mut self,
517        start_checkpoint: u64,
518        end_checkpoint: u64,
519    ) -> anyhow::Result<TransactionScan> {
520        if start_checkpoint >= end_checkpoint {
521            anyhow::bail!("empty Sui transaction checkpoint range");
522        }
523
524        let request = ListTransactionsRequest::default()
525            .with_read_mask(FieldMask::from_paths([
526                "timestamp",
527                "checkpoint",
528                "events.events.event_type",
529                "events.events.contents",
530            ]))
531            .with_start_checkpoint(start_checkpoint)
532            .with_end_checkpoint(end_checkpoint)
533            .with_filter(self.transaction_filter())
534            .with_options(
535                QueryOptions::default()
536                    .with_limit(PAGE_SIZE)
537                    .with_ordering(Ordering::Ascending),
538            );
539        let stream = self.client.list_transactions(request);
540        futures::pin_mut!(stream);
541
542        let mut all_transactions = Vec::new();
543        let mut completed_checkpoint = None;
544        let mut end_reason = None;
545        tracing::info!(
546            start_checkpoint,
547            end_checkpoint,
548            "starting Sui transaction scan"
549        );
550
551        while let Some(frame) = stream.next().await {
552            // The SDK validates watermarks and transparently resumes page
553            // limits and retryable partial streams before yielding frames.
554            let frame = frame.context("Sui ListTransactions stream failed")?;
555            if let Some(checkpoint) = frame
556                .watermark
557                .as_ref()
558                .and_then(|watermark| watermark.checkpoint)
559            {
560                completed_checkpoint = Some(checkpoint);
561            }
562            if let Some(transaction) = frame.transaction {
563                all_transactions.push(transaction);
564            }
565            if let Some(end) = frame.end {
566                end_reason = end
567                    .reason
568                    .and_then(|reason| QueryEndReason::try_from(reason).ok());
569            }
570        }
571
572        Ok(TransactionScan {
573            transactions: all_transactions,
574            completed_checkpoint,
575            end_reason: end_reason.context("Sui ListTransactions ended without QueryEnd")?,
576        })
577    }

579    fn transaction_filter(&self) -> TransactionFilter {
580        let event_types = [
581            format!(
582                "{}::withdrawal_queue::WithdrawalPickedForProcessing",
583                self.package_id
584            ),
585            format!("{}::deposit::DepositConfirmed", self.package_id),
586        ];
587        TransactionFilter::any(event_types.into_iter().map(tx_filter::event_type))
588    }

590    /// Parse the monitored events of one transaction into `events`. An event
591    /// that fails a check adds its findings to `findings` instead.
592    fn parse_transaction(
593        &self,
594        transaction: ExecutedTransaction,
595        events: &mut Vec<MonitorEvent>,
596        findings: &mut Vec<MonitorFinding>,
597    ) -> anyhow::Result<()> {
598        let timestamp = transaction
599            .timestamp
600            .context("Sui transaction is missing checkpoint timestamp")?;
601        let timestamp_ms =
602            proto_to_timestamp_ms(timestamp).context("invalid Sui transaction timestamp")?;
603        let timestamp_secs = unix_millis_to_seconds(timestamp_ms);
604        let sui_events = transaction
605            .events
606            .context("filtered Sui transaction is missing events")?
607            .events;
608
609        for sui_event in sui_events {
610            approval::parse_event(
611                &self.package_versions,
612                &self.hashi_btc_keys,
613                sui_event,
614                timestamp_secs,
615                events,
616                findings,
617            )?;
618        }
619        Ok(())
620    }

650    /// A ledger holding at most one object, answering every other id with
651    /// `miss`. Like a fullnode, it serves only the fields in the read mask.
652    #[derive(Clone)]
653    struct OneObjectLedger {
654        object: Option<Object>,
655        miss: tonic::Code,
656    }
657
658    #[tonic::async_trait]
659    impl LedgerService for OneObjectLedger {
660        async fn get_object(
661            &self,
662            request: tonic::Request<GetObjectRequest>,
663        ) -> Result<tonic::Response<GetObjectResponse>, tonic::Status> {
664            let request = request.into_inner();
665            let Some(stored) = self
666                .object
667                .as_ref()
668                .filter(|object| object.object_id == request.object_id)
669            else {
670                return Err(tonic::Status::new(self.miss, "no such object"));
671            };
672            let paths = request.read_mask.map(|mask| mask.paths).unwrap_or_default();
673            let mut object = Object::default();
674            if paths.iter().any(|path| path == "object_type") {
675                object.object_type = stored.object_type.clone();
676            }
677            if paths.iter().any(|path| path == "contents") {
678                object.contents = stored.contents.clone();
679            }
680            Ok(tonic::Response::new(GetObjectResponse::new(object)))
681        }
682    }

684    /// Answers the first requests with `failures`, in order, then defers to `ledger`.
685    #[derive(Clone)]
686    struct FlakyLedger {
687        failures: Arc<Mutex<VecDeque<tonic::Code>>>,
688        requests: Arc<Mutex<u32>>,
689        ledger: OneObjectLedger,
690    }
691
692    impl FlakyLedger {
693        fn new(failures: impl IntoIterator<Item = tonic::Code>, ledger: OneObjectLedger) -> Self {
694            Self {
695                failures: Arc::new(Mutex::new(failures.into_iter().collect())),
696                requests: Arc::default(),
697                ledger,
698            }
699        }
700    }
701
702    #[tonic::async_trait]
703    impl LedgerService for FlakyLedger {
704        async fn get_object(
705            &self,
706            request: tonic::Request<GetObjectRequest>,
707        ) -> Result<tonic::Response<GetObjectResponse>, tonic::Status> {
708            *self.requests.lock().unwrap() += 1;
709            let failure = self.failures.lock().unwrap().pop_front();
710            match failure {
711                Some(code) => Err(tonic::Status::new(code, "injected failure")),
712                None => self.ledger.get_object(request).await,
713            }
714        }
715    }

717    /// Checkpoints `lowest..=head`, one a second from `GENESIS_SECS`. Like a pruning
718    /// fullnode, it reports `lowest` and answers older checkpoints with `NotFound`.
719    #[derive(Clone)]
720    struct PrunedLedger {
721        lowest: u64,
722        head: u64,
723    }
724
725    const GENESIS_SECS: UnixSeconds = 1_700_000_000;
726
727    #[tonic::async_trait]
728    impl LedgerService for PrunedLedger {
729        async fn get_service_info(
730            &self,
731            _: tonic::Request<GetServiceInfoRequest>,
732        ) -> Result<tonic::Response<GetServiceInfoResponse>, tonic::Status> {
733            let mut info = GetServiceInfoResponse::default();
734            info.checkpoint_height = Some(self.head);
735            info.timestamp = Some(timestamp_ms_to_proto((GENESIS_SECS + self.head) * 1_000));
736            info.lowest_available_checkpoint = Some(self.lowest);
737            Ok(tonic::Response::new(info))
738        }
739
740        async fn get_checkpoint(
741            &self,
742            request: tonic::Request<GetCheckpointRequest>,
743        ) -> Result<tonic::Response<GetCheckpointResponse>, tonic::Status> {
744            let Some(CheckpointId::SequenceNumber(sequence)) = request.into_inner().checkpoint_id
745            else {
746                return Err(tonic::Status::invalid_argument(
747                    "expected a sequence number",
748                ));
749            };
750            if !(self.lowest..=self.head).contains(&sequence) {
751                return Err(tonic::Status::not_found("pruned"));
752            }
753            let mut summary = CheckpointSummary::default();
754            summary.timestamp = Some(timestamp_ms_to_proto((GENESIS_SECS + sequence) * 1_000));
755            let mut checkpoint = Checkpoint::default();
756            checkpoint.sequence_number = Some(sequence);
757            checkpoint.summary = Some(summary);
758            let mut response = GetCheckpointResponse::default();
759            response.checkpoint = Some(checkpoint);
760            Ok(tonic::Response::new(response))
761        }
762    }

764    async fn poller_for(ledger: impl LedgerService) -> SuiEventsPoller {
765        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
766        let addr = listener.local_addr().unwrap();
767        let incoming = futures::stream::unfold(listener, |listener| async move {
768            let result = listener.accept().await.map(|(stream, _)| stream);
769            Some((result, listener))
770        });
771        tokio::spawn(
772            tonic::transport::Server::builder()
773                .add_service(LedgerServiceServer::new(ledger))
774                .serve_with_incoming(incoming),
775        );
776        let config = SuiConfig {
777            rpc_url: format!("http://{addr}"),
778            package_id: PACKAGE_ID.to_string(),
779        };
780        SuiEventsPoller::new(&config, test_hashi_btc_keys(), 0).unwrap()
781    }

800    /// A poller over a ledger holding `txn` at `WID` that has scanned `[start, cursor)`.
801    pub async fn poller_scanned_with(
802        txn: WithdrawalTransaction,
803        start: UnixSeconds,
804        cursor: UnixSeconds,
805    ) -> SuiEventsPoller {
806        let mut poller = poller_for(OneObjectLedger {
807            object: Some(object_at_wid(PACKAGE_ID, &txn)),
808            miss: tonic::Code::NotFound,
809        })
810        .await;
811        poller.start_seconds = start;
812        poller.cursor_seconds = cursor;
813        poller
814    }
