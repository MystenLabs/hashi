4//! The monitor's own record of the UTXOs the bridge holds, built from Bitcoin facts.
5//!
6//! Why it exists: the committee writes Move's `utxo_records`. A compromised
7//! committee can mark a UTXO spent on Move without spending it, or keep a
8//! deposit out of Move, then spend the UTXO alone when the 60-day recovery
9//! leaf opens. Only a Bitcoin fact changes this pool, so the committee
10//! cannot remove a UTXO from it or keep one out.
11//!
12//! How it works:
13//! 1. The pool starts empty at the bridge's genesis. The snapshot keeps it
14//!    across restarts.
15//! 2. A deposit enters when Bitcoin confirms the request's output at the
16//!    bridge script of the claimed path.
17//! 3. A complete withdrawal (E1, E2, E3) removes its inputs and adds its
18//!    change. The guardian's E2 labels the change; the monitor verifies it.
19//! 4. In one pass, every add comes before every spend.
20//! 5. A second add of a held UTXO is a repeat. A second amount or script is
21//!    an `BtcRpcInconsistency`. A spend of a UTXO the pool does not hold is
22//!    `UtxoUnknownInputSpent`.
23//! 6. A `DepositApproved` must match a held deposit's amount and script.
24//! 7. Once a day, `check` asks Bitcoin about every held UTXO. It reports old,
25//!    spent and missing UTXOs.
26//! 8. The pool keeps no record of spent UTXOs. Move refuses a request after
27//!    the first DC, so every add for a UTXO comes before its spend.
28//!
29//! The pool can hold a UTXO that Move never holds, for example a deposit
30//! that AML screening rejects or a request with a wrong amount.
31//!
32//! Assumptions:
33//! - The first run starts at genesis, so the monitor sees every UTXO enter.
34//! - Sui and the guardian deliver each event once: no repeat and no miss.
35

55/// A UTXO the bridge holds, by Bitcoin's account.
56#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
57pub struct Utxo {
58    pub amount: u64,
59    /// The bridge script the output pays.
60    pub script_pubkey: ScriptBuf,
61    pub origin: UtxoOrigin,
62    /// The time of the block that holds the UTXO.
63    pub first_seen: UnixSeconds,
64}
65
66/// A held UTXO that Bitcoin reports spent, which no followed withdrawal
67/// explains. The auditor judges it once the Sui cursor is past `checked_at`.
68#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
69pub struct SuspectSpend {
70    /// When the Bitcoin lookups finished.
71    pub checked_at: UnixSeconds,
72    pub amount: u64,
73}

75/// The UTXOs the bridge holds, by Bitcoin's account.
76#[derive(Clone, Debug, Default, Eq, PartialEq)]
77pub struct UtxoPool {
78    utxos: BTreeMap<OutPoint, Utxo>,
79}

81/// A held UTXO as stored: a JSON map needs string keys, so a list.
82#[derive(Deserialize, Serialize)]
83pub struct UtxoWire {
84    outpoint: OutPoint,
85    #[serde(flatten)]
86    utxo: Utxo,
87}

89impl UtxoPool {
90    /// The held UTXOs, ready to store.
91    pub fn held_to_wire(&self) -> Vec<UtxoWire> {
92        self.utxos
93            .iter()
94            .map(|(outpoint, utxo)| UtxoWire {
95                outpoint: *outpoint,
96                utxo: utxo.clone(),
97            })
98            .collect()
99    }
100
101    /// A pool from its stored UTXOs.
102    pub fn from_wire(held: Vec<UtxoWire>) -> Self {
103        Self {
104            utxos: held
105                .into_iter()
106                .map(|wire| (wire.outpoint, wire.utxo))
107                .collect(),
108        }
109    }
110
111    /// Whether the pool holds the UTXO.
112    pub fn holds(&self, outpoint: &OutPoint) -> bool {
113        self.utxos.contains_key(outpoint)
114    }
115
116    /// The number of UTXOs held.
117    pub fn len(&self) -> usize {
118        self.utxos.len()
119    }
120
121    /// Whether no UTXO is held.
122    pub fn is_empty(&self) -> bool {
123        self.utxos.is_empty()
124    }
125
126    /// The earliest block timestamp among held UTXOs.
127    /// The sum of the held amounts, in satoshis.
128    pub fn value(&self) -> u64 {
129        self.utxos
130            .values()
131            .fold(0, |sum, utxo| sum.saturating_add(utxo.amount))
132    }
133
134    pub fn oldest_timestamp(&self) -> Option<UnixSeconds> {
135        self.utxos.values().map(|utxo| utxo.first_seen).min()
136    }
137
138    /// The outpoints held, in order.
139    pub fn outpoints(&self) -> Vec<OutPoint> {
140        self.utxos.keys().copied().collect()
141    }
142
143    /// Add a deposit Bitcoin confirmed.
144    pub fn add_deposit(&mut self, deposit: &ConfirmedDeposit) -> Option<MonitorFinding> {
145        self.add(
146            deposit.deposit_id.into(),
147            Utxo {
148                amount: deposit.amount,
149                script_pubkey: deposit.script_pubkey.clone(),
150                origin: UtxoOrigin::Deposit,
151                first_seen: deposit.block_time,
152            },
153        )
154    }
155
156    /// Check the Sui `DepositApproved` against the pool: the UTXO must be a
157    /// deposit the pool holds. See `SuiDepositApprovedEvent::judge`.
158    pub fn check_sui_approval(
159        &self,
160        sui: &SuiDepositApprovedEvent,
161        skew: u64,
162    ) -> Vec<MonitorFinding> {
163        let outpoint = OutPoint::from(sui.deposit_id);
164        let paid = self
165            .utxos
166            .get(&outpoint)
167            .filter(|utxo| utxo.origin == UtxoOrigin::Deposit)
168            .map(|utxo| ConfirmedDeposit {
169                deposit_id: sui.deposit_id,
170                amount: utxo.amount,
171                script_pubkey: utxo.script_pubkey.clone(),
172                block_time: utxo.first_seen,
173            });
174        sui.judge(paid.as_ref(), skew)
175    }
176
177    /// Apply withdrawals Bitcoin confirmed: every change output enters,
178    /// then every input is spent. See the module doc for why in that order.
179    pub fn add_withdrawals(&mut self, withdrawals: &[ConfirmedWithdrawal]) -> Vec<MonitorFinding> {
180        let mut findings = Vec::new();
181        for withdrawal in withdrawals {
182            findings.extend(self.add_change_outputs(withdrawal));
183        }
184        for withdrawal in withdrawals {
185            for input in &withdrawal.confirmation.inputs {
186                findings.extend(self.spend(*input, withdrawal.txid));
187            }
188        }
189        findings
190    }
191
192    /// Add a withdrawal's change outputs: every output from the first
193    /// change vout on.
194    fn add_change_outputs(&mut self, withdrawal: &ConfirmedWithdrawal) -> Vec<MonitorFinding> {
195        let outputs = &withdrawal.confirmation.outputs;
196        (withdrawal.first_change_vout..outputs.len())
197            .filter_map(|vout| {
198                let outpoint = OutPoint {
199                    txid: withdrawal.txid,
200                    vout: vout as u32,
201                };
202                self.add(
203                    outpoint,
204                    Utxo {
205                        amount: outputs[vout].value.to_sat(),
206                        script_pubkey: outputs[vout].script_pubkey.clone(),
207                        origin: UtxoOrigin::Change,
208                        first_seen: withdrawal.confirmation.block_time,
209                    },
210                )
211            })
212            .collect()
213    }
214
215    /// Add a UTXO. One already held with the same amount and script is a
216    /// repeat and keeps its entry. One that differs is a finding.
217    fn add(&mut self, outpoint: OutPoint, utxo: Utxo) -> Option<MonitorFinding> {
218        let Some(held) = self.utxos.get(&outpoint) else {
219            self.utxos.insert(outpoint, utxo);
220            return None;
221        };
222        let same = held.amount == utxo.amount && held.script_pubkey == utxo.script_pubkey;
223        (!same).then_some(MonitorFinding::BtcRpcInconsistency(
224            BtcRpcInconsistency::UtxoMismatch {
225                outpoint,
226                held_amount: held.amount,
227                seen_amount: utxo.amount,
228                held_script: held.script_pubkey.clone(),
229                seen_script: utxo.script_pubkey,
230            },
231        ))
232    }
233
234    /// Spend a UTXO by the transaction `txid`. One the pool does not hold is
235    /// a finding.
236    fn spend(&mut self, outpoint: OutPoint, txid: Txid) -> Option<MonitorFinding> {
237        self.utxos
238            .remove(&outpoint)
239            .is_none()
240            .then_some(MonitorFinding::UtxoUnknownInputSpent { outpoint, txid })
241    }
242
243    /// Check held outputs against Bitcoin results and return findings and suspect spends.
244    /// Use the lookup completion time `now` for ages and suspect timestamps.
245    /// The auditor judges suspects after Sui polling catches up.
246    pub fn check(
247        &self,
248        states: &HashMap<OutPoint, OutpointState>,
249        now: UnixSeconds,
250        timing: &Timing,
251    ) -> (Vec<MonitorFinding>, HashMap<OutPoint, SuspectSpend>) {
252        let mut findings = Vec::new();
253        let mut suspects = HashMap::new();
254        for (outpoint, utxo) in &self.utxos {
255            match states.get(outpoint) {
256                Some(OutpointState::Unspent { block_time }) => {
257                    let age_secs = now.saturating_sub(*block_time);
258                    if age_secs >= timing.utxo_max_age {
259                        findings.push(MonitorFinding::UtxoTooOld {
260                            outpoint: *outpoint,
261                            age_secs,
262                            threshold_secs: timing.utxo_max_age,
263                        });
264                    }
265                }
266                Some(OutpointState::Spent) => {
267                    suspects.insert(
268                        *outpoint,
269                        SuspectSpend {
270                            checked_at: now,
271                            amount: utxo.amount,
272                        },
273                    );
274                }
275                Some(OutpointState::Missing) | None => {
276                    findings.push(MonitorFinding::BtcRpcInconsistency(
277                        BtcRpcInconsistency::UtxoVanished {
278                            outpoint: *outpoint,
279                            first_seen: utxo.first_seen,
280                        },
281                    ));
282                }
283            }
284        }
285        (findings, suspects)
286    }
287}
