17/// Stable identifier for a monitored deposit: its Bitcoin outpoint.
18#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
19pub struct DepositId(OutPoint);
20
21impl DepositId {
22    pub fn new(txid: Txid, vout: u32) -> Self {
23        Self(OutPoint { txid, vout })
24    }
25
26    pub fn txid(self) -> Txid {
27        self.0.txid
28    }
29
30    pub fn vout(self) -> u32 {
31        self.0.vout
32    }
33}
34
35impl fmt::Display for DepositId {
36    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
37        self.0.fmt(formatter)
38    }
39}

41#[derive(Clone, Debug, Eq, PartialEq)]
42pub struct MonitorDepositEvent {
43    pub event_type: DepositEventType,
44    pub timestamp_secs: UnixSeconds,
45    pub deposit_id: DepositId,
46    /// Satoshis in the deposit output. Sui gives the claimed amount, Bitcoin
47    /// the paid amount.
48    pub amount: u64,
49    /// Script the deposit output pays. Sui gives the bridge address of the
50    /// claimed derivation path, Bitcoin the paid script.
51    pub script_pubkey: ScriptBuf,
52}
53
54impl MonitorDepositEvent {
55    /// E2 from Hashi's `DepositConfirmed`. Its Move payload has no
56    /// timestamp, so the containing checkpoint's time is used. The claim's
57    /// derivation path becomes the bridge script the state machine compares
58    /// with the paid output.
59    pub fn from_sui(
60        event: DepositConfirmed,
61        checkpoint_timestamp_secs: UnixSeconds,
62        keys: &HashiBTCKeys,
63    ) -> Self {
64        Self {
65            event_type: DepositEventType::E2HashiDeposited,
66            timestamp_secs: checkpoint_timestamp_secs,
67            deposit_id: DepositId::new(event.utxo.id.txid.into(), event.utxo.id.vout),
68            amount: event.utxo.amount,
69            script_pubkey: keys.script_pubkey(event.utxo.derivation_path.as_ref()),
70        }
71    }
72
73    /// E1 from Bitcoin's confirmation: the output at `deposit_id`'s vout, or
74    /// `None` if the transaction has no output there.
75    pub fn from_btc(confirmation: &BtcConfirmation, deposit_id: DepositId) -> Option<Self> {
76        let output = confirmation.outputs.get(deposit_id.vout() as usize)?;
77        Some(Self {
78            event_type: DepositEventType::E1BtcConfirmed,
79            timestamp_secs: confirmation.confirmed_time,
80            deposit_id,
81            amount: output.value.to_sat(),
82            script_pubkey: output.script_pubkey.clone(),
83        })
84    }
85}
86
87#[derive(Copy, Clone, Debug, Eq, PartialEq)]
88pub enum DepositEventType {
89    /// deposit confirmed event on btc
90    E1BtcConfirmed,
91    /// `DepositConfirmed` on Sui. Its timestamp is the containing checkpoint's
92    /// timestamp, supplied with the transaction by Sui gRPC.
93    E2HashiDeposited,
94}
