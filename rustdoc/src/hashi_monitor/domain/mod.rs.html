4//! The monitor's core types: the events it follows, their ids, and the
5//! cursors of its sources.
6//!
7//! Why it exists: every source, Sui, the guardian's S3 logs and Bitcoin, is
8//! parsed into one of these events, and the state machines reason about the
9//! events alone. Each event type has a constructor named for its source, so
10//! the mapping from a source's shape to the monitor's lives next to the type.
11//!
12//! The flows:
13//! - A withdrawal is three events: E1, the Hashi approval on Sui; E2, the
14//!   guardian's signature in its log; E3, the transaction confirmed on
15//!   Bitcoin. Every event states the transaction in Move's shape,
16//!   `WithdrawalTx`, so the sources can be compared as a whole.
17//! - A deposit is three events: E1, the user's `DepositRequested` on Sui;
18//!   E2, the output confirmed on Bitcoin; E3, the `DepositApproved` on
19//!   Sui, after which Move mints. Each request is followed on its own.
20//!
21//! Formatting of times lives in `time`, and the bridge keys in `keys`.
22

24pub mod utxo_pool;
25pub mod withdrawal;

47#[derive(Clone, Debug, Eq, PartialEq)]
48pub enum MonitorEvent {
49    Withdrawal(MonitorWithdrawalEvent),
50    DepositApproved(SuiDepositApprovedEvent),
51    DepositRequested(SuiDepositRequestedEvent),
52}
53
54impl fmt::Display for MonitorEvent {
55    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
56        match self {
57            Self::Withdrawal(event) => write!(
58                formatter,
59                "Withdrawal(type={:?}, wid={}, timestamp={}, btc_txid={})",
60                event.event_type,
61                event.wid,
62                utc_timestamp(event.timestamp_secs),
63                event.btc_txid,
64            ),
65            Self::DepositApproved(event) => write!(
66                formatter,
67                "DepositApproved(request_id={}, deposit_id={}, timestamp={}, amount={}, script_pubkey={:x})",
68                event.request_id,
69                event.deposit_id,
70                utc_timestamp(event.timestamp_secs),
71                event.amount,
72                event.script_pubkey,
73            ),
74            Self::DepositRequested(event) => write!(
75                formatter,
76                "DepositRequested(request_id={}, deposit_id={}, timestamp={}, amount={}, script_pubkey={:x})",
77                event.request_id,
78                event.deposit_id,
79                utc_timestamp(event.timestamp_secs),
80                event.amount,
81                event.script_pubkey,
82            ),
83        }
84    }
85}

87/// A transaction mined at least Hashi's confirmation threshold deep.
88#[derive(Clone, Debug, Deserialize, PartialEq, Serialize)]
89pub struct BtcConfirmation {
90    /// The time of the block that holds the transaction. The recovery leaf's
91    /// clock starts at this block, so a UTXO's age counts from here.
92    pub block_time: UnixSeconds,
93    /// The time of the block that gave the transaction the threshold depth.
94    /// The bridge acts on that block, so the Bitcoin events carry this time.
95    pub confirmed_time: UnixSeconds,
96    /// The outpoints the transaction spends, in order.
97    pub inputs: Vec<OutPoint>,
98    /// The transaction's outputs, in order.
99    pub outputs: Vec<TxOut>,
100}

102/// How a UTXO entered the monitor's UTXO pool.
103#[derive(Clone, Copy, Debug, Deserialize, Eq, PartialEq, Serialize)]
104pub enum UtxoOrigin {
105    /// A deposit Bitcoin confirmed at a bridge address, with a Sui
106    /// `DepositApproved`.
107    Deposit,
108    /// A deposit Bitcoin confirmed at a bridge address, with no Sui
109    /// `DepositApproved` yet. For example, a sanction or a user mistake keeps
110    /// a deposit here.
111    UnapprovedDeposit {
112        /// The time of the latest request for the deposit. The approval
113        /// deadline counts from it.
114        requested_at: UnixSeconds,
115        /// The amount that the latest request claims.
116        requested_amount: u64,
117    },
118    /// A change output of a withdrawal Bitcoin confirmed.
119    Change,
120}
121
122impl UtxoOrigin {
123    /// Whether Move's pool holds the UTXO: an approved deposit, or change.
124    /// Only these count in the pool value. Move adds an approved deposit
125    /// after the deposit delay.
126    pub fn in_move_pool(self) -> bool {
127        !matches!(self, Self::UnapprovedDeposit { .. })
128    }
129
130    fn is_change(self) -> bool {
131        self == Self::Change
132    }
133
134    /// The origin of a held UTXO after a second add with the origin `new`.
135    /// `None` means that a deposit and change claim one outpoint.
136    fn updated_by(self, new: Self) -> Option<Self> {
137        // 1. A deposit and change cannot share an outpoint.
138        if self.is_change() != new.is_change() {
139            return None;
140        }
141        // 2. Both are change: a repeat keeps the held origin.
142        if self.is_change() {
143            return Some(self);
144        }
145        // 3. Both are deposits.
146        match (self, new) {
147            // A second request for an approved deposit. Move mints only once.
148            (Self::Deposit, Self::Deposit | Self::UnapprovedDeposit { .. }) => Some(self),
149            // An approval approves the held deposit.
150            (Self::UnapprovedDeposit { .. }, Self::Deposit) => Some(Self::Deposit),
151            // The later of two requests stays.
152            (
153                Self::UnapprovedDeposit {
154                    requested_at: held_at,
155                    ..
156                },
157                Self::UnapprovedDeposit { requested_at, .. },
158            ) => Some(if requested_at > held_at { new } else { self }),
159            (Self::Change, _) | (_, Self::Change) => {
160                unreachable!("steps 1 and 2 return for change")
161            }
162        }
163    }
164}

166/// The time through which each source has been read.
167#[derive(Clone, Copy, Debug)]
168pub struct Cursors {
169    pub sui: UnixSeconds,
170    pub guardian: UnixSeconds,
171}
172
173impl Cursors {
174    pub fn for_event_type(&self, et: WithdrawalEventType) -> UnixSeconds {
175        match et {
176            WithdrawalEventType::E1HashiApproved => self.sui,
177            WithdrawalEventType::E2GuardianApproved => self.guardian,
178            WithdrawalEventType::E3BtcConfirmed => {
179                unreachable!("E3 cursor is tracked per-withdrawal via btc_checked_at")
180            }
181        }
182    }
