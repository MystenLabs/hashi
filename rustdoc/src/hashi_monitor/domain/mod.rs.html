4//! The monitor's core types: the events it follows, their ids, and the
5//! cursors of its sources.
6//!
7//! Why it exists: every source, Sui, the guardian's S3 logs and Bitcoin, is
8//! parsed into one of these events, and the state machines reason about the
9//! events alone. Each event type has a constructor named for its source, so
10//! the mapping from a source's shape to the monitor's lives next to the type.
11//!
12//! The flows:
13//! - A withdrawal is three events: E1, the Hashi approval on Sui; E2, the
14//!   guardian's signature in its log; E3, the transaction confirmed on
15//!   Bitcoin. Every event states the transaction in Move's shape,
16//!   `WithdrawalTx`, so the sources can be compared as a whole.
17//! - A deposit is two events: E1, the output confirmed on Bitcoin; E2,
18//!   Hashi's `DepositConfirmed` on Sui.
19//!
20//! Formatting of times lives in `time`, and the bridge keys in `keys`.
21
22pub mod deposit;
23pub mod withdrawal;

41#[derive(Clone, Debug, Eq, PartialEq)]
42pub enum MonitorEvent {
43    Withdrawal(MonitorWithdrawalEvent),
44    Deposit(MonitorDepositEvent),
45}
46
47impl fmt::Display for MonitorEvent {
48    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
49        match self {
50            Self::Withdrawal(event) => write!(
51                formatter,
52                "Withdrawal(type={:?}, wid={}, timestamp={}, btc_txid={})",
53                event.event_type,
54                event.wid,
55                utc_timestamp(event.timestamp_secs),
56                event.btc_txid,
57            ),
58            Self::Deposit(event) => write!(
59                formatter,
60                "Deposit(type={:?}, deposit_id={}, timestamp={}, amount={}, script_pubkey={:x})",
61                event.event_type,
62                event.deposit_id,
63                utc_timestamp(event.timestamp_secs),
64                event.amount,
65                event.script_pubkey,
66            ),
67        }
68    }
69}

71/// Correlation identifier for an event in the monitor's withdrawal or deposit model.
72/// This is not a chain-native event ID.
73#[derive(Clone, Copy, Debug, Eq, PartialEq)]
74pub enum MonitorEventId {
75    Withdrawal(WithdrawalID),
76    Deposit(DepositId),
77}
78
79impl fmt::Display for MonitorEventId {
80    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
81        match self {
82            Self::Withdrawal(wid) => write!(formatter, "wid={wid}"),
83            Self::Deposit(deposit_id) => write!(formatter, "deposit_id={deposit_id}"),
84        }
85    }
86}

94/// A transaction mined at least `MIN_CONFIRMATIONS` blocks deep.
95#[derive(Clone, Debug)]
96pub struct BtcConfirmation {
97    /// The time of the block that holds the transaction. The recovery leaf's
98    /// clock starts at this block, so a coin's age counts from here.
99    pub block_time: UnixSeconds,
100    /// The time of the block that gave the transaction `MIN_CONFIRMATIONS`.
101    /// The bridge acts on that block, so the Bitcoin events carry this time.
102    pub confirmed_time: UnixSeconds,
103    /// The transaction's outputs, in order.
104    pub outputs: Vec<TxOut>,
105}
106
107/// The time through which each source has been read.
108#[derive(Clone, Copy, Debug)]
109pub struct Cursors {
110    pub sui: UnixSeconds,
111    pub guardian: UnixSeconds,
112}
113
114impl Cursors {
115    pub fn for_event_type(&self, et: WithdrawalEventType) -> UnixSeconds {
116        match et {
117            WithdrawalEventType::E1HashiApproved => self.sui,
118            WithdrawalEventType::E2GuardianApproved => self.guardian,
119            WithdrawalEventType::E3BtcConfirmed => {
120                unreachable!("E3 cursor is tracked per-withdrawal via btc_checked_at")
121            }
122        }
123    }
124}
