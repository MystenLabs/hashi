4//! The continuous auditor's snapshot: the whole audit state, so a restart
5//! continues where the last run stopped.
6//!
7//! Why it exists: without it, every restart would re-audit from the bridge's
8//! genesis to rebuild the UTXO pool, and report every old finding again.
9//!
10//! How it works:
11//! 1. About once an hour, the auditor saves a `Snapshot`: the deployment
12//!    it audits, its `AuditState`, and the UTXO pool. The audit state is the
13//!    position of each poller, every open state machine, the suspect spends
14//!    of the last pool check. All of it is taken at one instant.
15//! 2. A restart loads the machines and resumes each poller at its saved
16//!    position. It reads no event twice and skips none, so no flow is cut
17//!    in two.
18//! 3. The bucket holds the snapshot as one object, `SNAPSHOT_KEY`. A save
19//!    replaces it whole.
20//!
21//! Assumptions:
22//! - Only the monitor writes the bucket.
23//! - A snapshot of another `SNAPSHOT_VERSION` fails the load. The operator
24//!   deletes it, and the next start is a first run from genesis.
25//! - The snapshot is durable. Without it, the next start is a first run from
26//!   genesis, which needs a Sui node that keeps the history.
27

48/// The object that holds the snapshot.
49const SNAPSHOT_KEY: &str = "snapshot.json";
50
51/// The format of the snapshot. Change it when an old snapshot cannot load,
52/// or loads with another meaning.
53const SNAPSHOT_VERSION: u32 = 1;
54
55/// What the continuous auditor keeps between runs.
56#[derive(Clone, Debug, PartialEq)]
57pub struct Snapshot {
58    /// The guardian bucket and Hashi package audited. Progress through one
59    /// deployment says nothing about another, such as a redeployed devnet.
60    pub guardian_bucket: String,
61    pub package_id: String,
62    pub state: AuditState,
63    pub pool: UtxoPool,
64}
65
66/// The auditor's state apart from the UTXO pool.
67#[derive(Clone, Debug, Deserialize, PartialEq, Serialize)]
68pub struct AuditState {
69    /// The first Sui checkpoint not yet scanned.
70    pub sui_next_checkpoint: u64,
71    /// The guardian directory read now, and its seq state.
72    pub guardian_cursor: GuardianCursor,
73    pub withdrawals: Vec<WithdrawalStateMachine>,
74    pub deposits: Vec<DepositStateMachine>,
75    /// The spends the last pool check saw, still waiting for the Sui cursor.
76    pub suspect_spends: Vec<(OutPoint, SuspectSpend)>,
77    /// Since when the hBTC supply has exceeded the pool value, if it does.
78    pub hbtc_supply_above_pool_since: Option<UnixSeconds>,
79    /// Move's time of the latest Sui `DepositApproved`.
80    pub last_deposit_approval: Option<UnixSeconds>,
81}
82
83impl Snapshot {
84    pub fn new(cfg: &Config, state: AuditState, pool: UtxoPool) -> Self {
85        Self {
86            guardian_bucket: cfg.deployment.bucket_info.name.clone(),
87            package_id: cfg.sui.package_id().to_string(),
88            state,
89            pool,
90        }
91    }
92
93    /// Whether the snapshot was saved auditing the deployment in `cfg`.
94    pub fn is_for(&self, cfg: &Config) -> bool {
95        self.guardian_bucket == cfg.deployment.bucket_info.name
96            && self.package_id.parse::<sui_sdk_types::Address>().ok() == Some(cfg.sui.package_id())
97    }
98}

100/// The object at `SNAPSHOT_KEY`.
101#[derive(Deserialize, Serialize)]
102struct StoredSnapshot {
103    version: u32,
104    guardian_bucket: String,
105    package_id: String,
106    #[serde(flatten)]
107    state: AuditState,
108    utxos: Vec<UtxoWire>,
109}
110
111/// The bucket that keeps the snapshot.
112pub struct SnapshotStore {
113    store: Box<dyn ObjectStore>,
114}
115
116impl SnapshotStore {
117    /// The GCS bucket `bucket`. Credentials come from the environment, or on
118    /// GKE from the pod's workload identity.
119    pub fn gcs(bucket: &str) -> anyhow::Result<Self> {
120        let store = GoogleCloudStorageBuilder::from_env()
121            .with_bucket_name(bucket)
122            .build()
123            .with_context(|| format!("failed to set up the snapshot bucket {bucket}"))?;
124        Ok(Self {
125            store: Box::new(store),
126        })
127    }
128
129    /// The saved snapshot, or `None` if there is none.
130    pub async fn load(&self) -> anyhow::Result<Option<Snapshot>> {
131        let bytes = match self.store.get(&Path::from(SNAPSHOT_KEY)).await {
132            Ok(object) => object.bytes().await,
133            Err(object_store::Error::NotFound { .. }) => return Ok(None),
134            Err(error) => Err(error),
135        }
136        .context("failed to read the snapshot")?;
137        #[derive(Deserialize)]
138        struct Version {
139            version: u32,
140        }
141        let Version { version } =
142            serde_json::from_slice(&bytes).context("failed to parse the snapshot")?;
143        anyhow::ensure!(
144            version == SNAPSHOT_VERSION,
145            "the snapshot has version {version}, but this monitor needs version {SNAPSHOT_VERSION}. Delete the snapshot to run again from genesis",
146        );
147        let stored: StoredSnapshot =
148            serde_json::from_slice(&bytes).context("failed to parse the snapshot")?;
149        Ok(Some(Snapshot {
150            guardian_bucket: stored.guardian_bucket,
151            package_id: stored.package_id,
152            state: stored.state,
153            pool: UtxoPool::from_wire(stored.utxos),
154        }))
155    }
156
157    /// Replace the saved snapshot of the deployment in `cfg`.
158    pub async fn save(
159        &self,
160        cfg: &Config,
161        state: AuditState,
162        pool: &UtxoPool,
163    ) -> anyhow::Result<()> {
164        let started = Instant::now();
165        // Before the save, so a save that runs out of memory still leaves its size.
166        tracing::info!(
167            withdrawals = state.withdrawals.len(),
168            deposits = state.deposits.len(),
169            suspect_spends = state.suspect_spends.len(),
170            utxos = pool.len(),
171            "saving the snapshot"
172        );
173        let stored = StoredSnapshot {
174            version: SNAPSHOT_VERSION,
175            guardian_bucket: cfg.deployment.bucket_info.name.clone(),
176            package_id: cfg.sui.package_id().to_string(),
177            state,
178            utxos: pool.held_to_wire(),
179        };
180        let bytes = serde_json::to_vec(&stored)?;
181        let bytes_written = bytes.len();
182        self.store
183            .put(&Path::from(SNAPSHOT_KEY), PutPayload::from(bytes))
184            .await
185            .context("failed to write the snapshot")?;
186
187        tracing::info!(
188            bytes_written,
189            elapsed_ms = started.elapsed().as_millis() as u64,
190            "saved the snapshot"
191        );
192        Ok(())
193    }
194}

219    /// A state with one open withdrawal, one open deposit and a suspect spend.
220    fn state() -> AuditState {
221        let approval = hashi_approval();
222        let mut withdrawal = WithdrawalStateMachine::new(approval.clone(), &Timing::for_tests());
223        let signature = crate::domain::MonitorWithdrawalEvent {
224            event_type: WithdrawalEventType::E2GuardianApproved,
225            timestamp_secs: approval.timestamp_secs + 60,
226            ..approval
227        };
228        assert!(
229            withdrawal
230                .add_event(signature, &Timing::for_tests())
231                .is_empty()
232        );
233        AuditState {
234            sui_next_checkpoint: 1_234,
235            guardian_cursor: GuardianCursor::new(S3HourDirectory::withdraw(1_700_000_000).unwrap()),
236            withdrawals: vec![withdrawal],
237            deposits: vec![DepositStateMachine::new(deposit_request())],
238            suspect_spends: vec![(
239                OutPoint {
240                    txid: txid(4),
241                    vout: 0,
242                },
243                SuspectSpend {
244                    checked_at: 1_700_000_100,
245                    amount: 2_000,
246                },
247            )],
248            hbtc_supply_above_pool_since: Some(1_700_000_600),
249            last_deposit_approval: Some(1_700_000_500),
250        }
251    }
252
253    /// A pool that holds one UTXO.
254    fn pool_with_a_utxo() -> UtxoPool {
255        let mut pool = UtxoPool::default();
256        let deposit = BtcConfirmedDeposit {
257            deposit_id: DepositId::new(txid(1), 0),
258            amount: 1_000,
259            script_pubkey: ScriptBuf::from_bytes(vec![0x51]),
260            block_time: 1_699_999_000,
261        };
262        assert!(pool.add_deposit(&unapproved(deposit)).is_empty());
263        pool
264    }

266    async fn put(store: &SnapshotStore, key: &Path, bytes: Vec<u8>) {
267        store.store.put(key, PutPayload::from(bytes)).await.unwrap();
268    }

277    #[tokio::test]
278    async fn a_snapshot_round_trips_through_the_store() {
279        let store = SnapshotStore::in_memory();
280        assert_eq!(store.load().await.unwrap(), None);
281
282        let snapshot = Snapshot::new(&Config::for_tests(), state(), pool_with_a_utxo());
283        save(&store, &snapshot).await;
284
285        assert_eq!(store.load().await.unwrap(), Some(snapshot));
286    }
287
288    #[tokio::test]
289    async fn a_snapshot_of_another_version_or_none_is_an_error() {
290        let store = SnapshotStore::in_memory();
291        let key = Path::from(SNAPSHOT_KEY);
292
293        put(&store, &key, br#"{"version": 0}"#.to_vec()).await;
294        let error = format!("{:#}", store.load().await.unwrap_err());
295        assert!(error.contains("Delete the snapshot"), "{error}");
296
297        put(&store, &key, b"{".to_vec()).await;
298        let error = format!("{:#}", store.load().await.unwrap_err());
299        assert!(error.contains("failed to parse the snapshot"), "{error}");
300    }
301
302    #[test]
303    fn a_snapshot_is_only_for_the_deployment_that_saved_it() {
304        let cfg = Config::for_tests();
305        let snapshot = Snapshot::new(&cfg, state(), UtxoPool::default());
306        assert!(snapshot.is_for(&cfg));
307
308        let mut redeployed = cfg.clone();
309        redeployed.sui.package_id = Some(sui_sdk_types::Address::new([0x22; 32]));
310        assert!(!snapshot.is_for(&redeployed));
311
312        let mut other_guardian = cfg;
313        other_guardian.deployment.bucket_info.name = "other-bucket".to_string();
314        assert!(!snapshot.is_for(&other_guardian));
315    }
