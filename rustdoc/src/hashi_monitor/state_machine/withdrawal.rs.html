4//! The state machine that follows one withdrawal through its three events.
5//!
6//! Why it exists: the withdrawal's events come from three sources at
7//! different times and in any order. The machine joins them by withdrawal id
8//! and judges each pair of neighbors by the timing bounds, so the auditors
9//! need not know the order the sources were read in.
10//!
11//! How it works:
12//! 1. The first event creates the machine. The machine expects the event's
13//!    missing neighbors, each by a deadline: a predecessor by the event's
14//!    time plus the predecessor's skew, a successor by the event's time plus
15//!    the successor's delay.
16//! 2. A later event is checked against the machine's wid, txid and
17//!    transaction. A contradiction is reported and the event is dropped.
18//! 3. A consistent event is kept. It is judged against each neighbor the
19//!    machine has seen, and it removes itself from the expected events.
20//! 4. The auditor looks the transaction up on Bitcoin while E3 is expected,
21//!    and asks for timing violations each tick: an expected event whose
22//!    deadline the source's cursor has passed is reported missing.
23//!
24//! Invariants:
25//! - `expected_events` never holds an event type in `seen_events`.
26//! - Findings are returned to the caller and not kept. A complete machine
27//!   may have reported findings earlier.
28

44/// The events seen and still expected for one withdrawal. See the module
45/// doc for how an event is added and judged.
46#[derive(Clone, Debug, Deserialize, PartialEq, Serialize)]
47pub struct WithdrawalStateMachine {
48    /// The events seen so far, at most one per type.
49    seen_events: Vec<MonitorWithdrawalEvent>,
50    /// Each `(event, deadline, relation)` entry means `event` is expected by
51    /// `deadline`; `relation` records whether it precedes or follows an event
52    /// already observed for this withdrawal.
53    expected_events: Vec<(WithdrawalEventType, UnixSeconds, EventRelation)>,
54    /// The time of the last Bitcoin lookup of the transaction.
55    btc_checked_at: Option<UnixSeconds>,
56    /// The withdrawal for the UTXO pool, once Bitcoin confirmed it.
57    confirmed: Option<ConfirmedWithdrawal>,
58    /// The withdrawal id. Every event must carry it.
59    wid: WithdrawalID,
60    /// The transaction id. Every event must carry it.
61    btc_txid: Txid,
62    /// The inputs and outputs stated by the first event seen. Every later
63    /// event must state the same.
64    tx: WithdrawalTx,
65}
66
67impl WithdrawalStateMachine {
68    /// A machine holding `event`. The first event cannot produce a finding.
69    pub fn new(event: MonitorWithdrawalEvent, timing: &Timing) -> Self {
70        let mut sm = Self {
71            seen_events: Vec::new(),
72            expected_events: Vec::new(),
73            btc_checked_at: None,
74            confirmed: None,
75            wid: event.wid,
76            btc_txid: event.btc_txid,
77            tx: event.tx.clone(),
78        };
79        assert!(
80            sm.add_event(event, timing).is_empty(),
81            "first event cannot produce a finding"
82        );
83        sm
84    }

86    pub fn get(&self, event_type: WithdrawalEventType) -> Option<&MonitorWithdrawalEvent> {
87        self.seen_events
88            .iter()
89            .find(|event| event.event_type == event_type)
90    }
91
92    pub fn btc_txid(&self) -> Txid {
93        self.btc_txid
94    }
95
96    /// The withdrawal's inputs and outputs, as every event states them.
97    pub fn tx(&self) -> &WithdrawalTx {
98        &self.tx
99    }
100
101    /// The withdrawal for the UTXO pool, or `None` until Bitcoin confirmed it.
102    pub fn confirmed(&self) -> Option<&ConfirmedWithdrawal> {
103        self.confirmed.as_ref()
104    }
105
106    /// The withdrawal Bitcoin confirmed, if any.
107    pub fn into_confirmed(self) -> Option<ConfirmedWithdrawal> {
108        self.confirmed
109    }

111    pub fn wid(&self) -> WithdrawalID {
112        self.wid
113    }

115    pub fn expects(&self, event_type: WithdrawalEventType) -> bool {
116        self.expected_events
117            .iter()
118            .any(|(event, _, _)| *event == event_type)
119    }
120
121    /// Permit Bitcoin lookups only after both the Sui and guardian approvals arrive.
122    /// Sui approves a withdrawal after the flows that made its inputs, so the monitor has read them.
123    /// Thus every flow this one spends from is open or closed when this one closes.
124    pub fn expects_btc(&self) -> bool {
125        self.get(WithdrawalEventType::E1HashiApproved).is_some()
126            && self.get(WithdrawalEventType::E2GuardianApproved).is_some()
127            && self.expects(WithdrawalEventType::E3BtcConfirmed)
128    }
129
130    /// Whether every expected event was seen, so the pool can take the flow
131    /// and the auditor can drop it.
132    ///
133    /// This does not mean that no timing finding was emitted during ingestion.
134    pub fn is_ready_for_gc(&self) -> bool {
135        self.expected_events.is_empty()
136    }
137
138    pub fn expected_event_types(&self) -> Vec<WithdrawalEventType> {
139        self.expected_events
140            .iter()
141            .map(|(event_type, _, _)| *event_type)
142            .collect()
143    }

145    /// The time of the earliest event seen.
146    pub fn earliest_event_at(&self) -> UnixSeconds {
147        self.seen_events
148            .iter()
149            .map(|e| e.timestamp_secs)
150            .min()
151            .expect("a machine holds its first event")
152    }

154    /// Add an event and update expectations for its immediate neighbors.
155    /// The auditor adds E1 and E2 as it reads Sui and the guardian log.
156    /// `try_fetch_btc_tx` adds E3 from Bitcoin's confirmation.
157    ///
158    /// A missing predecessor is expected by the event timestamp plus the
159    /// predecessor's clock skew; a missing successor is expected by the
160    /// event timestamp plus the successor's delay. A neighbor already seen is
161    /// checked against the pair's deadlines, so findings do not depend on
162    /// ingestion order.
163    ///
164    /// Event retention is based on structural validity.
165    /// `MonitorFinding::InvalidEventAdded` denotes a contradictory event, so
166    /// it is returned immediately without storing the event. A structurally
167    /// valid event is stored and updates subsequent expectations even when
168    /// its timing produces a finding.
169    ///
170    /// The return value contains every resulting finding. An empty vector means
171    /// the event was accepted without one.
172    pub fn add_event(
173        &mut self,
174        new_event: MonitorWithdrawalEvent,
175        timing: &Timing,
176    ) -> Vec<MonitorFinding> {
177        if let Some(existing_event) = self.get(new_event.event_type) {
178            return if *existing_event == new_event {
179                Vec::new()
180            } else {
181                vec![MonitorFinding::InvalidEventAdded(
182                    "duplicate event for same wid with different contents".to_string(),
183                )]
184            };
185        }
186
187        if self.wid != new_event.wid {
188            return vec![MonitorFinding::InvalidEventAdded("invalid wid".to_string())];
189        }
190
191        if self.btc_txid != new_event.btc_txid {
192            return vec![MonitorFinding::InvalidEventAdded(
193                "invalid btc_txid".to_string(),
194            )];
195        }
196
197        let mut timing_findings = Vec::new();
198        // The txid pins the inputs and outputs, but not which outputs Sui
199        // calls change. Sui's and the guardian's statements must be equal,
200        // else a committee can label a payout to itself as change.
201        if new_event.tx != self.tx {
202            let (sui, guardian) = match new_event.event_type {
203                WithdrawalEventType::E1HashiApproved => (new_event.tx.clone(), self.tx.clone()),
204                _ => (self.tx.clone(), new_event.tx.clone()),
205            };
206            timing_findings.push(MonitorFinding::WithdrawalMismatch {
207                wid: self.wid,
208                sui,
209                guardian,
210            });
211        }
212        if let Some(predecessor_event_type) = new_event.event_type.predecessor() {
213            match self.get(predecessor_event_type) {
214                Some(predecessor) => timing_findings.extend(neighbor_timing_findings(
215                    predecessor,
216                    &new_event,
217                    timing,
218                )),
219                None => self.expected_events.push((
220                    predecessor_event_type,
221                    timing.predecessor_deadline(&new_event),
222                    EventRelation::Predecessor,
223                )),
224            }
225        }
226        if let Some(successor_event_type) = new_event.event_type.successor() {
227            match self.get(successor_event_type) {
228                Some(successor) => {
229                    timing_findings.extend(neighbor_timing_findings(&new_event, successor, timing))
230                }
231                None => self.expected_events.push((
232                    successor_event_type,
233                    timing.successor_deadline(&new_event),
234                    EventRelation::Successor,
235                )),
236            }
237        }
238
239        // remove any previously stored expected events
240        self.expected_events
241            .retain(|(src, _, _)| *src != new_event.event_type);
242        // add to seen events
243        self.seen_events.push(new_event);
244        timing_findings
245    }
246
247    /// If expecting BTC confirmation, query BTC RPC and add the event if confirmed.
248    /// Returns the findings from the new event. An error is the RPC's.
249    ///
250    /// `checked_at` must be no later than the lookup.
251    pub fn try_fetch_btc_tx(
252        &mut self,
253        timing: &Timing,
254        btc_rpc_client: &BtcRpcClient,
255        checked_at: UnixSeconds,
256    ) -> anyhow::Result<Vec<MonitorFinding>> {
257        if !self.expects_btc() {
258            return Ok(Vec::new());
259        }
260
261        let confirmation = btc_rpc_client.lookup_confirmation(self.btc_txid)?;
262        self.btc_checked_at = Some(checked_at);
263        Ok(match confirmation {
264            Some(confirmation) => {
265                let e_btc = MonitorWithdrawalEvent::from_btc(
266                    &confirmation,
267                    self.wid,
268                    self.btc_txid,
269                    self.tx.clone(),
270                );
271                // E3 is expected only after E2, so E2 is present here. The
272                // pool takes the change layout from E2, the guardian's word.
273                let e2 = self
274                    .get(WithdrawalEventType::E2GuardianApproved)
275                    .expect("E3 is expected only after E2");
276                self.confirmed = Some(ConfirmedWithdrawal::from_confirmation(e2, confirmation));
277                self.add_event(e_btc, timing)
278            }
279            None => Vec::new(),
280        })
281    }
282
283    /// Check for timing violations given per-source cursors.
284    /// Only reports a missing event if its deadline has passed relative to the relevant cursor.
285    pub fn timing_violations(&self, cursors: &Cursors) -> Vec<MonitorFinding> {
286        let mut out = Vec::new();
287        for (event_type, deadline, relation) in &self.expected_events {
288            let cursor = match event_type {
289                WithdrawalEventType::E3BtcConfirmed => match self.btc_checked_at {
290                    Some(checked_at) => checked_at,
291                    None => {
292                        // Bitcoin and state checks have independent schedules.
293                        // Wait for the first lookup before evaluating absence.
294                        continue;
295                    }
296                },
297                _ => cursors.for_event_type(*event_type),
298            };
299            if *deadline <= cursor {
300                out.push(MonitorFinding::WithdrawalEventMissing {
301                    wid: self.wid,
302                    event_type: *event_type,
303                    relation: *relation,
304                    deadline: *deadline,
305                    cursor,
306                });
307            }
308        }
309        out
310    }
311}

313/// Both timing bounds between consecutive events.
314fn neighbor_timing_findings(
315    predecessor: &MonitorWithdrawalEvent,
316    successor: &MonitorWithdrawalEvent,
317    timing: &Timing,
318) -> Vec<MonitorFinding> {
319    let mut findings = Vec::new();
320    let deadline = timing.successor_deadline(predecessor);
321    if deadline < successor.timestamp_secs {
322        findings.push(MonitorFinding::WithdrawalEventLate {
323            event: successor.clone(),
324            relation: EventRelation::Successor,
325            deadline,
326            occurred_at: successor.timestamp_secs,
327        });
328    }
329    let deadline = timing.predecessor_deadline(successor);
330    if deadline < predecessor.timestamp_secs {
331        findings.push(MonitorFinding::WithdrawalEventLate {
332            event: predecessor.clone(),
333            relation: EventRelation::Predecessor,
334            deadline,
335            occurred_at: predecessor.timestamp_secs,
336        });
337    }
338    findings
339}

342impl WithdrawalStateMachine {
343    /// A complete machine: `approval` as E1, the same a minute later as
344    /// E2, and `confirmation` as E3, with no timing finding.
345    pub fn complete_for_tests(
346        approval: MonitorWithdrawalEvent,
347        confirmation: crate::domain::BtcConfirmation,
348        timing: &Timing,
349    ) -> Self {
350        let e2 = MonitorWithdrawalEvent {
351            event_type: WithdrawalEventType::E2GuardianApproved,
352            timestamp_secs: approval.timestamp_secs + 60,
353            ..approval.clone()
354        };
355        let e3 = MonitorWithdrawalEvent::from_btc(
356            &confirmation,
357            approval.wid,
358            approval.btc_txid,
359            approval.tx.clone(),
360        );
361        let mut sm = Self::new(approval, timing);
362        assert!(sm.add_event(e2.clone(), timing).is_empty());
363        assert!(sm.add_event(e3, timing).is_empty());
364        sm.confirmed = Some(ConfirmedWithdrawal::from_confirmation(&e2, confirmation));
365        assert!(sm.is_ready_for_gc());
366        sm
367    }

553    #[test]
554    fn timing_findings_do_not_depend_on_ingestion_order() {
555        let timing = Timing::for_tests();
556        let late = |event: &MonitorWithdrawalEvent, relation, deadline| {
557            MonitorFinding::WithdrawalEventLate {
558                event: event.clone(),
559                relation,
560                deadline,
561                occurred_at: event.timestamp_secs,
562            }
563        };
564
565        // The last two cases have a block time before the guardian signature,
566        // within and past E2's clock skew.
567        for (approved_at, signed_at, confirmed_at) in [
568            (100, 150, 300),
569            (100, 200, 400),
570            (100, 201, 300),
571            (100, 400_000, 400_100),
572            (110, 100, 300),
573            (111, 100, 301),
574            (100, 150, 120),
575            (100, 150, 119),
576        ] {
577            let events = [
578                event(WithdrawalEventType::E1HashiApproved, 5, approved_at, 5),
579                event(WithdrawalEventType::E2GuardianApproved, 5, signed_at, 5),
580                event(WithdrawalEventType::E3BtcConfirmed, 5, confirmed_at, 5),
581            ];
582            let [approval, signature, confirmation] = &events;
583            let mut expected = Vec::new();
584            if signed_at > approved_at + 100 {
585                expected.push(late(signature, EventRelation::Successor, approved_at + 100));
586            }
587            if approved_at > signed_at + 10 {
588                expected.push(late(approval, EventRelation::Predecessor, signed_at + 10));
589            }
590            if confirmed_at > signed_at + 200 {
591                expected.push(late(
592                    confirmation,
593                    EventRelation::Successor,
594                    signed_at + 200,
595                ));
596            }
597            if signed_at > confirmed_at + 30 {
598                expected.push(late(
599                    signature,
600                    EventRelation::Predecessor,
601                    confirmed_at + 30,
602                ));
603            }
604
605            for order in [
606                [0, 1, 2],
607                [0, 2, 1],
608                [1, 0, 2],
609                [1, 2, 0],
610                [2, 0, 1],
611                [2, 1, 0],
612            ] {
613                let mut sm = WithdrawalStateMachine::new(events[order[0]].clone(), &timing);
614                let findings: Vec<_> = order[1..]
615                    .iter()
616                    .flat_map(|&i| sm.add_event(events[i].clone(), &timing))
617                    .collect();
618                assert_eq!(findings.len(), expected.len(), "{order:?}: {findings:?}");
619                assert!(
620                    expected.iter().all(|finding| findings.contains(finding)),
621                    "{order:?}: {findings:?}"
622                );
623                assert!(sm.is_ready_for_gc(), "{order:?}");
624            }
625        }
626    }
