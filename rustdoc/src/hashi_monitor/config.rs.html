17/// Configuration shared by the batch and continuous monitor modes.
18///
19/// All duration values are expressed in seconds.
20#[derive(Clone, Debug, Deserialize)]
21pub struct Config {
22    /// Maximum allowed delay between consecutive events.
23    pub next_event_delays: NextEventDelays,
24
25    /// How far each event's successor may occur before it (default: 300s for E1,
26    /// 2h for E2). Keep E1's at least the guardian's 5-minute request clock tolerance.
27    #[serde(default = "default_clock_skews")]
28    pub clock_skews: ClockSkews,
29
30    /// How far a deposit's Bitcoin block time may be after its Sui confirmation
31    /// (default: 300s).
32    #[serde(default = "default_deposit_clock_skew")]
33    pub deposit_clock_skew: u64,
34
35    /// How far before the guardian audit start to search Sui for withdrawal
36    /// predecessor events (default: 1 hour).
37    #[serde(default = "default_withdrawal_predecessor_lookback")]
38    pub withdrawal_predecessor_lookback: u64,
39
40    pub deployment: DeploymentConfig,
41    /// Omit to use AWS's default credential chain.
42    pub s3_credentials: Option<S3Credentials>,
43    pub sui: SuiConfig,
44    pub btc: BtcConfig,
45}

67#[derive(Clone, Debug, Deserialize)]
68pub struct BtcConfig {
69    /// Bitcoin JSON-RPC endpoint.
70    ///
71    /// Prefix with `env:` to read the URL from an environment variable, which
72    /// keeps provider API keys out of YAML.
73    pub rpc_url: String,
74
75    /// Optional HTTP headers for the JSON-RPC provider.
76    ///
77    /// Values accept the same `env:` prefix as `rpc_url`.
78    #[serde(default)]
79    pub http_headers: BTreeMap<String, String>,
80}
81
82impl BtcConfig {
83    pub fn resolve_rpc_url(&self) -> anyhow::Result<String> {
84        resolve_env_reference("rpc_url", &self.rpc_url)
85    }
86
87    pub fn resolve_http_headers(&self) -> anyhow::Result<BTreeMap<String, String>> {
88        self.http_headers
89            .iter()
90            .map(|(name, value)| {
91                let value = resolve_env_reference(&format!("{name} header"), value)?;
92                Ok((name.clone(), value))
93            })
94            .collect()
95    }
96}
97
98fn resolve_env_reference(field: &str, value: &str) -> anyhow::Result<String> {
99    let Some(variable) = value.strip_prefix("env:") else {
100        return Ok(value.to_string());
101    };
102    anyhow::ensure!(
103        !variable.is_empty(),
104        "bitcoin {field} environment variable name is empty"
105    );
106    std::env::var(variable)
107        .with_context(|| format!("bitcoin {field} environment variable {variable} is not set"))
108}

110fn default_clock_skews() -> ClockSkews {
111    // E3's block time is set by its miner and need only exceed the median of the
112    // last 11 blocks, which trails real time by about an hour.
113    ClockSkews::new(vec![
114        (WithdrawalEventType::E1HashiApproved, 5 * 60),
115        (WithdrawalEventType::E2GuardianApproved, 2 * 60 * 60),
116    ])
117    .expect("one entry per non-terminal event")
118}
119
120fn default_deposit_clock_skew() -> u64 {
121    300
122}
123
124fn default_withdrawal_predecessor_lookback() -> u64 {
125    60 * 60
126}

128fn check_one_entry_per_non_terminal_event(
129    kind: &str,
130    inputs: &[(WithdrawalEventType, u64)],
131) -> anyhow::Result<()> {
132    let mut seen_sources = Vec::new();
133    for (source, _) in inputs {
134        if seen_sources.contains(source) {
135            return Err(anyhow!("duplicate {kind} entry for {source:?}"));
136        }
137        seen_sources.push(*source);
138    }
139
140    if seen_sources.contains(&WithdrawalEventType::TERMINAL_EVENT) {
141        return Err(anyhow!("{kind} for terminal event is not allowed"));
142    }
143
144    for source in WithdrawalEventType::NON_TERMINAL_EVENTS {
145        if !seen_sources.contains(&source) {
146            return Err(anyhow!("missing {kind} entry for {source:?}"));
147        }
148    }
149
150    Ok(())
151}

153impl NextEventDelays {
154    /// The constructor ensures that there is one entry for every non-terminal event.
155    pub fn new(inputs: Vec<(WithdrawalEventType, u64)>) -> anyhow::Result<Self> {
156        check_one_entry_per_non_terminal_event("delay", &inputs)?;
157        Ok(Self(inputs))
158    }
159
160    pub fn get_delay(&self, source: WithdrawalEventType) -> Option<u64> {
161        self.0
162            .iter()
163            .find(|(event_source, _)| *event_source == source)
164            .map(|(_, next_event_delay_secs)| *next_event_delay_secs)
165    }
166
167    pub fn max_delay(&self) -> u64 {
168        self.0
169            .iter()
170            .map(|(_, next_event_delay_secs)| *next_event_delay_secs)
171            .max()
172            .unwrap_or_default()
173    }
174}
175
176impl TryFrom<Vec<(WithdrawalEventType, u64)>> for NextEventDelays {
177    type Error = anyhow::Error;
178
179    fn try_from(entries: Vec<(WithdrawalEventType, u64)>) -> Result<Self, Self::Error> {
180        Self::new(entries)
181    }
182}

215impl Config {
216    pub fn load_yaml(path: &Path) -> anyhow::Result<Self> {
217        let bytes = std::fs::read(path)
218            .with_context(|| format!("failed to read config file at {}", path.display()))?;
219        let cfg = serde_yaml::from_slice(&bytes)
220            .with_context(|| format!("failed to parse config yaml at {}", path.display()))?;
221        Ok(cfg)
222    }
223
224    pub fn next_event_delay(&self, source: WithdrawalEventType) -> Option<u64> {
225        self.next_event_delays.get_delay(source)
226    }
227
228    pub fn predecessor_deadline(&self, event: &MonitorWithdrawalEvent) -> UnixSeconds {
229        let predecessor = event.event_type.predecessor().expect("has a predecessor");
230        event.timestamp_secs
231            + self
232                .clock_skews
233                .get_skew(predecessor)
234                .expect("a predecessor has a successor")
235    }
236
237    pub fn successor_deadline(&self, event: &MonitorWithdrawalEvent) -> UnixSeconds {
238        event.timestamp_secs
239            + self
240                .next_event_delay(event.event_type)
241                .expect("has a successor")
242    }
243}

250    fn sample_config_contains_deployment_without_credentials() {
251        let sample = include_str!("../audit.sample.yaml");
252        assert!(serde_yaml::from_str::<Config>(sample).is_err());

255        assert_eq!(config.deployment.bitcoin_network, bitcoin::Network::Signet);

264    #[test]
265    fn clock_skews_default_to_a_looser_bound_before_a_block_time() {
266        let config: Config = serde_yaml::from_str(&sample_config_yaml()).unwrap();
267
268        let skew = |source| config.clock_skews.get_skew(source);
269        assert_eq!(skew(WithdrawalEventType::E1HashiApproved), Some(300));
270        assert_eq!(skew(WithdrawalEventType::E2GuardianApproved), Some(7_200));
271        assert_eq!(config.deposit_clock_skew, 300);
272    }
