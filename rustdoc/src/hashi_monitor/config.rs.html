15/// Configuration shared by the batch and continuous monitor modes.
16///
17/// All duration values are expressed in seconds.
18#[derive(Clone, Debug, Deserialize)]
19pub struct Config {
20    /// Maximum allowed delay between consecutive events.
21    pub next_event_delays: NextEventDelays,
22
23    /// E_{i+1} is allowed to occur up to `clock_skew` before E_i (default: 300s).
24    #[serde(default = "default_clock_skew")]
25    pub clock_skew: u64,
26
27    /// How far before the guardian audit start to search Sui for withdrawal
28    /// predecessor events (default: 1 hour).
29    #[serde(default = "default_withdrawal_predecessor_lookback")]
30    pub withdrawal_predecessor_lookback: u64,
31
32    pub deployment: DeploymentConfig,
33    /// Omit to use AWS's default credential chain.
34    pub s3_credentials: Option<S3Credentials>,
35    pub sui: SuiConfig,
36    pub btc: BtcConfig,
37}

39/// The maximum allowed delay between an event and its successor.
40#[derive(Clone, Debug, Deserialize)]
41#[serde(try_from = "Vec<(WithdrawalEventType, u64)>")]
42pub struct NextEventDelays(Vec<(WithdrawalEventType, u64)>);

53#[derive(Clone, Debug, Deserialize)]
54pub struct BtcConfig {
55    /// Bitcoin JSON-RPC endpoint.
56    ///
57    /// Prefix with `env:` to read the URL from an environment variable, which
58    /// keeps provider API keys out of YAML.
59    pub rpc_url: String,
60
61    /// Optional HTTP headers for the JSON-RPC provider.
62    ///
63    /// Values accept the same `env:` prefix as `rpc_url`.
64    #[serde(default)]
65    pub http_headers: BTreeMap<String, String>,
66}
67
68impl BtcConfig {
69    pub fn resolve_rpc_url(&self) -> anyhow::Result<String> {
70        resolve_env_reference("rpc_url", &self.rpc_url)
71    }
72
73    pub fn resolve_http_headers(&self) -> anyhow::Result<BTreeMap<String, String>> {
74        self.http_headers
75            .iter()
76            .map(|(name, value)| {
77                let value = resolve_env_reference(&format!("{name} header"), value)?;
78                Ok((name.clone(), value))
79            })
80            .collect()
81    }
82}
83
84fn resolve_env_reference(field: &str, value: &str) -> anyhow::Result<String> {
85    let Some(variable) = value.strip_prefix("env:") else {
86        return Ok(value.to_string());
87    };
88    anyhow::ensure!(
89        !variable.is_empty(),
90        "bitcoin {field} environment variable name is empty"
91    );
92    std::env::var(variable)
93        .with_context(|| format!("bitcoin {field} environment variable {variable} is not set"))
94}
95
96fn default_clock_skew() -> u64 {
97    300
98}
99
100fn default_withdrawal_predecessor_lookback() -> u64 {
101    60 * 60
102}

104impl NextEventDelays {
105    /// The constructor ensures that there is one entry for every non-terminal event.
106    pub fn new(inputs: Vec<(WithdrawalEventType, u64)>) -> anyhow::Result<Self> {
107        let mut seen_sources = Vec::new();
108        for (source, _) in &inputs {
109            if seen_sources.contains(source) {
110                return Err(anyhow!(format!("duplicate delay entry for {:?}", source)));
111            }
112            seen_sources.push(*source);
113        }
114
115        if seen_sources.contains(&WithdrawalEventType::TERMINAL_EVENT) {
116            return Err(anyhow!(
117                "delay for terminal event is not allowed".to_string()
118            ));
119        }
120
121        for source in WithdrawalEventType::NON_TERMINAL_EVENTS {
122            if !seen_sources.contains(&source) {
123                return Err(anyhow!(format!("missing delay entry for {:?}", source)));
124            }
125        }
126
127        Ok(Self(inputs))
128    }
129
130    pub fn get_delay(&self, source: WithdrawalEventType) -> Option<u64> {
131        self.0
132            .iter()
133            .find(|(event_source, _)| *event_source == source)
134            .map(|(_, next_event_delay_secs)| *next_event_delay_secs)
135    }
136
137    pub fn max_delay(&self) -> u64 {
138        self.0
139            .iter()
140            .map(|(_, next_event_delay_secs)| *next_event_delay_secs)
141            .max()
142            .unwrap_or_default()
143    }
144}
145
146impl TryFrom<Vec<(WithdrawalEventType, u64)>> for NextEventDelays {
147    type Error = anyhow::Error;
148
149    fn try_from(entries: Vec<(WithdrawalEventType, u64)>) -> Result<Self, Self::Error> {
150        Self::new(entries)
151    }
152}

154impl Config {
155    pub fn load_yaml(path: &Path) -> anyhow::Result<Self> {
156        let bytes = std::fs::read(path)
157            .with_context(|| format!("failed to read config file at {}", path.display()))?;
158        let cfg = serde_yaml::from_slice(&bytes)
159            .with_context(|| format!("failed to parse config yaml at {}", path.display()))?;
160        Ok(cfg)
161    }
162
163    pub fn next_event_delay(&self, source: WithdrawalEventType) -> Option<u64> {
164        self.next_event_delays.get_delay(source)
165    }
166}

168#[cfg(test)]
169mod tests {
170    use super::*;
171
172    #[test]
173    fn sample_config_contains_deployment_without_credentials() {
174        let config: Config = serde_yaml::from_str(include_str!("../audit.sample.yaml")).unwrap();
175        assert_eq!(config.deployment.bitcoin_network, bitcoin::Network::Signet);
176        assert!(config.s3_credentials.is_none());
177    }
178
179    fn btc_config(rpc_url: &str, headers: &[(&str, &str)]) -> BtcConfig {
180        BtcConfig {
181            rpc_url: rpc_url.to_string(),
182            http_headers: headers
183                .iter()
184                .map(|(name, value)| (name.to_string(), value.to_string()))
185                .collect(),
186        }
187    }
188
189    #[test]
190    fn http_header_values_resolve_env_references() {
191        // Cargo and nextest set CARGO_PKG_NAME for test processes.
192        let cfg = btc_config(
193            "http://btc",
194            &[
195                ("Authorization", "env:CARGO_PKG_NAME"),
196                ("Origin", "https://example.com"),
197            ],
198        );
199
200        let headers = cfg.resolve_http_headers().unwrap();
201
202        assert_eq!(headers["Authorization"], env!("CARGO_PKG_NAME"));
203        assert_eq!(headers["Origin"], "https://example.com");
204    }
205
206    #[test]
207    fn unset_header_environment_variable_is_an_error() {
208        let cfg = btc_config(
209            "http://btc",
210            &[("Authorization", "env:HASHI_MONITOR_TEST_UNSET_VARIABLE")],
211        );
212
213        let error = cfg.resolve_http_headers().unwrap_err().to_string();
214
215        assert!(
216            error.contains("Authorization header")
217                && error.contains("HASHI_MONITOR_TEST_UNSET_VARIABLE"),
218            "{error}"
219        );
220    }
221
222    #[test]
223    fn empty_environment_variable_name_is_an_error() {
224        assert!(btc_config("env:", &[]).resolve_rpc_url().is_err());
225        assert!(
226            btc_config("http://btc", &[("Authorization", "env:")])
227                .resolve_http_headers()
228                .is_err()
229        );
230    }
