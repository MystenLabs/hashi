4//! The monitor's config file: the identity of one deployment.
5//!
6//! Why it exists: the file holds what differs between environments or changes
7//! over time, such as the guardian bucket, the accepted enclave builds, the
8//! Hashi package and the RPC endpoints. The timing bounds describe the
9//! protocol and are the same everywhere, so they live in `timing`, not here.
10//!
11//! How it works:
12//! 1. `load_yaml` parses the file into a YAML tree.
13//! 2. Every string value of the form `env:NAME` is replaced by the value of
14//!    the environment variable `NAME`. This keeps bucket names, endpoints and
15//!    credentials out of the file.
16//! 3. The tree is deserialized. Unknown keys are an error, so a typo cannot
17//!    be ignored.
18//!
19//! Assumptions:
20//! - `audit.sample.yaml` is the reference for the file's shape. The
21//!   deployment in sui-operations stores that shape verbatim.
22

33/// Configuration of the continuous monitor.
34#[derive(Clone, Debug, Deserialize)]
35#[serde(deny_unknown_fields)]
36pub struct Config {
37    /// The timing bounds in force. Not read from the file: production uses
38    /// the protocol set in `timing`, and tests set their own.
39    #[serde(skip)]
40    pub timing: Timing,
41
42    pub deployment: DeploymentConfig,
43    /// Omit to use AWS's default credential chain.
44    pub s3_credentials: Option<S3Credentials>,
45    pub sui: SuiConfig,
46    pub btc: BtcConfig,
47}
48
49#[derive(Clone, Debug, Deserialize)]
50#[serde(deny_unknown_fields)]
51pub struct SuiConfig {
52    /// Sui RPC endpoint.
53    pub rpc_url: String,
54
55    /// Original Hashi package: the event and object types the monitor reads keep
56    /// its address across upgrades.
57    pub package_id: String,
58}
59
60#[derive(Clone, Debug, Deserialize)]
61#[serde(deny_unknown_fields)]
62pub struct BtcConfig {
63    /// Bitcoin JSON-RPC endpoint.
64    pub rpc_url: String,
65
66    /// Optional HTTP headers for the JSON-RPC provider, such as the
67    /// authorization a hosted node needs.
68    #[serde(default)]
69    pub http_headers: BTreeMap<String, String>,
70}
71
72impl Config {
73    pub fn load_yaml(path: &Path) -> anyhow::Result<Self> {
74        let text = std::fs::read_to_string(path)
75            .with_context(|| format!("failed to read config file at {}", path.display()))?;
76        Self::from_yaml(&text)
77            .with_context(|| format!("failed to load config yaml at {}", path.display()))
78    }
79
80    /// Parse a config file's text. See the module doc for the `env:` rule.
81    pub fn from_yaml(text: &str) -> anyhow::Result<Self> {
82        let mut tree: serde_yaml::Value =
83            serde_yaml::from_str(text).context("failed to parse config yaml")?;
84        resolve_env_references(&mut tree, "")?;
85        serde_yaml::from_value(tree).context("failed to parse config yaml")
86    }
87}
88
89/// Replace every `env:NAME` string under `value` with the environment
90/// variable `NAME`. `path` names the value in errors.
91fn resolve_env_references(value: &mut serde_yaml::Value, path: &str) -> anyhow::Result<()> {
92    match value {
93        serde_yaml::Value::String(text) => {
94            if let Some(variable) = text.strip_prefix("env:") {
95                anyhow::ensure!(
96                    !variable.is_empty(),
97                    "{path}: environment variable name is empty"
98                );
99                *text = std::env::var(variable).with_context(|| {
100                    format!("{path}: environment variable {variable} is not set")
101                })?;
102            }
103        }
104        serde_yaml::Value::Mapping(mapping) => {
105            for (key, child) in mapping.iter_mut() {
106                let key = key.as_str().unwrap_or("?");
107                resolve_env_references(child, &format!("{path}.{key}"))?;
108            }
109        }
110        serde_yaml::Value::Sequence(items) => {
111            for (index, child) in items.iter_mut().enumerate() {
112                resolve_env_references(child, &format!("{path}[{index}]"))?;
113            }
114        }
115        _ => {}
116    }
117    Ok(())
118}

120#[cfg(test)]
121impl Config {
122    /// A regtest config with `Timing::for_tests`, unreachable RPC endpoints,
123    /// and explicit S3 credentials. Tests override the fields they exercise.
124    pub fn for_tests() -> Self {
125        Self {
126            timing: Timing::for_tests(),
127            deployment: DeploymentConfig {
128                bucket_info: hashi_types::guardian::S3BucketInfo {
129                    name: "bucket".to_string(),
130                    region: "us-east-1".to_string(),
131                },
132                retention_environment: hashi_types::guardian::S3RetentionEnvironment::Testnet,
133                bitcoin_network: bitcoin::Network::Regtest,
134                pcr_allowlist: hashi_types::guardian::PcrAllowlist::new(
135                    hashi_types::guardian::BuildPcrs::mock_for_testing("", 1),
136                    vec![],
137                )
138                .expect("valid PCR allowlist"),
139            },
140            s3_credentials: Some(S3Credentials {
141                access_key: "access-key".to_string(),
142                secret_key: "secret-key".to_string(),
143                session_token: None,
144            }),
145            sui: SuiConfig {
146                rpc_url: "http://sui".to_string(),
147                package_id: format!("0x{}", "11".repeat(32)),
148            },
149            btc: BtcConfig {
150                rpc_url: "http://btc".to_string(),
151                http_headers: BTreeMap::new(),
152            },
153        }
154    }
155}

157#[cfg(test)]
158mod tests {
159    use super::*;
160
161    fn sample_config_yaml() -> String {
162        include_str!("../audit.sample.yaml")
163            .replace("<PCR0_HEX_FROM_VERIFIED_BUILD>", &"11".repeat(48))
164    }
165
166    #[test]
167    fn the_sample_parses_once_its_placeholder_is_filled() {
168        let sample = include_str!("../audit.sample.yaml");
169        assert!(Config::from_yaml(sample).is_err());
170
171        // Cargo and nextest set CARGO_PKG_NAME for test processes.
172        let config = Config::from_yaml(&sample_config_yaml()).unwrap();
173
174        assert_eq!(config.deployment.bitcoin_network, bitcoin::Network::Signet);
175        assert!(config.s3_credentials.is_none());
176        assert_eq!(config.btc.rpc_url, env!("CARGO_PKG_NAME"));
177    }
178
179    #[test]
180    fn an_unknown_key_is_an_error() {
181        let yaml = format!("{}\ntyping: {{}}\n", sample_config_yaml());
182
183        let error = format!("{:#}", Config::from_yaml(&yaml).unwrap_err());
184
185        assert!(error.contains("unknown field `typing`"), "{error}");
186    }
187
188    #[test]
189    fn env_references_resolve_anywhere_in_the_file() {
190        let yaml = sample_config_yaml()
191            .replace(
192                "rpc_url: \"https://fullnode.testnet.sui.io:443\"",
193                "rpc_url: \"env:CARGO_PKG_NAME\"",
194            )
195            .replace(
196                "name: \"hashi-guardian-logs\"",
197                "name: \"env:CARGO_PKG_NAME\"",
198            )
199            .replace(
200                "  # http_headers:\n  #   Origin: \"https://example.com\"",
201                "  http_headers:\n    Origin: \"env:CARGO_PKG_NAME\"",
202            );
203
204        let config = Config::from_yaml(&yaml).unwrap();
205
206        assert_eq!(config.sui.rpc_url, env!("CARGO_PKG_NAME"));
207        assert_eq!(config.deployment.bucket_info.name, env!("CARGO_PKG_NAME"));
208        assert_eq!(config.btc.http_headers["Origin"], env!("CARGO_PKG_NAME"));
209    }
210
211    #[test]
212    fn an_unset_environment_variable_names_its_key() {
213        let yaml = sample_config_yaml().replace(
214            "rpc_url: \"env:CARGO_PKG_NAME\"",
215            "rpc_url: \"env:HASHI_MONITOR_TEST_UNSET_VARIABLE\"",
216        );
217
218        let error = format!("{:#}", Config::from_yaml(&yaml).unwrap_err());
219
220        assert!(
221            error.contains(".btc.rpc_url") && error.contains("HASHI_MONITOR_TEST_UNSET_VARIABLE"),
222            "{error}"
223        );
224    }
225
226    #[test]
227    fn an_empty_environment_variable_name_is_an_error() {
228        let yaml =
229            sample_config_yaml().replace("rpc_url: \"env:CARGO_PKG_NAME\"", "rpc_url: \"env:\"");
230
231        let error = format!("{:#}", Config::from_yaml(&yaml).unwrap_err());
232
233        assert!(error.contains(".btc.rpc_url"), "{error}");
234    }
235}
