4//! The auditor maintains the bridge's unspent transaction output (UTXO) pool.
5//!
6//! Why it exists:
7//! - Complete flows change the pool's holdings.
8//! - A due Bitcoin check starts after the pool update.
9//!
10//! How it works:
11//! 1. Remove closed state machines, then add their complete flows to the pool.
12//! 2. Check Bitcoin at the first maintenance tick.
13//! 3. Repeat successful checks after `UTXO_CHECK_INTERVAL` seconds.
14//! 4. Retry failed checks at the next pool maintenance tick.
15//!
16//! Assumptions:
17//! - Polling and maintenance ticks run sequentially.
18//!   For example, deposit D creates an output that withdrawal W spends. Bitcoin confirms both transactions.
19//!   Within one Bitcoin tick, the auditor processes withdrawal results before deposit results.
20//!   W therefore becomes complete while D still appears pending and its output is absent from the pool.
21//!   Maintenance during this gap could apply W without adding D and report `UtxoUnknownInputSpent`.
22//!   After the bitcoin tick finishes, maintenance takes both machines and adds D before applying W.
23
24use super::Auditor;
25use crate::domain::ConfirmedDeposit;
26use crate::domain::ConfirmedWithdrawal;
27use crate::domain::WithdrawalEventType;
28use crate::findings::MonitorFinding;
29use crate::time::human_timestamp_delta;
30use crate::time::utc_timestamp;
31use hashi_types::guardian::time::now_timestamp_secs;
32
33impl Auditor {
34    /// Update the pool from closed state machines, then run a due Bitcoin check.
35    /// Retry failed checks at the next maintenance tick.
36    pub fn tick_pool_maintenance(&mut self) -> anyhow::Result<()> {
37        let pool_findings = self.add_to_pool_and_gc();
38        self.report_findings("pool", &pool_findings);
39        self.metrics
40            .set_utxo_pool(self.utxo_pool.len(), self.utxo_pool.oldest_timestamp());
41        if self.pool_check_due(now_timestamp_secs()) {
42            self.check_utxo_pool()?;
43        }
44        Ok(())
45    }
46
47    /// Take closed machines and add their complete flows to the pool.
48    /// Call between ticks, after all Bitcoin results have been processed.
49    /// For complete withdrawals, dependent state machines must already be closed or ready to close.
50    pub fn add_to_pool_and_gc(&mut self) -> Vec<MonitorFinding> {
51        let mut findings = Vec::new();
52        let confirmed_deposits = self.take_closed_deposits();
53        let confirmed_withdrawals = self.take_closed_withdrawals();
54        for deposit in &confirmed_deposits {
55            findings.extend(self.utxo_pool.add_deposit(deposit));
56        }
57        findings.extend(self.utxo_pool.add_withdrawals(&confirmed_withdrawals));
58
59        findings
60    }
61
62    /// Remove the deposit machines that closed since the last call. Returns
63    /// the deposits Bitcoin confirmed, for the pool.
64    fn take_closed_deposits(&mut self) -> Vec<ConfirmedDeposit> {
65        let mut deposits = Vec::new();
66        for (request_id, sm) in self
67            .pending_deposits
68            .extract_if(|_, sm| sm.is_ready_for_gc())
69        {
70            let deposit_id = sm.deposit_id();
71            let deposit = sm.into_confirmed();
72            match &deposit {
73                Some(deposit) => tracing::debug!(
74                    deposit_id = %deposit.deposit_id,
75                    amount = deposit.amount,
76                    block_time = %utc_timestamp(deposit.block_time),
77                    "deposit request confirmed on Bitcoin at a bridge address"
78                ),
79                // Bitcoin answered without a UTXO, or the request lapsed.
80                None => tracing::debug!(
81                    %request_id,
82                    %deposit_id,
83                    "deposit request closed without a UTXO"
84                ),
85            }
86            deposits.extend(deposit);
87        }
88        deposits
89    }
90
91    /// Remove closed withdrawal machines and return their confirmed withdrawals for the pool.
92    fn take_closed_withdrawals(&mut self) -> Vec<ConfirmedWithdrawal> {
93        let mut withdrawals = Vec::new();
94        for (wid, sm) in self
95            .pending_withdrawals
96            .extract_if(|_, sm| sm.is_ready_for_gc())
97        {
98            let e1 = sm
99                .get(WithdrawalEventType::E1HashiApproved)
100                .expect("completed withdrawal has E1");
101            let e2 = sm
102                .get(WithdrawalEventType::E2GuardianApproved)
103                .expect("completed withdrawal has E2");
104            let e3 = sm
105                .get(WithdrawalEventType::E3BtcConfirmed)
106                .expect("completed withdrawal has E3");
107            tracing::debug!(
108                %wid,
109                e1_at = %utc_timestamp(e1.timestamp_secs),
110                e2_at = %utc_timestamp(e2.timestamp_secs),
111                e3_at = %utc_timestamp(e3.timestamp_secs),
112                e1_to_e2 = %human_timestamp_delta(e2.timestamp_secs, e1.timestamp_secs),
113                e2_to_e3 = %human_timestamp_delta(e3.timestamp_secs, e2.timestamp_secs),
114                "withdrawal flow is complete"
115            );
116            withdrawals.extend(sm.into_confirmed());
117        }
118        withdrawals
119    }
120}

122#[cfg(test)]
123mod tests {
124    use super::*;
125    use crate::audit::tests::auditor_with_pool;
126    use crate::audit::tests::complete_withdrawal;
127    use crate::test_utils::txid;
128    use bitcoin::OutPoint;
129    use hashi_types::guardian::WithdrawalID;
130
131    #[tokio::test]
132    async fn a_complete_withdrawal_spending_an_input_no_flow_names_is_a_finding() {
133        let mut auditor = auditor_with_pool().await;
134        let timing = auditor.cfg.timing.clone();
135        let unknown = OutPoint {
136            txid: txid(9),
137            vout: 1,
138        };
139        auditor.pending_withdrawals.insert(
140            WithdrawalID::new([7; 32]),
141            complete_withdrawal(7, 7, vec![unknown], &timing),
142        );
143
144        let findings = auditor.add_to_pool_and_gc();
145
146        assert_eq!(
147            findings,
148            vec![MonitorFinding::UtxoUnknownInputSpent {
149                outpoint: unknown,
150                txid: txid(7),
151            }]
152        );
153        assert!(auditor.pending_withdrawals.is_empty());
154    }
