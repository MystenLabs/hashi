4//! The auditor polls Sui, guardian logs, and Bitcoin through these methods.
5//!
6//! Why it exists:
7//! - `RpcPollers` owns the three source handles and records how far each source is checked.
8//! - Auditor tick handlers read each source and update the flow machines.
9//!
10//! How it works:
11//! 1. Read Sui events and guardian logs into the flow machines.
12//! 2. Before the mint, check that each fresh deposit approval names an unspent outpoint.
13//! 3. Check Bitcoin confirmations from each withdrawal's Sui approval on, also with no guardian log.
14//! 4. Find each source that is behind its limit. The auditor reports it as a `SourceStale` finding.
15//! 5. Leave pool updates and checks to the auditor scheduler.
16//! 6. At startup, catch up Sui, then the guardian, then Bitcoin. Retry a failed source with a
17//!    backoff.
18

41/// How far behind a source may fall before a `SourceStale` finding, in
42/// seconds. The guardian cursor is the start of the current hour directory,
43/// which is finished 10 minutes after the hour, at the next poll. So it
44/// normally lags up to about 75 minutes. Sui and Bitcoin lag a few minutes.
45pub(super) const SOURCE_STALE_LIMIT: u64 = 3 * 60 * 60;
46
47pub struct RpcPollers {
48    pub guardian_poller: GuardianWithdrawalsPoller,
49    pub sui_poller: SuiPoller,
50    pub btc_client: BtcRpcClient,
51    /// When the last full Bitcoin pass started. It starts at the startup
52    /// time. Thus a Bitcoin node that is down at startup becomes stale only
53    /// after `SOURCE_STALE_LIMIT`.
54    btc_checked_through: UnixSeconds,
55    /// Move's minimum time from a deposit approval to its mint, in seconds.
56    pub deposit_delay_secs: u64,
57}
58
59impl RpcPollers {
60    /// Initialize the source handles at their saved or genesis positions.
61    pub async fn new(
62        cfg: &Config,
63        guardian_start: GuardianCursor,
64        sui_start: u64,
65    ) -> anyhow::Result<Self> {
66        let mut guardian_poller = GuardianWithdrawalsPoller::new(cfg, guardian_start).await?;
67        let (hashi_btc_keys, hashi_object_id) = guardian_poller
68            .read_hashi_identity(cfg.deployment.bitcoin_network)
69            .await?;
70        let settings = read_hashi_settings(&cfg.sui, hashi_object_id).await?;
71        let confirmations = checked_confirmation_threshold(
72            settings.confirmation_threshold,
73            cfg.deployment.bitcoin_network,
74        )?;
75        tracing::info!(
76            confirmations,
77            deposit_delay_secs = settings.deposit_delay_secs,
78            "read Hashi's Bitcoin confirmation threshold and deposit delay"
79        );
80        let mut sui_poller = SuiPoller::new(&cfg.sui, hashi_btc_keys, sui_start)?;
81        sui_poller.find_hbtc_treasury_cap(hashi_object_id).await?;
82        Ok(Self {
83            guardian_poller,
84            sui_poller,
85            btc_client: BtcRpcClient::new(cfg, confirmations, SOURCE_STALE_LIMIT)?,
86            btc_checked_through: now_timestamp_secs(),
87            deposit_delay_secs: settings.deposit_delay_secs,
88        })
89    }
90
91    pub fn cursors(&self) -> Cursors {
92        Cursors {
93            sui: self.sui_read_until(),
94            guardian: self.guardian_read_until(),
95        }
96    }
97
98    /// A `SourceStale` finding for each source checked longer ago than
99    /// `SOURCE_STALE_LIMIT`.
100    pub fn stale_sources(&self, now: UnixSeconds) -> Vec<MonitorFinding> {
101        [
102            (SOURCE_SUI, self.sui_read_until()),
103            (SOURCE_GUARDIAN, self.guardian_read_until()),
104            (SOURCE_BTC, self.btc_read_until()),
105        ]
106        .into_iter()
107        .filter(|(_, checked_through)| now.saturating_sub(*checked_through) > SOURCE_STALE_LIMIT)
108        .map(|(source, checked_through)| MonitorFinding::SourceStale {
109            source,
110            checked_through,
111            limit_secs: SOURCE_STALE_LIMIT,
112        })
113        .collect()
114    }
115}

138impl Auditor {
139    pub async fn tick_sui(&mut self) -> anyhow::Result<()> {
140        // The head at the tick's start bounds the tick, so it ends.
141        let up_to = self.rpc_pollers.latest_sui_checkpoint().await?;
142        while let Some(poll) = self.rpc_pollers.sui_poller.poll(up_to).await? {
143            self.report_findings("sui", &poll.findings);
144            self.ingest_batch(poll.events);
145            self.metrics
146                .set_checked_through(SOURCE_SUI, self.rpc_pollers.sui_read_until());
147        }
148        Ok(())
149    }
150}

162impl Auditor {
163    /// Read every finished directory, then the new logs of the current one.
164    pub async fn tick_guardian(&mut self) -> anyhow::Result<()> {
165        loop {
166            let poll = self.rpc_pollers.guardian_poller.poll().await?;
167            self.report_findings("guardian", &poll.findings);
168            self.ingest_batch(poll.events);
169            self.metrics
170                .set_checked_through(SOURCE_GUARDIAN, self.rpc_pollers.guardian_read_until());
171            if !poll.dir_done {
172                return Ok(());
173            }
174        }
175    }
176}

188impl Auditor {
189    /// Collect Bitcoin lookup targets and log their counts.
190    fn txids_to_poll(&self) -> Vec<Txid> {
191        let withdrawals = self
192            .pending_withdrawals
193            .values()
194            .filter(|sm| sm.expects_btc());
195        let deposits = self.pending_deposits.values().filter(|sm| sm.expects_btc());
196        let withdrawal_count = withdrawals.clone().count();
197        let deposit_count = deposits.clone().count();
198        let unique_txids = withdrawals
199            .map(WithdrawalStateMachine::btc_txid)
200            .chain(deposits.map(DepositStateMachine::btc_txid))
201            .collect::<HashSet<_>>()
202            .into_iter()
203            .collect::<Vec<_>>();
204        tracing::info!(
205            withdrawal_count,
206            deposit_count,
207            unique_txids = unique_txids.len(),
208            "starting bitcoin confirmation lookups"
209        );
210        unique_txids
211    }
212
213    /// Pings Bitcoin RPC for all relevant withdrawals & deposits.
214    /// Returns domain findings and bubbles up infra errors. `started_at` is
215    /// taken before the lookups, so the machines record it as their check time.
216    fn fetch_btc_info(&mut self, started_at: UnixSeconds) -> anyhow::Result<Vec<MonitorFinding>> {
217        let tip_height = self.rpc_pollers.btc_client.synced_height()?;
218        // Report now, so that a later error in this tick does not lose them.
219        let fresh = self.check_fresh_approvals(tip_height, started_at)?;
220        self.report_findings("btc", &fresh);
221        // A "not confirmed yet" answer goes stale. A confirmed one is kept
222        // until no flow names its txid.
223        self.rpc_pollers.btc_client.forget_unconfirmed();
224        let unique_txids = self.txids_to_poll();
225        self.rpc_pollers
226            .btc_client
227            .prefetch_confirmations(&unique_txids, tip_height)?;
228        let mut findings = Vec::new();
229        for sm in self.pending_withdrawals.values_mut() {
230            findings.extend(sm.try_fetch_btc_tx(
231                &self.cfg.timing,
232                &self.rpc_pollers.btc_client,
233                started_at,
234            )?);
235        }
236
237        for sm in self.pending_deposits.values_mut() {
238            findings.extend(sm.try_fetch_btc_tx(
239                &self.cfg.timing,
240                &self.rpc_pollers.btc_client,
241                started_at,
242            )?);
243        }
244
245        Ok(findings)
246    }
247
248    /// Read Bitcoin confirmations and report findings for pending flows.
249    /// Complete flows' dependent machines must already be closed or ready to close when this tick finishes.
250    pub fn tick_btc(&mut self) -> anyhow::Result<()> {
251        let started_at = now_timestamp_secs();
252        let findings = self.fetch_btc_info(started_at)?;
253        // The machines Bitcoin answered stay until pool maintenance removes them.
254        tracing::info!(
255            withdrawals_waiting = self
256                .pending_withdrawals
257                .values()
258                .filter(|sm| sm.expects_btc())
259                .count(),
260            deposits_waiting = self
261                .pending_deposits
262                .values()
263                .filter(|sm| sm.expects_btc())
264                .count(),
265            "finished bitcoin confirmation lookups"
266        );
267        self.report_findings("btc", &findings);
268        self.rpc_pollers.btc_checked_through = started_at;
269        self.metrics.set_checked_through(SOURCE_BTC, started_at);
270        Ok(())
271    }
272}

274// ====================
275//   Initial catch-up
276// ====================
277
278/// The wait after the first failed try of an initial catch-up.
279const CATCH_UP_FIRST_BACKOFF: Duration = Duration::from_secs(1);
280/// The longest wait between two tries of a failed initial catch-up.
281const CATCH_UP_MAX_BACKOFF: Duration = Duration::from_secs(60);
282
283/// The wait before the next try of a failed initial catch-up. It doubles at
284/// each failure, up to `CATCH_UP_MAX_BACKOFF`.
285struct CatchUpBackoff {
286    delay: Duration,
287}
288
289impl CatchUpBackoff {
290    fn new() -> Self {
291        Self {
292            delay: CATCH_UP_FIRST_BACKOFF,
293        }
294    }
295
296    /// Log the failure, then wait before the next try.
297    async fn wait(&mut self, source: &'static str, error: anyhow::Error) {
298        tracing::warn!(
299            source,
300            ?error,
301            retry_in = ?self.delay,
302            "initial catch-up failed; retrying"
303        );
304        tokio::time::sleep(self.delay).await;
305        self.delay = (self.delay * 2).min(CATCH_UP_MAX_BACKOFF);
306    }
307}
308
309impl Auditor {
310    /// Catch up Sui, then the guardian, then Bitcoin. Retry a failed source
311    /// with a backoff until it is current. Sui and the guardian add the
312    /// machines, so Bitcoin comes last and looks up all of them.
313    pub async fn persistent_catch_up(&mut self) {
314        let mut backoff = CatchUpBackoff::new();
315        while let Err(error) = self.tick_sui().await {
316            backoff.wait(SOURCE_SUI, error).await;
317        }
318        tracing::info!(source = SOURCE_SUI, "finished initial catch-up");
319        let mut backoff = CatchUpBackoff::new();
320        while let Err(error) = self.tick_guardian().await {
321            backoff.wait(SOURCE_GUARDIAN, error).await;
322        }
323        tracing::info!(source = SOURCE_GUARDIAN, "finished initial catch-up");
324        let mut backoff = CatchUpBackoff::new();
325        while let Err(error) = self.tick_btc() {
326            backoff.wait(SOURCE_BTC, error).await;
327        }
328        tracing::info!(source = SOURCE_BTC, "finished initial catch-up");
329    }
330}

350#[cfg(test)]
351mod tests {
352    use super::*;
353
354    #[tokio::test]
355    async fn a_source_behind_its_limit_is_stale() {
356        let mut pollers = RpcPollers::for_tests(&Config::for_tests(), 1_700_000_000);
357        let sui = pollers.sui_read_until();
358        let guardian = pollers.guardian_read_until();
359        pollers.btc_checked_through = sui;
360        let stale = |now| {
361            pollers
362                .stale_sources(now)
363                .into_iter()
364                .map(|finding| match finding {
365                    MonitorFinding::SourceStale { source, .. } => source,
366                    other => panic!("unexpected {other}"),
367                })
368                .collect::<Vec<_>>()
369        };
370
371        // The guardian cursor is at the start of the hour, behind Sui.
372        assert!(guardian < sui);
373        assert!(stale(guardian + SOURCE_STALE_LIMIT).is_empty());
374        assert_eq!(
375            stale(guardian + SOURCE_STALE_LIMIT + 1),
376            vec![SOURCE_GUARDIAN]
377        );
378        assert_eq!(
379            stale(sui + SOURCE_STALE_LIMIT + 1),
380            vec![SOURCE_SUI, SOURCE_GUARDIAN, SOURCE_BTC]
381        );
382    }
