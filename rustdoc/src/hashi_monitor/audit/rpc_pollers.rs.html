4//! The auditor polls Sui, guardian logs, and Bitcoin through these methods.
5//!
6//! Why it exists:
7//! - `RpcPollers` owns the three source handles and records how far each source is checked.
8//! - Auditor tick handlers read each source and update the flow machines.
9//!
10//! How it works:
11//! 1. Read Sui events and guardian logs into the flow machines.
12//! 2. Before the mint, check that each fresh deposit approval names an unspent outpoint.
13//! 3. Check Bitcoin confirmations after each withdrawal has both Sui and guardian approvals.
14//! 4. Find each source that is behind its limit. The auditor reports it as a `SourceStale` finding.
15//! 5. Leave pool updates and checks to the auditor scheduler.
16

39/// How far behind a source may fall before a `SourceStale` finding, in
40/// seconds. The guardian cursor is the start of the current hour directory,
41/// which is finished 10 minutes after the hour, at the next poll. So it
42/// normally lags up to about 75 minutes. Sui and Bitcoin lag a few minutes.
43pub(super) const SOURCE_STALE_LIMIT: u64 = 3 * 60 * 60;
44
45pub struct RpcPollers {
46    pub guardian_poller: GuardianWithdrawalsPoller,
47    pub sui_poller: SuiPoller,
48    pub btc_client: BtcRpcClient,
49    /// When the last full Bitcoin pass started. It starts at the startup
50    /// time. Thus a Bitcoin node that is down at startup becomes stale only
51    /// after `SOURCE_STALE_LIMIT`.
52    btc_checked_through: UnixSeconds,
53    /// Move's minimum time from a deposit approval to its mint, in seconds.
54    pub deposit_delay_secs: u64,
55}
56
57impl RpcPollers {
58    /// Initialize the source handles at their saved or genesis positions.
59    pub async fn new(
60        cfg: &Config,
61        guardian_start: GuardianCursor,
62        sui_start: u64,
63    ) -> anyhow::Result<Self> {
64        let mut guardian_poller = GuardianWithdrawalsPoller::new(cfg, guardian_start).await?;
65        let (hashi_btc_keys, hashi_object_id) = guardian_poller
66            .read_hashi_identity(cfg.deployment.bitcoin_network)
67            .await?;
68        let settings = read_hashi_settings(&cfg.sui, hashi_object_id).await?;
69        let confirmations = checked_confirmation_threshold(
70            settings.confirmation_threshold,
71            cfg.deployment.bitcoin_network,
72        )?;
73        tracing::info!(
74            confirmations,
75            deposit_delay_secs = settings.deposit_delay_secs,
76            "read Hashi's Bitcoin confirmation threshold and deposit delay"
77        );
78        let mut sui_poller = SuiPoller::new(&cfg.sui, hashi_btc_keys, sui_start)?;
79        sui_poller.find_hbtc_treasury_cap(hashi_object_id).await?;
80        Ok(Self {
81            guardian_poller,
82            sui_poller,
83            btc_client: BtcRpcClient::new(cfg, confirmations, SOURCE_STALE_LIMIT)?,
84            btc_checked_through: now_timestamp_secs(),
85            deposit_delay_secs: settings.deposit_delay_secs,
86        })
87    }
88
89    pub fn cursors(&self) -> Cursors {
90        Cursors {
91            sui: self.sui_read_until(),
92            guardian: self.guardian_read_until(),
93        }
94    }
95
96    /// A `SourceStale` finding for each source checked longer ago than
97    /// `SOURCE_STALE_LIMIT`.
98    pub fn stale_sources(&self, now: UnixSeconds) -> Vec<MonitorFinding> {
99        [
100            (SOURCE_SUI, self.sui_read_until()),
101            (SOURCE_GUARDIAN, self.guardian_read_until()),
102            (SOURCE_BTC, self.btc_read_until()),
103        ]
104        .into_iter()
105        .filter(|(_, checked_through)| now.saturating_sub(*checked_through) > SOURCE_STALE_LIMIT)
106        .map(|(source, checked_through)| MonitorFinding::SourceStale {
107            source,
108            checked_through,
109            limit_secs: SOURCE_STALE_LIMIT,
110        })
111        .collect()
112    }
113}

136/// The wait after the first failed try of an initial catch-up.
137const CATCH_UP_FIRST_BACKOFF: Duration = Duration::from_secs(1);
138/// The longest wait between two tries of a failed initial catch-up.
139const CATCH_UP_MAX_BACKOFF: Duration = Duration::from_secs(60);
140
141/// The wait before the next try of a failed initial catch-up. It doubles at
142/// each failure, up to `CATCH_UP_MAX_BACKOFF`.
143struct CatchUpBackoff {
144    delay: Duration,
145}
146
147impl CatchUpBackoff {
148    fn new() -> Self {
149        Self {
150            delay: CATCH_UP_FIRST_BACKOFF,
151        }
152    }
153
154    /// Log the failure, then wait before the next try.
155    async fn wait(&mut self, source: &'static str, error: anyhow::Error) {
156        tracing::warn!(
157            source,
158            ?error,
159            retry_in = ?self.delay,
160            "initial catch-up failed; retrying"
161        );
162        tokio::time::sleep(self.delay).await;
163        self.delay = (self.delay * 2).min(CATCH_UP_MAX_BACKOFF);
164    }
165}
166
167impl Auditor {
168    /// Catch up Sui, then the guardian, then Bitcoin. Retry a failed source
169    /// with a backoff until it is current. Sui and the guardian add the
170    /// machines, so Bitcoin comes last and looks up all of them.
171    pub async fn catch_up_sources(&mut self) {
172        let mut backoff = CatchUpBackoff::new();
173        while let Err(error) = self.tick_sui().await {
174            backoff.wait(SOURCE_SUI, error).await;
175        }
176        tracing::info!(source = SOURCE_SUI, "finished initial catch-up");
177        let mut backoff = CatchUpBackoff::new();
178        while let Err(error) = self.tick_guardian().await {
179            backoff.wait(SOURCE_GUARDIAN, error).await;
180        }
181        tracing::info!(source = SOURCE_GUARDIAN, "finished initial catch-up");
182        let mut backoff = CatchUpBackoff::new();
183        while let Err(error) = self.tick_btc() {
184            backoff.wait(SOURCE_BTC, error).await;
185        }
186        tracing::info!(source = SOURCE_BTC, "finished initial catch-up");
187    }
188
189    pub async fn tick_sui(&mut self) -> anyhow::Result<()> {
190        // The head at the tick's start bounds the tick, so it ends.
191        let up_to = self.rpc_pollers.latest_sui_checkpoint().await?;
192        while let Some(poll) = self.rpc_pollers.sui_poller.poll(up_to).await? {
193            self.report_findings("sui", &poll.findings);
194            self.ingest_batch(poll.events);
195            self.metrics
196                .set_checked_through(SOURCE_SUI, self.rpc_pollers.sui_read_until());
197        }
198        Ok(())
199    }
200}

212impl Auditor {
213    /// Read every finished directory, then the new logs of the current one.
214    pub async fn tick_guardian(&mut self) -> anyhow::Result<()> {
215        loop {
216            let poll = self.rpc_pollers.guardian_poller.poll().await?;
217            self.report_findings("guardian", &poll.findings);
218            self.ingest_batch(poll.events);
219            self.metrics
220                .set_checked_through(SOURCE_GUARDIAN, self.rpc_pollers.guardian_read_until());
221            if !poll.dir_done {
222                return Ok(());
223            }
224        }
225    }
226}

238impl Auditor {
239    /// Collect Bitcoin lookup targets and log their counts.
240    fn txids_to_poll(&self) -> Vec<Txid> {
241        let withdrawals = self
242            .pending_withdrawals
243            .values()
244            .filter(|sm| sm.expects_btc());
245        let deposits = self.pending_deposits.values().filter(|sm| sm.expects_btc());
246        let withdrawal_count = withdrawals.clone().count();
247        let deposit_count = deposits.clone().count();
248        let unique_txids = withdrawals
249            .map(WithdrawalStateMachine::btc_txid)
250            .chain(deposits.map(DepositStateMachine::btc_txid))
251            .collect::<HashSet<_>>()
252            .into_iter()
253            .collect::<Vec<_>>();
254        tracing::info!(
255            withdrawal_count,
256            deposit_count,
257            unique_txids = unique_txids.len(),
258            "starting bitcoin confirmation lookups"
259        );
260        unique_txids
261    }
262
263    /// Pings Bitcoin RPC for all relevant withdrawals & deposits.
264    /// Returns domain findings and bubbles up infra errors. `started_at` is
265    /// taken before the lookups, so the machines record it as their check time.
266    fn fetch_btc_info(&mut self, started_at: UnixSeconds) -> anyhow::Result<Vec<MonitorFinding>> {
267        let tip_height = self.rpc_pollers.btc_client.synced_height()?;
268        // Report now, so that a later error in this tick does not lose them.
269        let fresh = self.check_fresh_approvals(tip_height, started_at)?;
270        self.report_findings("btc", &fresh);
271        // A "not confirmed yet" answer goes stale. A confirmed one is kept
272        // until no flow names its txid.
273        self.rpc_pollers.btc_client.forget_unconfirmed();
274        let unique_txids = self.txids_to_poll();
275        self.rpc_pollers
276            .btc_client
277            .prefetch_confirmations(&unique_txids, tip_height)?;
278        let mut findings = Vec::new();
279        for sm in self.pending_withdrawals.values_mut() {
280            findings.extend(sm.try_fetch_btc_tx(
281                &self.cfg.timing,
282                &self.rpc_pollers.btc_client,
283                started_at,
284            )?);
285        }
286
287        for sm in self.pending_deposits.values_mut() {
288            findings.extend(sm.try_fetch_btc_tx(
289                &self.cfg.timing,
290                &self.rpc_pollers.btc_client,
291                started_at,
292            )?);
293        }
294
295        Ok(findings)
296    }
297
298    /// Read Bitcoin confirmations and report findings for pending flows.
299    /// Complete flows' dependent machines must already be closed or ready to close when this tick finishes.
300    pub fn tick_btc(&mut self) -> anyhow::Result<()> {
301        let started_at = now_timestamp_secs();
302        let findings = self.fetch_btc_info(started_at)?;
303        // The machines Bitcoin answered stay until pool maintenance removes them.
304        tracing::info!(
305            withdrawals_waiting = self
306                .pending_withdrawals
307                .values()
308                .filter(|sm| sm.expects_btc())
309                .count(),
310            deposits_waiting = self
311                .pending_deposits
312                .values()
313                .filter(|sm| sm.expects_btc())
314                .count(),
315            "finished bitcoin confirmation lookups"
316        );
317        self.report_findings("btc", &findings);
318        self.rpc_pollers.btc_checked_through = started_at;
319        self.metrics.set_checked_through(SOURCE_BTC, started_at);
320        Ok(())
321    }
322}

342#[cfg(test)]
343mod tests {
344    use super::*;
345
346    #[tokio::test]
347    async fn a_source_behind_its_limit_is_stale() {
348        let mut pollers = RpcPollers::for_tests(&Config::for_tests(), 1_700_000_000);
349        let sui = pollers.sui_read_until();
350        let guardian = pollers.guardian_read_until();
351        pollers.btc_checked_through = sui;
352        let stale = |now| {
353            pollers
354                .stale_sources(now)
355                .into_iter()
356                .map(|finding| match finding {
357                    MonitorFinding::SourceStale { source, .. } => source,
358                    other => panic!("unexpected {other}"),
359                })
360                .collect::<Vec<_>>()
361        };
362
363        // The guardian cursor is at the start of the hour, behind Sui.
364        assert!(guardian < sui);
365        assert!(stale(guardian + SOURCE_STALE_LIMIT).is_empty());
366        assert_eq!(
367            stale(guardian + SOURCE_STALE_LIMIT + 1),
368            vec![SOURCE_GUARDIAN]
369        );
370        assert_eq!(
371            stale(sui + SOURCE_STALE_LIMIT + 1),
372            vec![SOURCE_SUI, SOURCE_GUARDIAN, SOURCE_BTC]
373        );
374    }
