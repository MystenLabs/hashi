4//! The auditor polls Sui, guardian logs, and Bitcoin through these methods.
5//!
6//! Why it exists:
7//! - `RpcPollers` owns the three source handles and records how far each source is checked.
8//! - Auditor tick handlers read each source and update the flow machines.
9//!
10//! How it works:
11//! 1. Read Sui events and guardian logs into the flow machines.
12//! 2. Check Bitcoin confirmations after each withdrawal has both Sui and guardian approvals.
13//! 3. Find each source that is behind its limit. The auditor reports it as a `SourceStale` finding.
14//! 4. Leave pool updates and checks to the auditor scheduler.
15

37/// How far behind a source may fall before a `SourceStale` finding, in
38/// seconds. The guardian cursor is the start of the current hour directory,
39/// which is finished 10 minutes after the hour, at the next poll. So it
40/// normally lags up to about 75 minutes. Sui and Bitcoin lag a few minutes.
41pub(super) const SOURCE_STALE_LIMIT: u64 = 3 * 60 * 60;
42
43pub struct RpcPollers {
44    pub guardian_poller: GuardianWithdrawalsPoller,
45    pub sui_poller: SuiPoller,
46    pub btc_client: BtcRpcClient,
47    /// When the last full Bitcoin pass started. It starts at the startup
48    /// time. Thus a Bitcoin node that is down at startup becomes stale only
49    /// after `SOURCE_STALE_LIMIT`.
50    btc_checked_through: UnixSeconds,
51}
52
53impl RpcPollers {
54    /// Initialize the source handles at their saved or genesis positions.
55    pub async fn new(
56        cfg: &Config,
57        guardian_start: GuardianCursor,
58        sui_start: u64,
59    ) -> anyhow::Result<Self> {
60        let mut guardian_poller = GuardianWithdrawalsPoller::new(cfg, guardian_start).await?;
61        let (hashi_btc_keys, hashi_object_id) = guardian_poller
62            .read_hashi_identity(cfg.deployment.bitcoin_network)
63            .await?;
64        let confirmations = checked_confirmation_threshold(
65            read_confirmation_threshold(&cfg.sui, hashi_object_id).await?,
66            cfg.deployment.bitcoin_network,
67        )?;
68        tracing::info!(confirmations, "read Hashi's Bitcoin confirmation threshold");
69        let mut sui_poller = SuiPoller::new(&cfg.sui, hashi_btc_keys, sui_start)?;
70        sui_poller.find_hbtc_treasury_cap(hashi_object_id).await?;
71        Ok(Self {
72            guardian_poller,
73            sui_poller,
74            btc_client: BtcRpcClient::new(cfg, confirmations)?,
75            btc_checked_through: now_timestamp_secs(),
76        })
77    }
78
79    pub fn cursors(&self) -> Cursors {
80        Cursors {
81            sui: self.sui_read_until(),
82            guardian: self.guardian_read_until(),
83        }
84    }
85
86    /// A `SourceStale` finding for each source checked longer ago than
87    /// `SOURCE_STALE_LIMIT`.
88    pub fn stale_sources(&self, now: UnixSeconds) -> Vec<MonitorFinding> {
89        [
90            (SOURCE_SUI, self.sui_read_until()),
91            (SOURCE_GUARDIAN, self.guardian_read_until()),
92            (SOURCE_BTC, self.btc_read_until()),
93        ]
94        .into_iter()
95        .filter(|(_, checked_through)| now.saturating_sub(*checked_through) > SOURCE_STALE_LIMIT)
96        .map(|(source, checked_through)| MonitorFinding::SourceStale {
97            source,
98            checked_through,
99            limit_secs: SOURCE_STALE_LIMIT,
100        })
101        .collect()
102    }
103}

109impl RpcPollers {
110    pub fn sui_read_until(&self) -> UnixSeconds {
111        self.sui_poller.read_until()
112    }
113
114    /// The newest checkpoint the Sui node serves.
115    async fn latest_sui_checkpoint(&mut self) -> anyhow::Result<u64> {
116        self.sui_poller.latest_checkpoint().await
117    }
118
119    /// Fail unless the Sui node still serves the start. See
120    /// `SuiPoller::ensure_start_is_served`.
121    pub async fn ensure_sui_serves_start(&mut self) -> anyhow::Result<()> {
122        self.sui_poller.ensure_start_is_served().await
123    }
124}
125
126impl Auditor {
127    pub async fn tick_sui(&mut self) -> anyhow::Result<()> {
128        // The head at the tick's start bounds the tick, so it ends.
129        let up_to = self.rpc_pollers.latest_sui_checkpoint().await?;
130        while let Some(poll) = self.rpc_pollers.sui_poller.poll(up_to).await? {
131            self.report_findings("sui", &poll.findings);
132            self.ingest_batch(poll.events);
133            self.metrics
134                .set_checked_through(SOURCE_SUI, self.rpc_pollers.sui_read_until());
135        }
136        Ok(())
137    }
138}

150impl Auditor {
151    /// Read every finished directory, then the new logs of the current one.
152    pub async fn tick_guardian(&mut self) -> anyhow::Result<()> {
153        loop {
154            let poll = self.rpc_pollers.guardian_poller.poll().await?;
155            self.report_findings("guardian", &poll.findings);
156            self.ingest_batch(poll.events);
157            self.metrics
158                .set_checked_through(SOURCE_GUARDIAN, self.rpc_pollers.guardian_read_until());
159            if !poll.dir_done {
160                return Ok(());
161            }
162        }
163    }
164}

176impl Auditor {
177    /// Collect Bitcoin lookup targets and log their counts.
178    fn txids_to_poll(&self) -> Vec<Txid> {
179        let withdrawals = self
180            .pending_withdrawals
181            .values()
182            .filter(|sm| sm.expects_btc());
183        let deposits = self.pending_deposits.values().filter(|sm| sm.expects_btc());
184        let withdrawal_count = withdrawals.clone().count();
185        let deposit_count = deposits.clone().count();
186        let unique_txids = withdrawals
187            .map(WithdrawalStateMachine::btc_txid)
188            .chain(deposits.map(DepositStateMachine::btc_txid))
189            .collect::<HashSet<_>>()
190            .into_iter()
191            .collect::<Vec<_>>();
192        tracing::info!(
193            withdrawal_count,
194            deposit_count,
195            unique_txids = unique_txids.len(),
196            "starting bitcoin confirmation lookups"
197        );
198        unique_txids
199    }
200
201    /// Pings Bitcoin RPC for all relevant withdrawals & deposits.
202    /// Returns domain findings and bubbles up infra errors. `started_at` is
203    /// taken before the lookups, so the machines record it as their check time.
204    fn fetch_btc_info(&mut self, started_at: UnixSeconds) -> anyhow::Result<Vec<MonitorFinding>> {
205        let tip_height = self.rpc_pollers.btc_client.synced_height()?;
206        // A "not confirmed yet" answer goes stale. A confirmed one is kept
207        // until no flow names its txid.
208        self.rpc_pollers.btc_client.forget_unconfirmed();
209        let unique_txids = self.txids_to_poll();
210        self.rpc_pollers
211            .btc_client
212            .prefetch_confirmations(&unique_txids, tip_height)?;
213        let mut findings = Vec::new();
214        for sm in self.pending_withdrawals.values_mut() {
215            findings.extend(sm.try_fetch_btc_tx(
216                &self.cfg.timing,
217                &self.rpc_pollers.btc_client,
218                started_at,
219            )?);
220        }
221
222        for sm in self.pending_deposits.values_mut() {
223            findings.extend(sm.try_fetch_btc_tx(
224                &self.cfg.timing,
225                &self.rpc_pollers.btc_client,
226                started_at,
227            )?);
228        }
229
230        Ok(findings)
231    }
232
233    /// Read Bitcoin confirmations and report findings for pending flows.
234    /// Complete flows' dependent machines must already be closed or ready to close when this tick finishes.
235    pub fn tick_btc(&mut self) -> anyhow::Result<()> {
236        let started_at = now_timestamp_secs();
237        let findings = self.fetch_btc_info(started_at)?;
238        self.report_findings("btc", &findings);
239        self.rpc_pollers.btc_checked_through = started_at;
240        self.metrics.set_checked_through(SOURCE_BTC, started_at);
241        Ok(())
242    }
243}

261#[cfg(test)]
262mod tests {
263    use super::*;
264
265    #[tokio::test]
266    async fn a_source_behind_its_limit_is_stale() {
267        let mut pollers = RpcPollers::for_tests(&Config::for_tests(), 1_700_000_000);
268        let sui = pollers.sui_read_until();
269        let guardian = pollers.guardian_read_until();
270        pollers.btc_checked_through = sui;
271        let stale = |now| {
272            pollers
273                .stale_sources(now)
274                .into_iter()
275                .map(|finding| match finding {
276                    MonitorFinding::SourceStale { source, .. } => source,
277                    other => panic!("unexpected {other}"),
278                })
279                .collect::<Vec<_>>()
280        };
281
282        // The guardian cursor is at the start of the hour, behind Sui.
283        assert!(guardian < sui);
284        assert!(stale(guardian + SOURCE_STALE_LIMIT).is_empty());
285        assert_eq!(
286            stale(guardian + SOURCE_STALE_LIMIT + 1),
287            vec![SOURCE_GUARDIAN]
288        );
289        assert_eq!(
290            stale(sui + SOURCE_STALE_LIMIT + 1),
291            vec![SOURCE_SUI, SOURCE_GUARDIAN, SOURCE_BTC]
292        );
293    }
