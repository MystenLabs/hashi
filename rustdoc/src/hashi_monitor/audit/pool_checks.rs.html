4//! The auditor checks its unspent transaction output (UTXO) pool against Bitcoin.
5//!
6//! Why it exists:
7//! - Bitcoin can report a spend before the auditor reads its Sui withdrawal record.
8//! - A suspect spend is a Bitcoin spend that the auditor has not yet judged.
9//!
10//! How it works:
11//! 1. Check Bitcoin at the first maintenance tick, then after `UTXO_CHECK_INTERVAL` seconds since the last successful check.
12//! 2. Keep suspect spends until Sui polling passes their lookup completion timestamps.
13//! 3. Clear spends that pending or completed withdrawals explain.
14//! 4. Report the remaining spends as findings.
15//! 5. Report `SourceStale` for the pool check when none has succeeded for
16//!    `UTXO_CHECK_INTERVAL` plus `SOURCE_STALE_LIMIT`. Before the first
17//!    success, count from the start of the audit loop, after the first
18//!    catch-up.
19
20use super::Auditor;
21use super::rpc_pollers::SOURCE_STALE_LIMIT;
22use crate::findings::MonitorFinding;
23use crate::metrics::SOURCE_UTXO;
24use bitcoin::OutPoint;
25use bitcoin::Txid;
26use hashi_types::guardian::time::UnixSeconds;
27use hashi_types::guardian::time::now_timestamp_secs;
28use std::collections::HashSet;
29
30/// How long after a successful pool check the next one is due, in seconds.
31/// Every tick runs on the one audit task, so a check delays the polls by
32/// the time its batched lookups take.
33const UTXO_CHECK_INTERVAL: u64 = 24 * 60 * 60;
34
35impl Auditor {
36    /// Check each pool UTXO on Bitcoin and keep unexplained spends for `check_suspect_spends`.
37    /// Report findings and record the check time after success.
38    pub fn check_utxo_pool(&mut self) -> anyhow::Result<()> {
39        let tip_height = self.rpc_pollers.btc_client.synced_height()?;
40        let pool = &self.utxo_pool;
41        let outpoints = pool.outpoints();
42        tracing::info!(utxos = outpoints.len(), "starting UTXO pool check");
43        let states = self
44            .rpc_pollers
45            .btc_client
46            .lookup_outpoint_states(&outpoints, tip_height, false)?;
47        // Unlike `tick_btc` confirmation checks, `gettxout` reads live state.
48        // The pool check does not use one Bitcoin snapshot and can see spends from newer blocks.
49        // Record completion time so Sui polling must cover the whole lookup period.
50        let now = now_timestamp_secs();
51        let (findings, suspects) = pool.check(&states, now, &self.cfg.timing);
52        tracing::info!(
53            utxos = outpoints.len(),
54            findings = findings.len(),
55            suspects = suspects.len(),
56            "finished UTXO pool check"
57        );
58        // A suspect already waiting keeps its earlier time.
59        for (outpoint, suspect) in suspects {
60            self.suspect_spends.entry(outpoint).or_insert(suspect);
61        }
62        self.report_findings(SOURCE_UTXO, &findings);
63        self.metrics.set_checked_through(SOURCE_UTXO, now);
64        self.last_successful_pool_check = Some(now);
65        Ok(())
66    }
67
68    /// Whether a pool check is due at `now`: none has succeeded in this run,
69    /// or the last one finished `UTXO_CHECK_INTERVAL` ago or more.
70    pub fn pool_check_due(&self, now: UnixSeconds) -> bool {
71        self.last_successful_pool_check
72            .is_none_or(|last| now.saturating_sub(last) >= UTXO_CHECK_INTERVAL)
73    }
74
75    /// A `SourceStale` finding when no pool check has succeeded for
76    /// `UTXO_CHECK_INTERVAL` plus `SOURCE_STALE_LIMIT` at `now`.
77    pub fn stale_pool_check(&self, now: UnixSeconds) -> Option<MonitorFinding> {
78        let checked_through = self.last_successful_pool_check.unwrap_or(self.started_at);
79        let limit_secs = UTXO_CHECK_INTERVAL + SOURCE_STALE_LIMIT;
80        (now.saturating_sub(checked_through) > limit_secs).then_some(MonitorFinding::SourceStale {
81            source: SOURCE_UTXO,
82            checked_through,
83            limit_secs,
84        })
85    }
86
87    /// The inputs of every withdrawal the monitor follows. It spends them
88    /// on Bitcoin before its flow completes, so a spend of one is explained.
89    fn pending_withdrawal_inputs(&self) -> HashSet<OutPoint> {
90        self.pending_withdrawals
91            .values()
92            .flat_map(|sm| sm.tx().inputs.iter())
93            .map(|input| OutPoint {
94                txid: Txid::from(input.id.txid),
95                vout: input.id.vout,
96            })
97            .collect()
98    }
99
100    /// Wait until Sui polling passes the lookup completion time.
101    /// Clear spends that a pending withdrawal names or that the pool already took.
102    /// Report the remaining spends as unexpected. Return the findings for tests.
103    pub fn check_suspect_spends(&mut self) -> Vec<MonitorFinding> {
104        let wsm_inputs = self.pending_withdrawal_inputs();
105        let sui_cursor = self.rpc_pollers.sui_read_until();
106        let mut findings = Vec::new();
107        let mut judged = 0;
108        for (outpoint, suspect) in self
109            .suspect_spends
110            .extract_if(|_, suspect| sui_cursor > suspect.checked_at)
111        {
112            judged += 1;
113            if wsm_inputs.contains(&outpoint) || !self.utxo_pool.holds(&outpoint) {
114                tracing::info!(%outpoint, "a withdrawal explains the spend the pool check saw");
115            } else {
116                findings.push(MonitorFinding::UtxoUnexpectedSpend {
117                    outpoint,
118                    amount: suspect.amount,
119                });
120            }
121        }
122        // Only when there is work, so a quiet pool adds no line.
123        if judged > 0 || !self.suspect_spends.is_empty() {
124            tracing::info!(
125                judged,
126                unexpected = findings.len(),
127                waiting_for_sui = self.suspect_spends.len(),
128                "checked the suspect spends"
129            );
130        }
131        self.report_findings(SOURCE_UTXO, &findings);
132        findings
133    }
134}

152    #[tokio::test]
153    async fn a_suspect_spend_waits_while_the_sui_cursor_is_not_past_its_time() {
154        let mut auditor = auditor_with_pool().await;
155        let cursor = auditor.rpc_pollers.sui_read_until();
156        let outpoint = OutPoint {
157            txid: txid(9),
158            vout: 1,
159        };
160        auditor.suspect_spends.insert(
161            outpoint,
162            SuspectSpend {
163                checked_at: cursor,
164                amount: 5_000,
165            },
166        );
167
168        assert!(auditor.check_suspect_spends().is_empty());
169        assert!(auditor.suspect_spends.contains_key(&outpoint));
170    }

172    #[tokio::test]
173    async fn a_suspect_spend_is_unexpected_once_the_sui_cursor_is_past_its_time() {
174        let mut auditor = auditor_with_pool().await;
175        let cursor = auditor.rpc_pollers.sui_read_until();
176        let outpoint = OutPoint {
177            txid: txid(9),
178            vout: 1,
179        };
180        auditor.suspect_spends.insert(
181            outpoint,
182            SuspectSpend {
183                checked_at: cursor - 1,
184                amount: 5_000,
185            },
186        );
187        assert!(
188            auditor
189                .utxo_pool
190                .add_deposit(&ConfirmedDeposit {
191                    deposit_id: DepositId::new(txid(9), 1),
192                    amount: 5_000,
193                    script_pubkey: p2wpkh(0x54),
194                    block_time: 900,
195                })
196                .is_none()
197        );
198
199        let findings = auditor.check_suspect_spends();
200
201        assert_eq!(
202            findings,
203            vec![MonitorFinding::UtxoUnexpectedSpend {
204                outpoint,
205                amount: 5_000,
206            }]
207        );
208        assert!(auditor.suspect_spends.is_empty());
209    }

237    #[tokio::test]
238    async fn a_suspect_spend_the_pool_took_a_withdrawal_for_is_cleared_without_a_finding() {
239        let mut auditor = auditor_with_pool().await;
240        let timing = auditor.cfg.timing.clone();
241        let cursor = auditor.rpc_pollers.sui_read_until();
242        // `withdrawal_tx` spends `txid(0x51):1`.
243        let outpoint = OutPoint {
244            txid: txid(0x51),
245            vout: 1,
246        };
247        auditor.suspect_spends.insert(
248            outpoint,
249            SuspectSpend {
250                checked_at: cursor - 1,
251                amount: 5_000,
252            },
253        );
254        // The withdrawal completed after the check: the pool took it, and
255        // the auditor no longer follows it.
256        let withdrawal = complete_withdrawal(7, 7, vec![outpoint], &timing);
257        let pool = &mut auditor.utxo_pool;
258        assert!(

264            })
265            .is_none()
266        );
267        assert!(
268            pool.add_withdrawals(&[withdrawal.confirmed().cloned().unwrap()])
269                .is_empty()
270        );
271
272        assert!(auditor.check_suspect_spends().is_empty());
273        assert!(auditor.suspect_spends.is_empty());
274    }

276    #[tokio::test]
277    async fn a_pool_check_that_does_not_succeed_is_stale() {
278        let cfg = Config::for_tests();
279        let mut auditor = auditor_with(&cfg, Arc::new(MonitorMetrics::new()));
280        let limit_secs = UTXO_CHECK_INTERVAL + SOURCE_STALE_LIMIT;
281        let stale = |checked_through| {
282            Some(MonitorFinding::SourceStale {
283                source: SOURCE_UTXO,
284                checked_through,
285                limit_secs,
286            })
287        };
288
289        // Before the first success, the run's start counts.
290        let start = auditor.started_at;
291        assert_eq!(auditor.stale_pool_check(start + limit_secs), None);
292        assert_eq!(
293            auditor.stale_pool_check(start + limit_secs + 1),
294            stale(start)
295        );
296
297        auditor.last_successful_pool_check = Some(start + 10);
298        assert_eq!(auditor.stale_pool_check(start + 10 + limit_secs), None);
299        assert_eq!(
300            auditor.stale_pool_check(start + 10 + limit_secs + 1),
301            stale(start + 10)
302        );
303    }

305    #[tokio::test]
306    async fn a_pool_check_is_due_until_one_succeeds_and_a_day_after() {
307        let cfg = Config::for_tests();
308        let mut auditor = auditor_with(&cfg, Arc::new(MonitorMetrics::new()));
309        assert!(auditor.pool_check_due(1_000));
310
311        auditor.last_successful_pool_check = Some(1_000);
312        assert!(!auditor.pool_check_due(1_000 + UTXO_CHECK_INTERVAL - 1));
313        assert!(auditor.pool_check_due(1_000 + UTXO_CHECK_INTERVAL));
314    }
