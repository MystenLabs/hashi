4//! The auditor checks that each deposit approval names an output that pays
5//! it, on the monitor's node and on a reference source.
6//!
7//! Why it exists:
8//! - Move refuses only the outpoints that it tracks. A dishonest committee
9//!   can approve a deposit that Bitcoin never paid, or spend an untracked
10//!   bridge output through the recovery leaf, then approve a deposit of it.
11//! - Whoever controls the peers of the monitor's node can withhold blocks
12//!   from it, so an honest payment looks missing there. The reference source
13//!   (see `rpc::esplora`) has other peers.
14//!
15//! How it works:
16//! 1. On ingest, keep each approval, with a deadline `APPROVAL_DEADLINE`
17//!    after the monitor read it.
18//! 2. At each Bitcoin tick, even when the node fails its sync check, look the
19//!    kept approvals up on both sources. A source that fails answers nothing.
20//! 3. While the approval is fresh, either source showing the outpoint spent,
21//!    also in the mempool, is `DepositUnbacked` (`Spent`). Withholding blocks
22//!    cannot fake a spend. An approval is fresh until `DELAY_END_MARGIN`
23//!    before its mint delay ends.
24//! 4. Otherwise, either source showing an output that pays the approval,
25//!    spent or not, settles it.
26//! 5. Neither source showing that by the deadline is `DepositUnbacked`
27//!    (`Unverified`).
28//!
29//! Each approval is settled once, then dropped.
30//!
31//! Assumptions:
32//! - An honest deposit is confirmed when Hashi approves it, so a working
33//!   source shows it paid at the first lookup. Both sources failing until the
34//!   deadline gives a false `Unverified`.
35//! - No honest transaction spends a deposit outpoint before its delay ends:
36//!   it joins Move's active pool at `confirm_deposit`. After that, a
37//!   withdrawal may spend it.
38//! - The snapshot does not keep the approvals. After a restart, the catch-up
39//!   reads the recent ones again and checks them once more.
40
41use super::Auditor;
42use crate::domain::SuiDepositApprovedEvent;
43use crate::findings::DepositUnbackedReason;
44use crate::findings::MonitorFinding;
45use crate::rpc::btc::OutpointState;
46use crate::rpc::esplora::ReferenceOutput;
47use bitcoin::OutPoint;

50use hashi_types::guardian::time::now_timestamp_secs;
51use std::collections::HashMap;

53/// How long before the mint delay ends a spend stops counting, in seconds.
54/// It covers lookup time and clock skew between the monitor and Sui.
55const DELAY_END_MARGIN: u64 = 2 * 60;
56
57/// How long after the monitor reads an approval a source must show its
58/// payment, in seconds. It leaves most of Move's ten-minute delay for the
59/// pause.
60const APPROVAL_DEADLINE: u64 = 3 * 60;
61
62/// Whether the delay of `event` ends more than `DELAY_END_MARGIN` after `now`.
63fn is_fresh(event: &SuiDepositApprovedEvent, now: UnixSeconds, delay: u64) -> bool {
64    now.saturating_add(DELAY_END_MARGIN) < event.timestamp_secs.saturating_add(delay)
65}
66
67impl Auditor {
68    /// Keep each request's first approval and deadline, even if another request names the same output.
69    pub(super) fn note_approval(&mut self, event: &SuiDepositApprovedEvent, now: UnixSeconds) {
70        self.unsettled_approvals
71            .entry(event.request_id)
72            .or_insert_with(|| (event.clone(), now.saturating_add(APPROVAL_DEADLINE)));
73    }
74
75    /// Judge each kept approval on the answers, at `finished`, when the
76    /// lookups returned. Keep only the approvals that neither source settled,
77    /// until their deadline.
78    fn judge_approvals(
79        &mut self,
80        node: &HashMap<OutPoint, OutpointState>,
81        reference: &HashMap<OutPoint, ReferenceOutput>,
82        finished: UnixSeconds,
83    ) -> Vec<MonitorFinding> {
84        let delay = self.rpc_pollers.deposit_delay_secs;
85        let mut findings = Vec::new();
86        self.unsettled_approvals.retain(|_, (event, deadline)| {
87            let outpoint = OutPoint::from(event.deposit_id);
88            let pays = |amount: u64, script_pubkey: &ScriptBuf| {
89                amount == event.amount && *script_pubkey == event.script_pubkey
90            };
91            let (node_spent, node_paid) = match node.get(&outpoint) {
92                Some(OutpointState::Spent { output }) => {
93                    (true, pays(output.value.to_sat(), &output.script_pubkey))
94                }
95                Some(OutpointState::Unspent { output, .. }) => {
96                    (false, pays(output.value.to_sat(), &output.script_pubkey))
97                }
98                Some(OutpointState::Missing) | None => (false, false),
99            };
100            let (reference_spent, reference_paid) = match reference.get(&outpoint) {
101                Some(ReferenceOutput::Spent {
102                    amount,
103                    script_pubkey,
104                }) => (true, pays(*amount, script_pubkey)),
105                Some(ReferenceOutput::Unspent {
106                    amount,
107                    script_pubkey,
108                }) => (false, pays(*amount, script_pubkey)),
109                Some(ReferenceOutput::Missing) | None => (false, false),
110            };
111            let unbacked = |reason| MonitorFinding::DepositUnbacked {
112                sui: event.clone(),
113                reason,
114            };
115            if (node_spent || reference_spent) && is_fresh(event, finished, delay) {
116                findings.push(unbacked(DepositUnbackedReason::Spent));
117                return false;
118            }
119            if node_paid || reference_paid {
120                return false;
121            }
122            if finished >= *deadline {
123                findings.push(unbacked(DepositUnbackedReason::Unverified));
124                return false;
125            }
126            true
127        });
128        findings
129    }
130
131    /// Look up the kept approvals on the node at `tip_height`, if it passed
132    /// its sync check, and on the reference source.
133    pub(super) fn check_approvals(&mut self, tip_height: Option<u64>) -> Vec<MonitorFinding> {
134        if self.unsettled_approvals.is_empty() {
135            return Vec::new();
136        }
137        let mut outpoints: Vec<_> = self
138            .unsettled_approvals
139            .values()
140            .map(|(event, _)| OutPoint::from(event.deposit_id))
141            .collect();
142        outpoints.sort_unstable();
143        outpoints.dedup();
144        tracing::info!(
145            approvals = self.unsettled_approvals.len(),
146            "checking deposit approvals"
147        );
148        let node = match tip_height {
149            Some(tip_height) => self
150                .rpc_pollers
151                .btc_client
152                .lookup_outpoint_states(&outpoints, tip_height, true)
153                .unwrap_or_else(|error| {
154                    tracing::warn!(?error, "node lookup of deposit approvals failed");
155                    HashMap::new()
156                }),
157            None => HashMap::new(),
158        };
159        let reference = self
160            .rpc_pollers
161            .reference_btc
162            .lookup_outputs(&outpoints)
163            .unwrap_or_else(|error| {
164                tracing::warn!(?error, "reference lookup of deposit approvals failed");
165                HashMap::new()
166            });
167        self.judge_approvals(&node, &reference, now_timestamp_secs())
168    }
169}

181    #[tokio::test]
182    async fn every_approval_is_kept_until_its_deadline() {
183        let mut auditor = auditor_with_pool().await;
184        let now = now_timestamp_secs();
185        // Read long after its mint.
186        let late = approval_of(&deposit_request(), now - 3_600);
187        auditor.ingest(MonitorEvent::DepositApproved(late.clone()));
188        assert_eq!(
189            auditor.unsettled_approvals[&late.request_id].1,
190            now + APPROVAL_DEADLINE
191        );
192    }

194    #[tokio::test]
195    async fn approvals_for_the_same_output_are_checked_separately() {
196        let mut auditor = auditor_with_pool().await;
197        let paid = approval_of(&deposit_request(), 1_000);
198        let other = SuiDepositApprovedEvent {
199            request_id: sui_sdk_types::Address::new([2; 32]),
200            script_pubkey: crate::test_utils::p2wpkh(0x55),
201            ..paid.clone()
202        };
203        auditor.note_approval(&paid, 1_000);
204        auditor.note_approval(&other, 1_001);
205        let reference = HashMap::from([(
206            OutPoint::from(paid.deposit_id),
207            ReferenceOutput::Unspent {
208                amount: paid.amount,
209                script_pubkey: paid.script_pubkey,
210            },
211        )]);
212        let deadline = 1_001 + APPROVAL_DEADLINE;
213        let node = HashMap::new();
214        assert!(
215            auditor
216                .judge_approvals(&node, &reference, deadline - 1)
217                .is_empty()
218        );
219        assert!(!auditor.unsettled_approvals.contains_key(&paid.request_id));
220        let findings = auditor.judge_approvals(&node, &reference, deadline);
221        assert_eq!(
222            findings,
223            [MonitorFinding::DepositUnbacked {
224                sui: other,
225                reason: DepositUnbackedReason::Unverified,
226            }]
227        );
228        assert!(findings[0].autopauses());
229    }

231    #[tokio::test]
232    async fn either_source_can_settle_or_condemn_an_approval() {
233        use DepositUnbackedReason::*;
234        let approval = approval_of(&deposit_request(), 1_000);
235        let output = |amount, script_pubkey| TxOut {
236            value: Amount::from_sat(amount),
237            script_pubkey,
238        };
239        let paid = output(approval.amount, approval.script_pubkey.clone());
240        let node_unspent = OutpointState::Unspent {
241            block_time: 900,
242            output: paid.clone(),
243        };
244        let node_spent = OutpointState::Spent {
245            output: paid.clone(),
246        };
247        let reference = |spent, amount| {
248            let script_pubkey = approval.script_pubkey.clone();
249            if spent {
250                ReferenceOutput::Spent {
251                    amount,
252                    script_pubkey,
253                }
254            } else {
255                ReferenceOutput::Unspent {
256                    amount,
257                    script_pubkey,
258                }
259            }
260        };
261        let amount = approval.amount;
262        // Read at 1_000, so the deadline is 1_180. Fresh until 1_480.
263        let deadline = 1_000 + APPROVAL_DEADLINE;
264        // Read at 5_000, after the mint, so the deadline is 5_180.
265        let late = 5_000;
266        let cases = [
267            // (read at, node, reference, judged at, finding, kept)
268            (
269                1_000,
270                Some(node_spent.clone()),
271                None,
272                1_000,
273                Some(Spent),
274                false,
275            ),
276            (
277                1_000,
278                Some(node_unspent.clone()),
279                Some(reference(true, amount)),
280                1_000,
281                Some(Spent),
282                false,
283            ),
284            (1_000, Some(node_unspent.clone()), None, 1_000, None, false),
285            // A node that peers withhold blocks from does not see the payment.
286            (
287                1_000,
288                Some(OutpointState::Missing),
289                Some(reference(false, amount)),
290                1_000,
291                None,
292                false,
293            ),
294            (
295                1_000,
296                Some(OutpointState::Missing),
297                Some(ReferenceOutput::Missing),
298                deadline - 1,
299                None,
300                true,
301            ),
302            (1_000, None, None, deadline, Some(Unverified), false),
303            // Another amount, or a real output that pays someone else, is
304            // not the approved payment.
305            (
306                1_000,
307                None,
308                Some(reference(false, amount + 1)),
309                deadline,
310                Some(Unverified),
311                false,
312            ),
313            (
314                1_000,
315                Some(OutpointState::Unspent {
316                    block_time: 900,
317                    output: output(amount, ScriptBuf::new()),
318                }),
319                None,
320                deadline,
321                Some(Unverified),
322                false,
323            ),
324            // After the mint, a withdrawal may spend the deposit.
325            (late, Some(node_spent.clone()), None, late, None, false),
326            (late, None, Some(reference(true, amount)), late, None, false),
327            (
328                late,
329                None,
330                None,
331                late + APPROVAL_DEADLINE,
332                Some(Unverified),
333                false,
334            ),
335        ];
336        for (read_at, node, reference, finished, reason, kept) in cases {
337            let case = format!("read at {read_at}: {node:?} {reference:?} at {finished}");
338            let mut auditor = auditor_with_pool().await;
339            auditor.note_approval(&approval, read_at);
340            let outpoint = OutPoint::from(approval.deposit_id);
341            let node = node
342                .map(|s| HashMap::from([(outpoint, s)]))
343                .unwrap_or_default();
344            let reference = reference
345                .map(|s| HashMap::from([(outpoint, s)]))
346                .unwrap_or_default();
347            let findings = auditor.judge_approvals(&node, &reference, finished);
348            let expected = reason
349                .map(|reason| MonitorFinding::DepositUnbacked {
350                    sui: approval.clone(),
351                    reason,
352                })
353                .into_iter()
354                .collect::<Vec<_>>();
355            assert_eq!(findings, expected, "{case}");
356            assert_eq!(auditor.unsettled_approvals.is_empty(), !kept, "{case}");
357        }
358    }
