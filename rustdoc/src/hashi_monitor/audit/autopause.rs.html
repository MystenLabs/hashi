4//! Pause Hashi when a reported finding autopauses. The pause starts at once,
5//! on a separate task, and only one runs at a time. A pause of an already
6//! paused Hashi does nothing.
7
8use super::Auditor;
9use crate::findings::MonitorFinding;
10use crate::metrics::AUTOPAUSE_FAILED;
11use crate::metrics::AUTOPAUSE_SUCCEEDED;
12use crate::pause::initiate_pause;
13use hashi_types::guardian::time::now_timestamp_secs;
14
15impl Auditor {
16    /// Pause Hashi on the first finding that autopauses, if autopause is
17    /// enabled and no pause is running.
18    pub(super) fn autopause(&mut self, findings: &[MonitorFinding]) {
19        if !self.cfg.pause.as_ref().is_some_and(|pause| pause.autopause) {
20            return;
21        }
22        if self
23            .pause_task
24            .as_ref()
25            .is_some_and(|task| !task.is_finished())
26        {
27            return;
28        }
29        let Some(finding) = findings.iter().find(|finding| finding.autopauses()) else {
30            return;
31        };
32        let cfg = self.cfg.clone();
33        let metrics = self.metrics.clone();
34        let finding = finding.to_string();
35        self.pause_task = Some(tokio::spawn(async move {
36            let result = match initiate_pause(&cfg).await {
37                Ok(false) => return,
38                Ok(true) => {
39                    tracing::error!(finding, "autopause paused Hashi");
40                    AUTOPAUSE_SUCCEEDED
41                }
42                Err(error) => {
43                    tracing::error!(finding, ?error, "autopause failed to pause Hashi");
44                    AUTOPAUSE_FAILED
45                }
46            };
47            metrics.record_autopause(result, now_timestamp_secs());
48        }));
49    }
50}

52#[cfg(test)]
53mod tests {
54    use super::*;
55    use crate::audit::tests::auditor_with;
56    use crate::config::Config;
57    use crate::config::PauseConfig;
58    use crate::metrics::MonitorMetrics;
59    use crate::test_utils::approval_of;
60    use crate::test_utils::deposit_request;
61    use std::sync::Arc;
62
63    fn autopause_auditor(autopause: bool) -> (Auditor, Arc<MonitorMetrics>) {
64        let mut cfg = Config::for_tests();
65        cfg.pause = Some(PauseConfig {
66            // Nothing listens here, so a pause fails.
67            transaction_rpc_url: "http://127.0.0.1:1".to_owned(),
68            object_id: None,
69            operators: Vec::new(),
70            autopause,
71        });
72        let metrics = Arc::new(MonitorMetrics::new());
73        (auditor_with(&cfg, metrics.clone()), metrics)
74    }
75
76    fn unbacked() -> MonitorFinding {
77        MonitorFinding::DepositUnbacked {
78            sui: approval_of(&deposit_request(), 200),
79            reason: crate::findings::DepositUnbackedReason::Unverified,
80        }
81    }
82
83    #[tokio::test]
84    async fn a_finding_starts_one_pause_at_a_time() {
85        let (mut auditor, metrics) = autopause_auditor(true);
86        auditor.report_findings("test", &[unbacked()]);
87        let first = auditor.pause_task.as_ref().unwrap().id();
88        auditor.report_findings("test", &[unbacked()]);
89        assert_eq!(auditor.pause_task.as_ref().unwrap().id(), first);
90
91        auditor.pause_task.take().unwrap().await.unwrap();
92        assert!(metrics.render().contains(r#"result="failed""#));
93    }
94
95    #[tokio::test]
96    async fn no_pause_when_disabled() {
97        let (mut auditor, _) = autopause_auditor(false);
98        auditor.report_findings("test", &[unbacked()]);
99        assert!(auditor.pause_task.is_none());
100    }
