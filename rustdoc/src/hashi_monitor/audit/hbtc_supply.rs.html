4//! The auditor checks that Bitcoin backs every hBTC on Sui.
5//!
6//! Why it exists: the other checks follow the events that the monitor reads.
7//! A mint path that the monitor does not follow, or a Move bug, changes the
8//! hBTC supply without such an event. One comparison of state catches all of
9//! them.
10//!
11//! How it works:
12//! 1. After the pool update in each maintenance tick, read the hBTC supply and sum the
13//!    held UTXO amounts. Export both as gauges.
14//! 2. While the supply exceeds the pool value, keep the time it started.
15//! 3. Report `HbtcSupplyExceedsPool` at each tick once that lasts
16//!    `HBTC_SUPPLY_GRACE`.
17//! 4. A failed read skips the check. When no read has succeeded for
18//!    `SOURCE_STALE_LIMIT`, counted from the start of the audit loop at the
19//!    earliest, the maintenance tick reports `SourceStale` for `hbtc_supply`.
20//!
21//! Assumptions:
22//! - In an honest bridge, the supply never exceeds the pool value. The pool
23//!   takes a deposit at its Bitcoin confirmation, and Move mints at least
24//!   `deposit_time_delay_ms` after the approval. Move burns a withdrawal at
25//!   E1, and the pool drops it at E3. The burned amount includes the miner
26//!   fee, so it equals what leaves the pool.
27//! - The supply exceeds the pool value only for a short time when the
28//!   monitor's Bitcoin side lags. `HBTC_SUPPLY_GRACE` is `SOURCE_STALE_LIMIT`:
29//!   a longer lag is a `SourceStale` finding already.
30
31use super::Auditor;
32use super::rpc_pollers::SOURCE_STALE_LIMIT;
33use crate::findings::MonitorFinding;
34use hashi_types::guardian::time::UnixSeconds;
35use hashi_types::guardian::time::now_timestamp_secs;
36
37/// How long the hBTC supply may exceed the pool value before a finding, in
38/// seconds.
39const HBTC_SUPPLY_GRACE: u64 = SOURCE_STALE_LIMIT;
40
41/// The `SourceStale` source name of the hBTC supply read.
42const SOURCE_HBTC_SUPPLY: &str = "hbtc_supply";
43
44impl Auditor {
45    /// Read the hBTC supply and compare it with the pool value. A failed read
46    /// skips the check; see `stale_hbtc_supply`.
47    pub async fn check_hbtc_supply(&mut self) {
48        let supply = match self.read_hbtc_supply().await {
49            Ok(supply) => supply,
50            Err(error) => {
51                tracing::warn!(?error, "hBTC supply read failed; retrying at the next tick");
52                return;
53            }
54        };
55        let pool_value = self.utxo_pool.value();
56        let (since, finding) = judge_hbtc_supply(
57            self.hbtc_supply_above_pool_since,
58            supply,
59            pool_value,
60            now_timestamp_secs(),
61        );
62        // Log only a change, so a steady state adds no line.
63        match (self.hbtc_supply_above_pool_since, since) {
64            (None, Some(_)) => tracing::info!(
65                hbtc_supply = supply,
66                pool_value,
67                grace_secs = HBTC_SUPPLY_GRACE,
68                "the hBTC supply exceeds the pool value; waiting for the grace period"
69            ),
70            (Some(_), None) => tracing::info!(
71                hbtc_supply = supply,
72                pool_value,
73                "the pool value backs the hBTC supply again"
74            ),
75            _ => {}
76        }
77        self.hbtc_supply_above_pool_since = since;
78        self.report_findings("hbtc_supply", finding.as_slice());
79    }
80
81    /// Read the hBTC supply and export it with the pool value.
82    pub async fn read_hbtc_supply(&mut self) -> anyhow::Result<u64> {
83        let supply = self.rpc_pollers.sui_poller.hbtc_supply().await?;
84        self.last_hbtc_supply_read = now_timestamp_secs();
85        self.metrics
86            .set_hbtc_backing(supply, self.utxo_pool.value());
87        Ok(supply)
88    }
89}
90
91impl Auditor {
92    /// A `SourceStale` finding when no hBTC supply read has succeeded for
93    /// `SOURCE_STALE_LIMIT` at `now`. The time counts from the later of the
94    /// last read and the start of the audit loop, because the first catch-up
95    /// does not read the supply.
96    pub fn stale_hbtc_supply(&self, now: UnixSeconds) -> Option<MonitorFinding> {
97        let checked_through = self.last_hbtc_supply_read.max(self.started_at);
98        (now.saturating_sub(checked_through) > SOURCE_STALE_LIMIT).then_some(
99            MonitorFinding::SourceStale {
100                source: SOURCE_HBTC_SUPPLY,
101                checked_through,
102                limit_secs: SOURCE_STALE_LIMIT,
103            },
104        )
105    }
106}
107
108/// Judge the hBTC supply against the pool value at `now`. Returns the new
109/// time since the supply exceeds the pool, and a finding once that lasts
110/// `HBTC_SUPPLY_GRACE`.
111fn judge_hbtc_supply(
112    above_since: Option<UnixSeconds>,
113    hbtc_supply: u64,
114    pool_value: u64,
115    now: UnixSeconds,
116) -> (Option<UnixSeconds>, Option<MonitorFinding>) {
117    if hbtc_supply <= pool_value {
118        return (None, None);
119    }
120    let since = above_since.unwrap_or(now);
121    let finding = (now.saturating_sub(since) >= HBTC_SUPPLY_GRACE).then_some(
122        MonitorFinding::HbtcSupplyExceedsPool {
123            hbtc_supply,
124            pool_value,
125            since,
126        },
127    );
128    (Some(since), finding)
129}

131#[cfg(test)]
132mod tests {
133    use super::*;
134    use crate::audit::tests::auditor_with;
135    use crate::config::Config;
136    use crate::metrics::MonitorMetrics;
137    use std::sync::Arc;
138
139    #[tokio::test]
140    async fn a_supply_read_that_does_not_succeed_is_stale() {
141        let cfg = Config::for_tests();
142        let mut auditor = auditor_with(&cfg, Arc::new(MonitorMetrics::new()));
143        let read_at = auditor.last_hbtc_supply_read;
144
145        assert_eq!(
146            auditor.stale_hbtc_supply(read_at + SOURCE_STALE_LIMIT),
147            None
148        );
149        assert_eq!(
150            auditor.stale_hbtc_supply(read_at + SOURCE_STALE_LIMIT + 1),
151            Some(MonitorFinding::SourceStale {
152                source: SOURCE_HBTC_SUPPLY,
153                checked_through: read_at,
154                limit_secs: SOURCE_STALE_LIMIT,
155            })
156        );
157
158        // A loop that starts after a long catch-up counts from its start.
159        auditor.started_at = read_at + 10 * SOURCE_STALE_LIMIT;
160        assert_eq!(
161            auditor.stale_hbtc_supply(auditor.started_at + SOURCE_STALE_LIMIT),
162            None
163        );
164    }
165
166    #[test]
167    fn a_supply_above_the_pool_is_reported_after_the_grace() {
168        // Below or equal is fine and clears an earlier excess.
169        assert_eq!(judge_hbtc_supply(Some(5), 100, 100, 10), (None, None));
170
171        // The first excess starts the grace.
172        assert_eq!(
173            judge_hbtc_supply(None, 101, 100, 1_000),
174            (Some(1_000), None)
175        );
176        let within = 1_000 + HBTC_SUPPLY_GRACE - 1;
177        assert_eq!(
178            judge_hbtc_supply(Some(1_000), 101, 100, within),
179            (Some(1_000), None)
180        );
181
182        let after = 1_000 + HBTC_SUPPLY_GRACE;
183        assert_eq!(
184            judge_hbtc_supply(Some(1_000), 101, 100, after),
185            (
186                Some(1_000),
187                Some(MonitorFinding::HbtcSupplyExceedsPool {
188                    hbtc_supply: 101,
189                    pool_value: 100,
190                    since: 1_000,
191                })
192            )
193        );
194    }
