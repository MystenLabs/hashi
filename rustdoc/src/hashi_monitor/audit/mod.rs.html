4//! The auditor monitors bridge events and reports findings.
5//!
6//! Why it exists:
7//! - The monitor checks deposits, withdrawals, and unspent transaction outputs (UTXOs).
8//! - Metrics provide findings and source progress for alerts.
9//!
10//! How it works:
11//! 1. Restore the source positions, flow machines, and UTXO pool from a snapshot.
12//! 2. Without a snapshot, start at the bridge genesis with an empty pool.
13//! 3. Fail at startup if the Sui node has pruned the start checkpoint.
14//! 4. Read Sui events and guardian logs into the flow machines.
15//! 5. Check Bitcoin confirmations after both withdrawal approvals arrive.
16//! 6. Update the pool every `POOL_MAINTENANCE_INTERVAL` and check Bitcoin when the daily check is due.
17//! 7. Check Bitcoin at the first maintenance tick. Repeat successful pool checks after `UTXO_CHECK_INTERVAL` seconds.
18//! 8. Judge suspect spends after the Sui cursor passes the Bitcoin lookup time.
19//! 9. Report timing violations and save snapshots every `STATE_TICK_INTERVAL`.
20//! 10. Log a failed source tick. The next tick of that source retries it.
21//! 11. At each state tick, report a source or a snapshot save that is behind its limit.
22//! 12. After each pool maintenance tick, check that the UTXO pool backs the hBTC supply.
23//!
24//! Assumptions:
25//! - Every tick runs on one task. The first catch-up blocks the state tick,
26//!   so a first run reports no stale source while it catches up.
27

69/// How often Sui is polled. A deposit approval must reach the monitor
70/// early in Move's delay before the mint, ten minutes by default.
71const SUI_POLL_INTERVAL: Duration = Duration::from_secs(60);
72
73/// How often Bitcoin is asked about the pending transactions. An approval
74/// is reported unbacked at the second empty lookup after it, so with
75/// `SUI_POLL_INTERVAL` the alert comes within five minutes of it.
76const BTC_POLL_INTERVAL: Duration = Duration::from_secs(2 * 60);
77
78/// How often the guardian's log is polled. Each poll lists the current
79/// directory and reads only its new logs. The withdrawal bounds are hours.
80const GUARDIAN_POLL_INTERVAL: Duration = Duration::from_secs(5 * 60);
81
82/// How often pool maintenance runs.
83const POOL_MAINTENANCE_INTERVAL: Duration = Duration::from_secs(5 * 60);
84
85/// How often the state checks run.
86const STATE_TICK_INTERVAL: Duration = Duration::from_secs(5 * 60);
87
88/// How long snapshot saves may fail before a `SnapshotStale` finding, in
89/// seconds. A failed save costs only a longer replay at the next restart.
90const SNAPSHOT_STALE_LIMIT: u64 = 12 * 60 * 60;
91
92pub fn log_findings(source: &'static str, phase: &'static str, findings: &[MonitorFinding]) {
93    for finding in findings.iter() {
94        tracing::error!(
95            source,
96            phase,
97            total = findings.len(),
98            finding = %finding,
99            "monitor finding"
100        );
101    }
102}

104pub struct Auditor {
105    cfg: Config,
106    pending_withdrawals: HashMap<WithdrawalID, WithdrawalStateMachine>,
107    /// Deposit requests Bitcoin is still asked about. Keyed by the Sui
108    /// request id, so a repeated event is ignored and each request has its
109    /// own.
110    pending_deposits: HashMap<Address, DepositStateMachine>,
111    /// The UTXOs the bridge holds, by Bitcoin's account. See `crate::domain::utxo_pool`. The
112    /// snapshot keeps it between runs.
113    utxo_pool: UtxoPool,
114    /// Spends waiting for the Sui cursor, by outpoint.
115    suspect_spends: HashMap<OutPoint, SuspectSpend>,
116    rpc_pollers: RpcPollers,
117    metrics: Arc<MonitorMetrics>,
118    /// Where the snapshot is kept between runs.
119    store: SnapshotStore,
120    /// When the last successful pool check finished. `None` until one
121    /// succeeds in this run.
122    last_successful_pool_check: Option<UnixSeconds>,
123    /// When the last snapshot save succeeded.
124    last_snapshot_save: UnixSeconds,
125    /// Since when the hBTC supply has exceeded the pool value, if it does.
126    hbtc_supply_above_pool_since: Option<UnixSeconds>,
127}

129impl Auditor {
130    /// Restore the snapshot, or start at the bridge genesis.
131    /// Fail if the store cannot be read or written, or Sui has removed the required history.
132    pub async fn new(
133        cfg: &Config,
134        metrics: Arc<MonitorMetrics>,
135        store: SnapshotStore,
136    ) -> anyhow::Result<Self> {
137        let (state, pool) = match this_deployments(cfg, store.load().await?) {
138            Some(snapshot) => {
139                tracing::info!(
140                    sui_next_checkpoint = snapshot.state.sui_next_checkpoint,
141                    guardian_cursor = %utc_timestamp(snapshot.state.guardian_cursor.read_until()),
142                    withdrawals = snapshot.state.withdrawals.len(),
143                    deposits = snapshot.state.deposits.len(),
144                    suspect_spends = snapshot.state.suspect_spends.len(),
145                    utxos_held = snapshot.pool.len(),
146                    "loaded the snapshot; resuming where it stopped"
147                );
148                (Some(snapshot.state), snapshot.pool)
149            }
150            None => (None, UtxoPool::default()),
151        };
152        let (guardian_start, sui_start) = match &state {
153            Some(state) => (state.guardian_cursor.clone(), state.sui_next_checkpoint),
154            None => {
155                let genesis = package_genesis(&cfg.sui).await?;
156                tracing::info!(
157                    genesis = %utc_timestamp(genesis.timestamp),
158                    genesis_checkpoint = genesis.checkpoint,
159                    "first run: starting at the bridge's genesis with an empty UTXO pool"
160                );
161                (
162                    GuardianCursor::new(S3HourDirectory::withdraw(genesis.timestamp)?),
163                    genesis.checkpoint,
164                )
165            }
166        };
167        let rpc_pollers = RpcPollers::new(cfg, guardian_start, sui_start).await?;
168        let mut auditor = Self {
169            cfg: cfg.clone(),
170            pending_withdrawals: HashMap::new(),
171            pending_deposits: HashMap::new(),
172            utxo_pool: pool,
173            suspect_spends: HashMap::new(),
174            rpc_pollers,
175            metrics,
176            store,
177            last_successful_pool_check: None,
178            last_snapshot_save: 0,
179            hbtc_supply_above_pool_since: None,
180        };
181        if let Some(state) = state {
182            auditor.restore(state);
183        }
184        // A pruned start leaves the pool incomplete.
185        auditor.rpc_pollers.ensure_sui_serves_start().await?;
186        // A setup that cannot read the supply fails here, not silently later.
187        let supply = auditor
188            .read_hbtc_supply()
189            .await
190            .context("failed to read the hBTC supply")?;
191        tracing::info!(hbtc_supply = supply, "read the hBTC supply");
192        auditor
193            .metrics
194            .set_checked_through(SOURCE_SUI, auditor.rpc_pollers.sui_read_until());
195        auditor
196            .metrics
197            .set_checked_through(SOURCE_GUARDIAN, auditor.rpc_pollers.guardian_read_until());
198        auditor
199            .metrics
200            .set_checked_through(SOURCE_BTC, auditor.rpc_pollers.btc_read_until());
201        // Check write access at startup. A missing bucket loads as empty.
202        auditor
203            .store
204            .save(&auditor.cfg, auditor.state(), auditor.utxo_pool())
205            .await?;
206        let now = now_timestamp_secs();
207        auditor.last_snapshot_save = now;
208        auditor.metrics.set_checked_through(SOURCE_SNAPSHOT, now);
209        Ok(auditor)
210    }

212    /// The pollers' positions, the open machines and the suspect spends, for
213    /// the snapshot.
214    fn state(&self) -> AuditState {
215        AuditState {
216            sui_next_checkpoint: self.rpc_pollers.sui_poller.next_item_to_poll(),
217            guardian_cursor: self.rpc_pollers.guardian_poller.next_item_to_poll(),
218            withdrawals: self.pending_withdrawals.values().cloned().collect(),
219            deposits: self.pending_deposits.values().cloned().collect(),
220            suspect_spends: self.suspect_spends.clone().into_iter().collect(),
221            hbtc_supply_above_pool_since: self.hbtc_supply_above_pool_since,
222        }
223    }
224
225    /// Hold the machines and the suspect spends a snapshot saved.
226    fn restore(&mut self, state: AuditState) {
227        for sm in state.withdrawals {
228            self.pending_withdrawals.insert(sm.wid(), sm);
229        }
230        for sm in state.deposits {
231            self.pending_deposits.insert(sm.request_id(), sm);
232        }
233        self.suspect_spends.extend(state.suspect_spends);
234        self.hbtc_supply_above_pool_since = state.hbtc_supply_above_pool_since;
235    }
236
237    fn ingest(&mut self, event: MonitorEvent) -> Vec<MonitorFinding> {
238        match event {
239            MonitorEvent::Withdrawal(event) => self.ingest_withdrawal(event),
240            MonitorEvent::DepositApproved(event) => self.ingest_deposit_approved(event),
241            MonitorEvent::DepositRequested(event) => self.ingest_deposit_request(event),
242        }
243    }
244
245    fn ingest_deposit_request(&mut self, event: SuiDepositRequestedEvent) -> Vec<MonitorFinding> {
246        match self.pending_deposits.entry(event.request_id) {
247            Entry::Occupied(entry) => entry.get().add_sui_request(&event).into_iter().collect(),
248            Entry::Vacant(entry) => {
249                entry.insert(DepositStateMachine::new(event));
250                Vec::new()
251            }
252        }
253    }
254
255    fn ingest_withdrawal(&mut self, event: MonitorWithdrawalEvent) -> Vec<MonitorFinding> {
256        let wid = event.wid;
257        match self.pending_withdrawals.entry(wid) {
258            Entry::Occupied(mut entry) => entry.get_mut().add_event(event, &self.cfg.timing),
259            Entry::Vacant(entry) => {
260                entry.insert(WithdrawalStateMachine::new(event, &self.cfg.timing));
261                Vec::new()
262            }
263        }
264    }
265
266    /// The Sui `DepositApproved`. Move approves only a request it holds, so
267    /// the request came first. While its machine is open, the machine keeps
268    /// the approval. Once it closed, the pool has its answer.
269    fn ingest_deposit_approved(&mut self, event: SuiDepositApprovedEvent) -> Vec<MonitorFinding> {
270        match self.pending_deposits.get_mut(&event.request_id) {
271            Some(sm) => sm.add_sui_approval(event, &self.cfg.timing),
272            None => self
273                .utxo_pool
274                .check_sui_approval(&event, self.cfg.timing.deposit_approval_skew),
275        }
276    }
277
278    pub fn ingest_batch(&mut self, events: Vec<MonitorEvent>) -> Vec<MonitorFinding> {
279        let mut findings = Vec::new();
280        for event in events {
281            findings.extend(self.ingest(event));
282        }
283        self.report_findings("ingest", &findings);
284        findings
285    }

287    fn detect_timing_violations(&self) -> Vec<MonitorFinding> {
288        let mut findings = Vec::new();
289        for sm in self.pending_withdrawals.values() {
290            findings.extend(sm.timing_violations(&self.rpc_pollers.cursors()));
291        }
292        findings
293    }

295    /// The open withdrawals whose first event is the earliest, for the
296    /// progress log. An operator sees what the audit still waits for.
297    fn oldest_open_withdrawals(&self) -> String {
298        let earliest = self
299            .pending_withdrawals
300            .values()
301            .map(WithdrawalStateMachine::earliest_event_at)
302            .min();
303        let oldest = self
304            .pending_withdrawals
305            .values()
306            .filter(|sm| Some(sm.earliest_event_at()) == earliest)
307            .map(|sm| {
308                format!(
309                    "wid={}, btc_txid={}, earliest_event_at={}, waiting_for={:?}",
310                    sm.wid(),
311                    sm.btc_txid(),
312                    utc_timestamp(sm.earliest_event_at()),
313                    sm.expected_event_types(),
314                )
315            })
316            .collect::<Vec<_>>();
317        if oldest.is_empty() {
318            "none".to_string()
319        } else {
320            oldest.join("; ")
321        }
322    }
323
324    /// Return the deposits with the earliest request timestamp for the progress log.
325    fn oldest_open_deposits(&self) -> String {
326        let earliest = self
327            .pending_deposits
328            .values()
329            .map(DepositStateMachine::requested_at)
330            .min();
331        let oldest = self
332            .pending_deposits
333            .values()
334            .filter(|sm| Some(sm.requested_at()) == earliest)
335            .map(|sm| {
336                format!(
337                    "request_id={}, deposit_id={}, requested_at={}",
338                    sm.request_id(),
339                    sm.deposit_id(),
340                    utc_timestamp(sm.requested_at()),
341                )
342            })
343            .collect::<Vec<_>>();
344        if oldest.is_empty() {
345            "none".to_string()
346        } else {
347            oldest.join("; ")
348        }
349    }

351    /// The number of open withdrawal and deposit machines.
352    fn open_flows(&self) -> (usize, usize) {
353        (self.pending_withdrawals.len(), self.pending_deposits.len())

356    pub fn utxo_pool(&self) -> &UtxoPool {
357        &self.utxo_pool
358    }

360    /// Save the snapshot. A failed save is logged and does not stop the
361    /// audit; the next state check saves again. Alerting watches the time
362    /// of the last good save.
363    async fn save_snapshot(&mut self) {
364        let saved = self
365            .store
366            .save(&self.cfg, self.state(), self.utxo_pool())
367            .await;
368        match saved {
369            Ok(()) => {
370                self.last_snapshot_save = now_timestamp_secs();
371                self.metrics
372                    .set_checked_through(SOURCE_SNAPSHOT, self.last_snapshot_save);
373            }
374            Err(error) => tracing::error!(?error, "failed to save the snapshot; continuing"),
375        }
376    }
377
378    /// A `SnapshotStale` finding if no save has succeeded within the limit.
379    fn stale_snapshot(&self, now: UnixSeconds) -> Option<MonitorFinding> {
380        (now.saturating_sub(self.last_snapshot_save) > SNAPSHOT_STALE_LIMIT).then_some(
381            MonitorFinding::SnapshotStale {
382                last_saved: self.last_snapshot_save,
383                limit_secs: SNAPSHOT_STALE_LIMIT,
384            },
385        )
386    }
387
388    fn report_findings(&self, phase: &'static str, findings: &[MonitorFinding]) {
389        log_findings("continuous", phase, findings);
390        self.metrics.record_findings(findings, now_timestamp_secs());
391    }

393    /// Log the outcome of one source's initial catch-up. A failure does not
394    /// stop the audit: the source's next tick retries it.
395    fn log_initial_catch_up(source: &'static str, result: anyhow::Result<()>) {
396        match result {
397            Ok(()) => tracing::info!(source, "finished initial catch-up"),
398            Err(error) => tracing::warn!(source, ?error, "initial catch-up failed; continuing"),
399        }
400    }
401
402    async fn tick_state_checks(&mut self) {
403        let violations = self.detect_timing_violations();
404        self.report_findings("timing", &violations);
405        let spends = self.judge_suspect_spends();
406        self.report_findings(SOURCE_UTXO, &spends);
407
408        let (withdrawals, deposits) = self.open_flows();
409        tracing::info!(
410            "continuous progress checkpoint:\n  guardian_cursor={}\n  sui_cursor={}\n  open_withdrawals={}\n  open_deposits={}\n  oldest_withdrawals={}\n  oldest_deposits={}",
411            utc_timestamp(self.rpc_pollers.guardian_read_until()),
412            utc_timestamp(self.rpc_pollers.sui_read_until()),
413            withdrawals,
414            deposits,
415            self.oldest_open_withdrawals(),
416            self.oldest_open_deposits(),
417        );
418        let stale = self.rpc_pollers.stale_sources(now_timestamp_secs());
419        self.report_findings("progress", &stale);
420
421        self.save_snapshot().await;
422        // After the save, so a save that succeeds now is not reported.
423        let stale = self.stale_snapshot(now_timestamp_secs());
424        self.report_findings("progress", stale.as_slice());
425    }

427    pub async fn run(&mut self) -> anyhow::Result<()> {
428        tracing::info!(
429            "starting continuous audit:\n  sui_start={}\n  guardian_start={}",
430            utc_timestamp(self.rpc_pollers.sui_read_until()),
431            utc_timestamp(self.rpc_pollers.guardian_read_until()),
432        );
433        let sui = self.tick_sui().await;
434        Self::log_initial_catch_up(SOURCE_SUI, sui);
435        let guardian = self.tick_guardian().await;
436        Self::log_initial_catch_up(SOURCE_GUARDIAN, guardian);
437        let btc = self.tick_btc();
438        Self::log_initial_catch_up(SOURCE_BTC, btc);
439
440        let mut sui_ticker = tokio::time::interval(SUI_POLL_INTERVAL);
441        let mut guardian_ticker = tokio::time::interval(GUARDIAN_POLL_INTERVAL);
442        let mut btc_ticker = tokio::time::interval(BTC_POLL_INTERVAL);
443        let mut pool_maintenance_ticker = tokio::time::interval(POOL_MAINTENANCE_INTERVAL);
444        let mut state_checks_ticker = tokio::time::interval(STATE_TICK_INTERVAL);
445
446        sui_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
447        guardian_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
448        btc_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
449        pool_maintenance_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
450        state_checks_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
451
452        // Consume each immediate tick so scheduled work starts after its interval.
453        sui_ticker.tick().await;
454        guardian_ticker.tick().await;
455        btc_ticker.tick().await;
456        pool_maintenance_ticker.tick().await;
457        state_checks_ticker.tick().await;
458
459        loop {
460            tokio::select! {
461                _ = sui_ticker.tick() => {
462                    if let Err(error) = self.tick_sui().await {
463                        tracing::warn!(source = "sui", ?error, "poll failed; continuing");
464                    }
465                }
466                _ = guardian_ticker.tick() => {
467                    if let Err(error) = self.tick_guardian().await {
468                        tracing::warn!(source = "guardian", ?error, "poll failed; continuing");
469                    }
470                }
471                _ = btc_ticker.tick() => {
472                    if let Err(error) = self.tick_btc() {
473                        tracing::warn!(source = "btc", ?error, "btc tick failed; continuing");
474                    }
475                }
476                _ = pool_maintenance_ticker.tick() => {
477                    if let Err(error) = self.tick_pool_maintenance() {
478                        tracing::warn!(source = SOURCE_UTXO, ?error, "pool maintenance failed; retrying at the next tick");
479                    }
480                    self.check_hbtc_supply().await;
481                }
482                _ = state_checks_ticker.tick() => {
483                    self.tick_state_checks().await;
484                }
485            }
486        }
487    }
488}

490/// The snapshot, if this deployment saved it. Otherwise `None`: the audit
491/// is a first run.
492fn this_deployments(cfg: &Config, snapshot: Option<Snapshot>) -> Option<Snapshot> {
493    match snapshot {
494        Some(snapshot) if snapshot.is_for(cfg) => Some(snapshot),
495        Some(snapshot) => {
496            tracing::warn!(
497                guardian_bucket = snapshot.guardian_bucket,
498                package_id = snapshot.package_id,
499                "ignoring the snapshot of another deployment"
500            );
501            None
502        }
503        None => {
504            tracing::info!("no snapshot");
505            None
506        }
507    }
508}

529    #[tokio::test]
530    async fn a_repeated_deposit_request_is_ingested_once() {
531        let approval = hashi_approval();
532        let mut auditor = auditor_with_pool().await;
533        let request = SuiDepositRequestedEvent {
534            timestamp_secs: approval.timestamp_secs,
535            deposit_id: DepositId::new(approval.btc_txid, 0),
536            ..deposit_request()
537        };
538
539        assert!(
540            auditor
541                .ingest(MonitorEvent::DepositRequested(request.clone()))
542                .is_empty()
543        );
544        assert!(
545            auditor
546                .ingest(MonitorEvent::DepositRequested(request.clone()))
547                .is_empty()
548        );
549        assert_eq!(auditor.pending_deposits.len(), 1);
550
551        // The same request id with other contents is a contradiction.
552        let contradiction = SuiDepositRequestedEvent {
553            amount: request.amount + 1,
554            ..request.clone()
555        };
556        assert!(matches!(
557            &auditor.ingest(MonitorEvent::DepositRequested(contradiction))[..],
558            [MonitorFinding::InvalidEventAdded(_)]
559        ));
560        assert_eq!(auditor.pending_deposits.len(), 1);
561
562        // Another request for the same UTXO has its own machine.
563        let again = SuiDepositRequestedEvent {
564            request_id: Address::new([0x62; 32]),
565            ..request
566        };
567        assert!(
568            auditor
569                .ingest(MonitorEvent::DepositRequested(again))
570                .is_empty()
571        );
572        assert_eq!(auditor.pending_deposits.len(), 2);
573    }

752    async fn a_restart_resumes_the_open_flows_and_the_pool() {
753        let cfg = Config::for_tests();
754        let mut auditor = auditor_with(&cfg, Arc::new(MonitorMetrics::new()));
755        let deposit = ConfirmedDeposit {
756            deposit_id: DepositId::new(txid(1), 0),
757            amount: 1_000,
758            script_pubkey: ScriptBuf::from_bytes(vec![0x51]),
759            block_time: 1_700_000_000,
760        };
761        assert!(auditor.utxo_pool.add_deposit(&deposit).is_none());
762        // An approval the guardian never signed: its E2 is overdue.
763        auditor.ingest_batch(vec![MonitorEvent::Withdrawal(hashi_approval())]);
764        auditor.tick_state_checks().await;
765        let missing = auditor.detect_timing_violations();
766        assert!(matches!(
767            missing[..],
768            [MonitorFinding::WithdrawalEventMissing {
769                event_type: WithdrawalEventType::E2GuardianApproved,

771            }]
772        ));
773
774        let snapshot = auditor.store.load().await.unwrap().unwrap();
775        assert_eq!(snapshot.pool, *auditor.utxo_pool());
776        let mut resumed = auditor_with(&cfg, Arc::new(MonitorMetrics::new()));
777        resumed.restore(snapshot.state);
778
779        assert_eq!(resumed.detect_timing_violations(), missing);
