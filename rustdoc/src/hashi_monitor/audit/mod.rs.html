4//! Auditor implementations.
5//! Goal: Attempt to match all withdrawals with a guardian event inside the input time window,
6//!       even if some corresponding events for that withdrawal occur outside the window.
7//! Core workflow:
8//!     - User inputs a window (either just start or both start & end).
9//!     - Guardian window is authoritative; Sui uses relaxed bounds for predecessor checks.
10//!     - At a desired frequency, auditors do:
11//!         - advance cursors
12//!         - call `if wsm.is_in_audit_window() { wsm.violations(&cursors) }` to identify errors.
13//!     - Currently we also report orphan E1 findings when they fall in the user window.
14//!
15//! Deposits are checked over the derived Sui polling range rather than gated by
16//! the withdrawal audit window.
17

54pub fn log_findings(source: &'static str, phase: &'static str, findings: &[MonitorFinding]) {
55    for finding in findings.iter() {
56        let category = finding.category();
57        match category {
58            FindingCategory::Safety => tracing::error!(
59                source,
60                phase,
61                %category,
62                total = findings.len(),
63                finding = %finding,
64                "monitor finding"
65            ),
66            FindingCategory::Liveness => tracing::warn!(
67                source,
68                phase,
69                %category,
70                total = findings.len(),
71                finding = %finding,
72                "monitor finding"
73            ),
74        }
75    }
76}

95impl fmt::Display for WithdrawalProgressBlockers {
96    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
97        let Some((first, rest)) = self.0.split_first() else {
98            return f.write_str("none");
99        };
100
101        write!(f, "{first}")?;
102        for blocker in rest {
103            write!(f, "; {blocker}")?;
104        }
105        Ok(())
106    }
107}

141impl AuditorCore {
142    pub async fn new(cfg: &Config, cursors: Cursors) -> anyhow::Result<Self> {
143        Ok(Self {
144            cfg: cfg.clone(),
145            pending_withdrawals: HashMap::new(),
146            pending_deposits: HashMap::new(),
147            guardian_poller: GuardianWithdrawalsPoller::new(cfg, cursors.guardian).await?,
148            sui_poller: SuiEventsPoller::new(&cfg.sui, cursors.sui)?,
149            btc_client: BtcRpcClient::new(cfg)?,
150        })
151    }
152
153    pub fn ingest(&mut self, event: MonitorEvent) -> Vec<MonitorFinding> {
154        match event {
155            MonitorEvent::Withdrawal(event) => self.ingest_withdrawal(event),
156            MonitorEvent::Deposit(event) => self.ingest_deposit(event),
157        }
158    }
159
160    fn ingest_withdrawal(&mut self, event: MonitorWithdrawalEvent) -> Vec<MonitorFinding> {
161        let wid = event.wid;
162        match self.pending_withdrawals.entry(wid) {
163            Entry::Occupied(mut entry) => entry.get_mut().add_event(event, &self.cfg),
164            Entry::Vacant(entry) => {
165                entry.insert(WithdrawalStateMachine::new(event, &self.cfg));
166                Vec::new()
167            }
168        }
169    }
170
171    fn ingest_deposit(&mut self, event: MonitorDepositEvent) -> Vec<MonitorFinding> {
172        let deposit_id = event.deposit_id;
173        match self.pending_deposits.entry(deposit_id) {
174            Entry::Occupied(entry) => {
175                if entry.get().hashi_deposit_event() != &event {
176                    return vec![MonitorFinding::InvalidEventAdded(
177                        "duplicate deposit event for same outpoint with different contents"
178                            .to_string(),
179                    )];
180                }
181            }
182            Entry::Vacant(entry) => {
183                entry.insert(DepositStateMachine::new(event, &self.cfg));
184            }
185        }
186        Vec::new()
187    }
188
189    pub fn ingest_batch(&mut self, events: Vec<MonitorEvent>) -> Vec<MonitorFinding> {
190        let mut findings = Vec::new();
191        for event in events {
192            findings.extend(self.ingest(event));
193        }
194        findings
195    }

197    /// Pings Bitcoin RPC for all relevant withdrawals & deposits.
198    /// Returns domain findings and bubbles up infra errors.
199    pub fn fetch_btc_info(
200        &mut self,
201        window: &impl AuditWindow,
202    ) -> anyhow::Result<Vec<MonitorFinding>> {
203        self.btc_client.ensure_synced()?;
204        self.btc_client.clear_confirmation_cache();
205        let withdrawal_count = self
206            .pending_withdrawals
207            .values()
208            .filter(|sm| {
209                sm.is_in_audit_window(window) && sm.expects(WithdrawalEventType::E3BtcConfirmed)
210            })
211            .count();
212        let deposit_count = self
213            .pending_deposits
214            .values()
215            .filter(|sm| sm.is_expecting_events())
216            .count();
217        let unique_txids = self
218            .pending_withdrawals
219            .values()
220            .filter(|sm| {
221                sm.is_in_audit_window(window) && sm.expects(WithdrawalEventType::E3BtcConfirmed)
222            })
223            .map(WithdrawalStateMachine::btc_txid)
224            .chain(
225                self.pending_deposits
226                    .values()
227                    .filter(|sm| sm.is_expecting_events())
228                    .map(DepositStateMachine::btc_txid),
229            )
230            .collect::<HashSet<_>>()
231            .into_iter()
232            .collect::<Vec<_>>();
233        tracing::info!(
234            withdrawal_count,
235            deposit_count,
236            unique_txids = unique_txids.len(),
237            "starting bitcoin confirmation lookups"
238        );
239        self.btc_client.prefetch_confirmations(&unique_txids)?;
240        let mut findings = Vec::new();
241        for sm in self.pending_withdrawals.values_mut() {
242            if !sm.is_in_audit_window(window) {
243                continue;
244            }
245
246            // Fetch BTC info for expecting withdrawals
247            if sm.expects(WithdrawalEventType::E3BtcConfirmed)
248                && let BtcFetchOutcome::Confirmed(new_findings) =
249                    sm.try_fetch_btc_tx(&self.cfg, &self.btc_client)?
250            {
251                findings.extend(new_findings);
252            }
253        }
254
255        for sm in self.pending_deposits.values_mut() {
256            if sm.is_expecting_events()
257                && let BtcFetchOutcome::Confirmed(new_findings) =
258                    sm.try_fetch_btc_tx(&self.btc_client)?
259            {
260                findings.extend(new_findings);
261            }
262        }
263
264        Ok(findings)
265    }

267    /// Fetches each overdue Hashi approval from Sui. One for the guardian's txid inside the
268    /// scanned range is also a `SuiScanMissedEvent`; one whose lookup fails stays missing.
269    pub async fn fetch_missing_hashi_approvals(
270        &mut self,
271        window: &impl AuditWindow,
272    ) -> Vec<MonitorFinding> {
273        let cursors = self.get_cursors();
274        let missing = self
275            .pending_withdrawals
276            .values()
277            .filter(|sm| sm.is_in_audit_window(window) && sm.is_missing_hashi_approval(&cursors))
278            .map(|sm| (sm.wid(), sm.btc_txid()))
279            .collect::<Vec<_>>();
280        let mut findings = Vec::new();
281        let mut approvals = Vec::new();
282        for (wid, btc_txid) in missing {
283            match self.sui_poller.fetch_withdrawal_approval(wid).await {
284                // A contradiction, not a scan miss: the state machine rejects even a scanned one.
285                Ok(Some(approval)) if approval.btc_txid != btc_txid => {
286                    findings.push(MonitorFinding::InvalidEventAdded(
287                        "invalid btc_txid".to_string(),
288                    ));
289                }
290                Ok(Some(approval)) => {
291                    if self.sui_poller.has_scanned(approval.timestamp_secs) {
292                        findings.push(MonitorFinding::SuiScanMissedEvent {
293                            event: MonitorEvent::Withdrawal(approval.clone()),
294                            cursor: cursors.sui,
295                        });
296                    } else {
297                        tracing::info!(
298                            %wid,
299                            approved_at = %utc_timestamp(approval.timestamp_secs),
300                            "fetched Hashi approval missing from the Sui event scan"
301                        );
302                    }
303                    approvals.push(MonitorEvent::Withdrawal(approval));
304                }
305                Ok(None) => {}
306                Err(error) => tracing::warn!(
307                    source = "sui",
308                    %wid,
309                    ?error,
310                    "Hashi approval lookup failed; reporting it missing"
311                ),
312            }
313        }
314        findings.extend(self.ingest_batch(approvals));
315        findings
316    }

318    pub fn detect_violations(&self, window: &impl AuditWindow) -> Vec<MonitorFinding> {
319        let mut findings = Vec::new();
320        for sm in self.pending_withdrawals.values() {
321            if !sm.is_in_audit_window(window) {
322                continue;
323            }
324
325            // Gather all violations so far
326            let violations = sm.violations(&self.get_cursors());
327            if !violations.is_empty() {
328                findings.extend(violations);
329            }
330        }
331        for sm in self.pending_deposits.values() {
332            let violations = sm.violations();
333            if !violations.is_empty() {
334                findings.extend(violations);
335            }
336        }
337        findings
338    }

340    pub fn garbage_collect(&mut self, window: &impl AuditWindow) {
341        let mut completed_withdrawals = Vec::new();
342        for (wid, sm) in &mut self.pending_withdrawals {
343            if !sm.is_in_audit_window(window) {
344                continue;
345            }
346            if !sm.is_expecting_events() {
347                let e1 = sm
348                    .get(WithdrawalEventType::E1HashiApproved)
349                    .expect("completed withdrawal has E1");
350                let e2 = sm
351                    .get(WithdrawalEventType::E2GuardianApproved)
352                    .expect("completed withdrawal has E2");
353                let e3 = sm
354                    .get(WithdrawalEventType::E3BtcConfirmed)
355                    .expect("completed withdrawal has E3");
356                tracing::info!(
357                    %wid,
358                    e1_at = %utc_timestamp(e1.timestamp_secs),
359                    e2_at = %utc_timestamp(e2.timestamp_secs),
360                    e3_at = %utc_timestamp(e3.timestamp_secs),
361                    e1_to_e2 = %human_timestamp_delta(e2.timestamp_secs, e1.timestamp_secs),
362                    e2_to_e3 = %human_timestamp_delta(e3.timestamp_secs, e2.timestamp_secs),
363                    "withdrawal flow is complete"
364                );
365                completed_withdrawals.push(*wid);
366            }
367        }
368        // Garbage collect
369        for wid in completed_withdrawals {
370            self.pending_withdrawals.remove(&wid);
371        }
372
373        let mut completed_deposits = Vec::new();
374        for (deposit_id, sm) in &mut self.pending_deposits {
375            if !sm.is_expecting_events() {
376                tracing::debug!(%deposit_id, "deposit flow is complete");
377                completed_deposits.push(*deposit_id);
378            }
379        }
380
381        for deposit_id in completed_deposits {
382            self.pending_deposits.remove(&deposit_id);
383        }
384    }

386    pub fn progress_watermarks(&self, window: &impl AuditWindow) -> ProgressWatermarks {
387        let mut verified_up_to_withdrawals = self.get_guardian_cursor();
388        let mut withdrawal_blockers = Vec::new();
389        for sm in self.pending_withdrawals.values() {
390            if !sm.is_in_audit_window(window) || !sm.is_expecting_events() {
391                continue;
392            }
393
394            if let Some(e2) = sm.get(WithdrawalEventType::E2GuardianApproved) {
395                if e2.timestamp_secs < verified_up_to_withdrawals {
396                    verified_up_to_withdrawals = e2.timestamp_secs;
397                    withdrawal_blockers.clear();
398                }
399                if e2.timestamp_secs == verified_up_to_withdrawals {
400                    withdrawal_blockers.push(WithdrawalProgressBlocker {
401                        wid: sm.wid(),
402                        btc_txid: sm.btc_txid(),
403                        guardian_approved_at: e2.timestamp_secs,
404                        waiting_for: sm.expected_event_types(),
405                    });
406                }
407            } else {
408                tracing::debug!(
409                    wid = %sm.wid(),
410                    "in-window withdrawal missing guardian anchor; skipping in verified_up_to computation"
411                );
412            }
413        }
414
415        let unresolved_deposit_floor = self
416            .pending_deposits
417            .values()
418            .filter(|sm| sm.is_expecting_events())
419            .map(|sm| sm.hashi_deposit_event().timestamp_secs)
420            .min()
421            .unwrap_or(u64::MAX);
422
423        let verified_up_to_deposits = self.get_sui_cursor().min(unresolved_deposit_floor);
424
425        ProgressWatermarks {
426            verified_up_to_withdrawals,
427            verified_up_to_deposits,
428            restart_start: verified_up_to_withdrawals.min(verified_up_to_deposits),
429            withdrawal_blockers: WithdrawalProgressBlockers(withdrawal_blockers),
430        }
431    }

503    /// An auditor holding a guardian approval for `btc_txid` of `looked_up_approval`'s
504    /// withdrawal, whose Sui scan covered `[start, cursor)` without its E1.
505    async fn auditor_missing_approval(
506        start: UnixSeconds,
507        cursor: UnixSeconds,
508        btc_txid: Txid,
509    ) -> AuditorCore {
510        let cfg: Config = serde_yaml::from_str(CONFIG).unwrap();
511        let mut auditor = AuditorCore {
512            pending_withdrawals: HashMap::new(),
513            pending_deposits: HashMap::new(),
514            guardian_poller: GuardianWithdrawalsPoller::for_tests(&cfg, 0),
515            sui_poller: poller_scanned(start, cursor).await,
516            btc_client: BtcRpcClient::new(&cfg).unwrap(),
517            cfg,
518        };
519        let approval = looked_up_approval();
520        let findings = auditor.ingest(MonitorEvent::Withdrawal(MonitorWithdrawalEvent {
521            event_type: WithdrawalEventType::E2GuardianApproved,
522            timestamp_secs: approval.timestamp_secs + 7,
523            btc_txid,
524            ..approval
525        }));
526        assert!(findings.is_empty());
527        auditor
