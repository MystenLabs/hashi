4//! The auditor monitors bridge events and reports findings.
5//!
6//! Why it exists:
7//! - The monitor checks deposits, withdrawals, and unspent transaction outputs (UTXOs).
8//! - Metrics provide findings and source progress for alerts.
9//!
10//! How it works:
11//! 1. Restore the source positions, flow machines, and UTXO pool from a snapshot.
12//! 2. Without a snapshot, start at the bridge genesis with an empty pool.
13//!    Fail at startup on the snapshot of another deployment.
14//! 3. Fail at startup if the Sui node has pruned the start checkpoint.
15//! 4. Read Sui events and guardian logs into the flow machines.
16//!    At startup, catch up Sui, then the guardian, then Bitcoin. Retry a failed source
17//!    until it is current. Then remove the closed machines and save the snapshot.
18//!    Only then start the loop, so the loop does not start with a large unanswered backlog.
19//! 5. Check Bitcoin confirmations from each withdrawal's Sui approval on.
20//!    Check that each fresh deposit approval names an unspent outpoint.
21//! 6. Every `MAINTENANCE_INTERVAL`, run one maintenance tick in this order:
22//!    1. Remove the closed machines and add their flows to the pool. Log the open flows.
23//!       This step is first, so each later step sees only open machines and a current pool.
24//!    2. Check Bitcoin when the daily pool check is due.
25//!    3. Check that the UTXO pool backs the hBTC supply.
26//!    4. Report timing violations and stalled deposit approvals, and judge suspect spends. These have no order rule.
27//!    5. Report a source, the daily pool check, or the hBTC supply read that is behind its limit.
28//!       This step comes after steps 2 and 3, so a check that succeeds in this tick is not stale.
29//!    6. Save the snapshot once `SNAPSHOT_SAVE_INTERVAL` seconds have passed since the last save.
30//!       Then report a save that is behind its limit. This step is last.
31//! 7. Check Bitcoin at the first maintenance tick. Repeat successful pool checks after `UTXO_CHECK_INTERVAL` seconds.
32//! 8. Judge suspect spends after the Sui cursor passes the Bitcoin lookup time.
33//! 9. In the loop, log a failed source tick. The next tick of that source retries it.
34//!
35//! Assumptions:
36//! - Every tick runs on one task. The first catch-up blocks the maintenance tick,
37//!   so a first run reports no stale source while it catches up.
38//! - A source that stays down at startup blocks the loop. The Sui `checked_through`
39//!   gauge then stops too, and its lag alert pages. Only the Sui gauge has a lag alert.
40

83/// How often Sui is polled. A deposit approval must reach the monitor
84/// early in Move's delay before the mint, ten minutes by default.
85const SUI_POLL_INTERVAL: Duration = Duration::from_secs(60);
86
87/// How often Bitcoin is asked about the pending transactions. An approval
88/// is reported unbacked at the second empty lookup after it, so with
89/// `SUI_POLL_INTERVAL` the alert comes within five minutes of it.
90const BTC_POLL_INTERVAL: Duration = Duration::from_secs(2 * 60);
91
92/// How often the guardian's log is polled. Each poll lists the current
93/// directory and reads only its new logs. The withdrawal bounds are hours.
94const GUARDIAN_POLL_INTERVAL: Duration = Duration::from_secs(5 * 60);
95
96/// How often the maintenance tick runs.
97const MAINTENANCE_INTERVAL: Duration = Duration::from_secs(5 * 60);
98
99/// How often the snapshot is saved, in seconds. A restart replays the
100/// history since the last save and reports its findings again.
101const SNAPSHOT_SAVE_INTERVAL: u64 = 60 * 60;
102
103/// How long snapshot saves may fail before a `SnapshotStale` finding, in
104/// seconds. A failed save costs only a longer replay at the next restart.
105const SNAPSHOT_STALE_LIMIT: u64 = 12 * 60 * 60;

107/// Log each finding: an ERROR line if it alerts in `mode`, else a WARN line.
108pub fn log_findings(
109    source: &'static str,
110    phase: &'static str,
111    findings: &[MonitorFinding],
112    mode: AlertMode,
113) {
114    for finding in findings.iter() {
115        if finding.alerts_in(mode) {
116            tracing::error!(
117                source,
118                phase,
119                total = findings.len(),
120                finding = %finding,
121                "monitor finding"
122            );
123        } else {
124            tracing::warn!(
125                source,
126                phase,
127                total = findings.len(),
128                finding = %finding,
129                "monitor finding"
130            );
131        }
132    }
133}

135pub struct Auditor {
136    cfg: Config,
137    pending_withdrawals: HashMap<WithdrawalID, WithdrawalStateMachine>,
138    /// Deposit requests Bitcoin is still asked about. Keyed by the Sui
139    /// request id, so a repeated event is ignored and each request has its
140    /// own.
141    pending_deposits: HashMap<Address, DepositStateMachine>,
142    /// The UTXOs the bridge holds, by Bitcoin's account. See `crate::domain::utxo_pool`. The
143    /// snapshot keeps it between runs.
144    utxo_pool: UtxoPool,
145    /// Spends waiting for the Sui cursor, by outpoint.
146    suspect_spends: HashMap<OutPoint, SuspectSpend>,
147    /// Deposit approvals whose mint delay has not ended, by outpoint. See
148    /// `fresh_approvals`.
149    fresh_approvals: HashMap<OutPoint, SuiDepositApprovedEvent>,
150    rpc_pollers: RpcPollers,
151    metrics: Arc<MonitorMetrics>,
152    /// Where the snapshot is kept between runs.
153    store: SnapshotStore,
154    /// When the last successful pool check finished. `None` until one
155    /// succeeds in this run.
156    last_successful_pool_check: Option<UnixSeconds>,
157    /// When the audit loop started, after the first catch-up. Staleness of a
158    /// check that has not succeeded yet counts from here.
159    started_at: UnixSeconds,
160    /// When the last snapshot save succeeded.
161    last_snapshot_save: UnixSeconds,
162    /// Since when the hBTC supply has exceeded the pool value, if it does.
163    hbtc_supply_above_pool_since: Option<UnixSeconds>,
164    /// When the hBTC supply was last read. Startup reads it.
165    last_hbtc_supply_read: UnixSeconds,
166    /// Move's time of the latest Sui `DepositApproved`.
167    last_deposit_approval: Option<UnixSeconds>,
168}

170impl Auditor {
171    /// Restore the snapshot, or start at the bridge genesis.
172    /// Fail if the store cannot be read or written, or Sui has removed the required history.
173    pub async fn new(
174        cfg: &Config,
175        metrics: Arc<MonitorMetrics>,
176        store: SnapshotStore,
177    ) -> anyhow::Result<Self> {
178        let (state, pool) = match this_deployments(cfg, store.load().await?)? {
179            Some(snapshot) => {
180                tracing::info!(
181                    sui_next_checkpoint = snapshot.state.sui_next_checkpoint,
182                    guardian_cursor = %utc_timestamp(snapshot.state.guardian_cursor.read_until()),
183                    withdrawals = snapshot.state.withdrawals.len(),
184                    deposits = snapshot.state.deposits.len(),
185                    suspect_spends = snapshot.state.suspect_spends.len(),
186                    utxos_held = snapshot.pool.len(),
187                    "loaded the snapshot; resuming where it stopped"
188                );
189                (Some(snapshot.state), snapshot.pool)
190            }
191            None => (None, UtxoPool::default()),
192        };
193        let (guardian_start, sui_start) = match &state {
194            Some(state) => (state.guardian_cursor.clone(), state.sui_next_checkpoint),
195            None => {
196                let genesis = package_genesis(&cfg.sui).await?;
197                tracing::info!(
198                    genesis = %utc_timestamp(genesis.timestamp),
199                    genesis_checkpoint = genesis.checkpoint,
200                    "first run: starting at the bridge's genesis with an empty UTXO pool"
201                );
202                (
203                    GuardianCursor::new(S3HourDirectory::withdraw(genesis.timestamp)?),
204                    genesis.checkpoint,
205                )
206            }
207        };
208        let rpc_pollers = RpcPollers::new(cfg, guardian_start, sui_start).await?;
209        let mut auditor = Self {
210            cfg: cfg.clone(),
211            pending_withdrawals: HashMap::new(),
212            pending_deposits: HashMap::new(),
213            utxo_pool: pool,
214            suspect_spends: HashMap::new(),
215            fresh_approvals: HashMap::new(),
216            rpc_pollers,
217            metrics,
218            store,
219            last_successful_pool_check: None,
220            started_at: now_timestamp_secs(),
221            last_snapshot_save: 0,
222            hbtc_supply_above_pool_since: None,
223            last_hbtc_supply_read: 0,
224            last_deposit_approval: None,
225        };
226        if let Some(state) = state {
227            auditor.restore(state);
228        }
229        // A pruned start leaves the pool incomplete.
230        auditor.rpc_pollers.ensure_sui_serves_start().await?;
231        // A setup that cannot read the supply fails here, not silently later.
232        let supply = auditor
233            .read_hbtc_supply()
234            .await
235            .context("failed to read the hBTC supply")?;
236        tracing::info!(hbtc_supply = supply, "read the hBTC supply");
237        auditor
238            .metrics
239            .set_checked_through(SOURCE_SUI, auditor.rpc_pollers.sui_read_until());
240        auditor
241            .metrics
242            .set_checked_through(SOURCE_GUARDIAN, auditor.rpc_pollers.guardian_read_until());
243        auditor
244            .metrics
245            .set_checked_through(SOURCE_BTC, auditor.rpc_pollers.btc_read_until());
246        // Check write access at startup. A missing bucket loads as empty.
247        auditor
248            .store
249            .save(&auditor.cfg, auditor.state(), auditor.utxo_pool())
250            .await?;
251        let now = now_timestamp_secs();
252        auditor.last_snapshot_save = now;
253        auditor.metrics.set_checked_through(SOURCE_SNAPSHOT, now);
254        Ok(auditor)
255    }

257    /// The pollers' positions, the open machines and the suspect spends, for
258    /// the snapshot.
259    fn state(&self) -> AuditState {
260        AuditState {
261            sui_next_checkpoint: self.rpc_pollers.sui_poller.next_item_to_poll(),
262            guardian_cursor: self.rpc_pollers.guardian_poller.next_item_to_poll(),
263            withdrawals: self.pending_withdrawals.values().cloned().collect(),
264            deposits: self.pending_deposits.values().cloned().collect(),
265            suspect_spends: self.suspect_spends.clone().into_iter().collect(),
266            hbtc_supply_above_pool_since: self.hbtc_supply_above_pool_since,
267            last_deposit_approval: self.last_deposit_approval,
268        }
269    }
270
271    /// Hold the machines and the suspect spends a snapshot saved.
272    fn restore(&mut self, state: AuditState) {
273        for sm in state.withdrawals {
274            self.pending_withdrawals.insert(sm.wid(), sm);
275        }
276        for sm in state.deposits {
277            self.pending_deposits.insert(sm.request_id(), sm);
278        }
279        self.suspect_spends.extend(state.suspect_spends);
280        self.hbtc_supply_above_pool_since = state.hbtc_supply_above_pool_since;
281        self.last_deposit_approval = state.last_deposit_approval;
282    }

284    fn ingest(&mut self, event: MonitorEvent) -> Vec<MonitorFinding> {
285        match event {
286            MonitorEvent::Withdrawal(event) => self.ingest_withdrawal(event),
287            MonitorEvent::DepositApproved(event) => self.ingest_deposit_approved(event),
288            MonitorEvent::DepositRequested(event) => self.ingest_deposit_request(event),
289        }
290    }
291
292    fn ingest_deposit_request(&mut self, event: SuiDepositRequestedEvent) -> Vec<MonitorFinding> {
293        match self.pending_deposits.entry(event.request_id) {
294            Entry::Occupied(entry) => entry.get().add_sui_request(&event).into_iter().collect(),
295            Entry::Vacant(entry) => {
296                entry.insert(DepositStateMachine::new(event));
297                Vec::new()
298            }
299        }
300    }
301
302    fn ingest_withdrawal(&mut self, event: MonitorWithdrawalEvent) -> Vec<MonitorFinding> {
303        let wid = event.wid;
304        match self.pending_withdrawals.entry(wid) {
305            Entry::Occupied(mut entry) => entry.get_mut().add_event(event, &self.cfg.timing),
306            Entry::Vacant(entry) => {
307                entry.insert(WithdrawalStateMachine::new(event, &self.cfg.timing));
308                Vec::new()
309            }
310        }
311    }
312
313    /// The Sui `DepositApproved`. Move approves only a request it holds, so
314    /// the request came first. While its machine is open, the machine keeps
315    /// the approval. Once it closed, the pool has its answer.
316    fn ingest_deposit_approved(&mut self, event: SuiDepositApprovedEvent) -> Vec<MonitorFinding> {
317        self.note_fresh_approval(&event, now_timestamp_secs());
318        self.last_deposit_approval = self.last_deposit_approval.max(Some(event.timestamp_secs));
319        match self.pending_deposits.get_mut(&event.request_id) {
320            Some(sm) => sm.add_sui_approval(event, &self.cfg.timing),
321            None => self
322                .utxo_pool
323                .check_sui_approval(&event, self.cfg.timing.deposit_approval_skew),
324        }
325    }

327    pub fn ingest_batch(&mut self, events: Vec<MonitorEvent>) -> Vec<MonitorFinding> {
328        let mut findings = Vec::new();
329        for event in events {
330            findings.extend(self.ingest(event));
331        }
332        self.report_findings("ingest", &findings);
333        findings
334    }

336    fn detect_timing_violations(&self) -> Vec<MonitorFinding> {
337        let mut findings = Vec::new();
338        for sm in self.pending_withdrawals.values() {
339            findings.extend(sm.timing_violations(&self.rpc_pollers.cursors()));
340        }
341        findings

344    /// The open withdrawals whose first event is the earliest, for the
345    /// GC log. An operator sees what the audit still waits for.
346    fn oldest_open_withdrawals(&self) -> String {
347        let earliest = self
348            .pending_withdrawals
349            .values()
350            .map(WithdrawalStateMachine::earliest_event_at)
351            .min();
352        let oldest = self
353            .pending_withdrawals
354            .values()
355            .filter(|sm| Some(sm.earliest_event_at()) == earliest)
356            .map(|sm| {
357                format!(
358                    "wid={}, btc_txid={}, earliest_event_at={}, waiting_for={:?}",
359                    sm.wid(),
360                    sm.btc_txid(),
361                    utc_timestamp(sm.earliest_event_at()),
362                    sm.expected_event_types(),
363                )
364            })
365            .collect::<Vec<_>>();
366        if oldest.is_empty() {
367            "none".to_string()
368        } else {
369            oldest.join("; ")
370        }
371    }

373    /// Return the deposits with the earliest request timestamp for the GC log.
374    fn oldest_open_deposits(&self) -> String {
375        let earliest = self
376            .pending_deposits
377            .values()
378            .map(DepositStateMachine::requested_at)
379            .min();
380        let oldest = self
381            .pending_deposits
382            .values()
383            .filter(|sm| Some(sm.requested_at()) == earliest)
384            .map(|sm| {
385                format!(
386                    "request_id={}, deposit_id={}, requested_at={}",
387                    sm.request_id(),
388                    sm.deposit_id(),
389                    utc_timestamp(sm.requested_at()),
390                )
391            })
392            .collect::<Vec<_>>();
393        if oldest.is_empty() {
394            "none".to_string()
395        } else {
396            oldest.join("; ")
397        }

405    pub fn utxo_pool(&self) -> &UtxoPool {
406        &self.utxo_pool

409    /// Save the snapshot. A failed save is logged and does not stop the
410    /// audit; the next maintenance tick saves again. Alerting watches the
411    /// time of the last good save.
412    async fn save_snapshot(&mut self) {
413        let saved = self
414            .store
415            .save(&self.cfg, self.state(), self.utxo_pool())
416            .await;
417        match saved {
418            Ok(()) => {
419                self.last_snapshot_save = now_timestamp_secs();
420                self.metrics
421                    .set_checked_through(SOURCE_SNAPSHOT, self.last_snapshot_save);
422            }
423            Err(error) => tracing::error!(?error, "failed to save the snapshot; continuing"),
424        }
425    }
426
427    /// A `SnapshotStale` finding if no save has succeeded within the limit.
428    fn stale_snapshot(&self, now: UnixSeconds) -> Option<MonitorFinding> {
429        (now.saturating_sub(self.last_snapshot_save) > SNAPSHOT_STALE_LIMIT).then_some(
430            MonitorFinding::SnapshotStale {
431                last_saved: self.last_snapshot_save,
432                limit_secs: SNAPSHOT_STALE_LIMIT,
433            },
434        )
435    }

437    /// Log the findings, and record the ones that alert in the metrics.
438    fn report_findings(&self, phase: &'static str, findings: &[MonitorFinding]) {
439        let mode = self.cfg.alert_mode;
440        log_findings("continuous", phase, findings, mode);
441        let alerts = findings.iter().filter(|finding| finding.alerts_in(mode));
442        self.metrics.record_findings(alerts, now_timestamp_secs());
443    }
444
445    /// Report the open withdrawals that missed a timing bound.
446    fn check_timing(&self) {
447        let violations = self.detect_timing_violations();
448        self.report_findings("timing", &violations);
449    }
450
451    /// Report a stall of deposit approvals.
452    fn check_deposit_approvals(&self) {
453        let stall = self.utxo_pool.approval_stall(
454            self.last_deposit_approval,
455            self.rpc_pollers.sui_read_until(),
456            &self.cfg.timing,
457        );
458        self.report_findings("deposits", stall.as_slice());
459    }
460
461    /// Report a source, the daily pool check, or the hBTC supply read that is behind its limit.
462    fn check_staleness(&self) {
463        let now = now_timestamp_secs();
464        let mut stale = self.rpc_pollers.stale_sources(now);
465        stale.extend(self.stale_pool_check(now));
466        stale.extend(self.stale_hbtc_supply(now));
467        self.report_findings("progress", &stale);
468    }
469
470    /// Save once `SNAPSHOT_SAVE_INTERVAL` has passed, then report a stale save.
471    async fn save_snapshot_if_due(&mut self) {
472        if now_timestamp_secs().saturating_sub(self.last_snapshot_save) >= SNAPSHOT_SAVE_INTERVAL {
473            self.save_snapshot().await;
474        }
475        // After the save, so a save that succeeds now is not reported.
476        let stale = self.stale_snapshot(now_timestamp_secs());
477        self.report_findings("progress", stale.as_slice());
478    }
479
480    /// Run the maintenance steps in order.
481    async fn tick_maintenance(&mut self) {
482        // First, so every later step sees only open machines and a current pool.
483        self.gc_and_update_pool();
484        self.check_utxo_pool_if_due();
485        self.check_hbtc_supply().await;
486        // These have no order rule.
487        self.check_timing();
488        self.check_deposit_approvals();
489        self.check_suspect_spends();
490        // After the checks, so a check that succeeds in this tick is not stale.
491        self.check_staleness();
492        // Last, so the snapshot holds the result of every step.
493        self.save_snapshot_if_due().await;
494    }
495
496    pub async fn run(&mut self) -> anyhow::Result<()> {
497        tracing::info!(
498            "starting continuous audit:\n  sui_start={}\n  guardian_start={}",
499            utc_timestamp(self.rpc_pollers.sui_read_until()),
500            utc_timestamp(self.rpc_pollers.guardian_read_until()),
501        );
502        self.persistent_catch_up().await;
503        // After the catch-up, so the snapshot holds only the open machines.
504        self.gc_and_update_pool();
505        self.save_snapshot().await;
506        // A first run can catch up for hours. That time is not a stale check.
507        self.started_at = now_timestamp_secs();
508        tracing::info!(
509            sui_cursor = %utc_timestamp(self.rpc_pollers.sui_read_until()),
510            guardian_cursor = %utc_timestamp(self.rpc_pollers.guardian_read_until()),
511            "finished the initial catch-up; starting the audit loop"
512        );
513
514        let mut sui_ticker = tokio::time::interval(SUI_POLL_INTERVAL);
515        let mut guardian_ticker = tokio::time::interval(GUARDIAN_POLL_INTERVAL);
516        let mut btc_ticker = tokio::time::interval(BTC_POLL_INTERVAL);
517        let mut maintenance_ticker = tokio::time::interval(MAINTENANCE_INTERVAL);
518
519        sui_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
520        guardian_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
521        btc_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
522        maintenance_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
523
524        // Consume each immediate tick so scheduled work starts after its interval.
525        sui_ticker.tick().await;
526        guardian_ticker.tick().await;
527        btc_ticker.tick().await;
528        maintenance_ticker.tick().await;
529
530        loop {
531            tokio::select! {
532                _ = sui_ticker.tick() => {
533                    if let Err(error) = self.tick_sui().await {
534                        tracing::warn!(source = "sui", ?error, "poll failed; continuing");
535                    }
536                }
537                _ = guardian_ticker.tick() => {
538                    if let Err(error) = self.tick_guardian().await {
539                        tracing::warn!(source = "guardian", ?error, "poll failed; continuing");
540                    }
541                }
542                _ = btc_ticker.tick() => {
543                    if let Err(error) = self.tick_btc() {
544                        tracing::warn!(source = "btc", ?error, "btc tick failed; continuing");
545                    }
546                }
547                _ = maintenance_ticker.tick() => {
548                    self.tick_maintenance().await;
549                }
550            }
551        }
552    }
553}

555/// The snapshot, if this deployment saved it, or `None` for a first run.
556/// Fail on the snapshot of another deployment, so a config mistake does not
557/// overwrite it.
558fn this_deployments(cfg: &Config, snapshot: Option<Snapshot>) -> anyhow::Result<Option<Snapshot>> {
559    match snapshot {
560        Some(snapshot) if snapshot.is_for(cfg) => Ok(Some(snapshot)),
561        Some(snapshot) => anyhow::bail!(
562            "the snapshot is for another deployment (guardian bucket {}, package {}); \
563             check the config, or delete snapshot.json to start again from genesis",
564            snapshot.guardian_bucket,
565            snapshot.package_id,
566        ),
567        None => {
568            tracing::info!("no snapshot");
569            Ok(None)
570        }
571    }
572}

598    #[tokio::test]
599    async fn a_repeated_deposit_request_is_ingested_once() {
600        let approval = hashi_approval();
601        let mut auditor = auditor_with_pool().await;
602        let request = SuiDepositRequestedEvent {
603            timestamp_secs: approval.timestamp_secs,
604            deposit_id: DepositId::new(approval.btc_txid, 0),
605            ..deposit_request()
606        };
607
608        assert!(
609            auditor
610                .ingest(MonitorEvent::DepositRequested(request.clone()))
611                .is_empty()
612        );
613        assert!(
614            auditor
615                .ingest(MonitorEvent::DepositRequested(request.clone()))
616                .is_empty()
617        );
618        assert_eq!(auditor.pending_deposits.len(), 1);
619
620        // The same request id with other contents is a contradiction.
621        let contradiction = SuiDepositRequestedEvent {
622            amount: request.amount + 1,
623            ..request.clone()
624        };
625        assert!(matches!(
626            &auditor.ingest(MonitorEvent::DepositRequested(contradiction))[..],
627            [MonitorFinding::DepositMismatch {
628                reason: DepositMismatchReason::RequestDiffers { .. },
629                ..
630            }]
631        ));
632        assert_eq!(auditor.pending_deposits.len(), 1);
633
634        // Another request for the same UTXO has its own machine.
635        let again = SuiDepositRequestedEvent {
636            request_id: Address::new([0x62; 32]),
637            ..request
638        };
639        assert!(
640            auditor
641                .ingest(MonitorEvent::DepositRequested(again))
642                .is_empty()
643        );
644        assert_eq!(auditor.pending_deposits.len(), 2);
645    }

683    #[tokio::test]
684    async fn a_sui_approval_after_its_request_closed_is_checked_against_the_pool() {
685        let mut auditor = auditor_with_pool().await;
686        let (_, sui) = deposit_request_and_approval(50_000);
687        let unbacked = auditor.ingest(MonitorEvent::DepositApproved(sui.clone()));
688        assert!(matches!(
689            &unbacked[..],
690            [MonitorFinding::DepositUnbacked { sui: unbacked }] if *unbacked == sui
691        ));
692
693        assert!(
694            auditor
695                .utxo_pool
696                .add_deposit(&unapproved(BtcConfirmedDeposit {
697                    deposit_id: sui.deposit_id,
698                    amount: 50_000,
699                    script_pubkey: sui.script_pubkey.clone(),
700                    block_time: 150,
701                }))
702                .is_empty()
703        );
704        assert_eq!(auditor.utxo_pool.value(), 0);
705        let outpoint = OutPoint::from(sui.deposit_id);
706        assert!(
707            auditor
708                .ingest(MonitorEvent::DepositApproved(sui))
709                .is_empty()
710        );
711        // The approval marks the held deposit approved, so it counts.
712        assert_eq!(
713            auditor.utxo_pool.get(&outpoint).unwrap().origin,
714            UtxoOrigin::Deposit
715        );
716        assert_eq!(auditor.utxo_pool.value(), 50_000);
717    }

719    #[tokio::test]
720    async fn an_open_request_keeps_its_sui_approval_until_bitcoin_answers() {
721        let mut auditor = auditor_with_pool().await;
722        // The request claims more than Bitcoin pays, and Hashi approves it.
723        let (request, sui) = deposit_request_and_approval(60_000);
724        let request_id = request.request_id;
725        assert!(
726            auditor
727                .ingest_batch(vec![
728                    MonitorEvent::DepositRequested(request),
729                    MonitorEvent::DepositApproved(sui.clone()),
730                ])
731                .is_empty()
732        );
733        assert!(auditor.gc_and_update_pool().is_empty());
734
735        // Bitcoin answers: the machine judges the held approval.
736        let findings = auditor
737            .pending_deposits
738            .get_mut(&request_id)
739            .unwrap()
740            .record_lookup(
741                &Timing::for_tests(),
742                0,
743                Some(&crate::test_utils::confirmation(50_000, p2wpkh(0x54))),
744            );
745        assert!(matches!(&findings[..], [MonitorFinding::DepositMismatch {
746                reason: DepositMismatchReason::PaymentDiffers { sui: claimed, btc },
747                ..
748            }] if *claimed == sui && btc.amount == 50_000));
749
750        assert!(auditor.gc_and_update_pool().is_empty());
751        assert!(auditor.pending_deposits.is_empty());
752        // Move mints on the approval, so the deposit counts.
753        assert_eq!(
754            auditor
755                .utxo_pool
756                .get(&OutPoint::from(sui.deposit_id))
757                .unwrap()
758                .origin,
759            UtxoOrigin::Deposit
760        );
761        assert_eq!(auditor.utxo_pool.value(), 50_000);
762    }

886    /// A guardian clock skew and a stale snapshot, reported in `mode`.
887    /// Returns the kinds that set the finding metric.
888    fn alerted_kinds(mode: AlertMode) -> Vec<&'static str> {
889        let mut cfg = Config::for_tests();
890        cfg.alert_mode = mode;
891        let metrics = Arc::new(MonitorMetrics::new());
892        let auditor = auditor_with(&cfg, metrics.clone());
893        let skew = MonitorFinding::GuardianClockSkew {
894            session_id: SessionID::from("session"),
895            object_key: "log".to_string(),
896            logged_at_ms: 0,
897            stored_at_ms: 60_000,
898        };
899        let stale = MonitorFinding::SnapshotStale {
900            last_saved: 0,
901            limit_secs: 3_600,
902        };
903        auditor.report_findings("test", &[skew, stale]);
904
905        let rendered = metrics.render();
906        ["GuardianClockSkew", "SnapshotStale"]
907            .into_iter()
908            .filter(|kind| rendered.contains(&format!("kind=\"{kind}\"")))
909            .collect()
910    }

939    #[tokio::test]
940    async fn a_restart_resumes_the_open_flows_and_the_pool() {
941        let cfg = Config::for_tests();
942        let mut auditor = auditor_with(&cfg, Arc::new(MonitorMetrics::new()));
943        let deposit = BtcConfirmedDeposit {
944            deposit_id: DepositId::new(txid(1), 0),
945            amount: 1_000,
946            script_pubkey: ScriptBuf::from_bytes(vec![0x51]),
947            block_time: 1_700_000_000,
948        };
949        assert!(
950            auditor
951                .utxo_pool
952                .add_deposit(&unapproved(deposit))
953                .is_empty()
954        );
955        // An approval the guardian never signed: its E2 is overdue.
956        auditor.ingest_batch(vec![MonitorEvent::Withdrawal(hashi_approval())]);
957        auditor.save_snapshot().await;
958        let missing = auditor.detect_timing_violations();
959        assert!(matches!(
960            missing[..],
961            [MonitorFinding::WithdrawalStuck {
962                event_type: WithdrawalEventType::E2GuardianApproved,
963                timing: EventTiming::Missing { .. },
964                ..
965            }]
966        ));
967
968        let snapshot = auditor.store.load().await.unwrap().unwrap();
969        assert_eq!(snapshot.pool, *auditor.utxo_pool());
970        let mut resumed = auditor_with(&cfg, Arc::new(MonitorMetrics::new()));
971        resumed.restore(snapshot.state);
972
973        assert_eq!(resumed.detect_timing_violations(), missing);
974    }
