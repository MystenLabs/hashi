4//! The auditor monitors bridge events and reports findings.
5//!
6//! Why it exists:
7//! - The monitor checks deposits, withdrawals, and unspent transaction outputs (UTXOs).
8//! - Metrics provide findings and source progress for alerts.
9//!
10//! How it works:
11//! 1. Restore the source positions, flow machines, and UTXO pool from a snapshot.
12//! 2. Without a snapshot, start at the bridge genesis with an empty pool.
13//!    Fail at startup on the snapshot of another deployment.
14//! 3. Fail at startup if the Sui node has pruned the start checkpoint.
15//! 4. Read Sui events and guardian logs into the flow machines.
16//!    At startup, catch up Sui, then the guardian, then Bitcoin. Retry a failed source
17//!    until it is current. Then remove the closed machines and save the snapshot.
18//!    Only then start the loop, so the loop does not start with a large unanswered backlog.
19//! 5. Check Bitcoin confirmations after both withdrawal approvals arrive.
20//!    Check that each fresh deposit approval names an unspent outpoint.
21//! 6. Every `MAINTENANCE_INTERVAL`, run one maintenance tick in this order:
22//!    1. Remove the closed machines and add their flows to the pool. Log the open flows.
23//!       This step is first, so each later step sees only open machines and a current pool.
24//!    2. Check Bitcoin when the daily pool check is due.
25//!    3. Check that the UTXO pool backs the hBTC supply.
26//!    4. Report timing violations and judge suspect spends. These two have no order rule.
27//!    5. Report a source, the daily pool check, or the hBTC supply read that is behind its limit.
28//!       This step comes after steps 2 and 3, so a check that succeeds in this tick is not stale.
29//!    6. Save the snapshot once `SNAPSHOT_SAVE_INTERVAL` seconds have passed since the last save.
30//!       Then report a save that is behind its limit. This step is last.
31//! 7. Check Bitcoin at the first maintenance tick. Repeat successful pool checks after `UTXO_CHECK_INTERVAL` seconds.
32//! 8. Judge suspect spends after the Sui cursor passes the Bitcoin lookup time.
33//! 9. In the loop, log a failed source tick. The next tick of that source retries it.
34//!
35//! Assumptions:
36//! - Every tick runs on one task. The first catch-up blocks the maintenance tick,
37//!   so a first run reports no stale source while it catches up.
38//! - A source that stays down at startup blocks the loop. The Sui `checked_through`
39//!   gauge then stops too, and its lag alert pages. Only the Sui gauge has a lag alert.

82/// How often Sui is polled. A deposit approval must reach the monitor
83/// early in Move's delay before the mint, ten minutes by default.
84const SUI_POLL_INTERVAL: Duration = Duration::from_secs(60);
85
86/// How often Bitcoin is asked about the pending transactions. An approval
87/// is reported unbacked at the second empty lookup after it, so with
88/// `SUI_POLL_INTERVAL` the alert comes within five minutes of it.
89const BTC_POLL_INTERVAL: Duration = Duration::from_secs(2 * 60);
90
91/// How often the guardian's log is polled. Each poll lists the current
92/// directory and reads only its new logs. The withdrawal bounds are hours.
93const GUARDIAN_POLL_INTERVAL: Duration = Duration::from_secs(5 * 60);
94
95/// How often the maintenance tick runs.
96const MAINTENANCE_INTERVAL: Duration = Duration::from_secs(5 * 60);
97
98/// How often the snapshot is saved, in seconds. A restart replays the
99/// history since the last save and reports its findings again.
100const SNAPSHOT_SAVE_INTERVAL: u64 = 60 * 60;
101
102/// How long snapshot saves may fail before a `SnapshotStale` finding, in
103/// seconds. A failed save costs only a longer replay at the next restart.
104const SNAPSHOT_STALE_LIMIT: u64 = 12 * 60 * 60;
105
106pub fn log_findings(source: &'static str, phase: &'static str, findings: &[MonitorFinding]) {
107    for finding in findings.iter() {
108        tracing::error!(
109            source,
110            phase,
111            total = findings.len(),
112            finding = %finding,
113            "monitor finding"
114        );
115    }
116}

118pub struct Auditor {
119    cfg: Config,
120    pending_withdrawals: HashMap<WithdrawalID, WithdrawalStateMachine>,
121    /// Deposit requests Bitcoin is still asked about. Keyed by the Sui
122    /// request id, so a repeated event is ignored and each request has its
123    /// own.
124    pending_deposits: HashMap<Address, DepositStateMachine>,
125    /// The UTXOs the bridge holds, by Bitcoin's account. See `crate::domain::utxo_pool`. The
126    /// snapshot keeps it between runs.
127    utxo_pool: UtxoPool,
128    /// Spends waiting for the Sui cursor, by outpoint.
129    suspect_spends: HashMap<OutPoint, SuspectSpend>,
130    /// Deposit approvals whose mint delay has not ended, by outpoint. See
131    /// `fresh_approvals`.
132    fresh_approvals: HashMap<OutPoint, SuiDepositApprovedEvent>,
133    rpc_pollers: RpcPollers,
134    metrics: Arc<MonitorMetrics>,
135    /// Where the snapshot is kept between runs.
136    store: SnapshotStore,
137    /// When the last successful pool check finished. `None` until one
138    /// succeeds in this run.
139    last_successful_pool_check: Option<UnixSeconds>,
140    /// When the audit loop started, after the first catch-up. Staleness of a
141    /// check that has not succeeded yet counts from here.
142    started_at: UnixSeconds,
143    /// When the last snapshot save succeeded.
144    last_snapshot_save: UnixSeconds,
145    /// Since when the hBTC supply has exceeded the pool value, if it does.
146    hbtc_supply_above_pool_since: Option<UnixSeconds>,
147    /// When the hBTC supply was last read. Startup reads it.
148    last_hbtc_supply_read: UnixSeconds,
149}

151impl Auditor {
152    /// Restore the snapshot, or start at the bridge genesis.
153    /// Fail if the store cannot be read or written, or Sui has removed the required history.
154    pub async fn new(
155        cfg: &Config,
156        metrics: Arc<MonitorMetrics>,
157        store: SnapshotStore,
158    ) -> anyhow::Result<Self> {
159        let (state, pool) = match this_deployments(cfg, store.load().await?)? {
160            Some(snapshot) => {
161                tracing::info!(
162                    sui_next_checkpoint = snapshot.state.sui_next_checkpoint,
163                    guardian_cursor = %utc_timestamp(snapshot.state.guardian_cursor.read_until()),
164                    withdrawals = snapshot.state.withdrawals.len(),
165                    deposits = snapshot.state.deposits.len(),
166                    suspect_spends = snapshot.state.suspect_spends.len(),
167                    utxos_held = snapshot.pool.len(),
168                    "loaded the snapshot; resuming where it stopped"
169                );
170                (Some(snapshot.state), snapshot.pool)
171            }
172            None => (None, UtxoPool::default()),
173        };
174        let (guardian_start, sui_start) = match &state {
175            Some(state) => (state.guardian_cursor.clone(), state.sui_next_checkpoint),
176            None => {
177                let genesis = package_genesis(&cfg.sui).await?;
178                tracing::info!(
179                    genesis = %utc_timestamp(genesis.timestamp),
180                    genesis_checkpoint = genesis.checkpoint,
181                    "first run: starting at the bridge's genesis with an empty UTXO pool"
182                );
183                (
184                    GuardianCursor::new(S3HourDirectory::withdraw(genesis.timestamp)?),
185                    genesis.checkpoint,
186                )
187            }
188        };
189        let rpc_pollers = RpcPollers::new(cfg, guardian_start, sui_start).await?;
190        let mut auditor = Self {
191            cfg: cfg.clone(),
192            pending_withdrawals: HashMap::new(),
193            pending_deposits: HashMap::new(),
194            utxo_pool: pool,
195            suspect_spends: HashMap::new(),
196            fresh_approvals: HashMap::new(),
197            rpc_pollers,
198            metrics,
199            store,
200            last_successful_pool_check: None,
201            started_at: now_timestamp_secs(),
202            last_snapshot_save: 0,
203            hbtc_supply_above_pool_since: None,
204            last_hbtc_supply_read: 0,
205        };
206        if let Some(state) = state {
207            auditor.restore(state);
208        }
209        // A pruned start leaves the pool incomplete.
210        auditor.rpc_pollers.ensure_sui_serves_start().await?;
211        // A setup that cannot read the supply fails here, not silently later.
212        let supply = auditor
213            .read_hbtc_supply()
214            .await
215            .context("failed to read the hBTC supply")?;
216        tracing::info!(hbtc_supply = supply, "read the hBTC supply");
217        auditor
218            .metrics
219            .set_checked_through(SOURCE_SUI, auditor.rpc_pollers.sui_read_until());
220        auditor
221            .metrics
222            .set_checked_through(SOURCE_GUARDIAN, auditor.rpc_pollers.guardian_read_until());
223        auditor
224            .metrics
225            .set_checked_through(SOURCE_BTC, auditor.rpc_pollers.btc_read_until());
226        // Check write access at startup. A missing bucket loads as empty.
227        auditor
228            .store
229            .save(&auditor.cfg, auditor.state(), auditor.utxo_pool())
230            .await?;
231        let now = now_timestamp_secs();
232        auditor.last_snapshot_save = now;
233        auditor.metrics.set_checked_through(SOURCE_SNAPSHOT, now);
234        Ok(auditor)
235    }

237    /// The pollers' positions, the open machines and the suspect spends, for
238    /// the snapshot.
239    fn state(&self) -> AuditState {
240        AuditState {
241            sui_next_checkpoint: self.rpc_pollers.sui_poller.next_item_to_poll(),
242            guardian_cursor: self.rpc_pollers.guardian_poller.next_item_to_poll(),
243            withdrawals: self.pending_withdrawals.values().cloned().collect(),
244            deposits: self.pending_deposits.values().cloned().collect(),
245            suspect_spends: self.suspect_spends.clone().into_iter().collect(),
246            hbtc_supply_above_pool_since: self.hbtc_supply_above_pool_since,
247        }
248    }
249
250    /// Hold the machines and the suspect spends a snapshot saved.
251    fn restore(&mut self, state: AuditState) {
252        for sm in state.withdrawals {
253            self.pending_withdrawals.insert(sm.wid(), sm);
254        }
255        for sm in state.deposits {
256            self.pending_deposits.insert(sm.request_id(), sm);
257        }
258        self.suspect_spends.extend(state.suspect_spends);
259        self.hbtc_supply_above_pool_since = state.hbtc_supply_above_pool_since;
260    }
261
262    fn ingest(&mut self, event: MonitorEvent) -> Vec<MonitorFinding> {
263        match event {
264            MonitorEvent::Withdrawal(event) => self.ingest_withdrawal(event),
265            MonitorEvent::DepositApproved(event) => self.ingest_deposit_approved(event),
266            MonitorEvent::DepositRequested(event) => self.ingest_deposit_request(event),
267        }
268    }
269
270    fn ingest_deposit_request(&mut self, event: SuiDepositRequestedEvent) -> Vec<MonitorFinding> {
271        match self.pending_deposits.entry(event.request_id) {
272            Entry::Occupied(entry) => entry.get().add_sui_request(&event).into_iter().collect(),
273            Entry::Vacant(entry) => {
274                entry.insert(DepositStateMachine::new(event));
275                Vec::new()
276            }
277        }
278    }
279
280    fn ingest_withdrawal(&mut self, event: MonitorWithdrawalEvent) -> Vec<MonitorFinding> {
281        let wid = event.wid;
282        match self.pending_withdrawals.entry(wid) {
283            Entry::Occupied(mut entry) => entry.get_mut().add_event(event, &self.cfg.timing),
284            Entry::Vacant(entry) => {
285                entry.insert(WithdrawalStateMachine::new(event, &self.cfg.timing));
286                Vec::new()
287            }
288        }
289    }
290
291    /// The Sui `DepositApproved`. Move approves only a request it holds, so
292    /// the request came first. While its machine is open, the machine keeps
293    /// the approval. Once it closed, the pool has its answer.
294    fn ingest_deposit_approved(&mut self, event: SuiDepositApprovedEvent) -> Vec<MonitorFinding> {
295        self.note_fresh_approval(&event, now_timestamp_secs());
296        match self.pending_deposits.get_mut(&event.request_id) {
297            Some(sm) => sm.add_sui_approval(event, &self.cfg.timing),
298            None => self
299                .utxo_pool
300                .check_sui_approval(&event, self.cfg.timing.deposit_approval_skew),
301        }
302    }
303
304    pub fn ingest_batch(&mut self, events: Vec<MonitorEvent>) -> Vec<MonitorFinding> {
305        let mut findings = Vec::new();
306        for event in events {
307            findings.extend(self.ingest(event));
308        }
309        self.report_findings("ingest", &findings);
310        findings
311    }

313    fn detect_timing_violations(&self) -> Vec<MonitorFinding> {
314        let mut findings = Vec::new();
315        for sm in self.pending_withdrawals.values() {
316            findings.extend(sm.timing_violations(&self.rpc_pollers.cursors()));
317        }
318        findings
319    }

321    /// The open withdrawals whose first event is the earliest, for the
322    /// GC log. An operator sees what the audit still waits for.
323    fn oldest_open_withdrawals(&self) -> String {
324        let earliest = self
325            .pending_withdrawals
326            .values()
327            .map(WithdrawalStateMachine::earliest_event_at)
328            .min();
329        let oldest = self
330            .pending_withdrawals
331            .values()
332            .filter(|sm| Some(sm.earliest_event_at()) == earliest)
333            .map(|sm| {
334                format!(
335                    "wid={}, btc_txid={}, earliest_event_at={}, waiting_for={:?}",
336                    sm.wid(),
337                    sm.btc_txid(),
338                    utc_timestamp(sm.earliest_event_at()),
339                    sm.expected_event_types(),
340                )
341            })
342            .collect::<Vec<_>>();
343        if oldest.is_empty() {
344            "none".to_string()
345        } else {
346            oldest.join("; ")
347        }
348    }

350    /// Return the deposits with the earliest request timestamp for the GC log.
351    fn oldest_open_deposits(&self) -> String {
352        let earliest = self
353            .pending_deposits
354            .values()
355            .map(DepositStateMachine::requested_at)
356            .min();
357        let oldest = self
358            .pending_deposits
359            .values()
360            .filter(|sm| Some(sm.requested_at()) == earliest)
361            .map(|sm| {
362                format!(
363                    "request_id={}, deposit_id={}, requested_at={}",
364                    sm.request_id(),
365                    sm.deposit_id(),
366                    utc_timestamp(sm.requested_at()),
367                )
368            })
369            .collect::<Vec<_>>();
370        if oldest.is_empty() {
371            "none".to_string()
372        } else {
373            oldest.join("; ")
374        }
375    }

382    pub fn utxo_pool(&self) -> &UtxoPool {
383        &self.utxo_pool
384    }
385
386    /// Save the snapshot. A failed save is logged and does not stop the
387    /// audit; the next maintenance tick saves again. Alerting watches the
388    /// time of the last good save.
389    async fn save_snapshot(&mut self) {
390        let saved = self
391            .store
392            .save(&self.cfg, self.state(), self.utxo_pool())
393            .await;
394        match saved {
395            Ok(()) => {
396                self.last_snapshot_save = now_timestamp_secs();
397                self.metrics
398                    .set_checked_through(SOURCE_SNAPSHOT, self.last_snapshot_save);
399            }
400            Err(error) => tracing::error!(?error, "failed to save the snapshot; continuing"),
401        }
402    }
403
404    /// A `SnapshotStale` finding if no save has succeeded within the limit.
405    fn stale_snapshot(&self, now: UnixSeconds) -> Option<MonitorFinding> {
406        (now.saturating_sub(self.last_snapshot_save) > SNAPSHOT_STALE_LIMIT).then_some(
407            MonitorFinding::SnapshotStale {
408                last_saved: self.last_snapshot_save,
409                limit_secs: SNAPSHOT_STALE_LIMIT,
410            },
411        )
412    }
413
414    fn report_findings(&self, phase: &'static str, findings: &[MonitorFinding]) {
415        log_findings("continuous", phase, findings);
416        self.metrics.record_findings(findings, now_timestamp_secs());
417    }
418
419    /// Report the open withdrawals that missed a timing bound.
420    fn check_timing(&self) {
421        let violations = self.detect_timing_violations();
422        self.report_findings("timing", &violations);
423    }
424
425    /// Report a source, the daily pool check, or the hBTC supply read that is behind its limit.
426    fn check_staleness(&self) {
427        let now = now_timestamp_secs();
428        let mut stale = self.rpc_pollers.stale_sources(now);
429        stale.extend(self.stale_pool_check(now));
430        stale.extend(self.stale_hbtc_supply(now));
431        self.report_findings("progress", &stale);
432    }
433
434    /// Save once `SNAPSHOT_SAVE_INTERVAL` has passed, then report a stale save.
435    async fn save_snapshot_if_due(&mut self) {
436        if now_timestamp_secs().saturating_sub(self.last_snapshot_save) >= SNAPSHOT_SAVE_INTERVAL {
437            self.save_snapshot().await;
438        }
439        // After the save, so a save that succeeds now is not reported.
440        let stale = self.stale_snapshot(now_timestamp_secs());
441        self.report_findings("progress", stale.as_slice());
442    }
443
444    /// Run the maintenance steps in order.
445    async fn tick_maintenance(&mut self) {
446        // First, so every later step sees only open machines and a current pool.
447        self.gc_and_update_pool();
448        self.check_utxo_pool_if_due();
449        self.check_hbtc_supply().await;
450        // These two have no order rule.
451        self.check_timing();
452        self.check_suspect_spends();
453        // After the checks, so a check that succeeds in this tick is not stale.
454        self.check_staleness();
455        // Last, so the snapshot holds the result of every step.
456        self.save_snapshot_if_due().await;
457    }
458
459    pub async fn run(&mut self) -> anyhow::Result<()> {
460        tracing::info!(
461            "starting continuous audit:\n  sui_start={}\n  guardian_start={}",
462            utc_timestamp(self.rpc_pollers.sui_read_until()),
463            utc_timestamp(self.rpc_pollers.guardian_read_until()),
464        );
465        self.catch_up_sources().await;
466        // After the catch-up, so the snapshot holds only the open machines.
467        self.gc_and_update_pool();
468        self.save_snapshot().await;
469        // A first run can catch up for hours. That time is not a stale check.
470        self.started_at = now_timestamp_secs();
471        tracing::info!(
472            sui_cursor = %utc_timestamp(self.rpc_pollers.sui_read_until()),
473            guardian_cursor = %utc_timestamp(self.rpc_pollers.guardian_read_until()),
474            "finished the initial catch-up; starting the audit loop"
475        );
476
477        let mut sui_ticker = tokio::time::interval(SUI_POLL_INTERVAL);
478        let mut guardian_ticker = tokio::time::interval(GUARDIAN_POLL_INTERVAL);
479        let mut btc_ticker = tokio::time::interval(BTC_POLL_INTERVAL);
480        let mut maintenance_ticker = tokio::time::interval(MAINTENANCE_INTERVAL);
481
482        sui_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
483        guardian_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
484        btc_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
485        maintenance_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
486
487        // Consume each immediate tick so scheduled work starts after its interval.
488        sui_ticker.tick().await;
489        guardian_ticker.tick().await;
490        btc_ticker.tick().await;
491        maintenance_ticker.tick().await;
492
493        loop {
494            tokio::select! {
495                _ = sui_ticker.tick() => {
496                    if let Err(error) = self.tick_sui().await {
497                        tracing::warn!(source = "sui", ?error, "poll failed; continuing");
498                    }
499                }
500                _ = guardian_ticker.tick() => {
501                    if let Err(error) = self.tick_guardian().await {
502                        tracing::warn!(source = "guardian", ?error, "poll failed; continuing");
503                    }
504                }
505                _ = btc_ticker.tick() => {
506                    if let Err(error) = self.tick_btc() {
507                        tracing::warn!(source = "btc", ?error, "btc tick failed; continuing");
508                    }
509                }
510                _ = maintenance_ticker.tick() => {
511                    self.tick_maintenance().await;
512                }
513            }
514        }
515    }
516}

518/// The snapshot, if this deployment saved it, or `None` for a first run.
519/// Fail on the snapshot of another deployment, so a config mistake does not
520/// overwrite it.
521fn this_deployments(cfg: &Config, snapshot: Option<Snapshot>) -> anyhow::Result<Option<Snapshot>> {
522    match snapshot {
523        Some(snapshot) if snapshot.is_for(cfg) => Ok(Some(snapshot)),
524        Some(snapshot) => anyhow::bail!(
525            "the snapshot is for another deployment (guardian bucket {}, package {}); \
526             check the config, or delete snapshot.json to start again from genesis",
527            snapshot.guardian_bucket,
528            snapshot.package_id,
529        ),
530        None => {
531            tracing::info!("no snapshot");
532            Ok(None)
533        }
534    }

582        };

586        ));
587        assert_eq!(auditor.pending_deposits.len(), 1);

590        let again = SuiDepositRequestedEvent {
591            request_id: Address::new([0x62; 32]),
592            ..request
593        };
594        assert!(

596                .ingest(MonitorEvent::DepositRequested(again))
